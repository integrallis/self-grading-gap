```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a dictionary to store the number of coins at each position
    coin_map = {}
    for l, r, c in coins:
        for position in range(l, r + 1):
            coin_map[position] = coin_map.get(position, 0) + c

    # Prepare to find the maximum coins in any k consecutive bags
    max_coins = 0
    current_coins = 0
    position_list = sorted(coin_map.keys())
    n = len(position_list)

    # Sliding window to count the coins in k consecutive bags
    left = 0
    for right in range(n):
        current_coins += coin_map[position_list[right]]
        # When we have more than k bags, slide the window
        while position_list[right] - position_list[left] + 1 > k:
            current_coins -= coin_map[position_list[left]]
            left += 1
        # Update max_coins if we have exactly k bags
        if position_list[right] - position_list[left] + 1 == k:
            max_coins = max(max_coins, current_coins)

    return max_coins

```
