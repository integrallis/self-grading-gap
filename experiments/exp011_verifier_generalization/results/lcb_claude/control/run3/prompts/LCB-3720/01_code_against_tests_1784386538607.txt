```python
from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    from collections import defaultdict, deque
    
    # Create a graph as an adjacency list with weights
    graph = defaultdict(list)
    for a, b, w in edges:
        graph[a].append((b, w))

    # Function to check if we can reach node 0 with a given maximum weight
    def canReachWithMaxWeight(max_weight):
        outgoing_count = [0] * n
        visited = set()
        queue = deque([0])  # Start BFS from node 0
        visited.add(0)

        while queue:
            current = queue.popleft()
            for neighbor, weight in graph[current]:
                if weight <= max_weight:
                    # allow outgoing connection  if not exceeding threshold
                    if outgoing_count[current] < threshold:
                        outgoing_count[current] += 1
                        if neighbor not in visited or outgoing_count[neighbor] < threshold:
                            visited.add(neighbor)
                            queue.append(neighbor)

        return len(visited) == n  # Check if all nodes are reachable from node 0

    # Binary search for the minimum maximum weight
    left, right = 1, 10**6  # weights are between 1 and 10^6
    answer = -1

    while left <= right:
        mid = (left + right) // 2
        if canReachWithMaxWeight(mid):
            answer = mid
            right = mid - 1
        else:
            left = mid + 1

    return answer

```
