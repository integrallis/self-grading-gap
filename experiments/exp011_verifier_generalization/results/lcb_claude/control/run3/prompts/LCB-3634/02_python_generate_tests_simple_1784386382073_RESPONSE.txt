from solution import calculateScore

def test_canonical_example_1():
    # spec: calculateScore("aczzx") -> 0
    # Explanation: 
    # i = 0: no j
    # i = 1: no j
    # i = 2: j = 0 (mirror of 'c' is 'x'); no valid j
    # i = 3: no j
    # i = 4: j = 1 (mirror of 'e' is 'v'); no valid j
    # Total score = 0
    assert calculateScore("aczzx") == 0

def test_canonical_example_2():
    # spec: calculateScore("abcdef") -> 0
    # Explanation:
    # For each index i, there is no index j that satisfies the conditions.
    # Thus, total score = 0
    assert calculateScore("abcdef") == 0

def test_edge_case_single_character():
    # spec: calculateScore("a") -> 0
    # Explanation: 
    # Single character, no j exists, total score = 0
    assert calculateScore("a") == 0

def test_edge_case_two_characters_no_mirror():
    # spec: calculateScore("ab") -> 0
    # Explanation: 
    # No valid j exists for either character, total score = 0
    assert calculateScore("ab") == 0

def test_edge_case_two_characters_mirror():
    # spec: calculateScore("az") -> 1
    # Explanation: 
    # i = 0: no j
    # i = 1: j = 0 (mirror of 'a' is 'z'); score += 1 - 0 = 1
    assert calculateScore("az") == 1

def test_complex_pattern():
    # spec: calculateScore("abczyx") -> 6
    # Explanation:
    # i = 0: no j
    # i = 1: no j
    # i = 2: no j
    # i = 3: j = 2 (mirror of 'z' is 'a'); score += 3 - 0 = 3
    # i = 4: j = 1 (mirror of 'y' is 'b'); score += 4 - 1 = 3
    # Total score = 3 + 3 = 6
    assert calculateScore("abczyx") == 6
