```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # Create a mirror mapping for a-z
    for i in range(len(s)):
        if marked[i]:  # Skip if already marked
            continue
        for j in range(i - 1, -1, -1):  # Look for a mirror character before i
            if marked[j]:  # Skip if j is already marked
                continue
            if mirror[s[i]] == s[j]:  # Check for mirror
                score += i - j  # Update score
                marked[i] = True  # Mark both i and j
                marked[j] = True
                break  # Move to the next i
    return score

```
