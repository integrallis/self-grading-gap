## Implementation

```python
from typing import List

def maxFrequency(nums: List[int], k: int) -> int:
    nums.sort()
    left = 0
    total = 0
    max_freq = 0

    for right in range(len(nums)):
        total += nums[right]
        while total + (right - left) * nums[right] > (right - left + 1) * k:
            total -= nums[left]
            left += 1
        max_freq = max(max_freq, right - left + 1)

    if k < nums[0]:
        return 0

    return max_freq
```
