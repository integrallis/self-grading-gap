from solution import subarraySum

def test_canonical_example_1():
    # spec: nums = [2, 3, 1]
    # subarrays:
    # index 0: nums[0] = [2] -> sum = 2
    # index 1: nums[0...1] = [2, 3] -> sum = 5
    # index 2: nums[1...2] = [3, 1] -> sum = 4
    # total sum = 2 + 5 + 4 = 11
    assert subarraySum([2, 3, 1]) == 11

def test_canonical_example_2():
    # spec: nums = [3, 1, 1, 2]
    # subarrays:
    # index 0: nums[0] = [3] -> sum = 3
    # index 1: nums[0...1] = [3, 1] -> sum = 4
    # index 2: nums[1...2] = [1, 1] -> sum = 2
    # index 3: nums[1...3] = [1, 1, 2] -> sum = 4
    # total sum = 3 + 4 + 2 + 4 = 13
    assert subarraySum([3, 1, 1, 2]) == 13
