from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n == 0:
        return 0
    # dp[i] will store the maximum length of subsequence ending at index i
    dp = [1] * n
    max_length = 1
    # iterate over every pair (j, i) with j < i
    for i in range(1, n):
        for j in range(i):
            # calculate the absolute difference
            diff_ij = abs(nums[i] - nums[j])
            # if the difference is non-increasing, update dp[i]
            if i == 1 or abs(nums[j] - nums[j-1]) >= diff_ij:
                dp[i] = max(dp[i], dp[j] + 1)
        max_length = max(max_length, dp[i])
    return max_length
```
