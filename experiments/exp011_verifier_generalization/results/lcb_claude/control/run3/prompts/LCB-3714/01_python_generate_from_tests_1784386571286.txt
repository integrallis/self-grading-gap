from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Total count of subsequences of length 1 to k
    for i in range(n):
        # Each element contributes as max and min
        # Maximum contributions
        max_contrib = (1 << (n - 1)) % MOD
        if n > k:
            max_contrib -= (1 << (n - k - 1)) % MOD
        # Minimum contributions
        min_contrib = (1 << (n - 1)) % MOD
        if n > k:
            min_contrib -= (1 << (n - k - 1)) % MOD
        total_sum += (nums[i] * (max_contrib + min_contrib)) % MOD

    return total_sum % MOD

```
