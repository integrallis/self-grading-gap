```python
from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Sort nums to easily determine min and max in subsequences
    nums.sort()
    
    # Calculate contributions of each number as min and max across valid subsequences
    for i in range(n):
        # Number of ways this number can be chosen as a max element in subsequences of size 1 to k
        max_ways = max(0, min(k, n - i)) * (1 << (i)) % MOD
        # Number of ways this number can be chosen as a min element in subsequences of size 1 to k
        min_ways = max(0, min(k, i + 1)) * (1 << (n - i - 1)) % MOD

        #Contribution to total sum
        contribution = (max_ways + min_ways) * nums[i] % MOD
        total_sum += contribution
        total_sum %= MOD

    return total_sum % MOD

```
