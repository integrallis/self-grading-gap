from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    dp = [[0] * 3 for _ in range(n)]

    # Initialize the first house costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the dp array for all houses
    for i in range(1, n):
        for j in range(3):
            # Cost of painting house i with color j
            dp[i][j] = cost[i][j] + min(
                dp[i - 1][(j + 1) % 3],  # Different color from previous house
                dp[i - 1][(j + 2) % 3]   # Different color from previous house
            )
            # Check equidistant constraint
            if i < n // 2:
                dp[i][j] += min(dp[n - 1 - i][k] for k in range(3) if k != j)

    # Return the minimum cost of painting all houses
    return min(dp[n - 1][j] for j in range(3))
```
