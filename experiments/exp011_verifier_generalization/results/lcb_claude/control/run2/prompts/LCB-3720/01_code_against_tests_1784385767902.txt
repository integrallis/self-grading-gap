```python
from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    from collections import defaultdict
    import heapq

    # Build the graph as an adjacency list with weights
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))

    # Function to check if a maximum weight can allow all nodes to reach node 0
    def canReachNode0(max_weight):
        # Create a new graph with filtered edges
        filtered_graph = defaultdict(list)
        for u in range(n):
            for v, w in graph[u]:
                if w <= max_weight:
                    filtered_graph[v].append(u)

        # BFS or DFS to see if we can reach node 0 from all nodes
        reachable = [False] * n
        stack = [0]
        reachable[0] = True
        while stack:
            node = stack.pop()
            for neighbor in filtered_graph[node]:
                if not reachable[neighbor]:
                    reachable[neighbor] = True
                    stack.append(neighbor)

        return all(reachable)

    # Binary search to find the minimum possible maximum edge weight
    left, right = 1, 10**6
    answer = -1
    while left <= right:
        mid = (left + right) // 2
        # Check if it's possible with the current mid as the maximum weight
        if canReachNode0(mid):
            answer = mid
            right = mid - 1
        else:
            left = mid + 1

    # Check if we can meet the threshold condition
    if answer != -1:
        # Check if we can limit outgoing edges to the threshold
        for w in range(1, answer + 1):
            count = defaultdict(int)
            for u, v, weight in edges:
                if weight <= answer:
                    count[u] += 1
            if any(c > threshold for c in count.values()):
                return -1
    return answer
```
