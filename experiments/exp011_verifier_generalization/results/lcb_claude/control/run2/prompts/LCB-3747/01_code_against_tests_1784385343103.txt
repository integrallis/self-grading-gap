## Implementation

```python
from typing import List

def maxAdjacentDistance(nums: List[int]) -> int:
    max_diff = 0
    n = len(nums)
    for i in range(n):
        for j in range(i + 1, n):
            diff = abs(nums[i] - nums[j])
            max_diff = max(max_diff, diff)
    return max_diff

```
