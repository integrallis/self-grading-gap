## Implementation

```python
from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Calculate contribution of each number in nums
    for num in nums:
        # Each number contributes to the total sum as maximum and minimum
        # in all combinations of size up to k
        # max contribution
        total_sum += num * (pow(2, min(k, n-1), MOD) - 1)
        # min contribution
        total_sum += num * (pow(2, min(k-1, n-1), MOD) - 1)
        total_sum %= MOD

    return total_sum
```
