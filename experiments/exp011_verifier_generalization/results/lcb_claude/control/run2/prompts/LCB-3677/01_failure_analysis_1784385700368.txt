from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[[float('-inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0]  # Starting point

    for i in range(m):
        for j in range(n):
            for k in range(3):  # Number of neutralizations used
                if i == 0 and j == 0:
                    continue
                if i > 0:
                    current = dp[i-1][j][k]
                    if coins[i][j] < 0 and k > 0:
                        dp[i][j][k] = max(dp[i][j][k], current + coins[i][j])  # Neutralize
                    elif coins[i][j] >= 0:
                        dp[i][j][k] = max(dp[i][j][k], current + coins[i][j])  # Gain coins
                if j > 0:
                    current = dp[i][j-1][k]
                    if coins[i][j] < 0 and k > 0:
                        dp[i][j][k] = max(dp[i][j][k], current + coins[i][j])  # Neutralize
                    elif coins[i][j] >= 0:
                        dp[i][j][k] = max(dp[i][j][k], current + coins[i][j])  # Gain coins
                if coins[i][j] < 0 and k > 0:
                    dp[i][j][k] = max(dp[i][j][k], dp[i][j][k-1])  # Neutralize the robber in this cell

    return max(dp[m-1][n-1])
