from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    # Initialize dp array: dp[i][j][k] = maximum coins at (i,j) using k neutralizations
    dp = [[[-float('inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0]  # Starting point

    for i in range(m):
        for j in range(n):
            for k in range(3):  # Number of neutralizations used
                if i == 0 and j == 0:
                    continue
                if i > 0:
                    if coins[i][j] < 0 and k > 0:
                        dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k-1] + coins[i][j])  # Neutralize
                    else:
                        dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k] + coins[i][j])  # Gain coins
                if j > 0:
                    if coins[i][j] < 0 and k > 0:
                        dp[i][j][k] = max(dp[i][j][k], dp[i][j-1][k-1] + coins[i][j])  # Neutralize
                    else:
                        dp[i][j][k] = max(dp[i][j][k], dp[i][j-1][k] + coins[i][j])  # Gain coins

    # Calculate the maximum coins possible at the bottom-right corner with any neutralization usage
    return max(dp[m-1][n-1])
```
