```python
from typing import List

def minCosts(cost: List[int]) -> List[int]:
    n = len(cost)
    answer = [0] * (n + 1)
    
    for i in range(n):
        if i == 0:
            answer[i] = cost[i]
        else:
            answer[i] = min(answer[i - 1], cost[i])
    answer[n] = 0  # The cost to reach the end is 0 since we are already there
    return answer[:n]  # Return only the first n elements

```
