from typing import List
from solution import minCosts

def test_canonical_example_1():
    # spec: cost = [5, 3, 4, 1, 3, 2] 
    # answer[0] = 5
    # answer[1] = 3
    # answer[2] = min(3 (swap with person 1) , 4 (swap with person 2) ) = 3
    # answer[3] = 1
    # answer[4] = min(1 (swap with person 3), 3 (swap with person 4)) = 1
    # answer[5] = min(1 (swap with person 3), 2 (swap with person 5)) = 1
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]

def test_canonical_example_2():
    # spec: cost = [1, 2, 4, 6, 7] 
    # answer[0] = 1
    # answer[1] = min(1 (swap with person 0), 2) = 1
    # answer[2] = min(1 (from person 0), 4) = 1
    # answer[3] = min(1 (from person 0), 6) = 1
