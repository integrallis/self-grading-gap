from typing import List
from solution import maxFrequency

def test_canonical_example_1():
    # spec: input = [1, 2, 3, 4, 5, 6], k = 1
    # By adding -5 to nums[2..5], we get [1, 2, -2, -1, 0, 1], where 1 appears 2 times.
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2

def test_canonical_example_2():
    # spec: input = [10, 2, 3, 4, 5, 5, 4, 3, 2, 2], k = 10
    # By adding 8 to nums[1..9], we get [10, 10, 11, 12, 13, 13, 12, 11, 10, 10], where 10 appears 4 times.
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4
