def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    for move in s:
        if move == 'N':
            y += 1
        elif move == 'S':
            y -= 1
        elif move == 'E':
            x += 1
        elif move == 'W':
            x -= 1
        current_distance = abs(x) + abs(y)
        max_distance = max(max_distance, current_distance)
    # Calculate the maximum possible distance by considering k changes
    max_possible_distance = max_distance + k
    return max_possible_distance

```
