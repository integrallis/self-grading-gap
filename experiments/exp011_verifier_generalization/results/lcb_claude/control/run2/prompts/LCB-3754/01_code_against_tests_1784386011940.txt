## Implementation

```python
def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    moves = []
    for move in s:
        if move == 'N':
            y += 1
        elif move == 'S':
            y -= 1
        elif move == 'E':
            x += 1
        elif move == 'W':
            x -= 1
        current_distance = abs(x) + abs(y)
        max_distance = max(max_distance, current_distance)
        moves.append((x, y))
        k -= min(k, abs(x) + abs(y) - current_distance)
        if k < 0:
            break
    # The best we can do with k additional changes is to extend the maximum distance calculated.
    return max_distance + k

```
