from typing import List
from solution import countPartitions

def test_canonical_example_1():
    # spec: Input: [10, 10, 3, 7, 6]
    # Partitions: 
    # [10], [10, 3, 7, 6] -> diff = 10 - 26 = -16 (even)
    # [10, 10], [3, 7, 6] -> diff = 20 - 16 = 4 (even)
    # [10, 10, 3], [7, 6] -> diff = 23 - 13 = 10 (even)
    # [10, 10, 3, 7], [6] -> diff = 30 - 6 = 24 (even)
    # Total = 4
    assert countPartitions([10, 10, 3, 7, 6]) == 4

def test_canonical_example_2():
    # spec: Input: [1, 2, 2]
    # Partitions: 
    # [1], [2, 2] -> diff = 1 - 4 = -3 (odd)
    # [1, 2], [2] -> diff = 3 - 2 = 1 (odd)
    # Total = 0
    assert countPartitions([1, 2, 2]) == 0

def test_canonical_example_3():
    # spec: Input: [2, 4, 6, 8]
    # Partitions: 
    # [2], [4, 6, 8] -> diff = 2 - 18 = -16 (even)
    # [2, 4], [6, 8] -> diff = 6 - 14 = -8 (even)
    # [2, 4, 6], [8] -> diff = 12 - 8 = 4 (even)
    # Total = 3
    assert countPartitions([2, 4, 6, 8]) == 3
