from typing import List

def countPartitions(nums: List[int]) -> int:
    total_count = 0
    left_sum = 0
    total_sum = sum(nums)
    for i in range(len(nums) - 1):
        left_sum += nums[i]
        right_sum = total_sum - left_sum
        if (left_sum - right_sum) % 2 == 0:
            total_count += 1
    return total_count

```
