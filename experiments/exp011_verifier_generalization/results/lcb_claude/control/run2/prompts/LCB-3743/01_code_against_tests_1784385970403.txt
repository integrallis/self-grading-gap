```python
from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_intervals = []
    last_end = 0

    # Collect free intervals
    for i in range(n):
        start = startTime[i]
        end = endTime[i]
        if start > last_end:
            free_intervals.append((last_end, start))
        last_end = end

    # Add the last free interval until event time
    if last_end < eventTime:
        free_intervals.append((last_end, eventTime))

    # Calculate gaps between meetings
    gaps = []
    for start, end in free_intervals:
        gaps.append(end - start)

    # We can maximize free time by moving meetings into gaps
    max_free_time = 0
    for i in range(len(gaps)):  # Iterate over gap sizes
        if i < k:
            max_free_time += gaps[i]
        else:
            break
    return max_free_time

```
