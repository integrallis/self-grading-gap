from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_intervals = []
    last_end = 0

    # Collect free intervals
    for i in range(n):
        start = startTime[i]
        end = endTime[i]
        if start > last_end:
            free_intervals.append((last_end, start))
        last_end = end

    # Add the last free interval until event time
    if last_end < eventTime:
        free_intervals.append((last_end, eventTime))

    # Calculate gaps
    gaps = []
    for start, end in free_intervals:
        gaps.append(end - start)

    # We can maximize free time by potentially moving meetings into free slots
    max_free_time = 0
    moveable_meetings = k

    for i in range(n):
        current_free_time = 0
        if i > 0:
            current_free_time += free_intervals[i-1][1] - free_intervals[i-1][0]
        if i < len(gaps):
            current_free_time += gaps[i]
        if moveable_meetings > 0:
            moveable_meetings -= 1
            if i + 1 < len(gaps) and free_intervals[i][1] < free_intervals[i+1][0]:
                moveable_meetings += 1  # If we can move the next meeting away
        max_free_time = max(max_free_time, current_free_time)

    return max_free_time
