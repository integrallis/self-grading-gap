from solution import calculateScore

def test_canonical_example_1():
    # Example: input "aczzx" -> output 5
    # Calculation:
    # i = 0: no j found
    # i = 1: no j found
    # i = 2: j = 0 (mirror of 'a' is 'z'), score += 2 - 0 = 2
    # i = 3: no j found
    # i = 4: j = 1 (mirror of 'c' is 'x'), score += 4 - 1 = 3
    # Total score = 2 + 3 = 5
    assert calculateScore("aczzx") == 5

def test_canonical_example_2():
    # Example: input "abcdef" -> output 0
    # Calculation:
    # For each index i, there is no j that satisfies the conditions
    # Thus, total score remains 0
    assert calculateScore("abcdef") == 0
