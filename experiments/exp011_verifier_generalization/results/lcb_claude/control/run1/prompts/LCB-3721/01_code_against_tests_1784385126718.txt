```python
from typing import List

def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers
    online_status = [True] * numberOfUsers
    offline_timers = [0] * numberOfUsers

    for event in events:
        event_type = event[0]
        timestamp = int(event[1])

        # Update online status before processing the event
        for i in range(numberOfUsers):
            if not online_status[i] and timestamp >= offline_timers[i]:
                online_status[i] = True

        if event_type == 'OFFLINE':
            user_id = int(event[2])
            online_status[user_id] = False
            offline_timers[user_id] = timestamp + 60

        elif event_type == 'MESSAGE':
            mentions_string = event[2]

            if mentions_string == 'ALL':
                for i in range(numberOfUsers):
                    mentions[i] += 1
            elif mentions_string == 'HERE':
                for i in range(numberOfUsers):
                    if online_status[i]:
                        mentions[i] += 1
            else:
                mentioned_ids = mentions_string.split()
                for mention in mentioned_ids:
                    user_id = int(mention[2:])  # Extract id<number>
                    mentions[user_id] += 1

    return mentions

```
