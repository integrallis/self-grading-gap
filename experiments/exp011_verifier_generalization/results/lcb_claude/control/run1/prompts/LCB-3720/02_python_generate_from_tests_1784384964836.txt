from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    from collections import defaultdict, deque
    import heapq

    # Create a graph and in-degree count
    graph = defaultdict(list)
    in_degree = [0] * n

    for u, v, w in edges:
        graph[u].append((v, w))
        in_degree[v] += 1

    # Check if we can reach node 0 from all nodes
    def can_reach_node_0(max_weight):
        visited = [False] * n
        queue = deque([0])
        visited[0] = True

        while queue:
            node = queue.popleft()
            for neighbor, weight in graph[node]:
                if not visited[neighbor] and weight <= max_weight:
                    visited[neighbor] = True
                    queue.append(neighbor)

        return all(visited)

    # Binary search for the minimum possible max weight
    left, right = 1, 10**6
    result = -1

    while left <= right:
        mid = (left + right) // 2

        # Create a filtered graph with the maximum weight `mid`
        filtered_graph = defaultdict(list)
        out_degree = [0] * n

        for u in range(n):
            for v, w in graph[u]:
                if w <= mid and out_degree[u] < threshold:
                    filtered_graph[u].append((v, w))
                    out_degree[u] += 1

        # Check if we can reach node 0 with this filtered graph
        visited = [False] * n
        queue = deque([0])
        visited[0] = True

        while queue:
            node = queue.popleft()
            for neighbor, weight in filtered_graph[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    queue.append(neighbor)

        if all(visited):
            result = mid
            right = mid - 1
        else:
            left = mid + 1

    return result

```
