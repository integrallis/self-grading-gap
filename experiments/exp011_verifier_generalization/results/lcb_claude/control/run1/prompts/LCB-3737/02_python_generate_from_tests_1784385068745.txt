from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize a 2D list to store the minimum cost to paint each house
    dp = [[0] * 3 for _ in range(n)]

    # Fill in the first house costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the dp table for each house
    for i in range(1, n):
        for j in range(3):
            # Calculate the minimum cost for painting house i with color j
            dp[i][j] = cost[i][j] + min(dp[i-1][(j+1)%3], dp[i-1][(j+2)%3])

    # To satisfy the equidistant condition, we need to consider two cases:
    # 1. Paint the first house with color 0
    # 2. Paint the first house with color 1
    first_case = dp[n-1][0] + min(dp[n//2-1][1], dp[n//2-1][2])
    second_case = dp[n-1][1] + min(dp[n//2-1][0], dp[n//2-1][2])
    third_case = dp[n-1][2] + min(dp[n//2-1][0], dp[n//2-1][1])

    # The result is the minimum cost of the three cases
    return min(first_case, second_case, third_case)
```
