from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize a 2D list to store the minimum cost to paint each house
    dp = [[0] * 3 for _ in range(n)]

    # Fill in the first house costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the dp table for each house
    for i in range(1, n):
        for j in range(3):
            # Calculate the minimum cost for painting house i with color j
            dp[i][j] = cost[i][j] + min(dp[i-1][(j+1)%3], dp[i-1][(j+2)%3])

    # To satisfy the equidistant condition, we must check the last half of houses
    result = float('inf')
    for j in range(3):
        # Paint the last house with color j and check the costs with the first half
        total_cost = dp[n-1][j] + min(dp[n//2-1][(j+1)%3], dp[n//2-1][(j+2)%3])
        result = min(result, total_cost)

    return result
```
