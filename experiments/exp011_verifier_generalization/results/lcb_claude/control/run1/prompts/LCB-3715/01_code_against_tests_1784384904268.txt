```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a list to hold the number of coins at each position
    total_coins = {}

    # Fill the total_coins dictionary with the number of coins in each bag
    for l, r, c in coins:
        for position in range(l, r + 1):
            total_coins[position] = total_coins.get(position, 0) + c

    # Sort the positions to allow for sliding window calculation
    sorted_positions = sorted(total_coins.keys())

    # Sliding window to find the maximum coins in k consecutive bags
    max_coins = 0
    current_coins = 0
    left = 0

    for right in range(len(sorted_positions)):
        current_coins += total_coins[sorted_positions[right]]

        # If we have more than k bags, move the left pointer
        while sorted_positions[right] - sorted_positions[left] + 1 > k:
            current_coins -= total_coins[sorted_positions[left]]
            left += 1

        # Update maximum coins if the current window is valid
        if sorted_positions[right] - sorted_positions[left] + 1 <= k:
            max_coins = max(max_coins, current_coins)

    return max_coins

```
