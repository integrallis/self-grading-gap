from solution import calculateScore

def test_canonical_example_1():
    # spec: calculateScore("aczzx") -> 5
    # Explanation:
    # i = 0. No j found, skip.
    # i = 1. No j found, skip.
    # i = 2. j = 0 (mirror 'a' is 'z'), score += 2 - 0 = 2.
    # i = 3. No j found, skip.
    # i = 4. j = 1 (mirror 'b' is 'y'), score += 4 - 1 = 3.
    # Total score = 2 + 3 = 5
    assert calculateScore("aczzx") == 5

def test_canonical_example_2():
    # spec: calculateScore("abcdef") -> 0
    # Explanation:
    # For each index i, there is no j < i such that s[j] is the mirror of s[i].
    # Thus, score remains 0.
    assert calculateScore("abcdef") == 0
