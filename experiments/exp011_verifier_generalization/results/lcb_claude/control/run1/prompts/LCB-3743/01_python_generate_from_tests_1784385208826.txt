from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_intervals = []

    # Calculate initial free time slots before and after meetings
    if startTime[0] > 0:
        free_intervals.append((0, startTime[0]))  # Free time before the first meeting
    for i in range(1, n):
        free_intervals.append((endTime[i - 1], startTime[i]))  # Free time between meetings
    if endTime[-1] < eventTime:
        free_intervals.append((endTime[-1], eventTime))  # Free time after the last meeting

    # Calculate the maximum free time possible with the ability to move k meetings
    free_times = []
    for start, end in free_intervals:
        free_times.append(end - start)  # Add free time lengths

    # Sort free_times in descending order to find the largest free time slots
    free_times.sort(reverse=True)

    # If we can move k meetings, we can create k free time slots from the largest ones
    return sum(free_times[:k]) if k <= len(free_times) else sum(free_times)
```
