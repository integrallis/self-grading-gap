from typing import List
from solution import minMaxSums

def test_canonical_example_1():
    # Input: nums = [1, 2, 3], k = 2
    # All subsequences with at most 2 elements:
    # [1], [2], [3], [1, 2], [1, 3], [2, 3]
    # Minimum/Maximum sums:
    # [1]: min=1, max=1 -> 2
    # [2]: min=2, max=2 -> 4
    # [3]: min=3, max=3 -> 6
    # [1, 2]: min=1, max=2 -> 3
    # [1, 3]: min=1, max=3 -> 4
    # [2, 3]: min=2, max=3 -> 5
    # Total = 2 + 4 + 6 + 3 + 4 + 5 = 24
    assert minMaxSums([1, 2, 3], 2) == 24

def test_canonical_example_2():
    # Input: nums = [5, 0, 6], k = 1
    # All subsequences with exactly 1 element:
    # [5], [0], [6]
    # Minimum/Maximum sums:
    # [5]: min=5, max=5 -> 10
    # [0]: min=0, max=0 -> 0
    # [6]: min=6, max=6 -> 12
    # Total = 10 + 0 + 12 = 22
    assert minMaxSums([5, 0, 6], 1) == 22

def test_canonical_example_3():
    # Input: nums = [1, 1, 1], k = 2
    # All subsequences with at most 2 elements:
    # [1], [1], [1], [1, 1], [1, 1], [1, 1]
    # Minimum/Maximum sums:
    # [1]: min=1, max=1 -> 2 (3 times, total 2*3=6)
    # [1, 1]: min=1, max=1 -> 2 (3 times, total 2*3=6)
    # Total = 6 + 6 = 12
    assert minMaxSums([1, 1, 1], 2) == 12
