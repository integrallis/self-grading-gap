from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Calculate contributions from elements as min and max
    for i in range(n):
        # Each element nums[i] can act as min in C(n-1, j) ways for j = 0 to k-1
        # It can act as max in C(n-1, j) ways for j = 0 to k-1
        min_contrib = nums[i] * (pow(2, min(k, n-1), MOD)) % MOD
        max_contrib = nums[i] * (pow(2, min(k, n-1), MOD)) % MOD
        total_sum = (total_sum + min_contrib + max_contrib) % MOD

    return total_sum

```
