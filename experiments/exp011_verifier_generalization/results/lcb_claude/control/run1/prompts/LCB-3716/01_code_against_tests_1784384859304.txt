```python
from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n == 0:
        return 0
    # dp[i] will store the length of the longest subsequence ending with nums[i]
    dp = [1] * n
    # Store the maximum length found
    max_length = 1
    # Check pairs to build the dp array
    for i in range(1, n):
        for j in range(i):
            diff1 = abs(nums[i] - nums[j])
            # We need to find the longest subsequence with non-increasing absolute differences
            for k in range(j):
                diff2 = abs(nums[j] - nums[k])
                if diff1 >= diff2:
                    dp[i] = max(dp[i], dp[j] + 1)
        max_length = max(max_length, dp[i])
    return max_length
```
