def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a dictionary to hold the number of coins at each bag position
    coin_map = {}
    
    # Fill the coin_map with the number of coins for each relevant position
    for l, r, c in coins:
        for i in range(l, r + 1):
            coin_map[i] = coin_map.get(i, 0) + c
    
    # Create a list of bags filled with coins based on the positions
    positions = sorted(coin_map.keys())
    total_bags = len(positions)

    # Calculate the maximum coins obtainable from any k consecutive bags
    max_coins = 0
    current_coins = 0
    start = 0

    # Iterate through the positions and calculate the sum of coins for k bags
    for end in range(total_bags):
        current_coins += coin_map[positions[end]]
        # If we have more than k bags, slide the window
        if end - start + 1 > k:
            current_coins -= coin_map[positions[start]]
            start += 1
        # Update max coins if we have exactly k bags
        if end - start + 1 == k:
            max_coins = max(max_coins, current_coins)

    return max_coins
