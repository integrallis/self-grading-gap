def test_single_element():
    # edge case: single element [42], k = 1
    # The only subsequence is [42], min = max = 42
    # Sum = 42 + 42 = 84
    assert minMaxSums([42], 1) == 84

def test_large_array():
    # edge case: maximum length array with increasing values

    # edge case: nums = [0, 0, 0], k = 2
    # All subsequences have min = max = 0
    # Total = 0
    assert minMaxSums([0, 0, 0], 2) == 0

def test_max_values():
    # edge case: nums = [10**9, 10**9, 10**9], k = 2
    # All subsequences have min = max = 10**9
    # Total = 10**9 * 3 = 3 * 10**9 % (10**9 + 7) = 299999992
    assert minMaxSums([10**9, 10**9, 10**9], 2) == 299999992
