from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Sort nums to easily find min and max
    nums.sort()

    # Calculate contributions for min and max elements
    for i in range(n):
        # Each element nums[i] can be min or max in subsequences of size 1 to k
        # Number of times nums[i] is the maximum element in k-sized subsequences:
        max_contribution = nums[i] * (pow(2, i, MOD) * pow(2, n - i - 1, MOD)) % MOD

        # Number of times nums[i] is the minimum element in k-sized subsequences:
        min_contribution = nums[i] * (pow(2, i, MOD) * pow(2, n - i - 1, MOD)) % MOD

        # Each element contributes to total_sum as both min and max
        total_sum = (total_sum + max_contribution + min_contribution) % MOD

    # Since we consider subsequences up to size k, we take k + 1 for 0 to k
    total_sum = total_sum * (k + 1) % MOD

    return total_sum
```
