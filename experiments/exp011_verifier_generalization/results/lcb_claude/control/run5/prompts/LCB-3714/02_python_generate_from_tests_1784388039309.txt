from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Calculate contributions of each element when it is the max element
    for i in range(n):
        # Number of ways to choose elements before nums[i] as min elements
        ways_as_max = pow(2, i, MOD)  # Choose any subset of elements before i
        # Number of ways to choose elements after nums[i] as max elements
        ways_as_min = pow(2, n - i - 1, MOD)  # Choose any subset of elements after i

        # Each element is counted in the contribution of being max and min
        total_sum += (nums[i] * ways_as_max * ways_as_min) % MOD
        total_sum %= MOD

    # Each element can be min or max in k-sized subsequences
    total_sum = (total_sum * (k + 1)) % MOD

    return total_sum
```
