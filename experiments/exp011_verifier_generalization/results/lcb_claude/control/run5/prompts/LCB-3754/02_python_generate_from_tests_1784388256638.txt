def maxDistance(s: str, k: int) -> int:
    # Initial position
    x, y = 0, 0
    # Count movements
    for char in s:
        if char == 'N':
            y += 1
        elif char == 'S':
            y -= 1
        elif char == 'E':
            x += 1
        elif char == 'W':
            x -= 1
    # Calculate current Manhattan distance
    current_distance = abs(x) + abs(y)
    # Calculate the maximum possible distance by changing k moves
    max_distance = current_distance + k
    return max_distance
```
