```python
from typing import List

def largestInteger(nums: List[int], k: int) -> int:
    from collections import defaultdict
    count = defaultdict(int)
    n = len(nums)

    # Count occurrences of each number in each subarray of size k
    for i in range(n - k + 1):
        unique = set()
        for j in range(i, i + k):
            if nums[j] not in unique:
                count[nums[j]] += 1
                unique.add(nums[j])

    # Find the largest integer that appears exactly once
    almost_missing = [num for num, c in count.items() if c == 1]
    return max(almost_missing) if almost_missing else -1

```
