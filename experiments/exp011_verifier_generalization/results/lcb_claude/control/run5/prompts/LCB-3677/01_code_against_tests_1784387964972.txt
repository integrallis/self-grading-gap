```python
from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[[-float('inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0]  # Starting point with no neutralization

    for i in range(m):
        for j in range(n):
            for k in range(3):  # k is the count of neutralizations used
                if i == 0 and j == 0:
                    continue
                # Calculate current cell value
                current_value = coins[i][j]
                # Update dp from top and left
                if i > 0:
                    dp[i][j][k] = max(dp[i][j][k], dp[i - 1][j][k] + current_value)
                if j > 0:
                    dp[i][j][k] = max(dp[i][j][k], dp[i][j - 1][k] + current_value)

                # If we encounter a robber
                if current_value < 0:
                    # Count neutralized value if we can neutralize
                    if k > 0:
                        dp[i][j][k] = max(dp[i][j][k], dp[i - 1][j][k - 1])
                        dp[i][j][k] = max(dp[i][j][k], dp[i][j - 1][k - 1])
                    # If we don't neutralize, subtract coins
                    dp[i][j][k] = max(dp[i][j][k], dp[i - 1][j][k] + current_value)
                    dp[i][j][k] = max(dp[i][j][k], dp[i][j - 1][k] + current_value)
                else:
                    # Updating without neutralization
                    dp[i][j][k] += current_value

    # Get the maximum value from the bottom-right cell with up to 2 neutralizations
    return max(dp[m - 1][n - 1][0], dp[m - 1][n - 1][1], dp[m - 1][n - 1][2])

```
