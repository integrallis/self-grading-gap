from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # Dynamic programming table to store minimum costs
    dp = [[0] * 3 for _ in range(n)]
    # Initialize the first house with the costs of painting it with each color
    dp[0] = cost[0]

    # Fill the dp table for all houses
    for i in range(1, n):
        for j in range(3):
            # Cost to paint house i with color j
            # Must not be the same color as the previous house
            # and also consider equidistant constraint
            dp[i][j] = cost[i][j] + min(
                dp[i - 1][(j + 1) % 3],
                dp[i - 1][(j + 2) % 3],
                (dp[i - 2][j] if i > 1 else float('inf'))
            )

    # The minimum cost for the last house considering the equidistant constraint
    return min(dp[n - 1][0], dp[n - 1][1], dp[n - 1][2])
```
