from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Function to check if we can reach node 0 with a given max weight
    def canReachWithMaxWeight(max_weight: int) -> bool:
        from collections import defaultdict, deque

        graph = defaultdict(list)
        for u, v, w in edges:
            if w <= max_weight:
                graph[u].append(v)  # Keep original direction

        # BFS or DFS to check reachability from all nodes to node 0
        queue = deque([0])
        visited = set([0])
        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if neighbor not in visited:
                    visited.add(neighbor)
                    queue.append(neighbor)

        return len(visited) == n  # Check if all nodes can reach node 0

    # Binary search over the possible maximum edge weights
    low, high = 1, 10**6
    result = -1
    while low <= high:
        mid = (low + high) // 2
        # Check if we can reach node 0 with the current mid weight
        if canReachWithMaxWeight(mid):
            result = mid
            high = mid - 1  # Try to find a smaller max weight
        else:
            low = mid + 1  # Increase the weight

    # After finding the minimum max weight, check if it satisfies the threshold condition
    if result != -1:
        # Build the graph again with the minimum max weight
        graph = defaultdict(list)
        out_degree = [0] * n
        for u, v, w in edges:
            if w <= result:
                graph[u].append(v)
                out_degree[u] += 1

        # Check if we exceed the threshold
        for out in out_degree:
            if out > threshold:
                return -1

    return result

```
