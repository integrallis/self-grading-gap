from typing import List
from solution import minCosts

def test_canonical_example_1():
    # spec: input [5, 3, 4, 1, 3, 2] 
    # costs: to reach 0 -> 5, to reach 1 -> 3, 
    # to reach 2 -> min(3 + free to 2) = 3, 
    # to reach 3 -> 1, 
    # to reach 4 -> min(1 + free to 4) = 1, 
    # to reach 5 -> min(1 + free to 5) = 1
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]

def test_canonical_example_2():
    # spec: input [1, 2, 4, 6, 7] 
    # costs: to reach 0 -> 1, 
    # to reach 1 -> min(1) = 1, 
    # to reach 2 -> min(1) = 1, 
    # to reach 3 -> min(1) = 1, 
    # to reach 4 -> min(1) = 1
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]
