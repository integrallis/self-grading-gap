from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_times = []

    # Calculate initial free times between meetings
    if startTime[0] > 0:
        free_times.append(startTime[0])  # Free time before the first meeting
    for i in range(1, n):
        free_times.append(startTime[i] - endTime[i - 1])  # Free time between meetings
    if endTime[-1] < eventTime:
        free_times.append(eventTime - endTime[-1])  # Free time after the last meeting

    # Sort free times in descending order
    free_times.sort(reverse=True)

    # If k is greater than or equal to the number of free periods, return the largest free period
    if k >= len(free_times):
        return free_times[0] if free_times else 0

    # We can reschedule k meetings, so we take the k largest free times
    return sum(free_times[:k]) - k
