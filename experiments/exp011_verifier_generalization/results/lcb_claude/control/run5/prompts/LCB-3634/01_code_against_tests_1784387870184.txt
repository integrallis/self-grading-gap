## Implementation

```python
def calculateScore(s: str) -> int:
    score = 0
    n = len(s)
    marked = [False] * n
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # 'a' to 'z' mirrors

    for i in range(n):
        if marked[i]:
            continue
        mirror_char = mirror[s[i]]
        for j in range(i - 1, -1, -1):
            if not marked[j] and s[j] == mirror_char:
                marked[i] = True
                marked[j] = True
                score += i - j
                break

    return score
```
