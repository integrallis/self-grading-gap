```python
from typing import List
import heapq


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Create a graph structure from the edges
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[u].append((v, w))

    # Function to check if we can reach node 0 with max weight allowed
    def canReachWithMaxWeight(max_weight: int) -> bool:
        # Using a priority queue for the modified Dijkstra's algorithm to limit edges
        reachable = [0] * n
        reachable[0] = 1  # Start from node 0
        count = 0

        pq = [(0, 0)]  # (current node, current weight)
        while pq:
            current, weight = heapq.heappop(pq)
            if weight > max_weight:
                continue
            count += 1
            for neighbor, edge_weight in graph[current]:
                if edge_weight <= max_weight and reachable[neighbor] < threshold:
                    reachable[neighbor] += 1
                    heapq.heappush(pq, (neighbor, edge_weight))
        return count == n  # All nodes must be reachable

    # Binary search over the possible max weights
    left, right = 1, 10**6
    answer = -1
    while left <= right:
        mid = (left + right) // 2
        if canReachWithMaxWeight(mid):
            answer = mid
            right = mid - 1  # Try for a smaller maximum weight
        else:
            left = mid + 1  # Increase the weight limit
    return answer
```
