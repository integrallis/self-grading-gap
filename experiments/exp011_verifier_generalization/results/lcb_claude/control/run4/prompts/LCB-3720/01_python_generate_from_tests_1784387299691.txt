from typing import List
import heapq


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Create a graph structure from the edges
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[u].append((v, w))

    # Function to check if we can reach node 0 with max weight allowed
    def canReachWithMaxWeight(max_weight: int) -> bool:
        # Using a modified BFS to check reachability to node 0
        reachable = [0] * n
        queue = []
        for i in range(n):
            if i != 0:
                queue.append(i)

        visited = set()  # To track visited nodes
        while queue:
            current = queue.pop(0)
            if current in visited:
                continue
            visited.add(current)
            for neighbor, edge_weight in graph[current]:
                if edge_weight <= max_weight:
                    if neighbor not in visited:
                        queue.append(neighbor)

        return 0 in visited

    # Binary search over the possible max weights
    left, right = 1, 10**6
    answer = -1
    while left <= right:
        mid = (left + right) // 2
        if canReachWithMaxWeight(mid):
            answer = mid
            right = mid - 1  # Try for a smaller maximum weight
        else:
            left = mid + 1  # Increase the weight limit
    return answer

```
