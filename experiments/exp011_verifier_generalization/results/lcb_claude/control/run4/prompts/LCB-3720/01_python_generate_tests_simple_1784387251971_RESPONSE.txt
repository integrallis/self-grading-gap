from typing import List
from solution import minMaxWeight

def test_canonical_example_1():
    # spec: n = 5, edges = [[1,0,1],[2,0,2],[3,0,1],[4,3,1],[2,1,1]], threshold = 2
    # Remove edge 2 -> 0. Max weight left is 1.
    assert minMaxWeight(5, [[1,0,1],[2,0,2],[3,0,1],[4,3,1],[2,1,1]], 2) == 1

def test_canonical_example_2():
    # spec: n = 5, edges = [[0,1,1],[0,2,2],[0,3,1],[0,4,1],[1,2,1],[1,4,1]], threshold = 1
    # Impossible for node 2 to reach node 0.
    assert minMaxWeight(5, [[0,1,1],[0,2,2],[0,3,1],[0,4,1],[1,2,1],[1,4,1]], 1) == -1

def test_canonical_example_3():
    # spec: n = 5, edges = [[1,2,1],[1,3,3],[1,4,5],[2,3,2],[3,4,2],[4,0,1]], threshold = 1
    # Remove edges 1 -> 3 and 1 -> 4. Max weight left is 2.
    assert minMaxWeight(5, [[1,2,1],[1,3,3],[1,4,5],[2,3,2],[3,4,2],[4,0,1]], 1) == 2

def test_canonical_example_4():
    # spec: n = 5, edges = [[1,2,1],[1,3,3],[1,4,5],[2,3,2],[4,0,1]], threshold = 1
    # Impossible for node 4 to reach node 0.
    assert minMaxWeight(5, [[1,2,1],[1,3,3],[1,4,5],[2,3,2],[4,0,1]], 1) == -1
