from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_times = []

    # Calculate free time intervals
    t = 0
    for i in range(n):
        if startTime[i] > t:
            free_times.append((t, startTime[i]))
        t = endTime[i]

    if t < eventTime:
        free_times.append((t, eventTime))

    # Aggregate the maximum free time from existing free intervals
    max_free_time = 0
    for start, end in free_times:
        max_free_time = max(max_free_time, end - start)

    # Look for reschedule options (gaps)
    reschedule_options = []
    for i in range(1, n):
        if endTime[i-1] < startTime[i]:
            reschedule_options.append(startTime[i] - endTime[i-1])

    # If k is greater than or equal to the number of reschedule options
    if k >= len(reschedule_options):
        total_free_time = max_free_time + sum(reschedule_options) + max(0, eventTime - t)
        return total_free_time

    # Sort the gap options in descending order and sum the top k
    reschedule_options.sort(reverse=True)
    additional_free_time = sum(reschedule_options[:k])

    return max_free_time + additional_free_time
