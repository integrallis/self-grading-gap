```python
def calculateScore(s: str) -> int:
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # a-z to z-a mapping
    marked = set()  # to track marked indices
    score = 0  # initialize score

    for i in range(len(s)):
        if i in marked:
            continue
        # find the closest unmarked index j where s[j] is the mirror of s[i]
        for j in range(i - 1, -1, -1):
            if j not in marked and s[j] == mirror[s[i]]:
                marked.add(i)
                marked.add(j)
                score += i - j
                break  # exit loop after finding the first valid j

    return score
```
