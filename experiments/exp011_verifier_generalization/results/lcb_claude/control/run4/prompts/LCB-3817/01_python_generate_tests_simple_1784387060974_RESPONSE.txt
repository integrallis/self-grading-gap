from solution import maxContainers

def test_canonical_example_1():
    # spec: maxContainers(2, 3, 15)
    # Total cells = 2*2 = 4
    # Total weight capacity = 15
    # Each container weight = 3
    # Maximum containers that can be loaded = min(4, 15 // 3) = min(4, 5) = 4
    assert maxContainers(2, 3, 15) == 4

def test_canonical_example_2():
    # spec: maxContainers(3, 5, 20)
    # Total cells = 3*3 = 9
    # Total weight capacity = 20
    # Each container weight = 5
    # Maximum containers that can be loaded = min(9, 20 // 5) = min(9, 4) = 4
    assert maxContainers(3, 5, 20) == 4
