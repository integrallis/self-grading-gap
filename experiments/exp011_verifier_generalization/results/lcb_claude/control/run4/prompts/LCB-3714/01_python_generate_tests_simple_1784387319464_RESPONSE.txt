from solution import minMaxSums

def test_canonical_example_1():
    # spec: nums = [1, 2, 3], k = 2
    # Subsequences are: [1], [2], [3], [1, 2], [1, 3], [2, 3]
    # Their sums: 2 (1+1) + 4 (2+2) + 6 (3+3) + 3 (1+2) + 4 (1+3) + 5 (2+3) = 24
    assert minMaxSums([1, 2, 3], 2) == 24
