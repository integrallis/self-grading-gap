from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    power_of_two = [1] * (n + 1)
    for i in range(1, n + 1):
        power_of_two[i] = (power_of_two[i - 1] * 2) % MOD

    for i in range(1, k + 1):
        for j in range(n):
            min_contrib = power_of_two[n - 1] * power_of_two[n - i] % MOD
            max_contrib = power_of_two[n - 1] * power_of_two[n - i] % MOD
            total_sum += (nums[j] * (min_contrib + max_contrib)) % MOD
            total_sum %= MOD
    return total_sum

```
