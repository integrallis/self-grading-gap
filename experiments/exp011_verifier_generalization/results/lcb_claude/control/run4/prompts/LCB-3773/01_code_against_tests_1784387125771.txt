```python
from typing import List

def minimumPairRemoval(nums: List[int]) -> int:
    operations = 0
    while True:
        # Find the adjacent pair with the minimum sum
        min_sum = float('inf')
        min_index = -1
        for i in range(len(nums) - 1):
            pair_sum = nums[i] + nums[i + 1]
            if pair_sum < min_sum:
                min_sum = pair_sum
                min_index = i

        # If no pair found or array is non-decreasing, break
        if min_index == -1 or all(nums[i] <= nums[i + 1] for i in range(len(nums) - 1)):
            break

        # Replace the pair with their sum
        nums[min_index] = nums[min_index] + nums[min_index + 1]
        nums.pop(min_index + 1)
        operations += 1
    return operations
```
