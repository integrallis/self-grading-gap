from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n == 0:
        return 0

    # Create a list to store the maximum length of subsequence ending at each index
    dp = [1] * n
    max_length = 1

    # Iterate through each pair of elements in nums
    for i in range(n):
        for j in range(i):
            # Calculate absolute difference
            diff1 = abs(nums[i] - nums[j])
            # Check if we can form a valid subsequence
            if dp[j] > 0:
                # Ensure the absolute differences are non-increasing
                if (i == 0) or (abs(nums[i] - nums[i-1]) >= diff1):
                    dp[i] = max(dp[i], dp[j] + 1)
                    max_length = max(max_length, dp[i])

    return max_length

```
