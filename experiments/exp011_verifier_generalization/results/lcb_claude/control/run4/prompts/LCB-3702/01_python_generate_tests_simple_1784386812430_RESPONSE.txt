from typing import List
from solution import maxLength

def test_canonical_example_1():
    # Input: [1, 2, 1, 2, 1, 1, 1]
    # Longest product equivalent subarray is [1, 2, 1, 1, 1]
    # prod([1, 2, 1, 1, 1]) = 2
    # gcd([1, 2, 1, 1, 1]) = 1
    # lcm([1, 2, 1, 1, 1]) = 2
    # prod == lcm * gcd -> 2 == 2 * 1
    # Expected output: 5
    assert maxLength([1, 2, 1, 2, 1, 1, 1]) == 5

def test_canonical_example_2():
    # Input: [2, 3, 4, 5, 6]
    # Longest product equivalent subarray is [3, 4, 5]
    # prod([3, 4, 5]) = 60
    # gcd([3, 4, 5]) = 1
    # lcm([3, 4, 5]) = 60
    # prod == lcm * gcd -> 60 == 60 * 1
    # Expected output: 3
    assert maxLength([2, 3, 4, 5, 6]) == 3

def test_canonical_example_3():
    # Input: [1, 2, 3, 1, 4, 5, 1]
    # Longest product equivalent subarray is [1, 2, 3, 1, 4, 5]
    # prod([1, 2, 3, 1, 4, 5]) = 120
    # gcd([1, 2, 3, 1, 4, 5]) = 1
    # lcm([1, 2, 3, 1, 4, 5]) = 60
    # prod != lcm * gcd, but subarray [2, 3, 1, 4, 5] also has prod == lcm * gcd
    # Expected output: 5
    assert maxLength([1, 2, 3, 1, 4, 5, 1]) == 5
