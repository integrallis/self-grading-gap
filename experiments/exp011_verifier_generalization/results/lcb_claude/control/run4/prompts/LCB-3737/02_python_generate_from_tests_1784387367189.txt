from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize a dp array to store the minimum cost for painting houses
    dp = [[0] * 3 for _ in range(n)]

    # Base case: fill the first house costs
    dp[0] = cost[0]

    for i in range(1, n):
        for j in range(3):
            # Calculate minimum cost for painting house i with color j
            dp[i][j] = cost[i][j] + min(
                dp[i - 1][(j + 1) % 3],  # Previous house different color
                dp[i - 1][(j + 2) % 3]   # Previous house different color
            )

    # We also need to consider the restriction on equidistant houses
    # Calculate the minimum cost for the two possible configurations
    min_cost = float('inf')
    for j in range(3):
        # For the first configuration (0 to n-1)
        total_cost1 = dp[n-1][j]
        for k in range(3):
            if j != k:
                total_cost1 += cost[n//2][k]  # To ensure the equidistant house is different
        min_cost = min(min_cost, total_cost1)

    # Now for the second configuration (1 to n-1)
    for j in range(3):
        total_cost2 = dp[n-1][j]
        for k in range(3):
            if j != k:
                total_cost2 += cost[n//2 - 1][k]  # To ensure the equidistant house is different
        min_cost = min(min_cost, total_cost2)

    return min_cost
```
