from typing import List
from solution import minCosts

def test_canonical_example_1():
    # Input: [5, 3, 4, 1, 3, 2]
    # Cost to reach each position:
    # 0: swap with 0 (cost 5)
    # 1: swap with 1 (cost 3)
    # 2: swap with 1 (cost 3), then free swap with 2 -> cost 3
    # 3: swap with 3 (cost 1)
    # 4: swap with 3 (cost 1), then free swap with 4 -> cost 1
    # 5: swap with 3 (cost 1), then free swap with 5 -> cost 1
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]

def test_canonical_example_2():
    # Input: [1, 2, 4, 6, 7]
    # Cost to reach each position:
    # 0: swap with 0 (cost 1)
    # 1: swap with 0 (cost 1), then free swap with 1 -> cost 1
    # 2: swap with 0 (cost 1), then free swap with 1 (cost 0), then free swap with 2 -> cost 1
    # 3: swap with 0 (cost 1), then free swap with 1 (cost 0), then free swap with 2 (cost 0), then free swap with 3 -> cost 1
    # 4: swap with 0 (cost 1), then free swap with 1 (cost 0), then free swap with 2 (cost 0), then free swap with 3 (cost 0), then free swap with 4 -> cost 1
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]
