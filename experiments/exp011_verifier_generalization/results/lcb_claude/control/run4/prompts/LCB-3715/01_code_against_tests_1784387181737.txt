```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a dictionary to hold the number of coins available at each bag position
    coin_map = {}

    # Fill the coin_map with the number of coins from each segment
    for l, r, c in coins:
        for pos in range(l, r + 1):
            coin_map[pos] = coin_map.get(pos, 0) + c

    # Sort the positions to compute the maximum coins over k consecutive bags
    sorted_positions = sorted(coin_map.keys())
    max_coins = 0
    current_sum = 0
    left = 0

    # Using a sliding window approach to find the maximum sum of coins in k consecutive bags
    for right in range(len(sorted_positions)):
        current_sum += coin_map[sorted_positions[right]]

        # If we reach k bags, update the max_coins and slide the window
        if right - left + 1 > k:
            current_sum -= coin_map[sorted_positions[left]]
            left += 1

        # Update max_coins if we have exactly k bags
        if right - left + 1 == k:
            max_coins = max(max_coins, current_sum)

    return max_coins
```
