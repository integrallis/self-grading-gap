def test_canonical_example_1():
    # spec: minMaxSums([1, 2, 3], 2) -> 24
    # All subsequences with at most 2 elements:
    # [1] -> min: 1, max: 1, sum: 2
    # [2] -> min: 2, max: 2, sum: 4
    # [3] -> min: 3, max: 3, sum: 6
    # [1, 2] -> min: 1, max: 2, sum: 3
    # [1, 3] -> min: 1, max: 3, sum: 4
    # [2, 3] -> min: 2, max: 3, sum: 5
    # Total: 2 + 4 + 6 + 3 + 4 + 5 = 24
