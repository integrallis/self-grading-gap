from typing import List
from solution import maximumCoins

def test_canonical_example_1():
    # Input: coins = [[8,10,1],[1,3,2],[5,6,4]], k = 4
    # Selecting bags at positions [3, 4, 5, 6] gives the maximum number of coins: 2 + 0 + 4 + 4 = 10
    assert maximumCoins([[8,10,1],[1,3,2],[5,6,4]], 4) == 10

def test_canonical_example_2():
    # Input: coins = [[1,10,3]], k = 2
    # Selecting bags at positions [1, 2] gives the maximum number of coins: 3 + 3 = 6
    assert maximumCoins([[1,10,3]], 2) == 6

def test_k_equals_1():
    # Input: coins = [[5,10,7]], k = 1
    # Selecting bag at position [5] gives the maximum number of coins: 7
    assert maximumCoins([[5,10,7]], 1) == 7

def test_k_exceeds_total_span():
    # Input: coins = [[1,3,5]], k = 10
    # Total span of segments is 3, so selecting 10 bags gives: 5 + 5 + 5 + 0 + 0 + 0 + 0 + 0 + 0 + 0 = 15
    assert maximumCoins([[1,3,5]], 10) == 15

def test_optimal_window_multiple_segments():
    # Input: coins = [[1,2,3],[4,5,2]], k = 3
    # Selecting bags at positions [1,2,3] gives the maximum number of coins: 3 + 3 + 0 = 6
    assert maximumCoins([[1,2,3],[4,5,2]], 3) == 6

    # Input: coins = [[1,2,5],[10,12,3]], k = 3
