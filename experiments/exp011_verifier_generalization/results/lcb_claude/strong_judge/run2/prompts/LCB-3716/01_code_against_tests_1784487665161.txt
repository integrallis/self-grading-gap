```python
from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n == 0:
        return 0

    # dp[i] will store the length of the longest subsequence ending at index i
    dp = [1] * n  # Every element is a valid subsequence of length 1
    max_length = 1

    for i in range(n):
        for j in range(i):
            diff_i_j = abs(nums[i] - nums[j])
            if dp[j] > 1:
                # Check the last difference in the subsequence
                last_diff = abs(nums[j] - nums[j - 1]) if j > 0 else float('inf')
                if diff_i_j <= last_diff:
                    dp[i] = max(dp[i], dp[j] + 1)
            else:
                dp[i] = max(dp[i], 2)  # At least nums[j] and nums[i] form a subsequence.
        max_length = max(max_length, dp[i])

    return max_length
```
