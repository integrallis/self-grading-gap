def test_canonical_example_1():

def test_single_element_array_odd():
    # spec: transformArray([5]) -> [1] (5 is odd)
    assert transformArray([5]) == [1]

def test_single_element_array_even():
    # spec: transformArray([2]) -> [0] (2 is even)
    assert transformArray([2]) == [0]

def test_all_even_numbers():
    # spec: transformArray([2, 4, 6]) -> [0, 0, 0] (all are even)

    # spec: transformArray([1, 3, 5]) -> [1, 1, 1] (all are odd)

    # spec: transformArray([999, 1000]) -> [0, 1] (999 is odd, 1000 is even)

def test_large_array():
    # spec: transformArray([2, 4, 6, 1, 3, 5] * 16 + [7])

    # After sorting: [0, 0, 0, 1, 1, 1] * 16 + [1] -> [0, 0, 0, ..., 1, 1]
    assert transformArray([2, 4, 6, 1, 3, 5] * 16 + [7]) == [0, 0, 0] * 16 + [1] * 97
