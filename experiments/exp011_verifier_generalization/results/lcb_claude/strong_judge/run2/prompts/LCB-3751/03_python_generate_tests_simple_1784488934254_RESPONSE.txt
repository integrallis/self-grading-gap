from typing import List
from solution import maxFrequency

def test_canonical_example_1():
    # spec: After adding -5 to nums[2..5], 1 has a frequency of 2 in [1, 2, -2, -1, 0, 1].
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2

def test_canonical_example_2():
    # spec: After adding 8 to nums[1..9], 10 has a frequency of 4 in [10, 10, 11, 12, 13, 13, 12, 11, 10, 10].
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4

def test_single_element_array_equal_k():
    # spec: Single element array where nums[0] = k
    assert maxFrequency([5], 5) == 1  # Only element equals k

def test_single_element_array_transform():
    # spec: Single element array where transformation is needed
    assert maxFrequency([3], 5) == 1  # Add 2 to make it equal to k

def test_k_already_multiple_times():
    # spec: k appears multiple times in the array
    assert maxFrequency([1, 1, 1, 2, 3], 1) >= 3  # At least 3 occurrences of k

def test_boundary_nums_elements_at_1():
    # spec: All elements are 1 and equals k
    assert maxFrequency([1, 1, 1], 1) == 3  # All elements equal to k

def test_boundary_nums_elements_at_50():
    # spec: All elements are 50 and equals k
    assert maxFrequency([50, 50, 50], 50) == 3  # All elements equal to k

def test_boundary_k_at_1():
    # spec: k is at its minimum boundary
    assert maxFrequency([2, 3, 4], 1) == 1  # Add -1 to any element to reach 1

def test_boundary_k_at_50():
    # spec: k is at its maximum boundary
    assert maxFrequency([1, 2, 3], 50) == 1  # Add 49 to make one element equal to k

def test_optimal_subarray_at_start():
    # spec: Optimal subarray is at the start of the array
    assert maxFrequency([1, 2, 3, 4], 3) == 2  # Add 1 to nums[0..1] to make two 3s

def test_optimal_subarray_at_end():
    # spec: Optimal subarray is at the end of the array
    assert maxFrequency([4, 3, 2, 1], 3) == 2  # Add -1 to nums[2..3] to make two 3s

def test_optimal_subarray_entire_array():
    # spec: Optimal subarray is the entire array
    assert maxFrequency([1, 2, 3], 2) == 1  # Cannot make all elements equal to 2 by adding same value

def test_all_elements_same_not_equal_k():
    # spec: All elements the same, but not equal to k

def test_no_operation_needed():
