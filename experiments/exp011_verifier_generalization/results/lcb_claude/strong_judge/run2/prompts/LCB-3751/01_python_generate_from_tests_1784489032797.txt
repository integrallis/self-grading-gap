from typing import List


def maxFrequency(nums: List[int], k: int) -> int:
    nums.sort()
    left = 0
    total = 0
    max_count = 0
    n = len(nums)

    for right in range(n):
        total += nums[right]
        while (right - left + 1) * k > total + (right - left) * k:
            total -= nums[left]
            left += 1
        max_count = max(max_count, right - left + 1)
    return max_count
```
