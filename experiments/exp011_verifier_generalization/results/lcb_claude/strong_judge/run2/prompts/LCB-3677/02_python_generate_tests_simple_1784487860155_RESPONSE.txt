from typing import List
from solution import maximumAmount

def test_canonical_example_1():
    # spec: maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) -> 8
    assert maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) == 8

def test_canonical_example_2():
    # spec: maximumAmount([[10, 10, 10], [10, 10, 10]]) -> 40
    assert maximumAmount([[10, 10, 10], [10, 10, 10]]) == 40

def test_single_cell_grid():
    # spec: maximumAmount([[5]]) -> 5
    assert maximumAmount([[5]]) == 5

def test_using_two_neutralizations():
    # spec: maximumAmount([[1, -1, -1], [1, -1, 1], [1, 1, 1]]) -> 6
    # Optimal path: Start (0,0) -> (0,1) -neutralize-> (0,2) -> (1,2) -> (2,2) -> (2,1) -> (2,0)
    assert maximumAmount([[1, -1, -1], [1, -1, 1], [1, 1, 1]]) == 6

def test_negative_final_total():
    # spec: maximumAmount([[1, -2], [-2, -1]]) -> -4
    # Optimal path: Start (0,0) -> (0,1) -> (1,1)
    assert maximumAmount([[1, -2], [-2, -1]]) == -4

def test_all_negative_values():
    # spec: maximumAmount([[-1, -2], [-3, -4]]) -> -4
    # Optimal path: Start (0,0) -> (1,0) -> (1,1)
    assert maximumAmount([[-1, -2], [-3, -4]]) == -4

def test_boundary_coin_values():
    # spec: maximumAmount([[1000, -1000], [-1000, 1000]]) -> 0
    # Optimal path: Start (0,0) -> (0,1) -> (1,1)
    assert maximumAmount([[1000, -1000], [-1000, 1000]]) == 0

def test_large_grid():
    # spec: maximumAmount([[1]*500]*500) -> 250000
    # All cells have 1 coin, the robot can traverse the entire grid
    assert maximumAmount([[1]*500]*500) == 250000
