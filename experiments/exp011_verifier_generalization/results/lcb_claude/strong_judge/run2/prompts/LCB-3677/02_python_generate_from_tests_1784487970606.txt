from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m = len(coins)
    n = len(coins[0]) if m > 0 else 0
    dp = [[-float('inf')] * (n + 1) for _ in range(m + 1)]
    dp[0][1] = 0

    for i in range(m):
        for j in range(n):
            value = coins[i][j]
            for neutralizations in range(3):
                if value < 0 and neutralizations > 0:
                    dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j] - value)
                else:
                    dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j] + value)

                if j < n - 1:
                    dp[i][j + 2] = max(dp[i][j + 2], dp[i][j + 1])
                if i < m - 1:
                    dp[i + 2][j] = max(dp[i + 2][j], dp[i + 1][j])

    return dp[m][n]

```
