from typing import List
from solution import maximumAmount

def test_canonical_example_1():
    # spec: maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) -> 8
    assert maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) == 8

def test_canonical_example_2():
    # spec: maximumAmount([[10, 10, 10], [10, 10, 10]]) -> 40
    assert maximumAmount([[10, 10, 10], [10, 10, 10]]) == 40

def test_single_cell_grid():
    # spec: maximumAmount([[5]]) -> 5
    assert maximumAmount([[5]]) == 5

def test_using_two_neutralizations():
    # spec: maximumAmount([[1, -1, -1], [1, -1, 1], [1, 1, 1]]) -> 5
    # Optimal path: Start (0,0) -> (1,0) -> (2,0) -> (2,1) -> (2,2)
    # No robbers on this path (total coins = 1 + 1 + 1 + 1 + 1 = 5)
    assert maximumAmount([[1, -1, -1], [1, -1, 1], [1, 1, 1]]) == 5

def test_negative_final_total():
    # spec: maximumAmount([[1, -2], [-2, -1]]) -> 1
    # Optimal path: Start (0,0) -> (0,1) -> (1,1)
    # Neutralize at (0,1): 1 + 0 + 0 = 1
    assert maximumAmount([[1, -2], [-2, -1]]) == 1

def test_all_negative_values():
    # spec: maximumAmount([[-1, -2], [-3, -4]]) -> -1
    # Optimal path: Start (0,0) -> (0,1) -> (1,1)
    # Neutralize two worst: -1 + 0 + 0 = -1
    assert maximumAmount([[-1, -2], [-3, -4]]) == -1

def test_boundary_coin_values():
    # spec: maximumAmount([[1000, -1000], [-1000, 1000]]) -> 2000
    # Optimal path: Start (0,0) -> (0,1) (neutralize) -> (1,1)
    assert maximumAmount([[1000, -1000], [-1000, 1000]]) == 2000

def test_large_grid():
    # spec: maximumAmount([[1]*500]*500) -> 999
    # All cells have 1 coin, the robot can traverse the entire grid
    assert maximumAmount([[1]*500]*500) == 999
