from typing import List
from solution import countMentions

def test_canonical_example_1():
    # Example 1: 
    # Input: numberOfUsers = 2, events = [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","71","HERE"]]
    # At timestamp 10: id1 and id0 mentioned -> mentions = [1, 1]
    # At timestamp 11: id0 goes offline -> mentions remain [1, 1]
    # At timestamp 71: HERE mentions all online users (id1 is online) -> mentions = [2, 2]
    assert countMentions(2, [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","71","HERE"]]) == [2, 2]

def test_canonical_example_2():
    # Example 2:

    # At timestamp 11: id0 goes offline -> mentions remain [1, 1]
    # At timestamp 12: ALL mentions all users (both id0 and id1) -> mentions = [2, 2]
