from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    if n == 0:
        return eventTime  # If no meetings, all time is free

    # Calculate total meeting duration
    total_meeting_duration = sum(end - start for start, end in zip(startTime, endTime))

    # If k is greater than or equal to the number of meetings, all can be rescheduled
    if k >= n:
        return eventTime - total_meeting_duration

    # Calculate free time at the start and end
    free_time_start = startTime[0] - 0  # Free time before the first meeting
    free_time_end = eventTime - endTime[-1]  # Free time after the last meeting

    # We can try to reduce the total meeting time by rescheduling k meetings
    # Sort meetings by their duration (smallest first)
    meeting_durations = sorted(end - start for start, end in zip(startTime, endTime))

    # Reduce the k smallest meeting durations to maximize free time
    reduction = sum(meeting_durations[i] for i in range(k))

    # Calculate the maximum free time possible
    max_free_time = free_time_start + free_time_end + (eventTime - total_meeting_duration + reduction)

    return max(max_free_time, 0)  # Ensure we don't return negative free time
```
