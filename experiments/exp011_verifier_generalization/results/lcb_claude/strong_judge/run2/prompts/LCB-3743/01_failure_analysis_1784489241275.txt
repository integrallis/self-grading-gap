from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    # Calculate initial free time gaps
    free_gaps = []
    if startTime[0] > 0:
        free_gaps.append((0, startTime[0]))  # Free time before first meeting
    for i in range(1, n):
        if endTime[i - 1] < startTime[i]:
            free_gaps.append((endTime[i - 1], startTime[i]))  # Gap between meetings
    if endTime[n - 1] < eventTime:
        free_gaps.append((endTime[n - 1], eventTime))  # Free time after last meeting

    # If k is greater than or equal to the number of meetings, all can be rescheduled
    if k >= n:
        total_duration = sum(end - start for start, end in zip(startTime, endTime))
        return eventTime - total_duration

    # Now we need to consider rescheduling to maximize free time
    max_free_time = 0
    for i in range(len(free_gaps)):
        gap_start, gap_end = free_gaps[i]
        gap_duration = gap_end - gap_start
        # If we can adjust meetings into this gap
        if i < len(free_gaps) - 1:
            next_gap_start, _ = free_gaps[i + 1]
            # Calculate how many meetings fit into this gap
            meetings_fit = 0
            for j in range(n):
                if startTime[j] < next_gap_start and endTime[j] > gap_start:
                    meetings_fit += 1
            # We can only move k meetings
            adjustments = min(meetings_fit, k)
            # Calculate new free time
            new_free_time = gap_duration + adjustments
            max_free_time = max(max_free_time, new_free_time)

    return max(max_free_time, *[gap[1] - gap[0] for gap in free_gaps])
