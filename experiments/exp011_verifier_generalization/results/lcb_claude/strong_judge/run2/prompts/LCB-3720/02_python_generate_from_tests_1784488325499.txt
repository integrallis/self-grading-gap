from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    from collections import defaultdict, deque
    import sys

    # Create adjacency list and weight mapping
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[v].append((u, w))  # Reverse graph to find paths to node 0

    # Function to check if all nodes can reach 0 under max_weight and threshold
    def canReachAll(max_weight):
        out_degree = [0] * n
        queue = deque([0])
        visited = [False] * n
        visited[0] = True

        while queue:
            current = queue.popleft()
            for neighbor, weight in graph[current]:
                if weight <= max_weight:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        queue.append(neighbor)
                    out_degree[neighbor] += 1
                    if out_degree[neighbor] > threshold:
                        return False

        return all(visited)

    # Binary search for the minimum possible maximum weight
    left, right = 1, 10**6
    answer = -1
    while left <= right:
        mid = (left + right) // 2
        if canReachAll(mid):
            answer = mid
            right = mid - 1
        else:
            left = mid + 1

    return answer
```
