from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    from collections import defaultdict, deque

    # Create adjacency list for the original graph
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))  # Forward graph for original edges

    # Function to check if all nodes can reach 0 under max_weight and threshold
    def canReachAll(max_weight):
        out_degree = [0] * n
        visited = [False] * n

        # Count out-degrees for nodes with edges that have weight <= max_weight
        for node in range(n):
            for neighbor, weight in graph[node]:
                if weight <= max_weight:
                    out_degree[node] += 1

        # Perform BFS for each node 1 to n-1 to see if they can reach node 0
        for start_node in range(1, n):
            if visited[start_node]:
                continue
            queue = deque([start_node])
            visited[start_node] = True

            while queue:
                current = queue.popleft()
                for neighbor, weight in graph[current]:
                    if weight <= max_weight:
                        if out_degree[current] >= threshold:
                            continue  # Skip if threshold exceeded
                        out_degree[current] += 1
                        if not visited[neighbor]:
                            visited[neighbor] = True
                            queue.append(neighbor)

        # Check if all nodes are visited
        return all(visited[1:])  # Skip node 0 since it doesn't need to reach itself

    # Binary search for the minimum possible maximum weight
    left, right = 1, 10**6
    answer = -1
    while left <= right:
        mid = (left + right) // 2
        if canReachAll(mid):
            answer = mid
            right = mid - 1
        else:
            left = mid + 1

    return answer
