## Implementation

```python
from typing import List

def minCosts(cost: List[int]) -> List[int]:
    n = len(cost)
    answer = [0] * n
    global_min = min(cost)  # Find the global minimum cost
    answer[0] = global_min  # The cost to reach position 0
    for i in range(1, n):
        answer[i] = min(answer[i - 1], cost[i])  # Minimum cost to reach position i
    return answer
```
