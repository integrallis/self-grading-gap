## Implementation

```python
from collections import Counter

def maxDifference(s: str) -> int:
    frequency = Counter(s)
    max_odd = float('-inf')
    min_even = float('inf')

    for count in frequency.values():
        if count % 2 == 0:  # even
            min_even = min(min_even, count)
        else:  # odd
            max_odd = max(max_odd, count)

    return max_odd - min_even

```
