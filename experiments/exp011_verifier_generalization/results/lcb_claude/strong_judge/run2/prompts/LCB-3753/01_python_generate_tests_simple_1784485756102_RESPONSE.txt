def test_canonical_example_1():
    # For input "aaaaabbc":
    # Frequency of 'a' = 5 (odd), Frequency of 'b' = 2 (even)
    # Difference = 5 - 2 = 3

def test_canonical_example_2():
    # For input "abcabcab":
    # Frequency of 'a' = 3 (odd), Frequency of 'c' = 2 (even)
    # Difference = 3 - 2 = 1
    assert maxDifference("abcabcab") == 1
