from solution import maxDifference

def test_canonical_example_1():
    # For input "aaaaabbc":
    # Frequency of 'a' = 5 (odd), Frequency of 'b' = 2 (even)
    # Difference = 5 - 2 = 3
    assert maxDifference("aaaaabbc") == 3

def test_canonical_example_2():
    # For input "abcabcab":
    # Frequency of 'a' = 3 (odd), Frequency of 'c' = 2 (even)
    # Difference = 3 - 2 = 1
    assert maxDifference("abcabcab") == 1

def test_minimum_length():
    # For input "aab":
    # Frequency of 'a' = 2 (even), Frequency of 'b' = 1 (odd)
    # Difference = 1 - 2 = -1, but we want the maximum difference which is 1 - 2 = -1
    # Thus, the maximum difference can only be 1 - 2 = -1, but since we need max, we consider it 0.
    assert maxDifference("aab") == -1

def test_maximum_length():
    # For input with 100 characters: 99 'a's and 2 'b's
    # Frequency of 'a' = 99 (odd), Frequency of 'b' = 2 (even)
    # Difference = 99 - 2 = 97
    assert maxDifference("a" * 99 + "b" * 2) == 97

def test_maximum_difference():
    # For input "aaabbbb":
    # Frequency of 'a' = 3 (odd), Frequency of 'b' = 4 (even)
    # Difference = 3 - 4 = -1, maximum difference considering odd - even
    assert maxDifference("aaabbbb") == -1

def test_multiple_characters():
    # For input "aabbccddd":
    # Frequencies: a=2 (even), b=2 (even), c=2 (even), d=3 (odd)
    # Maximum difference = 3 (odd) - 2 (even) = 1
    assert maxDifference("aabbccddd") == 1
