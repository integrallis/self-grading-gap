from typing import List

def maxSum(nums: List[int]) -> int:
    seen = set()
    max_sum = float('-inf')
    current_sum = 0
    left = 0

    for right in range(len(nums)):
        while nums[right] in seen:
            seen.remove(nums[left])
            current_sum -= nums[left]
            left += 1
        seen.add(nums[right])
        current_sum += nums[right]
        max_sum = max(max_sum, current_sum)

    return max_sum if max_sum != float('-inf') else 0
```

In the case of all-negative numbers with no duplicates, it accumulates all elements from left to right, getting progressively worse sums, and never considers taking just a single element from the middle.
