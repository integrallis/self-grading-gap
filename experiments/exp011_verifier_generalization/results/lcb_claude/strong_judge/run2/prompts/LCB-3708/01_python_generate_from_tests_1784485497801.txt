from typing import List

def zigzagTraversal(grid: List[List[int]]) -> List[int]:
    result = []
    for i in range(len(grid)):
        if i % 2 == 0:
            # Traverse left to right
            for j in range(0, len(grid[i]), 2):
                result.append(grid[i][j])
        else:
            # Traverse right to left
            for j in range(len(grid[i]) - 2, -1, -2):
                result.append(grid[i][j])
            # Append the last element of the row if it exists
            if len(grid[i]) % 2 == 1:
                result.append(grid[i][-1])
    return result
```
