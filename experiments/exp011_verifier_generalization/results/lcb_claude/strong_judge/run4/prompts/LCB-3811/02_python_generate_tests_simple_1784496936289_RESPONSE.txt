from solution import reverseDegree

def test_canonical_example_1():
    # 'a' = 26, 'b' = 25, 'c' = 24
    # (26*1) + (25*2) + (24*3) = 26 + 50 + 72 = 148
    assert reverseDegree("abc") == 148

def test_canonical_example_2():
    # 'z' = 1, 'a' = 26, 'z' = 1, 'a' = 26
    # (1*1) + (26*2) + (1*3) + (26*4) = 1 + 52 + 3 + 104 = 160
    assert reverseDegree("zaza") == 160

def test_single_character():
    # 'a' = 26
    # (26*1) = 26
    assert reverseDegree("a") == 26

def test_long_string():
    # Testing a longer string to verify scaling
    # 'a' = 26, 'b' = 25, ..., 'z' = 1
    # For "az": (26*1) + (1*2) = 26 + 2 = 28
    assert reverseDegree("az") == 28

def test_full_alphabet():
    # Testing the whole alphabet from 'a' to 'z'
    # Sum for each character: 
    # (26*1) + (25*2) + (24*3) + (23*4) + (22*5) + (21*6) + (20*7) + (19*8) + (18*9) + (17*10)
    # + (16*11) + (15*12) + (14*13) + (13*14) + (12*15) + (11*16) + (10*17) + (9*18) + (8*19) + (7*20)
    # + (6*21) + (5*22) + (4*23) + (3*24) + (2*25) + (1*26) = 351
    assert reverseDegree("abcdefghijklmnopqrstuvwxyz") == 351
