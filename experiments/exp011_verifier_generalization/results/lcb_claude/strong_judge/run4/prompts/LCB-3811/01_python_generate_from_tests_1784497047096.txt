def reverseDegree(s: str) -> int:
    total = 0
    for i, char in enumerate(s):
        position_in_reversed_alphabet = 26 - (ord(char) - ord('a')) + 1
        position_in_string = i + 1
        total += position_in_reversed_alphabet * position_in_string
    return total

```
