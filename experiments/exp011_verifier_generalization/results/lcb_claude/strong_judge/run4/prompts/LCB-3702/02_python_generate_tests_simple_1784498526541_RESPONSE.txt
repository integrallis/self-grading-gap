from typing import List
from solution import maxLength

def test_canonical_example_1():
    # spec: longest product equivalent subarray in [1, 2, 1, 2, 1, 1, 1] is [1, 2, 1, 1, 1]
    # prod([1, 2, 1, 1, 1]) = 2, gcd([1, 2, 1, 1, 1]) = 1, lcm([1, 2, 1, 1, 1]) = 2
    assert maxLength([1, 2, 1, 2, 1, 1, 1]) == 5

def test_canonical_example_2():
    # spec: longest product equivalent subarray in [2, 3, 4, 5, 6] is [3, 4, 5]
    # prod([3, 4, 5]) = 60, gcd([3, 4, 5]) = 1, lcm([3, 4, 5]) = 60
    assert maxLength([2, 3, 4, 5, 6]) == 3

def test_canonical_example_3():
    # spec: longest product equivalent subarray in [1, 2, 3, 1, 4, 5, 1] is [1, 2, 3, 1, 4, 5] (6 elements)
    # prod([1, 2, 3, 1, 4, 5]) = 120, gcd([1, 2, 3, 1, 4, 5]) = 1, lcm([1, 2, 3, 1, 4, 5]) = 120
    assert maxLength([1, 2, 3, 1, 4, 5, 1]) == 5

def test_minimum_array_length():
    # spec: minimum length array [1, 2] should return 1 as both single elements are product equivalent
    assert maxLength([1, 2]) >= 1

def test_identical_elements():
    # spec: with identical elements [3, 3, 3, 3], the whole array is product equivalent
    # prod([3, 3, 3, 3]) = 81, gcd([3, 3, 3, 3]) = 3, lcm([3, 3, 3, 3]) = 3
    assert maxLength([3, 3, 3, 3]) == 4

def test_entire_array_product_equivalent():
    # spec: entire array [2, 2, 2] should return full length as all are identical and product equivalent
    assert maxLength([2, 2, 2]) == 3

def test_single_element_subarrays():
    # spec: in array [1, 2, 3], only single elements [1], [2], [3] are product equivalent
    assert maxLength([1, 2, 3]) >= 1
