from solution import calculateScore

def test_canonical_example_1():
    # spec: calculateScore("aczzx") -> 5
    # Explanation: 
    # i = 2: closest j = 0 (mirror of 'a' is 'z'), score += 2 - 0 = 2
    # i = 4: closest j = 1 (mirror of 'c' is 'x'), score += 4 - 1 = 3
    assert calculateScore("aczzx") == 5

def test_canonical_example_2():
    # spec: calculateScore("abcdef") -> 0
    # Explanation: 
    # There are no indices j < i that are mirrors of any s[i].
    assert calculateScore("abcdef") == 0

def test_single_character():
    # spec: calculateScore("a") -> 0
    # Explanation:
    # There are no indices j < i that can be selected, so score remains 0.
    assert calculateScore("a") == 0

def test_all_characters_can_be_paired():
    # spec: calculateScore("az") -> 1
    # Explanation:
    # i = 1: closest j = 0 (mirror of 'a' is 'z'), score += 1 - 0 = 1
    assert calculateScore("az") == 1

def test_repeated_characters():
    # spec: calculateScore("azza") -> 4
    # Explanation:
    # i = 1: closest j = 0 (mirror of 'a' is 'z'), score += 1 - 0 = 1
    # i = 3: closest j = 2 (mirror of 'z' is 'a'), score += 3 - 2 = 1
    assert calculateScore("azza") == 2

def test_multiple_mirrors():
    # spec: calculateScore("abzy") -> 4
    # Explanation:
    # i = 1: closest j = 0 (mirror of 'a' is 'z'), score += 1 - 0 = 1
    # i = 3: closest j = 2 (mirror of 'b' is 'y'), score += 3 - 2 = 1
    assert calculateScore("abzy") == 2
