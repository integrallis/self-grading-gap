```python
from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize the dp array where dp[i][j] = minimum cost to paint up to house i with color j
    dp = [[0] * 3 for _ in range(n)]

    # Base case: for the first house, the cost is just the cost of painting it
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the dp array with the minimum costs
    for i in range(1, n):
        for j in range(3):
            # Choose the minimum cost for painting house i with color j
            dp[i][j] = cost[i][j] + min(
                dp[i - 1][(j + 1) % 3],  # Previous house different color
                dp[i - 1][(j + 2) % 3]   # Previous house different color
            )

    # Now we need to account for the equidistant condition
    min_cost = float('inf')
    for first_color in range(3):
        for second_color in range(3):
            if first_color != second_color:  # Ensure first and second color are different
                total_cost = dp[n - 1][first_color] + dp[n - 2][second_color]  # Last two houses with different colors
                min_cost = min(min_cost, total_cost)

    return min_cost

```
