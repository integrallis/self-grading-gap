from typing import List
from solution import sumOfGoodNumbers

def test_canonical_example_1():
    # Spec: nums = [1, 3, 2, 1, 5, 4], k = 2
    # Good numbers: nums[1] = 3 (greater than nums[-1] (not exists) and nums[3] = 1)
    #               nums[4] = 5 (greater than nums[2] = 2 and nums[6] (not exists))
    #               nums[5] = 4 (greater than nums[3] = 1 and nums[7] (not exists))
    # Sum = 3 + 5 + 4 = 12
    assert sumOfGoodNumbers([1, 3, 2, 1, 5, 4], 2) == 12

def test_canonical_example_2():
    # Spec: nums = [2, 1], k = 1
    # Good number: nums[0] = 2 (greater than nums[1] = 1)
    # Sum = 2
    assert sumOfGoodNumbers([2, 1], 1) == 2

def test_edge_case_no_neighbors():
    # Edge case: nums = [5, 3, 7], k = 2
    # Good numbers: nums[0] = 5 (no left neighbor, right neighbor nums[2] = 7)
    #               nums[1] = 3 (greater than nums[-1] (not exists) and nums[3] (not exists))
    #               nums[2] = 7 (left neighbor nums[0] = 5, no right neighbor)
    # Sum = 5 + 3 + 7 = 15
    assert sumOfGoodNumbers([5, 3, 7], 2) == 15

def test_all_elements_good():
    # All elements good: nums = [5, 1, 5, 1], k = 2
    # Good numbers: nums[0] = 5 (no left neighbor, right neighbor nums[2] = 5)
    #               nums[1] = 1 (greater than nums[-1] (not exists) and nums[3] = 1)
    #               nums[2] = 5 (left neighbor nums[0] = 5, no right neighbor)
    #               nums[3] = 1 (greater than nums[1] = 1)
    # Sum = 5 + 1 + 5 + 1 = 12
    assert sumOfGoodNumbers([5, 1, 5, 1], 2) == 12

def test_no_elements_good():
    # No elements good: nums = [1, 5, 1], k = 1
    # Good numbers: nums[0] = 1 (greater than nums[1] = 5)
    #               nums[1] = 5 (greater than nums[0] = 1)
    #               nums[2] = 1 (greater than nums[1] = 5)
    # Sum = 0 (none are good)
    assert sumOfGoodNumbers([1, 5, 1], 1) == 0
