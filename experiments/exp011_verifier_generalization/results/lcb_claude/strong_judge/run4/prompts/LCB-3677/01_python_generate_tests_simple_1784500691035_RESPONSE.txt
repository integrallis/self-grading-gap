from typing import List
from solution import maximumAmount

def test_canonical_example_1():
    # Input: coins = [[0, 1, -1], [1, -2, 3], [2, -3, 4]]
    # Explanation: The optimal path is (0,0) -> (0,1) -> (1,1, neutralize) -> (1,2) -> (2,2)
    # Total coins = 0 + 1 + 0 + 3 + 4 = 8
    assert maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) == 8

def test_canonical_example_2():
    # Input: coins = [[10, 10, 10], [10, 10, 10]]
    # Explanation: The optimal path is (0,0) -> (0,1) -> (0,2) -> (1,2)
    # Total coins = 10 + 10 + 10 + 10 = 40
    assert maximumAmount([[10, 10, 10], [10, 10, 10]]) == 40
