## Implementation

```python
def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]
    prefix_length = len(prefix)
    suffix_length = len(suffix)

    for i in range(len(s) - prefix_length - suffix_length + 1):
        if s[i:i + prefix_length] == prefix:
            for j in range(i + prefix_length, len(s) - suffix_length + 1):
                if s[j:j + suffix_length] == suffix:
                    return True
    return False
```
