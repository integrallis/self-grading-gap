## Implementation

```python
from typing import List

def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index+1:]
    # Check if there is a match for the prefix
    for i in range(len(s) - len(prefix) - len(suffix) + 1):
        if s[i:i + len(prefix)] == prefix:
            # Check if the suffix matches
            if s[i + len(prefix):].endswith(suffix):
                return True
    return False
```
