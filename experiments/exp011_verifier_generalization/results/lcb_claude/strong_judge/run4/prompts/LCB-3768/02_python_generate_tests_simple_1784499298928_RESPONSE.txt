from solution import hasSameDigits

def test_canonical_example_1():
    # spec: hasSameDigits("3902") -> true
    # First operation: "3902" -> "292"
    # Second operation: "292" -> "11"
    # Since '1' == '1', output is true
    assert hasSameDigits("3902") is True

def test_canonical_example_2():
    # spec: hasSameDigits("34789") -> false
    # First operation: "34789" -> "7157"
    # Second operation: "7157" -> "862"
    # Third operation: "862" -> "48"
    # Since '4' != '8', output is false
    assert hasSameDigits("34789") is False

def test_three_digit_input():
    # spec: hasSameDigits("123") -> false
    # First operation: "123" -> "35" (1+2=3, 2+3=5)
    # Final check: '3' != '5', output is false
    assert hasSameDigits("123") is False

def test_identical_digits():
    # spec: hasSameDigits("3333") -> true
    # First operation: "3333" -> "66" (3+3=6, 3+3=6)
    # Second operation: "66" -> "00" (6+6=12 -> 2)
    # Since '0' == '0', output is true
    assert hasSameDigits("3333") is True

def test_with_zeros():
    # spec: hasSameDigits("1002") -> true
    # First operation: "1002" -> "02" (1+0=1, 0+0=0, 0+2=2)
    # Second operation: "02" -> "2" (0+2=2 but we need two digits)
    # Since '0' == '0', output is true
    assert hasSameDigits("1002") is True

def test_large_input():
    # spec: hasSameDigits("1234567890") -> false
    # First operation: "1234567890" -> "357911" (1+2=3, 2+3=5, 3+4=7, ...)
    # Continues until the final two digits
    # Final check will eventually show that digits are different
    assert hasSameDigits("1234567890") is False

def test_false_with_minimum_length():
    # spec: hasSameDigits("145") -> false
    # First operation: "145" -> "49" (1+4=5, 4+5=9)
