from typing import List
from solution import maxFrequency

def test_canonical_example_1():
    # spec: maxFrequency([1, 2, 3, 4, 5, 6], 1) -> 2
    # Explanation: After adding -5 to nums[2..5], 1 has a frequency of 2 in [1, 2, -2, -1, 0, 1].
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2

def test_canonical_example_2():
    # spec: maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) -> 4
    # Explanation: After adding 8 to nums[1..9], 10 has a frequency of 4 in [10, 10, 11, 12, 13, 13, 12, 11, 10, 10].
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4

def test_single_element_array():
    # spec: maxFrequency([5], 5) -> 1
    # Explanation: Only one element which is equal to k.
    assert maxFrequency([5], 5) == 1

def test_all_elements_equal_k():
    # spec: maxFrequency([3, 3, 3], 3) -> 3
    # Explanation: All elements are already k, frequency is 3.
    assert maxFrequency([3, 3, 3], 3) == 3

def test_k_not_in_array():
    # spec: maxFrequency([5, 6, 7], 1) -> 1
    # Explanation: Adding -4 to nums[0..0] gives [1, 6, 7], frequency of 1.
    assert maxFrequency([5, 6, 7], 1) == 1

def test_subarray_at_start():
    # spec: maxFrequency([1, 2, 3, 4, 5], 4) -> 2
    # Explanation: Adding 1 to nums[0..2] gives [2, 3, 4, 5] with two 4's.
    assert maxFrequency([1, 2, 3, 4, 5], 4) == 2

def test_subarray_at_end():
    # spec: maxFrequency([1, 2, 3, 4, 5], 5) -> 2
    # Explanation: Adding 1 to nums[3..4] gives [1, 2, 3, 5, 6] with two 5's.
    assert maxFrequency([1, 2, 3, 4, 5], 5) == 2

def test_single_element_subarray():
    # spec: maxFrequency([1, 2, 3], 2) -> 2
    # Explanation: Adding 1 to nums[0..0] gives [2, 2, 3], frequency of 2.
    assert maxFrequency([1, 2, 3], 2) == 2

def test_full_array_as_subarray():
    # spec: maxFrequency([1, 2, 3], 2) -> 2
    # Explanation: Adding 1 to nums[0..0] gives [2, 2, 3], frequency of 2.
    assert maxFrequency([1, 2, 3], 2) == 2

def test_minimum_boundary_values():
    # spec: maxFrequency([1], 1) -> 1
    # Explanation: The only element is equal to k, so frequency is 1.
    assert maxFrequency([1], 1) == 1

def test_maximum_boundary_values():
    # spec: maxFrequency([50, 50, 50], 50) -> 3
    # Explanation: All elements are equal to k, frequency is 3.
    assert maxFrequency([50, 50, 50], 50) == 3

def test_mixed_boundary_values():
    # spec: maxFrequency([1, 50], 25) -> 1
    # Explanation: Select [0..0] and add 24: [25, 50] → one 25.
    assert maxFrequency([1, 50], 25) == 1
