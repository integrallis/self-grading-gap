```python
from typing import List

def maxFrequency(nums: List[int], k: int) -> int:
    left = 0
    total = 0
    frequency = 0
    count = [0] * 51  # Since nums[i] ranges from 1 to 50

    for right in range(len(nums)):
        count[nums[right]] += 1
        total += nums[right]

        # Calculate the cost to make all elements in the window equal to k
        while total + (right - left + 1) * k > (right - left + 1) * k:
            total -= nums[left]
            left += 1

        frequency = max(frequency, right - left + 1)

    return frequency
```
