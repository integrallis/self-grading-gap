```python
def maxDistance(s: str, k: int) -> int:
    # Initialize the starting position and max distance
    x, y = 0, 0
    max_distance = 0
    # Movement mapping
    movement = {'N': (0, 1), 'S': (0, -1), 'E': (1, 0), 'W': (-1, 0)}
    # Track the movements
    for i in range(len(s)):
        dx, dy = movement[s[i]]
        x += dx
        y += dy
        # Calculate current Manhattan distance
        current_distance = abs(x) + abs(y)
        # Update the max distance found so far
        max_distance = max(max_distance, current_distance)
        # Calculate the potential maximum distance if we change up to k moves
        remaining_changes = k - (i + 1)  # Changes left after this move
        if remaining_changes >= 0:
            max_distance = max(max_distance, current_distance + remaining_changes)
        else:
            # Calculate additional distance we could achieve by changing moves
            max_distance = max(max_distance, current_distance + min(remaining_changes + (i + 1), 0))
    return max_distance
```
