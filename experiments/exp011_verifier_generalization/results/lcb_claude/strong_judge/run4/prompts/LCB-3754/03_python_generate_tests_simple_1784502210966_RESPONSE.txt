from solution import maxDistance

def test_canonical_example_1():
    # spec: maxDistance("NWSE", 1) -> 3
    assert maxDistance("NWSE", 1) == 3

def test_canonical_example_2():
    # spec: maxDistance("NSWWEW", 3) -> 6
    assert maxDistance("NSWWEW", 3) == 6

def test_k_is_zero():
    # spec: maxDistance("NSEW", 0) -> 1
    # No changes allowed, max distance occurs after first move N to position (0,1).
    assert maxDistance("NSEW", 0) == 1

def test_k_equals_length():
    # spec: maxDistance("SSSS", 4) -> 4
    # All can be changed to 'N', resulting in a maximum distance of 4.
    assert maxDistance("SSSS", 4) == 4

def test_single_character():
    # spec: maxDistance("N", 0) -> 1
    # With one movement north and no changes allowed, the max distance is 1.
    assert maxDistance("N", 0) == 1

def test_straight_line():
    # spec: maxDistance("NNNN", 0) -> 4
    # Moving north 4 times with no changes allowed, max distance is 4.
    assert maxDistance("NNNN", 0) == 4

def test_cancel_out():
    # spec: maxDistance("NSNS", 0) -> 1
    # Movements cancel each other out, max distance is 1 after first N.
    assert maxDistance("NSNS", 0) == 1

def test_k_one_long_string():
    # spec: maxDistance("NNNEEWW", 1) -> 6
    # Changing one 'W' to 'E', maximum distance becomes 6.
    assert maxDistance("NNNEEWW", 1) == 6

def test_max_distance_at_final_position():
    # spec: maxDistance("NSEW", 1) -> 3
    # Changing one character can lead to maximum distance of 3 during movement.
    assert maxDistance("NSEW", 1) == 3

def test_optimal_strategy_both_axes():
    # spec: maxDistance("NNSS", 2) -> 4
    # Changing two S to N allows reaching (0,4), max distance 4.
    assert maxDistance("NNSS", 2) == 4

def test_long_string_performance():
    # spec: maxDistance("E" * (10**5), 0) -> 100000
    # All moves to the east with no changes, max distance is 100000.
    assert maxDistance("E" * (10**5), 0) == 100000
