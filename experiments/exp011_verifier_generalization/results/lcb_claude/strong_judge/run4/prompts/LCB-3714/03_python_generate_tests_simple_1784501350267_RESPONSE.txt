def test_large_values_modulo():
    # spec: minMaxSums([10**9] * 70, 70) 
    # Each subsequence has min = max = 10^9, total subsequences = 2^70 - 1
    # Total = (2^70 - 1) * 2 * 10^9
    # We need to compute this modulo (10^9 + 7)
    MOD = 10**9 + 7
    total_subsequences = (pow(2, 70, MOD) - 1) % MOD
    expected_value = (total_subsequences * 2 * 10**9) % MOD
    assert minMaxSums([10**9] * 70, 70) == expected_value

def test_k_equals_length():
    # spec: minMaxSums([1, 2, 3, 4], 4) 
    # Totaling subsequences:
    # Size 1: 20 (1+2+3+4)
    # Size 2: 30 (1+2, 1+3, 1+4, 2+3, 2+4, 3+4)
    # Size 3: 20 (1+2+3, 1+2+4, 1+3+4, 2+3+4)
    # Size 4: 5 (1+2+3+4)
    # Total = 20 + 30 + 20 + 5 = 75
    assert minMaxSums([1, 2, 3, 4], 4) == 75

def test_single_element_array():
    # spec: minMaxSums([10], 1) -> 20 (10 + 10)
    assert minMaxSums([10], 1) == 20
