from typing import List


def minMaxSums(nums: List[int], k: int) -> int:
    MOD = 10**9 + 7
    n = len(nums)
    total_sum = 0

    # Calculate how many times each number is counted as min and max
    for i in range(n):
        # Each number nums[i] can be the minimum in subsequences of size 1 to k
        # Count of subsequences including nums[i] as min
        count_min = (pow(2, i, MOD) * (pow(2, n - i - 1, MOD) - 1 + MOD) % MOD) % MOD
        # Count of subsequences including nums[i] as max
        count_max = (pow(2, n - i - 1, MOD) * (pow(2, i, MOD) - 1 + MOD) % MOD) % MOD

        # Add to total sum
        total_sum = (total_sum + nums[i] * (count_min + count_max)) % MOD

    return total_sum

```
