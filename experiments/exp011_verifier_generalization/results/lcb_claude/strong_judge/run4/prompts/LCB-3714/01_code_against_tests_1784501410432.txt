```python
from typing import List


def minMaxSums(nums: List[int], k: int) -> int:
    MOD = 10**9 + 7
    n = len(nums)
    total_sum = 0
    
    # Sort nums to handle min and max efficiently
    nums.sort()

    # Precompute powers of 2 up to k
    power_of_2 = [1] * (k + 1)
    for i in range(1, k + 1):
        power_of_2[i] = (power_of_2[i - 1] * 2) % MOD

    # Calculate contributions for each element
    for i in range(n):
        # Each number nums[i] can be a minimum in subsequences of size 1 to k
        # Count of subsequences of size j that can include nums[i] as min
        for j in range(1, k + 1):
            # The number of ways to choose (j-1) elements from the remaining (n-i-1) elements
            if j <= n - i:
                count_min = power_of_2[j - 1]
                total_sum = (total_sum + nums[i] * count_min) % MOD
            
            # The number of ways to choose (j-1) elements from the previous (i) elements
            if j <= i + 1:
                count_max = power_of_2[j - 1]
                total_sum = (total_sum + nums[i] * count_max) % MOD

    return total_sum % MOD

```
