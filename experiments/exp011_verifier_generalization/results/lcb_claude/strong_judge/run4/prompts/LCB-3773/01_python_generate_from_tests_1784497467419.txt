from typing import List

def minimumPairRemoval(nums: List[int]) -> int:
    operations = 0
    while True:
        n = len(nums)
        # Check if the array is already non-decreasing
        if all(nums[i] <= nums[i + 1] for i in range(n - 1)):
            break
        # Find the pair with the minimum sum among adjacent violating pairs
        min_sum = float('inf')
        min_index = -1
        for i in range(n - 1):
            if nums[i] > nums[i + 1]:  # Only consider violating pairs
                pair_sum = nums[i] + nums[i + 1]
                if pair_sum < min_sum:
                    min_sum = pair_sum
                    min_index = i
        # If no violating pairs found, break
        if min_index == -1:
            break
        # Replace the pair with their sum
        nums[min_index] = min_sum
        del nums[min_index + 1]
        operations += 1
    return operations
```
