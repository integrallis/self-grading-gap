from typing import List
from solution import maximumCoins

def test_canonical_example_1():
    # coins = [[8, 10, 1], [1, 3, 2], [5, 6, 4]], k = 4
    # Selecting bags at positions [3, 4, 5, 6] gives the maximum number of coins: 2 + 0 + 4 + 4 = 10
    assert maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) == 10

def test_canonical_example_2():
    # coins = [[1, 10, 3]], k = 2
    # Selecting bags at positions [1, 2] gives the maximum number of coins: 3 + 3 = 6
    assert maximumCoins([[1, 10, 3]], 2) == 6

def test_k_equals_1():
    # coins = [[1, 5, 3]], k = 1
    # Selecting bag at position [1] gives the maximum number of coins: 3
    assert maximumCoins([[1, 5, 3]], 1) == 3

def test_k_spans_multiple_segments():
    # coins = [[1, 2, 5], [4, 5, 3]], k = 4
    # Selecting bags at positions [2, 3, 4, 5] gives the maximum number of coins: 0 + 0 + 3 + 3 = 6
    assert maximumCoins([[1, 2, 5], [4, 5, 3]], 4) == 6

def test_k_larger_than_all_segments():
    # coins = [[1, 2, 5], [4, 5, 3]], k = 10
    # All bags combined cover positions 1 to 5, so the result is 5 + 0 + 0 + 3 + 3 = 11, but we can't cover 10 bags.
    # The answer is still limited to the maximum coins we can find from existing bags
    assert maximumCoins([[1, 2, 5], [4, 5, 3]], 10) == 11

def test_varying_coin_values():
    # coins = [[1, 3, 2], [4, 6, 5]], k = 5
    # Selecting bags at positions [1, 2, 3, 4, 5] gives: 2 + 2 + 2 + 5 + 5 = 16
    assert maximumCoins([[1, 3, 2], [4, 6, 5]], 5) == 16

    # coins = [[1, 3, 3]], k = 3
