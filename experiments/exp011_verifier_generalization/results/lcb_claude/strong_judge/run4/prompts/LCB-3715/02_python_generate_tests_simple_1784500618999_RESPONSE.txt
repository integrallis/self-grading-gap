from typing import List
from solution import maximumCoins

def test_canonical_example_1():
    # coins = [[8, 10, 1],[1, 3, 2],[5, 6, 4]], k = 4
    # Selecting bags at positions [3, 4, 5, 6] gives the maximum number of coins:
    # 2 (from bag 3) + 0 (from bag 4) + 4 (from bag 5) + 4 (from bag 6) = 10
    assert maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) == 10

def test_canonical_example_2():
    # coins = [[1, 10, 3]], k = 2
    # Selecting bags at positions [1, 2] gives the maximum number of coins:
    # 3 (from bag 1) + 3 (from bag 2) = 6
    assert maximumCoins([[1, 10, 3]], 2) == 6

def test_k_equals_1():
    # coins = [[1, 5, 3]], k = 1
    # Selecting bag at position [1] gives the maximum number of coins: 3 (from bag 1)
    assert maximumCoins([[1, 5, 3]], 1) == 3

def test_k_greater_than_total_range():
    # coins = [[1, 3, 2], [5, 7, 4]], k = 10
    # There are not enough bags to collect, the maximum is 0
    assert maximumCoins([[1, 3, 2], [5, 7, 4]], 10) == 0

def test_window_in_gap_between_segments():
    # coins = [[1, 2, 5], [4, 5, 10]], k = 3
    # Selecting bags at positions [3, 4, 5] gives the maximum number of coins: 
    # 0 (from bag 3) + 10 (from bag 4) + 10 (from bag 5) = 10
    assert maximumCoins([[1, 2, 5], [4, 5, 10]], 3) == 10

def test_multiple_segments_optimizing_window():
    # coins = [[1, 2, 5], [4, 5, 10], [7, 8, 2]], k = 5
    # Optimal bags to select: [1, 2, 4, 5, 6] gives: 5 + 5 + 10 + 0 + 0 = 20
    assert maximumCoins([[1, 2, 5], [4, 5, 10], [7, 8, 2]], 5) == 20
