from typing import List
from solution import largestInteger

def test_canonical_example_1():
    # spec: largestInteger([3, 9, 2, 1, 7], 3) -> 7
    # 1 appears in 2 subarrays: [9, 2, 1], [2, 1, 7]
    # 2 appears in 3 subarrays: [3, 9, 2], [9, 2, 1], [2, 1, 7]
    # 3 appears in 1 subarray: [3, 9, 2]
    # 7 appears in 1 subarray: [2, 1, 7]
    # 9 appears in 2 subarrays: [3, 9, 2], [9, 2, 1]
    assert largestInteger([3, 9, 2, 1, 7], 3) == 7

def test_canonical_example_2():
    # spec: largestInteger([3, 9, 7, 2, 1, 7], 4) -> 3
    # 1 appears in 2 subarrays: [9, 7, 2, 1], [7, 2, 1, 7]
    # 2 appears in 3 subarrays: [3, 9, 7, 2], [9, 7, 2, 1], [7, 2, 1, 7]
    # 3 appears in 1 subarray: [3, 9, 7, 2]
    # 7 appears in 3 subarrays: [3, 9, 7, 2], [9, 7, 2, 1], [7, 2, 1, 7]
    # 9 appears in 2 subarrays: [3, 9, 7, 2], [9, 7, 2, 1]
    assert largestInteger([3, 9, 7, 2, 1, 7], 4) == 3

def test_canonical_example_3():
    # spec: largestInteger([0, 0], 1) -> -1
    # 0 appears in 2 subarrays: [0] (only one subarray of size 1)
    # No integer appears in exactly one subarray of size 1
    assert largestInteger([0, 0], 1) == -1

def test_k_equals_nums_length():
    # spec: largestInteger([1, 2, 3], 3) -> 3
    # Only one subarray: [1, 2, 3]
    # 1 appears in 1 subarray: [1, 2, 3]
    # 2 appears in 1 subarray: [1, 2, 3]
    # 3 appears in 1 subarray: [1, 2, 3]
    # Largest integer is 3
    assert largestInteger([1, 2, 3], 3) == 3

def test_single_element_array():
    # spec: largestInteger([5], 1) -> 5
    # Only one subarray: [5]
    # 5 appears in 1 subarray: [5]
    assert largestInteger([5], 1) == 5
