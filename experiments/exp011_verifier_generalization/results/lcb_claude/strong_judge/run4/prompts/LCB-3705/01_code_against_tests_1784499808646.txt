## Implementation

```python
from typing import List


def largestInteger(nums: List[int], k: int) -> int:
    count_map = {}
    n = len(nums)

    # Count occurrences of each number in all subarrays of size k
    for i in range(n - k + 1):
        subarray = nums[i:i + k]
        unique_in_subarray = set(subarray)
        for num in unique_in_subarray:
            if num in count_map:
                count_map[num] += 1
            else:
                count_map[num] = 1

    # Find the largest integer that appears in exactly one subarray
    largest_missing = -1
    for num, count in count_map.items():
        if count == 1 and num > largest_missing:
            largest_missing = num

    return largest_missing
```
