from solution import maxDifference

def test_canonical_example_1():
    # Input: "aaaaabbc" -> 'a' (5) is odd, 'b' (2) is even. Difference: 5 - 2 = 3
    assert maxDifference("aaaaabbc") == 3

def test_canonical_example_2():
    # Input: "abcabcab" -> 'a' (3) is odd, 'c' (2) is even. Difference: 3 - 2 = 1
    assert maxDifference("abcabcab") == 1

def test_minimum_string_length():
    # Input: "abb" -> 'a' (1) is odd, 'b' (2) is even. Difference: 1 - 2 = -1
    assert maxDifference("abb") == -1

def test_multiple_odd_and_even_frequencies():
    # Input: "aaabbbcccc" -> 'a' (3) is odd, 'b' (3) is odd, 'c' (4) is even. Difference: max(3, 3) - 4 = 3 - 4 = -1
    assert maxDifference("aaabbbcccc") == -1

def test_large_odd_and_small_even_frequency():
    # Input: "aaaaaaabbb" -> 'a' (7) is odd, 'b' (3) is odd. Difference: max(7, 3) - min(3) = 7 - 3 = 4
    assert maxDifference("aaaaaaabbb") == 4

def test_three_distinct_characters():
    # Input: "aaabbbbbcd" -> 'a' (3) is odd, 'b' (5) is odd, 'c' (1) is odd, 'd' (1) is odd. Difference: max(3, 5, 1, 1) - min(0) = not valid, but needs at least 1 even.
    assert maxDifference("aabbbc") == 1

def test_long_string():
    # Input: "a" * 50 + "b" * 50 -> 'a' (50) is even, 'b' (50) is even. No valid pairs, should not happen according to constraints. 
    # Therefore, we need a valid configuration, thus we use "a" * 51 + "b" * 49
    assert maxDifference("a" * 51 + "b" * 49) == 1  # max(51) - min(49) = 2
