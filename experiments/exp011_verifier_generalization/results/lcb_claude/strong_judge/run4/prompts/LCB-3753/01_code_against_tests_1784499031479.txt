## Implementation

```python
from collections import Counter

def maxDifference(s: str) -> int:
    frequency = Counter(s)
    odd_frequencies = [freq for freq in frequency.values() if freq % 2 == 1]
    even_frequencies = [freq for freq in frequency.values() if freq % 2 == 0]
    max_odd = max(odd_frequencies) if odd_frequencies else 0
    min_even = min(even_frequencies) if even_frequencies else 0
    return max_odd - min_even

```
