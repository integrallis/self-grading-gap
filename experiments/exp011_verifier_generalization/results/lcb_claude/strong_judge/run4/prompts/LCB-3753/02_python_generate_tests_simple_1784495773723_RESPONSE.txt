from solution import maxDifference

def test_canonical_example_1():
    # Example: input "aaaaabbc" 
    # 'a' has an odd frequency of 5, 'b' has an even frequency of 2.
    # Maximum difference is 5 - 2 = 3
    assert maxDifference("aaaaabbc") == 3

def test_canonical_example_2():
    # Example: input "abcabcab" 
    # 'a' has an odd frequency of 3, 'c' has an even frequency of 2.
    # Maximum difference is 3 - 2 = 1
    assert maxDifference("abcabcab") == 1

def test_minimum_length():
    # Input: "aab"
    # 'a' has an even frequency of 2, 'b' has an odd frequency of 1.
    # Maximum difference is 1 - 2 = -1
    assert maxDifference("aab") == -1

def test_multiple_odd_even_characters():
    # Input: "aaabbbccccddddde"
    # 'a' has an odd frequency of 3, 'b' has an odd frequency of 3,
    # 'c' has an even frequency of 4, 'd' has an odd frequency of 5,
    # 'e' has an odd frequency of 1.
    # Maximum difference is 5 (from 'd') - 4 (from 'c') = 1
    assert maxDifference("aaabbbccccddddde") == 1

def test_minimum_even_with_max_odd():
    # Input: "aaaaaaabbcde"
    # 'a' has an odd frequency of 7, 'b' has an even frequency of 2,
    # 'c', 'd', 'e' have an odd frequency of 1 each (not used).
    # Maximum difference is 7 - 2 = 5
    assert maxDifference("aaaaaaabbcde") == 5
