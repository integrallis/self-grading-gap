from typing import List
from solution import minCosts

def test_canonical_example_1():
    # Input: [5, 3, 4, 1, 3, 2]
    # Explanation:
    # To position 0: swap with person 0 for cost 5
    # To position 1: swap with person 1 for cost 3
    # To position 2: swap with person 1 for cost 3, then swap with person 2 for free
    # To position 3: swap with person 3 for cost 1
    # To position 4: swap with person 3 for cost 1, then swap with person 4 for free
    # To position 5: swap with person 3 for cost 1, then swap with person 5 for free
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]

def test_canonical_example_2():
    # Input: [1, 2, 4, 6, 7]
    # Explanation:
    # To position 0: swap with person 0 for cost 1
    # To position 1: swap with person 0 for cost 1, then swap with person 1 for free
    # To position 2: swap with person 0 for cost 1, then swap with person 2 for free
    # To position 3: swap with person 0 for cost 1, then swap with person 3 for free
    # To position 4: swap with person 0 for cost 1, then swap with person 4 for free
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]

def test_single_element():
    # Input: [5]
    # Explanation:
    # To position 0: swap with person 0 for cost 5
    assert minCosts([5]) == [5]

def test_descending_costs():
    # Input: [5, 4, 3, 2, 1]
    # Explanation:
    # Each position is cheapest by direct swap with the respective person.
    assert minCosts([5, 4, 3, 2, 1]) == [5, 4, 3, 2, 1]

def test_uniform_costs():
    # Input: [3, 3, 3]
    # Explanation:
    # Each position can be reached by swapping with person 0 at cost 3, then all swaps are free.
    assert minCosts([3, 3, 3]) == [3, 3, 3]
