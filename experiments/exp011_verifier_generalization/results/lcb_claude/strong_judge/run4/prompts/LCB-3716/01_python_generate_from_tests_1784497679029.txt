from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = [[1] * (n + 1) for _ in range(n)]
    max_length = 1

    for i in range(n):
        for j in range(i):
            diff = abs(nums[i] - nums[j])
            # Update dp[i][k] if we can extend the subsequence
            for k in range(1, n + 1):
                if dp[j][k] > 0 and (k == 1 or diff <= abs(nums[dp[j][k] - 1] - nums[dp[j][k - 1] - 1])):
                    dp[i][k] = max(dp[i][k], dp[j][k] + 1)
                    max_length = max(max_length, dp[i][k])

    return max_length
```
