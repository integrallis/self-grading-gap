from typing import List
from collections import defaultdict, deque


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Create a reversed graph as an adjacency list
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[v].append((u, w))  # reverse the direction for backward reachability to node 0

    # Define a binary search over the possible maximum weights
    left, right = 1, 10**6
    result = -1

    while left <= right:
        mid = (left + right) // 2

        # Check if we can satisfy the conditions with max weight `mid`
        if canReachWithMaxWeight(graph, n, threshold, mid):
            result = mid  # we can satisfy the conditions, try for a smaller max weight
            right = mid - 1
        else:
            left = mid + 1

    return result


def canReachWithMaxWeight(graph: List[List[tuple]], n: int, threshold: int, maxWeight: int) -> bool:
    out_edges_count = [0] * n
    visited_count = [0] * n
    queue = deque([0])  # Start BFS from node 0
    reachable = [False] * n
    reachable[0] = True

    # BFS to check reachability
    while queue:
        node = queue.popleft()
        for neighbor, weight in graph[node]:
            # Check edge weight and outgoing edge count
            if weight <= maxWeight:
                if not reachable[neighbor] and visited_count[neighbor] < threshold:
                    reachable[neighbor] = True
                    visited_count[neighbor] += 1
                    queue.append(neighbor)

    return all(reachable)
```
