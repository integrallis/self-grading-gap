from typing import List
import heapq


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Create a graph as an adjacency list
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[u].append((v, w))  # original direction for reachability to node 0

    # Define a binary search over the possible maximum weights
    left, right = 1, 10**6
    result = -1

    while left <= right:
        mid = (left + right) // 2

        # Check if we can satisfy the conditions with max weight `mid`
        if canReachWithMaxWeight(graph, n, threshold, mid):
            result = mid  # we can satisfy the conditions, try for a smaller max weight
            right = mid - 1
        else:
            left = mid + 1

    return result


def canReachWithMaxWeight(graph: List[List[tuple]], n: int, threshold: int, maxWeight: int) -> bool:
    out_edges_count = [0] * n
    queue = []
    reachable = [False] * n
    reachable[0] = True

    # Count outgoing edges and initialize queue
    for u in range(n):
        for v, w in graph[u]:
            if w <= maxWeight:
                out_edges_count[u] += 1
                if out_edges_count[u] <= threshold:
                    queue.append(u)

    # BFS/DFS to check reachability
    while queue:
        node = queue.pop()
        for neighbor, weight in graph[node]:
            if weight <= maxWeight and not reachable[neighbor]:
                reachable[neighbor] = True
                # Count outgoing edges for the neighbor
                out_edges_count[neighbor] += 1
                if out_edges_count[neighbor] <= threshold:
                    queue.append(neighbor)

    return all(reachable)
```
