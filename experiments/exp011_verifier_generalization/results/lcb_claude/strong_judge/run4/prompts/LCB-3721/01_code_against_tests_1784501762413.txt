```python
from typing import List

def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers
    online_status = [True] * numberOfUsers
    offline_until = [0] * numberOfUsers

    for event in events:
        event_type, timestamp, *data = event
        timestamp = int(timestamp)

        # Update online status based on OFFLINE events
        if event_type == "OFFLINE":
            user_id = int(data[0][2:])  # extract id number from "id<number>"
            online_status[user_id] = False
            offline_until[user_id] = timestamp + 60

        # Handle MESSAGE events
        elif event_type == "MESSAGE":
            # Process any users going back online at this timestamp
            for i in range(numberOfUsers):
                if not online_status[i] and timestamp >= offline_until[i]:
                    online_status[i] = True

            mentions_string = data[0]
            if mentions_string == "ALL":
                for i in range(numberOfUsers):
                    mentions[i] += 1
            elif mentions_string == "HERE":
                for i in range(numberOfUsers):
                    if online_status[i]:
                        mentions[i] += 1
            else:
                ids = mentions_string.split()
                for uid in ids:
                    user_id = int(uid[2:])  # extract id number from "id<number>"
                    mentions[user_id] += 1

    return mentions
```
