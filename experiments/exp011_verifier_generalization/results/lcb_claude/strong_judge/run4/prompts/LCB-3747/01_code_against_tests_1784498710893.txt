## Implementation

```python
from typing import List

def maxAdjacentDistance(nums: List[int]) -> int:
    max_distance = 0
    # Calculate the distance between adjacent elements, including circular pair
    for i in range(len(nums)):
        # Circular index with modulo
        distance = abs(nums[i] - nums[(i + 1) % len(nums)])
        max_distance = max(max_distance, distance)
    return max_distance

```
