from typing import List

def subarraySum(nums: List[int]) -> int:
    total_sum = 0
    for i in range(len(nums)):
        start = max(0, i - nums[i])  # Determine where the subarray starts
        subarray_sum = sum(nums[start:i + 1])  # Calculate the sum of the subarray
        total_sum += subarray_sum  # Accumulate the sum
    return total_sum
```
