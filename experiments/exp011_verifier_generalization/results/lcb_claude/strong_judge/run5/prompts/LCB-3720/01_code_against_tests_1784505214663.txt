from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    from collections import defaultdict, deque

    # Function to check if we can reach node 0 and satisfy the threshold
    def canReachWithMaxWeight(max_weight):
        # Keep track of incoming edges count
        incoming_count = [0] * n
        filtered_graph = defaultdict(list)

        # Create a filtered graph with edges below max_weight
        for u, v, w in edges:
            if w <= max_weight:
                filtered_graph[u].append(v)
                incoming_count[v] += 1

        # Check if any node exceeds the threshold
        if any(count > threshold for count in incoming_count):
            return False

        # BFS to check if all nodes can reach node 0
        visited = [False] * n
        queue = deque([0])
        visited[0] = True

        while queue:
            node = queue.popleft()
            for neighbor in filtered_graph[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    queue.append(neighbor)

        return all(visited[i] for i in range(n))

    # Binary search on the maximum weight
    left, right = 1, 10**6
    answer = -1

    while left <= right:
        mid = (left + right) // 2
        if canReachWithMaxWeight(mid):
            answer = mid
            right = mid - 1
        else:
            left = mid + 1

    return answer
