from typing import List
from solution import minMaxWeight

def test_canonical_example_1():
    # Input: n = 5, edges = [[1,0,1],[2,0,2],[3,0,1],[4,3,1],[2,1,1]], threshold = 2
    # Output: 1
    assert minMaxWeight(5, [[1,0,1],[2,0,2],[3,0,1],[4,3,1],[2,1,1]], 2) == 1

def test_canonical_example_2():
    # Input: n = 5, edges = [[0,1,1],[0,2,2],[0,3,1],[0,4,1],[1,2,1],[1,4,1]], threshold = 1
    # Output: -1
    assert minMaxWeight(5, [[0,1,1],[0,2,2],[0,3,1],[0,4,1],[1,2,1],[1,4,1]], 1) == -1

def test_canonical_example_3():

def test_canonical_example_4():
    # Input: n = 5, edges = [[1,2,1],[1,3,3],[1,4,5],[2,3,2],[4,0,1]], threshold = 1
    # Output: -1
    assert minMaxWeight(5, [[1,2,1],[1,3,3],[1,4,5],[2,3,2],[4,0,1]], 1) == -1

def test_minimum_graph_size():
    # Input: n = 2, edges = [[1, 0, 1]], threshold = 1
    # Output: 1 (single edge with weight 1, node 0 reachable from node 1)
    assert minMaxWeight(2, [[1, 0, 1]], 1) == 1

def test_no_edge_removal_needed():
    # Input: n = 4, edges = [[1, 0, 1], [2, 0, 2], [3, 0, 1]], threshold = 3
    # Output: 2 (all nodes can reach 0 and max weight is 2)
    assert minMaxWeight(4, [[1, 0, 1], [2, 0, 2], [3, 0, 1]], 3) == 2

def test_maximum_threshold():
    # Input: n = 5, edges = [[1, 0, 1], [2, 0, 2], [3, 0, 3], [4, 0, 4]], threshold = 4
    # Output: 4 (all nodes can reach 0, max weight is 4)
    assert minMaxWeight(5, [[1, 0, 1], [2, 0, 2], [3, 0, 3], [4, 0, 4]], 4) == 4

def test_multiple_edges_same_nodes():
    # Input: n = 3, edges = [[1, 0, 1], [2, 0, 2], [1, 0, 3]], threshold = 2
    # Output: 2 (both edges from 1 to 0 can be removed, max weight becomes 2)
    assert minMaxWeight(3, [[1, 0, 1], [2, 0, 2], [1, 0, 3]], 2) == 2
