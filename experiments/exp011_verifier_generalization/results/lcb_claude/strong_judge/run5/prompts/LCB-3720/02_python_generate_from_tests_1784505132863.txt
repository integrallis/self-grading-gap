from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    from collections import defaultdict, deque
    import heapq

    # Create an adjacency list for the graph
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))

    # Function to check if we can reach node 0 and satisfy the threshold
    def canReachWithMaxWeight(max_weight):
        # Keep track of outgoing edges count
        outgoing_count = [0] * n
        # Create a filtered graph with edges below max_weight
        filtered_graph = defaultdict(list)

        for u in range(n):
            for v, w in graph[u]:
                if w <= max_weight:
                    filtered_graph[u].append(v)
                    outgoing_count[u] += 1

        # Check if any node exceeds the threshold
        if any(count > threshold for count in outgoing_count):
            return False

        # BFS or DFS to check reachability to node 0
        visited = [False] * n
        queue = deque([0])
        visited[0] = True

        while queue:
            node = queue.popleft()
            for neighbor in filtered_graph[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    queue.append(neighbor)

        return all(visited[i] for i in range(1, n))

    # Binary search on the weight
    left, right = 1, 10**6
    answer = -1

    while left <= right:
        mid = (left + right) // 2
        if canReachWithMaxWeight(mid):
            answer = mid
            right = mid - 1
        else:
            left = mid + 1

    return answer
```
