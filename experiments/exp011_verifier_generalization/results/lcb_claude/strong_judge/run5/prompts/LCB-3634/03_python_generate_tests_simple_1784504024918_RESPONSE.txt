from solution import calculateScore

def test_canonical_example_1():
    # spec: calculateScore("aczzx") -> 5
    # i=0: no j, skip
    # i=1: no j, skip
    # i=2: j=0 (mirror: a-z), score += 2 - 0 = 2
    # i=3: no j, skip
    # i=4: j=1 (mirror: c-x), score += 4 - 1 = 3
    # Total score = 2 + 3 = 5
    assert calculateScore("aczzx") == 5

def test_canonical_example_2():
    # spec: calculateScore("abcdef") -> 0
    # i=0: no j, skip
    # i=1: no j, skip
    # i=2: no j, skip
    # i=3: no j, skip
    # i=4: no j, skip
    # i=5: no j, skip
    # Total score = 0
    assert calculateScore("abcdef") == 0

def test_single_character():
    # spec: calculateScore("a") -> 0
    # i=0: no j < 0, skip
    # Total score = 0
    assert calculateScore("a") == 0

def test_fully_pairable_string():
    # spec: calculateScore("az") -> 1
    # i=0: no j, skip
    # i=1: j=0 (mirror: a-z), score += 1 - 0 = 1
    # Total score = 1
    assert calculateScore("az") == 1

def test_multiple_unmarked_mirrors():
    # spec: calculateScore("aaz") -> 1
    # i=0: no j, skip
    # i=1: j=0 (mirror: a-z), score += 1 - 0 = 1
    # i=2: no j (j=1 is marked), skip
    # Total score = 1
    assert calculateScore("aaz") == 1

def test_all_mirrors():
    # spec: calculateScore("azby") -> 2
    # i=0: no j, skip
    # i=1: j=0 (mirror: a-z), score += 1 - 0 = 1
    # i=2: no j, skip
    # i=3: j=2 (mirror: b-y), score += 3 - 2 = 1
    # Total score = 1 + 1 = 2
    assert calculateScore("azby") == 2
