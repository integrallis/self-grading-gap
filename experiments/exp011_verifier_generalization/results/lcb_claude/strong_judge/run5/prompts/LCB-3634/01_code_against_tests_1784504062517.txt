```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # a-z to z-a

    for i in range(len(s)):
        if marked[i]:  # Skip if already marked
            continue
        for j in range(i - 1, -1, -1):  # Look for unmarked j < i
            if not marked[j] and mirror[s[i]] == s[j]:
                marked[i] = marked[j] = True
                score += i - j
                break
    return score
```
