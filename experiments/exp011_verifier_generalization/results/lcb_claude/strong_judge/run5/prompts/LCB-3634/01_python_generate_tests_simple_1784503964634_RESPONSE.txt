from solution import calculateScore

def test_canonical_example_1():
    # spec: calculateScore("aczzx") -> 5
    # i=0: no j, skip
    # i=1: no j, skip
    # i=2: j=0 (mirror: a-z), score += 2 - 0 = 2
    # i=3: no j, skip
    # i=4: j=1 (mirror: b-y), score += 4 - 1 = 3
    # Total score = 2 + 3 = 5
    assert calculateScore("aczzx") == 5

def test_canonical_example_2():
    # spec: calculateScore("abcdef") -> 0
    # i=0: no j, skip
    # i=1: no j, skip
    # i=2: no j, skip
    # i=3: no j, skip
    # i=4: no j, skip
    # i=5: no j, skip
    # Total score = 0
    assert calculateScore("abcdef") == 0
