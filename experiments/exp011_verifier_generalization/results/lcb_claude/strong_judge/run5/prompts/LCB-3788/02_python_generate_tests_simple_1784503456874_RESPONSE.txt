def test_canonical_example_1():
    # spec: maxSum([1, 2, 3, 4, 5]) -> 15

def test_canonical_example_2():
    # spec: maxSum([1, 1, 0, 1, 1]) -> 1 
    # Explanation: Delete the elements 1's and select the unique element [1] to obtain the maximum sum.
    assert maxSum([1, 1, 0, 1, 1]) == 1

def test_canonical_example_3():
    # spec: maxSum([1, 2, -1, -2, 1, 0, -1]) -> 3 
    # Explanation: Delete -1 and -2, and select the subarray [2, 1] to obtain the maximum sum.
    assert maxSum([1, 2, -1, -2, 1, 0, -1]) == 3

def test_single_element_array():
    # spec: maxSum([42]) -> 42

    # spec: maxSum([-5, -3, -8]) -> -3 
    # Explanation: The maximum sum is the least negative number, which is -3.
    assert maxSum([-5, -3, -8]) == -3

    # spec: maxSum([5, 5, 3]) -> 5
