from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[[float('-inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0]

    for i in range(m):
        for j in range(n):
            for k in range(3):
                if i == 0 and j == 0:
                    continue
                if i > 0:
                    if coins[i][j] >= 0:
                        dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k] + coins[i][j])
                    else:
                        # Take the penalty choice
                        dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k] + coins[i][j])  
                        if k < 2:
                            dp[i][j][k + 1] = max(dp[i][j][k + 1], dp[i-1][j][k])  # Neutralize
                if j > 0:
                    if coins[i][j] >= 0:
                        dp[i][j][k] = max(dp[i][j][k], dp[i][j-1][k] + coins[i][j])
                    else:
                        # Take the penalty choice
                        dp[i][j][k] = max(dp[i][j][k], dp[i][j-1][k] + coins[i][j])  
                        if k < 2:
                            dp[i][j][k + 1] = max(dp[i][j][k + 1], dp[i][j-1][k])  # Neutralize

    max_profit = float('-inf')
    for k in range(3):  # Check all neutralization states on the last cell
        max_profit = max(max_profit, dp[m-1][n-1][k])
    return max_profit

```
