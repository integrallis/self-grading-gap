```python
def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    for direction in s:
        if direction == 'N':
            y += 1
        elif direction == 'S':
            y -= 1
        elif direction == 'E':
            x += 1
        elif direction == 'W':
            x -= 1
        max_distance = max(max_distance, abs(x) + abs(y))

    total_moves = len(s)
    distance_with_changes = max_distance + min(k, total_moves)
    return distance_with_changes

```
