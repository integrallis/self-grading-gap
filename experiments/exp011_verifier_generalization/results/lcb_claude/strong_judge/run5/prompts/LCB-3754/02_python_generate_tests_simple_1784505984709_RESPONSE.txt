from solution import maxDistance

def test_canonical_example_1():
    # Example: input "NWSE", k = 1
    # Explanation:
    # Move sequence: N -> W -> (change S to N) -> E
    # Positions: (0,1) -> (-1,1) -> (-1,2) -> (0,2)
    # Distances: 1, 2, 3, 2
    # Maximum distance = 3
    assert maxDistance("NWSE", 1) == 3

def test_canonical_example_2():
    # Example: input "NSWWEW", k = 3
    # Explanation:
    # Change S to N, E to W: "NNWWWW"
    # Positions: (0,1) -> (0,2) -> (-1,2) -> (-2,2) -> (-3,2) -> (-4,2)
    # Distances: 1, 2, 3, 4, 5, 6
    # Maximum distance = 6
    assert maxDistance("NSWWEW", 3) == 6

def test_k_zero():
    # No changes allowed
    # Path: N -> S -> E -> W results in positions (0,1) -> (0,0) -> (1,0) -> (0,0)
    # Maximum distance = 1
    assert maxDistance("NSEW", 0) == 1

def test_k_equals_length():
    # Can change all characters
    # All 'S' to 'N' results in moving (0,0) -> (0,1) -> (0,2) -> (0,3) -> (0,4)
    # Maximum distance = 4
    assert maxDistance("SSSS", 4) == 4

def test_minimum_length():
    # Only one move, maximum distance is 1
    assert maxDistance("N", 0) == 1

def test_all_moves_same_direction():
    # All moves north
    # Maximum distance is simply the length of the string
    assert maxDistance("NNNN", 0) == 4

def test_opposing_moves():
    # Two steps north, two steps south
    # Path: N -> S -> N -> S results in positions (0,1) -> (0,0) -> (0,1) -> (0,0)
    # Change S to N results in (0,2)
    # Maximum distance = 4
    assert maxDistance("NSNS", 2) == 4

def test_already_optimal():
    # The path is already optimal, no changes needed
    # Path: N -> N -> E -> E results in positions (0,1) -> (0,2) -> (1,2) -> (2,2)
    # Maximum distance = 4
    assert maxDistance("NNEE", 0) == 4

def test_large_string():
    # A long string to test scalability
    # Path made of all movements north and east
    assert maxDistance("N" * 50000 + "E" * 50000, 0) == 100000
