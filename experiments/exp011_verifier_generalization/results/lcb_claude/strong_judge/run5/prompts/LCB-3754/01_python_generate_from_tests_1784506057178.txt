```python
def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    for direction in s:
        if direction == 'N':
            y += 1
        elif direction == 'S':
            y -= 1
        elif direction == 'E':
            x += 1
        elif direction == 'W':
            x -= 1
        max_distance = max(max_distance, abs(x) + abs(y))

    num_changes = 0
    # Count opposing moves
    for direction in s:
        if direction == 'N' or direction == 'S':
            num_changes += 1
        elif direction == 'E' or direction == 'W':
            num_changes += 1

    # Maximum distance will be the max distance achieved + the changes we can make
    max_possible_distance = max_distance + min(num_changes, k)
    return max_possible_distance
```
