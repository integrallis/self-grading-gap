from solution import maxDistance

def test_canonical_example_1():
    # Example: input "NWSE", k = 1
    # Explanation:
    # Move sequence: N -> W -> (change S to N) -> E
    # Positions: (0,1) -> (-1,1) -> (-1,2) -> (0,2)
    # Distances: 1, 2, 3, 2
    # Maximum distance = 3
    assert maxDistance("NWSE", 1) == 3

def test_canonical_example_2():
    # Example: input "NSWWEW", k = 3
    # Explanation:
    # Change S to N, E to W: "NNWWWW"
    # Positions: (0,1) -> (0,0) -> (-1,0) -> (-2,0) -> (-3,0) -> (-3,1)
    # Distances: 1, 0, 1, 2, 3, 4
    # Maximum distance = 6
    assert maxDistance("NSWWEW", 3) == 6
