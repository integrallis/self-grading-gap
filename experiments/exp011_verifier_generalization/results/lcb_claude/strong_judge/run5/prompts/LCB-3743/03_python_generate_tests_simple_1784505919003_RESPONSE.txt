def test_canonical_example_1():
    # Example: input '5\n1\n[1, 3]\n[2, 5]' -> output '2'
    # Reschedule the meeting at [1, 2] to [2, 3], leaving no meetings during the time [0, 2].
    assert maxFreeTime(5, 1, [1, 3], [2, 5]) == 2

def test_canonical_example_2():
    # Example: input '10\n1\n[0, 2, 9]\n[1, 4, 10]' -> output '6'

    assert maxFreeTime(10, 1, [0, 2, 9], [1, 4, 10]) == 6
