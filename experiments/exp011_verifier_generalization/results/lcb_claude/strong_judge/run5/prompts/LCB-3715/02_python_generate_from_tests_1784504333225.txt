from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create an array to store the number of coins at each position
    max_position = 0
    for l, r, c in coins:
        max_position = max(max_position, r)

    # Initialize a list to hold the number of coins from position 1 to max_position
    coin_count = [0] * (max_position + 1)
    for l, r, c in coins:
        for i in range(l, r + 1):
            coin_count[i] += c

    # Use a sliding window to find the maximum coins in any k consecutive bags
    max_coins = 0
    current_coins = sum(coin_count[1:min(k + 1, max_position + 1)])
    max_coins = current_coins

    for i in range(1, max_position - k + 2):
        # Slide the window right by removing the leftmost element and adding the new rightmost element
        if i + k - 1 <= max_position:
            current_coins = current_coins - coin_count[i] + coin_count[i + k - 1]
            max_coins = max(max_coins, current_coins)

    return max_coins
```
