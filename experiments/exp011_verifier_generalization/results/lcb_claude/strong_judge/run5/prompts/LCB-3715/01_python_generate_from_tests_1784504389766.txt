from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    max_coins = 0
    candidates = set()

    # Collect candidate starting points
    for l, r, c in coins:
        candidates.add(1)  # Always consider starting from position 1
        candidates.add(max(1, l))  # Start at l
        candidates.add(max(1, l - k + 1))  # Start before l
        candidates.add(max(1, r - k + 1))  # Start before r
        candidates.add(max(1, r))  # Start at r

    # Evaluate each candidate starting position
    for start in candidates:
        current_coins = 0
        end = start + k - 1

        for l, r, c in coins:
            if r < start:
                continue
            if l > end:
                break
            # Calculate overlap
            overlap_start = max(l, start)
            overlap_end = min(r, end)
            current_coins += c * (overlap_end - overlap_start + 1)

        max_coins = max(max_coins, current_coins)

    return max_coins
```
