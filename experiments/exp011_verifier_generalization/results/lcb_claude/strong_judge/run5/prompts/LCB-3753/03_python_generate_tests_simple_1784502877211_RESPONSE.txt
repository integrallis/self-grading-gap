from solution import maxDifference

def test_canonical_example_1():
    # spec: maxDifference("aaaaabbc") -> 3
    # 'a' has frequency 5 (odd), 'b' has frequency 2 (even). 
    # Difference = 5 - 2 = 3
    assert maxDifference("aaaaabbc") == 3

def test_canonical_example_2():
    # spec: maxDifference("abcabcab") -> 1
    # 'a' has frequency 3 (odd), 'c' has frequency 2 (even). 
    # Difference = 3 - 2 = 1
    assert maxDifference("abcabcab") == 1

def test_minimum_length_boundary():
    # Test with minimum length (3 characters): "aab"
    # 'a' has frequency 2 (even), 'b' has frequency 1 (odd).
    # Difference = 1 - 2 = -1
    assert maxDifference("aab") == -1

def test_maximum_length_boundary():
    # Test with maximum length (100 characters): "a" * 50 + "b" * 49 + "c"
    # 'a' has frequency 50 (even), 'b' has frequency 49 (odd), 'c' has frequency 1 (odd).
    # Difference = max(49, 1) - min(50) = 49 - 50 = -1
    assert maxDifference("a" * 50 + "b" * 49 + "c") == -1

def test_multiple_even_frequencies():
    # Test with multiple even-frequency characters: "aabbbbccccd"
    # 'a' has frequency 2 (even), 'b' has frequency 4 (even), 'c' has frequency 4 (even)
    # Let's use 'd' with frequency 1 (odd).
    # Difference = max(1) - min(2, 4, 4) = 1 - 2 = -1
    assert maxDifference("aabbbbccccd") == -1

def test_large_odd_minus_small_even():
    # Test case where the smallest odd is larger than the largest even: "aabbccddeee"
    # 'a' has frequency 2 (even), 'b' has frequency 2 (even), 'c' has frequency 2 (even), 'd' has frequency 2 (even), 'e' has frequency 3 (odd)
    # Difference = 3 - 2 = 1
    assert maxDifference("aabbccddeee") == 1
