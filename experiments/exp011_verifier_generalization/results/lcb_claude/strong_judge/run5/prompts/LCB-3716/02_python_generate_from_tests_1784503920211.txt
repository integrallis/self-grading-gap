from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n < 2:
        return 0  # not expected to reach here as per constraints

    # dp[i] will hold the length of the longest subsequence ending at index i
    dp = [1] * n
    max_length = 1

    for i in range(n):
        for j in range(i):
            diff1 = abs(nums[i] - nums[j])
            valid = True
            current_length = 1
            last_diff = float('inf')

            # Build the sequence backwards from i to j
            for k in range(j, -1, -1):
                diff2 = abs(nums[k] - nums[k - 1]) if k > 0 else float('inf')
                if diff1 > last_diff:
                    break
                if diff1 >= diff2:
                    current_length += 1
                    last_diff = diff2
                else:
                    break

            dp[i] = max(dp[i], current_length)
            max_length = max(max_length, dp[i])

    return max_length
```
