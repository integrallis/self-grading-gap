from typing import List
from solution import minMaxSums

def test_canonical_example_1():
    # spec: nums = [1, 2, 3], k = 2
    # subsequences: [1], [2], [3], [1, 2], [1, 3], [2, 3]
    # min + max for all: 2 + 4 + 6 + 3 + 4 + 5 = 24
    assert minMaxSums([1, 2, 3], 2) == 24

def test_canonical_example_2():
    # spec: nums = [5, 0, 6], k = 1
    # subsequences: [5], [0], [6]
    # min + max for all: 5 + 5 + 0 + 0 + 6 + 6 = 22
    assert minMaxSums([5, 0, 6], 1) == 22

def test_canonical_example_3():
    # spec: nums = [1, 1, 1], k = 2
    # subsequences: [1], [1], [1], [1, 1], [1, 1], [1, 1]
    # min + max for all: 2 * 3 + 6 = 12
    assert minMaxSums([1, 1, 1], 2) == 12

def test_k_equals_array_length():
    # spec: nums = [1, 2], k = 2
    # subsequences: [1], [2], [1, 2]
    # min + max for all: 1 + 1 + 2 + 2 + 3 = 9
    assert minMaxSums([1, 2], 2) == 9

def test_single_element_array():
    # spec: nums = [5], k = 1
    # subsequences: [5]
    # min + max for all: 5 + 5 = 10
    assert minMaxSums([5], 1) == 10

def test_large_output():
    # spec: nums = [10**9] * 70, k = 70
    # all elements are the same, min = max = 10**9
    # number of subsequences = 2**70 - 1 (all non-empty subsets)
    # total = (min + max) * number of non-empty subsequences
    # total = (10**9 + 10**9) * (2**70 - 1)
    # large output, needs to be taken modulo 10**9 + 7
    expected_output = (2 * (10**9) * (2**70 - 1)) % (10**9 + 7)
    assert minMaxSums([10**9] * 70, 70) == expected_output
