```python
from typing import List

def minMaxSums(nums: List[int], k: int) -> int:
    MOD = 10**9 + 7
    n = len(nums)
    total = 0
    
    # Iterate through all elements for subsequences with length 1 to k
    for i in range(1, k + 1):
        # Calculate how many times each number appears as min and max
        # For each number in sorted order, it appears:
        total += sum(nums) * pow(2, n - i, MOD) % MOD
        total += sum(nums) * pow(2, n - i, MOD) % MOD
    
    return total * 2 % MOD
```
