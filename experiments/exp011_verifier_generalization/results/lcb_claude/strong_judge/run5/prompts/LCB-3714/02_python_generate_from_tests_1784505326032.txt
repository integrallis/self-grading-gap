from typing import List


def minMaxSums(nums: List[int], k: int) -> int:
    MOD = 10**9 + 7
    n = len(nums)
    nums.sort()
    total = 0

    # Iterate through the first k smallest elements (for min)
    for i in range(min(k, n)):
        total += nums[i] * pow(2, n - 1 - i, MOD) % MOD
        total %= MOD

    # Iterate through the last k largest elements (for max)
    for i in range(1, min(k, n) + 1):
        total += nums[-i] * pow(2, n - 1 - (n - i), MOD) % MOD
        total %= MOD

    return total * 2 % MOD

```
