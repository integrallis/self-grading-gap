from typing import List


def minMaxSums(nums: List[int], k: int) -> int:
    MOD = 10**9 + 7
    n = len(nums)
    nums.sort()
    total = 0

    # Calculate contribution from the minimum elements
    for i in range(min(k, n)):
        total += nums[i] * pow(2, n - 1 - i, MOD) % MOD
        total %= MOD

    # Calculate contribution from the maximum elements
    for i in range(1, min(k, n) + 1):
        total += nums[-i] * pow(2, n - 1 - (n - i), MOD) % MOD
        total %= MOD

    return total % MOD
```
