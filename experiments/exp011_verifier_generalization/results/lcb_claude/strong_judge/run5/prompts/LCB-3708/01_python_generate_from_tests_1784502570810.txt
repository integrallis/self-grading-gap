from typing import List

def zigzagTraversal(grid: List[List[int]]) -> List[int]:
    result = []
    for i in range(len(grid)):
        if i % 2 == 0:
            # Traverse right
            for j in range(0, len(grid[i]), 2):
                result.append(grid[i][j])
        else:
            # Traverse left
            for j in range(len(grid[i])-1, -1, -2):
                result.append(grid[i][j])
    return result

```
