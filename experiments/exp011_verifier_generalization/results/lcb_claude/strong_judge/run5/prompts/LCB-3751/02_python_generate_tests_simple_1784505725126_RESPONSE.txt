from typing import List
from solution import maxFrequency

def test_canonical_example_1():
    # Example: input [1, 2, 3, 4, 5, 6], k = 1
    # After adding -5 to nums[2..5], we get [1, 2, -2, -1, 0, 1]
    # The value 1 appears 2 times.
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2

def test_canonical_example_2():
    # Example: input [10, 2, 3, 4, 5, 5, 4, 3, 2, 2], k = 10
    # After adding 8 to nums[1..9], we get [10, 10, 11, 12, 13, 13, 12, 11, 10, 10]
    # The value 10 appears 4 times.
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4

def test_single_element_array():
    # Input: [5], k = 5
    # The value 5 already appears 1 time.
    assert maxFrequency([5], 5) == 1

def test_single_element_needs_transformation():
    # Input: [10], k = 5
    # After adding -5 to nums[0..0], we get [5]
    # The value 5 appears 1 time.
    assert maxFrequency([10], 5) == 1

def test_k_at_upper_boundary():
    # Input: [1, 2, 3], k = 50
    # After adding 47 to the entire array, we get [48, 49, 50]
    # The value 50 appears 1 time.
    assert maxFrequency([1, 2, 3], 50) == 1

def test_all_elements_already_k():
    # Input: [5, 5, 5], k = 5
    # The value 5 already appears 3 times.
    assert maxFrequency([5, 5, 5], 5) == 3
