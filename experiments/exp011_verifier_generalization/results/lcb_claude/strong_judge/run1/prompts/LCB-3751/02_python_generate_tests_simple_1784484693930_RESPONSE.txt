from typing import List
from solution import maxFrequency

def test_canonical_example_1():
    # spec: maxFrequency([1, 2, 3, 4, 5, 6], 1) -> 2
    # Explanation: After adding -5 to nums[2..5], 1 has a frequency of 2 in [1, 2, -2, -1, 0, 1].
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2

def test_canonical_example_2():
    # spec: maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) -> 4
    # Explanation: After adding 8 to nums[1..9], 10 has a frequency of 4 in [10, 10, 11, 12, 13, 13, 12, 11, 10, 10].
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4

def test_min_array_length():
    # spec: maxFrequency([5], 5) -> 1
    # Explanation: The array has only one element which is equal to k.
    assert maxFrequency([5], 5) == 1

def test_k_already_in_array():
    # spec: maxFrequency([1, 1, 2, 3], 1) -> 3
    # Explanation: The optimal operation is to add 0 to index 0..1, resulting in [1, 1, 2, 3] where 1 appears 2 times.
    assert maxFrequency([1, 1, 2, 3], 1) == 3

def test_k_is_minimum():
    # spec: maxFrequency([1, 2, 3, 4, 5], 1) -> 1
    # Explanation: After adding -1 to nums[0..0], we get [0, 2, 3, 4, 5], where the value 1 appears once.
    assert maxFrequency([1, 2, 3, 4, 5], 1) == 1

def test_k_is_maximum():
    # spec: maxFrequency([1, 2, 3, 4, 5], 50) -> 1
    # Explanation: After adding 45 to nums[0..4], we get [46, 47, 48, 49, 50], where the value 50 appears once.
    assert maxFrequency([1, 2, 3, 4, 5], 50) == 1
