def test_canonical_example_1():
    # spec: input [5, 3, 4, 1, 3, 2]
    # i = 0: swap with person 0 for cost 5
    # i = 1: swap with person 1 for cost 3
    # i = 2: swap with person 1 for cost 3, then swap with person 2 for free
    # i = 3: swap with person 3 for cost 1
    # i = 4: swap with person 3 for cost 1, then swap with person 4 for free
    # i = 5: swap with person 3 for cost 1, then swap with person 5 for free
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]
