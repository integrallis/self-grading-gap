from typing import List
from solution import minMaxSums

def test_canonical_example_1():
    # spec: nums = [1, 2, 3], k = 2
    # Subsequences: [1], [2], [3], [1, 2], [1, 3], [2, 3]
    # Sums: 2 (1+1), 4 (2+2), 6 (3+3), 3 (1+2), 4 (1+3), 5 (2+3)
    # Total = 2 + 4 + 6 + 3 + 4 + 5 = 24
    assert minMaxSums([1, 2, 3], 2) == 24

def test_canonical_example_2():
    # spec: nums = [5, 0, 6], k = 1
    # Subsequences: [5], [0], [6]
    # Sums: 10 (5+5), 0 (0+0), 12 (6+6)
    # Total = 10 + 0 + 12 = 22
    assert minMaxSums([5, 0, 6], 1) == 22
