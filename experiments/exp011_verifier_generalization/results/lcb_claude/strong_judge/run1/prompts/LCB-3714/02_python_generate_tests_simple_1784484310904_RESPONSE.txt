from typing import List
from solution import minMaxSums

def test_canonical_example_1():
    # spec: nums = [1, 2, 3], k = 2
    # Subsequences: [1], [2], [3], [1, 2], [1, 3], [2, 3]
    # Sums: 2 (1+1), 4 (2+2), 6 (3+3), 3 (1+2), 4 (1+3), 5 (2+3)
    # Total = 2 + 4 + 6 + 3 + 4 + 5 = 24
    assert minMaxSums([1, 2, 3], 2) == 24

def test_canonical_example_2():
    # spec: nums = [5, 0, 6], k = 1
    # Subsequences: [5], [0], [6]
    # Sums: 10 (5+5), 0 (0+0), 12 (6+6)
    # Total = 10 + 0 + 12 = 22
    assert minMaxSums([5, 0, 6], 1) == 22

def test_canonical_example_3():
    # spec: nums = [1, 1, 1], k = 2
    # Subsequences: [1], [1], [1], [1, 1], [1, 1], [1, 1]
    # Sums: 2 (1+1), 2 (1+1), 2 (1+1), 2 (1+1), 2 (1+1), 2 (1+1)
    # Total = 2 + 2 + 2 + 2 + 2 + 2 = 12
    assert minMaxSums([1, 1, 1], 2) == 12

def test_large_values_modulo():
    # nums = [10**9, 10**9, 10**9], k = 2
    # Subsequences: [10**9], [10**9], [10**9], [10**9, 10**9], [10**9, 10**9], [10**9, 10**9]
    # Sums: 2 * (10**9 + 10**9) = 2 * 2 * 10**9 = 4 * 10**9
    # There are 3 single element subsequences contributing 10**9 each
    # Total = 3 * 10**9 + 6 * (10**9) = 9 * 10**9
    # 9 * 10**9 % (10**9 + 7) = 9 * 10**9 - 9 * (10**9 + 7) // (10**9 + 7) = 9 * 10**9 - 9 * 7
    # Total = 9 * 10**9 - 63 = 8999999937
    assert minMaxSums([10**9, 10**9, 10**9], 2) == 8999999937

def test_k_equals_length():
    # nums = [1, 2, 3], k = 3
    # All subsequences: [1], [2], [3], [1, 2], [1, 3], [2, 3], [1, 2, 3]
    # Sums: 2 (1+1), 4 (2+2), 6 (3+3), 3 (1+2), 4 (1+3), 5 (2+3), 4 (1+3)
    # Total = 2 + 4 + 6 + 3 + 4 + 5 + 6 = 30
    assert minMaxSums([1, 2, 3], 3) == 30

def test_single_element_array():
    # nums = [42], k = 1
    # Subsequences: [42]
    # Sums: 42 + 42 = 84
    # Total = 84
    assert minMaxSums([42], 1) == 84
