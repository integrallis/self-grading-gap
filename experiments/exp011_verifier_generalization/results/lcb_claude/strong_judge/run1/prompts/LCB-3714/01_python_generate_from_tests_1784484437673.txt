from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Sort the array to easily access min and max elements
    nums.sort()

    # Calculate contributions for subsequences of size 1 to k
    for size in range(1, k + 1):
        # Calculate the number of ways to choose 'size' elements
        from math import comb
        count = comb(n, size)

        # Each number contributes as minimum in `count` subsequences and as maximum in `count` subsequences
        for i in range(n):
            # As minimum
total_sum += (nums[i] * comb(n - 1 - i, size - 1)) % MOD
            # As maximum
            total_sum += (nums[n - 1 - i] * comb(i, size - 1)) % MOD

    return total_sum % MOD
```
