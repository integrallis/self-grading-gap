from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Calculate contributions for subsequences of size 1 to k
    for size in range(1, k + 1):
        # Calculate the number of ways to choose 'size' elements
        from math import comb
        count = comb(n, size)

        # For each number, it contributes to min and max in its respective subsequences
        for num in nums:
            # Each number will appear in the min and max positions the same number of times
            total_sum += (num * count * 2) % MOD
            total_sum %= MOD

    return total_sum

```
