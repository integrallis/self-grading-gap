from collections import Counter

def maxDifference(s: str) -> int:
    freq = Counter(s)
    max_odd = float('-inf')
    min_even = float('inf')
    max_even = float('-inf')
    min_odd = float('inf')
    
    for count in freq.values():
        if count % 2 == 0:
            min_even = min(min_even, count)
            max_even = max(max_even, count)
        else:
            max_odd = max(max_odd, count)
            min_odd = min(min_odd, count)
    
    return max(max_odd - min_even, max_even - min_odd)
```
