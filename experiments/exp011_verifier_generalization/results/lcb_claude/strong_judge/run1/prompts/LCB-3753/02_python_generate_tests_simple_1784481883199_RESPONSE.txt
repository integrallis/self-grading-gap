from solution import maxDifference

def test_canonical_example_1():
    # Example: input "aaaaabbc" -> 
    # 'a' frequency = 5 (odd), 'b' frequency = 2 (even) -> max difference = 5 - 2 = 3
    assert maxDifference("aaaaabbc") == 3

def test_canonical_example_2():
    # Example: input "abcabcab" -> 
    # 'a' frequency = 3 (odd), 'c' frequency = 2 (even) -> max difference = 3 - 2 = 1
    assert maxDifference("abcabcab") == 1

def test_minimum_length_string():
    # Minimum length example: input "aab" -> 
    # 'a' frequency = 2 (even), 'b' frequency = 1 (odd) -> max difference = 1 - 2 = -1
    assert maxDifference("aab") == -1

def test_multiple_even_frequency_characters():
    # Test with multiple even frequencies: input "aaaabbbbcc" -> 
    # 'a' frequency = 4 (even), 'b' frequency = 4 (even), 'c' frequency = 2 (even)
    # No odd frequencies, but this violates the constraint of having at least one odd
    # So instead, we use "aaaaabbbbc" where a=5 (odd), b=4 (even), c=1 (odd)
    # max difference = 5 - 4 = 1
    assert maxDifference("aaaaabbbbc") == 1

def test_large_difference():
    # Test where the maximum odd frequency is large: input "aaaaabbbbcdeeeeeeeeeee" -> 
    # 'a' frequency = 5 (odd), 'b' frequency = 4 (even), 'c' frequency = 1 (odd), 'd' frequency = 1 (odd), 'e' frequency = 12 (even)
    # max difference = 12 - 4 = 8
    assert maxDifference("aaaaabbbbcdeeeeeeeeeee") == 8
