from solution import maxDistance

def test_canonical_example_1():
    # Input: "NWSE", k = 1
    # Explanation: Change s[2] from 'S' to 'N'. The string s becomes "NWNE".
    # Maximum Manhattan distance: 3
    assert maxDistance("NWSE", 1) == 3

def test_canonical_example_2():
    # Input: "NSWWEW", k = 3
    # Explanation: Change s[1] from 'S' to 'N', and s[4] from 'E' to 'W'.
    # The string s becomes "NNWWWW". Maximum Manhattan distance: 6
    assert maxDistance("NSWWEW", 3) == 6

def test_k_zero_no_changes():
    # Input: "NNNN", k = 0
    # Explanation: No changes allowed. The string remains "NNNN".
    # Maximum Manhattan distance: 4 (0,0) -> (0,1) -> (0,2) -> (0,3) -> (0,4)
    assert maxDistance("NNNN", 0) == 4

def test_k_equals_length():
    # Input: "SWNE", k = 4
    # Explanation: Can change all characters. Changing all to 'N' gives "NNNN".
    # Maximum Manhattan distance: 4
    assert maxDistance("SWNE", 4) == 4

def test_minimum_length_string():
    # Input: "N", k = 0
    # Explanation: No changes allowed. The string is "N". 
    # Maximum Manhattan distance: 1
    assert maxDistance("N", 0) == 1

def test_single_direction_sequence():
    # Input: "NNNN", k = 0
    # Explanation: No changes allowed. The string remains "NNNN".
    # Maximum Manhattan distance: 4
    assert maxDistance("NNNN", 0) == 4

def test_optimal_solution_with_fewer_changes():
    # Input: "NSEW", k = 1
    # Explanation: Change s[3] from 'W' to 'E' gives "NSE".
    # Maximum Manhattan distance: 3
    assert maxDistance("NSEW", 1) == 3
