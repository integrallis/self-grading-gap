```python
def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    # Count movements
    for move in s:
        if move == 'N':
            y += 1
        elif move == 'S':
            y -= 1
        elif move == 'E':
            x += 1
        elif move == 'W':
            x -= 1
        # Calculate the current distance
        current_distance = abs(x) + abs(y)
        # Update max_distance considering changes
        max_distance = max(max_distance, current_distance)
    # Apply k changes to maximize the distance
    max_distance += k
    return max_distance
```
