```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # mirror dict for 'a' to 'z'

    for i in range(len(s)):
        if marked[i]:  # Skip if this index is already marked
            continue
        for j in range(i - 1, -1, -1):  # Check previous indices
            if marked[j]:  # Skip if j is marked
                continue
            if mirror[s[i]] == s[j]:  # Check if s[j] is the mirror of s[i]
                marked[i] = marked[j] = True  # Mark both indices
                score += i - j  # Update score
                break  # Stop after finding the closest j

    return score

```
