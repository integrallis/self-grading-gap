from solution import calculateScore

def test_canonical_example_1():
    # s = "aczzx"
    # i = 0: no j found, skip
    # i = 1: no j found, skip
    # i = 2: j = 0 (mirror of 'z' is 'a'), mark (0, 2), score += 2 - 0 = 2
    # i = 3: no j found, skip
    # i = 4: j = 1 (mirror of 'x' is 'c'), mark (1, 4), score += 4 - 1 = 3
    # Total score = 2 + 3 = 5
    assert calculateScore("aczzx") == 5

def test_canonical_example_2():
    # s = "abcdef"
    # For each index i, there is no index j that satisfies the conditions.
    # Total score = 0
    assert calculateScore("abcdef") == 0

def test_single_character():
    # s = "a"
    # There are no prior indices to find a mirror, so total score = 0
    assert calculateScore("a") == 0

def test_closest_mirror():
    # s = "aza"
    # i = 0: no j found, skip
    # i = 1: j = 0 (mirror of 'a' is 'z'), mark (0, 1), score += 1 - 0 = 1
    # i = 2: j = 0 (mirror of 'a' is 'z'), but j is already marked, skip
    # Total score = 1
    assert calculateScore("aza") == 1
