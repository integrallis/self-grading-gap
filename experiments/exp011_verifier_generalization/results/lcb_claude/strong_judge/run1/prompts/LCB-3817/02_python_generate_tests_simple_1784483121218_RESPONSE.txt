from solution import maxContainers

def test_canonical_example_1():
    # spec: n = 2, w = 3, maxWeight = 15
    # total cells = 2 * 2 = 4, total weight = 4 * 3 = 12, maxWeight = 15
    # Since 12 <= 15, we can load all 4 containers.
    assert maxContainers(2, 3, 15) == 4

def test_canonical_example_2():
    # spec: n = 3, w = 5, maxWeight = 20
    # total cells = 3 * 3 = 9, total weight = 4 * 5 = 20, maxWeight = 20
    # Since 20 <= 20, we can load 4 containers.
    assert maxContainers(3, 5, 20) == 4

def test_minimum_deck_size():
    # spec: n = 1, w = 1, maxWeight = 1
    # total cells = 1 * 1 = 1, total weight = 1 * 1 = 1, maxWeight = 1
    # Since 1 <= 1, we can load 1 container.
    assert maxContainers(1, 1, 1) == 1

def test_maxWeight_less_than_weight():
    # spec: n = 2, w = 5, maxWeight = 3
    # total cells = 2 * 2 = 4, total weight = 0 * 5 = 0
    # Since maxWeight < w, we cannot load any containers.
    assert maxContainers(2, 5, 3) == 0

def test_floor_division_case():
    # spec: n = 5, w = 7, maxWeight = 25
    # total cells = 5 * 5 = 25, total weight = 3 * 7 = 21, maxWeight = 25
    # We can load at most 3 containers since 25 // 7 = 3
    assert maxContainers(5, 7, 25) == 3

def test_single_cell_deck_limited_by_weight():
    # spec: n = 1, w = 10, maxWeight = 5
    # total cells = 1 * 1 = 1, total weight = 0 * 10 = 0
    # Since maxWeight < w, we cannot load any containers.
    assert maxContainers(1, 10, 5) == 0

def test_large_deck_small_weight_limit():
    # spec: n = 10, w = 2, maxWeight = 10
    # total cells = 10 * 10 = 100, total weight = 5 * 2 = 10
    # We can load at most 5 containers since 10 // 2 = 5
    assert maxContainers(10, 2, 10) == 5
