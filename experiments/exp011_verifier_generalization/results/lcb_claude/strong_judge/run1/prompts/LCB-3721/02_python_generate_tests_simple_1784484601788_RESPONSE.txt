from typing import List
from solution import countMentions

def test_canonical_example_1():
    # Input: 2, [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "71", "HERE"]]
    # At timestamp 10, id1 and id0 are mentioned. mentions = [1, 1]
    # At timestamp 11, id0 goes offline.
    # At timestamp 71, "HERE" is mentioned. id0 is offline, so mentions = [2, 2]
    assert countMentions(2, [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "71", "HERE"]]) == [2, 2]

def test_canonical_example_2():
    # Input: 2, [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "12", "ALL"]]
    # At timestamp 10, id1 and id0 are mentioned. mentions = [1, 1]
    # At timestamp 11, id0 goes offline.
    # At timestamp 12, "ALL" is mentioned. Both id0 and id1 are mentioned, mentions = [2, 2]
    assert countMentions(2, [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "12", "ALL"]]) == [2, 2]

def test_canonical_example_3():
    # Input: 2, [["OFFLINE", "10", "0"], ["MESSAGE", "12", "HERE"]]
    # At timestamp 10, id0 goes offline.
    # At timestamp 12, "HERE" is mentioned. id0 is offline, so mentions = [0, 1]
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "12", "HERE"]]) == [0, 1]

def test_duplicate_id_mentions():
    # Input: 2, [["MESSAGE", "10", "id0 id0 id0"]]
    # At timestamp 10, id0 is mentioned three times. mentions = [3, 0]
    assert countMentions(2, [["MESSAGE", "10", "id0 id0 id0"]]) == [3, 0]

def test_multiple_users_offline():
    # Input: 3, [["OFFLINE", "10", "0"], ["OFFLINE", "15", "1"], ["MESSAGE", "20", "id0 id1 id2"]]
    # At timestamp 10, id0 goes offline. At timestamp 15, id1 goes offline. At timestamp 20, all users are offline.
    # mentions = [1, 1, 1] should count only id2.
    assert countMentions(3, [["OFFLINE", "10", "0"], ["OFFLINE", "15", "1"], ["MESSAGE", "20", "id0 id1 id2"]]) == [1, 1, 1]

def test_more_than_two_ids_mentioned():
    # Input: 3, [["MESSAGE", "10", "id0 id1 id2"]]
    # At timestamp 10, id0, id1, and id2 are mentioned once each. mentions = [1, 1, 1]
    assert countMentions(3, [["MESSAGE", "10", "id0 id1 id2"]]) == [1, 1, 1]

def test_same_timestamp_offline_and_message():
    # Input: 2, [["OFFLINE", "10", "0"], ["MESSAGE", "10", "HERE"]]
    # At timestamp 10, id0 goes offline, and "HERE" mentions only online users (id1). mentions = [0, 1]
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "10", "HERE"]]) == [0, 1]
