```python
from typing import List

def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers
    online_status = [True] * numberOfUsers
    offline_till = [0] * numberOfUsers

    for event in events:
        event_type, timestamp, *details = event
        timestamp = int(timestamp)

        if event_type == 'OFFLINE':
            user_id = int(details[0])
            online_status[user_id] = False
            offline_till[user_id] = timestamp + 60

        elif event_type == 'MESSAGE':
            # Process any offline updates before this message
            for user_id in range(numberOfUsers):
                if not online_status[user_id] and timestamp >= offline_till[user_id]:
                    online_status[user_id] = True

            if details[0] == 'ALL':
                for user_id in range(numberOfUsers):
                    mentions[user_id] += 1
            elif details[0] == 'HERE':
                for user_id in range(numberOfUsers):
                    if online_status[user_id]:
                        mentions[user_id] += 1
            else:
                for mention in details[0].split():
                    user_id = int(mention[2:])
                    mentions[user_id] += 1

    return mentions

```
