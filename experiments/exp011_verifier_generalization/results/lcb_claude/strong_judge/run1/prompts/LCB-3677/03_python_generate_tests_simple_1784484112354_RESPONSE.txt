from typing import List
from solution import maximumAmount

def test_canonical_example_1():
    # spec: coins = [[0,1,-1],[1,-2,3],[2,-3,4]]
    # An optimal path for maximum coins is:
    # Start at (0, 0) with 0 coins (total coins = 0).
    # Move to (0, 1), gaining 1 coin (total coins = 1).
    # Move to (1, 1), where there's a robber stealing 2 coins. 
    # The robot uses one neutralization here, avoiding the robbery (total coins = 1).
    # Move to (1, 2), gaining 3 coins (total coins = 4).
    # Move to (2, 2), gaining 4 coins (total coins = 8).
    assert maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) == 8

def test_canonical_example_2():
    # spec: coins = [[10,10,10],[10,10,10]]
    # An optimal path for maximum coins is:
    # Start at (0, 0) with 10 coins (total coins = 10).
    # Move to (0, 1), gaining 10 coins (total coins = 20).
    # Move to (0, 2), gaining another 10 coins (total coins = 30).
    # Move to (1, 2), gaining the final 10 coins (total coins = 40).
    assert maximumAmount([[10, 10, 10], [10, 10, 10]]) == 40

def test_minimal_grid():
    # spec: coins = [[5]]
    # The only cell contains 5 coins, so the output should be 5.
    assert maximumAmount([[5]]) == 5

def test_minimal_grid_with_robber():
    # spec: coins = [[-5]]
    # The only cell contains a robber stealing 5 coins, the robot can neutralize it,
    # so the optimal output should be 0 (neutralizing the robber).
    assert maximumAmount([[-5]]) == 0

def test_two_neutralizations():
    # coins = [[0, -1, -1], [-1, -1, 0], [-1, 0, 0]]
    # An optimal path is (0,0) → (0,1) → (0,2) → (1,2) → (2,2)
    # Neutralizing (0,1) and (0,2) gives: 0 - 1 + 0 + 0 + 0 = 0
    assert maximumAmount([[0, -1, -1], [-1, -1, 0], [-1, 0, 0]]) == 0

def test_extreme_values():
    # coins = [[1000, -1000], [-1000, 1000]]
    # An optimal path is (0,0) → (0,1) → (1,1)
    # This results in: 1000 - 1000 + 1000 = 2000 (neutralizing (0,1))
    assert maximumAmount([[1000, -1000], [-1000, 1000]]) == 2000

def test_starting_with_robber():
    # coins = [[-1, 2], [2, 3]]
    # The robot must neutralize the starting point (-1) to maximize coins.
    # Path: (0,0) → (0,1) → (1,1)
    # After neutralizing the robber, the coins will be: 0 + 2 + 3 = 5
    assert maximumAmount([[-1, 2], [2, 3]]) == 5

def test_using_one_neutralization_optimal():
    # coins = [[1, -5, 1], [1, 1, 1]]
    # Optimal path is (0,0) → (0,1) (neutralize) → (0,2) → (1,2)
    # This results in: 1 + 0 + 1 + 1 = 5
    assert maximumAmount([[1, -5, 1], [1, 1, 1]]) == 5

    # Optimal path is (0,0) → (0,1) → (1,1) → (1,0)
    # This results in: 1 + 2 + 4 + 3 = 10
    assert maximumAmount([[1, 2], [3, 4]]) == 10
