```python
from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = [[1] * 301 for _ in range(n)]  # dp[i][j] will hold the length of the longest subsequence ending with nums[i] and last difference j
    max_length = 1

    for i in range(n):
        for j in range(i):
            diff = abs(nums[i] - nums[j])
            for length in range(1, 301):
                if diff <= length:
                    dp[i][diff] = max(dp[i][diff], dp[j][length] + 1)
                    max_length = max(max_length, dp[i][diff])

    return max_length

```
