from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    # Initialize free time and meetings
    free_time_intervals = []
    last_end_time = 0

    # Collect all free time intervals
    for i in range(n):
        if startTime[i] > last_end_time:
            free_time_intervals.append((last_end_time, startTime[i]))
        last_end_time = max(last_end_time, endTime[i])

    # Check free time after the last meeting
    if last_end_time < eventTime:
        free_time_intervals.append((last_end_time, eventTime))

    # If no meetings can be rescheduled, return the maximum free time
    if k == 0:
        return max((end - start for start, end in free_time_intervals), default=0)

    # Calculate the maximum free time possible by rescheduling k meetings
    max_free_time = 0
    for i in range(len(free_time_intervals) - 1):
        gap = free_time_intervals[i + 1][0] - free_time_intervals[i][1]
        if gap > 0:
            max_free_time = max(max_free_time, gap)

    return max_free_time

```
