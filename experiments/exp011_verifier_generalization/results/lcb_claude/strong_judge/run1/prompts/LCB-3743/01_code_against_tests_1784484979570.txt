```python
from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    end_points = []

    # Collect all meetings' end points
    for i in range(n):
        end_points.append((startTime[i], endTime[i]))

    # Initialize free intervals and counters
    free_intervals = []
    last_end_time = 0

    # Iterate over meetings to find free intervals
    for start, end in end_points:
        if start > last_end_time:
            # There is a free interval
            free_intervals.append((last_end_time, start))
        last_end_time = max(last_end_time, end)

    # Add the final free time interval if applicable
    if last_end_time < eventTime:
        free_intervals.append((last_end_time, eventTime))

    # Create cost array for how much time we can shift back
    cost = []
    for i in range(n):
        shift_start = startTime[i] - (endTime[i] if i > 0 else 0)  # shifting to the beginning of free spots
        if i > 0:
            shift_start = max(0, shift_start)  # can only shift back to 0 or more
        cost.append(shift_start)

    # Sort costs in descending order and sum the largest k costs
    cost.sort(reverse=True)
    max_free_time = sum(cost[:k])

    return max(max_free_time, max((end - start for start, end in free_intervals), default=0))
```
