from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a list to hold the coin counts for each bag position
    max_position = 0
    for l, r, c in coins:
        max_position = max(max_position, r)

    # Initialize a list of zeros for the coin counts
    coin_count = [0] * (max_position + 1)

    # Fill the coin_count array based on the coins input
    for l, r, c in coins:
        for i in range(l, r + 1):
            coin_count[i] += c

    # Calculate the maximum coins that can be collected in k consecutive bags
    max_coins = 0
    current_coins = sum(coin_count[1:min(k + 1, max_position + 1)])  # Initial sum for the first k bags
    max_coins = current_coins

    for i in range(1, max_position - k + 2):  # Sliding window
        if i + k <= max_position:
            current_coins = current_coins - coin_count[i] + coin_count[i + k]
            max_coins = max(max_coins, current_coins)

    return max_coins

```
