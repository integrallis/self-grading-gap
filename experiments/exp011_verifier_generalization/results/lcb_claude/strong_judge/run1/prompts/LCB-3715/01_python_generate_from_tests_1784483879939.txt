from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a list to hold the coin counts for each bag position
    max_position = 0
    min_position = float('inf')
    for l, r, c in coins:
        max_position = max(max_position, r)
        min_position = min(min_position, l)

    # Initialize a list of zeros for the coin counts/potential bag positions
    coin_count = [0] * (max_position + 1)

    # Fill the coin_count array based on the coins input
    for l, r, c in coins:
        for i in range(l, r + 1):
            coin_count[i] += c

    # Calculate the maximum coins that can be collected in k consecutive bags
    max_coins = 0

    # Consider bags starting from position beyond the first segment to the last, allowing out of bounds as 0
    for start in range(min_position - k + 1, max_position + 1):
        current_coins = 0
        for i in range(k):
            if start + i < len(coin_count):
                current_coins += coin_count[start + i]
        max_coins = max(max_coins, current_coins)

    return max_coins

```
