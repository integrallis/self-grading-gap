from typing import List

def maxSum(nums: List[int]) -> int:
    unique_elements = set()
    max_sum = 0
    current_sum = 0
    for num in nums:
        if num not in unique_elements:
            unique_elements.add(num)
            current_sum += num
        else:
            if current_sum > max_sum:
                max_sum = current_sum
            # Start a new sum calculation
            current_sum = num if num not in unique_elements else 0
            unique_elements.clear()
            unique_elements.add(num)
            current_sum = num
    return max(max_sum, current_sum)
```
