from typing import List
from solution import minCosts

def test_canonical_example_1():
    # spec: input [5, 3, 4, 1, 3, 2] 
    # costs to reach each position are:
    # 0: swap with 0 for 5
    # 1: swap with 1 for 3
    # 2: swap with 1 for 3, then swap with 2 for free
    # 3: swap with 3 for 1
    # 4: swap with 3 for 1, then swap with 4 for free
    # 5: swap with 3 for 1, then swap with 5 for free
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]

def test_canonical_example_2():
    # spec: input [1, 2, 4, 6, 7]
    # costs to reach each position are:
    # 0: swap with 0 for 1
    # 1: swap with 0 for 1
    # 2: swap with 0 for 1
    # 3: swap with 0 for 1
    # 4: swap with 0 for 1
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]

def test_single_element():
    # spec: input [5]
    # costs to reach each position are:
    # 0: swap with 0 for 5
    assert minCosts([5]) == [5]

def test_monotonically_decreasing_costs():
    # spec: input [10, 5, 1]
    # costs to reach each position are:
    # 0: swap with 0 for 10
    # 1: swap with 1 for 5
    # 2: swap with 2 for 1
    assert minCosts([10, 5, 1]) == [10, 5, 1]

    # spec: input [5, 5, 5]
