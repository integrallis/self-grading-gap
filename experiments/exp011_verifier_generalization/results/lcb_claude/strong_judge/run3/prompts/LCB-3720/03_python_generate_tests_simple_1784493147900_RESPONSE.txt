from typing import List
from solution import minMaxWeight

def test_canonical_example_1():
    # Input: n = 5, edges = [[1,0,1],[2,0,2],[3,0,1],[4,3,1],[2,1,1]], threshold = 2
    # Output: 1
    assert minMaxWeight(5, [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]], 2) == 1

def test_canonical_example_2():
    # Input: n = 5, edges = [[0,1,1],[0,2,2],[0,3,1],[0,4,1],[1,2,1],[1,4,1]], threshold = 1
    # Output: -1
    assert minMaxWeight(5, [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]], 1) == -1

def test_canonical_example_3():
    # Input: n = 5, edges = [[1,2,1],[1,3,3],[1,4,5],[2,3,2],[3,4,2],[4,0,1]], threshold = 1
    # Output: 2
    assert minMaxWeight(5, [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]], 1) == 2

def test_canonical_example_4():
    # Input: n = 5, edges = [[1,2,1],[1,3,3],[1,4,5],[2,3,2],[4,0,1]], threshold = 1
    # Output: -1
    assert minMaxWeight(5, [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]], 1) == -1

def test_minimum_graph_size():
    # Input: n = 2, edges = [[1,0,1]], threshold = 1
    # Output: 1 (node 1 can reach node 0 with weight 1)
    assert minMaxWeight(2, [[1, 0, 1]], 1) == 1

def test_no_edge_removal_needed():
    # Input: n = 3, edges = [[1,0,1],[2,0,1]], threshold = 1
    # Output: 1 (both edges can be kept, max weight = 1)
    assert minMaxWeight(3, [[1, 0, 1], [2, 0, 1]], 1) == 1

def test_maximum_threshold():
    # Input: n = 4, edges = [[1,0,1],[2,1,2],[3,2,3],[3,0,4]], threshold = 3
    # Output: 4 (can keep all edges, max weight = 4)
    assert minMaxWeight(4, [[1, 0, 1], [2, 1, 2], [3, 2, 3], [3, 0, 4]], 3) == 4

def test_multiple_edges_between_same_nodes():
    # Input: n = 3, edges = [[1,0,1],[1,0,2],[2,1,3]], threshold = 1
    # Output: 3 (keep one of the edges 1->0 with weight 1, and 2->1 with weight 3)
    assert minMaxWeight(3, [[1, 0, 1], [1, 0, 2], [2, 1, 3]], 1) == 3

def test_large_weights():
    # Input: n = 3, edges = [[1, 0, 1000000], [2, 1, 999999]], threshold = 1
    # Output: 999999 (must remove edge 1->0 to keep node 2 connected)
    assert minMaxWeight(3, [[1, 0, 1000000], [2, 1, 999999]], 1) == 999999

def test_threshold_blocks_path():
    # Input: n = 4, edges = [[1, 0, 1], [2, 1, 2], [3, 2, 3]], threshold = 2
    # Output: -1 (removing edges to satisfy threshold makes node 3 unable to reach node 0)
    assert minMaxWeight(4, [[1, 0, 1], [2, 1, 2], [3, 2, 3]], 2) == -1
