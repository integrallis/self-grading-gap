from typing import List

def minimumPairRemoval(nums: List[int]) -> int:
    operations = 0
    while True:
        # Check if the array is already non-decreasing
        if all(nums[i] <= nums[i + 1] for i in range(len(nums) - 1)):
            break

        # Find the leftmost index where the array is not non-decreasing
        min_index = -1
        for i in range(len(nums) - 1):
            if nums[i] > nums[i + 1]:
                min_index = i
                break

        # If no index found, it means array is non-decreasing
        if min_index == -1:
            break

        # Replace the pair with their sum
        nums[min_index] = nums[min_index] + nums[min_index + 1]
        del nums[min_index + 1]
        operations += 1

    return operations
