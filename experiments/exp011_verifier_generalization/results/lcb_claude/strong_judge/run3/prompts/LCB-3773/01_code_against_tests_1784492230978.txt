from typing import List


def minimumPairRemoval(nums: List[int]) -> int:
    from collections import deque
    initial_tuple = tuple(nums)
    queue = deque()  # for BFS
    queue.append((initial_tuple, 0))  # (current_array, operations_count)
    visited = set()  # to avoid reprocessing same array
    visited.add(initial_tuple)

    while queue:
        current, operations = queue.popleft()

        # Check if the array is non-decreasing
        if all(current[i] <= current[i + 1] for i in range(len(current) - 1)):
            return operations

        # Try merging all adjacent pairs
        for i in range(len(current) - 1):
            # Create new array after merging current[i] and current[i+1]
            new_list = list(current[:i]) + [current[i] + current[i + 1]] + list(current[i + 2:])
            new_tuple = tuple(new_list)

            # If this new arrangement hasn't been visited, add to the queue
            if new_tuple not in visited:
                visited.add(new_tuple)
                queue.append((new_tuple, operations + 1))

    return -1  # Should not reach here if input is valid
