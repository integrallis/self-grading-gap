from solution import maxContainers

def test_canonical_example_1():
    # spec: n = 2, w = 3, maxWeight = 15
    # The deck has 4 cells, each can hold a container weighing 3.
    # Total weight of 4 containers = 4 * 3 = 12, which is ≤ 15.
    assert maxContainers(2, 3, 15) == 4

def test_canonical_example_2():
    # spec: n = 3, w = 5, maxWeight = 20
    # The deck has 9 cells, each can hold a container weighing 5.
    # Total weight of 4 containers = 4 * 5 = 20, which is ≤ 20.
    assert maxContainers(3, 5, 20) == 4

def test_minimum_constraints():
    # spec: n = 1, w = 1, maxWeight = 1
    # The deck has 1 cell, which can hold 1 container weighing 1.
    # Total weight of 1 container = 1 * 1 = 1, which is ≤ 1.
    assert maxContainers(1, 1, 1) == 1

def test_zero_containers_case():
    # spec: n = 2, w = 10, maxWeight = 5
    # The deck has 4 cells, but each container weighs 10.
    # Total weight of 0 containers = 0, which is ≤ 5.
    assert maxContainers(2, 10, 5) == 0

def test_exact_deck_capacity():
    # spec: n = 2, w = 5, maxWeight = 20
    # The deck has 4 cells, each can hold a container weighing 5.
    # Total weight of 4 containers = 4 * 5 = 20, which is ≤ 20.
    assert maxContainers(2, 5, 20) == 4

def test_large_n_value():
    # spec: n = 1000, w = 1, maxWeight = 1000
    # The deck has 1000 * 1000 = 1000000 cells, each can hold a container weighing 1.
    # Maximum containers = min(1000000, floor(1000/1)) = min(1000000, 1000) = 1000.
    assert maxContainers(1000, 1, 1000) == 1000

def test_large_maxWeight_value():
    # spec: n = 10, w = 1, maxWeight = 10^9
    # The deck has 10 * 10 = 100 cells, each can hold a container weighing 1.
    # Maximum containers = min(100, floor(10^9/1)) = min(100, 1000000000) = 100.
    assert maxContainers(10, 1, 10**9) == 100
