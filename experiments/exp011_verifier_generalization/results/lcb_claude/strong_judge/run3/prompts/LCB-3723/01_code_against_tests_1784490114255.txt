## Implementation

```python
from typing import List

def sumOfGoodNumbers(nums: List[int], k: int) -> int:
    total = 0
    n = len(nums)
    for i in range(n):
        good = True
        if i - k >= 0:
            good = good and nums[i] > nums[i - k]
        if i + k < n:
            good = good and nums[i] > nums[i + k]
        if good:
            total += nums[i]
    return total
```
