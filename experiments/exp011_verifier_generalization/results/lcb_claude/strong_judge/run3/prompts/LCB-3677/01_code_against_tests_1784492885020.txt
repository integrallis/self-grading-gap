```python
from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    # DP table to store maximum coins with up to 2 neutralizations
    dp = [[[-float('inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0]  # Starting point, no neutralizations used

    for i in range(m):
        for j in range(n):
            for k in range(3):  # k = number of neutralizations used
                if i == 0 and j == 0:
                    continue
                # Coming from the top
                if i > 0:
                    coins_here = coins[i][j]
                    if coins_here < 0 and k > 0:
                        coins_here = 0  # Neutralize robber
                    dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k] + coins_here)
                # Coming from the left
                if j > 0:
                    coins_here = coins[i][j]
                    if coins_here < 0 and k > 0:
                        coins_here = 0  # Neutralize robber
                    dp[i][j][k] = max(dp[i][j][k], dp[i][j-1][k] + coins_here)
                # Using a neutralization if there are remaining
                if coins[i][j] < 0 and k > 0:
                    dp[i][j][k] = max(dp[i][j][k], dp[i][j][k-1] - coins[i][j])

    # The result is the maximum coins at the bottom-right corner with up to 2 neutralizations
    return max(dp[m-1][n-1][0], dp[m-1][n-1][1], dp[m-1][n-1][2])
```
