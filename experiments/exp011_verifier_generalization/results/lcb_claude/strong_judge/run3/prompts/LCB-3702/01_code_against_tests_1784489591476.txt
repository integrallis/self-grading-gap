## Implementation

```python
from typing import List
import math


def maxLength(nums: List[int]) -> int:
    n = len(nums)
    max_length = 0
    for i in range(n):
        product = 1
        current_gcd = nums[i]
        for j in range(i, n):
            product *= nums[j]
            current_gcd = math.gcd(current_gcd, nums[j])
            lcm = product // current_gcd
            if product == lcm * current_gcd:
                max_length = max(max_length, j - i + 1)
    return max_length
```
