def test_canonical_example_1():
    # spec: maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) -> 10

    assert maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) == 10

    # spec: maximumCoins([[1, 10, 3]], 2) -> 6

    # spec: maximumCoins([[5, 10, 3]], 1) -> 3
