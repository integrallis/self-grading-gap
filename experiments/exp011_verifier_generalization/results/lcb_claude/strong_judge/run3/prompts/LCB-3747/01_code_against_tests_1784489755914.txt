## Implementation

```python
from typing import List

def maxAdjacentDistance(nums: List[int]) -> int:
    max_distance = 0
    n = len(nums)
    for i in range(n):
        # Calculate the absolute difference between the current element and the next (circular)
        distance = abs(nums[i] - nums[(i + 1) % n])
        max_distance = max(max_distance, distance)
    return max_distance

```
