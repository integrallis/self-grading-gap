def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # Create mirror mapping for 'a' to 'z'
    for i in range(len(s)):
        if marked[i]:  # Skip if already marked
            continue
        for j in range(i - 1, -1, -1):  # Find the closest unmarked index j < i
            if not marked[j] and mirror[s[i]] == s[j]:  # Check mirror condition
                marked[i] = marked[j] = True  # Mark both indices
                score += i - j  # Update score
                break  # Stop looking for j after finding the closest
    return score
