from typing import List

def maxFrequency(nums: List[int], k: int) -> int:
    count = 0
    for num in nums:
        if num == k:
            count += 1
    nums.sort()
    left = 0
    max_freq = count
    for right in range(len(nums)):
        if nums[right] > k:
            continue
        while left < right and nums[left] < k:
            left += 1
        max_freq = max(max_freq, right - left + 1 + count)
    return max_freq
```
