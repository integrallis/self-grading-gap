## Implementation

```python
def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]
    # Check if the string starts with the prefix and ends with the suffix
    for i in range(len(s) - len(suffix) + 1):
        if s[i:i + len(prefix)] == prefix:
            if s[i + len(prefix):].endswith(suffix):
                return True
    return False

```
