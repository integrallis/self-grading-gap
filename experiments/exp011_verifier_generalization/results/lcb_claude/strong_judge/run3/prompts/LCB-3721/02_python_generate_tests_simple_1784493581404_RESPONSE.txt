from typing import List
from solution import countMentions

def test_canonical_example_1():
    # spec: countMentions(2, [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "71", "HERE"]]) 
    # mentions at 10: id1 -> 1, id0 -> 1; at 11: id0 goes offline; at 71: id1 -> 2, id0 -> 2 (back online)
    assert countMentions(2, [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "71", "HERE"]]) == [2, 2]

def test_canonical_example_2():
    # spec: countMentions(2, [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "12", "ALL"]])
    # mentions at 10: id1 -> 1, id0 -> 1; at 11: id0 goes offline; at 12: ALL mentions id0 -> 2, id1 -> 2
    assert countMentions(2, [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "12", "ALL"]]) == [2, 2]

def test_canonical_example_3():
    # spec: countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "12", "HERE"]])
    # at 10: id0 goes offline; at 12: HERE mentions id1 -> 1, id0 (offline) -> 0
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "12", "HERE"]]) == [0, 1]

def test_duplicate_mentions_in_message():
    # spec: countMentions(2, [["MESSAGE", "10", "id0 id0 id1"]])
    # mentions at 10: id0 -> 2 (twice), id1 -> 1
    assert countMentions(2, [["MESSAGE", "10", "id0 id0 id1"]]) == [2, 1]

def test_multiple_offline_cycles():
    # spec: countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "12", "HERE"], ["OFFLINE", "20", "0"], ["MESSAGE", "80", "id1"]])
    # at 10: id0 goes offline; at 12: HERE mentions id1 -> 1, id0 (offline) -> 0; at 20: id0 goes offline again
    # at 80: id1 mentions id1 -> 2
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "12", "HERE"], ["OFFLINE", "20", "0"], ["MESSAGE", "80", "id1"]]) == [0, 2]

def test_single_user():
    # spec: countMentions(1, [["MESSAGE", "10", "id0"]])
    # mentions at 10: id0 -> 1
    assert countMentions(1, [["MESSAGE", "10", "id0"]]) == [1]

def test_message_at_boundary_of_offline_period():
    # spec: countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "70", "HERE"]])
    # at 10: id0 goes offline; at 70: HERE mentions id1 -> 1, id0 (offline) -> 0
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "70", "HERE"]]) == [0, 1]

def test_mixed_mentions():
    # spec: countMentions(2, [["MESSAGE", "10", "id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "12", "ALL"], ["MESSAGE", "13", "HERE"]])
    # at 10: id0 -> 1; at 11: id0 goes offline; at 12: ALL mentions id0 -> 2, id1 -> 1; at 13: HERE mentions id1 -> 2, id0 (offline) -> 2
    assert countMentions(2, [["MESSAGE", "10", "id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "12", "ALL"], ["MESSAGE", "13", "HERE"]]) == [2, 2]
