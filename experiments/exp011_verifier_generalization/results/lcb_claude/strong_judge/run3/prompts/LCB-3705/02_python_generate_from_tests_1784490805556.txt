from typing import List


def largestInteger(nums: List[int], k: int) -> int:
    from collections import defaultdict
    subarray_count = defaultdict(int)
    n = len(nums)

    # Count appearances of each number in all subarrays of size k
    for i in range(n - k + 1):
        unique_in_subarray = set(nums[i:i + k])
        for num in unique_in_subarray:
            subarray_count[num] += 1

    # Find the largest number that appears in exactly one subarray of size k
    largest_missing = -1
    for num, count in subarray_count.items():
        if count == 1:
            largest_missing = max(largest_missing, num)

    return largest_missing
```
