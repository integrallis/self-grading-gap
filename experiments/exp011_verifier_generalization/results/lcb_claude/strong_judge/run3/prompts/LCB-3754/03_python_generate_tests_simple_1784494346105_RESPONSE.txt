from solution import maxDistance

def test_canonical_example_1():
    # example: input "NWSE", k = 1
    # Change s[2] from 'S' to 'N'. The string s becomes "NWNE".
    # Maximum Manhattan distance is 3.
    assert maxDistance("NWSE", 1) == 3

def test_canonical_example_2():
    # example: input "NSWWEW", k = 3
    # Change s[1] from 'S' to 'N', and s[4] from 'E' to 'W'. The string s becomes "NNWWWW".
    # Maximum Manhattan distance is 6.
    assert maxDistance("NSWWEW", 3) == 6

def test_minimum_length():
    # example: input "N", k = 0
    # Maximum Manhattan distance is 1.
    assert maxDistance("N", 0) == 1

def test_no_changes_allowed():
    # example: input "NS", k = 0
    # Maximum Manhattan distance is 1 (only one move can be used).
    assert maxDistance("NS", 0) == 1

def test_all_changes_allowed():
    # example: input "SSSS", k = 4
    # Changing all 'S' to 'N' gives maximum Manhattan distance of 4.
    assert maxDistance("SSSS", 4) == 4

def test_single_direction_string():
    # example: input "NNNN", k = 0
    # Maximum Manhattan distance is 4 (all moves north).
    assert maxDistance("NNNN", 0) == 4

def test_opposing_moves():
    # example: input "NSNS", k = 2
    # Change both 'S' to 'N' gives maximum Manhattan distance of 4.
    assert maxDistance("NSNS", 2) == 4

def test_optimal_solution_requires_fewer_changes():
    # example: input "NNS", k = 1
    # Changing one 'S' to 'N' gives maximum Manhattan distance of 3.
    assert maxDistance("NNS", 1) == 3

def test_max_distance_occurs_at_final_position():
    # example: input "NNEE", k = 0
    # Maximum Manhattan distance is 4 (at position (2, 2)).
    assert maxDistance("NNEE", 0) == 4

def test_max_distance_occurs_in_middle():
    # example: input "NSSSEE", k = 2
    # Changing two 'S' to 'N' gives maximum Manhattan distance of 4.
    assert maxDistance("NSSSEE", 2) == 4

def test_longer_string():
    # example: input "N" * 100000, k = 0
    # Maximum Manhattan distance is 100000 (all moves north).
    assert maxDistance("N" * 100000, 0) == 100000

def test_all_four_directions():
    # example: input "NEWS", k = 2
    # Changing 'E' to 'W' and 'S' to 'N' gives maximum Manhattan distance of 4.
    assert maxDistance("NEWS", 2) == 4
