```python
def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    for move in s:
        if move == 'N':
            y += 1
        elif move == 'S':
            y -= 1
        elif move == 'E':
            x += 1
        elif move == 'W':
            x -= 1
        # Calculate current Manhattan distance
        current_distance = abs(x) + abs(y)
        # Update maximum distance considering changes allowed
        max_distance = max(max_distance, current_distance)
        if k > 0:
            # Consider the effect of changing the last move to the opposite direction
            max_distance = max(max_distance, current_distance + 2)
            k -= 1  # Simulate using a change
    return max_distance
```
