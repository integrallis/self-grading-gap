from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    dp = [[0] * 3 for _ in range(n)]
    dp[0] = cost[0]

    for i in range(1, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(
                dp[i-1][(j+1) % 3],
                dp[i-1][(j+2) % 3]
            )

    # Handle equidistant houses
    dp[n-1][0] = min(dp[n-1][0], cost[n-1][0] + min(dp[n-2][1], dp[n-2][2]))
    dp[n-1][1] = min(dp[n-1][1], cost[n-1][1] + min(dp[n-2][0], dp[n-2][2]))
    dp[n-1][2] = min(dp[n-1][2], cost[n-1][2] + min(dp[n-2][0], dp[n-2][1]))

    return min(dp[n-1])

```
