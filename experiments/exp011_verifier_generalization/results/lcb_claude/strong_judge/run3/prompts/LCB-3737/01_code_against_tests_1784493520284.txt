```python
from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    dp = [[0] * 3 for _ in range(n)]
    dp[0] = cost[0]

    for i in range(1, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(
                dp[i-1][(j+1) % 3],
                dp[i-1][(j+2) % 3]
            )

    # Handle equidistant houses for the first half
    for j in range(3):
        dp[n//2-1][j] = min(dp[n//2-1][j], cost[n//2-1][j] + min(
            dp[0][(j+1) % 3], dp[0][(j+2) % 3]
        ))

    # Final adjustments for the last house based on the first house
    for j in range(3):
        dp[n-1][j] = cost[n-1][j] + min(
            dp[n-2][(j+1) % 3], dp[n-2][(j+2) % 3],
            dp[n//2-1][(j+1) % 3], dp[n//2-1][(j+2) % 3]
        )

    return min(dp[n-1])
```
