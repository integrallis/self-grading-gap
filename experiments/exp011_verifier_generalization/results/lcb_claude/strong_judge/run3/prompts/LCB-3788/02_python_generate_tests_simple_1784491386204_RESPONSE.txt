from typing import List
from solution import maxSum

def test_canonical_example_1():
    # spec: maxSum([1, 2, 3, 4, 5]) -> 15 (sum of all elements)
    assert maxSum([1, 2, 3, 4, 5]) == 15

def test_canonical_example_2():
    # spec: maxSum([1, 1, 0, 1, 1]) -> 1 (delete duplicates and select [1])
    assert maxSum([1, 1, 0, 1, 1]) == 1

def test_canonical_example_3():
    # spec: maxSum([1, 2, -1, -2, 1, 0, -1]) -> 3 (delete -1 and -2, select [2, 1])
    assert maxSum([1, 2, -1, -2, 1, 0, -1]) == 3

def test_single_element_array():
    # spec: maxSum([5]) -> 5 (only one element, return it)
    assert maxSum([5]) == 5

def test_all_negative_array():
    # spec: maxSum([-5, -3, -8]) -> -3 (maximum single element)
    assert maxSum([-5, -3, -8]) == -3

def test_boundary_values():
    # spec: maxSum([100, -100, 50]) -> 150 (select [100, 50])
    assert maxSum([100, -100, 50]) == 150

def test_maximum_length_array():
    # spec: maxSum with unique elements in maximum length array
    assert maxSum(list(range(1, 101))) == sum(range(1, 101))  # sum of 1 to 100
