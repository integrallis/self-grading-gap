from typing import List

def maxSum(nums: List[int]) -> int:
    unique_elements = set()
    max_sum = float('-inf')
    current_sum = 0

    for num in nums:
        if num not in unique_elements:
            unique_elements.add(num)
            current_sum += num
        else:
            max_sum = max(max_sum, current_sum)
            # Reset the current sum and unique elements
            current_sum = num
            unique_elements = {num}

    # Final check to update max_sum with the last seen sum
    max_sum = max(max_sum, current_sum)

    return max_sum
```
