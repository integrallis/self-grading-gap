from solution import maxDifference

def test_canonical_example_1():
    # Input: "aaaaabbc"
    # Frequencies: a: 5 (odd), b: 2 (even), c: 1 (odd)
    # Max difference: 5 (odd) - 2 (even) = 3
    assert maxDifference("aaaaabbc") == 3

def test_canonical_example_2():
    # Input: "abcabcab"
    # Frequencies: a: 3 (odd), b: 2 (even), c: 2 (even)
    # Max difference: 3 (odd) - 2 (even) = 1
    assert maxDifference("abcabcab") == 1

def test_minimum_length_string():
    # Input: "aab"
    # Frequencies: a: 2 (even), b: 1 (odd)
    # Max difference: 1 (odd) - 2 (even) = -1, but we need max odd - even
    assert maxDifference("aab") == -1

def test_larger_frequency_difference():
    # Input: "aaaaaaaaabb"
    # Frequencies: a: 9 (odd), b: 2 (even)
    # Max difference: 9 (odd) - 2 (even) = 7
    assert maxDifference("aaaaaaaaabb") == 7

def test_multiple_valid_pairs():
    # Input: "aaabbbccdd"
    # Frequencies: a: 3 (odd), b: 3 (odd), c: 2 (even), d: 2 (even)
    # Max difference: max(3 (odd) - 2 (even), 3 (odd) - 2 (even)) = 1
    assert maxDifference("aaabbbccdd") == 1
