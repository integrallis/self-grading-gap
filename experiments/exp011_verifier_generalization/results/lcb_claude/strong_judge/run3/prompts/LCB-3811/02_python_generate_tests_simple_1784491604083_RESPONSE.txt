def test_canonical_example_1():
    # spec: reverseDegree("abc") -> 148
    # Calculation:
    # 'a' -> 26 * 1 = 26
    # 'b' -> 25 * 2 = 50
    # 'c' -> 24 * 3 = 72
    # Total = 26 + 50 + 72 = 148
    assert reverseDegree("abc") == 148
