from typing import List
from collections import Counter
from itertools import permutations

def totalNumbers(digits: List[int]) -> int:
    count = Counter(digits)
    distinct_even_numbers = set()

    for last_digit in count:
        if last_digit % 2 != 0:
            continue
        # For a valid last digit, form permutations with the other digits
        for first_digit in count:
            if first_digit == 0:
                continue
            for middle_digit in count:
                # Create the number only if first and middle are not equal to last
                if (first_digit != last_digit and middle_digit != last_digit and
                        (first_digit != middle_digit or count[first_digit] > 1)):
                    number = (first_digit, middle_digit, last_digit)
                    distinct_even_numbers.add(number)

    return len(distinct_even_numbers)
