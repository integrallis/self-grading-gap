from typing import List
from collections import Counter

def totalNumbers(digits: List[int]) -> int:
    count = Counter(digits)
    even_digits = [d for d in count if d % 2 == 0]
    total_count = 0

    for first_digit in count:
        if first_digit == 0:
            continue
        for middle_digit in count:
            for last_digit in even_digits:
                # Calculate usage count
                used = Counter([first_digit, middle_digit, last_digit])
                if all(used[d] <= count[d] for d in used):
                    total_count += 1

    return total_count
```
