def test_large_numbers_modulo():
    # Testing with large values that should require modulo operation
    # input [10**9, 10**9, 10**9], k = 2
    # The sum of min and max for all subsequences with at most 2 elements:
    # Each subsequence will contribute a sum of 2*10**9, hence:
    # There are 3 single element subsequences (10**9 + 10**9 + 10**9) +
    # 3 subsequences of 2 elements (2 * 10**9 + 2 * 10**9 + 2 * 10**9 = 6 * 10**9)
    # Total = 9 * 10**9, which is 9000000000
    # Result modulo 10**9 + 7 = 9000000000 % 1000000007 = 999999986
    assert minMaxSums([10**9, 10**9, 10**9], 2) == 999999986

def test_k_equals_length():
    # Testing the case where k equals length of nums
    # input [1, 2, 3], k = 3
    # All subsequences = [1], [2], [3], [1, 2], [1, 3], [2, 3], [1, 2, 3]
    # Their min + max sums: 2 + 4 + 6 + 3 + 4 + 5 + 6 = 28
    assert minMaxSums([1, 2, 3], 3) == 28
