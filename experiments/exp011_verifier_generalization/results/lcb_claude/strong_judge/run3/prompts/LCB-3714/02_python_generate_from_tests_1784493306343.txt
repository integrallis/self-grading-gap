from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    for i in range(1, k + 1):
        # Calculate the contribution of each element as min and max
        for j in range(n):
            # Count how many times nums[j] will be the minimum
            min_contrib = nums[j] * (pow(2, j) * pow(2, n - j - 1) % MOD) % MOD
            # Count how many times nums[j] will be the maximum
            max_contrib = nums[j] * (pow(2, j) * pow(2, n - j - 1) % MOD) % MOD
            total_sum = (total_sum + min_contrib + max_contrib) % MOD
    return total_sum
```
