from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    # Calculate contributions for each element as min and max
    for i in range(n):
        # Each element nums[i] can be a min or max in any subsequence
        # Count how many subsequences can be formed with nums[i] as min
        min_count = (pow(2, i, MOD) * pow(2, n - i - 1, MOD)) % MOD
        # Count how many subsequences can be formed with nums[i] as max
        max_count = (pow(2, i, MOD) * pow(2, n - i - 1, MOD)) % MOD
        # We only want to consider subsequences of sizes up to k
        for size in range(1, k + 1):
            total_sum += (nums[i] * (min_count + max_count)) % MOD
            total_sum %= MOD
    return total_sum

```
