from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = [1] * n  # dp[i] will hold the length of the longest subsequence ending at index i
    max_length = 1

    for i in range(n):
        for j in range(i):
            diff1 = abs(nums[i] - nums[j])  # Calculate the absolute difference
            if dp[j] >= 1:  # We can consider any subsequence of length 1 or greater
                # We find the last difference in the subsequence ending at j
                last_dp = 1
                for k in range(j):
                    if dp[k] >= last_dp:
                        last_dp = dp[k]
                last_diff = abs(nums[j] - nums[j - 1]) if j > 0 else float('inf')
                if diff1 >= last_diff:
                    dp[i] = max(dp[i], dp[j] + 1)
                    max_length = max(max_length, dp[i])
    return max_length
```
