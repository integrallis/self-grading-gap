from typing import List
import pytest
from solution import longestSubsequence


def test_canonical_example_1():
    # spec: nums = [16,6,3] -> 3
    # The longest subsequence is [16, 6, 3] with absolute adjacent differences [10, 3]
    # 10 >= 3, so this is valid
    assert longestSubsequence([16, 6, 3]) == 3


def test_canonical_example_2():
    # spec: nums = [6,5,3,4,2,1] -> 4
    # The longest subsequence is [6, 4, 2, 1] with absolute adjacent differences [2, 2, 1]
    # 2 >= 2 >= 1, so this is valid
    assert longestSubsequence([6, 5, 3, 4, 2, 1]) == 4


def test_canonical_example_3():
    # spec: nums = [10,20,10,19,10,20] -> 5
    # The longest subsequence is [10, 20, 10, 19, 10] with absolute adjacent differences [10, 10, 9, 9]
    # 10 >= 10 >= 9 >= 9, so this is valid
    assert longestSubsequence([10, 20, 10, 19, 10, 20]) == 5


def test_minimum_length_array():
    # spec: 2 <= nums.length, minimum valid input is length 2
    # Any subsequence of length 1 or 2 is valid (no constraint on first difference)
    # For [1, 2], the subsequence [1, 2] has length 2
    assert longestSubsequence([1, 2]) == 2


def test_all_same_elements():
    # All elements the same: [5, 5, 5]
    # Subsequence [5, 5, 5] has differences [0, 0]
    # 0 >= 0, so length is 3
    assert longestSubsequence([5, 5, 5]) == 3


def test_strictly_increasing_differences():
    # nums = [1, 2, 4, 7]
    # Subsequence [1, 2, 4, 7] has differences [1, 2, 3]
    # 1 >= 2 is false, so we can't use all elements
    # Best subsequence could be [1, 2] (diff [1]) or [2, 4] (diff [2]) or any pair, length 2
    assert longestSubsequence([1, 2, 4, 7]) == 2


def test_strictly_decreasing_differences():
    # nums = [1, 11, 14, 15]
    # Subsequence [1, 11, 14, 15] has differences [10, 3, 1]
    # 10 >= 3 >= 1, so all elements can be used, length 4
    assert longestSubsequence([1, 11, 14, 15]) == 4


def test_alternating_pattern():
    # nums = [1, 10, 2, 9, 3]
    # Subsequence [1, 10, 2, 9] has differences [9, 8, 7]
    # 9 >= 8 >= 7, so length 4
    assert longestSubsequence([1, 10, 2, 9, 3]) == 4


def test_single_large_jump_then_small():
    # nums = [1, 100, 99]
    # Subsequence [1, 100, 99] has differences [99, 1]
    # 99 >= 1, so length 3
    assert longestSubsequence([1, 100, 99]) == 3


def test_constant_difference():
    # nums = [10, 20, 30, 40]
    # Subsequence [10, 20, 30, 40] has differences [10, 10, 10]
    # 10 >= 10 >= 10, so length 4
    assert longestSubsequence([10, 20, 30, 40]) == 4
