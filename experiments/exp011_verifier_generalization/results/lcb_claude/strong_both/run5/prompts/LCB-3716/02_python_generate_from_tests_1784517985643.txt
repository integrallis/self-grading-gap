from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    # dp[i] will hold the length of the longest subsequence ending at index i
    dp = [1] * n
    # This will store the maximum length found
    max_length = 1

    # Iterate through the nums array
    for i in range(n):
        # Check pairs (j, i) with j < i
        for j in range(i):
            # Calculate the absolute difference
            diff1 = abs(nums[i] - nums[j])
            # If current value can extend the subsequence
            if dp[j] > 1:
                # Look for the last difference in the subsequence
                last_diff = abs(nums[j] - nums[j - 1]) if j > 0 else float('inf')
                # Compare with the last difference
                if diff1 >= last_diff:
                    dp[i] = max(dp[i], dp[j] + 1)
                    max_length = max(max_length, dp[i])

    return max_length

```
