```python
from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = [1] * n  # dp[i] tracks the length of the longest subsequence ending at index i
    last_diff = [[0] * (n) for _ in range(n)]  # last_diff[i][j] stores the last difference for subsequence ending at j with the previous element at i
    max_length = 1

    for i in range(n):
        for j in range(i):
            diff = abs(nums[i] - nums[j])  # Calculate the absolute difference between nums[i] and nums[j]
            if dp[j] >= 1:  # Only consider valid subsequences
                valid = True
                # Check the last difference in the subsequence ending at j
                for k in range(1, j + 1):
                    if last_diff[j][k] and last_diff[j][k] < diff:
                        valid = True
                    else:
                        valid = False
                        break
                if valid:
                    dp[i] = max(dp[i], dp[j] + 1)
                    max_length = max(max_length, dp[i])
                last_diff[j][i] = diff
    return max_length
```
