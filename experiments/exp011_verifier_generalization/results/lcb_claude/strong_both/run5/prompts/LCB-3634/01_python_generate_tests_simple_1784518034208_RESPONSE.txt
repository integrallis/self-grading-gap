from solution import calculateScore


def test_canonical_example_1():
    # spec: s = "aczzx" -> 5
    # i=0: 'a', no mirror match before
    # i=1: 'c', no mirror match before
    # i=2: 'z', mirror is 'a' at j=0 (unmarked), mark both, score += 2-0 = 2
    # i=3: 'z', mirror is 'a' at j=0 (marked), no unmarked match, skip
    # i=4: 'x', mirror is 'c' at j=1 (unmarked), mark both, score += 4-1 = 3
    # total = 2 + 3 = 5
    assert calculateScore("aczzx") == 5


def test_canonical_example_2():
    # spec: s = "abcdef" -> 0
    # No character has its mirror before it
    assert calculateScore("abcdef") == 0


def test_single_character():
    # Single character has no j < i, so score is 0
    assert calculateScore("a") == 0


def test_mirror_pair_adjacent():
    # s = "az": i=0 'a' (no j), i=1 'z' (mirror 'a' at j=0), score = 1-0 = 1
    assert calculateScore("az") == 1


def test_mirror_pair_separated():
    # s = "abz": i=0 'a', i=1 'b', i=2 'z' (mirror 'a' at j=0), score = 2-0 = 2
    assert calculateScore("abz") == 2


def test_multiple_same_char_finds_closest():
    # s = "azza": 
    # i=0 'a', i=1 'z' (mirror 'a' at j=0), score = 1, mark 0,1
    # i=2 'z' (mirror 'a', j=0 marked, no match), i=3 'a' (mirror 'z' at j=1 marked, no match)
    # score = 1
    assert calculateScore("azza") == 1


def test_closest_unmarked_index():
    # s = "aaaz": i=0 'a', i=1 'a', i=2 'a', i=3 'z' (mirror 'a')
    # closest unmarked j with 'a' is j=2, score = 3-2 = 1
    assert calculateScore("aaaz") == 1


def test_two_pairs():
    # s = "azby": 
    # i=0 'a', i=1 'z' (mirror 'a' at j=0), score = 1, mark 0,1
    # i=2 'b', i=3 'y' (mirror 'b' at j=2), score += 3-2 = 1
    # total = 2
    assert calculateScore("azby") == 2


def test_no_mirrors():
    # s = "aaa": all same character, 'a' mirrors to 'z', no 'z' present
    assert calculateScore("aaa") == 0


def test_reverse_alphabet_pairs():
    # s = "zyxwvu": all pairs are mirrors but in reverse order (no j < i matches)
    assert calculateScore("zyxwvu") == 0


def test_forward_alphabet_pairs():
    # s = "abcxyz": 
    # i=3 'x' (mirror 'c' at j=2), score = 3-2 = 1, mark 2,3
    # i=4 'y' (mirror 'b' at j=1), score += 4-1 = 3
    # i=5 'z' (mirror 'a' at j=0), score += 5-0 = 5
    # total = 1 + 3 + 5 = 9
    assert calculateScore("abcxyz") == 9


def test_middle_mirror():
    # 'm' and 'n' are mirrors (m is 13th, n is 14th from start; 'm' is 14th from end, 'n' is 13th from end)
    # s = "mn": i=1 'n' (mirror 'm' at j=0), score = 1
    assert calculateScore("mn") == 1


def test_repeated_mirrors_interleaved():
    # s = "azazaz":
    # i=0 'a', i=1 'z' (mirror 'a' at j=0), score = 1, mark 0,1
    # i=2 'a', i=3 'z' (mirror 'a' at j=2), score += 3-2 = 1
    # i=4 'a', i=5 'z' (mirror 'a' at j=4), score += 5-4 = 1
    # total = 1 + 1 + 1 = 3
    assert calculateScore("azazaz") == 3
