from typing import List


def maxFrequency(nums: List[int], k: int) -> int:
    max_freq = 0
    n = len(nums)
    # Iterate over all possible contiguous subarrays
    for i in range(n):
        for j in range(i, n):
            # Calculate the delta needed to make the subarray elements k
            delta = k - nums[i]
            # Count how many elements in the entire array can become k
            count = 0
            for m in range(n):
                if m >= i and m <= j:
                    # Apply delta to the current element in the subarray
                    if nums[m] + delta == k:
                        count += 1
                else:
                    # Elements outside the subarray remain unchanged
                    if nums[m] == k:
                        count += 1
            max_freq = max(max_freq, count)
    return max_freq

```
