def test_canonical_example_1():
    # spec: nums = [1,2,3,4,5,6], k = 1

    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4

    # nums = [5], k = 5
    # The element is already k, frequency is 1

    # nums = [3], k = 5
    # After adding 2 to nums[0..0], we get [5], frequency is 1

    # We can convert all elements to 5 by adding 4 to nums[0..2], frequency is 3
    assert maxFrequency([1, 2, 3], 5) == 3

    # nums = [5, 1, 5, 2], k = 5
    # Initially 2 elements are k. We can add 3 to nums[1..3] to get [5, 4, 8, 5] - still 2
    # Or add 4 to nums[1..1] to get [5, 5, 5, 2] - frequency is 3
    assert maxFrequency([5, 1, 5, 2], 5) == 3

    # nums = [10, 20, 30], k = 15
    # Add 5 to nums[0..0] to get [15, 20, 30], frequency is 1
    # Add -5 to nums[1..1] to get [10, 15, 30], frequency is 1
    # Add -15 to nums[2..2] to get [10, 20, 15], frequency is 1
    # Add 5 to nums[0..1] to get [15, 25, 30], frequency is 1
    # Add -5 to nums[1..2] to get [10, 15, 25], frequency is 1
    # Add -15 to nums[0..2] to get [-5, 5, 15], frequency is 1
    assert maxFrequency([10, 20, 30], 15) == 1

    assert maxFrequency([10, 5, 10, 3, 10], 10) == 4
