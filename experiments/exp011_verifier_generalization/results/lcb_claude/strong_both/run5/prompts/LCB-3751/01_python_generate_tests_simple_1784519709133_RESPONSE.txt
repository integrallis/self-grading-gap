from typing import List
import pytest
from solution import maxFrequency


def test_canonical_example_1():
    # spec: nums = [1,2,3,4,5,6], k = 1
    # After adding -5 to nums[2..5], 1 has a frequency of 2 in [1, 2, -2, -1, 0, 1]
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2


def test_canonical_example_2():
    # spec: nums = [10,2,3,4,5,5,4,3,2,2], k = 10
    # After adding 8 to nums[1..9], 10 has a frequency of 4 in [10, 10, 11, 12, 13, 13, 12, 11, 10, 10]
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4


def test_single_element_already_k():
    # nums = [5], k = 5
    # The element is already k, frequency is 1
    assert maxFrequency([5], 5) == 1


def test_single_element_not_k():
    # nums = [3], k = 5
    # After adding 2 to nums[0..0], we get [5], frequency is 1
    assert maxFrequency([3], 5) == 1


def test_all_elements_already_k():
    # nums = [7, 7, 7], k = 7
    # All elements are already k, frequency is 3
    assert maxFrequency([7, 7, 7], 7) == 3


def test_no_elements_initially_k():
    # nums = [1, 2, 3], k = 5
    # We can convert all elements to 5 by adding 4 to nums[0..2], frequency is 3
    assert maxFrequency([1, 2, 3], 5) == 3


def test_some_elements_initially_k():
    # nums = [5, 1, 5, 2], k = 5
    # Initially 2 elements are k. We can add 3 to nums[1..3] to get [5, 4, 8, 5] - still 2
    # Or add 4 to nums[1..1] to get [5, 5, 5, 2] - frequency is 3
    assert maxFrequency([5, 1, 5, 2], 5) == 3


def test_convert_contiguous_subarray():
    # nums = [10, 20, 30], k = 15
    # Add 5 to nums[0..0] to get [15, 20, 30], frequency is 1
    # Add -5 to nums[1..1] to get [10, 15, 30], frequency is 1
    # Add -15 to nums[2..2] to get [10, 20, 15], frequency is 1
    # Add 5 to nums[0..1] to get [15, 25, 30], frequency is 1
    # Add -5 to nums[1..2] to get [10, 15, 25], frequency is 1
    # Add -15 to nums[0..2] to get [-5, 5, 15], frequency is 1
    assert maxFrequency([10, 20, 30], 15) == 1


def test_two_elements_same():
    # nums = [3, 3], k = 5
    # Add 2 to nums[0..1] to get [5, 5], frequency is 2
    assert maxFrequency([3, 3], 5) == 2


def test_two_elements_different():
    # nums = [3, 7], k = 5
    # Add 2 to nums[0..0] to get [5, 7], frequency is 1
    # Add -2 to nums[1..1] to get [3, 5], frequency is 1
    # Add 2 to nums[0..1] to get [5, 9], frequency is 1
    assert maxFrequency([3, 7], 5) == 1


def test_multiple_k_values_scattered():
    # nums = [10, 5, 10, 3, 10], k = 10
    # Initially 3 elements are 10
    # Add 5 to nums[1..1] to get [10, 10, 10, 3, 10], frequency is 4
    # Add 7 to nums[3..3] to get [10, 5, 10, 10, 10], frequency is 4
    assert maxFrequency([10, 5, 10, 3, 10], 10) == 4


def test_optimal_subarray_in_middle():
    # nums = [1, 10, 10, 10, 1], k = 10
    # Initially 3 elements are 10
    # Add 9 to nums[0..0] to get [10, 10, 10, 10, 1], frequency is 4
    # Add 9 to nums[4..4] to get [1, 10, 10, 10, 10], frequency is 4
    # Add 9 to nums[0..4] to get [10, 19, 19, 19, 10], frequency is 2
    assert maxFrequency([1, 10, 10, 10, 1], 10) == 4
