from typing import List
import pytest
from solution import minMaxWeight


def test_canonical_example_1():
    # spec: n=5, edges=[[1,0,1],[2,0,2],[3,0,1],[4,3,1],[2,1,1]], threshold=2 -> 1
    # Remove edge 2->0. Maximum weight among remaining edges is 1.
    assert minMaxWeight(5, [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]], 2) == 1


def test_canonical_example_2():
    # spec: n=5, edges=[[0,1,1],[0,2,2],[0,3,1],[0,4,1],[1,2,1],[1,4,1]], threshold=1 -> -1
    # Impossible to reach node 0 from node 2.
    assert minMaxWeight(5, [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]], 1) == -1


def test_canonical_example_3():
    # spec: n=5, edges=[[1,2,1],[1,3,3],[1,4,5],[2,3,2],[3,4,2],[4,0,1]], threshold=1 -> 2
    # Remove edges 1->3 and 1->4. Maximum weight among remaining edges is 2.
    assert minMaxWeight(5, [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]], 1) == 2


def test_canonical_example_4():
    # spec: n=5, edges=[[1,2,1],[1,3,3],[1,4,5],[2,3,2],[4,0,1]], threshold=1 -> -1
    # Impossible to satisfy all conditions.
    assert minMaxWeight(5, [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]], 1) == -1


def test_simple_two_nodes_direct_path():
    # n=2, one edge from node 1 to node 0 with weight 5, threshold=1
    # Node 1 can reach node 0. Maximum weight is 5.
    assert minMaxWeight(2, [[1, 0, 5]], 1) == 5


def test_simple_two_nodes_no_path():
    # n=2, one edge from node 0 to node 1 with weight 5, threshold=1
    # Node 1 cannot reach node 0. Return -1.
    assert minMaxWeight(2, [[0, 1, 5]], 1) == -1


def test_three_nodes_linear_path():
    # n=3, edges: 1->0 (weight 3), 2->1 (weight 2), threshold=1
    # Path: 2->1->0. Maximum weight is 3.
    assert minMaxWeight(3, [[1, 0, 3], [2, 1, 2]], 1) == 3


def test_multiple_edges_select_smaller_weight():
    # n=3, edges: 1->0 (weight 10), 1->0 (weight 1), 2->1 (weight 5), threshold=1
    # Can use the edge with weight 1 from 1->0. Maximum weight is 5.
    assert minMaxWeight(3, [[1, 0, 10], [1, 0, 1], [2, 1, 5]], 1) == 5


def test_threshold_constraint_forces_removal():
    # n=3, edges: 1->0 (weight 1), 1->2 (weight 2), 1->0 (weight 3), 2->0 (weight 4), threshold=1
    # Node 1 has 3 outgoing edges but threshold=1. Must keep only 1 edge from node 1.
    # Node 2 must reach 0, so keep 2->0 (weight 4).
    # Node 1 must reach 0, so keep 1->0 (weight 1 or 3).
    # Maximum weight would be 4 (from 2->0) if we keep 1->0 with weight 1.
    assert minMaxWeight(3, [[1, 0, 1], [1, 2, 2], [1, 0, 3], [2, 0, 4]], 1) == 4


def test_all_nodes_direct_to_zero():
    # n=4, edges: 1->0 (weight 2), 2->0 (weight 3), 3->0 (weight 1), threshold=1
    # All nodes can directly reach 0. Maximum weight is 3.
    assert minMaxWeight(4, [[1, 0, 2], [2, 0, 3], [3, 0, 1]], 1) == 3


def test_threshold_allows_multiple_outgoing():
    # n=4, edges: 1->0 (weight 5), 1->2 (weight 1), 2->0 (weight 2), 3->0 (weight 3), threshold=2
    # Node 1 can have 2 outgoing edges. Keep 1->2 and 1->0. Maximum weight is 5.
    # Or keep 1->2 only (and 2->0). Maximum weight is 2.
    assert minMaxWeight(4, [[1, 0, 5], [1, 2, 1], [2, 0, 2], [3, 0, 3]], 2) == 3


def test_complex_graph_with_cycle():
    # n=4, edges forming a cycle with path to 0
    # 1->2 (weight 1), 2->3 (weight 2), 3->1 (weight 3), 3->0 (weight 4), threshold=1
    # All nodes 1, 2, 3 must reach 0. Path: 1->2->3->0. Maximum weight is 4.
    assert minMaxWeight(4, [[1, 2, 1], [2, 3, 2], [3, 1, 3], [3, 0, 4]], 1) == 4


def test_impossible_disconnected_component():
    # n=3, edges: 1->2 (weight 1), threshold=1
    # Node 1 and 2 cannot reach node 0. Return -1.
    assert minMaxWeight(3, [[1, 2, 1]], 1) == -1


def test_node_zero_with_outgoing_edges_only():
    # n=3, edges: 0->1 (weight 1), 0->2 (weight 2), threshold=2
    # Nodes 1 and 2 cannot reach node 0. Return -1.
    assert minMaxWeight(3, [[0, 1, 1], [0, 2, 2]], 2) == -1
