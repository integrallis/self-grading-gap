from typing import List
import pytest
from solution import minimumPairRemoval


def test_canonical_example_1():
    # spec: [5,2,3,1] -> 2
    # (3,1) sum=4 -> [5,2,4]
    # (2,4) sum=6 -> [5,6]
    # 2 operations to make non-decreasing
    assert minimumPairRemoval([5, 2, 3, 1]) == 2


def test_canonical_example_2():
    # spec: [1,2,2] -> 0
    # already non-decreasing
    assert minimumPairRemoval([1, 2, 2]) == 0


def test_single_element():
    # single element is already non-decreasing
    assert minimumPairRemoval([5]) == 0


def test_two_elements_already_sorted():
    # [1, 2] is already non-decreasing
    assert minimumPairRemoval([1, 2]) == 0


def test_two_elements_need_merge():
    # [2, 1] is not non-decreasing
    # only pair (2,1) sum=3 -> [3]
    # 1 operation
    assert minimumPairRemoval([2, 1]) == 1


def test_all_same_elements():
    # [3, 3, 3, 3] is already non-decreasing
    assert minimumPairRemoval([3, 3, 3, 3]) == 0


def test_strictly_increasing():
    # [1, 2, 3, 4] is already non-decreasing
    assert minimumPairRemoval([1, 2, 3, 4]) == 0


def test_strictly_decreasing():
    # [4, 3, 2, 1] requires merging
    # (2,1) sum=3 -> [4,3,3]
    # (3,3) sum=6 (leftmost if tied) -> [4,6]
    # 2 operations
    assert minimumPairRemoval([4, 3, 2, 1]) == 2


def test_negative_numbers():
    # [-5, -3, -1] is already non-decreasing
    assert minimumPairRemoval([-5, -3, -1]) == 0


def test_negative_numbers_need_merge():
    # [1, -1, 2] is not non-decreasing
    # (1,-1) sum=0 -> [0,2]
    # already non-decreasing, 1 operation
    assert minimumPairRemoval([1, -1, 2]) == 1


def test_mixed_positive_negative():
    # [-1, 3, 2] is not non-decreasing
    # (3,2) sum=5 -> [-1,5]
    # already non-decreasing, 1 operation
    assert minimumPairRemoval([-1, 3, 2]) == 1


def test_zero_in_array():
    # [0, 1, 2] is already non-decreasing
    assert minimumPairRemoval([0, 1, 2]) == 0


def test_leftmost_when_tied():
    # [2, 1, 2, 1] 
    # (2,1) and (2,1) both sum to 3, choose leftmost
    # (2,1) at index 0-1 -> [3,2,1]
    # (2,1) at index 1-2 sum=3 -> [3,3]
    # 2 operations
    assert minimumPairRemoval([2, 1, 2, 1]) == 2
