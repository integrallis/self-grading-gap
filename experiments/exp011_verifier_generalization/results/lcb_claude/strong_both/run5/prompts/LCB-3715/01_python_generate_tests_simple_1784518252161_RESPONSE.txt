from typing import List
import pytest
from solution import maximumCoins


def test_canonical_example_1():
    # spec: coins = [[8,10,1],[1,3,2],[5,6,4]], k = 4 -> 10
    # Selecting bags at positions [3, 4, 5, 6] gives: 2 + 0 + 4 + 4 = 10
    assert maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) == 10


def test_canonical_example_2():
    # spec: coins = [[1,10,3]], k = 2 -> 6
    # Selecting bags at positions [1, 2] gives: 3 + 3 = 6
    assert maximumCoins([[1, 10, 3]], 2) == 6


def test_single_segment_k_equals_segment_length():
    # One segment [1, 5, 2], k = 5 covers entire segment: 2*5 = 10
    assert maximumCoins([[1, 5, 2]], 5) == 10


def test_single_segment_k_less_than_segment_length():
    # One segment [1, 10, 5], k = 3 can take any 3 consecutive: 5*3 = 15
    assert maximumCoins([[1, 10, 5]], 3) == 15


def test_single_segment_k_greater_than_segment_length():
    # One segment [5, 7, 3] (length 3), k = 5
    # Best window includes all of segment: 3*3 = 9 (positions 5,6,7 and 2 others with 0 coins)
    assert maximumCoins([[5, 7, 3]], 5) == 9


def test_multiple_segments_k_spans_gap():
    # Two segments [1, 2, 5] and [5, 6, 3], k = 4
    # Window [2, 3, 4, 5] gives: 5 + 0 + 0 + 3 = 8
    # Window [3, 4, 5, 6] gives: 0 + 0 + 3 + 3 = 6
    # Window [1, 2, 3, 4] gives: 5 + 5 + 0 + 0 = 10
    assert maximumCoins([[1, 2, 5], [5, 6, 3]], 4) == 10


def test_k_equals_one():
    # k = 1, should return maximum coin value in any single bag
    # Segments: [1, 3, 2], [5, 7, 10], max single bag has 10 coins
    assert maximumCoins([[1, 3, 2], [5, 7, 10]], 1) == 10


def test_non_overlapping_segments_different_order():
    # Segments given in different order: [5, 6, 4], [1, 3, 2], [8, 10, 1]
    # Same as canonical example 1, just different input order
    # Best window [3, 4, 5, 6]: 2 + 0 + 4 + 4 = 10
    assert maximumCoins([[5, 6, 4], [1, 3, 2], [8, 10, 1]], 4) == 10


def test_large_k_covers_all_segments():
    # Segments: [1, 2, 3], [5, 6, 4], k = 10
    # Window must include gaps; best is probably [1..10]: 3*2 + 4*2 = 14
    assert maximumCoins([[1, 2, 3], [5, 6, 4]], 10) == 14


def test_window_ending_at_segment_boundary():
    # Segment [10, 15, 5], k = 3
    # Window [13, 14, 15] all within segment: 5*3 = 15
    assert maximumCoins([[10, 15, 5]], 3) == 15


def test_window_starting_before_all_segments():
    # Segment [10, 12, 7], k = 5
    # Window [8, 9, 10, 11, 12] gives: 0 + 0 + 7 + 7 + 7 = 21
    assert maximumCoins([[10, 12, 7]], 5) == 21


def test_multiple_segments_no_overlap_best_within_one():
    # Segments [1, 3, 1], [10, 15, 10], k = 4
    # Best window entirely in second segment: 10*4 = 40
    assert maximumCoins([[1, 3, 1], [10, 15, 10]], 4) == 40


def test_window_spans_multiple_segments():
    # Segments [1, 2, 5], [3, 4, 3], [5, 6, 7], k = 6
    # Window [1, 2, 3, 4, 5, 6]: 5*2 + 3*2 + 7*2 = 30
    assert maximumCoins([[1, 2, 5], [3, 4, 3], [5, 6, 7]], 6) == 30
