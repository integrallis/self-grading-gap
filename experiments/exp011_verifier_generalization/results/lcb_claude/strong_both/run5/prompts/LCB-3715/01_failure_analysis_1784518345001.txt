from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a list to hold the number of coins at each position
    max_position = 0
    for l, r, c in coins:
        max_position = max(max_position, r)

    coins_count = [0] * (max_position + 1)

    # Fill the coins_count array based on the input segments
    for l, r, c in coins:
        for i in range(l, r + 1):
            coins_count[i] += c

    # Now we find the maximum coins we can collect in any k consecutive bags
    max_coins = 0
    current_coins = sum(coins_count[1:k + 1])  # Initialize with the first k elements
    max_coins = current_coins

    for i in range(1, max_position - k + 2):  # Slide the window
        current_coins -= coins_count[i]  # Remove the leftmost element
        current_coins += coins_count[i + k - 1]  # Add the new rightmost element
        max_coins = max(max_coins, current_coins)

    return max_coins
