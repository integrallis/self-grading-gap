```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a list to hold the number of coins at each position
    if not coins:
        return 0
    min_position = min(l for l, r, c in coins)
    max_position = max(r for l, r, c in coins)
    offset = min_position
    coins_count = [0] * (max_position - min_position + 1)

    # Fill the coins_count array based on the input segments
    for l, r, c in coins:
        for i in range(l, r + 1):
            coins_count[i - offset] += c

    # Now we find the maximum coins we can collect in any k consecutive bags
    max_coins = 0

    # Slide the window over all possible starting positions
    for start in range(min_position - k + 1, max_position + 1):
        window_sum = sum(coins_count[max(0, i - offset)] if min_position <= i <= max_position else 0 for i in range(start, start + k))
        max_coins = max(max_coins, window_sum)

    return max_coins

```
