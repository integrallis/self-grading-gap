import pytest
from solution import hasSpecialSubstring


def test_canonical_example_1():
    # spec: s = "aaabaaa", k = 3 -> true
    # The substring s[4..6] == "aaa" satisfies all conditions
    assert hasSpecialSubstring("aaabaaa", 3) is True


def test_canonical_example_2():
    # spec: s = "abc", k = 2 -> false
    # No substring of length 2 consists of one distinct character
    assert hasSpecialSubstring("abc", 2) is False


def test_single_character_string_k_equals_1():
    # s = "a", k = 1 -> true
    # Substring "a" at index 0 has length 1, all same char, no neighbors
    assert hasSpecialSubstring("a", 1) is True


def test_repeated_character_at_start():
    # s = "aaa", k = 2 -> true
    # Substring "aa" at index 0-1 has length 2, all same char, no char before,
    # char after is 'a' but we only take length k=2 so after is 'a' which is same
    # Wait, let me reconsider: s[0:2] = "aa", char after at s[2] = 'a' which is same
    # So this should be false. But s = "aaa", k = 3: s[0:3] = "aaa", no char after, 
    # no char before -> true
    assert hasSpecialSubstring("aaa", 3) is True


def test_repeated_character_at_end():
    # s = "baaa", k = 3 -> true
    # Substring s[1:4] = "aaa", char before is 'b' (different), no char after
    assert hasSpecialSubstring("baaa", 3) is True


def test_repeated_character_in_middle():
    # s = "baaac", k = 3 -> true
    # Substring s[1:4] = "aaa", char before is 'b' (different), char after is 'c' (different)
    assert hasSpecialSubstring("baaac", 3) is True


def test_repeated_character_surrounded_by_same():
    # s = "aaaaa", k = 3 -> false
    # Any substring of length 3 will have same char before or after
    # s[0:3] = "aaa", no before, after is 'a' (same) -> false
    # s[1:4] = "aaa", before is 'a' (same), after is 'a' (same) -> false
    # s[2:5] = "aaa", before is 'a' (same), no after -> false
    assert hasSpecialSubstring("aaaaa", 3) is False


def test_alternating_characters():
    # s = "ababab", k = 1 -> true
    # s[0:1] = "a", no before, after is 'b' (different) -> true
    assert hasSpecialSubstring("ababab", 1) is True


def test_no_repeated_characters():
    # s = "abcdef", k = 2 -> false
    # No substring of length 2 has all same characters
    assert hasSpecialSubstring("abcdef", 2) is False


def test_k_equals_string_length():
    # s = "aaaa", k = 4 -> true
    # Substring s[0:4] = "aaaa", all same, no before, no after
    assert hasSpecialSubstring("aaaa", 4) is True


def test_multiple_groups_of_repeated_chars():
    # s = "aabbbaa", k = 2 -> true
    # s[2:4] = "bb", before is 'a' (different), after is 'b' (same) -> false
    # s[3:5] = "bb", before is 'b' (same), after is 'a' (different) -> false
    # s[5:7] = "aa", before is 'b' (different), no after -> true
    assert hasSpecialSubstring("aabbbaa", 2) is True


def test_single_isolated_character():
    # s = "abac", k = 1 -> true
    # s[0:1] = "a", no before, after is 'b' (different) -> true
    assert hasSpecialSubstring("abac", 1) is True


def test_repeated_chars_not_isolated():
    # s = "aabaa", k = 2 -> false
    # s[0:2] = "aa", no before, after is 'b' (different) -> true
    assert hasSpecialSubstring("aabaa", 2) is True


def test_exactly_k_repeated_surrounded_by_different():
    # s = "abbbba", k = 4 -> true
    # s[1:5] = "bbbb", before is 'a' (different), after is 'a' (different)
    assert hasSpecialSubstring("abbbba", 4) is True


def test_k_1_with_repeated_start():
    # s = "aab", k = 1 -> true
    # s[1:2] = "a", before is 'a' (same), after is 'b' (different) -> false
    # s[2:3] = "b", before is 'a' (different), no after -> true
    assert hasSpecialSubstring("aab", 1) is True
