from typing import List
from solution import countPartitions


def test_canonical_example_1():
    # spec: nums = [10,10,3,7,6] -> 4
    # Partitions:
    # [10], [10,3,7,6]: 10 - 26 = -16 (even)
    # [10,10], [3,7,6]: 20 - 16 = 4 (even)
    # [10,10,3], [7,6]: 23 - 13 = 10 (even)
    # [10,10,3,7], [6]: 30 - 6 = 24 (even)
    assert countPartitions([10, 10, 3, 7, 6]) == 4


def test_canonical_example_2():
    # spec: nums = [1,2,2] -> 0
    # No partition results in an even sum difference
    assert countPartitions([1, 2, 2]) == 0


def test_canonical_example_3():
    # spec: nums = [2,4,6,8] -> 3
    # All partitions result in an even sum difference
    assert countPartitions([2, 4, 6, 8]) == 3


def test_minimum_length_array():
    # spec: 2 <= n, so minimum is [a, b]
    # Only one partition: [a], [b]
    # difference = a - b
    # if a=2, b=2: diff=0 (even) -> 1
    assert countPartitions([2, 2]) == 1


def test_minimum_length_array_odd_difference():
    # spec: 2 <= n, so minimum is [a, b]
    # if a=1, b=2: diff=-1 (odd) -> 0
    assert countPartitions([1, 2]) == 0


def test_all_same_elements():
    # All elements are the same value
    # [4,4,4,4]: each partition has even difference (0, 0, 0)
    # 3 partitions total
    assert countPartitions([4, 4, 4, 4]) == 3


def test_alternating_odd_even():
    # nums = [1,2,1,2]
    # partition 0: [1], [2,1,2]: 1 - 5 = -4 (even)
    # partition 1: [1,2], [1,2]: 3 - 3 = 0 (even)
    # partition 2: [1,2,1], [2]: 4 - 2 = 2 (even)
    assert countPartitions([1, 2, 1, 2]) == 3


def test_all_odd_numbers():
    # nums = [1,3,5]
    # partition 0: [1], [3,5]: 1 - 8 = -7 (odd)
    # partition 1: [1,3], [5]: 4 - 5 = -1 (odd)
    assert countPartitions([1, 3, 5]) == 0


def test_all_even_numbers():
    # nums = [2,4,6]
    # partition 0: [2], [4,6]: 2 - 10 = -8 (even)
    # partition 1: [2,4], [6]: 6 - 6 = 0 (even)
    assert countPartitions([2, 4, 6]) == 2


def test_single_even_single_odd():
    # nums = [2,3]
    # partition 0: [2], [3]: 2 - 3 = -1 (odd)
    assert countPartitions([2, 3]) == 0


def test_mixed_values():
    # nums = [5,10,15,20]
    # partition 0: [5], [10,15,20]: 5 - 45 = -40 (even)
    # partition 1: [5,10], [15,20]: 15 - 35 = -20 (even)
    # partition 2: [5,10,15], [20]: 30 - 20 = 10 (even)
    assert countPartitions([5, 10, 15, 20]) == 3


def test_larger_array_no_even_differences():
    # nums = [1,1,1,1,1]
    # Each partition: odd - even = odd or even - odd = odd
    # partition 0: [1], [1,1,1,1]: 1 - 4 = -3 (odd)
    # partition 1: [1,1], [1,1,1]: 2 - 3 = -1 (odd)
    # partition 2: [1,1,1], [1,1]: 3 - 2 = 1 (odd)
    # partition 3: [1,1,1,1], [1]: 4 - 1 = 3 (odd)
    assert countPartitions([1, 1, 1, 1, 1]) == 0
