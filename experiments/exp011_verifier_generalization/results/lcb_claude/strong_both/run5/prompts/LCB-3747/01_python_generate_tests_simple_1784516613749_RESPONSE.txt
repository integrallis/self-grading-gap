from typing import List
import pytest
from solution import maxAdjacentDistance


def test_canonical_example_1():
    # spec: nums = [1,2,4] -> 3
    # circular array: adjacent pairs are (1,2), (2,4), (4,1)
    # differences: |1-2|=1, |2-4|=2, |4-1|=3
    # maximum is 3
    assert maxAdjacentDistance([1, 2, 4]) == 3


def test_canonical_example_2():
    # spec: nums = [-5,-10,-5] -> 5
    # circular array: adjacent pairs are (-5,-10), (-10,-5), (-5,-5)
    # differences: |-5-(-10)|=5, |-10-(-5)|=5, |-5-(-5)|=0
    # maximum is 5
    assert maxAdjacentDistance([-5, -10, -5]) == 5


def test_two_elements():
    # minimum length is 2
    # only one pair: (10, 20)
    # difference: |10-20| = 10
    assert maxAdjacentDistance([10, 20]) == 10


def test_all_same_elements():
    # all elements are the same
    # all adjacent differences are 0
    assert maxAdjacentDistance([5, 5, 5, 5]) == 0


def test_alternating_positive_negative():
    # nums = [10, -10, 10, -10]
    # adjacent pairs: (10,-10), (-10,10), (10,-10), (-10,10)
    # all differences are |10-(-10)| = 20
    assert maxAdjacentDistance([10, -10, 10, -10]) == 20


def test_increasing_sequence():
    # nums = [1, 2, 3, 4, 5]
    # adjacent pairs: (1,2), (2,3), (3,4), (4,5), (5,1)
    # differences: 1, 1, 1, 1, |5-1|=4
    # maximum is 4 (circular wrap)
    assert maxAdjacentDistance([1, 2, 3, 4, 5]) == 4


def test_decreasing_sequence():
    # nums = [5, 4, 3, 2, 1]
    # adjacent pairs: (5,4), (4,3), (3,2), (2,1), (1,5)
    # differences: 1, 1, 1, 1, |1-5|=4
    # maximum is 4 (circular wrap)
    assert maxAdjacentDistance([5, 4, 3, 2, 1]) == 4


def test_negative_numbers():
    # nums = [-1, -5, -3]
    # adjacent pairs: (-1,-5), (-5,-3), (-3,-1)
    # differences: |-1-(-5)|=4, |-5-(-3)|=2, |-3-(-1)|=2
    # maximum is 4
    assert maxAdjacentDistance([-1, -5, -3]) == 4


def test_mixed_positive_negative_zero():
    # nums = [0, -10, 10]
    # adjacent pairs: (0,-10), (-10,10), (10,0)
    # differences: |0-(-10)|=10, |-10-10|=20, |10-0|=10
    # maximum is 20
    assert maxAdjacentDistance([0, -10, 10]) == 20


def test_max_at_circular_boundary():
    # nums = [100, 1, 2, -100]
    # adjacent pairs: (100,1), (1,2), (2,-100), (-100,100)
    # differences: 99, 1, 102, 200
    # maximum is 200 (circular wrap between last and first)
    assert maxAdjacentDistance([100, 1, 2, -100]) == 200
