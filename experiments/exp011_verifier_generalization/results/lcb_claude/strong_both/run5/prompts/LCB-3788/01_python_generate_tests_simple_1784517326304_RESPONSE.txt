from typing import List
from solution import maxSum


def test_canonical_example_1():
    # spec: [1,2,3,4,5] -> 15 (select entire array, all unique, sum = 1+2+3+4+5 = 15)
    assert maxSum([1, 2, 3, 4, 5]) == 15


def test_canonical_example_2():
    # spec: [1,1,0,1,1] -> 1 (delete all but one element with value 1, select [1], sum = 1)
    assert maxSum([1, 1, 0, 1, 1]) == 1


def test_canonical_example_3():
    # spec: [1,2,-1,-2,1,0,-1] -> 3 (delete -1,-2, select [2,1] from remaining, sum = 2+1 = 3)
    assert maxSum([1, 2, -1, -2, 1, 0, -1]) == 3


def test_single_element():
    # single element array: must select it, sum = 5
    assert maxSum([5]) == 5


def test_single_element_negative():
    # single element array with negative value: must select it, sum = -10
    assert maxSum([-10]) == -10


def test_all_negative():
    # all negative values: best is to select single largest (least negative), sum = -1
    assert maxSum([-5, -3, -1, -4]) == -1


def test_all_same_positive():
    # all same positive value: select one element, sum = 7
    assert maxSum([7, 7, 7, 7]) == 7


def test_all_same_negative():
    # all same negative value: select one element, sum = -3
    assert maxSum([-3, -3, -3]) == -3


def test_two_elements_unique():
    # two unique elements: select both, sum = 3 + 5 = 8
    assert maxSum([3, 5]) == 8


def test_two_elements_duplicate():
    # two duplicate elements: select one, sum = 4
    assert maxSum([4, 4]) == 4


def test_mixed_with_zero():
    # array with zeros and positives: select positive elements that are unique, sum = 1+2+3 = 6
    assert maxSum([1, 0, 2, 0, 3]) == 6


def test_alternating_duplicates():
    # duplicates alternating with unique: e.g., [5,1,5,2] -> can select [5,1,2] by deleting one 5, sum = 8
    assert maxSum([5, 1, 5, 2]) == 8


def test_negative_followed_by_positive():
    # negative at start, positives after: select positives, sum = 10+20 = 30
    assert maxSum([-5, 10, 20]) == 30


def test_positive_followed_by_negative():
    # positives followed by negatives: select positives, sum = 10+20 = 30
    assert maxSum([10, 20, -5, -10]) == 30


def test_subarray_in_middle():
    # best subarray in middle: [-1, 5, 6, -2] -> select [5,6], sum = 11
    assert maxSum([-1, 5, 6, -2]) == 11


def test_duplicates_with_negatives():
    # duplicates with negatives: [2, -1, 2, -1] -> select [2], sum = 2
    assert maxSum([2, -1, 2, -1]) == 2


def test_long_sequence_unique():
    # longer sequence all unique positives: sum = 1+2+3+4+5+6 = 21
    assert maxSum([1, 2, 3, 4, 5, 6]) == 21


def test_duplicates_at_boundaries():
    # duplicates at start and end: [3, 1, 2, 3] -> can select [3,1,2] or [1,2,3] by deleting one 3, sum = 6
    assert maxSum([3, 1, 2, 3]) == 6
