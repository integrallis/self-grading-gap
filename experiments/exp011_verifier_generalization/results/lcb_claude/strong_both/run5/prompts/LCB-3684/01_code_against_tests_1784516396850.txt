```python
def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]
    prefix_len = len(prefix)
    suffix_len = len(suffix)
    for i in range(len(s) - prefix_len + 1):
        if s[i:i + prefix_len] == prefix:
            # Check if there is enough space for suffix
            if i + prefix_len + suffix_len > len(s):
                continue
            if s[i + prefix_len - 1:i + prefix_len - 1 + suffix_len] == suffix:
                return True
            # Allow for the empty case of suffix
            if suffix_len == 0:
                return True
    return False
```
