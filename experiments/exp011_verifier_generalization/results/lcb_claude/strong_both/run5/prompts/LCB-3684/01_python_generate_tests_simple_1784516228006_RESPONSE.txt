import pytest
from solution import hasMatch


def test_canonical_example_1():
    # spec: s = "leetcode", p = "ee*e" -> true
    # By replacing the '*' with "tcod", the substring "eetcode" matches the pattern.
    assert hasMatch("leetcode", "ee*e") is True


def test_canonical_example_2():
    # spec: s = "car", p = "c*v" -> false
    # There is no substring matching the pattern.
    assert hasMatch("car", "c*v") is False


def test_canonical_example_3():
    # spec: s = "luck", p = "u*" -> true
    # The substrings "u", "uc", and "uck" match the pattern.
    assert hasMatch("luck", "u*") is True


def test_star_at_beginning():
    # Pattern "*abc" should match any substring ending with "abc"
    # s = "xyzabc" has substring "abc" which matches when * is empty string
    assert hasMatch("xyzabc", "*abc") is True


def test_star_at_end():
    # Pattern "abc*" should match any substring starting with "abc"
    # s = "abcxyz" has substring "abc" which matches when * is empty string
    assert hasMatch("abcxyz", "abc*") is True


def test_star_in_middle():
    # Pattern "a*c" should match substrings starting with "a" and ending with "c"
    # s = "abc" has substring "abc" which matches when * is "b"
    assert hasMatch("abc", "a*c") is True


def test_star_only():
    # Pattern "*" can match any single character (non-empty substring)
    # When * is replaced with any character, it forms a non-empty substring
    assert hasMatch("a", "*") is True


def test_no_match_prefix():
    # Pattern "xyz*" requires substring starting with "xyz"
    # s = "abc" has no such substring
    assert hasMatch("abc", "xyz*") is False


def test_no_match_suffix():
    # Pattern "*xyz" requires substring ending with "xyz"
    # s = "abc" has no such substring
    assert hasMatch("abc", "*xyz") is False

    # Pattern "ab*cd" can match "abcd" when * is replaced with empty string
    # s = "abcd" has substring "abcd"
    assert hasMatch("abcd", "ab*cd") is True


def test_star_replaced_with_multiple_chars():
    # Pattern "a*z" should match substring starting with "a" and ending with "z"
    # s = "abcdefz" has substring "abcdefz" which matches when * is "bcdef"
    assert hasMatch("abcdefz", "a*z") is True


def test_pattern_longer_than_string_prefix_suffix():
    # Pattern "abc*xyz" needs at least 6 chars (abc + xyz)
    # s = "ab" is too short
    assert hasMatch("ab", "abc*xyz") is False


def test_multiple_occurrences():
    # Pattern "a*a" should match any substring starting and ending with "a"
    # s = "banana" has substrings "ana", "anana" that match
    assert hasMatch("banana", "a*a") is True


def test_exact_match_with_empty_star():
    # Pattern "test*" with s = "test" matches when * is empty

    assert hasMatch("test", "test*") is True

    # Pattern "bc*de" requires substring starting with "bc" and ending with "de"

def test_no_overlap_possible():
    # Pattern "a*a" requires at least one 'a'
    # s = "bbb" has no 'a' characters
    assert hasMatch("bbb", "a*a") is False


def test_single_char_pattern_with_star():
    # Pattern "a*" should match any substring starting with "a"
    # s = "bac" has substring "a" which matches when * is empty
    assert hasMatch("bac", "a*") is True


def test_single_char_string():
    # Pattern "a*b" needs substring starting with "a" and ending with "b"
    # s = "a" cannot have such substring (needs at least "ab")
    assert hasMatch("a", "a*b") is False


def test_same_char_before_and_after_star():
    # Pattern "x*x" should match substrings starting and ending with "x"
    # s = "x" has substring "x" but it's only one char, need at least two positions
    # Actually "x*x" with * as empty would be "xx", which is not a substring of "x"
    # But we need a substring that starts with x and ends with x
    # "x" itself matches if we think of it as x + (empty) + x where both x's are the same position
    # However, the pattern "x*x" means: x, then anything, then x
    # For a single "x", this would only work if the pattern can match with * being empty and both x's being the same character
    # This is ambiguous. Let's use a clearer case.
    assert hasMatch("xyx", "x*x") is True


def test_pattern_repeated_chars():
    # Pattern "aa*bb" requires substring starting with "aa" and ending with "bb"
    # s = "aabb" has substring "aabb" which matches when * is empty
    assert hasMatch("aabb", "aa*bb") is True
