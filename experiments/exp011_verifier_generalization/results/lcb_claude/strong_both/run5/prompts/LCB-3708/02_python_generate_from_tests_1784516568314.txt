from typing import List

def zigzagTraversal(grid: List[List[int]]) -> List[int]:
    result = []
    rows = len(grid)
    cols = len(grid[0])

    for i in range(rows):
        if i % 2 == 0:
            # Traverse left to right
            for j in range(0, cols, 2):
                result.append(grid[i][j])
        else:
            # Traverse right to left
            for j in range(cols - 1, -1, -2):
                result.append(grid[i][j])

    return result
```
