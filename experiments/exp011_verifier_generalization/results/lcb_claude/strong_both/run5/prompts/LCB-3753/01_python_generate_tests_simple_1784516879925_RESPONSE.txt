import pytest
from solution import maxDifference


def test_canonical_example_1():
    # spec: "aaaaabbc" -> 3
    # 'a' appears 5 times (odd), 'b' appears 2 times (even), 'c' appears 1 time (odd)
    # max difference is max_odd - min_even = 5 - 2 = 3
    assert maxDifference("aaaaabbc") == 3


def test_canonical_example_2():
    # spec: "abcabcab" -> 1
    # 'a' appears 3 times (odd), 'b' appears 3 times (odd), 'c' appears 2 times (even)
    # max difference is max_odd - min_even = 3 - 2 = 1
    assert maxDifference("abcabcab") == 1


def test_single_odd_single_even_minimum_length():
    # "aab" has 'a' appearing 2 times (even) and 'b' appearing 1 time (odd)
    # max difference is 1 - 2 = -1
    assert maxDifference("aab") == -1


def test_high_odd_low_even():
    # "aaaaaabb" has 'a' appearing 6 times (even) and 'b' appearing 2 times (even)
    # Need to find odd frequency: none in this case, so try "aaaaaaabb"
    # "aaaaaaabb" has 'a' appearing 7 times (odd) and 'b' appearing 2 times (even)
    # max difference is 7 - 2 = 5
    assert maxDifference("aaaaaaabb") == 5


def test_multiple_even_multiple_odd():
    # "aabbcccddd" has 'a' appearing 2 times (even), 'b' appearing 2 times (even),
    # 'c' appearing 3 times (odd), 'd' appearing 3 times (odd)
    # max difference is max_odd - min_even = 3 - 2 = 1
    assert maxDifference("aabbcccddd") == 1


def test_large_odd_large_even():
    # "aaaaaabbbbbb" has 'a' appearing 6 times (even) and 'b' appearing 6 times (even)
    # Need odd, so "aaaaaabbbbbbc"
    # 'a' appears 6 times (even), 'b' appears 6 times (even), 'c' appears 1 time (odd)
    # max difference is 1 - 6 = -5
    assert maxDifference("aaaaaabbbbbbc") == -5


def test_all_different_characters():
    # "abc" has all characters appearing 1 time (odd each), no even
    # Need at least one even, so "abcc"
    # 'a' appears 1 time (odd), 'b' appears 1 time (odd), 'c' appears 2 times (even)
    # max difference is max_odd - min_even = 1 - 2 = -1
    assert maxDifference("abcc") == -1


def test_max_difference_with_low_even_high_odd():
    # "aaaaaaaaabbc" has 'a' appearing 9 times (odd), 'b' appearing 2 times (even), 'c' appearing 1 time (odd)
    # max difference is max_odd - min_even = 9 - 2 = 7
    assert maxDifference("aaaaaaaaabbc") == 7


def test_negative_difference():
    # "aabbbbbbbc" has 'a' appearing 2 times (even), 'b' appearing 7 times (odd), 'c' appearing 1 time (odd)
    # max difference is max_odd - min_even = 7 - 2 = 5
    assert maxDifference("aabbbbbbbc") == 5


def test_even_zero_not_counted():
    # "aaabbc" has 'a' appearing 3 times (odd), 'b' appearing 2 times (even), 'c' appearing 1 time (odd)
    # max difference is max_odd - min_even = 3 - 2 = 1
    assert maxDifference("aaabbc") == 1
