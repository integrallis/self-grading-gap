from typing import List
import pytest
from solution import transformArray


def test_canonical_example_1():
    # spec: [4,3,2,1] -> [0,0,1,1]
    # Replace evens (4,2) with 0, odds (3,1) with 1: [0,1,0,1]
    # Sort: [0,0,1,1]
    assert transformArray([4, 3, 2, 1]) == [0, 0, 1, 1]


def test_canonical_example_2():
    # spec: [1,5,1,4,2] -> [0,0,1,1,1]
    # Replace evens (4,2) with 0, odds (1,5,1) with 1: [1,1,1,0,0]
    # Sort: [0,0,1,1,1]
    assert transformArray([1, 5, 1, 4, 2]) == [0, 0, 1, 1, 1]


def test_single_even_number():
    # [2] -> even becomes 0 -> [0]
    assert transformArray([2]) == [0]


def test_single_odd_number():
    # [3] -> odd becomes 1 -> [1]
    assert transformArray([3]) == [1]


def test_all_even_numbers():
    # [2,4,6,8] -> all become 0 -> [0,0,0,0]
    assert transformArray([2, 4, 6, 8]) == [0, 0, 0, 0]


def test_all_odd_numbers():
    # [1,3,5,7] -> all become 1 -> [1,1,1,1]
    assert transformArray([1, 3, 5, 7]) == [1, 1, 1, 1]


def test_mixed_unsorted():
    # [5,2,8,1,3] -> odds (5,1,3) to 1, evens (2,8) to 0 -> [1,0,0,1,1]
    # Sort: [0,0,1,1,1]
    assert transformArray([5, 2, 8, 1, 3]) == [0, 0, 1, 1, 1]


def test_already_sorted_mixed():
    # [1,2,3,4] -> [1,0,1,0] -> sort -> [0,0,1,1]
    assert transformArray([1, 2, 3, 4]) == [0, 0, 1, 1]


def test_large_numbers():
    # [1000,999,500,1] -> evens (1000,500) to 0, odds (999,1) to 1
    # [0,1,0,1] -> sort -> [0,0,1,1]
    assert transformArray([1000, 999, 500, 1]) == [0, 0, 1, 1]


def test_repeated_values():
    # [2,2,3,3] -> evens (2,2) to 0, odds (3,3) to 1
    # [0,0,1,1] -> sort -> [0,0,1,1]
    assert transformArray([2, 2, 3, 3]) == [0, 0, 1, 1]
