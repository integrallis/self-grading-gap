```python
from typing import List

MOD = 10**9 + 7

from math import comb

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    # Sort the numbers to easily identify min and max in subsequences
    nums.sort()
    # Calculate contributions of each element as min and max
    for i in range(n):
        # Each element nums[i] contributes as min in all subsequences of length 1 to k that include it
        contribution_as_min = (nums[i] * comb(n - 1 - i, k - 1)) % MOD
        total_sum += contribution_as_min
        total_sum %= MOD
        # Each element nums[i] contributes as max in all subsequences of length 1 to k that include it
        contribution_as_max = (nums[i] * comb(i, k - 1)) % MOD
        total_sum += contribution_as_max
        total_sum %= MOD
    return total_sum % MOD

```
