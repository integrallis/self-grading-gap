from typing import List
from solution import minMaxSums


def test_canonical_example_1():
    # spec: nums = [1,2,3], k = 2 -> 24
    # Subsequences with at most 2 elements:
    # [1]: min=1, max=1, sum=2
    # [2]: min=2, max=2, sum=4
    # [3]: min=3, max=3, sum=6
    # [1,2]: min=1, max=2, sum=3
    # [1,3]: min=1, max=3, sum=4
    # [2,3]: min=2, max=3, sum=5
    # Total: 2+4+6+3+4+5 = 24
    assert minMaxSums([1, 2, 3], 2) == 24


def test_canonical_example_2():
    # spec: nums = [5,0,6], k = 1 -> 22
    # Subsequences with exactly 1 element (at most 1):
    # [5]: min=5, max=5, sum=10
    # [0]: min=0, max=0, sum=0
    # [6]: min=6, max=6, sum=12
    # Total: 10+0+12 = 22
    assert minMaxSums([5, 0, 6], 1) == 22


def test_canonical_example_3():
    # spec: nums = [1,1,1], k = 2 -> 12
    # Subsequences with at most 2 elements:
    # [1] appears 3 times: each has min=1, max=1, sum=2, total=6
    # [1,1] appears 3 times: each has min=1, max=1, sum=2, total=6
    # Total: 6+6 = 12
    assert minMaxSums([1, 1, 1], 2) == 12


def test_single_element_array():
    # nums = [5], k = 1
    # Only one subsequence: [5] with min=5, max=5, sum=10
    assert minMaxSums([5], 1) == 10


def test_k_equals_array_length():
    # nums = [1,2], k = 2
    # Subsequences: [1], [2], [1,2]
    # [1]: min=1, max=1, sum=2
    # [2]: min=2, max=2, sum=4
    # [1,2]: min=1, max=2, sum=3
    # Total: 2+4+3 = 9
    assert minMaxSums([1, 2], 2) == 9


def test_all_elements_same():
    # nums = [7,7], k = 1
    # Subsequences: [7], [7]
    # Each has min=7, max=7, sum=14
    # Total: 14+14 = 28
    assert minMaxSums([7, 7], 1) == 28


def test_with_zero():
    # nums = [0,1], k = 2
    # Subsequences: [0], [1], [0,1]
    # [0]: min=0, max=0, sum=0
    # [1]: min=1, max=1, sum=2
    # [0,1]: min=0, max=1, sum=1
    # Total: 0+2+1 = 3
    assert minMaxSums([0, 1], 2) == 3


def test_larger_array():
    # nums = [1,2,3,4], k = 1
    # Subsequences with 1 element: [1], [2], [3], [4]
    # [1]: 2, [2]: 4, [3]: 6, [4]: 8
    # Total: 2+4+6+8 = 20
    assert minMaxSums([1, 2, 3, 4], 1) == 20


def test_modulo_operation():
    # Test that result is returned modulo 10^9 + 7
    # Using large values that would require modulo
    # nums = [10**9, 10**9], k = 2
    # Subsequences: [10^9], [10^9], [10^9, 10^9]
    # Each single: min=10^9, max=10^9, sum=2*10^9
    # Pair: min=10^9, max=10^9, sum=2*10^9
    # Total: 2*10^9 + 2*10^9 + 2*10^9 = 6*10^9
    # Modulo 10^9 + 7: 6*10^9 % (10^9 + 7) = 6000000000 % 1000000007 = 4999999972
    MOD = 10**9 + 7
    assert minMaxSums([10**9, 10**9], 2) == (6 * 10**9) % MOD
