from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    # Calculate contributions of each element as min and max
    for i in range(n):
        # Each element nums[i] contributes as a min
        # in (k choose j) subsequences where j is from 0 to k-1
        num_contrib_as_min = 0
        for j in range(k):
            num_contrib_as_min += comb(n-1, j)
            num_contrib_as_min %= MOD
        total_sum += (nums[i] * num_contrib_as_min) % MOD
        total_sum %= MOD
        # Each element nums[i] contributes as a max
        num_contrib_as_max = 0
        for j in range(k):
            num_contrib_as_max += comb(n-1, j)
            num_contrib_as_max %= MOD
        total_sum += (nums[i] * num_contrib_as_max) % MOD
        total_sum %= MOD
    return total_sum
