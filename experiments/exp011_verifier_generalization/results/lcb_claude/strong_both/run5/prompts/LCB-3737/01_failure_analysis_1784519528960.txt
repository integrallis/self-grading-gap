from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize a DP table with 2 rows (current and previous)
    dp = [[0] * 3 for _ in range(2)]

    # Fill the dp for the first house
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the dp for the rest of the houses
    for i in range(1, n):
        for j in range(3):
            # Cost to paint the current house with color j
            dp[i % 2][j] = cost[i][j] + min(
                dp[(i - 1) % 2][(j + 1) % 3],  # Different color than previous house
                dp[(i - 1) % 2][(j + 2) % 3]   # Different color than previous house
            )

        # Handle equidistant constraint
        if i % 2 == 1:  # Only consider even indexed in the dp
            dp[i % 2][0] = min(dp[i % 2][0], dp[(i - 2) % 2][1] + cost[i][0])
            dp[i % 2][1] = min(dp[i % 2][1], dp[(i - 2) % 2][0] + cost[i][1])
            dp[i % 2][2] = min(dp[i % 2][2], dp[(i - 2) % 2][2] + cost[i][2])

    # The answer will be the minimum cost after painting all houses
    return min(dp[(n - 1) % 2])
