from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize a DP table
    dp = [[float('inf')] * 3 for _ in range(n)]

    # Fill the dp for the first house
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the dp for the rest of the houses
    for i in range(1, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(dp[i - 1][k] for k in range(3) if k != j)

    # Check equidistant constraint
    min_cost = float('inf')
    for k in range(n // 2):
        for j in range(3):
            for m in range(3):
                if j != m:
                    min_cost = min(min_cost, dp[k][j] + dp[n - 1 - k][m])

    return min_cost

```
