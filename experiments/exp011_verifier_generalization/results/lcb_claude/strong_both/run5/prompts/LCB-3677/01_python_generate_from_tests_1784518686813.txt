from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    # dp[i][j][k] will store the maximum coins at (i, j) with k neutralizations used
    dp = [[[-float('inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0]  # start at (0, 0) and must take the value here

    for i in range(m):
        for j in range(n):
            for k in range(3):  # k is the number of neutralizations used
                if i == 0 and j == 0:
                    continue
                # Determine the value of the current cell
                if coins[i][j] < 0:
                    # It's a robber
                    if k > 0:
                        current_value = 0  # neutralize this robber
                    else:
                        current_value = coins[i][j]  # robber steals coins
                else:
                    current_value = coins[i][j]  # gain coins

                # Move from the left cell (i, j-1)
                if j > 0:
                    dp[i][j][k] = max(dp[i][j][k], dp[i][j-1][k] + current_value)
                # Move from the top cell (i-1, j)
                if i > 0:
                    dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k] + current_value)
                # Use neutralization from the left cell (i, j-1)
                if j > 0 and k > 0:
                    dp[i][j][k] = max(dp[i][j][k], dp[i][j-1][k-1] + 0)
                # Use neutralization from the top cell (i-1, j)
                if i > 0 and k > 0:
                    dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k-1] + 0)

    # The result will be the maximum coins at the bottom-right corner with up to 2 neutralizations
    return max(dp[m-1][n-1][0], dp[m-1][n-1][1], dp[m-1][n-1][2])

The correct approach: `dp[i][j][k]` should represent "maximum coins reaching (i,j) having used exactly k neutralizations." Each transition should explicitly choose whether to neutralize the current cell (transition from k-1 to k) or take its value (stay at k).
