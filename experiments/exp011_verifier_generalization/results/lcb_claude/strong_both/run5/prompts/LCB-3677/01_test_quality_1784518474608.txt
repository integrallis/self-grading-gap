    # Without neutralization: -1 + -2 + -4 = -7 or -1 + -3 + -4 = -8

    # Or path (0,0) -> (0,1) -> (1,1): -1 + neutralize(-2) + neutralize(-4) = -1
    # Or path (0,0) -> (0,1) -> (1,1): neutralize(-1) + neutralize(-2) + -4 = -4
    # Best: neutralize the two largest negatives on shortest path
    assert maximumAmount([[-1, -2], [-3, -4]]) == -1
