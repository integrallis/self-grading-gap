from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_intervals = []

    # Calculate initial free time intervals
    if startTime[0] > 0:
        free_intervals.append((0, startTime[0]))

    for i in range(1, n):
        free_intervals.append((endTime[i - 1], startTime[i]))

    if endTime[-1] < eventTime:
        free_intervals.append((endTime[-1], eventTime))

    # Initial free time lengths
    free_lengths = [end - start for start, end in free_intervals]

    # If we can reschedule up to k meetings
    if k == 0:
        return max(free_lengths, default=0)

    # To maximize free time, we need to consider the gaps and reschedule meetings
    total_meeting_time = sum(end - start for start, end in zip(startTime, endTime))
    total_event_time = eventTime

    # The maximum free time is the total event time minus occupied meeting time
    max_free_time = total_event_time - total_meeting_time

    # If k is large enough to reschedule all meetings, return the max free time possible
    if k >= n:
        return max(max_free_time, max(free_lengths, default=0))

    # If we cannot fully reschedule, we need to find the best way to move meetings
    # Sort free intervals by their length (descending)
    free_lengths.sort(reverse=True)

    # We can create additional free time by moving meetings
    # The maximum we can gain is by moving the k meetings that contribute the most to free time
    max_gain = sum(free_lengths[:k])

    return max(max_free_time, max_gain)
