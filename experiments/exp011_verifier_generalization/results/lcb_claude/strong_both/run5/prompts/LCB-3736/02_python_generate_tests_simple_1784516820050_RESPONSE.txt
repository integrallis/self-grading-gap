import pytest
from solution import findValidPair


def test_canonical_example_1():
    # spec: "2523533" -> "23"
    # '2' appears 2 times, '3' appears 3 times
    # "23" is the first valid pair from left to right
    assert findValidPair("2523533") == "23"


def test_canonical_example_2():
    # spec: "221" -> "21"
    # '2' appears 2 times, '1' appears 1 time
    # "21" is the first valid pair from left to right
    assert findValidPair("221") == "21"


def test_canonical_example_3():
    # spec: "22" -> ""
    # No valid adjacent pairs (digits must be different)
    assert findValidPair("22") == ""


def test_no_valid_pair_digits_dont_match_count():
    # '1' appears 2 times (not 1), '2' appears 2 times
    # Neither digit satisfies count requirement
    assert findValidPair("1122") == ""


def test_valid_pair_at_start():
    # '1' appears 1 time, '2' appears 2 times
    # "12" at position 0-1 is valid
    assert findValidPair("1223") == "12"


def test_valid_pair_at_end():
    # '3' appears 3 times, '1' appears 1 time
    # "31" at position 3-4 is the first valid pair
    assert findValidPair("3331") == "31"


def test_multiple_valid_pairs_returns_first():
    # '1' appears 1 time, '2' appears 2 times
    # "12" at position 0-1 is first, "21" at position 1-2 is second
    # Should return "12"
    assert findValidPair("122") == "12"


def test_adjacent_same_digits_not_valid():
    # '3' appears 3 times, but "33" has same digits
    # Must find a pair with different digits
    assert findValidPair("333") == ""


def test_digit_count_matches_but_not_adjacent():
    # '1' appears 1 time, '3' appears 1 time (not 3), '2' appears 2 times
    # "13" invalid ('3' count mismatch), "32" invalid ('3' count mismatch), "22" (same digits)
    # No valid pair
    assert findValidPair("1322") == ""


def test_first_digit_valid_second_not():
    # '2' appears 2 times, '3' appears 2 times (not 3)
    # "23" is not valid because '3' doesn't appear 3 times
    assert findValidPair("2233") == ""


def test_second_digit_valid_first_not():
    # '1' appears 2 times (not 1), '2' appears 2 times
    # "12" is not valid because '1' doesn't appear 1 time
    assert findValidPair("1122") == ""


def test_minimum_length_string_with_valid_pair():
    # '1' appears 1 time, '2' appears 2 times
    # Minimum length = 2 according to constraints
    # But we need both to satisfy counts, so minimum practical is 3
    assert findValidPair("122") == "12"


def test_longer_string_with_valid_pair_in_middle():
    # '5' appears 5 times, '1' appears 2 times (not 1)
    # "51" is not valid because '1' doesn't appear 1 time
    # No valid pair
    assert findValidPair("5555511") == ""


def test_all_same_digit():
    # All '4's, no valid pair (digits must be different)
    assert findValidPair("4444") == ""


def test_valid_pair_with_larger_digits():
    # '7' appears 7 times, '1' appears 1 time
    # "71" is a valid pair
    assert findValidPair("77777771") == "71"
