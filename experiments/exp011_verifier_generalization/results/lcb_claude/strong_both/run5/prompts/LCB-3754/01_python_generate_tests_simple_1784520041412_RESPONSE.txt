import pytest
from solution import maxDistance


def test_canonical_example_1():
    # spec: s = "NWSE", k = 1 -> 3
    # Change s[2] from 'S' to 'N' to get "NWNE"
    # Path: (0,0) -> N:(0,1) -> W:(-1,1) -> N:(-1,2) -> E:(0,2)
    # Distances: 1, 2, 3, 2 -> max is 3
    assert maxDistance("NWSE", 1) == 3


def test_canonical_example_2():
    # spec: s = "NSWWEW", k = 3 -> 6
    # Change s[1] from 'S' to 'N', and s[4] from 'E' to 'W' to get "NNWWWW"
    # Max Manhattan distance achieved is 6
    assert maxDistance("NSWWEW", 3) == 6


def test_single_move_north_no_changes():
    # s = "N", k = 0 -> move to (0,1), distance = 1
    assert maxDistance("N", 0) == 1


def test_single_move_south_no_changes():
    # s = "S", k = 0 -> move to (0,-1), distance = 1
    assert maxDistance("S", 0) == 1


def test_single_move_east_no_changes():
    # s = "E", k = 0 -> move to (1,0), distance = 1
    assert maxDistance("E", 0) == 1


def test_single_move_west_no_changes():
    # s = "W", k = 0 -> move to (-1,0), distance = 1
    assert maxDistance("W", 0) == 1


def test_no_changes_allowed():
    # s = "NSEW", k = 0 -> (0,0) -> N:(0,1) -> S:(0,0) -> E:(1,0) -> W:(0,0)
    # Distances: 1, 0, 1, 0 -> max is 1
    assert maxDistance("NSEW", 0) == 1


def test_all_changes_allowed():
    # s = "SSSS", k = 4 -> can change all to go in one direction, e.g., all N
    # Path: N:(0,1) -> N:(0,2) -> N:(0,3) -> N:(0,4)
    # Max distance is 4
    assert maxDistance("SSSS", 4) == 4


def test_two_moves_same_direction():
    # s = "NN", k = 0 -> (0,0) -> N:(0,1) -> N:(0,2)
    # Distances: 1, 2 -> max is 2
    assert maxDistance("NN", 0) == 2


def test_opposing_moves_with_changes():
    # s = "NS", k = 1 -> change S to N to get "NN"
    # Path: N:(0,1) -> N:(0,2)
    # Distances: 1, 2 -> max is 2
    assert maxDistance("NS", 1) == 2


def test_longer_path_no_changes():
    # s = "NNNN", k = 0 -> move north 4 times
    # Path: (0,1), (0,2), (0,3), (0,4)
    # Max distance is 4
    assert maxDistance("NNNN", 0) == 4


def test_changes_exceed_length():
    # s = "SE", k = 5 -> can change both moves
    # Optimal: change to "NN" -> distances 1, 2 -> max is 2
    assert maxDistance("SE", 5) == 2


def test_mixed_moves_with_some_changes():
    # s = "EEWW", k = 2 -> can change 2 moves
    # Could change last 2 W's to E's: "EEEE" -> distances 1, 2, 3, 4 -> max is 4
    assert maxDistance("EEWW", 2) == 4


def test_k_is_zero():
    # s = "NESW", k = 0 -> no changes allowed
    # Path: (0,0) -> N:(0,1) -> E:(1,1) -> S:(1,0) -> W:(0,0)
    # Distances: 1, 2, 1, 0 -> max is 2
    assert maxDistance("NESW", 0) == 2


def test_single_character_with_k():
    # s = "S", k = 1 -> can change to any direction, max distance still 1
    assert maxDistance("S", 1) == 1
