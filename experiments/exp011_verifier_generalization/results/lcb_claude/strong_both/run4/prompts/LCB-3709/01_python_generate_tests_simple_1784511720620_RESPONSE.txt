import pytest
from solution import hasSpecialSubstring


def test_canonical_example_1():
    # spec: s = "aaabaaa", k = 3 -> true
    # The substring s[4..6] == "aaa" satisfies the conditions
    assert hasSpecialSubstring("aaabaaa", 3) is True


def test_canonical_example_2():
    # spec: s = "abc", k = 2 -> false
    # No substring of length 2 that consists of one distinct character
    assert hasSpecialSubstring("abc", 2) is False


def test_substring_at_start_no_char_before():
    # Substring "aaa" at start (positions 0-2), no char before, char after is 'b' (different)
    assert hasSpecialSubstring("aaab", 3) is True


def test_substring_at_end_no_char_after():
    # Substring "bbb" at end (positions 1-3), char before is 'a' (different), no char after
    assert hasSpecialSubstring("abbb", 3) is True


def test_substring_in_middle_both_boundaries_different():
    # Substring "cc" in middle (positions 1-2), char before 'a', char after 'd' (both different)
    assert hasSpecialSubstring("accd", 2) is True


def test_substring_length_1():
    # k=1: any single character is valid if boundaries are different or don't exist
    # "a" at position 0, no char before, char after is 'b' (different)
    assert hasSpecialSubstring("ab", 1) is True


def test_substring_length_1_entire_string():
    # k=1: single character string, no boundaries
    assert hasSpecialSubstring("x", 1) is True


def test_substring_fails_same_char_before():
    # "aaa" at positions 0-2 has char after at position 3 which is 'a' (same)
    assert hasSpecialSubstring("aaaa", 3) is False


def test_substring_fails_same_char_after():
    # Any substring of "aaaa" with k=2 will have same char before or after
    assert hasSpecialSubstring("aaaa", 2) is False


def test_multiple_same_chars_with_valid_substring():
    # "bb" at positions 3-4, char before is 'a' (different), char after is 'c' (different)
    assert hasSpecialSubstring("aabbc", 2) is True


def test_entire_string_is_substring():
    # k equals string length, entire string is one char, no boundaries
    assert hasSpecialSubstring("zzz", 3) is True


def test_alternating_chars_no_valid_substring():
    # No two consecutive same characters
    assert hasSpecialSubstring("ababab", 2) is False


def test_long_run_with_boundaries():
    # "ddd" at positions 2-4, char before is 'c' (different), char after is 'e' (different)
    assert hasSpecialSubstring("ccdddde", 3) is True


def test_k_equals_string_length_mixed_chars():
    # k equals string length but not all same char
    assert hasSpecialSubstring("aab", 3) is False


def test_valid_substring_followed_by_different():
    # "ee" at positions 0-1, no char before, char after is 'f' (different)
    assert hasSpecialSubstring("eef", 2) is True
