from typing import List

def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers
    online_status = [True] * numberOfUsers
    offline_timestamps = [0] * numberOfUsers

    for event in events:
        event_type, timestamp, *args = event
        timestamp = int(timestamp)

        if event_type == "OFFLINE":
            user_id = int(args[0])
            online_status[user_id] = False
            offline_timestamps[user_id] = timestamp + 60
        elif event_type == "MESSAGE":
            for user_id in process_message(args[0], online_status, timestamp, numberOfUsers):
                mentions[user_id] += 1

    return mentions


def process_message(mentions_string: str, online_status: List[bool], timestamp: int, numberOfUsers: int) -> List[int]:
    mentioned_users = set()

    if mentions_string == "ALL":
        mentioned_users = set(range(numberOfUsers))
    elif mentions_string == "HERE":
        mentioned_users = {i for i in range(numberOfUsers) if online_status[i]}
    else:
        tokens = mentions_string.split()
        for token in tokens:
            if token.startswith("id"):
                user_id = int(token[2:])
                if user_id < numberOfUsers:
                    mentioned_users.add(user_id)
            # Add additional conditions if necessary

    # Filter out users who are offline at the time of the message
    return [user for user in mentioned_users if online_status[user] or (timestamp >= offline_timestamps[user])]
