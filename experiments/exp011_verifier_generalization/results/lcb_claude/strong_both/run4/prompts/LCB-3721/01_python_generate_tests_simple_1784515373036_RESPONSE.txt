from typing import List
import pytest
from solution import countMentions


def test_canonical_example_1():
    # Example 1: numberOfUsers = 2, events = [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","71","HERE"]]
    # At timestamp 10: id1 and id0 mentioned -> mentions = [1, 1]
    # At timestamp 11: id0 goes offline
    # At timestamp 71: id0 comes back online (11 + 60 = 71), "HERE" mentions all online users -> mentions = [2, 2]
    assert countMentions(2, [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "71", "HERE"]]) == [2, 2]


def test_canonical_example_2():
    # Example 2: numberOfUsers = 2, events = [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","12","ALL"]]
    # At timestamp 10: id1 and id0 mentioned -> mentions = [1, 1]
    # At timestamp 11: id0 goes offline
    # At timestamp 12: "ALL" mentions all users (including offline) -> mentions = [2, 2]
    assert countMentions(2, [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "12", "ALL"]]) == [2, 2]


def test_canonical_example_3():
    # Example 3: numberOfUsers = 2, events = [["OFFLINE","10","0"],["MESSAGE","12","HERE"]]
    # At timestamp 10: id0 goes offline
    # At timestamp 12: "HERE" mentions only online users (id0 is offline, id1 is online) -> mentions = [0, 1]
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "12", "HERE"]]) == [0, 1]


def test_single_user_all_mention():

    # 1 user online, MESSAGE with HERE -> mentions = [1]
    assert countMentions(1, [["MESSAGE", "5", "HERE"]]) == [1]


def test_single_user_id_mention():
    # 1 user, MESSAGE with id0 -> mentions = [1]
    assert countMentions(1, [["MESSAGE", "5", "id0"]]) == [1]


def test_multiple_id_mentions_same_message():

    assert countMentions(1, [["MESSAGE", "5", "id0 id0"]]) == [2]


def test_user_offline_not_mentioned_by_here():
    # User offline at time of HERE mention is not counted
    # At timestamp 10: id0 goes offline (until 70)
    # At timestamp 20: HERE mentions only online users (id1) -> mentions = [0, 1]
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "20", "HERE"]]) == [0, 1]


def test_user_offline_mentioned_by_all():
    # User offline at time of ALL mention is still counted
    # At timestamp 10: id0 goes offline
    # At timestamp 20: ALL mentions all users -> mentions = [1, 1]
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "20", "ALL"]]) == [1, 1]


def test_user_offline_mentioned_by_id():
    # User offline can still be mentioned by id
    # At timestamp 10: id0 goes offline
    # At timestamp 20: id0 mentioned directly -> mentions = [1, 0]
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "20", "id0"]]) == [1, 0]


def test_user_comes_back_online_after_60_units():
    # User goes offline at timestamp 10, comes back at 70
    # At timestamp 70: HERE should include id0 (exactly at comeback time)
    # At timestamp 10: id0 goes offline
    # At timestamp 70: id0 is back online, HERE mentions both -> mentions = [1, 1]
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "70", "HERE"]]) == [1, 1]


def test_multiple_offline_events_different_users():
    # Multiple users can be offline
    # At timestamp 10: id0 goes offline
    # At timestamp 15: id1 goes offline
    # At timestamp 20: HERE mentions no one (all offline) -> mentions = [0, 0]
    assert countMentions(2, [["OFFLINE", "10", "0"], ["OFFLINE", "15", "1"], ["MESSAGE", "20", "HERE"]]) == [0, 0]


def test_no_mentions():

    # Three MESSAGE events with ALL, each user mentioned 3 times

def test_here_with_all_online():

def test_multiple_users_many_events():
    # Complex scenario with 3 users
    # At timestamp 5: id0, id1, id2 mentioned -> mentions = [1, 1, 1]
    # At timestamp 10: id1 goes offline
    # At timestamp 15: HERE mentions id0 and id2 (id1 offline) -> mentions = [2, 1, 2]
    # At timestamp 20: ALL mentions all -> mentions = [3, 2, 3]
    assert countMentions(3, [
        ["MESSAGE", "5", "id0 id1 id2"],
        ["OFFLINE", "10", "1"],
        ["MESSAGE", "15", "HERE"],
        ["MESSAGE", "20", "ALL"]
    ]) == [3, 2, 3]
