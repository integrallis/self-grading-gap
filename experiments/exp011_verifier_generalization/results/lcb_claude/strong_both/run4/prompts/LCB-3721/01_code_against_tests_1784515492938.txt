from typing import List

def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers
    online_status = [True] * numberOfUsers
    offline_timestamps = [0] * numberOfUsers

    for event in events:
        event_type, timestamp, *args = event
        timestamp = int(timestamp)

        if event_type == "OFFLINE":
            user_id = int(args[0])
            online_status[user_id] = False
            offline_timestamps[user_id] = timestamp + 60
        elif event_type == "MESSAGE":
            process_message(args[0], online_status, timestamp, mentions, numberOfUsers, offline_timestamps)

    return mentions


def process_message(mentions_string: str, online_status: List[bool], timestamp: int, mentions: List[int], numberOfUsers: int, offline_timestamps: List[int]) -> None:
    if mentions_string == "ALL":
        for user_id in range(numberOfUsers):
            mentions[user_id] += 1
    elif mentions_string == "HERE":
        for user_id in range(numberOfUsers):
            if online_status[user_id] and timestamp >= offline_timestamps[user_id]:
                mentions[user_id] += 1
    else:
        tokens = mentions_string.split()
        for token in tokens:
            if token.startswith("id"):
                user_id = int(token[2:])
                if user_id < numberOfUsers:
                    mentions[user_id] += 1
