```python
from typing import List

def maxSum(nums: List[int]) -> int:
    duplicates = {x for x in nums if nums.count(x) > 1}
    max_sum = float('-inf')  # Initialize max_sum to negative infinity
    n = len(nums)
    for i in range(n):
        current_sum = 0
        seen = set()
        for j in range(i, n):
            if nums[j] not in seen and nums[j] not in duplicates:
                seen.add(nums[j])
                current_sum += nums[j]
                max_sum = max(max_sum, current_sum)
            else:
                break  # Stop if we encounter a duplicate
    return max_sum
```
