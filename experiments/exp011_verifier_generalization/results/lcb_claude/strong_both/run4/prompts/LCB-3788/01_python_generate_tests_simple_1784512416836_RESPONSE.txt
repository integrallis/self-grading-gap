from typing import List
import pytest
from solution import maxSum


def test_canonical_example_1():
    # spec: nums = [1,2,3,4,5] -> 15
    # Select the entire array without deleting any element
    assert maxSum([1, 2, 3, 4, 5]) == 15


def test_canonical_example_2():
    # spec: nums = [1,1,0,1,1] -> 1
    # Delete elements to get [1], which has sum 1
    assert maxSum([1, 1, 0, 1, 1]) == 1


def test_canonical_example_3():
    # spec: nums = [1,2,-1,-2,1,0,-1] -> 3
    # Delete some elements and select subarray [2, 1] for sum 3
    assert maxSum([1, 2, -1, -2, 1, 0, -1]) == 3


def test_single_element():
    # Minimum array size is 1, single element is the answer
    # Cannot delete all elements, so return the single element
    assert maxSum([5]) == 5


def test_single_element_negative():
    # Single negative element must be returned
    assert maxSum([-10]) == -10


def test_all_same_positive():
    # All elements are the same positive value
    # Best subarray is a single element
    assert maxSum([3, 3, 3, 3]) == 3


def test_all_same_negative():
    # All elements are the same negative value
    # Best subarray is a single element
    assert maxSum([-5, -5, -5]) == -5


def test_two_unique_elements():
    # Two different elements, can select both
    assert maxSum([10, 20]) == 30


def test_two_same_elements():
    # Two identical elements, can only select one
    assert maxSum([7, 7]) == 7


def test_alternating_positive_negative():
    # Elements alternate between positive and negative
    # Should select only positive elements that don't repeat
    assert maxSum([5, -3, 5, -3]) == 5


def test_all_negative():
    # All negative elements with no duplicates
    # Should select the least negative (maximum) element
    assert maxSum([-1, -2, -3]) == -1


def test_mixed_with_duplicates():
    # Mix of positive and negative with duplicates
    # Should find best unique subarray
    assert maxSum([2, 3, 2, 1, 3]) == 6  # [2, 3, 1] or [3, 2, 1]


def test_zero_in_array():
    # Array contains zero
    assert maxSum([0, 5, 0]) == 5


def test_only_zeros():
    # All elements are zero
    assert maxSum([0, 0, 0]) == 0


def test_large_positive_sum():
    # No duplicates, all positive
    assert maxSum([10, 20, 30, 40]) == 100


def test_negative_then_positive():
    # Negative elements followed by positive
    # Should select positive elements
    assert maxSum([-5, -10, 3, 4, 5]) == 12  # [3, 4, 5]


def test_unique_subarray_in_middle():
    # Best unique subarray is in the middle
    assert maxSum([1, 1, 5, 6, 7, 1, 1]) == 18  # [5, 6, 7]


def test_max_value_element():
    # Test with maximum allowed value
    assert maxSum([100]) == 100


def test_min_value_element():
    # Test with minimum allowed value
    assert maxSum([-100]) == -100


def test_boundary_values():
    # Mix of boundary values
    assert maxSum([100, -100, 100]) == 100
