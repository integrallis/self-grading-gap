from typing import List


def maxSum(nums: List[int]) -> int:
    seen = set()
    max_sum = 0
    current_sum = 0
    for num in nums:
        if num not in seen:
            seen.add(num)
            current_sum += num
        else:
            max_sum = max(max_sum, current_sum)
            # Reset the current sum for the new subarray
            current_sum = num
            seen = {num}
    max_sum = max(max_sum, current_sum)
    return max_sum
```
