from typing import *

def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]
    for i in range(len(s)):
        # Check if the prefix matches
        if s.startswith(prefix, i):
            # Check if the suffix matches after the prefix
            if s[i + len(prefix):].startswith(suffix):
                return True
    return False
```
