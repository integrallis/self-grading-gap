import pytest
from solution import hasMatch


def test_canonical_example_1():
    # spec: s = "leetcode", p = "ee*e" -> true
    # By replacing '*' with "tcod", substring "eetcode" matches
    assert hasMatch("leetcode", "ee*e") is True


def test_canonical_example_2():
    # spec: s = "car", p = "c*v" -> false
    # No substring matching the pattern
    assert hasMatch("car", "c*v") is False


def test_canonical_example_3():
    # spec: s = "luck", p = "u*" -> true
    # Substrings "u", "uc", "uck" match the pattern
    assert hasMatch("luck", "u*") is True


def test_star_at_beginning():
    # pattern "*a" should match any substring ending with 'a'
    # e.g., in "bca", substring "a" matches (star replaced with empty)
    assert hasMatch("bca", "*a") is True


def test_star_at_end():
    # pattern "a*" should match any substring starting with 'a'
    # e.g., in "abc", substring "a", "ab", "abc" match
    assert hasMatch("abc", "a*") is True


def test_star_in_middle():
    # pattern "a*c" should match substrings starting with 'a' and ending with 'c'
    # e.g., in "abcd", substring "abc" matches (star replaced with "b")
    assert hasMatch("abcd", "a*c") is True


def test_only_star():
    # pattern "*" can match any substring of length >= 1
    # since substring is non-empty and star can be replaced with any sequence
    assert hasMatch("a", "*") is True


def test_star_replaced_with_empty_string():
    # pattern "ab*cd" where star is replaced with empty gives "abcd"
    # should match if "abcd" is substring of s
    assert hasMatch("abcd", "ab*cd") is True


def test_no_match_prefix():
    # pattern "x*y" where 'x' doesn't appear in s
    assert hasMatch("abc", "x*y") is False


def test_no_match_suffix():
    # pattern "a*z" where substring starts with 'a' but 'z' doesn't appear after
    assert hasMatch("abc", "a*z") is False


def test_pattern_longer_than_string():
    # pattern "abc*def" (min length 6) but s has length 3
    assert hasMatch("xyz", "abc*def") is False


def test_exact_match_with_empty_star():
    # pattern matches s exactly when star is replaced with empty
    assert hasMatch("hello", "hel*lo") is True


def test_multiple_positions():
    # pattern can match at different positions in s
    # "a*b" should match "ab" which appears in "xaby"
    assert hasMatch("xaby", "a*b") is True


def test_repeated_characters():
    # pattern "a*a" in string "aaa"
    # can match "aa" (star -> empty), "aaa" (star -> "a")
    assert hasMatch("aaa", "a*a") is True


def test_single_char_string_single_char_pattern():
    # s = "a", p = "*", star can be replaced with empty to match "a"? No, empty pattern
    # Actually p = "*" means star can be replaced to form any non-empty substring
    # Since substring must be non-empty, we need the replacement to form at least 1 char
    # Wait, the pattern itself after replacement must be a substring
    # p = "*" -> replace star with "a" -> "a" is substring of "a"
    assert hasMatch("a", "*") is True


def test_no_overlap_needed():
    # pattern "b*d" in "abcde" should match "bcd" (star -> "c")
    assert hasMatch("abcde", "b*d") is True


def test_prefix_and_suffix_dont_exist():
    # pattern "x*y" but neither x nor y exist in s
    assert hasMatch("abc", "x*y") is False
