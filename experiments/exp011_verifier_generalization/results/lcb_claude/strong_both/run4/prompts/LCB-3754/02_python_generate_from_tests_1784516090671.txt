```python
def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    for char in s:
        if char == 'N':
            y += 1
        elif char == 'S':
            y -= 1
        elif char == 'E':
            x += 1
        elif char == 'W':
            x -= 1
        # Calculate current Manhattan distance
        current_distance = abs(x) + abs(y)
        # Update maximum distance
        max_distance = max(max_distance, current_distance)
    # Potential maximum distance with k changes
    max_distance += k
    return max_distance
```
