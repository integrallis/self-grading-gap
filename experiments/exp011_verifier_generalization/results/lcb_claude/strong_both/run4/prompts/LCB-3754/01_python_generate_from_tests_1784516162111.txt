def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    # Store the positions after each move
    positions = [(0, 0)]
    for char in s:
        if char == 'N':
            y += 1
        elif char == 'S':
            y -= 1
        elif char == 'E':
            x += 1
        elif char == 'W':
            x -= 1
        positions.append((x, y))
        # Calculate current Manhattan distance
        current_distance = abs(x) + abs(y)
        # Update maximum distance
        max_distance = max(max_distance, current_distance)

    # Check maximum distances with k changes
    for i in range(len(positions)):
        if i + k < len(positions):
            # Adjust the position by changing up to k moves
            new_x, new_y = positions[i]
            # Calculate distance if we were to change k moves to the optimal direction
            max_distance = max(max_distance, abs(new_x) + abs(new_y) + k)
    return max_distance

```
