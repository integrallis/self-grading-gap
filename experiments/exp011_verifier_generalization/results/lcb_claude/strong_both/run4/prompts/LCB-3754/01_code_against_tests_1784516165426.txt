```python
def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    positions = [(0, 0)]
    for char in s:
        if char == 'N':
            y += 1
        elif char == 'S':
            y -= 1
        elif char == 'E':
            x += 1
        elif char == 'W':
            x -= 1
        positions.append((x, y))
        max_distance = max(max_distance, abs(x) + abs(y))

    # Using k changes to maximize Manhattan distance
    for i in range(len(positions)):
        # Calculate total changes available
        distance_with_changes = abs(positions[i][0]) + abs(positions[i][1]) + k
        max_distance = max(max_distance, distance_with_changes)

    return max_distance

```
