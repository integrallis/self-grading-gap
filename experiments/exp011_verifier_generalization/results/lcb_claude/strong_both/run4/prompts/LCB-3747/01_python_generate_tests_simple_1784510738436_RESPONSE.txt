from solution import maxAdjacentDistance
from typing import List


def test_canonical_example_1():
    # spec: nums = [1,2,4] -> 3
    # Because nums is circular, nums[0] and nums[2] are adjacent.
    # They have the maximum absolute difference of |4 - 1| = 3.
    assert maxAdjacentDistance([1, 2, 4]) == 3


def test_canonical_example_2():
    # spec: nums = [-5,-10,-5] -> 5
    # The adjacent elements nums[0] and nums[1] have the maximum
    # absolute difference of |-5 - (-10)| = 5.
    assert maxAdjacentDistance([-5, -10, -5]) == 5


def test_two_elements_positive():
    # nums = [5, 10]
    # adjacent pairs: (5,10) and circular (10,5)
    # |5 - 10| = 5, |10 - 5| = 5
    # max = 5
    assert maxAdjacentDistance([5, 10]) == 5


def test_two_elements_negative():
    # nums = [-3, -8]
    # adjacent pairs: (-3,-8) and circular (-8,-3)
    # |-3 - (-8)| = 5, |-8 - (-3)| = 5
    # max = 5
    assert maxAdjacentDistance([-3, -8]) == 5


def test_all_same_elements():
    # nums = [7, 7, 7, 7]
    # all adjacent differences are |7 - 7| = 0
    # max = 0
    assert maxAdjacentDistance([7, 7, 7, 7]) == 0


def test_increasing_sequence():
    # nums = [1, 2, 3, 4, 5]
    # consecutive differences: |1-2|=1, |2-3|=1, |3-4|=1, |4-5|=1
    # circular difference: |5-1|=4
    # max = 4
    assert maxAdjacentDistance([1, 2, 3, 4, 5]) == 4


def test_decreasing_sequence():
    # nums = [10, 8, 6, 4, 2]
    # consecutive differences: |10-8|=2, |8-6|=2, |6-4|=2, |4-2|=2
    # circular difference: |2-10|=8
    # max = 8
    assert maxAdjacentDistance([10, 8, 6, 4, 2]) == 8


def test_mixed_positive_negative():
    # nums = [10, -5, 15]
    # adjacent pairs: (10,-5), (-5,15), (15,10) circular
    # |10 - (-5)| = 15, |-5 - 15| = 20, |15 - 10| = 5
    # max = 20
    assert maxAdjacentDistance([10, -5, 15]) == 20


def test_single_peak():
    # nums = [1, 100, 1]
    # adjacent pairs: (1,100), (100,1), (1,1) circular
    # |1 - 100| = 99, |100 - 1| = 99, |1 - 1| = 0
    # max = 99
    assert maxAdjacentDistance([1, 100, 1]) == 99


def test_boundary_values():
    # nums = [-100, 100]
    # adjacent pairs: (-100,100) and circular (100,-100)
    # |-100 - 100| = 200, |100 - (-100)| = 200
    # max = 200
    assert maxAdjacentDistance([-100, 100]) == 200
