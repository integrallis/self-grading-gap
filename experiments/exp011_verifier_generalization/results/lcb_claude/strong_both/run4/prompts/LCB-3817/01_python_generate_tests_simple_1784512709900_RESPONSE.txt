import pytest
from solution import maxContainers


def test_canonical_example_1():
    # Example 1: n=2, w=3, maxWeight=15
    # deck has 2*2=4 cells, each container weighs 3
    # total weight if all loaded: 4*3=12 <= 15
    # so can load all 4 containers
    assert maxContainers(2, 3, 15) == 4


def test_canonical_example_2():
    # Example 2: n=3, w=5, maxWeight=20
    # deck has 3*3=9 cells, each container weighs 5
    # max containers without exceeding maxWeight=20: floor(20/5)=4
    # (4 containers would weigh 20, 5 would weigh 25 > 20)
    assert maxContainers(3, 5, 20) == 4


def test_single_cell_deck():
    # n=1 means 1x1=1 cell deck
    # if w=10 and maxWeight=10, can load 1 container (1*10=10 <= 10)
    assert maxContainers(1, 10, 10) == 1


def test_weight_limit_allows_all_containers():
    # n=4 means 4*4=16 cells
    # if w=2 and maxWeight=100, total weight would be 16*2=32 <= 100
    # so can load all 16 containers
    assert maxContainers(4, 2, 100) == 16


def test_weight_limit_allows_zero_containers():
    # n=5 means 5*5=25 cells, w=10, maxWeight=5
    # floor(5/10)=0, cannot load even one container
    assert maxContainers(5, 10, 5) == 0


def test_weight_limit_allows_partial_containers():
    # n=10 means 10*10=100 cells, w=7, maxWeight=50
    # floor(50/7)=7 containers can be loaded
    assert maxContainers(10, 7, 50) == 7


def test_exact_weight_match():
    # n=3 means 3*3=9 cells, w=5, maxWeight=45
    # floor(45/5)=9, exactly matches deck capacity
    assert maxContainers(3, 5, 45) == 9


def test_large_deck_small_weight_limit():
    # n=100 means 100*100=10000 cells, w=1, maxWeight=50
    # floor(50/1)=50 containers, much less than deck capacity
    assert maxContainers(100, 1, 50) == 50


def test_minimum_inputs():
    # n=1, w=1, maxWeight=1 (all minimum valid values per constraints)
    # 1*1=1 cell, floor(1/1)=1 container
    assert maxContainers(1, 1, 1) == 1


def test_heavy_containers():
    # n=2 means 2*2=4 cells, w=500, maxWeight=1000
    # floor(1000/500)=2 containers (less than deck capacity of 4)
    assert maxContainers(2, 500, 1000) == 2


def test_large_max_weight():
    # n=10 means 10*10=100 cells, w=100, maxWeight=1000000000
    # floor(1000000000/100)=10000000, but deck only has 100 cells
    # so can load all 100 containers
    assert maxContainers(10, 100, 1000000000) == 100
