from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Calculate the contribution of each number as min and max
    for i in range(n):
        # For nums[i] as minimum
        min_count = (1 << min(k, n - 1)) - 1
        max_count = (1 << min(k, n - 1)) - 1

        # Each element can be combined with other elements
        # Count of subsequences where nums[i] is the minimum
        total_sum += nums[i] * min_count
        total_sum %= MOD

        # Count of subsequences where nums[i] is the maximum
        total_sum += nums[i] * max_count
        total_sum %= MOD

    return total_sum % MOD

```
