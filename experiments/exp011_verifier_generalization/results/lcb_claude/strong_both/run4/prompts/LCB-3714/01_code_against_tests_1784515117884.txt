```python
from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Sort the array to easily find min and max in subsequences
    nums.sort()

    # Precompute factorials and inverse factorials for binomial coefficients
    fact = [1] * (n + 1)
    for i in range(2, n + 1):
        fact[i] = fact[i - 1] * i % MOD

    inv_fact = [1] * (n + 1)
    inv_fact[n] = pow(fact[n], MOD - 2, MOD)
    for i in range(n - 1, 0, -1):
        inv_fact[i] = inv_fact[i + 1] * (i + 1) % MOD

    # To calculate binomial coefficients C(n, k)
    def binom(n: int, k: int) -> int:
        if k > n or k < 0:
            return 0
        return fact[n] * inv_fact[k] % MOD * inv_fact[n - k] % MOD

    # Iterate for subsequence lengths from 1 to k
    for length in range(1, k + 1):
        for i in range(n):
            min_contrib = nums[i] * binom(n - i - 1, length - 1)
            max_contrib = nums[i] * binom(i, length - 1)
            total_sum += (min_contrib + max_contrib) % MOD
            total_sum %= MOD

    return total_sum

```
