from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Sort the array to easily find min and max in subsequences
    nums.sort()

    # Iterate over the elements to calculate their contributions
    for i in range(n):
        # Contribution of nums[i] as minimum
        # Number of ways to choose the rest (k-1) elements from (n-1) elements
        min_count = min(k, n - i)  # Count of subsequences where nums[i] is min
        total_sum += nums[i] * min_count
        total_sum %= MOD

        # Contribution of nums[i] as maximum
        # Number of ways to choose the rest (k-1) elements from (n-1) elements
        max_count = min(k, i + 1)  # Count of subsequences where nums[i] is max
        total_sum += nums[i] * max_count
        total_sum %= MOD
