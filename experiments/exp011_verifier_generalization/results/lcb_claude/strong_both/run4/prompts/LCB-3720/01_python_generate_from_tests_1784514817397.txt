def test_all_nodes_directly_to_zero():
    # n=3, edges=[[1,0,3],[2,0,7]], threshold=1
    # All nodes can reach 0 directly, max weight is 7
    assert minMaxWeight(3, [[1, 0, 3], [2, 0, 7]], 1) == 7


def test_chain_to_node_zero():
    # n=3, edges=[[2,1,2],[1,0,3]], threshold=1
    # Node 2 -> 1 -> 0, all nodes can reach 0, max weight is 3
    assert minMaxWeight(3, [[2, 1, 2], [1, 0, 3]], 1) == 3


def test_threshold_forces_edge_removal():
    # n=3, edges=[[1,0,5],[1,2,3],[2,0,2]], threshold=1
    # Node 1 has 2 outgoing edges, must remove one. Keep lower weights to minimize max.
    # Keep 1->2 (weight 3) and 2->0 (weight 2), max weight is 3
    assert minMaxWeight(3, [[1, 0, 5], [1, 2, 3], [2, 0, 2]], 1) == 3

    # n=3, edges=[[1,0,10],[1,2,1],[2,0,2]], threshold=2

    assert minMaxWeight(3, [[1, 0, 10], [1, 2, 1], [2, 0, 2]], 2) == 2

def test_node_zero_edges_ignored_for_reachability():
    # n=3, edges=[[0,1,1],[1,0,2],[2,1,3]], threshold=1
    # Node 0's outgoing edges don't help other nodes reach 0.
    # Nodes 1 and 2 can reach 0: 1->0 directly, 2->1->0
    assert minMaxWeight(3, [[0, 1, 1], [1, 0, 2], [2, 1, 3]], 1) == 3


def test_threshold_exactly_met():

    # Each node has at most 1 outgoing edge, all can reach 0, max weight is 3
    assert minMaxWeight(4, [[1, 0, 1], [2, 0, 2], [3, 2, 3]], 1) == 3


def test_large_threshold_no_restriction():

    # 1->0 (weight 1), 2->0 (weight 3), max weight is 3
    assert minMaxWeight(3, [[1, 0, 1], [1, 2, 5], [2, 0, 3]], 10) == 3
