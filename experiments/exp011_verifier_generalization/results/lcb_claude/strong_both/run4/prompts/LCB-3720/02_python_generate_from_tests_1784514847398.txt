from typing import List
import heapq


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Create adjacency list and count outgoing edges
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[u].append((v, w))

    # Check if we can reach node 0 from all nodes
    def can_reach_zero(max_weight):
        visited = [False] * n
        queue = [0]
        visited[0] = True
        while queue:
            node = queue.pop(0)
            for neighbor, weight in graph[node]:
                if not visited[neighbor] and weight <= max_weight:
                    visited[neighbor] = True
                    queue.append(neighbor)
        return all(visited)

    # Binary search on the maximum weight
    left, right = 1, 10**6
    answer = -1
    while left <= right:
        mid = (left + right) // 2
        if can_reach_zero(mid):
            # Check outgoing edges constraint
            outgoing_counts = [0] * n
            valid = True
            for u in range(n):
                for v, w in graph[u]:
                    if w <= mid:
                        outgoing_counts[u] += 1
                        if outgoing_counts[u] > threshold:
                            valid = False
                            break
                if not valid:
                    break
            if valid:
                answer = mid
                right = mid - 1
            else:
                left = mid + 1
        else:
            left = mid + 1

    return answer
```
