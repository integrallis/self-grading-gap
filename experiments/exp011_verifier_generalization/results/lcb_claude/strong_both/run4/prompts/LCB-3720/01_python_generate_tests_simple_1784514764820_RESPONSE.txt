from typing import List
import pytest
from solution import minMaxWeight


def test_canonical_example_1():
    # spec: n=5, edges=[[1,0,1],[2,0,2],[3,0,1],[4,3,1],[2,1,1]], threshold=2 -> 1
    # Remove edge 2->0. Max weight among remaining edges is 1.
    assert minMaxWeight(5, [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]], 2) == 1


def test_canonical_example_2():
    # spec: n=5, edges=[[0,1,1],[0,2,2],[0,3,1],[0,4,1],[1,2,1],[1,4,1]], threshold=1 -> -1
    # Impossible to reach node 0 from node 2.
    assert minMaxWeight(5, [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]], 1) == -1


def test_canonical_example_3():
    # spec: n=5, edges=[[1,2,1],[1,3,3],[1,4,5],[2,3,2],[3,4,2],[4,0,1]], threshold=1 -> 2
    # Remove edges 1->3 and 1->4. Max weight among remaining edges is 2.
    assert minMaxWeight(5, [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]], 1) == 2


def test_canonical_example_4():
    # spec: n=5, edges=[[1,2,1],[1,3,3],[1,4,5],[2,3,2],[4,0,1]], threshold=1 -> -1
    # Impossible to satisfy all conditions.
    assert minMaxWeight(5, [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]], 1) == -1


def test_simple_two_node_graph():
    # n=2, single edge from node 1 to node 0, threshold=1 -> weight of that edge
    # All nodes (just node 1) can reach node 0, threshold satisfied.
    assert minMaxWeight(2, [[1, 0, 5]], 1) == 5


def test_all_nodes_directly_to_zero():
    # n=3, edges=[[1,0,3],[2,0,7]], threshold=1
    # All nodes can reach 0 directly, max weight is 7
    assert minMaxWeight(3, [[1, 0, 3], [2, 0, 7]], 1) == 7


def test_chain_to_node_zero():
    # n=3, edges=[[2,1,2],[1,0,3]], threshold=1
    # Node 2 -> 1 -> 0, all nodes can reach 0, max weight is 3
    assert minMaxWeight(3, [[2, 1, 2], [1, 0, 3]], 1) == 3


def test_threshold_forces_edge_removal():
    # n=3, edges=[[1,0,5],[1,2,3],[2,0,2]], threshold=1
    # Node 1 has 2 outgoing edges, must remove one. Keep lower weights to minimize max.
    # Keep 1->2 (weight 3) and 2->0 (weight 2), max weight is 3
    assert minMaxWeight(3, [[1, 0, 5], [1, 2, 3], [2, 0, 2]], 1) == 3


def test_multiple_paths_choose_minimum_max():
    # n=3, edges=[[1,0,10],[1,2,1],[2,0,2]], threshold=2
    # Can keep all edges (threshold allows), or remove some.
    # Keeping 1->2 and 2->0 gives max weight 2 (better than keeping 1->0 with weight 10)
    assert minMaxWeight(3, [[1, 0, 10], [1, 2, 1], [2, 0, 2]], 2) == 2


def test_impossible_disconnected_node():
    # n=3, edges=[[1,0,1]], threshold=1
    # Node 2 has no outgoing edges and cannot reach node 0
    assert minMaxWeight(3, [[1, 0, 1]], 1) == -1


def test_node_zero_edges_ignored_for_reachability():
    # n=3, edges=[[0,1,1],[1,0,2],[2,1,3]], threshold=1
    # Node 0's outgoing edges don't help other nodes reach 0.
    # Nodes 1 and 2 can reach 0: 1->0 directly, 2->1->0
    assert minMaxWeight(3, [[0, 1, 1], [1, 0, 2], [2, 1, 3]], 1) == 3


def test_threshold_exactly_met():
    # n=4, edges=[[1,0,1],[2,0,2],[3,2,3]], threshold=1
    # Each node has at most 1 outgoing edge, all can reach 0, max weight is 3
    assert minMaxWeight(4, [[1, 0, 1], [2, 0, 2], [3, 2, 3]], 1) == 3


def test_large_threshold_no_restriction():
    # n=3, edges=[[1,0,1],[1,2,5],[2,0,3]], threshold=10
    # Threshold is very large, so no restriction. Choose path to minimize max weight.
    # 1->0 (weight 1), 2->0 (weight 3), max weight is 3
    assert minMaxWeight(3, [[1, 0, 1], [1, 2, 5], [2, 0, 3]], 10) == 3
