from typing import List
import pytest
from solution import minCosts


def test_canonical_example_1():
    # spec: cost = [5,3,4,1,3,2] -> [5,3,3,1,1,1]
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]


def test_canonical_example_2():
    # spec: cost = [1,2,4,6,7] -> [1,1,1,1,1]
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]


def test_single_person():
    # n=1: only person 0, must pay cost[0] to reach position 0
    assert minCosts([5]) == [5]


def test_two_people_increasing():
    # n=2: cost=[3,5]
    # position 0: pay 3
    # position 1: pay 5
    assert minCosts([3, 5]) == [3, 5]


def test_two_people_decreasing():
    # n=2: cost=[5,3]
    # position 0: pay 5
    # position 1: pay 3 OR pay 5 then swap free -> min is 3
    assert minCosts([5, 3]) == [5, 3]


def test_all_same_cost():
    # n=3: cost=[2,2,2]
    # position 0: pay 2
    # position 1: pay 2
    # position 2: pay 2
    assert minCosts([2, 2, 2]) == [2, 2, 2]


def test_strictly_decreasing():
    # n=4: cost=[10,7,5,3]
    # position 0: pay 10
    # position 1: pay 7
    # position 2: pay 5
    # position 3: pay 3
    assert minCosts([10, 7, 5, 3]) == [10, 7, 5, 3]


def test_strictly_increasing():
    # n=4: cost=[3,5,7,10]
    # position 0: pay 3
    # position 1: pay 3, swap free to 1 -> cost 3
    # position 2: pay 3, swap free to 2 -> cost 3
    # position 3: pay 3, swap free to 3 -> cost 3
    assert minCosts([3, 5, 7, 10]) == [3, 3, 3, 3]


def test_minimum_cost_values():
    # n=3: cost=[1,1,1] (minimum possible cost per constraint)
    # all positions cost 1
    assert minCosts([1, 1, 1]) == [1, 1, 1]


def test_maximum_cost_values():
    # n=3: cost=[100,100,100] (maximum possible cost per constraint)
    # all positions cost 100
    assert minCosts([100, 100, 100]) == [100, 100, 100]


def test_valley_shape():
    # n=5: cost=[5,3,1,3,5]
    # position 0: pay 5
    # position 1: pay 3
    # position 2: pay 1
    # position 3: pay 1, swap free to 3 -> cost 1
    # position 4: pay 1, swap free to 4 -> cost 1
    assert minCosts([5, 3, 1, 3, 5]) == [5, 3, 1, 1, 1]


def test_peak_shape():
    # n=5: cost=[1,3,5,3,1]
    # position 0: pay 1
    # position 1: pay 1, swap free to 1 -> cost 1
    # position 2: pay 1, swap free to 2 -> cost 1
    # position 3: pay 1, swap free to 3 -> cost 1
    # position 4: pay 1, swap free to 4 -> cost 1
    assert minCosts([1, 3, 5, 3, 1]) == [1, 1, 1, 1, 1]
