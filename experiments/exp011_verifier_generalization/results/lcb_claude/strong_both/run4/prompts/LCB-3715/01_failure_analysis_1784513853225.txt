def maximumCoins(coins: List[List[int]], k: int) -> int:

    coin_map = {}
    min_pos = float('inf')
    max_pos = float('-inf')
    
    for l, r, c in coins:
        for pos in range(l, r + 1):
            coin_map[pos] = coin_map.get(pos, 0) + c
        min_pos = min(min_pos, l)
        max_pos = max(max_pos, r)
    
    # Create an array to hold the cumulative sums of coins
    max_coins = 0
    current_sum = 0

    # Calculate the total coins for the first k bags
    for i in range(min_pos, min_pos + k):
        current_sum += coin_map.get(i, 0)
    max_coins = current_sum
