from typing import List
import pytest
from solution import maximumCoins


def test_canonical_example_1():
    # spec: coins = [[8,10,1],[1,3,2],[5,6,4]], k = 4 -> 10
    # Selecting bags at positions [3, 4, 5, 6] gives: 2 + 0 + 4 + 4 = 10
    assert maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) == 10


def test_canonical_example_2():
    # spec: coins = [[1,10,3]], k = 2 -> 6
    # Selecting bags at positions [1, 2] gives: 3 + 3 = 6
    assert maximumCoins([[1, 10, 3]], 2) == 6


def test_single_segment_k_equals_segment_length():
    # Single segment [1, 5, 10] with k=5 covers entire segment: 10*5 = 50
    assert maximumCoins([[1, 5, 10]], 5) == 50


def test_single_segment_k_less_than_segment_length():
    # Single segment [1, 10, 3] with k=3 can pick any 3 consecutive: 3*3 = 9
    assert maximumCoins([[1, 10, 3]], 3) == 9


def test_single_segment_k_greater_than_segment_length():
    # Single segment [5, 7, 2] (length 3) with k=5 includes 2 empty bags: 2*3 = 6
    assert maximumCoins([[5, 7, 2]], 5) == 6


def test_multiple_segments_no_overlap():
    # Segments [1, 2, 5], [5, 6, 3] with k=3
    # Best window: [1, 2, 3] = 5 + 5 + 0 = 10 or [5, 6, 7] = 3 + 3 + 0 = 6
    # Maximum is 10
    assert maximumCoins([[1, 2, 5], [5, 6, 3]], 3) == 10


def test_k_equals_one():
    # k=1 means selecting one bag; best is the bag with highest coin value
    # Segments [1, 3, 2], [5, 7, 10]: best single bag has 10 coins
    assert maximumCoins([[1, 3, 2], [5, 7, 10]], 1) == 10


def test_large_k_spanning_all_segments():
    # Segments [1, 2, 1], [4, 5, 2] with k=10
    # Total coins: 1*2 + 2*2 = 6, rest are empty bags
    assert maximumCoins([[1, 2, 1], [4, 5, 2]], 10) == 6


def test_segments_with_gaps():
    # Segments [1, 1, 5], [10, 10, 7] with k=2
    # Best options: [1, 2] = 5 + 0 = 5, [10, 11] = 7 + 0 = 7
    # Maximum is 7
    assert maximumCoins([[1, 1, 5], [10, 10, 7]], 2) == 7


def test_window_covering_multiple_segments():
    # Segments [1, 2, 3], [3, 4, 5], [5, 6, 2] with k=6
    # Window [1, 2, 3, 4, 5, 6] = 3 + 3 + 5 + 5 + 2 + 2 = 20
    assert maximumCoins([[1, 2, 3], [3, 4, 5], [5, 6, 2]], 6) == 20


def test_optimal_window_starts_before_first_segment():
    # Segment [5, 7, 4] with k=5
    # Window [3, 4, 5, 6, 7] = 0 + 0 + 4 + 4 + 4 = 12
    # Window [5, 6, 7, 8, 9] = 4 + 4 + 4 + 0 + 0 = 12
    assert maximumCoins([[5, 7, 4]], 5) == 12


def test_optimal_window_ends_after_last_segment():
    # Segment [1, 3, 2] with k=5
    # Window [1, 2, 3, 4, 5] = 2 + 2 + 2 + 0 + 0 = 6
    assert maximumCoins([[1, 3, 2]], 5) == 6


def test_window_entirely_in_gap():
    # Segments [1, 1, 10], [10, 10, 10] with k=2
    # Window [5, 6] = 0 + 0 = 0 (entirely in gap)
    assert maximumCoins([[1, 1, 10], [10, 10, 10]], 2) == 0


def test_multiple_segments_optimal_partial_overlap():
    # Segments [2, 4, 1], [6, 8, 3] with k=3
    # Window [6, 7, 8] = 3 + 3 + 3 = 9
    assert maximumCoins([[2, 4, 1], [6, 8, 3]], 3) == 9
