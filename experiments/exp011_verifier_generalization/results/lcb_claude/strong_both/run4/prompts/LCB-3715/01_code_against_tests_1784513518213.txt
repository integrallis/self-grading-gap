```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a dictionary to hold the coin counts at each position
    coin_map = {}
    for l, r, c in coins:
        for pos in range(l, r + 1):
            coin_map[pos] = coin_map.get(pos, 0) + c

    # Create a sorted list of positions
    positions = sorted(coin_map.keys())
    max_coins = 0
    current_coins = 0
    left = 0

    # Use a sliding window to calculate the max coins in k bags
    for right in range(len(positions)):
        # Add the current position's coins to the window
        current_coins += coin_map[positions[right]]

        # If the window exceeds k bags, move the left pointer
        while positions[right] - positions[left] + 1 > k:
            current_coins -= coin_map[positions[left]]
            left += 1

        # Update max_coins if the current window is valid
        if positions[right] - positions[left] + 1 <= k:
            max_coins = max(max_coins, current_coins)

    return max_coins
```
