from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a dictionary to hold the coin counts at each position
    coin_map = {}
    min_pos = float('inf')
    max_pos = float('-inf')
    
    for l, r, c in coins:
        for pos in range(l, r + 1):
            coin_map[pos] = coin_map.get(pos, 0) + c
        min_pos = min(min_pos, l)
        max_pos = max(max_pos, r)
    
    max_coins = 0
    
    # Check all starting positions from min_pos - k + 1 to max_pos
    for start in range(min_pos - k + 1, max_pos + 1):
        current_coins = sum(coin_map.get(pos, 0) for pos in range(start, start + k))
        max_coins = max(max_coins, current_coins)

    return max_coins

```
