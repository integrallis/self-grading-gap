```python
from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[[float('-inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0]  # Starting point
    if coins[0][0] < 0:
        dp[0][0][1] = 0  # Option to neutralize the starting cell

    for i in range(m):
        for j in range(n):
            for k in range(3):  # k: number of neutralizations used
                if i == 0 and j == 0:
                    continue
                if i > 0:
                    if coins[i][j] < 0 and k > 0:
                        current = max(dp[i-1][j][k] + coins[i][j], dp[i-1][j][k-1] + 0)  # Neutralize
                    else:
                        current = dp[i-1][j][k] + coins[i][j]  # Don't neutralize
                    dp[i][j][k] = max(dp[i][j][k], current)
                if j > 0:
                    if coins[i][j] < 0 and k > 0:
                        current = max(dp[i][j-1][k] + coins[i][j], dp[i][j-1][k-1] + 0)  # Neutralize
                    else:
                        current = dp[i][j-1][k] + coins[i][j]  # Don't neutralize
                    dp[i][j][k] = max(dp[i][j][k], current)

    result = max(dp[m-1][n-1])  # Maximum coins at the bottom-right corner with up to 2 neutralizations
    return result

```
