import pytest
from solution import calculateScore


def test_canonical_example_1():
    # spec: "aczzx" -> 5
    # i=0 'a': no mirror 'z' before it
    # i=1 'c': no mirror 'x' before it
    # i=2 'z': mirror 'a' at j=0 (unmarked), mark both, score += 2-0 = 2
    # i=3 'z': no unmarked mirror 'a' before it (j=0 is marked)
    # i=4 'x': mirror 'c' at j=1 (unmarked), mark both, score += 4-1 = 3
    # total score = 2 + 3 = 5
    assert calculateScore("aczzx") == 5


def test_canonical_example_2():
    # spec: "abcdef" -> 0
    # No character has its mirror before it
    assert calculateScore("abcdef") == 0


def test_single_character():
    # Single character has no j < i, so score is 0
    assert calculateScore("a") == 0


def test_mirror_pair_adjacent():
    # "az": i=1, 'z' finds mirror 'a' at j=0, score = 1-0 = 1
    assert calculateScore("az") == 1


def test_mirror_pair_separated():
    # "abz": i=2, 'z' finds mirror 'a' at j=0, score = 2-0 = 2
    assert calculateScore("abz") == 2


def test_multiple_same_mirror_pairs():
    # "azaz": 
    # i=1 'z': mirror 'a' at j=0, mark 0,1, score = 1
    # i=2 'a': no unmarked mirror 'z' before it (j=1 is marked)
    # i=3 'z': mirror 'a' at j=2, mark 2,3, score += 3-2 = 1
    # total = 2
    assert calculateScore("azaz") == 2


def test_closest_unmarked_mirror():
    # "aaz": 
    # i=1 'a': no mirror 'z' before it
    # i=2 'z': closest unmarked mirror 'a' is at j=1, mark 1,2, score = 2-1 = 1
    assert calculateScore("aaz") == 1


def test_no_mirrors_present():
    # "aaa": all same character, 'a' mirrors to 'z', no 'z' present
    assert calculateScore("aaa") == 0


def test_all_mirrors_but_wrong_order():
    # "zyxabc": mirrors are (z,a), (y,b), (x,c) but all mirrors come after
    assert calculateScore("zyxabc") == 0


def test_complex_matching():
    # "abccba":
    # i=0 'a': no mirror before
    # i=1 'b': no mirror before
    # i=2 'c': no mirror before
    # i=3 'c': no mirror 'x' before
    # i=4 'b': no mirror 'y' before
    # i=5 'a': no mirror 'z' before
    assert calculateScore("abccba") == 0


def test_mirror_self():
    # 'm' and 'n' are mirrors of each other (m=12, n=13, mirrors: m->n, n->m)
    # "mn": i=1 'n' finds mirror 'm' at j=0, score = 1
    assert calculateScore("mn") == 1


def test_multiple_potential_mirrors_pick_closest():
    # "aaaz":
    # i=1 'a': no mirror 'z' before
    # i=2 'a': no mirror 'z' before
    # i=3 'z': closest unmarked mirror 'a' is at j=2, score = 3-2 = 1
    assert calculateScore("aaaz") == 1


def test_marking_prevents_reuse():
    # "azaazz":
    # i=1 'z': mirror 'a' at j=0, mark 0,1, score = 1
    # i=2 'a': no unmarked mirror 'z' before (j=1 marked)
    # i=3 'a': no unmarked mirror 'z' before
    # i=4 'z': mirror 'a' at j=3, mark 3,4, score += 4-3 = 1
    # i=5 'z': mirror 'a' at j=2, mark 2,5, score += 5-2 = 3
    # total = 1 + 1 + 3 = 5
    assert calculateScore("azaazz") == 5
