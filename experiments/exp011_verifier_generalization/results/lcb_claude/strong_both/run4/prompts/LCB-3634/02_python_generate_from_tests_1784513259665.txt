```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    for i in range(len(s)):
        mirror = chr(219 - ord(s[i]))  # Calculate the mirror character.
        for j in range(i - 1, -1, -1):  # Look for the closest unmarked j < i.
            if not marked[j] and s[j] == mirror:
                marked[i] = marked[j] = True  # Mark both indices.
                score += i - j  # Add the difference to the score.
                break  # Stop after finding the closest unmarked mirror.
    return score

```
