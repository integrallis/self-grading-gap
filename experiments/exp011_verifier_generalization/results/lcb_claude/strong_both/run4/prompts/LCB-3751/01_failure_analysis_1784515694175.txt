def test_canonical_example_2():
    # spec: nums = [10,2,3,4,5,5,4,3,2,2], k = 10 -> 4

    # We can't make both 5 with a single operation on a contiguous subarray unless we convert entire array

    assert maxFrequency([3, 7], 5) == 1

from typing import List

def maxFrequency(nums: List[int], k: int) -> int:
    nums.sort()
    left = 0
    total = 0
    max_freq = 0
    for right in range(len(nums)):
        total += nums[right]
        while nums[right] * (right - left + 1) - total > k:
            total -= nums[left]
            left += 1
        max_freq = max(max_freq, right - left + 1)
    return max_freq
