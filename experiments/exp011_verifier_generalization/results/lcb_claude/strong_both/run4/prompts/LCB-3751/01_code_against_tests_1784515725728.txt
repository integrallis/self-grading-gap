```python
from typing import List

def maxFrequency(nums: List[int], k: int) -> int:
    count_k = nums.count(k)
    max_freq = count_k
    for num in set(nums):
        if num == k:
            continue
        required_x = k - num
        # Count how many we can convert to k
        current_freq = count_k
        for i in range(len(nums)):
            if nums[i] == num:
                current_freq += 1
            elif nums[i] < num:
                # If the element is less than num, we cannot convert it to k
                continue
            elif nums[i] > num:
                # If the element is greater than num, check if we can convert it
                if (nums[i] - required_x) == k:
                    current_freq += 1
        max_freq = max(max_freq, current_freq)
    return max_freq
```
