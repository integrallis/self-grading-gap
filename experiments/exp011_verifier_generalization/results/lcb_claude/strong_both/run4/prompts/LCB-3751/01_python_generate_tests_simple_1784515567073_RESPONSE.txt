from typing import List
import pytest
from solution import maxFrequency


def test_canonical_example_1():
    # spec: nums = [1,2,3,4,5,6], k = 1 -> 2
    # After adding -5 to nums[2..5], 1 has a frequency of 2 in [1, 2, -2, -1, 0, 1]
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2


def test_canonical_example_2():
    # spec: nums = [10,2,3,4,5,5,4,3,2,2], k = 10 -> 4
    # After adding 8 to nums[1..9], 10 has a frequency of 4 in [10, 10, 11, 12, 13, 13, 12, 11, 10, 10]
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4


def test_single_element_already_k():
    # Single element array where the element is already k
    # No operation needed, frequency is 1
    assert maxFrequency([5], 5) == 1


def test_single_element_not_k():
    # Single element array where the element is not k
    # Add (k - element) to make it k, frequency is 1
    assert maxFrequency([3], 5) == 1


def test_all_elements_already_k():
    # All elements are already k
    # No operation needed, frequency is the length of the array
    assert maxFrequency([7, 7, 7, 7], 7) == 4


def test_no_k_initially_convert_all():
    # No element is k initially, we can convert all to k by selecting entire array
    # Add (k - any_element) to all, frequency becomes len(nums)
    assert maxFrequency([1, 1, 1], 5) == 3


def test_some_elements_already_k():
    # Some elements are already k, we can add more by converting a subarray
    # e.g., [10, 5, 5], k=10: we have 1 copy of 10, can convert both 5s to 10 by adding 5 to nums[1..2]
    # Result: [10, 10, 10], frequency is 3
    assert maxFrequency([10, 5, 5], 10) == 3


def test_convert_prefix_subarray():
    # Convert a prefix subarray to k
    # [2, 2, 10], k=10: add 8 to nums[0..1] -> [10, 10, 10], frequency is 3
    assert maxFrequency([2, 2, 10], 10) == 3


def test_convert_suffix_subarray():
    # Convert a suffix subarray to k
    # [10, 3, 3], k=10: add 7 to nums[1..2] -> [10, 10, 10], frequency is 3
    assert maxFrequency([10, 3, 3], 10) == 3


def test_convert_middle_subarray():
    # Convert a middle subarray to k
    # [10, 5, 5, 10], k=10: add 5 to nums[1..2] -> [10, 10, 10, 10], frequency is 4
    assert maxFrequency([10, 5, 5, 10], 10) == 4


def test_optimal_is_partial_conversion():
    # Sometimes converting only part of array is optimal
    # [1, 2, 2, 10], k=2: we have 2 copies of 2, adding to any subarray won't increase frequency of 2
    # unless we convert other elements. Converting [1] to 2 gives us [2, 2, 2, 10] with frequency 3
    assert maxFrequency([1, 2, 2, 10], 2) == 3


def test_two_elements_same():
    # Two identical elements, convert to k
    # [5, 5], k=3: add -2 to nums[0..1] -> [3, 3], frequency is 2
    assert maxFrequency([5, 5], 3) == 2


def test_two_elements_different():
    # Two different elements, convert one or both to k
    # [3, 7], k=5: add 2 to nums[0..0] -> [5, 7] or add -2 to nums[1..1] -> [3, 5]
    # Either way frequency is 1. Or add 2 to nums[0..1] -> [5, 9], frequency is 1
    # Best is to convert both: this requires them to have same value after operation
    # We can't make both 5 with a single operation on a contiguous subarray unless we convert entire array
    # [3, 7], k=5: add 2 to nums[0..1] -> [5, 9], frequency 1. Or add -2 to nums[0..1] -> [1, 5], frequency 1
    # Actually we CAN convert entire array: add 2 to nums[0..1] -> [5, 9] (freq 1) or add -2 -> [1, 5] (freq 1)
    # We need same x for the subarray. To get [5, 5]: x=2 for first, x=-2 for second. Can't do both.
    # So best is frequency 1
    assert maxFrequency([3, 7], 5) == 1


def test_already_has_max_frequency():
    # Array already has multiple k values
    # [1, 1, 1], k=1: frequency is already 3, no operation needed (or operate on empty-ish range)
    assert maxFrequency([1, 1, 1], 1) == 3
