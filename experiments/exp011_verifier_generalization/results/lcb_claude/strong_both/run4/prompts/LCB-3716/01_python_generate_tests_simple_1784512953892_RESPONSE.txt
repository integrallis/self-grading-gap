from typing import List
import pytest
from solution import longestSubsequence


def test_canonical_example_1():
    # spec: nums = [16,6,3] -> 3
    # The longest subsequence is [16, 6, 3] with absolute adjacent differences [10, 3]
    # 10 >= 3, so this is valid and has length 3
    assert longestSubsequence([16, 6, 3]) == 3


def test_canonical_example_2():
    # spec: nums = [6,5,3,4,2,1] -> 4
    # The longest subsequence is [6, 4, 2, 1] with absolute adjacent differences [2, 2, 1]
    # 2 >= 2 >= 1, so this is valid and has length 4
    assert longestSubsequence([6, 5, 3, 4, 2, 1]) == 4


def test_canonical_example_3():
    # spec: nums = [10,20,10,19,10,20] -> 5
    # The longest subsequence is [10, 20, 10, 19, 10] with absolute adjacent differences [10, 10, 9, 9]
    # 10 >= 10 >= 9 >= 9, so this is valid and has length 5
    assert longestSubsequence([10, 20, 10, 19, 10, 20]) == 5


def test_minimum_length_array():
    # spec: 2 <= nums.length, so minimum valid input has 2 elements
    # Any two elements form a valid subsequence of length 2
    assert longestSubsequence([1, 2]) == 2


def test_all_same_values():
    # All elements the same: [5, 5, 5]
    # Any subsequence will have all differences = 0, which is non-increasing
    # Maximum length is 3 (all elements)
    assert longestSubsequence([5, 5, 5]) == 3


def test_strictly_increasing():
    # nums = [1, 2, 3, 4, 5]
    # Differences between consecutive pairs: 1, 1, 1, 1
    # A subsequence like [1, 2, 3, 4, 5] has differences [1, 1, 1, 1] which is non-increasing (1 >= 1 >= 1 >= 1)
    assert longestSubsequence([1, 2, 3, 4, 5]) == 5


def test_strictly_decreasing():
    # nums = [5, 4, 3, 2, 1]
    # Differences: 1, 1, 1, 1
    # Full sequence [5, 4, 3, 2, 1] has differences [1, 1, 1, 1] which is non-increasing
    assert longestSubsequence([5, 4, 3, 2, 1]) == 5


def test_large_then_small_differences():
    # nums = [1, 11, 12]
    # Subsequence [1, 11, 12]: differences [10, 1] -> 10 >= 1, valid, length 3
    assert longestSubsequence([1, 11, 12]) == 3


def test_increasing_differences_requires_skip():
    # nums = [1, 2, 5]
    # Full subsequence [1, 2, 5]: differences [1, 3] -> 1 >= 3 is FALSE
    # Best subsequences: [1, 2] with diff [1], or [1, 5] with diff [4], or [2, 5] with diff [3]
    # All have length 2
    assert longestSubsequence([1, 2, 5]) == 2


def test_alternating_large_small():
    # nums = [1, 10, 11, 20]
    # Try [1, 10, 11]: differences [9, 1] -> 9 >= 1, valid, length 3
    # Try [1, 10, 20]: differences [9, 10] -> 9 >= 10 is FALSE
    # Try [1, 10, 11, 20]: differences [9, 1, 9] -> 9 >= 1 >= 9 is FALSE
    # Best is likely [1, 10, 11] with length 3
    assert longestSubsequence([1, 10, 11, 20]) == 3


def test_single_large_jump():
    # nums = [1, 100, 101, 102]
    # [1, 100, 101, 102]: differences [99, 1, 1] -> 99 >= 1 >= 1, valid, length 4
    assert longestSubsequence([1, 100, 101, 102]) == 4


def test_two_elements_same():
    # nums = [5, 5]
    # Subsequence [5, 5]: difference [0], valid, length 2
    assert longestSubsequence([5, 5]) == 2


def test_zigzag_pattern():
    # nums = [1, 5, 2, 6, 3]
    # Try [1, 5, 2]: differences [4, 3] -> 4 >= 3, valid, length 3
    # Try [1, 5, 2, 6]: differences [4, 3, 4] -> 4 >= 3 >= 4 is FALSE
    # Try [1, 5, 2, 3]: differences [4, 3, 1] -> 4 >= 3 >= 1, valid, length 4
    assert longestSubsequence([1, 5, 2, 6, 3]) == 4
