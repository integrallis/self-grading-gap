```python
from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = [0] * (n + 1)  # dp[i] will hold the length of the longest valid subsequence ending at index i
    max_length = 0

    for i in range(n):
        for j in range(i):
            if dp[j] == 0:
                continue
            diff1 = abs(nums[i] - nums[j])
            if j == 0 or (dp[j] > 0 and abs(nums[j] - nums[j - 1]) >= diff1):
                dp[i] = max(dp[i], dp[j] + 1)
        dp[i] = max(dp[i], 1)  # Every number itself can be a subsequence
        max_length = max(max_length, dp[i])

    return max_length + 1 if max_length > 0 else 0
```
