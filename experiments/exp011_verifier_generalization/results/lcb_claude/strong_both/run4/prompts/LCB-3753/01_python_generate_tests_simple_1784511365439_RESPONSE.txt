import pytest
from solution import maxDifference


def test_canonical_example_1():
    # spec: "aaaaabbc" -> 3
    # 'a' has frequency 5 (odd), 'b' has frequency 2 (even)
    # max difference = 5 - 2 = 3
    assert maxDifference("aaaaabbc") == 3


def test_canonical_example_2():
    # spec: "abcabcab" -> 1
    # 'a' has frequency 3 (odd), 'c' has frequency 2 (even)
    # max difference = 3 - 2 = 1
    assert maxDifference("abcabcab") == 1


def test_single_odd_single_even_minimum_length():
    # s = "aab": 'a' freq 2 (even), 'b' freq 1 (odd)
    # max difference = 1 - 2 = -1
    assert maxDifference("aab") == -1


def test_multiple_odd_multiple_even():
    # s = "aaabbb": 'a' freq 3 (odd), 'b' freq 3 (odd)
    # But we need one odd and one even. This violates constraint.
    # s = "aaaabbbb": 'a' freq 4 (even), 'b' freq 4 (even)
    # This also violates constraint.
    # s = "aaaabbb": 'a' freq 4 (even), 'b' freq 3 (odd)
    # max difference = 3 - 4 = -1
    assert maxDifference("aaaabbb") == -1


def test_max_odd_min_even():
    # s = "aaaaaaaab": 'a' freq 8 (even), 'b' freq 1 (odd)
    # max difference = 1 - 8 = -7
    # s = "aaaaaaaaab": 'a' freq 9 (odd), 'b' freq 1 (odd)
    # s = "aaaaaaaaabc": 'a' freq 9 (odd), 'b' freq 1 (odd), 'c' freq 1 (odd)
    # s = "aaaaaaaabbc": 'a' freq 9 (odd), 'b' freq 2 (even), 'c' freq 1 (odd)
    # max difference = 9 - 2 = 7
    assert maxDifference("aaaaaaaabbc") == 7


def test_all_different_characters():
    # s = "abcd": 'a' freq 1 (odd), 'b' freq 1 (odd), 'c' freq 1 (odd), 'd' freq 1 (odd)
    # No even frequency, violates constraint
    # s = "aabcd": 'a' freq 2 (even), 'b' freq 1 (odd), 'c' freq 1 (odd), 'd' freq 1 (odd)
    # max difference = 1 - 2 = -1
    assert maxDifference("aabcd") == -1


def test_even_greater_than_odd():
    # s = "aaaaab": 'a' freq 5 (odd), 'b' freq 1 (odd)
    # s = "aaaaaab": 'a' freq 6 (even), 'b' freq 1 (odd)
    # max difference = 1 - 6 = -5
    assert maxDifference("aaaaaab") == -5


def test_multiple_characters_with_different_frequencies():
    # s = "aaabbbcccc": 'a' freq 3 (odd), 'b' freq 3 (odd), 'c' freq 4 (even)
    # max difference = 3 - 4 = -1
    assert maxDifference("aaabbbcccc") == -1


def test_large_odd_small_even():
    # s = "aaaaaaaaaaaaaaaaaabb": 'a' freq 18 (even), 'b' freq 2 (even)
    # s = "aaaaaaaaaaaaaaaaaaaab": 'a' freq 20 (even), 'b' freq 1 (odd)
    # max difference = 1 - 20 = -19
    # s = "aaaaaaaaaaaaaaaaaaabb": 'a' freq 19 (odd), 'b' freq 2 (even)
    # max difference = 19 - 2 = 17
    assert maxDifference("aaaaaaaaaaaaaaaaaaabb") == 17


def test_three_characters_mixed_parities():
    # s = "aaabbccc": 'a' freq 3 (odd), 'b' freq 2 (even), 'c' freq 3 (odd)
    # max difference candidates: 3 - 2 = 1
    assert maxDifference("aaabbccc") == 1
