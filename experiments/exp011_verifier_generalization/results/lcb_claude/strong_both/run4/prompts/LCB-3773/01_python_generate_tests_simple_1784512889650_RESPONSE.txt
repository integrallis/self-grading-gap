from typing import List
import pytest
from solution import minimumPairRemoval


def test_canonical_example_1():
    # spec: nums = [5,2,3,1] -> 2
    # (3,1) sum=4 -> [5,2,4]
    # (2,4) sum=6 -> [5,6]
    # 2 operations total
    assert minimumPairRemoval([5, 2, 3, 1]) == 2


def test_canonical_example_2():
    # spec: nums = [1,2,2] -> 0
    # already non-decreasing
    assert minimumPairRemoval([1, 2, 2]) == 0


def test_single_element_already_sorted():
    # single element is trivially non-decreasing
    assert minimumPairRemoval([5]) == 0


def test_two_elements_already_sorted():
    # [1, 2] is already non-decreasing
    assert minimumPairRemoval([1, 2]) == 0


def test_two_elements_need_merge():
    # [2, 1] is not non-decreasing
    # merge (2,1) -> [3]
    # 1 operation
    assert minimumPairRemoval([2, 1]) == 1


def test_all_equal_elements():
    # [3, 3, 3, 3] is already non-decreasing
    assert minimumPairRemoval([3, 3, 3, 3]) == 0


def test_strictly_increasing():
    # [1, 2, 3, 4, 5] is already non-decreasing
    assert minimumPairRemoval([1, 2, 3, 4, 5]) == 0


def test_strictly_decreasing():
    # [5, 4, 3, 2, 1] requires merging
    # pairs: (5,4)=9, (4,3)=7, (3,2)=5, (2,1)=3
    # leftmost minimum: (2,1) -> [5,4,3,3]
    # pairs: (5,4)=9, (4,3)=7, (3,3)=6
    # leftmost minimum: (3,3) -> [5,4,6]
    # pairs: (5,4)=9, (4,6)=10
    # leftmost minimum: (5,4) -> [9,6]
    # pairs: (9,6)=15
    # leftmost minimum: (9,6) -> [15]
    # But [9,6] is not non-decreasing, so merge -> [15]
    # Total operations needed
    assert minimumPairRemoval([5, 4, 3, 2, 1]) == 4


def test_negative_numbers():
    # [-5, -3, -1] is already non-decreasing
    assert minimumPairRemoval([-5, -3, -1]) == 0


def test_mixed_negative_positive():
    # [-1, 2, 1] is not non-decreasing (2 > 1 violates)
    # pairs: (-1,2)=1, (2,1)=3
    # leftmost minimum: (-1,2) -> [1,1]
    # [1,1] is non-decreasing
    assert minimumPairRemoval([-1, 2, 1]) == 1


def test_with_zeros():
    # [0, 0, 0] is already non-decreasing
    assert minimumPairRemoval([0, 0, 0]) == 0


def test_leftmost_selection_matters():
    # [3, 1, 1, 3]
    # pairs: (3,1)=4, (1,1)=2, (1,3)=4
    # minimum sum is 2, only one such pair at index 1
    # merge (1,1) -> [3,2,3]
    # pairs: (3,2)=5, (2,3)=5
    # both have same sum, choose leftmost: (3,2) -> [5,3]
    # pairs: (5,3)=8
    # [5,3] not non-decreasing, merge -> [8]
    assert minimumPairRemoval([3, 1, 1, 3]) == 3


def test_large_values():
    # [1000, -1000] not non-decreasing
    # merge (1000,-1000) -> [0]
    assert minimumPairRemoval([1000, -1000]) == 1


def test_complex_case():
    # [10, 1, 2, 20]
    # pairs: (10,1)=11, (1,2)=3, (2,20)=22
    # minimum: (1,2) -> [10,3,20]
    # [10,3,20] not non-decreasing (10>3)
    # pairs: (10,3)=13, (3,20)=23
    # minimum: (10,3) -> [13,20]
    # [13,20] is non-decreasing
    assert minimumPairRemoval([10, 1, 2, 20]) == 2
