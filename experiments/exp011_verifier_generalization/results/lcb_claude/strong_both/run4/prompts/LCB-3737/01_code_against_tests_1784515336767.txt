```python
from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    dp = [[0] * 3 for _ in range(n)]
    dp[0] = cost[0]

    for i in range(1, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(dp[i-1][(j+1) % 3], dp[i-1][(j+2) % 3])

    half_n = n // 2
    result = float('inf')

    for j in range(3):
        result = min(result, dp[n-1][j] + dp[half_n - 1][(j + 1) % 3] + dp[half_n - 1][(j + 2) % 3])

    return result
```
