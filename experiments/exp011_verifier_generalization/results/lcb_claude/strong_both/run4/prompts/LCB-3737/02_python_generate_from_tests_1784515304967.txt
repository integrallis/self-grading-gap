from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    dp = [[0] * 3 for _ in range(n)]
    dp[0] = cost[0]
    dp[1][0] = cost[1][0] + min(dp[0][1], dp[0][2])  # Color 1
    dp[1][1] = cost[1][1] + min(dp[0][0], dp[0][2])  # Color 2
    dp[1][2] = cost[1][2] + min(dp[0][0], dp[0][1])  # Color 3

    for i in range(2, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(dp[i-1][(j+1) % 3], dp[i-1][(j+2) % 3])

    last_cost = [dp[n-1][0], dp[n-1][1], dp[n-1][2]]
    return min(last_cost[0] + last_cost[1], last_cost[0] + last_cost[2], last_cost[1] + last_cost[2])
