def test_canonical_example_1():
    # spec: n=4, cost=[[3,5,7],[6,2,9],[4,8,1],[7,3,5]] -> 9
    # optimal painting: [1, 2, 3, 2] with costs [3, 2, 1, 3] = 3+2+1+3 = 9
    n = 4

    assert minCost(n, cost) == 9
