import pytest
from solution import hasSameDigits


def test_canonical_example_1():
    # spec: "3902" -> "292" -> "11" -> True (same digits)
    assert hasSameDigits("3902") is True


def test_canonical_example_2():
    # spec: "34789" -> "7157" -> "862" -> "48" -> False (different digits)
    assert hasSameDigits("34789") is False


def test_minimum_length_string_same_digits():
    # Minimum length is 3, after one operation becomes 2 digits
    # "111" -> (1+1)%10=2, (1+1)%10=2 -> "22" -> True
    assert hasSameDigits("111") is True


def test_minimum_length_string_different_digits():
    # "123" -> (1+2)%10=3, (2+3)%10=5 -> "35" -> False
    assert hasSameDigits("123") is False

def test_string_with_zeros():
    # "000" -> (0+0)%10=0, (0+0)%10=0 -> "00" -> True
    assert hasSameDigits("000") is True


def test_string_with_modulo_wrapping():
    # "999" -> (9+9)%10=8, (9+9)%10=8 -> "88" -> True
    assert hasSameDigits("999") is True


def test_longer_string_leading_to_same():
    # "12345" -> (1+2)%10=3, (2+3)%10=5, (3+4)%10=7, (4+5)%10=9 -> "3579"
    # "3579" -> (3+5)%10=8, (5+7)%10=2, (7+9)%10=6 -> "826"
    # "826" -> (8+2)%10=0, (2+6)%10=8 -> "08" -> False
    assert hasSameDigits("12345") is False


def test_longer_string_with_repeating_pattern():
    # "5555" -> (5+5)%10=0, (5+5)%10=0, (5+5)%10=0 -> "000"
    # "000" -> (0+0)%10=0, (0+0)%10=0 -> "00" -> True
    assert hasSameDigits("5555") is True


def test_string_with_mixed_digits():
    # "2468" -> (2+4)%10=6, (4+6)%10=0, (6+8)%10=4 -> "604"
    # "604" -> (6+0)%10=6, (0+4)%10=4 -> "64" -> False
    assert hasSameDigits("2468") is False


def test_maximum_single_digit_sum():
    # "9990" -> (9+9)%10=8, (9+9)%10=8, (9+0)%10=9 -> "889"
    # "889" -> (8+8)%10=6, (8+9)%10=7 -> "67" -> False
    assert hasSameDigits("9990") is False
