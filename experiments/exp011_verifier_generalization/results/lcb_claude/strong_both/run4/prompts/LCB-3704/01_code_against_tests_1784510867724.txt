```python
from typing import List

def countPartitions(nums: List[int]) -> int:
    total_sum = sum(nums)
    count = 0
    left_sum = 0
    for i in range(len(nums) - 1):  # Loop until the second last index
        left_sum += nums[i]  # Update the left subarray sum
        right_sum = total_sum - left_sum  # Compute the right subarray sum
        if (left_sum - right_sum) % 2 == 0:
            count += 1  # Check if the difference is even
    return count
```
