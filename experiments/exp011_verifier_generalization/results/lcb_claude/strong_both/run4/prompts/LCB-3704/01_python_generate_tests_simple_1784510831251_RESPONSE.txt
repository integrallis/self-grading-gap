import pytest

from solution import countPartitions


def test_canonical_example_1():
    # spec: nums = [10,10,3,7,6]
    # partition at i=0: [10] vs [10,3,7,6] -> 10 - 26 = -16 (even)
    # partition at i=1: [10,10] vs [3,7,6] -> 20 - 16 = 4 (even)
    # partition at i=2: [10,10,3] vs [7,6] -> 23 - 13 = 10 (even)
    # partition at i=3: [10,10,3,7] vs [6] -> 30 - 6 = 24 (even)
    # expected: 4
    assert countPartitions([10, 10, 3, 7, 6]) == 4
