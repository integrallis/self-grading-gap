import pytest
from typing import List
from solution import maxLength


def test_canonical_example_1():
    # spec: nums = [1,2,1,2,1,1,1] -> 5
    # longest product equivalent subarray is [1, 2, 1, 1, 1]
    assert maxLength([1, 2, 1, 2, 1, 1, 1]) == 5


def test_canonical_example_2():
    # spec: nums = [2,3,4,5,6] -> 3
    # longest product equivalent subarray is [3, 4, 5]
    assert maxLength([2, 3, 4, 5, 6]) == 3


def test_canonical_example_3():
    # spec: nums = [1,2,3,1,4,5,1] -> 5
    assert maxLength([1, 2, 3, 1, 4, 5, 1]) == 5


def test_minimum_length_array():
    # spec: 2 <= nums.length, so minimum is 2 elements
    # [1, 1]: prod=1, gcd=1, lcm=1, 1 == 1*1, so product equivalent
    assert maxLength([1, 1]) == 2


def test_all_ones():
    # [1, 1, 1, 1]: prod=1, gcd=1, lcm=1, 1 == 1*1, so product equivalent
    assert maxLength([1, 1, 1, 1]) == 4


def test_two_element_product_equivalent():
    # [2, 2]: prod=4, gcd=2, lcm=2, 4 == 2*2, so product equivalent
    assert maxLength([2, 2]) == 2


def test_two_element_not_product_equivalent():
    # [2, 3]: prod=6, gcd=1, lcm=6, 6 == 1*6, so product equivalent
    # [2, 4]: prod=8, gcd=2, lcm=4, 8 == 2*4, so product equivalent
    # [3, 4]: prod=12, gcd=1, lcm=12, 12 == 1*12, so product equivalent
    # Actually any two coprime numbers work
    assert maxLength([2, 3]) == 2


def test_single_element_with_ones():
    # [1, 5]: prod=5, gcd=1, lcm=5, 5 == 1*5, so product equivalent
    assert maxLength([1, 5]) == 2


def test_array_with_multiple_product_equivalent_subarrays():
    # [3, 4, 5] from example 2: prod=60, gcd=1, lcm=60, 60 == 1*60
    # Test if we can find this in a longer array
    assert maxLength([2, 3, 4, 5, 2]) == 3


def test_max_value_constraint():
    # spec: 1 <= nums[i] <= 10
    # [10, 10]: prod=100, gcd=10, lcm=10, 100 == 10*10, so product equivalent
    assert maxLength([10, 10]) == 2


def test_array_starting_with_one():
    # [1, x] is always product equivalent for any x
    # [1, 7]: prod=7, gcd=1, lcm=7, 7 == 1*7
    assert maxLength([1, 7]) == 2


def test_three_same_elements():
    # [3, 3, 3]: prod=27, gcd=3, lcm=3, 27 != 3*3=9, not product equivalent
    # But [3, 3]: prod=9, gcd=3, lcm=3, 9 == 3*3, product equivalent
    result = maxLength([3, 3, 3])
    # Should be at least 2 (any pair of 3s)
    assert result >= 2


def test_mixed_values():
    # [1, 2, 3]: prod=6, gcd=1, lcm=6, 6 == 1*6, product equivalent
    assert maxLength([1, 2, 3]) >= 3
