from typing import List
from itertools import permutations

def totalNumbers(digits: List[int]) -> int:
    even_digits = [d for d in set(digits) if d % 2 == 0]
    count = 0
    for last_digit in even_digits:
        for perm in permutations(digits, 3):
            if perm[2] == last_digit and perm[0] != 0:
                count += 1
    return count // 6  # Each combination is counted 6 times due to permutations

```
