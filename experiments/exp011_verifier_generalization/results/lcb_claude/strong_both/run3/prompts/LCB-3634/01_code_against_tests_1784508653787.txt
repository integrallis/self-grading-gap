```python
def calculateScore(s: str) -> int:
    mirrored = {chr(i + 97): chr(122 - i) for i in range(26)}  # mapping letters to their mirrors
    marked = [False] * len(s)  # tracking marked indices
    score = 0

    for i in range(len(s)):
        if marked[i]:  # skip if already marked
            continue
        mirror_char = mirrored[s[i]]  # find mirror character
        found_mirror = False  # flag for found mirror
        # find the closest unmarked index j < i
        for j in range(i - 1, -1, -1):
            if s[j] == mirror_char:
                found_mirror = True
                if not marked[j]:  # check for unmarked mirror
                    marked[i] = True  # mark index i
                    marked[j] = True  # mark index j
                    score += i - j  # update score
                    break  # exit loop after finding the closest
        if found_mirror:
            marked[i] = True  # mark index i if we found a mirror (even if marked)

    return score
```
