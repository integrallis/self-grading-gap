import pytest
from solution import calculateScore


def test_canonical_example_1():
    # spec: s = "aczzx"
    # i=0 'a': no mirror found, skip
    # i=1 'c': no mirror found, skip
    # i=2 'z': mirror is 'a' at j=0 (unmarked), mark both, score += 2-0 = 2
    # i=3 'z': mirror is 'a', but j=0 is marked, skip
    # i=4 'x': mirror is 'c' at j=1 (unmarked), mark both, score += 4-1 = 3
    # total score = 2 + 3 = 5
    assert calculateScore("aczzx") == 5


def test_canonical_example_2():
    # spec: s = "abcdef"
    # No character has its mirror present before it
    # total score = 0
    assert calculateScore("abcdef") == 0


def test_single_character():
    # s = "a", only one character, no j < i exists
    # score = 0
    assert calculateScore("a") == 0


def test_two_mirror_characters():
    # s = "az", i=1 'z' has mirror 'a' at j=0
    # score = 1 - 0 = 1
    assert calculateScore("az") == 1


def test_two_non_mirror_characters():
    # s = "ab", 'b' mirror is 'y', not present
    # score = 0
    assert calculateScore("ab") == 0


def test_multiple_mirrors_picks_closest():
    # s = "aza", 
    # i=1 'z': mirror 'a' at j=0, mark both, score += 1-0 = 1
    # i=2 'a': mirror 'z' at j=1 but j=1 is marked, skip
    # score = 1
    assert calculateScore("aza") == 1


def test_mirror_pairs_reversed_order():
    # s = "za", i=1 'a' has mirror 'z' at j=0
    # score = 1 - 0 = 1
    assert calculateScore("za") == 1


def test_same_character_not_its_own_mirror():
    # s = "nn", 'n' mirror is 'm' (n=14th letter, mirror is 13th='m')
    # no mirrors found, score = 0
    assert calculateScore("nn") == 0


def test_middle_character_is_own_mirror():
    # s = "mm", 'm' is 13th letter (0-indexed: 12)
    # mirror of 'm' is 'n' (25-12=13th letter='n')
    # Actually 'a'=0, 'z'=25, so 'm'=12, mirror=25-12=13='n'
    # So "mm" has no mirrors, score = 0
    assert calculateScore("mm") == 0


def test_multiple_pairs():
    # s = "abzy"
    # i=0 'a': no j<0
    # i=1 'b': mirror 'y' not found before
    # i=2 'z': mirror 'a' at j=0, mark both, score += 2-0 = 2
    # i=3 'y': mirror 'b' at j=1 (unmarked), mark both, score += 3-1 = 2
    # total = 4
    assert calculateScore("abzy") == 4


def test_marked_index_not_reused():
    # s = "azaz"
    # i=1 'z': mirror 'a' at j=0, mark both, score += 1
    # i=2 'a': mirror 'z' at j=1 but marked, skip
    # i=3 'z': mirror 'a' at j=2 but marked, and j=0 is marked, skip
    # score = 1
    assert calculateScore("azaz") == 1


def test_closest_unmarked_index():
    # s = "aazz"
    # i=1 'a': mirror 'z' not found before
    # i=2 'z': mirror 'a', closest is j=1, mark both, score += 2-1 = 1
    # i=3 'z': mirror 'a', j=1 marked, try j=0 (unmarked), mark both, score += 3-0 = 3
    # total = 4
    assert calculateScore("aazz") == 4


def test_all_same_non_mirror_character():
    # s = "aaaa", 'a' mirror is 'z', not present
    # score = 0
    assert calculateScore("aaaa") == 0


def test_longer_sequence_with_gaps():
    # s = "abcxyz"
    # i=3 'x': mirror 'c' at j=2, score += 3-2 = 1
    # i=4 'y': mirror 'b' at j=1, score += 4-1 = 3
    # i=5 'z': mirror 'a' at j=0, score += 5-0 = 5
    # total = 9
    assert calculateScore("abcxyz") == 9
