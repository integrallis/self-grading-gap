from solution import largestInteger
from typing import List


def test_canonical_example_1():
    # spec: nums = [3,9,2,1,7], k = 3 -> 7
    # 3 appears in 1 subarray, 7 appears in 1 subarray, largest is 7
    assert largestInteger([3, 9, 2, 1, 7], 3) == 7


def test_canonical_example_2():
    # spec: nums = [3,9,7,2,1,7], k = 4 -> 3
    # 3 appears in 1 subarray of size 4, it is the only such integer
    assert largestInteger([3, 9, 7, 2, 1, 7], 4) == 3


def test_canonical_example_3():
    # spec: nums = [0,0], k = 1 -> -1
    # No integer appears in exactly one subarray of size 1
    assert largestInteger([0, 0], 1) == -1


def test_single_element_array():
    # nums = [5], k = 1
    # Only one subarray [5], 5 appears in exactly 1 subarray
    assert largestInteger([5], 1) == 5


def test_all_elements_appear_in_multiple_subarrays():
    # nums = [1,1,1], k = 2
    # Subarrays: [1,1], [1,1]
    # 1 appears in 2 subarrays, no integer appears in exactly 1
    assert largestInteger([1, 1, 1], 2) == -1


def test_k_equals_array_length():
    # nums = [3,5,1], k = 3
    # Only one subarray [3,5,1], all elements appear in exactly 1 subarray
    # Largest is 5
    assert largestInteger([3, 5, 1], 3) == 5


def test_multiple_integers_appear_once():
    # nums = [1,2,3,4], k = 2
    # Subarrays: [1,2], [2,3], [3,4]
    # 1 appears in 1 subarray: [1,2]
    # 2 appears in 2 subarrays: [1,2], [2,3]
    # 3 appears in 2 subarrays: [2,3], [3,4]
    # 4 appears in 1 subarray: [3,4]
    # Largest among {1,4} is 4
    assert largestInteger([1, 2, 3, 4], 2) == 4


def test_duplicate_values_in_different_positions():
    # nums = [5,3,5,3], k = 2
    # Subarrays: [5,3], [3,5], [5,3]
    # 3 appears in 3 subarrays: [5,3], [3,5], [5,3]
    # 5 appears in 3 subarrays: [5,3], [3,5], [5,3]
    # No integer appears in exactly 1 subarray
    assert largestInteger([5, 3, 5, 3], 2) == -1


def test_zero_values():
    # nums = [0,1,2], k = 2
    # Subarrays: [0,1], [1,2]
    # 0 appears in 1 subarray: [0,1]
    # 1 appears in 2 subarrays: [0,1], [1,2]
    # 2 appears in 1 subarray: [1,2]
    # Largest among {0,2} is 2
    assert largestInteger([0, 1, 2], 2) == 2


def test_large_k_small_array():
    # nums = [10,20], k = 2
    # Only one subarray [10,20]
    # Both 10 and 20 appear in exactly 1 subarray
    # Largest is 20
    assert largestInteger([10, 20], 2) == 20


def test_element_at_boundary():
    # nums = [7,8,9,10], k = 3
    # Subarrays: [7,8,9], [8,9,10]
    # 7 appears in 1 subarray: [7,8,9]
    # 8 appears in 2 subarrays: [7,8,9], [8,9,10]
    # 9 appears in 2 subarrays: [7,8,9], [8,9,10]
    # 10 appears in 1 subarray: [8,9,10]
    # Largest among {7,10} is 10
    assert largestInteger([7, 8, 9, 10], 3) == 10
