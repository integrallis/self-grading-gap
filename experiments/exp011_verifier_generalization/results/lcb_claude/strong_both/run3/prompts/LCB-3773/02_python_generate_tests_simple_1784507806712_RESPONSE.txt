from typing import List
import pytest
from solution import minimumPairRemoval


def test_canonical_example_1():
    # spec: nums = [5,2,3,1] -> 2
    # The pair (3,1) has the minimum sum of 4. After replacement, nums = [5,2,4].
    # The pair (2,4) has the minimum sum of 6. After replacement, nums = [5,6].
    assert minimumPairRemoval([5, 2, 3, 1]) == 2


def test_canonical_example_2():
    # spec: nums = [1,2,2] -> 0
    # The array nums is already sorted.
    assert minimumPairRemoval([1, 2, 2]) == 0


def test_single_element():
    # spec: 1 <= nums.length, single element is trivially non-decreasing
    assert minimumPairRemoval([5]) == 0


def test_two_elements_non_decreasing():
    # spec: [1, 2] is already non-decreasing
    assert minimumPairRemoval([1, 2]) == 0


def test_two_elements_decreasing():
    # spec: [2, 1] requires one merge to become [3]
    assert minimumPairRemoval([2, 1]) == 1


def test_already_non_decreasing():
    # spec: [1, 2, 3, 4] is already non-decreasing
    assert minimumPairRemoval([1, 2, 3, 4]) == 0


def test_negative_numbers():
    # spec: -1000 <= nums[i] <= 1000, test with negative values
    # [-5, -3, -2] is already non-decreasing
    assert minimumPairRemoval([-5, -3, -2]) == 0


def test_negative_numbers_need_merging():
    # spec: [5, -3, 2] -> minimum sum is (-3, 2) = -1, giving [5, -1]
    # then (5, -1) = 4, giving [4] which is non-decreasing
    assert minimumPairRemoval([5, -3, 2]) == 2


def test_all_same_elements():
    # spec: [3, 3, 3, 3] is already non-decreasing
    assert minimumPairRemoval([3, 3, 3, 3]) == 0


def test_reverse_sorted():
    # spec: [5, 4, 3, 2, 1] requires multiple operations
    # minimum pair (2,1)=3 -> [5,4,3,3]
    # minimum pair (3,3)=6 -> [5,4,6]
    # minimum pair (4,6)=10 -> [5,10] which is non-decreasing
    assert minimumPairRemoval([5, 4, 3, 2, 1]) == 3


def test_mixed_positive_negative():
    # spec: [-10, 5, 3] -> minimum sum (-10,5)=-5 (leftmost), giving [-5, 3]
    # then (-5,3)=-2 -> [-2] which is non-decreasing
    assert minimumPairRemoval([-10, 5, 3]) == 2


def test_zero_values():
    # spec: [0, 0, 0] is non-decreasing
    assert minimumPairRemoval([0, 0, 0]) == 0


def test_leftmost_selection_on_tie():
    # spec: if multiple pairs have the same minimum sum, choose leftmost
    # [1, 2, 1, 2] -> pairs (1,2)=3, (2,1)=3, (1,2)=3
    # leftmost is (1,2) at index 0, giving [3, 1, 2]
    # then (1,2)=3 -> [3, 3] which is non-decreasing
    assert minimumPairRemoval([1, 2, 1, 2]) == 2
