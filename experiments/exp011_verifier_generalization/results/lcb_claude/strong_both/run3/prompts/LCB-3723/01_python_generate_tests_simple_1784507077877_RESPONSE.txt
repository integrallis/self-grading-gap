import pytest
from typing import List
from solution import sumOfGoodNumbers


def test_canonical_example_1():
    # spec: nums = [1,3,2,1,5,4], k = 2
    # good elements: nums[1]=3 (3>1 at i-2=0), nums[4]=5 (5>2 at i-2=2), nums[5]=4 (i+2=7 doesn't exist, i-2=3, 4>1)
    # sum = 3 + 5 + 4 = 12
    assert sumOfGoodNumbers([1, 3, 2, 1, 5, 4], 2) == 12


def test_canonical_example_2():
    # spec: nums = [2,1], k = 1

    # nums[1]=3: i+1=2 doesn't exist, i-1=0 exists (5), 3<5, not good
    # sum = 5
    assert sumOfGoodNumbers([5, 3], 1) == 5


def test_single_element_at_boundary():
    # nums = [10, 20, 30], k = 1
    # nums[0]=10: i-1=-1 doesn't exist, i+1=1 exists (20), 10<20, not good
    # nums[1]=20: i-1=0 exists (10), i+1=2 exists (30), 20>10 and 20<30, not good
    # nums[2]=30: i+1=3 doesn't exist, i-1=1 exists (20), 30>20, good
    # sum = 30
    assert sumOfGoodNumbers([10, 20, 30], 1) == 30

    # nums = [1, 2, 3, 4], k = 2

    # nums[1]=2: i-2=-1 doesn't exist, i+2=3 exists (4), 2<4, not good
    # nums[2]=3: i-2=0 exists (1), i+2=4 doesn't exist, 3>1, good

    # sum = 3 + 4 = 7
    assert sumOfGoodNumbers([1, 2, 3, 4], 2) == 7

    # nums[1]=200: i-2=-1 doesn't exist, i+2=3 doesn't exist, good (neither exists)
    # sum = 100 + 200 = 300

    # all elements equal, so none is strictly greater than neighbors
    # nums[0]=5: i-1 doesn't exist, i+1=1 (5), 5==5 not strictly greater, not good
    # nums[1]=5: i-1=0 (5), i+1=2 (5), 5==5, not good
    # nums[2]=5: i-1=1 (5), i+1=3 (5), 5==5, not good
    # nums[3]=5: i+1 doesn't exist, i-1=2 (5), 5==5, not good

    assert sumOfGoodNumbers([5, 5, 5, 5], 1) == 0
