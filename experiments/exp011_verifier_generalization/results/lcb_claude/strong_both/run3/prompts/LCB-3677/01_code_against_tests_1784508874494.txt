```python
from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[-float('inf')] * (n + 1) for _ in range(m + 1)]
    dp[0][1] = 0  # Starting point, before the first cell

    for i in range(m):
        for j in range(n):
            coin = coins[i][j]
            # Calculate current cell's value
            if coin >= 0:
                # If non-negative, just add it
                dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j + 1] + coin, dp[i + 1][j] + coin)
            else:
                # If negative, we potentially lose coins
                # We can neutralize this robber if we haven't used both
                for used in range(3):  # 0, 1, or 2 neutralizations used
                    if used < 2:
                        dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j + 1] - coin, dp[i + 1][j] - coin)
                    # Continuing without neutralization
                    dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j + 1] + coin, dp[i + 1][j] + coin)

    return max(dp[m][n], 0)  # Return maximum coins or 0 if all are negative
```
