from typing import List
import pytest
from solution import countPartitions


def test_canonical_example_1():
    # spec: nums = [10,10,3,7,6] -> 4
    # Partitions:
    # i=0: [10] vs [10,3,7,6], sums: 10 vs 26, diff: -16 (even)
    # i=1: [10,10] vs [3,7,6], sums: 20 vs 16, diff: 4 (even)
    # i=2: [10,10,3] vs [7,6], sums: 23 vs 13, diff: 10 (even)
    # i=3: [10,10,3,7] vs [6], sums: 30 vs 6, diff: 24 (even)
    assert countPartitions([10, 10, 3, 7, 6]) == 4
