from solution import maxDifference


def test_canonical_example_1():
    # spec: s = "aaaaabbc"
    # 'a' has frequency 5 (odd), 'b' has frequency 2 (even), 'c' has frequency 1 (odd)
    # max difference = 5 - 2 = 3
    assert maxDifference("aaaaabbc") == 3


def test_canonical_example_2():
    # spec: s = "abcabcab"
    # 'a' has frequency 3 (odd), 'b' has frequency 3 (odd), 'c' has frequency 2 (even)
    # max difference = 3 - 2 = 1
    assert maxDifference("abcabcab") == 1


def test_single_char_odd_single_char_even():
    # s = "aab"
    # 'a' has frequency 2 (even), 'b' has frequency 1 (odd)
    # max difference = 1 - 2 = -1
    assert maxDifference("aab") == -1


def test_high_odd_low_even():
    # s = "aaaaaabb"
    # 'a' has frequency 7 (odd), 'b' has frequency 2 (even)
    # max difference = 7 - 2 = 5
    assert maxDifference("aaaaaabb") == 5


def test_low_odd_high_even():
    # s = "aaaaabbbbb"
    # 'a' has frequency 5 (odd), 'b' has frequency 5 (odd)
    # Need at least one even frequency
    # s = "aaaabbbb"
    # 'a' has frequency 4 (even), 'b' has frequency 4 (even)
    # Need at least one odd frequency
    # s = "aaaabbbbc"
    # 'a' has frequency 4 (even), 'b' has frequency 4 (even), 'c' has frequency 1 (odd)
    # max difference = 1 - 4 = -3
    assert maxDifference("aaaabbbbc") == -3


def test_multiple_chars_various_frequencies():
    # s = "aabbcccddd"
    # 'a' has frequency 2 (even), 'b' has frequency 2 (even), 'c' has frequency 3 (odd), 'd' has frequency 3 (odd)
    # max difference = 3 - 2 = 1
    assert maxDifference("aabbcccddd") == 1


def test_minimum_length_string():
    # s = "aab" (length 3, minimum constraint)
    # 'a' has frequency 2 (even), 'b' has frequency 1 (odd)
    # max difference = 1 - 2 = -1
    assert maxDifference("aab") == -1


def test_all_same_except_one():
    # s = "aaaaaab"
    # 'a' has frequency 6 (even), 'b' has frequency 1 (odd)
    # max difference = 1 - 6 = -5
    assert maxDifference("aaaaaab") == -5

    # s = "aabbccddde"
