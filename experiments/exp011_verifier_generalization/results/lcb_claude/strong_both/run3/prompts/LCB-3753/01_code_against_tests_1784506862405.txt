## Implementation

```python
from collections import Counter

def maxDifference(s: str) -> int:
    frequencies = Counter(s)
    odd_freq = [freq for freq in frequencies.values() if freq % 2 == 1]
    even_freq = [freq for freq in frequencies.values() if freq % 2 == 0]
    # Safeguard against empty lists
    max_odd = max(odd_freq) if odd_freq else 0
    min_even = min(even_freq) if even_freq else 0
    return max_odd - min_even
```
