from typing import List
import pytest
from solution import longestSubsequence


def test_canonical_example_1():
    # spec: nums = [16,6,3] -> 3
    # The longest subsequence is [16, 6, 3] with absolute adjacent differences [10, 3]
    # 10 >= 3, so non-increasing
    assert longestSubsequence([16, 6, 3]) == 3


def test_canonical_example_2():
    # spec: nums = [6,5,3,4,2,1] -> 4
    # The longest subsequence is [6, 4, 2, 1] with absolute adjacent differences [2, 2, 1]
    # 2 >= 2 >= 1, so non-increasing
    assert longestSubsequence([6, 5, 3, 4, 2, 1]) == 4


def test_canonical_example_3():
    # spec: nums = [10,20,10,19,10,20] -> 5
    # The longest subsequence is [10, 20, 10, 19, 10] with absolute adjacent differences [10, 10, 9, 9]
    # 10 >= 10 >= 9 >= 9, so non-increasing
    assert longestSubsequence([10, 20, 10, 19, 10, 20]) == 5


def test_minimum_length_array():
    # spec: 2 <= nums.length
    # Minimum case: array of length 2, any subsequence of length 2 is valid
    # since there's only one difference and it's trivially non-increasing
    assert longestSubsequence([1, 2]) == 2


def test_all_same_elements():
    # All elements are the same, so all differences are 0
    # The entire array forms a valid subsequence with differences [0, 0, 0]
    # 0 >= 0 >= 0, so non-increasing
    assert longestSubsequence([5, 5, 5, 5]) == 4


def test_strictly_increasing_values():
    # nums = [1, 2, 3, 4, 5]
    # Subsequence [1, 2, 3, 4, 5] has differences [1, 1, 1, 1]
    # 1 >= 1 >= 1 >= 1, so the entire array is valid
    assert longestSubsequence([1, 2, 3, 4, 5]) == 5


def test_strictly_decreasing_values():
    # nums = [5, 4, 3, 2, 1]
    # Subsequence [5, 4, 3, 2, 1] has differences [1, 1, 1, 1]
    # 1 >= 1 >= 1 >= 1, so the entire array is valid
    assert longestSubsequence([5, 4, 3, 2, 1]) == 5


def test_increasing_then_decreasing_differences():
    # nums = [1, 2, 4, 5]
    # Subsequence [1, 2, 4, 5] has differences [1, 2, 1]
    # 1 >= 2 is false, so we need a different subsequence
    # Subsequence [1, 2, 4] has differences [1, 2] -> 1 >= 2 is false
    # Subsequence [1, 4, 5] has differences [3, 1] -> 3 >= 1 is true, length 3
    # Subsequence [2, 4, 5] has differences [2, 1] -> 2 >= 1 is true, length 3
    assert longestSubsequence([1, 2, 4, 5]) == 3


def test_large_initial_difference():
    # nums = [1, 100, 101]
    # Subsequence [1, 100, 101] has differences [99, 1] -> 99 >= 1, valid, length 3
    assert longestSubsequence([1, 100, 101]) == 3


def test_alternating_pattern():
    # nums = [1, 10, 2, 9, 3]
    # Subsequence [1, 10, 2, 9] has differences [9, 8, 7] -> 9 >= 8 >= 7, valid, length 4
    assert longestSubsequence([1, 10, 2, 9, 3]) == 4


def test_single_large_jump():
    # nums = [1, 2, 3, 100]
    # Subsequence [1, 2, 3] has differences [1, 1] -> valid, length 3
    # Subsequence [3, 100] has difference [97] -> valid, length 2
    # Subsequence [1, 100] has difference [99] -> valid, length 2
    # Subsequence [1, 2, 100] has differences [1, 98] -> 1 >= 98 is false
    # Best is [1, 2, 3] with length 3
    assert longestSubsequence([1, 2, 3, 100]) == 3


def test_boundary_values():
    # spec: 1 <= nums[i] <= 300
    # Test with boundary values
    # nums = [1, 300, 1]
    # Subsequence [1, 300, 1] has differences [299, 299] -> 299 >= 299, valid, length 3
    assert longestSubsequence([1, 300, 1]) == 3


def test_repeated_differences():
    # nums = [1, 3, 5, 7, 9]
    # Differences are all 2: [2, 2, 2, 2]
    # 2 >= 2 >= 2 >= 2, so entire array is valid
    assert longestSubsequence([1, 3, 5, 7, 9]) == 5
