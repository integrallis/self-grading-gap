def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n < 2:
        return n
    # Create a DP array to store the length of the longest subsequence ending at each index
    dp = [1] * n

    # Iterate through each element to build the DP solution
    for i in range(n):
        for j in range(i):
            # Start building the subsequence from j to i
            current_length = 2
            prev_diff = abs(nums[i] - nums[j])
            last_idx = i

            # Check subsequent elements to maintain non-increasing order of differences
            for k in range(i + 1, n):
                if abs(nums[k] - nums[last_idx]) <= prev_diff:
                    current_length += 1
                    prev_diff = abs(nums[k] - nums[last_idx])
                    last_idx = k
