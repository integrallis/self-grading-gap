import pytest
from solution import hasSpecialSubstring


def test_canonical_example_1():
    # spec: s = "aaabaaa", k = 3 -> true
    # The substring s[4..6] == "aaa" satisfies all conditions
    assert hasSpecialSubstring("aaabaaa", 3) is True


def test_canonical_example_2():
    # spec: s = "abc", k = 2 -> false
    # No substring of length 2 consists of one distinct character
    assert hasSpecialSubstring("abc", 2) is False


def test_single_character_string_k_equals_1():
    # s = "a", k = 1 -> the substring "a" has length 1, all same char,
    # no char before or after, so conditions are satisfied
    assert hasSpecialSubstring("a", 1) is True


def test_all_same_characters_k_equals_length():
    # s = "aaaa", k = 4 -> entire string is one char, no char before/after
    assert hasSpecialSubstring("aaaa", 4) is True


def test_all_same_characters_k_less_than_length():
    # s = "aaaa", k = 2 -> substring "aa" at positions [0:2] or [2:4]
    # At [0:2]: no char before, char after is 'a' (same), fails
    # At [1:3]: char before is 'a' (same), fails
    # At [2:4]: char before is 'a' (same), no char after, fails
    # All fail because they're surrounded by same character
    assert hasSpecialSubstring("aaaa", 2) is False


def test_special_substring_at_start():
    # s = "aaab", k = 3 -> substring "aaa" at [0:3]
    # All same char, no char before, char after is 'b' (different)
    assert hasSpecialSubstring("aaab", 3) is True


def test_special_substring_at_end():
    # s = "baaa", k = 3 -> substring "aaa" at [1:4]
    # All same char, char before is 'b' (different), no char after
    assert hasSpecialSubstring("baaa", 3) is True


def test_special_substring_in_middle():
    # s = "baaac", k = 3 -> substring "aaa" at [1:4]
    # All same char, char before is 'b' (different), char after is 'c' (different)
    assert hasSpecialSubstring("baaac", 3) is True


def test_substring_with_different_chars():
    # s = "abcd", k = 2 -> no substring of length 2 has all same chars
    assert hasSpecialSubstring("abcd", 2) is False


def test_repeating_pattern_no_valid_substring():
    # s = "ababab", k = 2 -> substrings of length 2 are "ab", "ba", "ab", "ba", "ab"
    # None consist of one distinct character
    assert hasSpecialSubstring("ababab", 2) is False


def test_multiple_runs_first_fails_second_succeeds():
    # s = "aaabbb", k = 3 -> "aaa" at [0:3] has char after 'b' (different), no char before
    # Also "bbb" at [3:6] has char before 'a' (different), no char after
    assert hasSpecialSubstring("aaabbb", 3) is True


def test_run_too_short():
    # s = "aabaa", k = 3 -> runs are "aa", "b", "aa" - none have length >= 3
    # Can't form a valid substring of length 3 with all same chars
    assert hasSpecialSubstring("aabaa", 3) is False


def test_exact_run_but_surrounded_by_same():
    # s = "aaaaa", k = 3 -> any substring of length 3:
    # [0:3] "aaa": no char before, char after is 'a' (same) - fails
    # [1:4] "aaa": char before is 'a' (same) - fails
    # [2:5] "aaa": char before is 'a' (same) - fails
    assert hasSpecialSubstring("aaaaa", 3) is False


def test_k_equals_1_with_different_neighbors():
    # s = "aba", k = 1 -> substring "b" at [1:2]
    # All same (trivially), char before is 'a' (different), char after is 'a' (different)
    assert hasSpecialSubstring("aba", 1) is True


def test_k_equals_1_at_boundary():
    # s = "ab", k = 1 -> substring "a" at [0:1]
    # All same, no char before, char after is 'b' (different)
    assert hasSpecialSubstring("ab", 1) is True
