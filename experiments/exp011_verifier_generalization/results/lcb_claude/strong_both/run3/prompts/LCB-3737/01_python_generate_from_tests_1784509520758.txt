from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    dp = [[float('inf')] * 3 for _ in range(n)]
    # Initialize the first house's costs
    for j in range(3):
        dp[0][j] = cost[0][j]
    # Fill the DP table
    for i in range(1, n):
        for j in range(3):
            for k in range(3):
                if j != k and (i < n // 2 or (n - 1 - i) // 2 != (n // 2 - 1 - j) // 2):
                    dp[i][j] = min(dp[i][j], dp[i-1][k] + cost[i][j])
    # Find the minimum cost considering the restrictions
    min_cost = float('inf')
    for j in range(3):
        if j != (n - 1) // 2:
            min_cost = min(min_cost, dp[n-1][j])
    return min_cost

```
