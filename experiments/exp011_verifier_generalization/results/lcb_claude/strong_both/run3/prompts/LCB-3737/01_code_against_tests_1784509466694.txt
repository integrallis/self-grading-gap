from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    dp = [[0] * 3 for _ in range(n)]
    # Initialize the first house's costs
    for j in range(3):
        dp[0][j] = cost[0][j]
    # Fill the DP table
    for i in range(1, n):
        for j in range(3):
            # Calculate the minimum cost for painting house i with color j
            dp[i][j] = min(
                dp[i-1][(j+1) % 3],
                dp[i-1][(j+2) % 3]
            ) + cost[i][j]
    # Get the minimum cost for the first half of the houses
    min_cost_first_half = min(dp[n//2-1])
    # Reset DP for the second half by keeping the first half costs in mind
    for j in range(3):
        dp[0][j] = cost[0][j]
    for i in range(1, n // 2):
        for j in range(3):
            dp[i][j] = min(
                dp[i-1][(j+1) % 3],
                dp[i-1][(j+2) % 3]
            ) + cost[i][j]
    # Get the minimum costs for the second half considering the first half
    min_cost_second_half = float('inf')
    for j in range(3):
        for k in range(3):
            if j != k:
                min_cost_second_half = min(min_cost_second_half, dp[n//2-1][j] + dp[n-1][k])
    return min(min_cost_first_half, min_cost_second_half)
