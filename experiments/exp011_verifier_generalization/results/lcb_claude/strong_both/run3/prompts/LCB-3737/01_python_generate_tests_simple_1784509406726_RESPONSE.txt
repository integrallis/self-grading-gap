def test_canonical_example_1():
    # Example 1: n=4, optimal painting [1,2,3,2] with costs [3,2,1,3], total=9

    assert minCost(n, cost) == 9

def test_minimal_case_two_houses():
    # n=2: two houses at positions 0 and 1 are adjacent and equidistant
    # Must paint different colors. Min cost among valid pairs.
    # cost[0][0]=1, cost[1][1]=2 -> colors 1,2, cost=3
    # cost[0][0]=1, cost[1][2]=3 -> colors 1,3, cost=4
    # cost[0][1]=4, cost[1][0]=5 -> colors 2,1, cost=9
    # cost[0][1]=4, cost[1][2]=3 -> colors 2,3, cost=7
    # cost[0][2]=6, cost[1][0]=5 -> colors 3,1, cost=11
    # cost[0][2]=6, cost[1][1]=2 -> colors 3,2, cost=8
    # Minimum is 3
    n = 2
    cost = [[1, 4, 6], [5, 2, 3]]
    assert minCost(n, cost) == 3
