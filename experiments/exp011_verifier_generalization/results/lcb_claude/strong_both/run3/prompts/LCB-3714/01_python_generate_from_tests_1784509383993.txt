from typing import List


def minMaxSums(nums: List[int], k: int) -> int:
    MOD = 10**9 + 7
    n = len(nums)
    total_sum = 0
    nums.sort()

    # Calculate the contribution of each number as minimum and maximum
    for i in range(n):
        min_contrib = nums[i] * (pow(2, min(k, n - 1), MOD) - 1) % MOD
        max_contrib = nums[i] * (pow(2, min(k, n - 1), MOD) - 1) % MOD
        total_sum = (total_sum + min_contrib + max_contrib) % MOD

    total_sum = (total_sum + sum(nums) * (pow(2, k, MOD) - 1)) % MOD
    return total_sum % MOD
```
