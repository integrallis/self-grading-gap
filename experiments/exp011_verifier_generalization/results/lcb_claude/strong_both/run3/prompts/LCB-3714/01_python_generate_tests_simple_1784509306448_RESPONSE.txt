from solution import minMaxSums
from typing import List


def test_canonical_example_1():
    # spec: nums = [1,2,3], k = 2 -> 24
    # Subsequences with at most 2 elements:
    # [1]: min=1, max=1, sum=2
    # [2]: min=2, max=2, sum=4
    # [3]: min=3, max=3, sum=6
    # [1,2]: min=1, max=2, sum=3
    # [1,3]: min=1, max=3, sum=4
    # [2,3]: min=2, max=3, sum=5
    # Total: 2+4+6+3+4+5 = 24
    assert minMaxSums([1, 2, 3], 2) == 24


def test_canonical_example_2():
    # spec: nums = [5,0,6], k = 1 -> 22
    # Subsequences with at most 1 element:
    # [5]: min=5, max=5, sum=10
    # [0]: min=0, max=0, sum=0
    # [6]: min=6, max=6, sum=12
    # Total: 10+0+12 = 22
    assert minMaxSums([5, 0, 6], 1) == 22


def test_canonical_example_3():
    # spec: nums = [1,1,1], k = 2 -> 12
    # Subsequences with at most 2 elements:
    # [1] appears 3 times: 3 * (1+1) = 6
    # [1,1] appears 3 times: 3 * (1+1) = 6
    # Total: 6+6 = 12
    assert minMaxSums([1, 1, 1], 2) == 12


def test_single_element_array_k_1():
    # Single element [5], k=1
    # Subsequence [5]: min=5, max=5, sum=10
    assert minMaxSums([5], 1) == 10


def test_two_elements_k_1():
    # nums = [2, 4], k = 1
    # [2]: min=2, max=2, sum=4
    # [4]: min=4, max=4, sum=8
    # Total: 12
    assert minMaxSums([2, 4], 1) == 12


def test_two_elements_k_2():
    # nums = [2, 4], k = 2
    # [2]: min=2, max=2, sum=4
    # [4]: min=4, max=4, sum=8
    # [2,4]: min=2, max=4, sum=6
    # Total: 18
    assert minMaxSums([2, 4], 2) == 18


def test_k_equals_array_length():
    # nums = [1, 2, 3], k = 3
    # All possible subsequences with at most 3 elements
    # Size 1: [1]=2, [2]=4, [3]=6
    # Size 2: [1,2]=3, [1,3]=4, [2,3]=5
    # Size 3: [1,2,3]=1+3=4
    # Total: 2+4+6+3+4+5+4 = 28
    assert minMaxSums([1, 2, 3], 3) == 28


def test_with_zeros():
    # nums = [0, 1], k = 2
    # [0]: min=0, max=0, sum=0
    # [1]: min=1, max=1, sum=2
    # [0,1]: min=0, max=1, sum=1
    # Total: 3
    assert minMaxSums([0, 1], 2) == 3


def test_all_same_elements():
    # nums = [5, 5], k = 2
    # [5] appears 2 times: 2 * (5+5) = 20
    # [5,5]: min=5, max=5, sum=10
    # Total: 30
    assert minMaxSums([5, 5], 2) == 30


def test_large_values_modulo():
    # Test that modulo 10^9 + 7 is applied
    # nums = [1000000000, 1000000000], k = 1
    # [1000000000] appears 2 times: 2 * (1000000000 + 1000000000) = 4000000000
    # 4000000000 % (10^9 + 7) = 4000000000 - 3*(10^9+7) = 4000000000 - 3000000021 = 999999979
    assert minMaxSums([1000000000, 1000000000], 1) == 999999979
