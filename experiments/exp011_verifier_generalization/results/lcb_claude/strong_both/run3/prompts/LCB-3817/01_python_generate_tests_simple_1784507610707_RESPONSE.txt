import pytest
from solution import maxContainers


def test_canonical_example_1():
    # spec: n=2, w=3, maxWeight=15
    # deck has 2*2=4 cells, each container weighs 3
    # total weight if all loaded: 4*3=12 <= 15, so can load all 4
    assert maxContainers(2, 3, 15) == 4


def test_canonical_example_2():
    # spec: n=3, w=5, maxWeight=20
    # deck has 3*3=9 cells, each container weighs 5
    # max containers without exceeding 20: floor(20/5)=4
    # 4*5=20 <= 20, so answer is 4
    assert maxContainers(3, 5, 20) == 4


def test_single_cell_deck():
    # n=1 means 1*1=1 cell deck
    # w=10, maxWeight=10: can fit 1 container (1*10=10 <= 10)
    assert maxContainers(1, 10, 10) == 1


def test_weight_limit_smaller_than_deck_capacity():
    # n=5 means 5*5=25 cells, w=2, maxWeight=30
    # max containers: floor(30/2)=15 (limited by weight, not deck size)
    assert maxContainers(5, 2, 30) == 15


def test_weight_limit_allows_all_containers():
    # n=3 means 3*3=9 cells, w=1, maxWeight=100
    # can fit all 9 containers (9*1=9 <= 100)
    assert maxContainers(3, 1, 100) == 9


def test_weight_limit_exactly_fits_all_containers():
    # n=2 means 2*2=4 cells, w=7, maxWeight=28
    # all 4 containers fit exactly (4*7=28 <= 28)
    assert maxContainers(2, 7, 28) == 4


def test_weight_limit_allows_zero_containers():
    # n=2 means 2*2=4 cells, w=100, maxWeight=50
    # cannot fit even 1 container (1*100=100 > 50), so 0
    assert maxContainers(2, 100, 50) == 0


def test_large_deck_small_weight():
    # n=10 means 10*10=100 cells, w=1, maxWeight=50
    # can fit 50 containers (50*1=50 <= 50)
    assert maxContainers(10, 1, 50) == 50


def test_weight_exactly_one_container():
    # n=5 means 5*5=25 cells, w=15, maxWeight=15
    # can fit exactly 1 container (1*15=15 <= 15)
    assert maxContainers(5, 15, 15) == 1


def test_minimum_inputs():
    # n=1, w=1, maxWeight=1 (all minimum constraint values)
    # 1*1=1 cell, 1 container of weight 1, total=1 <= 1
    assert maxContainers(1, 1, 1) == 1


def test_large_maxWeight():
    # n=4 means 4*4=16 cells, w=10, maxWeight=1000000000
    # all 16 containers fit (16*10=160 <= 1000000000)
    assert maxContainers(4, 10, 1000000000) == 16
