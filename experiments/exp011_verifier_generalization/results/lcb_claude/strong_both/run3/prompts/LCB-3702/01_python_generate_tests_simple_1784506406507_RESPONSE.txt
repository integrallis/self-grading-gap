from typing import List
import pytest
from solution import maxLength


def test_canonical_example_1():
    # spec: nums = [1,2,1,2,1,1,1] -> 5
    # The longest product equivalent subarray is [1, 2, 1, 1, 1]
    assert maxLength([1, 2, 1, 2, 1, 1, 1]) == 5


def test_canonical_example_2():
    # spec: nums = [2,3,4,5,6] -> 3
    # The longest product equivalent subarray is [3, 4, 5]
    assert maxLength([2, 3, 4, 5, 6]) == 3


def test_canonical_example_3():
    # spec: nums = [1,2,3,1,4,5,1] -> 5
    assert maxLength([1, 2, 3, 1, 4, 5, 1]) == 5


def test_minimum_length_array():
    # spec: 2 <= nums.length, so test with length 2
    # [1, 1]: prod=1, gcd=1, lcm=1, 1 == 1*1 -> product equivalent
    assert maxLength([1, 1]) == 2


def test_single_element_subarray():
    # For a single element x: prod=x, gcd=x, lcm=x, so x == x*x only if x=1
    # [2, 3]: length-1 subarrays fail (2 != 2*2, 3 != 3*3)
    # [2, 3]: prod=6, gcd=1, lcm=6, 6 == 1*6 -> product equivalent
    assert maxLength([2, 3]) == 2


def test_all_ones():
    # All 1s: for any subarray, prod=1, gcd=1, lcm=1, 1 == 1*1 -> always product equivalent
    # So the entire array should be the answer
    assert maxLength([1, 1, 1, 1]) == 4


def test_two_element_product_equivalent():
    # [2, 2]: prod=4, gcd=2, lcm=2, 4 == 2*2 -> product equivalent
    assert maxLength([2, 2]) == 2


def test_two_element_not_product_equivalent():
    # [2, 4]: prod=8, gcd=2, lcm=4, 8 == 2*4 -> product equivalent
    assert maxLength([2, 4]) == 2


def test_three_coprime_numbers():
    # [2, 3, 5]: prod=30, gcd=1, lcm=30, 30 == 1*30 -> product equivalent
    assert maxLength([2, 3, 5]) == 3


def test_with_max_value():
    # spec: 1 <= nums[i] <= 10
    # [10, 10]: prod=100, gcd=10, lcm=10, 100 == 10*10 -> product equivalent
    assert maxLength([10, 10]) == 2


def test_array_with_no_long_product_equivalent():
    # [5, 6, 7, 8, 9]: need to find longest product equivalent subarray
    # Individual elements (except 1) don't satisfy x == x*x
    # [5, 6]: prod=30, gcd=1, lcm=30, 30 == 1*30 -> product equivalent
    assert maxLength([5, 6, 7, 8, 9]) >= 2


def test_mixed_with_ones():
    # [1, 3, 1]: prod=3, gcd=1, lcm=3, 3 == 1*3 -> product equivalent
    assert maxLength([1, 3, 1]) == 3


def test_longer_array_all_same():
    # [3, 3, 3]: prod=27, gcd=3, lcm=3, 27 != 3*3=9 -> not product equivalent
    # But [3, 3]: prod=9, gcd=3, lcm=3, 9 == 3*3 -> product equivalent
    assert maxLength([3, 3, 3]) == 2
