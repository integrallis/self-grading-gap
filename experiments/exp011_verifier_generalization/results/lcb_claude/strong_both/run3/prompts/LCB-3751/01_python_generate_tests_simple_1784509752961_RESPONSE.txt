from typing import List
import pytest
from solution import maxFrequency


def test_canonical_example_1():
    # spec: nums = [1,2,3,4,5,6], k = 1 -> 2
    # After adding -5 to nums[2..5], 1 has a frequency of 2 in [1, 2, -2, -1, 0, 1]
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2


def test_canonical_example_2():
    # spec: nums = [10,2,3,4,5,5,4,3,2,2], k = 10 -> 4
    # After adding 8 to nums[1..9], 10 has a frequency of 4 in [10, 10, 11, 12, 13, 13, 12, 11, 10, 10]
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4


def test_single_element_already_equal_to_k():
    # nums = [5], k = 5
    # No operation needed, frequency is already 1
    assert maxFrequency([5], 5) == 1


def test_single_element_not_equal_to_k():
    # nums = [3], k = 7
    # Add 4 to nums[0..0], result is [7], frequency is 1
    assert maxFrequency([3], 7) == 1


def test_all_elements_already_equal_to_k():
    # nums = [3, 3, 3], k = 3
    # No operation needed (or add 0 to any subarray), frequency is 3
    assert maxFrequency([3, 3, 3], 3) == 3


def test_all_elements_same_but_not_k():
    # nums = [5, 5, 5], k = 10
    # Add 5 to nums[0..2], result is [10, 10, 10], frequency is 3
    assert maxFrequency([5, 5, 5], 10) == 3


def test_modify_prefix_subarray():
    # nums = [2, 2, 5], k = 7
    # Add 5 to nums[0..1], result is [7, 7, 5], frequency is 2
    assert maxFrequency([2, 2, 5], 7) == 2


def test_modify_suffix_subarray():
    # nums = [5, 2, 2], k = 7
    # Add 5 to nums[1..2], result is [5, 7, 7], frequency is 2
    assert maxFrequency([5, 2, 2], 7) == 2


def test_modify_middle_subarray():
    # nums = [1, 2, 2, 1], k = 5
    # Add 3 to nums[1..2], result is [1, 5, 5, 1], frequency is 2
    assert maxFrequency([1, 2, 2, 1], 5) == 2


def test_no_modification_needed():
    # nums = [7, 3, 7, 1, 7], k = 7
    # No operation needed (or add 0), frequency is already 3
    assert maxFrequency([7, 3, 7, 1, 7], 7) == 3


def test_maximize_with_existing_k_values():
    # nums = [5, 3, 3, 5], k = 5
    # Initially has 2 occurrences of 5
    # Add 2 to nums[1..2], result is [5, 5, 5, 5], frequency is 4
    assert maxFrequency([5, 3, 3, 5], 5) == 4


def test_two_elements_different():
    # nums = [1, 2], k = 3
    # Add 2 to nums[0..0], result is [3, 2], frequency is 1
    # Or add 1 to nums[1..1], result is [1, 3], frequency is 1
    # Or add 2 to nums[0..1], result is [3, 4], frequency is 1
    assert maxFrequency([1, 2], 3) == 1


def test_two_elements_same():
    # nums = [4, 4], k = 10
    # Add 6 to nums[0..1], result is [10, 10], frequency is 2
    assert maxFrequency([4, 4], 10) == 2


def test_scattered_values():
    # nums = [1, 5, 1, 5, 1], k = 1
    # Already has 3 occurrences of 1
    # Or add -4 to nums[1..3], result is [1, 1, 1, 1, 1], frequency is 5
    assert maxFrequency([1, 5, 1, 5, 1], 1) == 5
