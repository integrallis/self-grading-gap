from typing import List


def maxFrequency(nums: List[int], k: int) -> int:
    count = 0
    for num in nums:
        if num == k:
            count += 1
    return count + sum((k - num) for num in nums if num < k) // (k - min((num for num in nums if num < k), default=1))
