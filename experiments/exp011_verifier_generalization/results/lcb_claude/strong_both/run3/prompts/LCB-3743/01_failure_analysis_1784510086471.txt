from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_intervals = []
    last_end = 0

    for i in range(n):
        if startTime[i] > last_end:
            free_intervals.append((last_end, startTime[i]))
        last_end = max(last_end, endTime[i])

    if last_end < eventTime:
        free_intervals.append((last_end, eventTime))

    max_free = 0
    current_gaps = [end - start for start, end in free_intervals]

    for i in range(len(current_gaps)):
        if k > 0:
            if i < len(current_gaps) - 1:
                combined_gap = current_gaps[i] + current_gaps[i + 1]
                max_free = max(max_free, combined_gap)
                k -= 1
            else:
                max_free = max(max_free, current_gaps[i])
        else:
            max_free = max(max_free, current_gaps[i])

    return max_free
