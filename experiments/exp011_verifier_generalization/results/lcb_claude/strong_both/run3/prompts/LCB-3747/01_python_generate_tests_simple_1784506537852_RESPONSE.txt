from solution import maxAdjacentDistance
from typing import List


def test_canonical_example_1():
    # spec: nums = [1,2,4] -> 3
    # adjacent pairs: (1,2), (2,4), (4,1)
    # differences: |1-2|=1, |2-4|=2, |4-1|=3
    # maximum is 3
    assert maxAdjacentDistance([1, 2, 4]) == 3


def test_canonical_example_2():
    # spec: nums = [-5,-10,-5] -> 5
    # adjacent pairs: (-5,-10), (-10,-5), (-5,-5)
    # differences: |-5-(-10)|=5, |-10-(-5)|=5, |-5-(-5)|=0
    # maximum is 5
    assert maxAdjacentDistance([-5, -10, -5]) == 5


def test_two_elements_positive():
    # nums = [1, 5]

    # maximum is 4
    assert maxAdjacentDistance([1, 5]) == 4

    # nums = [-10, -3]

    # maximum is 7

    # nums = [0, 1, 2, 10]

    # differences: |0-1|=1, |1-2|=1, |2-10|=8, |10-0|=10

    # maximum is 8 (at circular boundary)
