import pytest
from solution import findValidPair


def test_canonical_example_1():
    # spec: "2523533" -> "23"
    # Digit '2' appears 2 times and digit '3' appears 3 times.
    # Each digit in the pair "23" appears in s exactly as many times as its numeric value.
    assert findValidPair("2523533") == "23"


def test_canonical_example_2():
    # spec: "221" -> "21"
    # Digit '2' appears 2 times and digit '1' appears 1 time.
    assert findValidPair("221") == "21"


def test_canonical_example_3():
    # spec: "22" -> ""
    # There are no valid adjacent pairs.
    assert findValidPair("22") == ""


def test_first_valid_pair_when_multiple_exist():
    # "1123" has pairs "11", "12", "23"
    # "11" - invalid (digits are equal)
    # "12" - '1' appears 2 times (should be 1), invalid
    # "23" - '2' appears 1 time (should be 2), invalid
    # No valid pair
    assert findValidPair("1123") == ""


def test_valid_pair_at_beginning():
    # "1233" has pairs "12", "23", "33"
    # "12" - '1' appears 1 time, '2' appears 1 time (should be 2), invalid
    # "23" - '2' appears 1 time (should be 2), invalid
    # "33" - invalid (digits are equal)
    assert findValidPair("1233") == ""


def test_valid_pair_in_middle():
    # "31123" 
    # '1' appears 2 times, '2' appears 1 time, '3' appears 2 times
    # pairs: "31", "11", "12", "23"
    # "31" - '3' appears 2 times (should be 3), invalid
    # "11" - invalid (digits are equal)
    # "12" - '1' appears 2 times (should be 1), invalid
    # "23" - '2' appears 1 time (should be 2), invalid
    assert findValidPair("31123") == ""


def test_minimum_length_string():
    # "12" - '1' appears 1 time, '2' appears 1 time (should be 2)
    # invalid
    assert findValidPair("12") == ""


def test_all_same_digit():
    # "333" - all pairs have equal digits, invalid
    assert findValidPair("333") == ""


def test_valid_pair_with_digit_1():
    # "112" - '1' appears 2 times (should be 1), '2' appears 1 time (should be 2)
    # pairs: "11", "12"
    # "11" - invalid (equal digits)
    # "12" - both invalid counts
    assert findValidPair("112") == ""


def test_valid_pair_different_positions():
    # "12211" 
    # '1' appears 3 times, '2' appears 2 times
    # pairs: "12", "22", "21", "11"
    # "12" - '1' appears 3 times (should be 1), invalid
    # "22" - invalid (equal digits)
    # "21" - '2' appears 2 times (valid), '1' appears 3 times (should be 1), invalid
    assert findValidPair("12211") == ""


def test_valid_pair_at_end():
    # "3321"
    # '1' appears 1 time, '2' appears 1 time, '3' appears 2 times
    # pairs: "33", "32", "21"
    # "33" - invalid (equal digits)
    # "32" - '3' appears 2 times (should be 3), invalid
    # "21" - '2' appears 1 time (should be 2), invalid
    assert findValidPair("3321") == ""


def test_longer_string_with_valid_pair():
    # "4444123"
    # '1' appears 1 time, '2' appears 1 time, '3' appears 1 time, '4' appears 4 times
    # First different adjacent pair: "44" (invalid, equal)
    # "44" invalid, "44" invalid, "41" - '4' appears 4 times (valid), '1' appears 1 time (valid)
    # "41" is valid!
    assert findValidPair("4444123") == "41"
