```python
from collections import Counter

def findValidPair(s: str) -> str:
    count = Counter(s)
    for i in range(len(s) - 1):
        if s[i] != s[i + 1]:  # Check if adjacent digits are different
            first_digit = int(s[i])
            second_digit = int(s[i + 1])
            if count[s[i]] == first_digit and count[s[i + 1]] == second_digit:
                return s[i:i + 2]  # Return the first valid pair found
    return ""  # Return an empty string if no valid pair is found
```
