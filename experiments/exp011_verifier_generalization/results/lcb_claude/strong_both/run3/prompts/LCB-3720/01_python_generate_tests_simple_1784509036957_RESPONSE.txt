from typing import List
import pytest
from solution import minMaxWeight


def test_canonical_example_1():
    # Example 1: n=5, edges=[[1,0,1],[2,0,2],[3,0,1],[4,3,1],[2,1,1]], threshold=2
    # Expected output: 1
    # Remove edge 2->0 (weight 2). Max weight in remaining edges is 1.
    assert minMaxWeight(5, [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]], 2) == 1


def test_canonical_example_2():
    # Example 2: n=5, edges=[[0,1,1],[0,2,2],[0,3,1],[0,4,1],[1,2,1],[1,4,1]], threshold=1
    # Expected output: -1
    # Impossible to reach node 0 from node 2
    assert minMaxWeight(5, [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]], 1) == -1


def test_canonical_example_3():
    # Example 3: n=5, edges=[[1,2,1],[1,3,3],[1,4,5],[2,3,2],[3,4,2],[4,0,1]], threshold=1
    # Expected output: 2
    # Remove edges 1->3 and 1->4. Max weight in remaining edges is 2.
    assert minMaxWeight(5, [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]], 1) == 2


def test_canonical_example_4():
    # Example 4: n=5, edges=[[1,2,1],[1,3,3],[1,4,5],[2,3,2],[4,0,1]], threshold=1
    # Expected output: -1
    # Impossible to satisfy all conditions
    assert minMaxWeight(5, [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]], 1) == -1


def test_simple_two_nodes():
    # n=2, single edge from node 1 to node 0
    # Node 0 is reachable from node 1, threshold is satisfied
    assert minMaxWeight(2, [[1, 0, 5]], 1) == 5


def test_all_nodes_directly_to_zero():
    # n=3, all nodes have direct edge to node 0
    # All nodes can reach 0, each has 1 outgoing edge
    assert minMaxWeight(3, [[1, 0, 3], [2, 0, 7]], 1) == 7


def test_threshold_constraint_forces_removal():
    # n=3, node 1 has 2 outgoing edges but threshold=1
    # Must remove one edge, choose the one with higher weight to minimize max
    assert minMaxWeight(3, [[1, 0, 5], [1, 2, 3], [2, 0, 1]], 1) == 3


def test_chain_to_zero():
    # n=4, nodes form a chain: 3->2->1->0
    # All nodes can reach 0 through the chain
    assert minMaxWeight(4, [[3, 2, 2], [2, 1, 3], [1, 0, 1]], 1) == 3


def test_multiple_edges_same_pair_different_weights():
    # Multiple edges between same pair of nodes with unique weights
    # Choose the one with minimum weight
    assert minMaxWeight(2, [[1, 0, 10], [1, 0, 5], [1, 0, 3]], 3) == 3


def test_no_path_to_zero_from_some_node():
    # n=3, node 2 has no path to node 0
    # Should return -1
    assert minMaxWeight(3, [[1, 0, 1], [2, 1, 1]], 1) == -1


def test_threshold_equals_n_minus_one():
    # threshold = n-1, very permissive
    # Node can have up to n-1 outgoing edges
    assert minMaxWeight(3, [[1, 0, 2], [2, 0, 3], [1, 2, 1]], 2) == 3


def test_single_weight_value():
    # All edges have the same weight
    # If all conditions can be satisfied, return that weight
    assert minMaxWeight(3, [[1, 0, 5], [2, 1, 5]], 1) == 5


def test_node_zero_has_outgoing_edges():
    # Node 0 can have outgoing edges, but they don't affect reachability to 0
    # Other nodes must still reach 0
    assert minMaxWeight(3, [[0, 1, 10], [1, 0, 2], [2, 0, 3]], 1) == 3
