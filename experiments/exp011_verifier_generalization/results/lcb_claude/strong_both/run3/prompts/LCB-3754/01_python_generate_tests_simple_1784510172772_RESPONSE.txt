import pytest
from solution import maxDistance


def test_canonical_example_1():
    # spec: s = "NWSE", k = 1
    # Change s[2] from 'S' to 'N' -> "NWNE"
    # Positions: (0,1), (-1,1), (-1,2), (0,2)
    # Distances: 1, 2, 3, 2
    # Maximum distance = 3
    assert maxDistance("NWSE", 1) == 3


def test_canonical_example_2():
    # spec: s = "NSWWEW", k = 3
    # Change s[1] from 'S' to 'N', s[4] from 'E' to 'W' -> "NNWWWW"
    # Maximum distance = 6
    assert maxDistance("NSWWEW", 3) == 6


def test_single_move_north_no_changes():
    # s = "N", k = 0
    # Position after move: (0, 1)
    # Distance = 1
    assert maxDistance("N", 0) == 1


def test_single_move_south_no_changes():
    # s = "S", k = 0
    # Position after move: (0, -1)
    # Distance = 1
    assert maxDistance("S", 0) == 1


def test_single_move_east_no_changes():
    # s = "E", k = 0
    # Position after move: (1, 0)
    # Distance = 1
    assert maxDistance("E", 0) == 1


def test_single_move_west_no_changes():
    # s = "W", k = 0
    # Position after move: (-1, 0)
    # Distance = 1
    assert maxDistance("W", 0) == 1


def test_opposing_moves_no_changes():
    # s = "NS", k = 0
    # Positions: (0, 1), (0, 0)
    # Distances: 1, 0
    # Maximum = 1
    assert maxDistance("NS", 0) == 1


def test_all_north_no_changes():
    # s = "NNN", k = 0
    # Positions: (0, 1), (0, 2), (0, 3)
    # Distances: 1, 2, 3
    # Maximum = 3
    assert maxDistance("NNN", 0) == 3


def test_all_same_direction_with_changes():
    # s = "NNNN", k = 4 (can change all)
    # Can keep as is or change; optimal is to keep moving in same direction
    # Maximum = 4
    assert maxDistance("NNNN", 4) == 4


def test_k_equals_string_length():
    # s = "NSEW", k = 4
    # Can change all characters to move in one direction
    # If all become N: positions (0,1), (0,2), (0,3), (0,4)
    # Maximum = 4
    assert maxDistance("NSEW", 4) == 4


def test_two_moves_one_change():
    # s = "NS", k = 1
    # Can change S to N -> "NN"
    # Positions: (0, 1), (0, 2)
    # Distances: 1, 2
    # Maximum = 2
    assert maxDistance("NS", 1) == 2


def test_mixed_directions_no_changes():
    # s = "NESW", k = 0
    # Positions: (0,1), (1,1), (1,0), (0,0)
    # Distances: 1, 2, 1, 0
    # Maximum = 2
    assert maxDistance("NESW", 0) == 2


def test_zero_changes_allowed():
    # s = "EEEE", k = 0
    # Positions: (1,0), (2,0), (3,0), (4,0)
    # Distances: 1, 2, 3, 4
    # Maximum = 4
    assert maxDistance("EEEE", 0) == 4


def test_square_path():
    # s = "NESS", k = 0
    # Positions: (0,1), (1,1), (1,0), (1,-1)
    # Distances: 1, 2, 1, 2
    # Maximum = 2
    assert maxDistance("NESS", 0) == 2


def test_diagonal_movement():
    # s = "NE", k = 0
    # Positions: (0,1), (1,1)
    # Distances: 1, 2
    # Maximum = 2
    assert maxDistance("NE", 0) == 2
