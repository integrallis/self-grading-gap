```python
def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]
    # Iterate over every possible starting position for the prefix
    for i in range(len(s) - len(prefix) + 1):
        # Check if the current position matches the prefix
        if s.startswith(prefix, i):
            # Check if the remaining string after the prefix can end with the suffix
            if s[i + len(prefix):].endswith(suffix):
                return True
    return False

```
