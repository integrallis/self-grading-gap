import pytest
from solution import hasMatch


def test_canonical_example_1():
    # spec: s = "leetcode", p = "ee*e" -> true
    # By replacing the '*' with "tcod", the substring "eetcode" matches the pattern.
    assert hasMatch("leetcode", "ee*e") is True


def test_canonical_example_2():
    # spec: s = "car", p = "c*v" -> false
    # There is no substring matching the pattern.
    assert hasMatch("car", "c*v") is False


def test_canonical_example_3():
    # spec: s = "luck", p = "u*" -> true
    # The substrings "u", "uc", and "uck" match the pattern.
    assert hasMatch("luck", "u*") is True


def test_star_at_beginning():
    # p = "*a" should match any substring ending with "a"
    # s = "ba" contains substring "a" which matches (star replaced with empty string)
    assert hasMatch("ba", "*a") is True


def test_star_at_end():
    # p = "a*" should match any substring starting with "a"
    # s = "abc" contains substring "a", "ab", "abc" which all match
    assert hasMatch("abc", "a*") is True


def test_star_in_middle():
    # p = "a*c" should match substrings starting with "a" and ending with "c"
    # s = "abcd" contains substring "abc" which matches (star replaced with "b")
    assert hasMatch("abcd", "a*c") is True


def test_star_only():
    # p = "*" can match any substring with zero or more characters
    # Since substrings are non-empty, this should match any non-empty substring
    # s = "a" contains substring "a" (star replaced with "a")
    assert hasMatch("a", "*") is True


def test_no_match_prefix():
    # p = "z*a" requires prefix "z" which doesn't exist in s
    assert hasMatch("abc", "z*a") is False


def test_no_match_suffix():
    # p = "a*z" requires suffix "z" which doesn't exist in s
    assert hasMatch("abc", "a*z") is False


def test_exact_match_with_empty_star():
    # p = "a*b" with s = "ab" matches when star is replaced with empty string
    assert hasMatch("ab", "a*b") is True


def test_multiple_possible_matches():
    # p = "a*a" with s = "aaa" has multiple matching substrings: "aa", "aaa"
    assert hasMatch("aaa", "a*a") is True


def test_pattern_longer_than_string_no_match():
    # p = "ab*cd" requires at least "abcd" (4 chars) but s = "abc" (3 chars)
    assert hasMatch("abc", "ab*cd") is False


def test_star_replacement_with_multiple_chars():
    # p = "l*k" with s = "luck" matches "luck" (star replaced with "uc")
    assert hasMatch("luck", "l*k") is True


def test_single_char_string_single_char_pattern():
    # p = "*" with s = "a" matches "a" (star replaced with "a")
    assert hasMatch("a", "*") is True


def test_pattern_not_substring():
    # p = "abc*def" requires "abc...def" as substring
    # s = "abcxyzdef" contains "abcxyzdef" which matches
    assert hasMatch("abcxyzdef", "abc*def") is True


def test_pattern_not_substring_false():
    # p = "abc*def" requires "abc...def" as substring
    # s = "abcxyz" doesn't contain suffix "def"
    assert hasMatch("abcxyz", "abc*def") is False
