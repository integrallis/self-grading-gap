from solution import hasSameDigits


def test_canonical_example_1():
    # spec: s = "3902"
    # First operation: "3902" -> "292" (pairs: 3+9=12%10=2, 9+0=9, 0+2=2)
    # Second operation: "292" -> "11" (pairs: 2+9=11%10=1, 9+2=11%10=1)
    # Final two digits are "11", both are '1', so output is true
    assert hasSameDigits("3902") is True


def test_canonical_example_2():
    # spec: s = "34789"
    # First operation: "34789" -> "7157" (3+4=7, 4+7=11%10=1, 7+8=15%10=5, 8+9=17%10=7)
    # Second operation: "7157" -> "862" (7+1=8, 1+5=6, 5+7=12%10=2)
    # Third operation: "862" -> "48" (8+6=14%10=4, 6+2=8)
    # Final two digits are "48", '4' != '8', so output is false
    assert hasSameDigits("34789") is False


def test_minimum_length_string():
    # spec: constraint is 3 <= s.length, so minimum is 3 digits
    # s = "111" -> "22" (1+1=2, 1+1=2)
    # Final two digits are "22", both are '2', so output is true
    assert hasSameDigits("111") is True
