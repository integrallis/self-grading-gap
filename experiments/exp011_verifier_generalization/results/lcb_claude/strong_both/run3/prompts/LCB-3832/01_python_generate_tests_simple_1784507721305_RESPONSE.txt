from typing import List
import pytest
from solution import minCosts


def test_canonical_example_1():
    # Example 1: cost = [5,3,4,1,3,2] -> [5,3,3,1,1,1]
    # i=0: swap with person 0, cost=5
    # i=1: swap with person 1, cost=3
    # i=2: swap with person 1 (cost=3), then swap with person 2 (free), total=3
    # i=3: swap with person 3, cost=1
    # i=4: swap with person 3 (cost=1), then swap with person 4 (free), total=1
    # i=5: swap with person 3 (cost=1), then swap with person 5 (free), total=1
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]


def test_canonical_example_2():
    # Example 2: cost = [1,2,4,6,7] -> [1,1,1,1,1]
    # Swap with person 0 (cost=1), then reach any position for free
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]


def test_single_person():
    # n=1: one person in front at position 0
    # To reach position 0, we must swap with person 0
    assert minCosts([5]) == [5]


def test_all_same_cost():
    # All persons have the same cost
    # Position 0: pay cost[0]=3
    # Position 1: pay cost[1]=3
    # Position 2: pay min(cost[0], cost[1], cost[2])=3, then free swaps
    assert minCosts([3, 3, 3]) == [3, 3, 3]


def test_decreasing_costs():
    # Costs decrease from left to right
    # Position 0: pay 10
    # Position 1: pay 5
    # Position 2: pay 1
    # Position 3: pay 1, then free swap to position 3
    assert minCosts([10, 5, 1, 2]) == [10, 5, 1, 1]


def test_increasing_costs():
    # Costs increase from left to right
    # Position 0: pay 1
    # All other positions: pay 1, then free swaps
    assert minCosts([1, 2, 3, 4, 5]) == [1, 1, 1, 1, 1]


def test_two_people():
    # n=2: two people at positions 0 and 1
    # Position 0: pay cost[0]=7
    # Position 1: pay cost[1]=3
    assert minCosts([7, 3]) == [7, 3]


def test_minimum_in_middle():
    # Minimum cost is in the middle
    # Position 0: pay 5
    # Position 1: pay 4
    # Position 2: pay 1
    # Position 3: pay 1, then free swap
    # Position 4: pay 1, then free swaps
    assert minCosts([5, 4, 1, 3, 6]) == [5, 4, 1, 1, 1]
