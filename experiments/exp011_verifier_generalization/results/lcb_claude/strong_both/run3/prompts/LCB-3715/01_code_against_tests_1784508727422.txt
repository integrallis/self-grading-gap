```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a map to hold coin counts at each position
    coin_map = {}  
    for l, r, c in coins:
        for i in range(l, r + 1):
            coin_map[i] = coin_map.get(i, 0) + c

    # Create a sorted list of unique positions from the coin_map
    positions = sorted(coin_map.keys())
    n = len(positions)

    # To find the maximum coins in a window of size k
    max_coins = 0
    current_coins = 0
    left = 0

    for right in range(n):
        # Add coins in the current position to current_coins
        current_coins += coin_map[positions[right]]

        # Check if the current window exceeds k
        while positions[right] - positions[left] + 1 > k:
            current_coins -= coin_map[positions[left]]
            left += 1

        # Update max_coins with the current valid window
        max_coins = max(max_coins, current_coins)

    return max_coins

```
