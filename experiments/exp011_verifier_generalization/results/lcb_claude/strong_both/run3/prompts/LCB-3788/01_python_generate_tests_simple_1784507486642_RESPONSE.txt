from typing import List
import pytest
from solution import maxSum


def test_canonical_example_1():
    # spec: nums = [1,2,3,4,5] -> 15
    # Select the entire array without deleting any element
    assert maxSum([1, 2, 3, 4, 5]) == 15


def test_canonical_example_2():
    # spec: nums = [1,1,0,1,1] -> 1
    # Delete elements to get [1], sum = 1
    assert maxSum([1, 1, 0, 1, 1]) == 1


def test_canonical_example_3():
    # spec: nums = [1,2,-1,-2,1,0,-1] -> 3
    # Delete elements nums[2] == -1 and nums[3] == -2, select subarray [2, 1]
    # sum = 2 + 1 = 3
    assert maxSum([1, 2, -1, -2, 1, 0, -1]) == 3


def test_single_element():
    # spec: 1 <= nums.length, single element array
    # The only subarray is [5], sum = 5
    assert maxSum([5]) == 5


def test_single_element_negative():
    # spec: -100 <= nums[i], single negative element
    # The only subarray is [-50], sum = -50
    assert maxSum([-50]) == -50


def test_all_negative_values():
    # spec: -100 <= nums[i] <= 100, can have all negative
    # Best we can do is select the least negative single element
    # [-5, -10, -3] -> select [-3], sum = -3
    assert maxSum([-5, -10, -3]) == -3


def test_all_same_positive():
    # spec: elements can repeat, all same positive value
    # [7, 7, 7] -> can only select one 7 in subarray (must be unique), sum = 7
    assert maxSum([7, 7, 7]) == 7


def test_mix_positive_negative_zero():
    # spec: can have positive, negative, and zero
    # [5, -2, 3, 0, -1, 4] -> select [5, 3, 4] (after deletions), sum = 12
    # Or select subarray [5] then after deletions [3, 0, 4], sum = 7
    # Or original positions: [5, -2, 3, 0, -1, 4] delete -2, 0, -1 -> [5, 3, 4], sum = 12
    assert maxSum([5, -2, 3, 0, -1, 4]) == 12


def test_duplicates_with_positive_values():
    # spec: duplicates allowed, need unique subarray
    # [3, 1, 3, 2] -> delete one 3, select [3, 1, 2] or [1, 3, 2], sum = 6
    assert maxSum([3, 1, 3, 2]) == 6


def test_all_zeros():
    # spec: -100 <= nums[i] <= 100, zero is valid
    # [0, 0, 0] -> select [0], sum = 0
    assert maxSum([0, 0, 0]) == 0


def test_two_elements_same():
    # spec: duplicates possible
    # [10, 10] -> select [10], sum = 10
    assert maxSum([10, 10]) == 10


def test_two_elements_different():

    # [10, 20] -> select [10, 20], sum = 30
    assert maxSum([10, 20]) == 30


def test_negative_then_positive():
    # spec: can have negative and positive

    assert maxSum([-5, 10]) == 10


def test_boundary_values():
    # spec: -100 <= nums[i] <= 100
    # [-100, 100, -100] -> select [100], sum = 100

def test_increasing_sequence():

    # [1, 2, 3] -> select all, sum = 6
    assert maxSum([1, 2, 3]) == 6


def test_decreasing_sequence():
    # spec: general case with unique decreasing values
    # [5, 4, 3, 2, 1] -> select all, sum = 15
    assert maxSum([5, 4, 3, 2, 1]) == 15
