```python
from typing import List

def maxSum(nums: List[int]) -> int:
    max_sum = float('-inf')  # Start with the smallest value to handle negative sums
    current_sum = 0
    unique_elements = set()
    left = 0

    for right in range(len(nums)):
        while nums[right] in unique_elements:
            unique_elements.remove(nums[left])
            current_sum -= nums[left]
            left += 1

        unique_elements.add(nums[right])
        current_sum += nums[right]
        max_sum = max(max_sum, current_sum)

    return max(max_sum, 0)  # Ensure at least returning 0 for non-negative sums

```
