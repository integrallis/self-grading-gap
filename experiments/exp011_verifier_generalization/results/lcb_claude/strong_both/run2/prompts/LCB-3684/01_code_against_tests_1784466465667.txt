## Implementation

```python
def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]
    for i in range(len(s) - len(prefix) + 1):
        if s[i:i + len(prefix)] == prefix:
            for j in range(i + len(prefix), len(s) - len(suffix) + 1):
                if s[j:j + len(suffix)] == suffix:
                    return True
    return False

```
