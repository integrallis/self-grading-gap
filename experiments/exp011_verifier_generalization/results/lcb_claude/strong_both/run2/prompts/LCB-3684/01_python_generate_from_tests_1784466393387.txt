A substring is a contiguous non-empty sequence of characters within a string.

Output: true
Explanation:
By replacing the '*' with "tcod", the substring "eetcode" matches the pattern.

Example 2:

Input: s = "car", p = "c*v"
Output: false
Explanation:
There is no substring matching the pattern.

Example 3:

Input: s = "luck", p = "u*"
Output: true
Explanation:
The substrings "u", "uc", and "uck" match the pattern.

 
Constraints:

1 <= s.length <= 50
1 <= p.length <= 50 
s contains only lowercase English letters.

def test_star_at_end():
    # p = "abc*" means "abc" followed by any suffix
    # s = "zabc" contains substring "abc" (with * = "")
    assert hasMatch("zabc", "abc*") is True


def test_star_in_middle():
    # p = "a*c" means "a" followed by any characters followed by "c"
    # s = "abxc" contains substring "axc" (with * = "")

def test_star_replaced_by_empty_string():
    # p = "ab*cd" can match "abcd" when * = ""
    # s = "xabcdy" contains substring "abcd"
    assert hasMatch("xabcdy", "ab*cd") is True


def test_star_replaced_by_multiple_chars():
    # p = "a*b" can match "axxxb" when * = "xxx"
    # s = "zaxxxbc" contains substring "axxxb"
    assert hasMatch("zaxxxbc", "a*b") is True


def test_no_match_missing_prefix():
    # p = "abc*" requires prefix "abc"
    # s = "xbc" does not contain any substring starting with "abc"
    assert hasMatch("xbc", "abc*") is False


def test_no_match_missing_suffix():
    # p = "*xyz" requires suffix "xyz"
    # s = "abcxy" does not contain any substring ending with "xyz"
    assert hasMatch("abcxy", "*xyz") is False


def test_single_char_pattern_with_star():
    # p = "*" means any sequence (including empty is valid for *)
    # But pattern must match a non-empty substring
    # s = "a" contains substring "a" (with * = "a")
    # Actually, p = "*" has only star, so prefix = "", suffix = ""
    # Any substring of s matches (empty prefix + anything + empty suffix)
    assert hasMatch("a", "*") is True


def test_pattern_longer_than_string_no_star_replacement():
    # p = "abc*def" has prefix "abc" and suffix "def" (6 chars minimum)
    # s = "abcd" is too short to contain "abcdef"
    assert hasMatch("abcd", "abc*def") is False


def test_pattern_matches_entire_string():
    # p = "a*c" can match entire string "abc" when * = "b"
    assert hasMatch("abc", "a*c") is True


def test_pattern_matches_at_start():
    # p = "ab*" matches substring "ab" at start of "abcd" when * = ""
    assert hasMatch("abcd", "ab*") is True


def test_pattern_matches_at_end():
    # p = "*cd" matches substring "cd" at end of "abcd" when * = ""
    assert hasMatch("abcd", "*cd") is True


def test_multiple_possible_matches():
    # p = "a*a" can match "aa", "aba", "abba", etc.
    # s = "baa" contains substring "aa" (with * = "")
    assert hasMatch("baa", "a*a") is True


def test_no_match_different_chars():
    # p = "x*y" requires 'x' before and 'y' after
    # s = "abc" has neither 'x' nor 'y'
    assert hasMatch("abc", "x*y") is False


def test_star_with_same_char_before_and_after():
    # p = "a*a" in s = "aaa" matches "aa" (with * = "")
    assert hasMatch("aaa", "a*a") is True


def test_minimal_string_and_pattern():
    # s = "a", p = "*" should match (substring "a" with * = "a")
    # But actually p = "*" means prefix="" + * + suffix=""
    # So any substring matches
    assert hasMatch("a", "*") is True


def test_pattern_requires_specific_chars_not_present():
    # p = "z*z" requires 'z' at both ends
    # s = "abc" has no 'z'
