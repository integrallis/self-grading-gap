```python
from typing import List

def countPartitions(nums: List[int]) -> int:
    total_count = 0
    total_sum = sum(nums)
    left_sum = 0
    for i in range(len(nums) - 1):  # Iterate through possible partition indices
        left_sum += nums[i]  # Update the left subarray sum
        right_sum = total_sum - left_sum  # Calculate the right subarray sum
        if (left_sum - right_sum) % 2 == 0:  # Check if the difference is even
            total_count += 1  # Increment the valid partition count
    return total_count

```
