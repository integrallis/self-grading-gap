from typing import List
from solution import countPartitions


def test_canonical_example_1():
    # spec: nums = [10,10,3,7,6]
    # Partitions:
    # i=0: [10] sum=10, [10,3,7,6] sum=26, diff=10-26=-16 (even)
    # i=1: [10,10] sum=20, [3,7,6] sum=16, diff=20-16=4 (even)
    # i=2: [10,10,3] sum=23, [7,6] sum=13, diff=23-13=10 (even)
    # i=3: [10,10,3,7] sum=30, [6] sum=6, diff=30-6=24 (even)
    # All 4 partitions have even difference -> output 4
    assert countPartitions([10, 10, 3, 7, 6]) == 4
