```python
from typing import List


def largestInteger(nums: List[int], k: int) -> int:
    from collections import defaultdict

    count = defaultdict(int)
    n = len(nums)

    # Iterate through all subarrays of size k
    for i in range(n - k + 1):
        seen = set()
        for j in range(k):
            seen.add(nums[i + j])
        for num in seen:
            count[num] += 1

    # Find the largest number that appears in exactly one subarray
    largest_missing = -1
    for num, c in count.items():
        if c == 1:
            largest_missing = max(largest_missing, num)

    return largest_missing
```
