import pytest
from typing import List
from solution import largestInteger


def test_canonical_example_1():
    # spec: nums=[3,9,2,1,7], k=3 -> 7
    # 3 appears in 1 subarray: [3,9,2]
    # 9 appears in 2 subarrays: [3,9,2], [9,2,1]
    # 2 appears in 3 subarrays: [3,9,2], [9,2,1], [2,1,7]
    # 1 appears in 2 subarrays: [9,2,1], [2,1,7]
    # 7 appears in 1 subarray: [2,1,7]
    # largest almost missing (exactly 1 subarray) is 7
    assert largestInteger([3, 9, 2, 1, 7], 3) == 7


def test_canonical_example_2():
    # spec: nums=[3,9,7,2,1,7], k=4 -> 3
    # 3 appears in 1 subarray: [3,9,7,2]
    # 9 appears in 2 subarrays: [3,9,7,2], [9,7,2,1]
    # 7 appears in 3 subarrays: [3,9,7,2], [9,7,2,1], [7,2,1,7]
    # 2 appears in 3 subarrays: [3,9,7,2], [9,7,2,1], [7,2,1,7]
    # 1 appears in 2 subarrays: [9,7,2,1], [7,2,1,7]
    # largest almost missing (exactly 1 subarray) is 3
    assert largestInteger([3, 9, 7, 2, 1, 7], 4) == 3


def test_canonical_example_3():
    # spec: nums=[0,0], k=1 -> -1
    # 0 appears in 2 subarrays of size 1: [0], [0]
    # no integer appears in exactly one subarray
    assert largestInteger([0, 0], 1) == -1


def test_single_element_array():
    # nums=[5], k=1 -> 5
    # only one subarray [5], 5 appears in exactly 1 subarray
    assert largestInteger([5], 1) == 5


def test_k_equals_array_length():
    # nums=[1,2,3], k=3 -> 3
    # only one subarray [1,2,3], all elements appear in exactly 1 subarray
    # largest is 3
    assert largestInteger([1, 2, 3], 3) == 3


def test_all_elements_appear_multiple_times():
    # nums=[1,2,1,2], k=2
    # subarrays: [1,2], [2,1], [1,2]
    # 1 appears in all 3 subarrays
    # 2 appears in all 3 subarrays
    # no element appears in exactly 1 subarray
    assert largestInteger([1, 2, 1, 2], 2) == -1


def test_multiple_almost_missing_integers():
    # nums=[5,3,1], k=2
    # subarrays: [5,3], [3,1]
    # 5 appears in 1 subarray: [5,3]
    # 3 appears in 2 subarrays: [5,3], [3,1]
    # 1 appears in 1 subarray: [3,1]
    # almost missing: 5, 1 -> largest is 5
    assert largestInteger([5, 3, 1], 2) == 5


def test_element_at_boundary():
    # nums=[10,20,30,40], k=2
    # subarrays: [10,20], [20,30], [30,40]
    # 10 appears in 1 subarray: [10,20]
    # 20 appears in 2 subarrays: [10,20], [20,30]
    # 30 appears in 2 subarrays: [20,30], [30,40]
    # 40 appears in 1 subarray: [30,40]
    # almost missing: 10, 40 -> largest is 40
    assert largestInteger([10, 20, 30, 40], 2) == 40


def test_duplicate_values_different_positions():
    # nums=[7,5,7], k=2
    # subarrays: [7,5], [5,7]
    # 7 appears in 2 subarrays: [7,5], [5,7]
    # 5 appears in 2 subarrays: [7,5], [5,7]
    # no element appears in exactly 1 subarray
    assert largestInteger([7, 5, 7], 2) == -1


def test_zero_values():
    # nums=[0,1,2], k=2
    # subarrays: [0,1], [1,2]
    # 0 appears in 1 subarray: [0,1]
    # 1 appears in 2 subarrays: [0,1], [1,2]
    # 2 appears in 1 subarray: [1,2]
    # almost missing: 0, 2 -> largest is 2
    assert largestInteger([0, 1, 2], 2) == 2
