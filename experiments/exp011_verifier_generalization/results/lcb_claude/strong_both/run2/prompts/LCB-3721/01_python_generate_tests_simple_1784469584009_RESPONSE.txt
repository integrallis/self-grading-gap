from typing import List
import pytest
from solution import countMentions


def test_canonical_example_1():
    # Example 1: numberOfUsers = 2, events = [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","71","HERE"]]
    # At timestamp 10: id1 and id0 mentioned -> mentions = [1,1]
    # At timestamp 11: id0 goes offline (until 71)
    # At timestamp 71: id0 back online, "HERE" mentions all online users (both) -> mentions = [2,2]
    assert countMentions(2, [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "71", "HERE"]]) == [2, 2]


def test_canonical_example_2():
    # Example 2: numberOfUsers = 2, events = [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","12","ALL"]]
    # At timestamp 10: id1 and id0 mentioned -> mentions = [1,1]
    # At timestamp 11: id0 goes offline
    # At timestamp 12: "ALL" mentions all users (including offline) -> mentions = [2,2]
    assert countMentions(2, [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "12", "ALL"]]) == [2, 2]


def test_canonical_example_3():
    # Example 3: numberOfUsers = 2, events = [["OFFLINE","10","0"],["MESSAGE","12","HERE"]]
    # At timestamp 10: id0 goes offline (until 70)

    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "12", "HERE"]]) == [0, 1]
