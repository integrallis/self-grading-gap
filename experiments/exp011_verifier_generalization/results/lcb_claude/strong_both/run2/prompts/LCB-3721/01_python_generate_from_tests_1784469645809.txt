Full problem statement:

You are given an integer numberOfUsers representing the total number of users and an array events of size n x 3.
Each events[i] can be either of the following two types:

Message Event: ["MESSAGE", "timestamp_i", "mentions_string_i"]

This event indicates that a set of users was mentioned in a message at timestamp_i.
The mentions_string_i string can contain one of the following tokens:
		
id<number>: where <number> is an integer in range [0,numberOfUsers - 1]. There can be multiple ids separated by a single whitespace and may contain duplicates. This can mention even the offline users.
ALL: mentions all users.
HERE: mentions all online users.




Offline Event: ["OFFLINE", "timestamp_i", "id_i"]

This event indicates that the user id_i had become offline at timestamp_i for 60 time units. The user will automatically be online again at time timestamp_i + 60.



Return an array mentions where mentions[i] represents the number of mentions the user with id i has across all MESSAGE events.
All users are initially online, and if a user goes offline or comes back online, their status change is processed before handling any message event that occurs at the same timestamp.
Note that a user can be mentioned multiple times in a single message event, and each mention should be counted separately.
 
Example 1:

Input: numberOfUsers = 2, events = [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","71","HERE"]]
Output: [2,2]
Explanation:
Initially, all users are online.
At timestamp 10, id1 and id0 are mentioned. mentions = [1,1]
At timestamp 11, id0 goes offline.
At timestamp 71, id0 comes back online and "HERE" is mentioned. mentions = [2,2]

Example 2:

Input: numberOfUsers = 2, events = [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","12","ALL"]]
Output: [2,2]
Explanation:
Initially, all users are online.
At timestamp 10, id1 and id0 are mentioned. mentions = [1,1]
At timestamp 11, id0 goes offline.
At timestamp 12, "ALL" is mentioned. This includes offline users, so both id0 and id1 are mentioned. mentions = [2,2]

Example 3:

Input: numberOfUsers = 2, events = [["OFFLINE","10","0"],["MESSAGE","12","HERE"]]
Output: [0,1]
Explanation:
Initially, all users are online.
At timestamp 10, id0 goes offline.
At timestamp 12, "HERE" is mentioned. Because id0 is still offline, they will not be mentioned. mentions = [0,1]

 
Constraints:

1 <= numberOfUsers <= 100
1 <= events.length <= 100
events[i].length == 3
events[i][0] will be one of MESSAGE or OFFLINE.
1 <= int(events[i][1]) <= 10^5
The number of id<number> mentions in any "MESSAGE" event is between 1 and 100.
0 <= <number> <= numberOfUsers - 1
It is guaranteed that the user id referenced in the OFFLINE event is online at the time the event occurs.

from typing import List
import pytest
from solution import countMentions


def test_canonical_example_1():
    # Example 1: numberOfUsers = 2, events = [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","71","HERE"]]

def test_canonical_example_2():
    # Example 2: numberOfUsers = 2, events = [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","12","ALL"]]
    # At timestamp 10: id1 and id0 mentioned -> mentions = [1,1]
    # At timestamp 11: id0 goes offline
    # At timestamp 12: "ALL" mentions all users (including offline) -> mentions = [2,2]
    assert countMentions(2, [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "12", "ALL"]]) == [2, 2]


def test_canonical_example_3():
    # Example 3: numberOfUsers = 2, events = [["OFFLINE","10","0"],["MESSAGE","12","HERE"]]
    # At timestamp 10: id0 goes offline (until 70)
    # At timestamp 12: "HERE" mentions only online users (id1) -> mentions = [0,1]
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "12", "HERE"]]) == [0, 1]


def test_single_user_single_message():
    # numberOfUsers = 1, single MESSAGE event mentioning id0
    # mentions = [1]
    assert countMentions(1, [["MESSAGE", "5", "id0"]]) == [1]


def test_multiple_mentions_same_user_same_message():
    # numberOfUsers = 2, MESSAGE with "id0 id0 id1"
    # Each mention counted separately: id0 gets 2, id1 gets 1
    assert countMentions(2, [["MESSAGE", "10", "id0 id0 id1"]]) == [2, 1]


def test_all_mentions_all_users():
    # numberOfUsers = 3, MESSAGE with "ALL"
    # All 3 users mentioned once each
    assert countMentions(3, [["MESSAGE", "10", "ALL"]]) == [1, 1, 1]


def test_here_mentions_online_users_only():
    # numberOfUsers = 3, id1 offline, MESSAGE with "HERE"
    # Only id0 and id2 (online) mentioned
    assert countMentions(3, [["OFFLINE", "5", "1"], ["MESSAGE", "10", "HERE"]]) == [1, 0, 1]


def test_user_comes_back_online_after_60_units():
    # numberOfUsers = 2, id0 offline at 10, MESSAGE at 70 with "HERE"
    # id0 comes back online at 70 (10+60), so both mentioned
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "70", "HERE"]]) == [1, 1]


def test_user_still_offline_before_60_units():
    # numberOfUsers = 2, id0 offline at 10, MESSAGE at 69 with "HERE"
    # id0 still offline at 69 (comes back at 70), only id1 mentioned
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "69", "HERE"]]) == [0, 1]


def test_multiple_offline_events():
    # numberOfUsers = 3, id0 and id2 offline, MESSAGE with "HERE"
    # Only id1 online and mentioned
    assert countMentions(3, [["OFFLINE", "5", "0"], ["OFFLINE", "6", "2"], ["MESSAGE", "10", "HERE"]]) == [0, 1, 0]


def test_explicit_id_mentions_offline_user():
    # numberOfUsers = 2, id0 offline, MESSAGE with "id0"
    # Explicit id mentions work even for offline users
    assert countMentions(2, [["OFFLINE", "5", "0"], ["MESSAGE", "10", "id0"]]) == [1, 0]


def test_mixed_id_and_here():
    # numberOfUsers = 3, MESSAGE with "id0 HERE"
    # id0 mentioned explicitly (1), HERE mentions all online (all 3) -> id0 gets 2 total, others 1
    assert countMentions(3, [["MESSAGE", "10", "id0 HERE"]]) == [2, 1, 1]


def test_mixed_id_and_all():
    # numberOfUsers = 3, MESSAGE with "id1 ALL"
    # id1 mentioned explicitly (1), ALL mentions all (3) -> id1 gets 2 total, others 1
    assert countMentions(3, [["MESSAGE", "10", "id1 ALL"]]) == [1, 2, 1]


def test_no_messages():
    # numberOfUsers = 2, only OFFLINE events, no MESSAGE events
    # No mentions for anyone
    assert countMentions(2, [["OFFLINE", "10", "0"]]) == [0, 0]


def test_empty_events():
    # numberOfUsers = 1, no events
    # No mentions for anyone
    assert countMentions(1, []) == [0]


def test_multiple_messages_accumulate():
    # numberOfUsers = 2, two MESSAGE events
    # First: id0, second: id1 -> mentions = [1, 1]
    assert countMentions(2, [["MESSAGE", "10", "id0"], ["MESSAGE", "20", "id1"]]) == [1, 1]


def test_offline_at_same_timestamp_processed_before_message():
    # numberOfUsers = 2, OFFLINE and MESSAGE at same timestamp 10
    # Offline processed first, so HERE doesn't mention the offline user
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "10", "HERE"]]) == [0, 1]


def test_online_at_same_timestamp_processed_before_message():
    # numberOfUsers = 2, user goes offline at 10, comes back at 70, MESSAGE at 70
    # User is online when message processed (comes online at 70 before message)
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "70", "HERE"]]) == [1, 1]


def test_many_users():
    # numberOfUsers = 5, MESSAGE with "ALL"
    # All 5 users mentioned
    assert countMentions(5, [["MESSAGE", "10", "ALL"]]) == [1, 1, 1, 1, 1]


def test_multiple_ids_in_message():
    # numberOfUsers = 4, MESSAGE with "id0 id2 id1 id3"
    # Each user mentioned once
    assert countMentions(4, [["MESSAGE", "10", "id0 id2 id1 id3"]]) == [1, 1, 1, 1]


def test_duplicate_ids_counted_separately():
    # numberOfUsers = 3, MESSAGE with "id0 id1 id0 id1 id0"
    # id0 mentioned 3 times, id1 mentioned 2 times, id2 not mentioned
    assert countMentions(3, [["MESSAGE", "10", "id0 id1 id0 id1 id0"]]) == [3, 2, 0]
