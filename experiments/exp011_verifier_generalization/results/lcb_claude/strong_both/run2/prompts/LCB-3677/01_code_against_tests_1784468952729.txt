```python
from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m = len(coins)
    n = len(coins[0]) if m > 0 else 0

    # dp[i][j][k] is the maximum coins that can be collected to (i, j) using k neutralizations
    dp = [[[-float('inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = max(coins[0][0], 0)  # Start position, neutralizing if negative
    dp[0][0][1] = 0
    dp[0][0][2] = 0

    for i in range(m):
        for j in range(n):
            for k in range(3):
                if i == 0 and j == 0:
                    continue
                if i > 0:
                    previous_value = dp[i-1][j][k]
                    if coins[i][j] < 0 and k > 0:
                        previous_value = max(previous_value, dp[i-1][j][k-1])
                    dp[i][j][k] = max(dp[i][j][k], previous_value + coins[i][j])
                if j > 0:
                    previous_value = dp[i][j-1][k]
                    if coins[i][j] < 0 and k > 0:
                        previous_value = max(previous_value, dp[i][j-1][k-1])
                    dp[i][j][k] = max(dp[i][j][k], previous_value + coins[i][j])

    return max(dp[m-1][n-1])
```
