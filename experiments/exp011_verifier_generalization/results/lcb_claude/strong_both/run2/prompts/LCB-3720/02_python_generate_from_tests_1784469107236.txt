from typing import List
import sys


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Sort edges based on their weights
    edges.sort(key=lambda x: x[2])
    left, right = 0, len(edges) - 1
    answer = -1

    while left <= right:
        mid = (left + right) // 2
        max_weight = edges[mid][2]
        graph = [[] for _ in range(n)]

        # Build graph with edges that have weights <= max_weight
        for i in range(len(edges)):
            if edges[i][2] <= max_weight:
                graph[edges[i][0]].append(edges[i][1])

        # Check if we can reach node 0 from all other nodes
        if canReachZero(n, graph, threshold):
            answer = max_weight
            right = mid - 1
        else:
            left = mid + 1

    return answer


def canReachZero(n: int, graph: List[List[int]], threshold: int) -> bool:
    from collections import deque
    out_degree = [0] * n
    for u in range(n):
        out_degree[u] = len(graph[u])

    queue = deque([0])
    visited = set([0])

    while queue:
        u = queue.popleft()
        for v in graph[u]:
            if v not in visited:
                visited.add(v)
                queue.append(v)

    # Check if all nodes can be visited and respect the threshold
    for u in range(n):
        if u not in visited:
            return False
        if len(graph[u]) > threshold:
            return False

    return True

```
