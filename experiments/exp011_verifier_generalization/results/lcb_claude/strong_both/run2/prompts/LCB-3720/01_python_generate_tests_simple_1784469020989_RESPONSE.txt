from typing import List
import pytest
from solution import minMaxWeight


def test_canonical_example_1():
    """
    Example 1: n=5, edges=[[1,0,1],[2,0,2],[3,0,1],[4,3,1],[2,1,1]], threshold=2
    Expected output: 1
    Explanation: Remove edge 2->0. Max weight among remaining edges is 1.
    """
    assert minMaxWeight(5, [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]], 2) == 1


def test_canonical_example_2():
    """
    Example 2: n=5, edges=[[0,1,1],[0,2,2],[0,3,1],[0,4,1],[1,2,1],[1,4,1]], threshold=1
    Expected output: -1
    Explanation: Impossible to reach node 0 from node 2.
    """
    assert minMaxWeight(5, [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]], 1) == -1


def test_canonical_example_3():
    """
    Example 3: n=5, edges=[[1,2,1],[1,3,3],[1,4,5],[2,3,2],[3,4,2],[4,0,1]], threshold=1
    Expected output: 2
    Explanation: Remove edges 1->3 and 1->4. Max weight among remaining edges is 2.
    """
    assert minMaxWeight(5, [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]], 1) == 2


def test_canonical_example_4():
    """
    Example 4: n=5, edges=[[1,2,1],[1,3,3],[1,4,5],[2,3,2],[4,0,1]], threshold=1
    Expected output: -1
    Explanation: Impossible to satisfy all conditions.
    """
    assert minMaxWeight(5, [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]], 1) == -1


def test_simple_two_nodes():
    """
    Simple case: 2 nodes, one edge from node 1 to node 0.
    All nodes can reach node 0, max weight is the edge weight.
    """
    assert minMaxWeight(2, [[1, 0, 5]], 1) == 5


def test_three_nodes_linear_path():
    """
    Three nodes in a line: 2 -> 1 -> 0
    All nodes can reach 0, max weight is max of two edge weights.
    """
    assert minMaxWeight(3, [[1, 0, 3], [2, 1, 4]], 1) == 4


def test_threshold_reduces_max_weight():
    """
    When threshold allows multiple outgoing edges, can choose lower weight edges.
    Node 1 has two edges to 0 with weights 10 and 2.
    With threshold=1, can only use one edge, so choose weight 2.
    """
    assert minMaxWeight(2, [[1, 0, 10], [1, 0, 2]], 1) == 2


def test_multiple_paths_to_zero():
    """
    Multiple paths from a node to 0, can choose path with lower max weight.
    Node 2 can go 2->0 (weight 5) or 2->1->0 (weights 1,1).
    Second path has lower max weight.
    """
    assert minMaxWeight(3, [[2, 0, 5], [2, 1, 1], [1, 0, 1]], 1) == 1


def test_all_edges_needed():
    """
    Case where all edges are needed and none can be removed.
    Each node has exactly one outgoing edge forming a path to 0.
    """
    assert minMaxWeight(4, [[1, 0, 3], [2, 1, 2], [3, 2, 1]], 1) == 3


def test_threshold_constraint_violated():
    """
    If a node needs more than threshold outgoing edges to ensure reachability,
    it becomes impossible. Node 1 needs path to 0 but has no outgoing edges.
    """
    assert minMaxWeight(3, [[2, 1, 1]], 1) == -1


def test_node_zero_has_outgoing_edges():
    """
    Node 0 can have outgoing edges but they don't affect reachability to 0.
    Only edges that help reach 0 matter.
    """
    assert minMaxWeight(3, [[0, 1, 100], [1, 0, 5], [2, 0, 3]], 1) == 5


def test_larger_threshold_allows_more_edges():
    """
    With higher threshold, can use more edges which might reduce max weight.
    Node 1 can use both outgoing edges when threshold=2.
    """
    assert minMaxWeight(4, [[1, 0, 10], [1, 2, 3], [2, 0, 4], [3, 1, 2]], 2) == 4
