from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_intervals = []

    # Initial free time before the first meeting
    if startTime[0] > 0:
        free_intervals.append((0, startTime[0]))

    # Free time between meetings
    for i in range(n - 1):
        if endTime[i] < startTime[i + 1]:
            free_intervals.append((endTime[i], startTime[i + 1]))

    # Free time after the last meeting
    if endTime[-1] < eventTime:
        free_intervals.append((endTime[-1], eventTime))

    # Calculate initial gaps
    initial_gaps = [end - start for start, end in free_intervals]
    max_free_time = max(initial_gaps) if initial_gaps else 0

    # Rescheduling meetings
    for i in range(n):
        for j in range(k):
            # Move current meeting forward as much as we can
            if i - j < 0:
                continue
            move_start = max(0, startTime[i - j] - (endTime[i - j] - startTime[i - j]))
            move_end = move_start + (endTime[i - j] - startTime[i - j])
            if (i - j > 0 and move_end > startTime[i - j]) or (i - j < n - 1 and move_start < endTime[i - j + 1]):
                continue  # Can't overlap with previous or next meeting

            # Calculate new free intervals and gaps
            new_free_intervals = [(0, move_start)] + free_intervals
            new_free_intervals[i - j] = (move_start, move_end)
            new_free_intervals = sorted(new_free_intervals)

            # Calculate new gaps
            new_gaps = [max(0, new_free_intervals[m][0] - new_free_intervals[m - 1][1]) for m in range(1, len(new_free_intervals))]
            max_free_time = max(max_free_time, *new_gaps)

    return max_free_time
```
