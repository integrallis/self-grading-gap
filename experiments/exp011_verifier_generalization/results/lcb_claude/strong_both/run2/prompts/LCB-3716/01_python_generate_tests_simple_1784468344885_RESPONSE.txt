from typing import List
import pytest
from solution import longestSubsequence


def test_canonical_example_1():
    # spec: nums = [16,6,3] -> 3
    # The longest subsequence is [16, 6, 3] with absolute adjacent differences [10, 3]
    # 10 >= 3, so differences are non-increasing
    assert longestSubsequence([16, 6, 3]) == 3


def test_canonical_example_2():
    # spec: nums = [6,5,3,4,2,1] -> 4
    # The longest subsequence is [6, 4, 2, 1] with absolute adjacent differences [2, 2, 1]
    # 2 >= 2 >= 1, so differences are non-increasing
    assert longestSubsequence([6, 5, 3, 4, 2, 1]) == 4


def test_canonical_example_3():
    # spec: nums = [10,20,10,19,10,20] -> 5
    # The longest subsequence is [10, 20, 10, 19, 10] with absolute adjacent differences [10, 10, 9, 9]
    # 10 >= 10 >= 9 >= 9, so differences are non-increasing
    assert longestSubsequence([10, 20, 10, 19, 10, 20]) == 5


def test_minimum_length_array():
    # spec: 2 <= nums.length, so minimum is 2 elements
    # Any 2-element subsequence has one difference, which is trivially non-increasing
    # So the answer should be at least 2
    assert longestSubsequence([1, 2]) == 2


def test_all_same_elements():
    # All elements equal means all differences are 0
    # Differences: [0, 0, 0, ...] which is non-increasing (0 >= 0 >= 0 ...)
    # So entire array is valid
    assert longestSubsequence([5, 5, 5, 5]) == 4


def test_strictly_increasing_sequence():
    # nums = [1, 2, 3, 4]
    # Differences for entire sequence: [1, 1, 1] - non-increasing (equal values count)
    # So length should be 4
    assert longestSubsequence([1, 2, 3, 4]) == 4


def test_strictly_decreasing_sequence():
    # nums = [4, 3, 2, 1]
    # Differences for entire sequence: [1, 1, 1] - non-increasing
    # So length should be 4
    assert longestSubsequence([4, 3, 2, 1]) == 4


def test_increasing_differences():
    # nums = [1, 2, 4, 7]
    # Full sequence differences: [1, 2, 3] - this is increasing, not non-increasing
    # We need to find a subsequence where differences are non-increasing
    # [1, 2] has diff [1], length 2
    # Any single pair works, so minimum answer is 2
    assert longestSubsequence([1, 2, 4, 7]) >= 2


def test_large_then_small_differences():
    # nums = [1, 10, 11, 12]
    # [1, 10, 11, 12] has differences [9, 1, 1] - non-increasing (9 >= 1 >= 1)
    # So length should be 4
    assert longestSubsequence([1, 10, 11, 12]) == 4


def test_alternating_values():
    # nums = [1, 5, 1, 5]
    # [1, 5, 1, 5] has differences [4, 4, 4] - non-increasing
    # So length should be 4
    assert longestSubsequence([1, 5, 1, 5]) == 4


def test_single_large_jump():
    # nums = [1, 100, 101]
    # [1, 100, 101] has differences [99, 1] - non-increasing (99 >= 1)
    # So length should be 3
    assert longestSubsequence([1, 100, 101]) == 3
