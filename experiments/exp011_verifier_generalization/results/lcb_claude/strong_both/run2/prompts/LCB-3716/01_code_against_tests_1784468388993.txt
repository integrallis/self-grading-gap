```python
from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n < 2:
        return n
    dp = [1] * n  # Each element is a subsequence of length 1
    for i in range(n):
        for j in range(i):
            diff1 = abs(nums[i] - nums[j])
            if j > 0:
                prev_diff = abs(nums[j] - nums[j-1])
                if diff1 <= prev_diff:
                    dp[i] = max(dp[i], dp[j] + 1)
            else:
                dp[i] = max(dp[i], dp[j] + 1)
    return max(dp)
```
