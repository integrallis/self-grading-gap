from typing import List
import pytest
from solution import minCosts


def test_canonical_example_1():
    # spec: cost = [5,3,4,1,3,2] -> [5,3,3,1,1,1]
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]


def test_canonical_example_2():
    # spec: cost = [1,2,4,6,7] -> [1,1,1,1,1]
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]


def test_single_person():
    # n=1: only one person at position 0, we start at position 1
    # To reach position 0, we swap with person 0 for cost[0]
    assert minCosts([5]) == [5]


def test_two_people_increasing_cost():
    # n=2: people at positions 0,1, we start at position 2
    # To reach position 0: swap with person 0 for cost 3
    # To reach position 1: min(swap with person 1 for cost 4, swap with person 0 for cost 3 then person 1 swaps for free) = min(4, 3) = 3
    assert minCosts([3, 4]) == [3, 3]


def test_two_people_decreasing_cost():
    # n=2: people at positions 0,1, we start at position 2
    # To reach position 0: min(swap with person 0 for cost 5, swap with person 1 for cost 2 then person 0 swaps for free) = min(5, 2) = 2
    # To reach position 1: swap with person 1 for cost 2
    assert minCosts([5, 2]) == [2, 2]


def test_all_same_cost():
    # If all costs are the same, we still need to find minimum paths
    # To reach position i, we can either pay cost[i] directly or go through cheaper earlier positions
    # With all costs = 3: position 0 costs 3, position 1 costs 3, position 2 costs 3
    assert minCosts([3, 3, 3]) == [3, 3, 3]


def test_decreasing_costs():
    # Costs strictly decreasing: each position can be reached by swapping with that person
    # or by swapping with a later cheaper person then moving back for free
    # cost = [5,4,3,2,1]
    # pos 0: min(5, 4, 3, 2, 1) = 1 (swap with person 4, then move back for free)
    # pos 1: min(4, 3, 2, 1) = 1
    # pos 2: min(3, 2, 1) = 1
    # pos 3: min(2, 1) = 1
    # pos 4: 1
    assert minCosts([5, 4, 3, 2, 1]) == [1, 1, 1, 1, 1]


def test_increasing_costs():
    # Costs strictly increasing: best to swap with person 0, then move back for free
    # cost = [1,2,3,4,5]
    # pos 0: 1
    # pos 1: min(2, 1) = 1 (swap with person 0, then person 1 swaps for free)
    # pos 2: min(3, 1) = 1
    # pos 3: min(4, 1) = 1
    # pos 4: min(5, 1) = 1
    assert minCosts([1, 2, 3, 4, 5]) == [1, 1, 1, 1, 1]


def test_minimum_constraint_values():
    # Minimum constraint: n=1, cost[i]=1
    assert minCosts([1]) == [1]


def test_maximum_constraint_value():
    # Test with maximum cost value = 100
    assert minCosts([100]) == [100]


def test_valley_pattern():
    # High costs at edges, low in middle
    # cost = [10, 5, 1, 5, 10]
    # pos 0: min(10, 5, 1, ...) = 1 (swap with person 2 for cost 1, then move back for free)
    # pos 1: min(5, 1, ...) = 1 (swap with person 2 for cost 1, then move back for free)
    # pos 2: min(1) = 1
    # pos 3: min(5, 1) = 1 (swap with person 2 for cost 1, then person 3 swaps for free)
    # pos 4: min(10, 1) = 1
    assert minCosts([10, 5, 1, 5, 10]) == [1, 1, 1, 1, 1]


def test_peak_pattern():
    # Low costs at edges, high in middle
    # cost = [1, 10, 100, 10, 1]
    # pos 0: 1
    # pos 1: min(10, 1) = 1
    # pos 2: min(100, 1) = 1
    # pos 3: min(10, 1) = 1
    # pos 4: min(1, 1) = 1
    assert minCosts([1, 10, 100, 10, 1]) == [1, 1, 1, 1, 1]


def test_maximum_length_constraint():
    # Test with n=100 (maximum constraint for array length)
    # Create array of length 100 with decreasing costs, last element is 1
    # All positions can be reached via the cheapest person (at position 99) for cost 1
    cost = list(range(100, 0, -1))  # [100, 99, 98, ..., 2, 1]
    expected = [1] * 100  # All positions reachable for cost 1
    assert minCosts(cost) == expected
