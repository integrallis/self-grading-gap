Full problem statement:

There are an infinite amount of bags on a number line, one bag for each coordinate. Some of these bags contain coins.
You are given a 2D array coins, where coins[i] = [l_i, r_i, c_i] denotes that every bag from l_i to r_i contains c_i coins.
The segments that coins contain are non-overlapping.
You are also given an integer k.
Return the maximum amount of coins you can obtain by collecting k consecutive bags.
 
Example 1:

Input: coins = [[8,10,1],[1,3,2],[5,6,4]], k = 4
Output: 10
Explanation:
Selecting bags at positions [3, 4, 5, 6] gives the maximum number of coins: 2 + 0 + 4 + 4 = 10.

Example 2:

Input: coins = [[1,10,3]], k = 2
Output: 6
Explanation:
Selecting bags at positions [1, 2] gives the maximum number of coins: 3 + 3 = 6.

 
Constraints:

1 <= coins.length <= 10^5
1 <= k <= 10^9
coins[i] == [l_i, r_i, c_i]
1 <= l_i <= r_i <= 10^9
1 <= c_i <= 1000
The given segments are non-overlapping.

def test_canonical_example_1():

    assert maximumCoins([[1, 10, 3]], 2) == 6


def test_single_segment_k_equals_segment_length():
    # Single segment [1, 5, 2] with k=5
    # Positions [1, 2, 3, 4, 5] all have 2 coins: 2*5 = 10
    assert maximumCoins([[1, 5, 2]], 5) == 10


def test_single_segment_k_less_than_segment_length():
    # Single segment [1, 10, 5] with k=3
    # Any 3 consecutive positions in [1, 10]: 5*3 = 15
    assert maximumCoins([[1, 10, 5]], 3) == 15


def test_single_segment_k_greater_than_segment_length():
    # Single segment [5, 7, 3] with k=5
    # Best 5 consecutive includes the segment: positions [5, 6, 7, 8, 9]
    # Coins: 3 + 3 + 3 + 0 + 0 = 9
    assert maximumCoins([[5, 7, 3]], 5) == 9


def test_multiple_non_overlapping_segments():
    # Segments [1, 2, 5] and [10, 12, 3] with k=3
    # Best is positions [1, 2, 3]: 5 + 5 + 0 = 10
    # Or positions [10, 11, 12]: 3 + 3 + 3 = 9
    # Max is 10
    assert maximumCoins([[1, 2, 5], [10, 12, 3]], 3) == 10


def test_k_spans_multiple_segments():
    # Segments [1, 2, 1], [4, 5, 2], [7, 8, 3] with k=8
    # Positions [1, 2, 3, 4, 5, 6, 7, 8]: 1+1+0+2+2+0+3+3 = 12
    assert maximumCoins([[1, 2, 1], [4, 5, 2], [7, 8, 3]], 8) == 12


def test_k_equals_one():
    # k=1 means select single bag with most coins
    # Segments [1, 3, 2], [5, 6, 5], [10, 12, 1]
    # Best single bag has 5 coins
    assert maximumCoins([[1, 3, 2], [5, 6, 5], [10, 12, 1]], 1) == 5


def test_large_k_covers_all_segments():
    # Segments [1, 2, 3], [5, 6, 2] with k=100
    # All segments combined: 3*2 + 2*2 = 10, rest are zeros
    assert maximumCoins([[1, 2, 3], [5, 6, 2]], 100) == 10


def test_segments_with_gaps():
    # Segments [1, 1, 10], [5, 5, 8], [9, 9, 6] with k=5
    # Positions [5, 6, 7, 8, 9]: 8+0+0+0+6 = 14
    assert maximumCoins([[1, 1, 10], [5, 5, 8], [9, 9, 6]], 5) == 14


def test_window_starting_before_first_segment():
    # Segment [10, 12, 4] with k=5
    # Window [8, 9, 10, 11, 12]: 0+0+4+4+4 = 12
    assert maximumCoins([[10, 12, 4]], 5) == 12


def test_window_ending_after_last_segment():
    # Segment [1, 3, 5] with k=6
    # Window [1, 2, 3, 4, 5, 6]: 5+5+5+0+0+0 = 15
    assert maximumCoins([[1, 3, 5]], 6) == 15


def test_two_segments_k_in_between():
    # Segments [1, 2, 4], [8, 9, 6] with k=4
    # Best window [6, 7, 8, 9]: 0+0+6+6 = 12
    assert maximumCoins([[1, 2, 4], [8, 9, 6]], 4) == 12
