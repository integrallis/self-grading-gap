```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a list to hold the number of coins at each position
    max_position = 0
    for l, r, c in coins:
        max_position = max(max_position, r)
    coin_count = [0] * (max_position + 2)  # +2 to avoid index error for max_position+1

    # Fill the coin_count array based on the segments
    for l, r, c in coins:
        for i in range(l, r + 1):
            coin_count[i] += c

    # Use a sliding window to calculate the maximum coins we can collect
    max_coins = 0
    current_coins = sum(coin_count[0:k])  # Initial window sum
    max_coins = current_coins

    for i in range(0, max_position + 1 - k):  # Slide the window
        current_coins = current_coins - coin_count[i] + coin_count[i + k]
        max_coins = max(max_coins, current_coins)

    return max_coins
```
