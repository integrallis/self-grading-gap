import pytest
from solution import maxContainers


def test_canonical_example_1():
    # spec: n=2, w=3, maxWeight=15
    # deck has 2*2=4 cells, each container weighs 3
    # total weight if all loaded: 4*3=12, which <= 15
    # output: 4
    assert maxContainers(2, 3, 15) == 4


def test_canonical_example_2():
    # spec: n=3, w=5, maxWeight=20
    # deck has 3*3=9 cells, each container weighs 5
    # max containers without exceeding maxWeight: 20//5=4
    # output: 4
    assert maxContainers(3, 5, 20) == 4


def test_single_cell_deck():
    # n=1: deck has 1*1=1 cell
    # w=10, maxWeight=10
    # can load 1 container (weight 10 <= 10)
    assert maxContainers(1, 10, 10) == 1


def test_weight_capacity_limits_containers():
    # n=5: deck has 5*5=25 cells
    # w=7, maxWeight=50
    # max containers: 50//7=7 (limited by weight, not deck size)
    assert maxContainers(5, 7, 50) == 7


def test_deck_size_limits_containers():
    # n=2: deck has 2*2=4 cells
    # w=1, maxWeight=1000
    # max containers: min(4, 1000//1)=4 (limited by deck size)
    assert maxContainers(2, 1, 1000) == 4


def test_exact_capacity_match():
    # n=3: deck has 3*3=9 cells
    # w=2, maxWeight=18
    # max containers: 18//2=9, deck can hold 9
    # all 9 containers can be loaded
    assert maxContainers(3, 2, 18) == 9


def test_insufficient_capacity_for_single_container():
    # n=10: deck has 10*10=100 cells
    # w=100, maxWeight=50
    # max containers: 50//100=0 (cannot even load 1 container)
    assert maxContainers(10, 100, 50) == 0


def test_large_deck_small_weight():
    # n=10: deck has 10*10=100 cells
    # w=1, maxWeight=30
    # max containers: 30//1=30 (limited by weight)
    assert maxContainers(10, 1, 30) == 30


def test_minimum_values():
    # n=1, w=1, maxWeight=1
    # deck has 1 cell, can load 1 container of weight 1
    assert maxContainers(1, 1, 1) == 1


def test_heavy_containers_large_capacity():
    # n=4: deck has 4*4=16 cells
    # w=500, maxWeight=1000
    # max containers: 1000//500=2
    assert maxContainers(4, 500, 1000) == 2


def test_very_large_capacity():
    # n=2: deck has 2*2=4 cells
    # w=10, maxWeight=1000000000
    # max containers: min(4, 1000000000//10)=4 (limited by deck)
    assert maxContainers(2, 10, 1000000000) == 4
