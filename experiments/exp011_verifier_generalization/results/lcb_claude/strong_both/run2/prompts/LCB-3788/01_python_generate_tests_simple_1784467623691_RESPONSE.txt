import pytest

def test_canonical_example_1():
    # spec: nums = [1,2,3,4,5] -> 15
    # All elements are unique, sum = 1+2+3+4+5 = 15
    assert maxSum([1, 2, 3, 4, 5]) == 15


def test_canonical_example_2():
    # spec: nums = [1,1,0,1,1] -> 1
    # After deletions, select subarray with unique elements that maximizes sum
    # Best is to keep one instance of 1, giving sum = 1
    assert maxSum([1, 1, 0, 1, 1]) == 1


def test_canonical_example_3():
    # spec: nums = [1,2,-1,-2,1,0,-1] -> 3
    # Delete elements to get subarray with unique elements
    # Best subarray is [2, 1] with sum = 2+1 = 3
    assert maxSum([1, 2, -1, -2, 1, 0, -1]) == 3

    # Single element array with positive value
