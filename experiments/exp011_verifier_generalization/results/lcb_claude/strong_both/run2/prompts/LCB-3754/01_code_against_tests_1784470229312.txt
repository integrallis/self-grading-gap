```python
def maxDistance(s: str, k: int) -> int:
    # Initialize coordinates and maximum distance
    x, y = 0, 0
    max_dist = 0

    # Process each movement in the string
    for char in s:
        # Update coordinates based on the movement character
        if char == 'N':
            y += 1
        elif char == 'S':
            y -= 1
        elif char == 'E':
            x += 1
        elif char == 'W':
            x -= 1

        # Calculate the current Manhattan distance
        current_distance = abs(x) + abs(y)
        # Update maximum distance
        max_dist = max(max_dist, current_distance)

    # If changes are allowed, calculate the maximum possible distance
    # Every change can potentially add 2 to the distance (from -1 to +1 in either x or y direction)
    possible_max_distance = max_dist + min(k, len(s)) * 2

    return possible_max_distance
```
