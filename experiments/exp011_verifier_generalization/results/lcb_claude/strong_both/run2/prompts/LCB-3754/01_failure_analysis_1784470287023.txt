def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0  # Initialize coordinates
    max_dist = 0  # Initialize maximum distance
    movements = []  # To track the movements

    # Process each movement in the string
    for char in s:
        # Update coordinates based on the movement character
        if char == 'N':
            y += 1
        elif char == 'S':
            y -= 1
        elif char == 'E':
            x += 1
        elif char == 'W':
            x -= 1

        # Calculate the current Manhattan distance
        current_distance = abs(x) + abs(y)
        # Update maximum distance
        max_dist = max(max_dist, current_distance)
        movements.append((x, y))  # Store the current position

    # Calculate the maximum possible distance considering k changes
    for i in range(len(movements)):  # Iterate over the recorded movements
        # Calculate the maximum distance we could achieve by changing the last k movements
        for j in range(1, k + 1):
            if i - j >= 0:
                new_x = movements[i - j][0]
                new_y = movements[i - j][1]
                # Calculate the distance after changing the direction of the last j movements
                distance_after_changes = abs(new_x) + abs(new_y) + j
                max_dist = max(max_dist, distance_after_changes)  # Update maximum distance
