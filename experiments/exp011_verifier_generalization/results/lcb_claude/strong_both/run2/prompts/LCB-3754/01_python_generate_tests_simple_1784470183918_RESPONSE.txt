import pytest
from solution import maxDistance


def test_canonical_example_1():
    # spec: s = "NWSE", k = 1 -> 3
    # Change s[2] from 'S' to 'N' to get "NWNE"
    # Positions: (0,1), (-1,1), (-1,2), (0,2)
    # Distances: 1, 2, 3, 2
    # Maximum: 3
    assert maxDistance("NWSE", 1) == 3


def test_canonical_example_2():
    # spec: s = "NSWWEW", k = 3 -> 6
    # Change s[1] from 'S' to 'N', and s[4] from 'E' to 'W' to get "NNWWWW"
    # Maximum Manhattan distance achieved is 6
    assert maxDistance("NSWWEW", 3) == 6


def test_single_move_north():
    # s = "N", k = 0
    # Position after move: (0, 1)
    # Distance: 1
    assert maxDistance("N", 0) == 1


def test_single_move_south():
    # s = "S", k = 0
    # Position after move: (0, -1)
    # Distance: 1
    assert maxDistance("S", 0) == 1


def test_single_move_east():
    # s = "E", k = 0
    # Position after move: (1, 0)
    # Distance: 1
    assert maxDistance("E", 0) == 1


def test_single_move_west():
    # s = "W", k = 0
    # Position after move: (-1, 0)
    # Distance: 1
    assert maxDistance("W", 0) == 1


def test_no_changes_allowed():
    # s = "NSEW", k = 0
    # Positions: (0,1), (0,0), (1,0), (0,0)
    # Distances: 1, 0, 1, 0
    # Maximum: 1
    assert maxDistance("NSEW", 0) == 1


def test_all_changes_allowed():
    # s = "NS", k = 2
    # Can change both to go in same direction, e.g., "NN"
    # Positions: (0,1), (0,2)
    # Distances: 1, 2
    # Maximum: 2
    assert maxDistance("NS", 2) == 2


def test_multiple_north_moves():
    # s = "NNN", k = 0
    # Positions: (0,1), (0,2), (0,3)
    # Distances: 1, 2, 3
    # Maximum: 3
    assert maxDistance("NNN", 0) == 3


def test_alternating_opposite_directions():
    # s = "NSNS", k = 0

    # Distances: 1, 0, 1, 0
    # Maximum: 1
    assert maxDistance("NSNS", 0) == 1

    # s = "SSSS", k = 4

    # Positions: (0,1), (0,2), (0,3), (0,4)
    # Distances: 1, 2, 3, 4
    # Maximum: 4
    assert maxDistance("SSSS", 4) == 4


def test_some_changes_to_extend_path():
    # s = "NNS", k = 1
    # Change s[2] from 'S' to 'N' to get "NNN"
    # Positions: (0,1), (0,2), (0,3)
    # Distances: 1, 2, 3
    # Maximum: 3
    assert maxDistance("NNS", 1) == 3


def test_east_west_movements():

    # Positions: (1,0), (2,0), (3,0)
    # Distances: 1, 2, 3
    # Maximum: 3
    assert maxDistance("EEW", 1) == 3


def test_diagonal_movement():
    # s = "NE", k = 0
    # Positions: (0,1), (1,1)
    # Distances: 1, 2
    # Maximum: 2
    assert maxDistance("NE", 0) == 2


def test_larger_k_than_needed():

    # Already optimal, or can keep "NN"
