import pytest
from solution import hasSpecialSubstring


def test_canonical_example_1():
    # spec: s = "aaabaaa", k = 3 -> true
    # substring s[4..6] == "aaa" satisfies: length 3, all same char, char before is 'b' (different)
    assert hasSpecialSubstring("aaabaaa", 3) is True


def test_canonical_example_2():
    # spec: s = "abc", k = 2 -> false
    # no substring of length 2 with one distinct character satisfies conditions
    assert hasSpecialSubstring("abc", 2) is False


def test_single_character_string_k_equals_1():
    # s = "a", k = 1 -> substring "a" at index 0 has length 1, all same char, no chars before/after
    assert hasSpecialSubstring("a", 1) is True


def test_all_same_characters_k_equals_length():
    # s = "aaaa", k = 4 -> entire string is one char, no chars before/after
    assert hasSpecialSubstring("aaaa", 4) is True


def test_all_same_characters_k_less_than_length():
    # s = "aaaa", k = 3 -> substring "aaa" at index 0 works (no char before, char after is 'a' which is same)
    # but substring at index 1 has 'a' before and after, so doesn't work
    # Actually s[0..2] = "aaa" has no char before but s[3]='a' after, so fails
    # s[1..3] = "aaa" has s[0]='a' before, so fails
    # This should be false since all substrings of length 3 either have same char before or after
    assert hasSpecialSubstring("aaaa", 3) is False


def test_substring_at_start_no_char_before():
    # s = "aaab", k = 3 -> s[0..2] = "aaa", no char before, char after is 'b' (different)
    assert hasSpecialSubstring("aaab", 3) is True


def test_substring_at_end_no_char_after():
    # s = "baaa", k = 3 -> s[1..3] = "aaa", char before is 'b' (different), no char after
    assert hasSpecialSubstring("baaa", 3) is True


def test_substring_in_middle_different_chars_around():
    # s = "baaac", k = 3 -> s[1..3] = "aaa", char before is 'b', char after is 'c' (both different)
    assert hasSpecialSubstring("baaac", 3) is True


def test_substring_same_char_before():
    # s = "aaaab", k = 3 -> s[0..2] has char after 'a' (same), s[1..3] has char before 'a' (same)
    # s[0..2] = "aaa" has no char before but s[3]='a' after (same char), fails
    # s[1..3] = "aaa" has s[0]='a' before (same char), fails
    # should be false
    assert hasSpecialSubstring("aaaab", 3) is False


def test_substring_same_char_after():
    # s = "baaaa", k = 3 -> s[1..3] = "aaa", char before is 'b' (different), char after is 'a' (same)
    # should be false
    assert hasSpecialSubstring("baaaa", 3) is False


def test_multiple_distinct_chars_no_repeats():
    # s = "abcde", k = 1 -> s[0]="a", no char before (start), char after is 'b' (different)
    assert hasSpecialSubstring("abcde", 1) is True


def test_two_char_groups_separated():
    # s = "aabaa", k = 2 -> s[0..1]="aa" (no before, after is 'b'), s[3..4]="aa" (before is 'b', no after)
    assert hasSpecialSubstring("aabaa", 2) is True


def test_k_equals_1_in_middle_different_neighbors():
    # s = "aba", k = 1 -> s[1]="b", char before is 'a', char after is 'a' (both different from 'b')
    assert hasSpecialSubstring("aba", 1) is True


def test_k_equals_1_same_neighbors():
    # s = "aaa", k = 1 -> s[0]="a" (no before, after is 'a' same), s[1]="a" (before 'a' same, after 'a' same), s[2]="a" (before 'a' same, no after)
    # all positions fail
    assert hasSpecialSubstring("aaa", 1) is False


def test_longer_string_with_valid_substring():
    # s = "abbbbaaa", k = 4 -> s[1..4]="bbbb", char before is 'a', char after is 'a' (both different)
    assert hasSpecialSubstring("abbbbaaa", 4) is True


def test_exact_k_length_group_isolated():
    # s = "aabbbcc", k = 3 -> s[2..4]="bbb", char before is 'a', char after is 'c' (both different)
    assert hasSpecialSubstring("aabbbcc", 3) is True
