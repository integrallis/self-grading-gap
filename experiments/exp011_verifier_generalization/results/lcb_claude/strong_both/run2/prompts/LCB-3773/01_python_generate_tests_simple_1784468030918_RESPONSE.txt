from typing import List
import pytest
from solution import minimumPairRemoval


def test_canonical_example_1():
    # spec: [5,2,3,1] -> 2
    # (3,1) sum=4 -> [5,2,4]
    # (2,4) sum=6 -> [5,6]
    # 2 operations to make non-decreasing
    assert minimumPairRemoval([5, 2, 3, 1]) == 2


def test_canonical_example_2():
    # spec: [1,2,2] -> 0
    # Already non-decreasing
    assert minimumPairRemoval([1, 2, 2]) == 0


def test_single_element():
    # Single element array is already non-decreasing

def test_two_elements_non_decreasing():
    # [1, 2] is already non-decreasing

def test_two_elements_decreasing():
    # [2, 1] needs one merge -> [3]
    # One operation to make non-decreasing (single element)

def test_already_sorted_ascending():
    # [1, 2, 3, 4] is already non-decreasing
    assert minimumPairRemoval([1, 2, 3, 4]) == 0


def test_reverse_sorted():
    # [4, 3, 2, 1]
    # Minimum pair (2,1) sum=3 -> [4,3,3]
    # Minimum pair (3,3) sum=6 (leftmost at index 1) -> [4,6]
    # Minimum pair (4,6) sum=10 -> [10]
    # 3 operations
    assert minimumPairRemoval([4, 3, 2, 1]) == 3

    # (3,3) sum=6, (3,2) sum=5, minimum is 5 -> [3, 5]
    # 2 operations
    assert minimumPairRemoval([1, 2, 3, 2]) == 2

    # Minimum sum is 2, leftmost is at idx 0 -> [2, 1, 2, 1]
    # Pairs: (2,1)=3 at idx 0, (1,2)=3 at idx 1, (2,1)=3 at idx 2
    # Minimum sum is 3, leftmost is at idx 0 -> [3, 2, 1]
    # Pairs: (3,2)=5, (2,1)=3
    # Minimum sum is 3 -> [3, 3]
    # Already non-decreasing, 3 operations
    assert minimumPairRemoval([1, 1, 1, 2, 1]) == 3
