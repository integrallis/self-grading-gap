from typing import List
import pytest
from solution import maxFrequency


def test_canonical_example_1():
    # spec: nums = [1,2,3,4,5,6], k = 1 -> output 2
    # After adding -5 to nums[2..5], 1 has a frequency of 2 in [1, 2, -2, -1, 0, 1]
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2


def test_canonical_example_2():
    # spec: nums = [10,2,3,4,5,5,4,3,2,2], k = 10 -> output 4
    # After adding 8 to nums[1..9], 10 has a frequency of 4 in [10, 10, 11, 12, 13, 13, 12, 11, 10, 10]
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4


def test_single_element_array_matching_k():
    # nums = [5], k = 5
    # Initial frequency of 5 is 1, no operation needed
    assert maxFrequency([5], 5) == 1


def test_single_element_array_not_matching_k():
    # nums = [3], k = 7
    # Can add 4 to nums[0..0] to make it [7], frequency of 7 becomes 1
    assert maxFrequency([3], 7) == 1


def test_all_elements_already_equal_to_k():
    # nums = [5, 5, 5, 5], k = 5
    # All elements are already 5, frequency is 4
    assert maxFrequency([5, 5, 5, 5], 5) == 4


def test_can_convert_entire_array_to_k():
    # nums = [3, 3, 3], k = 10
    # Add 7 to nums[0..2] to get [10, 10, 10], frequency of 10 is 3
    assert maxFrequency([3, 3, 3], 10) == 3


def test_some_elements_already_k():
    # nums = [1, 5, 1, 5], k = 1
    # Initial frequency of 1 is 2, adding -4 to nums[1..3] gives [1, 1, -3, 1], frequency is 3
    # Or adding -4 to nums[1..1] gives [1, 1, 1, 5], frequency is 3
    assert maxFrequency([1, 5, 1, 5], 1) == 3


def test_convert_subarray_in_middle():
    # nums = [10, 3, 3, 10], k = 10
    # Add 7 to nums[1..2] to get [10, 10, 10, 10], frequency of 10 is 4
    assert maxFrequency([10, 3, 3, 10], 10) == 4


def test_no_operation_better_than_any_operation():
    # nums = [7, 7, 7, 1], k = 7
    # Initial frequency of 7 is 3, any operation on subarray won't improve this
    assert maxFrequency([7, 7, 7, 1], 7) == 3


def test_convert_prefix_subarray():
    # nums = [2, 2, 2, 5], k = 10
    # Add 8 to nums[0..2] to get [10, 10, 10, 5], frequency of 10 is 3
    assert maxFrequency([2, 2, 2, 5], 10) == 3


def test_convert_suffix_subarray():
    # nums = [5, 2, 2, 2], k = 10
    # Add 8 to nums[1..3] to get [5, 10, 10, 10], frequency of 10 is 3
    assert maxFrequency([5, 2, 2, 2], 10) == 3


def test_two_element_array():
    # nums = [1, 2], k = 5
    # Add 4 to nums[0..0] to get [5, 2], frequency of 5 is 1
    # Add 3 to nums[1..1] to get [1, 5], frequency of 5 is 1
    # Add 4 to nums[0..1] to get [5, 6], frequency of 5 is 1
    # Add 3 to nums[0..1] to get [4, 5], frequency of 5 is 1
    assert maxFrequency([1, 2], 5) == 1


def test_array_with_repeated_values():
    # nums = [3, 3, 5, 5, 3], k = 3
    # Initial frequency of 3 is 3, can add -2 to nums[2..3] to get [3, 3, 3, 3, 3], frequency is 5
    assert maxFrequency([3, 3, 5, 5, 3], 3) == 5
