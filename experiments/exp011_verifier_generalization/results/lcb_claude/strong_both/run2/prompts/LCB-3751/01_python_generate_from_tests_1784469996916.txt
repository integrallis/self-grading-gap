from typing import List

def maxFrequency(nums: List[int], k: int) -> int:
    max_freq = nums.count(k)  # Start with the initial frequency of k
    n = len(nums)
    for i in range(n):  # Iterate over every starting index of the subarray
        for j in range(i, n):  # Iterate over every ending index of the subarray
            if nums[i] == nums[j]:  # Only consider if the subarray can be made equal
                add_value = k - nums[i]  # Calculate the value to add to make them equal to k
                new_freq = sum(1 for x in nums if x + (add_value if i <= nums.index(x) <= j else 0) == k)
                max_freq = max(max_freq, new_freq)  # Update the maximum frequency
    return max_freq
```
