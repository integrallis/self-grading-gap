from typing import List

def maxFrequency(nums: List[int], k: int) -> int:
    freq = [0] * 51  # Since nums[i] is in range 1 to 50
    for num in nums:
        freq[num] += 1
    max_freq = 0
    for x in range(1, 51):  # Check each possible value from 1 to 50
        if x + k <= 50:
            total = freq[x] + freq[x + k]
            for j in range(1, k + 1):
                total += freq[x + j]
            max_freq = max(max_freq, total)
    return max_freq
```
