from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # We will use dynamic programming to solve the problem
    dp = [[0] * 3 for _ in range(n)]
    # Initialize the first house costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the dp table
    for i in range(1, n):
        for j in range(3):
            # Cost of painting the i-th house with color j
            dp[i][j] = cost[i][j] + min(
                dp[i-1][(j+1) % 3],  # Color not the same as the previous house
                dp[i-1][(j+2) % 3]   # Ensure it's not the same as the previous
            )
            # Ensure equidistant house constraint
            if i < n // 2:
                equidistant_index = n - 1 - i
                if equidistant_index >= 0:
                    dp[equidistant_index][j] = min(
                        dp[equidistant_index][j],
                        cost[equidistant_index][j] + min(
                            dp[i][(j+1) % 3],
                            dp[i][(j+2) % 3]
                        )
                    )

    # The answer is the minimum cost of painting the last house
    return min(dp[n-1])
```
