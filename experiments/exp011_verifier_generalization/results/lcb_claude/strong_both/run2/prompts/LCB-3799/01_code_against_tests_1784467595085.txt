```python
from typing import List
from itertools import permutations

def totalNumbers(digits: List[int]) -> int:
    even_digits = [d for d in digits if d % 2 == 0]
    distinct_numbers = set()
    for last_digit in even_digits:
        # Create a new list of digits without the last digit
        remaining_digits = digits.copy()
        remaining_digits.remove(last_digit)
        # Generate permutations of the remaining digits for the first two positions
        for perm in permutations(remaining_digits, 2):
            first, middle = perm
            if first != 0:  # Ensure no leading zero
                number = first * 100 + middle * 10 + last_digit
                distinct_numbers.add(number)
    return len(distinct_numbers)
```
