```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirror_index = {chr(i): chr(219 - i) for i in range(97, 123)}  # a-z to z-a mapping

    for i in range(len(s)):
        if marked[i]:  # Skip if already marked
            continue
        mirror_char = mirror_index[s[i]]
        for j in range(i - 1, -1, -1):  # Check for unmarked characters before i
            if marked[j]:
                continue
            if s[j] == mirror_char:
                marked[i] = marked[j] = True  # Mark both
                score += i - j
                break
    return score

```
