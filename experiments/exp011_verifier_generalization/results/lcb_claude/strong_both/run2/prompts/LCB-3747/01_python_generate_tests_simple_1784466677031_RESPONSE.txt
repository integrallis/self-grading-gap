from solution import maxAdjacentDistance
from typing import List


def test_canonical_example_1():
    # spec: maxAdjacentDistance([1,2,4]) -> 3
    # Explanation: circular array, so pairs are (1,2), (2,4), (4,1)
    # |1-2| = 1, |2-4| = 2, |4-1| = 3, max = 3
    assert maxAdjacentDistance([1, 2, 4]) == 3


def test_canonical_example_2():
    # spec: maxAdjacentDistance([-5,-10,-5]) -> 5
    # Explanation: pairs are (-5,-10), (-10,-5), (-5,-5)
    # |-5-(-10)| = 5, |-10-(-5)| = 5, |-5-(-5)| = 0, max = 5
    assert maxAdjacentDistance([-5, -10, -5]) == 5


def test_two_elements_positive():
    # Two elements with positive values
    # pairs: (3,7), (7,3) - same difference
    # |3-7| = 4
    assert maxAdjacentDistance([3, 7]) == 4


def test_two_elements_negative():
    # Two elements with negative values
    # pairs: (-8,-2), (-2,-8)
    # |-8-(-2)| = 6
    assert maxAdjacentDistance([-8, -2]) == 6


def test_all_same_elements():
    # All elements are the same
    # All differences are 0
    assert maxAdjacentDistance([5, 5, 5, 5]) == 0


def test_alternating_values():
    # Alternating high and low values
    # [1, 10, 1, 10]: pairs (1,10), (10,1), (1,10), (10,1)
    # |1-10| = 9
    assert maxAdjacentDistance([1, 10, 1, 10]) == 9


def test_increasing_sequence():
    # Strictly increasing sequence
    # [1,2,3,4,5]: consecutive diffs are 1, but wrapping (5,1) = 4
    # |1-2| = 1, |2-3| = 1, |3-4| = 1, |4-5| = 1, |5-1| = 4
    assert maxAdjacentDistance([1, 2, 3, 4, 5]) == 4


def test_decreasing_sequence():
    # Strictly decreasing sequence
    # [10,8,6,4,2]: consecutive diffs are 2, but wrapping (2,10) = 8
    # |10-8| = 2, |8-6| = 2, |6-4| = 2, |4-2| = 2, |2-10| = 8
    assert maxAdjacentDistance([10, 8, 6, 4, 2]) == 8

    # [-10, 20, -5]: pairs (-10,20), (20,-5), (-5,-10)

    assert maxAdjacentDistance([-10, 20, -5]) == 30

    # Array includes zero
    # [0, 50, -50]: pairs (0,50), (50,-50), (-50,0)
    # |0-50| = 50, |50-(-50)| = 100, |-50-0| = 50
    assert maxAdjacentDistance([0, 50, -50]) == 100


def test_boundary_values():
    # Test with boundary values from constraints
