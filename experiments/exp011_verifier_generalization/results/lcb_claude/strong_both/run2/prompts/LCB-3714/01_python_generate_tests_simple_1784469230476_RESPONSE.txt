from typing import List
import pytest
from solution import minMaxSums


def test_canonical_example_1():
    # spec: nums = [1,2,3], k = 2 -> 24
    # Subsequences with at most 2 elements:
    # [1]: min=1, max=1, sum=2
    # [2]: min=2, max=2, sum=4
    # [3]: min=3, max=3, sum=6
    # [1,2]: min=1, max=2, sum=3
    # [1,3]: min=1, max=3, sum=4
    # [2,3]: min=2, max=3, sum=5
    # Total: 2+4+6+3+4+5 = 24
    assert minMaxSums([1, 2, 3], 2) == 24


def test_canonical_example_2():
    # spec: nums = [5,0,6], k = 1 -> 22
    # Subsequences with exactly 1 element (at most 1):
    # [5]: min=5, max=5, sum=10
    # [0]: min=0, max=0, sum=0
    # [6]: min=6, max=6, sum=12
    # Total: 10+0+12 = 22
    assert minMaxSums([5, 0, 6], 1) == 22


def test_canonical_example_3():
    # spec: nums = [1,1,1], k = 2 -> 12
    # Subsequences with at most 2 elements:
    # [1]: appears 3 times, each contributes min=1, max=1, sum=2, total=6
    # [1,1]: appears 3 times (C(3,2)=3), each contributes min=1, max=1, sum=2, total=6
    # Total: 6+6 = 12
    assert minMaxSums([1, 1, 1], 2) == 12


def test_single_element_array():
    # Single element [5], k=1: only subsequence is [5], min=5, max=5, sum=10
    assert minMaxSums([5], 1) == 10


def test_k_equals_array_length():
    # nums = [1,2], k = 2 (k equals length)
    # Subsequences: [1], [2], [1,2]
    # [1]: min=1, max=1, sum=2
    # [2]: min=2, max=2, sum=4
    # [1,2]: min=1, max=2, sum=3
    # Total: 2+4+3 = 9
    assert minMaxSums([1, 2], 2) == 9


def test_large_numbers_with_modulo():
    # Test that modulo 10^9 + 7 is applied
    # nums = [1000000000, 1000000000], k = 2
    # [1000000000]: min=max=1000000000, sum=2000000000
    # This appears twice (one for each element)
    # [1000000000, 1000000000]: min=max=1000000000, sum=2000000000
    # Total before modulo: 2*2000000000 + 2000000000 = 6000000000
    # 6000000000 % (10^9 + 7) = 6000000000 - 5*(10^9 + 7) = 6000000000 - 5000000035 = 999999965
    assert minMaxSums([1000000000, 1000000000], 2) == 999999965


def test_all_zeros():
    # nums = [0, 0, 0], k = 2
    # All subsequences have min=0, max=0, sum=0
    # Total: 0
    assert minMaxSums([0, 0, 0], 2) == 0


def test_two_elements_k_one():
    # nums = [3, 7], k = 1
    # Subsequences: [3], [7]
    # [3]: min=3, max=3, sum=6
    # [7]: min=7, max=7, sum=14
    # Total: 6+14 = 20
    assert minMaxSums([3, 7], 1) == 20


def test_multiple_identical_elements():
    # nums = [2, 2], k = 2
    # Subsequences: [2] (appears twice), [2, 2] (appears once)
    # [2]: min=2, max=2, sum=4, appears 2 times -> 8
    # [2, 2]: min=2, max=2, sum=4, appears 1 time -> 4
    # Total: 8+4 = 12
    assert minMaxSums([2, 2], 2) == 12


def test_k_larger_than_needed():
    # nums = [1, 2], k = 5 (k > length, so all subsequences)
    # Subsequences: [1], [2], [1,2]
    # [1]: min=1, max=1, sum=2
    # [2]: min=2, max=2, sum=4
    # [1,2]: min=1, max=2, sum=3
    # Total: 2+4+3 = 9
    assert minMaxSums([1, 2], 5) == 9
