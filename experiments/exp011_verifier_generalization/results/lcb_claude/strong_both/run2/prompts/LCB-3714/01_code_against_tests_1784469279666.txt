```python
from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Precompute the contributions of each number
    for i in range(n):
        # Each number will contribute as a minimum and maximum
        # in various combinations of subsequences.
        # Contributions as minimums
        min_contributions = 1 << (min(k, n - 1)) - 1
        # Contributions as maximums
        max_contributions = 1 << (min(k, n - 1)) - 1

        # If the number can be the only number in a subsequence
        if k > 0:
            min_contributions += 1
            max_contributions += 1

        total_sum += (min_contributions * nums[i] + max_contributions * nums[i]) % MOD
        total_sum %= MOD

    return total_sum % MOD

```
