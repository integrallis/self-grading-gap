    # [1,1]: appears 3 times (C(3,2)=3), each contributes min=1, max=1, sum=2, total=6
    # Total: 6+6 = 12
    assert minMaxSums([1, 1, 1], 2) == 12


def test_single_element_array():
    # Single element [5], k=1: only subsequence is [5], min=5, max=5, sum=10
    assert minMaxSums([5], 1) == 10


def test_k_equals_array_length():
    # nums = [1,2], k = 2 (k equals length)
    # Subsequences: [1], [2], [1,2]
    # [1]: min=1, max=1, sum=2
    # [2]: min=2, max=2, sum=4
    # [1,2]: min=1, max=2, sum=3
    # Total: 2+4+3 = 9
    assert minMaxSums([1, 2], 2) == 9


def test_large_numbers_with_modulo():
    # Test that modulo 10^9 + 7 is applied
    # nums = [1000000000, 1000000000], k = 2
    # [1000000000]: min=max=1000000000, sum=2000000000
    # This appears twice (one for each element)
    # [1000000000, 1000000000]: min=max=1000000000, sum=2000000000
    # Total before modulo: 2*2000000000 + 2000000000 = 6000000000
    # 6000000000 % (10^9 + 7) = 6000000000 - 5*(10^9 + 7) = 6000000000 - 5000000035 = 999999965
    assert minMaxSums([1000000000, 1000000000], 2) == 999999965

    # [2, 2]: min=2, max=2, sum=4, appears 1 time -> 4
    # Total: 8+4 = 12
    assert minMaxSums([2, 2], 2) == 12


def test_k_larger_than_needed():

    # [1,2]: min=1, max=2, sum=3
    # Total: 2+4+3 = 9
    assert minMaxSums([1, 2], 5) == 9
