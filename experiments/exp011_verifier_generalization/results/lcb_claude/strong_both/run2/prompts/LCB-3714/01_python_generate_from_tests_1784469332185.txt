```python
from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    
    # Count the contribution of each number as min and max in subsequences
    for i in range(n):
        # Calculate the number of subsequences of size j that include nums[i]
        min_contributions = 0
        max_contributions = 0
        for j in range(1, k + 1):
            if j <= n:
                # Selecting j-1 from the rest of the n-1 numbers
                combinations = comb(n - 1, j - 1)
                min_contributions += combinations
                max_contributions += combinations
        total_sum += (min_contributions * nums[i] + max_contributions * nums[i]) % MOD
        total_sum %= MOD

    return total_sum % MOD


def comb(n: int, k: int) -> int:
    if k > n or k < 0:
        return 0
    if k == 0 or k == n:
        return 1
    k = min(k, n - k)  # Take advantage of symmetry
    c = 1
    for i in range(k):
        c = c * (n - i) // (i + 1)
    return c
```
