from solution import maxAdjacentDistance
from typing import List


def test_canonical_example_1():
    # spec: [1,2,4] -> 3 (|4 - 1| = 3 between first and last elements)
    assert maxAdjacentDistance([1, 2, 4]) == 3


def test_canonical_example_2():
    # spec: [-5,-10,-5] -> 5 (|-5 - (-10)| = 5 between adjacent elements)
    assert maxAdjacentDistance([-5, -10, -5]) == 5


def test_two_elements():
    # minimum length array (2 elements): adjacent difference is |5 - 3| = 2
    assert maxAdjacentDistance([3, 5]) == 2


def test_all_same_elements():
    # all elements same: max absolute difference is |5 - 5| = 0
    assert maxAdjacentDistance([5, 5, 5, 5]) == 0


def test_increasing_sequence():
    # increasing sequence [1,2,3,4]: consecutive diffs are 1, wrap-around is |4-1| = 3
    assert maxAdjacentDistance([1, 2, 3, 4]) == 3


def test_decreasing_sequence():
    # decreasing sequence [10,7,4,1]: consecutive diffs are 3, wrap-around is |1-10| = 9
    assert maxAdjacentDistance([10, 7, 4, 1]) == 9


def test_negative_numbers():
    # negative numbers [-10, -20, -15]: diffs are |-10-(-20)| = 10, |-20-(-15)| = 5, |-15-(-10)| = 5

    # mixed: [10, -10, 20]: diffs are |10-(-10)| = 20, |-10-20| = 30, |20-10| = 10
    assert maxAdjacentDistance([10, -10, 20]) == 30


def test_wrap_around_is_maximum():
    # wrap-around maximum: [0, 1, 2, 100]: wrap |100-0| = 100 is max
