import pytest
from solution import findValidPair


def test_canonical_example_1():
    # spec: s = "2523533" -> "23"
    # '2' appears 2 times, '3' appears 3 times
    # "23" is the first valid pair (adjacent, different, counts match values)
    assert findValidPair("2523533") == "23"


def test_canonical_example_2():
    # spec: s = "221" -> "21"
    # '2' appears 2 times, '1' appears 1 time
    # "21" is a valid pair
    assert findValidPair("221") == "21"


def test_canonical_example_3():
    # spec: s = "22" -> ""
    # No valid adjacent pairs (digits must be different)
    assert findValidPair("22") == ""


def test_minimum_length_valid_pair():
    # s = "12": '1' appears 1 time, '2' appears 1 time
    # "12" is valid only if '2' appears 2 times, but it appears 1 time
    # So no valid pair
    assert findValidPair("12") == ""


def test_minimum_length_with_valid_pair():
    # s = "21": '2' appears 1 time, '1' appears 1 time
    # For "21" to be valid: '2' must appear 2 times (it appears 1), '1' must appear 1 time (it does)
    # Not valid
    assert findValidPair("21") == ""


def test_all_same_digit():
    # s = "333": '3' appears 3 times (matches its value)
    # But no adjacent pair has different digits
    assert findValidPair("333") == ""


def test_valid_pair_at_start():
    # s = "2134": '2' appears 1 time, '1' appears 1 time, '3' appears 1 time, '4' appears 1 time
    # No digit appears as many times as its value except '1'
    # "21" has '2' appearing 1 time (needs 2), '1' appearing 1 time (correct) - not valid
    assert findValidPair("2134") == ""


def test_valid_pair_in_middle():
    # s = "52235": '5' appears 2 times, '2' appears 2 times, '3' appears 1 time
    # "22" - same digits, not valid
    # "23" - '2' appears 2 times (correct), '3' appears 1 time (needs 3) - not valid
    # "35" - '3' appears 1 time (needs 3), '5' appears 2 times (needs 5) - not valid
    assert findValidPair("52235") == ""


def test_first_valid_pair_when_multiple_exist():
    # s = "211333": '2' appears 1 time, '1' appears 2 times, '3' appears 3 times
    # "21" - '2' appears 1 time (needs 2), not valid
    # "11" - same digits, not valid
    # "13" - '1' appears 2 times (needs 1), not valid
    # "33" - same digits, not valid
    assert findValidPair("211333") == ""


def test_valid_pair_with_correct_counts():
    # s = "1133322": '1' appears 2 times, '3' appears 3 times, '2' appears 2 times
    # "11" - same digits, not valid
    # "13" - '1' appears 2 times (needs 1), '3' appears 3 times (correct) - not valid
    # "33" - same digits, not valid
    # "32" - '3' appears 3 times (correct), '2' appears 2 times (correct) - VALID!
    assert findValidPair("1133322") == "32"


def test_later_valid_pair():
    # s = "44112": '4' appears 2 times, '1' appears 2 times, '2' appears 1 time
    # "44" - same digits, not valid
    # "41" - '4' appears 2 times (needs 4), not valid
    # "11" - same digits, not valid
    # "12" - '1' appears 2 times (needs 1), not valid
    assert findValidPair("44112") == ""


def test_digit_one_appears_once():
    # s = "2211": '2' appears 2 times, '1' appears 2 times
    # "22" - same digits, not valid
    # "21" - '2' appears 2 times (correct), '1' appears 2 times (needs 1) - not valid
    # "11" - same digits, not valid
    assert findValidPair("2211") == ""


def test_multiple_different_valid_pairs():
    # s = "1223": '1' appears 1 time, '2' appears 2 times, '3' appears 1 time
    # "12" - '1' appears 1 time (correct), '2' appears 2 times (correct) - VALID!
    assert findValidPair("1223") == "12"
