from typing import List

def maxSum(nums: List[int]) -> int:
    seen = set()
    max_sum = 0
    current_sum = 0

    for num in nums:
        if num not in seen:
            seen.add(num)
            current_sum += num
        else:
            max_sum = max(max_sum, current_sum)
            while num in seen:
                seen.remove(nums.pop(0))
                current_sum -= nums[0] if nums else 0
            seen.add(num)
            current_sum += num
    return max(max_sum, current_sum)
```
