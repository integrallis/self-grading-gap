from typing import List
import pytest
from solution import maxSum


def test_canonical_example_1():
    # spec: nums = [1,2,3,4,5] -> 15
    # Select the entire array without deleting any element to obtain the maximum sum.
    assert maxSum([1, 2, 3, 4, 5]) == 15


def test_canonical_example_2():
    # spec: nums = [1,1,0,1,1] -> 1
    # Delete the element nums[0] == 1, nums[1] == 1, nums[2] == 0, and nums[3] == 1.
    # Select the entire array [1] to obtain the maximum sum.
    assert maxSum([1, 1, 0, 1, 1]) == 1


def test_canonical_example_3():
    # spec: nums = [1,2,-1,-2,1,0,-1] -> 3
    # Delete the elements nums[2] == -1 and nums[3] == -2, and select the subarray [2, 1]
    # from [1, 2, 1, 0, -1] to obtain the maximum sum.
    assert maxSum([1, 2, -1, -2, 1, 0, -1]) == 3


def test_single_element():
    # Single element array: only option is to select that element
    # 1 <= nums.length, so single element is valid
    assert maxSum([5]) == 5


def test_single_negative_element():
    # Single element array with negative value
    # Must return that element since array cannot be empty
    assert maxSum([-10]) == -10


def test_all_negative_elements():
    # All negative elements: best strategy is to select the largest single element
    # Maximum sum would be the least negative value
    assert maxSum([-5, -3, -7]) == -3


def test_all_same_positive_elements():
    # All elements are the same positive value
    # Can only select one element to keep uniqueness
    assert maxSum([4, 4, 4, 4]) == 4


def test_two_elements_unique():
    # Two unique elements: select both
    assert maxSum([3, 7]) == 10


def test_two_elements_duplicate():
    # Two duplicate elements: can only select one
    assert maxSum([5, 5]) == 5


def test_mix_positive_and_negative_with_duplicates():
    # Mixed values with duplicates: need to find best unique subarray
    # [3, -1, 3] -> can select [3] for sum 3, or delete first 3 and select [-1, 3] for sum 2
    # Best is [3] = 3
    assert maxSum([3, -1, 3]) == 3


def test_alternating_positive_negative():
    # Alternating positive and negative values
    # [5, -2, 3, -1] -> can select [5, -2, 3] = 6 or [5, 3] after deleting -2
    # Actually [5, 3] requires selecting from modified array after deletion
    # Best unique subarray: [5, -2, 3] = 6
    assert maxSum([5, -2, 3, -1]) == 6


def test_zero_in_array():
    # Array with zeros
    # [1, 0, 2] -> select [1, 0, 2] = 3
    assert maxSum([1, 0, 2]) == 3


def test_all_zeros():
    # All zeros: can only select one zero
    assert maxSum([0, 0, 0]) == 0


def test_positive_then_duplicates():
    # Positive values followed by duplicates
    # [10, 20, 10] -> can select [10, 20] = 30 or [20, 10] = 30
    assert maxSum([10, 20, 10]) == 30


def test_large_values_within_constraint():
    # Test with boundary values: -100 <= nums[i] <= 100
    assert maxSum([100, -100, 100]) == 100


def test_subarray_in_middle():
    # Best subarray is in the middle after deletions
    # [-5, 10, 20, -5] -> select [10, 20] = 30
    assert maxSum([-5, 10, 20, -5]) == 30


def test_need_to_skip_negative():
    # [1, 2, -10, 3, 4] -> could select [1, 2, 3, 4] after deleting -10 = 10
    # Or select [1, 2] = 3, or [3, 4] = 7
    # After deleting -10: [1, 2, 3, 4] -> select all unique elements = 10
    assert maxSum([1, 2, -10, 3, 4]) == 10
