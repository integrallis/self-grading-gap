from typing import List

def maxSum(nums: List[int]) -> int:
    max_sum = float('-inf')
    n = len(nums)
    seen = set()

    for left in range(n):
        current_sum = 0
        seen.clear()  # Reset seen set for each left position
        for right in range(left, n):
            if nums[right] not in seen:
                seen.add(nums[right])
                current_sum += nums[right]
                max_sum = max(max_sum, current_sum)
            else:
                break  # Stop if we hit a duplicate

    return max_sum
```
