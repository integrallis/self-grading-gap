from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize dp array to store minimum costs
    dp = [[0] * 3 for _ in range(n)]

    # Base case for the first house
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill dp array while considering adjacency and equidistant constraints
    for i in range(1, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(dp[i-1][(j+1) % 3], dp[i-1][(j+2) % 3])

    # Calculate minimum cost considering equidistant constraints
    min_cost = float('inf')
    for j in range(3):
        for k in range(3):
            if j != k:  # Ensure different colors for equidistant houses
                min_cost = min(min_cost, dp[n // 2 - 1][j] + dp[n - 1][k])

    return min_cost

```
