from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize dp arrays for two colorings
    dp1 = [[0] * 3 for _ in range(n)]  # Cost for first coloring pattern
    dp2 = [[0] * 3 for _ in range(n)]  # Cost for second coloring pattern

    # Base case for the first house
    for j in range(3):
        dp1[0][j] = cost[0][j]

    # Fill dp1 for the first half of the houses
    for i in range(1, n // 2):
        for j in range(3):
            dp1[i][j] = cost[i][j] + min(dp1[i-1][(j+1)%3], dp1[i-1][(j+2)%3])

    # Base case for the second half of the houses
    for j in range(3):
        dp2[n//2][j] = cost[n//2][j]

    # Fill dp2 for the second half of the houses
    for i in range(n//2 + 1, n):
        for j in range(3):
            dp2[i][j] = cost[i][j] + min(dp2[i-1][(j+1)%3], dp2[i-1][(j+2)%3])

    # Calculate the minimum costs considering equidistant constraints
    min_cost = float('inf')
    for j in range(3):
        for k in range(3):
            if j != k:  # Ensure different colors for equidistant houses
                min_cost = min(min_cost, dp1[n//2-1][j] + dp2[n-1][k])

    return min_cost
```
