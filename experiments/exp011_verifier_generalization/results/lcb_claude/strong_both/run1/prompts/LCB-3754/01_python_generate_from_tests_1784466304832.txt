def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    moves = []

    for move in s:
        if move == 'N':
            y += 1
        elif move == 'S':
            y -= 1
        elif move == 'E':
            x += 1
        elif move == 'W':
            x -= 1
        moves.append((x, y))
        max_distance = max(max_distance, abs(x) + abs(y))

    # Calculate how many steps we can change
    total_moves = len(s)
    # Use k changes to maximize the distance
    max_distance_with_changes = max_distance + k

    return max(max_distance, max_distance_with_changes)

```
