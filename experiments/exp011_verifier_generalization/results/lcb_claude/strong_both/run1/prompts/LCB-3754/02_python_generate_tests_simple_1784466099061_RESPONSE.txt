import pytest

def test_canonical_example_1():
    # spec: s = "NWSE", k = 1 -> output 3
    # Change s[2] from 'S' to 'N'. String becomes "NWNE".
    # Positions: (0,1), (-1,1), (-1,2), (0,2)
    # Manhattan distances: 1, 2, 3, 2
    # Maximum: 3
    assert maxDistance("NWSE", 1) == 3
