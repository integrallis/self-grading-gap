def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0

    for move in s:
        if move == 'N':
            y += 1
        elif move == 'S':
            y -= 1
        elif move == 'E':
            x += 1
        elif move == 'W':
            x -= 1

        # Calculate current distance
        current_distance = abs(x) + abs(y)
        # Check the maximum distance we can achieve by changing up to k moves
        max_distance = max(max_distance, current_distance + min(k, len(s) - (s.index(move) + 1)))

    return max_distance

```
