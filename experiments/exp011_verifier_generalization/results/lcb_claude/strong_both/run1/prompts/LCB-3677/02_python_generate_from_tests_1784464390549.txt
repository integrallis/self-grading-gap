from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[[float('-inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0] if coins[0][0] >= 0 else 0

    for i in range(m):
        for j in range(n):
            for k in range(3):  # k is the number of neutralizations used
                if i == 0 and j == 0:
                    continue
                current_value = coins[i][j]
                if current_value < 0:
                    if k > 0:
                        # Option to neutralize this robber
                        dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k-1] + 0)
                        if j > 0:
                            dp[i][j][k] = max(dp[i][j][k], dp[i][j-1][k-1] + 0)
                    # Normal robbery (not neutralized)
                    if k < 3:
                        dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k] - abs(current_value))
                        if j > 0:
                            dp[i][j][k] = max(dp[i][j][k], dp[i][j-1][k] - abs(current_value))
                else:
                    # Normal coin collection
                    dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k] + current_value)
                    if j > 0:
                        dp[i][j][k] = max(dp[i][j][k], dp[i][j-1][k] + current_value)

    # Get the maximum value at the bottom-right corner with up to 2 neutralizations
    return max(dp[m-1][n-1][0], dp[m-1][n-1][1], dp[m-1][n-1][2])
```
