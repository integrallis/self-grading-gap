```python
from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    # Calculate total contributions from each number for subsequences of size 1 to k
    for i in range(n):
        # Each nums[i] contributes as a minimum in (n - i) * 2^(n - i - 1) subsequences
        # It contributes as a maximum in (i + 1) * 2^(i) subsequences
        min_contrib = (n - i) * pow(2, n - i - 1, MOD) % MOD
        max_contrib = (i + 1) * pow(2, i, MOD) % MOD
        total_sum = (total_sum + nums[i] * (min_contrib + max_contrib)) % MOD
    return total_sum

```
