from typing import List
import pytest
from solution import minMaxSums


def test_canonical_example_1():
    # spec: nums = [1,2,3], k = 2 -> 24
    # Subsequences with at most 2 elements:
    # [1]: min=1, max=1, sum=2
    # [2]: min=2, max=2, sum=4
    # [3]: min=3, max=3, sum=6
    # [1,2]: min=1, max=2, sum=3
    # [1,3]: min=1, max=3, sum=4
    # [2,3]: min=2, max=3, sum=5
    # Total: 2+4+6+3+4+5 = 24
    assert minMaxSums([1, 2, 3], 2) == 24


def test_canonical_example_2():
    # spec: nums = [5,0,6], k = 1 -> 22
    # Subsequences with exactly 1 element:
    # [5]: min=5, max=5, sum=10
    # [0]: min=0, max=0, sum=0
    # [6]: min=6, max=6, sum=12
    # Total: 10+0+12 = 22
    assert minMaxSums([5, 0, 6], 1) == 22


def test_canonical_example_3():
    # spec: nums = [1,1,1], k = 2 -> 12
    # Subsequences with at most 2 elements:
    # [1] appears 3 times: each contributes min=1, max=1, sum=2, total=6
    # [1,1] appears 3 times: each contributes min=1, max=1, sum=2, total=6
    # Total: 6+6 = 12
    assert minMaxSums([1, 1, 1], 2) == 12


def test_single_element_array():
    # nums = [5], k = 1
    # Only subsequence: [5] with min=5, max=5, sum=10
    assert minMaxSums([5], 1) == 10


def test_k_equals_array_length():
    # nums = [1, 2], k = 2
    # Subsequences: [1], [2], [1,2]
    # [1]: min=1, max=1, sum=2
    # [2]: min=2, max=2, sum=4
    # [1,2]: min=1, max=2, sum=3
    # Total: 2+4+3 = 9
    assert minMaxSums([1, 2], 2) == 9


def test_large_values_with_modulo():
    # nums = [1000000000, 1000000000], k = 2
    # [1000000000]: appears twice, each sum = 2000000000
    # [1000000000, 1000000000]: sum = 2000000000
    # Total: 6000000000
    # 6000000000 % (10^9 + 7) = 6000000000 % 1000000007 = 5999999986
    assert minMaxSums([1000000000, 1000000000], 2) == 5999999986


def test_all_zeros():
    # nums = [0, 0, 0], k = 2
    # All subsequences have min=0, max=0, sum=0
    # Total: 0
    assert minMaxSums([0, 0, 0], 2) == 0


def test_two_element_array_k_one():
    # nums = [3, 7], k = 1
    # [3]: min=3, max=3, sum=6
    # [7]: min=7, max=7, sum=14
    # Total: 6+14 = 20
    assert minMaxSums([3, 7], 1) == 20


def test_different_values():
    # nums = [1, 2], k = 1
    # [1]: min=1, max=1, sum=2
    # [2]: min=2, max=2, sum=4
    # Total: 2+4 = 6
    assert minMaxSums([1, 2], 1) == 6
