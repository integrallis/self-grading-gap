from typing import List

MOD = 10**9 + 7


def binomial_coefficient(n: int, k: int) -> int:
    if k > n or k < 0:
        return 0
    if k == 0 or k == n:
        return 1
    num = den = 1
    for i in range(k):
        num = num * (n - i) % MOD
        den = den * (i + 1) % MOD
    return num * pow(den, MOD - 2, MOD) % MOD


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    # Sort the numbers to handle min and max contributions
    nums.sort()
    # Iterate over each number as a possible min or max
    for i in range(n):
        # For nums[i] as a minimum
        for size in range(1, k + 1):
            min_contrib = binomial_coefficient(n - i - 1, size - 1)
            # For nums[i] as maximum
            max_contrib = binomial_coefficient(i, size - 1)
            total_sum = (total_sum + nums[i] * (min_contrib + max_contrib)) % MOD
    return total_sum
