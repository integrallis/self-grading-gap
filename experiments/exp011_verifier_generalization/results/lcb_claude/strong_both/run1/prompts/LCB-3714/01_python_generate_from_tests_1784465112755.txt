from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    # Sort the numbers to handle min and max contributions
    nums.sort()
    # Calculate contributions for min and max for subsequences of size 1 to k
    for i in range(n):
        # Each nums[i] contributes as min in subsequences of size 1 to k
        min_contrib = min(1, k) * pow(2, min(k - 1, n - 1), MOD) % MOD
        # Each nums[i] contributes as max in subsequences of size 1 to k
        max_contrib = min(1, k) * pow(2, min(k - 1, n - 1), MOD) % MOD
        total_sum = (total_sum + nums[i] * (min_contrib + max_contrib)) % MOD
    return total_sum

```
