import pytest
from solution import calculateScore


def test_canonical_example_1():
    # spec: s = "aczzx" -> 5
    # i=0 'a': no mirror before
    # i=1 'c': no mirror before
    # i=2 'z': mirror is 'a' at j=0, mark both, score += 2-0 = 2
    # i=3 'z': mirror is 'a', but j=0 is marked, no unmarked mirror
    # i=4 'x': mirror is 'c' at j=1, mark both, score += 4-1 = 3
    # total = 2 + 3 = 5
    assert calculateScore("aczzx") == 5


def test_canonical_example_2():
    # spec: s = "abcdef" -> 0
    # No character has its mirror before it
    assert calculateScore("abcdef") == 0


def test_single_character():
    # Single character, no j < i exists
    assert calculateScore("a") == 0


def test_mirror_pair_adjacent():
    # s = "az": i=1, 'z' has mirror 'a' at j=0, score = 1-0 = 1
    assert calculateScore("az") == 1


def test_mirror_pair_separated():
    # s = "abz": i=2, 'z' has mirror 'a' at j=0, score = 2-0 = 2
    assert calculateScore("abz") == 2


def test_two_pairs():
    # s = "azby": 
    # i=1, 'z' has mirror 'a' at j=0, mark both, score = 1
    # i=2, 'b' has mirror 'y', none before it
    # i=3, 'y' has mirror 'b' at j=2, mark both, score += 3-2 = 1
    # total = 2
    assert calculateScore("azby") == 2


def test_closest_unmarked_index():
    # s = "aaz": multiple 'a's before 'z'
    # i=2, 'z' has mirror 'a', closest is j=1, score = 2-1 = 1
    assert calculateScore("aaz") == 1


def test_marked_indices_not_reused():
    # s = "azza": 
    # i=1, 'z' has mirror 'a' at j=0, mark both, score = 1
    # i=2, 'z' has mirror 'a', but j=0 is marked, no unmarked mirror
    # i=3, 'a' has mirror 'z' at j=2 (j=1 is marked), score += 3-2 = 2
    # total = 3
    assert calculateScore("azza") == 3


def test_same_letter_is_not_mirror():
    # s = "aa": 'a' mirror is 'z', not 'a'
    assert calculateScore("aa") == 0


def test_middle_letter():
    # s = "mn": 'm' mirror is 'n', 'n' mirror is 'm'
    # i=1, 'n' has mirror 'm' at j=0, score = 1
    assert calculateScore("mn") == 1


def test_multiple_pairs_all_match():
    # s = "azbyc": 
    # i=1, 'z' mirror 'a' at j=0, score = 1
    # i=2, 'b' mirror 'y', none before
    # i=3, 'y' mirror 'b' at j=2, score += 1
    # i=4, 'c' mirror 'x', none before
    # total = 2
    assert calculateScore("azbyc") == 2


def test_all_same_character():
    # s = "aaaa": 'a' mirror is 'z', no 'z' present
    assert calculateScore("aaaa") == 0


def test_reverse_alphabet_pairs():
    # s = "zyxwvu": each char's mirror comes after, so no matches
    assert calculateScore("zyxwvu") == 0


def test_long_distance_match():
    # s = "abcdefghijklmnopqrstuvwxyza"
    # i=26, 'a' mirror is 'z' at j=25, score = 26-25 = 1
    assert calculateScore("abcdefghijklmnopqrstuvwxyza") == 1
