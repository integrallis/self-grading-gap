If no such index j exists for the index i, move on to the next index without making any changes.

Return the total score at the end of the process.

```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # Create mirror mapping for 'a' to 'z'

    for i in range(len(s)):
        if marked[i]:  # Skip if already marked
            continue
        for j in range(i - 1, -1, -1):  # Look for closest unmarked mirror
            if marked[j]:  # Skip if marked
                continue
            if mirror[s[i]] == s[j]:  # Check for mirror match
                marked[i] = marked[j] = True  # Mark both
                score += (i - j)  # Update score with corrected calculation
                break  # Stop once we find the first

    return score
```

def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # Create mirror mapping for 'a' to 'z'

    for i in range(len(s)):
        if marked[i]:  # Skip if already marked
            continue
        for j in range(i - 1, -1, -1):  # Look for closest unmarked mirror
            if marked[j]:  # Skip if marked
                continue
            if mirror[s[i]] == s[j]:  # Check for mirror match
                marked[i] = marked[j] = True  # Mark both
