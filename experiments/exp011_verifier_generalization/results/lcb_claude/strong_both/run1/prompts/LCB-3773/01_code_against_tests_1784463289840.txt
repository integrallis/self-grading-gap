```python
from typing import List


def minimumPairRemoval(nums: List[int]) -> int:
    operations = 0
    while True:
        # Check if the array is already non-decreasing
        if all(nums[i] <= nums[i + 1] for i in range(len(nums) - 1)):
            break
        # Find the minimum adjacent pair
        min_sum = float('inf')
        min_index = -1
        for i in range(len(nums) - 1):
            current_sum = nums[i] + nums[i + 1]
            if current_sum < min_sum:
                min_sum = current_sum
                min_index = i
        # Merge the pair with the minimum sum
        nums[min_index] += nums[min_index + 1]
        del nums[min_index + 1]
        operations += 1
    return operations

```
