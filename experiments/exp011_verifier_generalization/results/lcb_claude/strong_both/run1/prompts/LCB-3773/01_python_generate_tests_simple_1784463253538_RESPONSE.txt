def test_canonical_example_1():
    # spec: [5,2,3,1] -> 2
    # Operation 1: pair (3,1) has min sum 4, nums = [5,2,4]
    # Operation 2: pair (2,4) has min sum 6, nums = [5,6]
    # Array is now non-decreasing in 2 operations
    assert minimumPairRemoval([5, 2, 3, 1]) == 2


def test_canonical_example_2():
    # spec: [1,2,2] -> 0
    # Array is already non-decreasing, no operations needed
    assert minimumPairRemoval([1, 2, 2]) == 0


def test_single_element():
    # Single element array is already non-decreasing
    assert minimumPairRemoval([5]) == 0


def test_two_elements_sorted():
    # Two element array already non-decreasing

    # Array [3, 2, 1] requires operations
    # Operation 1: pair (2,1) has min sum 3, nums = [3,3]

    assert minimumPairRemoval([3, 2, 1]) == 1


def test_negative_numbers():
    # Array with negative numbers [-5, -2, -3]
    # Pair (-5,-2) has sum -7, pair (-2,-3) has sum -5
    # Operation 1: pair (-5,-2) has min sum -7, nums = [-7,-3]
    # Array is non-decreasing in 1 operation
    assert minimumPairRemoval([-5, -2, -3]) == 1
