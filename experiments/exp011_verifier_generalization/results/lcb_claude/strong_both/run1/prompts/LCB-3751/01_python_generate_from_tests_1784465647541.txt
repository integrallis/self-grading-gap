    # Element already equals k, frequency = 1

    # Add 4 to nums[0..0] to get [7], frequency = 1

    # Add 5 to nums[0..2] to get [10, 10, 10], frequency = 3
    assert maxFrequency([5, 5, 5], 10) == 3


def test_transform_prefix_only():
    # nums = [1, 1, 1, 5], k = 3
    # Add 2 to nums[0..2] to get [3, 3, 3, 5], frequency = 3
    assert maxFrequency([1, 1, 1, 5], 3) == 3


def test_transform_suffix_only():
    # nums = [5, 1, 1, 1], k = 3
    # Add 2 to nums[1..3] to get [5, 3, 3, 3], frequency = 3

def test_transform_middle_subarray():
    # nums = [1, 5, 5, 1], k = 3
    # Add -2 to nums[1..2] to get [1, 3, 3, 1], frequency = 2
    assert maxFrequency([1, 5, 5, 1], 3) == 2

    # Add 5 to nums[2..3] to get [10, 10, 10, 10], frequency = 4

    # Add 2 to nums[2..2] to get [1, 2, 5], frequency = 1
    # Add 2 to nums[0..2] to get [3, 4, 5], frequency = 1

    # Add 5 to nums[1..2] to get [10, 10, 10], frequency = 3
    assert maxFrequency([10, 5, 5], 10) == 3


def test_two_elements_same_value():
    # nums = [7, 7], k = 7
    # Already have 2 occurrences of k=7
    assert maxFrequency([7, 7], 7) == 2


def test_two_elements_transform_both():
    # nums = [3, 3], k = 8
    # Add 5 to nums[0..1] to get [8, 8], frequency = 2
