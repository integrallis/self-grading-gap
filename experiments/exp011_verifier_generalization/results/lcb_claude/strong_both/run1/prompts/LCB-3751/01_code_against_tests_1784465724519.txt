```python
from typing import List

def maxFrequency(nums: List[int], k: int) -> int:
    count_k = nums.count(k)
    max_freq = count_k
    n = len(nums)
    left = 0
    while left < n:
        if nums[left] == k:
            left += 1
            continue
        right = left
        while right < n and nums[right] == nums[left]:
            right += 1
        length = right - left
        if nums[left] < k:
            # The needed value to reach k
            needed = k - nums[left]
            if length >= needed:
                max_freq = max(max_freq, length)
        left = right
    return max_freq
```
