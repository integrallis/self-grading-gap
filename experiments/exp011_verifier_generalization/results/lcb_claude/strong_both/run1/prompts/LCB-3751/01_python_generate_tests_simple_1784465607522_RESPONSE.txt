from typing import List
import pytest
from solution import maxFrequency


def test_canonical_example_1():
    # spec: nums = [1,2,3,4,5,6], k = 1
    # After adding -5 to nums[2..5], 1 has a frequency of 2 in [1, 2, -2, -1, 0, 1]
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2


def test_canonical_example_2():
    # spec: nums = [10,2,3,4,5,5,4,3,2,2], k = 10
    # After adding 8 to nums[1..9], 10 has a frequency of 4 in [10, 10, 11, 12, 13, 13, 12, 11, 10, 10]
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4


def test_single_element_already_equals_k():
    # nums = [5], k = 5
    # Element already equals k, frequency = 1
    assert maxFrequency([5], 5) == 1


def test_single_element_needs_transformation():
    # nums = [3], k = 7
    # Add 4 to nums[0..0] to get [7], frequency = 1
    assert maxFrequency([3], 7) == 1


def test_all_elements_already_equal_k():
    # nums = [3, 3, 3, 3], k = 3
    # All elements already equal k, frequency = 4
    assert maxFrequency([3, 3, 3, 3], 3) == 4


def test_transform_entire_array():
    # nums = [5, 5, 5], k = 10
    # Add 5 to nums[0..2] to get [10, 10, 10], frequency = 3
    assert maxFrequency([5, 5, 5], 10) == 3


def test_transform_prefix_only():
    # nums = [1, 1, 1, 5], k = 3
    # Add 2 to nums[0..2] to get [3, 3, 3, 5], frequency = 3
    assert maxFrequency([1, 1, 1, 5], 3) == 3


def test_transform_suffix_only():
    # nums = [5, 1, 1, 1], k = 3
    # Add 2 to nums[1..3] to get [5, 3, 3, 3], frequency = 3
    assert maxFrequency([5, 1, 1, 1], 3) == 3


def test_transform_middle_subarray():
    # nums = [1, 5, 5, 1], k = 3
    # Add -2 to nums[1..2] to get [1, 3, 3, 1], frequency = 2
    assert maxFrequency([1, 5, 5, 1], 3) == 2


def test_no_transformation_needed():
    # nums = [1, 2, 1, 2, 1], k = 1
    # Without transformation, k=1 appears 3 times; can't improve by transforming
    assert maxFrequency([1, 2, 1, 2, 1], 1) == 3


def test_transform_to_create_more_k_values():
    # nums = [10, 10, 5, 5], k = 10
    # Add 5 to nums[2..3] to get [10, 10, 10, 10], frequency = 4
    assert maxFrequency([10, 10, 5, 5], 10) == 4


def test_mixed_values_transform_some():
    # nums = [1, 2, 3], k = 5
    # Add 4 to nums[0..0] to get [5, 2, 3], frequency = 1
    # Add 3 to nums[1..1] to get [1, 5, 3], frequency = 1
    # Add 2 to nums[2..2] to get [1, 2, 5], frequency = 1
    # Add 2 to nums[0..2] to get [3, 4, 5], frequency = 1
    assert maxFrequency([1, 2, 3], 5) == 1


def test_k_appears_once_can_add_more():
    # nums = [10, 5, 5], k = 10
    # Add 5 to nums[1..2] to get [10, 10, 10], frequency = 3
    assert maxFrequency([10, 5, 5], 10) == 3


def test_two_elements_same_value():
    # nums = [7, 7], k = 7
    # Already have 2 occurrences of k=7
    assert maxFrequency([7, 7], 7) == 2


def test_two_elements_transform_both():
    # nums = [3, 3], k = 8
    # Add 5 to nums[0..1] to get [8, 8], frequency = 2
    assert maxFrequency([3, 3], 8) == 2
