```python
from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n < 2:
        return n

    # dp array where dp[i] will store the length of the longest subsequence ending with nums[i]
    dp = [1] * n

    # Iterate over all pairs of indices to check possible subsequences
    for i in range(n):
        for j in range(i):
            # Calculate the absolute difference
            diff1 = abs(nums[i] - nums[j])
            # We only consider valid subsequences where the current difference is less than or equal to the previous one
            if j == 0 or abs(nums[j] - nums[j - 1]) >= diff1:
                dp[i] = max(dp[i], dp[j] + 1)

    # The result is the maximum length found in dp
    return max(dp)
```
