from typing import List
import pytest
from solution import longestSubsequence


def test_canonical_example_1():
    # spec: nums = [16,6,3] -> 3
    # The longest subsequence is [16, 6, 3] with absolute differences [10, 3]
    # 10 >= 3, so this is valid and has length 3
    assert longestSubsequence([16, 6, 3]) == 3


def test_canonical_example_2():
    # spec: nums = [6,5,3,4,2,1] -> 4
    # The longest subsequence is [6, 4, 2, 1] with absolute differences [2, 2, 1]
    # 2 >= 2 >= 1, so this is valid and has length 4
    assert longestSubsequence([6, 5, 3, 4, 2, 1]) == 4


def test_canonical_example_3():
    # spec: nums = [10,20,10,19,10,20] -> 5
    # The longest subsequence is [10, 20, 10, 19, 10] with absolute differences [10, 10, 9, 9]
    # 10 >= 10 >= 9 >= 9, so this is valid and has length 5
    assert longestSubsequence([10, 20, 10, 19, 10, 20]) == 5


def test_minimal_length_array():
    # spec constraint: 2 <= nums.length
    # Minimal case with 2 elements - any two elements form a valid subsequence of length 2
    # (only one difference, which is trivially non-increasing)
    assert longestSubsequence([1, 2]) == 2


def test_all_same_elements():
    # All elements are the same, so differences are all 0
    # [5, 5, 5] -> differences [0, 0], which is non-increasing (0 >= 0)
    assert longestSubsequence([5, 5, 5]) == 3


def test_strictly_increasing_sequence():
    # [1, 2, 3, 4, 5] 
    # Full sequence has differences [1, 1, 1, 1] which is non-increasing (all equal)
    assert longestSubsequence([1, 2, 3, 4, 5]) == 5


def test_strictly_decreasing_sequence():
    # [5, 4, 3, 2, 1]
    # Full sequence has differences [1, 1, 1, 1] which is non-increasing (all equal)
    assert longestSubsequence([5, 4, 3, 2, 1]) == 5


def test_large_then_small_differences():
    # [1, 10, 11] -> differences [9, 1]
    # 9 >= 1, so full sequence is valid, length 3
    assert longestSubsequence([1, 10, 11]) == 3


def test_small_then_large_differences():
    # [1, 2, 10] -> differences [1, 8]
    # 1 >= 8 is false, so we can't take the full sequence
    # Best subsequences: [1, 2] with diff [1], or [1, 10] with diff [9], or [2, 10] with diff [8]
    # All have length 2
    assert longestSubsequence([1, 2, 10]) == 2


def test_alternating_pattern():
    # [1, 100, 2, 99, 3]
    # Subsequence [1, 100, 2, 99, 3] has differences [99, 98, 97, 96]
    # 99 >= 98 >= 97 >= 96, so this is valid, length 5
    assert longestSubsequence([1, 100, 2, 99, 3]) == 5


def test_equal_differences_throughout():
    # [1, 3, 5, 7, 9] -> differences [2, 2, 2, 2]
    # All differences equal, so non-increasing, length 5
    assert longestSubsequence([1, 3, 5, 7, 9]) == 5


def test_boundary_values():
    # spec: 1 <= nums[i] <= 300
    # Test with boundary values
    assert longestSubsequence([1, 300]) == 2
    assert longestSubsequence([300, 1]) == 2
