from typing import List
import pytest
from solution import countPartitions


def test_canonical_example_1():
    # spec: nums = [10,10,3,7,6] -> 4
    # Partitions:
    # [10], [10, 3, 7, 6]: sum_diff = 10 - 26 = -16 (even)
    # [10, 10], [3, 7, 6]: sum_diff = 20 - 16 = 4 (even)
    # [10, 10, 3], [7, 6]: sum_diff = 23 - 13 = 10 (even)
    # [10, 10, 3, 7], [6]: sum_diff = 30 - 6 = 24 (even)
    assert countPartitions([10, 10, 3, 7, 6]) == 4


def test_canonical_example_2():
    # spec: nums = [1,2,2] -> 0
    # No partition results in an even sum difference
    assert countPartitions([1, 2, 2]) == 0


def test_canonical_example_3():
    # spec: nums = [2,4,6,8] -> 3
    # All partitions result in an even sum difference
    assert countPartitions([2, 4, 6, 8]) == 3


def test_minimum_length_array():
    # spec constraint: 2 <= n, so test n=2
    # [1], [2]: sum_diff = 1 - 2 = -1 (odd)
    assert countPartitions([1, 2]) == 0


def test_minimum_length_array_even_difference():
    # spec constraint: 2 <= n, so test n=2 with even difference
    # [2], [4]: sum_diff = 2 - 4 = -2 (even)
    assert countPartitions([2, 4]) == 1


def test_all_same_elements():
    # All elements are the same
    # [1], [1, 1, 1]: sum_diff = 1 - 3 = -2 (even)
    # [1, 1], [1, 1]: sum_diff = 2 - 2 = 0 (even)
    # [1, 1, 1], [1]: sum_diff = 3 - 1 = 2 (even)
    assert countPartitions([1, 1, 1, 1]) == 3


def test_all_even_numbers():
    # All even numbers should result in all even differences
    # [2], [2, 2]: sum_diff = 2 - 4 = -2 (even)
    # [2, 2], [2]: sum_diff = 4 - 2 = 2 (even)
    assert countPartitions([2, 2, 2]) == 2


def test_all_odd_numbers():
    # All odd numbers
    # [1], [1, 1]: sum_diff = 1 - 2 = -1 (odd)

    assert countPartitions([1, 1, 1]) == 0


def test_mixed_odd_even():
    # Mixed odd and even numbers
    # [1], [2, 3]: sum_diff = 1 - 5 = -4 (even)
    # [1, 2], [3]: sum_diff = 3 - 3 = 0 (even)
    assert countPartitions([1, 2, 3]) == 2

    # [1, 1, 1], [1, 1]: sum_diff = 3 - 2 = 1 (odd)
    # [1, 1, 1, 1], [1]: sum_diff = 4 - 1 = 3 (odd)
    assert countPartitions([1, 1, 1, 1, 1]) == 0
