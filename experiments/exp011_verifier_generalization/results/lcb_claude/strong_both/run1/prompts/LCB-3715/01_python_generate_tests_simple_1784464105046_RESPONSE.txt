from typing import List
import pytest
from solution import maximumCoins


def test_canonical_example_1():
    # spec: coins = [[8,10,1],[1,3,2],[5,6,4]], k = 4
    # Selecting bags at positions [3, 4, 5, 6] gives: 2 + 0 + 4 + 4 = 10
    assert maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) == 10


def test_canonical_example_2():
    # spec: coins = [[1,10,3]], k = 2
    # Selecting bags at positions [1, 2] gives: 3 + 3 = 6
    assert maximumCoins([[1, 10, 3]], 2) == 6


def test_single_segment_k_equals_segment_length():
    # segment [1, 5] with 2 coins each, k = 5
    # positions [1, 2, 3, 4, 5] gives 2 + 2 + 2 + 2 + 2 = 10
    assert maximumCoins([[1, 5, 2]], 5) == 10


def test_single_segment_k_larger_than_segment():
    # segment [3, 5] with 3 coins each, k = 10
    # best window includes segment [3, 4, 5] with 3+3+3=9, rest are 0
    # any window of 10 consecutive bags that includes all 3 positions gives 9
    assert maximumCoins([[3, 5, 3]], 10) == 9


def test_multiple_segments_no_overlap():
    # segments [1, 2, 5] and [5, 6, 3], k = 3
    # window [1, 2, 3] gives 5 + 5 + 0 = 10
    # window [4, 5, 6] gives 0 + 3 + 3 = 6
    # window [5, 6, 7] gives 3 + 3 + 0 = 6
    assert maximumCoins([[1, 2, 5], [5, 6, 3]], 3) == 10


def test_k_equals_one():
    # coins = [[2, 4, 7], [8, 10, 3]], k = 1
    # best single bag is any from [2, 4] with 7 coins
    assert maximumCoins([[2, 4, 7], [8, 10, 3]], 1) == 7


def test_gap_between_segments():
    # segments [1, 2, 4] and [10, 12, 5], k = 5
    # window [1, 2, 3, 4, 5] gives 4 + 4 + 0 + 0 + 0 = 8
    # window [8, 9, 10, 11, 12] gives 0 + 0 + 5 + 5 + 5 = 15
    assert maximumCoins([[1, 2, 4], [10, 12, 5]], 5) == 15


def test_window_spans_multiple_segments():
    # segments [1, 1, 10], [3, 3, 20], [5, 5, 30], k = 5
    # window [1, 2, 3, 4, 5] gives 10 + 0 + 20 + 0 + 30 = 60
    assert maximumCoins([[1, 1, 10], [3, 3, 20], [5, 5, 30]], 5) == 60


def test_all_bags_have_same_coins():
    # segment [1, 100, 1], k = 10
    # any window of 10 consecutive bags gives 1 * 10 = 10
    assert maximumCoins([[1, 100, 1]], 10) == 10


def test_k_larger_than_all_segments_combined():
    # segments [1, 2, 5], [10, 11, 3], k = 20
    # total coins from all segments: 5*2 + 3*2 = 10 + 6 = 16
    # best window includes both segments
    assert maximumCoins([[1, 2, 5], [10, 11, 3]], 20) == 16


def test_optimal_window_at_boundary():
    # segments [5, 10, 2], k = 6
    # window [5, 6, 7, 8, 9, 10] gives 2*6 = 12
    # window [10, 11, 12, 13, 14, 15] gives 2 + 0 + 0 + 0 + 0 + 0 = 2
    assert maximumCoins([[5, 10, 2]], 6) == 12
