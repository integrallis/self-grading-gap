```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a dictionary to store coins in bags by their positions
    coin_map = {}
    for l, r, c in coins:
        for position in range(l, r + 1):
            coin_map[position] = coin_map.get(position, 0) + c

    # Create a sorted list of positions and their corresponding coin counts
    positions = sorted(coin_map.keys())
    n = len(positions)
    max_coins = 0

    # Use a sliding window approach to find the maximum coins in any k consecutive bags
    left = 0
    current_coins = 0

    for right in range(n):
        current_coins += coin_map[positions[right]]

        # While the window size exceeds k, move the left pointer
        while positions[right] - positions[left] + 1 > k:
            current_coins -= coin_map[positions[left]]
            left += 1

        # Update the maximum coins found
        max_coins = max(max_coins, current_coins)

    return max_coins

```
