```python
def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]
    # Check for the prefix match
    for i in range(len(s) - len(prefix) - len(suffix) + 1):
        if s[i:i + len(prefix)] == prefix:
            # Check for the suffix match after the prefix
            if s[i + len(prefix):].endswith(suffix):
                return True
    return False

```
