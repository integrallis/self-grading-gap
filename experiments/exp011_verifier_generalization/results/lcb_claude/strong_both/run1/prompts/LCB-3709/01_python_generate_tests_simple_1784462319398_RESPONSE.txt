import pytest
from solution import hasSpecialSubstring


def test_canonical_example_1():
    # spec: s = "aaabaaa", k = 3 -> true
    # substring s[4..6] == "aaa" satisfies all conditions
    assert hasSpecialSubstring("aaabaaa", 3) is True


def test_canonical_example_2():
    # spec: s = "abc", k = 2 -> false
    # no substring of length 2 with one distinct character
    assert hasSpecialSubstring("abc", 2) is False


def test_single_character_string_k_equals_1():
    # s = "a", k = 1
    # substring "a" at position 0: all same char, no before/after chars
    assert hasSpecialSubstring("a", 1) is True


def test_all_same_character_k_equals_length():
    # s = "aaaa", k = 4
    # entire string is all 'a', no before/after chars to check
    assert hasSpecialSubstring("aaaa", 4) is True


def test_all_same_character_k_less_than_length():
    # s = "aaaa", k = 3
    # substring s[0..2] = "aaa": all same, no before, after is 'a' (same) -> fails
    # substring s[1..3] = "aaa": before is 'a' (same) -> fails
    # no valid substring exists
    assert hasSpecialSubstring("aaaa", 3) is False


def test_special_substring_at_start():
    # s = "aaabbb", k = 3
    # substring s[0..2] = "aaa": all same, no before, after is 'b' (different) -> valid
    assert hasSpecialSubstring("aaabbb", 3) is True


def test_special_substring_at_end():
    # s = "bbbccc", k = 3
    # substring s[3..5] = "ccc": all same, before is 'b' (different), no after -> valid
    assert hasSpecialSubstring("bbbccc", 3) is True


def test_special_substring_in_middle():
    # s = "abbbcd", k = 3
    # substring s[1..3] = "bbb": all same, before is 'a' (different), after is 'c' (different) -> valid
    assert hasSpecialSubstring("abbbcd", 3) is True


def test_no_special_substring_because_before_char_same():
    # s = "aabbb", k = 2
    # substring s[0..1] = "aa": all same, no before, after is 'b' (different) -> valid
    assert hasSpecialSubstring("aabbb", 2) is True


def test_no_special_substring_because_after_char_same():
    # s = "aaabba", k = 2
    # substring s[0..1] = "aa": all same, no before, after is 'a' (same) -> fails
    # substring s[1..2] = "aa": before is 'a' (same) -> fails
    # substring s[3..4] = "bb": all same, before is 'a' (different), after is 'a' (different) -> valid
    assert hasSpecialSubstring("aaabba", 2) is True


def test_alternating_characters():
    # s = "ababab", k = 2
    # no substring of length 2 with all same characters
    assert hasSpecialSubstring("ababab", 2) is False


def test_k_equals_1_with_different_neighbors():
    # s = "aba", k = 1
    # substring s[0] = "a": no before, after is 'b' (different) -> valid
    assert hasSpecialSubstring("aba", 1) is True


def test_k_equals_1_with_same_neighbors():
    # s = "aaa", k = 1
    # substring s[0] = "a": no before, after is 'a' (same) -> fails
    # substring s[1] = "a": before is 'a' (same) -> fails
    # substring s[2] = "a": before is 'a' (same), no after -> fails
    assert hasSpecialSubstring("aaa", 1) is False


def test_multiple_runs_none_valid():
    # s = "aabbcc", k = 3
    # no substring of length 3 with all same characters
    assert hasSpecialSubstring("aabbcc", 3) is False


def test_longer_run_with_valid_substring():
    # s = "aaaaaa", k = 3
    # substring s[0..2] = "aaa": all same, no before, after is 'a' (same) -> fails
    # all other substrings of length 3 have same char before or after
    assert hasSpecialSubstring("aaaaaa", 3) is False


def test_exact_length_run_bounded_by_different():
    # s = "baaac", k = 3
    # substring s[1..3] = "aaa": all same, before is 'b' (different), after is 'c' (different) -> valid
    assert hasSpecialSubstring("baaac", 3) is True
