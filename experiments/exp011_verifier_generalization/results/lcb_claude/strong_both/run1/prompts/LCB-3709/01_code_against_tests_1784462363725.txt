## Implementation

```python
def hasSpecialSubstring(s: str, k: int) -> bool:
    n = len(s)
    for i in range(n - k + 1):
        substring = s[i:i + k]
        if len(set(substring)) == 1:  # Check if all characters are the same
            before_char = s[i - 1] if i > 0 else None
            after_char = s[i + k] if i + k < n else None
            if (before_char is None or before_char != substring[0]) and (
                after_char is None or after_char != substring[0]):
                return True
    return False
```
