from typing import List
import itertools

def totalNumbers(digits: List[int]) -> int:
    even_digits = [d for d in digits if d % 2 == 0]
    count = 0
    for last_digit in set(even_digits):
        if digits.count(last_digit) > 0:
            remaining_digits = digits.copy()
            remaining_digits.remove(last_digit)
            for first_digit in set(remaining_digits):
                if first_digit == 0:
                    continue
                remaining_for_middle = remaining_digits.copy()
                remaining_for_middle.remove(first_digit)
                count += len(set(itertools.permutations(remaining_for_middle, 1))) * len(set(itertools.permutations(remaining_for_middle, 1)))
                count += len(set(itertools.permutations(remaining_for_middle, 1)))
    return count // 2  # Each number is counted twice in permutations
