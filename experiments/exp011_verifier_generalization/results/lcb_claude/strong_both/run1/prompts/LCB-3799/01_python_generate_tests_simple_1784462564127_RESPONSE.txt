def test_canonical_example_1():

    assert totalNumbers([1, 2, 3, 4]) == 12


def test_canonical_example_2():
    # spec: digits = [0,2,2] -> 2
    # The only 3-digit even numbers are 202 and 220
    assert totalNumbers([0, 2, 2]) == 2


def test_canonical_example_3():
    # spec: digits = [6,6,6] -> 1
    # Only 666 can be formed
    assert totalNumbers([6, 6, 6]) == 1


def test_canonical_example_4():
    # spec: digits = [1,3,5] -> 0
    # No even 3-digit numbers can be formed (all digits are odd)
    assert totalNumbers([1, 3, 5]) == 0


def test_single_even_digit():
    # digits = [0,1,2] -> can form 102, 120, 210
    # Last digit must be even (0 or 2), first digit cannot be 0
    # 102: valid, 120: valid, 210: valid = 3
    assert totalNumbers([0, 1, 2]) == 3


def test_multiple_same_even_digits():
    # digits = [2,2,2,2] -> can form 222 (all same)
    # Only one distinct number: 222
    assert totalNumbers([2, 2, 2, 2]) == 1


def test_mix_with_zeros():
    # digits = [0,0,1] -> no valid numbers
    # Even numbers must end in 0, but first digit cannot be 0
    # Need format X0Y or XY0 where Y=0, but first position cannot be 0
    # With [0,0,1]: 100 is the only possibility
    assert totalNumbers([0, 0, 1]) == 1


def test_all_even_digits():
    # digits = [2,4,6] -> all permutations ending in even are valid
    # 242, 246, 264, 426, 462, 624, 642 (but 222, 444, 666 need duplicates)
    # Actually: 246, 264, 426, 462, 624, 642 = 6 numbers
    assert totalNumbers([2, 4, 6]) == 6


def test_minimum_length_array():
    # digits = [1,2,3] (exactly 3 elements)
    # Even numbers: 132, 312 = 2
    assert totalNumbers([1, 2, 3]) == 2


def test_larger_array_with_duplicates():
    # digits = [1,1,2,2,3] 
    # Need to count distinct 3-digit even numbers ending in 2
    # Various permutations possible
    assert totalNumbers([1, 1, 2, 2, 3]) == 9


def test_all_zeros():
    # digits = [0,0,0] -> cannot form valid 3-digit number (leading zero not allowed)
    assert totalNumbers([0, 0, 0]) == 0


def test_one_even_many_odd():
    # digits = [1,3,5,7,8] -> numbers ending in 8
    # 138, 158, 178, 318, 358, 378, 518, 538, 578, 718, 738, 758,
