from typing import List

def zigzagTraversal(grid: List[List[int]]) -> List[int]:
    result = []
    for i in range(len(grid)):
        row = grid[i]
        if i % 2 == 0:
            # Even row: left to right
            indices = range(0, len(row))
        else:
            # Odd row: right to left
            indices = range(len(row) - 1, -1, -1)
        for j in indices:
            if j % 2 == 0:
                result.append(row[j])
    return result
```
