```python
from typing import List
import pytest
from solution import minCosts


def test_canonical_example_1():
    # Example 1: cost = [5,3,4,1,3,2] -> [5,3,3,1,1,1]
    # Position 0: swap with person 0 for cost 5
    # Position 1: swap with person 1 for cost 3
    # Position 2: swap with person 1 for cost 3, then swap with person 2 for free
    # Position 3: swap with person 3 for cost 1
    # Position 4: swap with person 3 for cost 1, then swap with person 4 for free
    # Position 5: swap with person 3 for cost 1, then swap with person 5 for free
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]


def test_canonical_example_2():
    # Example 2: cost = [1,2,4,6,7] -> [1,1,1,1,1]
    # Swap with person 0 for cost 1, then reach any position for free
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]


def test_single_person():
    # With one person, we can only reach position 0 by swapping with person 0
    # Minimum cost to reach position 0 is cost[0]
    assert minCosts([5]) == [5]


def test_descending_costs():
    # When costs are descending, each position can be reached by swapping directly
    # or by swapping with earlier person and moving forward for free
    # Position 0: swap with person 0 for cost 10
    # Position 1: swap with person 1 for cost 5
    # Position 2: swap with person 2 for cost 1
    assert minCosts([10, 5, 1]) == [10, 5, 1]


def test_ascending_costs():
    # When costs are ascending, later positions benefit from earlier cheaper swaps
    # Position 0: swap with person 0 for cost 1
    # Position 1: swap with person 0 for cost 1, then move forward for free
    # Position 2: swap with person 0 for cost 1, then move forward for free
    assert minCosts([1, 5, 10]) == [1, 1, 1]


def test_all_same_cost():
    # When all costs are the same, minimum is that cost for position 0,
    # and same cost for all other positions (can't improve)
    # Position 0: cost 3
    # Position 1: cost 3
    # Position 2: cost 3
    assert minCosts([3, 3, 3]) == [3, 3, 3]


def test_minimum_in_middle():
    # Minimum cost in the middle affects later positions
    # Position 0: swap with person 0 for cost 5
    # Position 1: swap with person 1 for cost 2
    # Position 2: swap with person 1 for cost 2, then move forward for free
    # Position 3: swap with person 1 for cost 2, then move forward for free
    assert minCosts([5, 2, 6, 8]) == [5, 2, 2, 2]


def test_two_people():
    # With two people, check both direct and indirect paths
    # Position 0: swap with person 0 for cost 3
    # Position 1: swap with person 1 for cost 2
    assert minCosts([3, 2]) == [3, 2]


def test_two_people_first_cheaper():
    # First person cheaper, second position can be reached via first
    # Position 0: swap with person 0 for cost 1
    # Position 1: swap with person 0 for cost 1, then move forward for free
    assert minCosts([1, 5]) == [1, 1]
```
