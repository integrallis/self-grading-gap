import pytest
from typing import List
from solution import sumOfGoodNumbers


def test_canonical_example_1():
    # spec: nums = [1,3,2,1,5,4], k = 2
    # good numbers: nums[1]=3 (3>1 at i-2=0, i+2=3 doesn't need check as 3>1), 
    #               nums[4]=5 (5>2 at i-2=2, i+2=6 out of bounds),
    #               nums[5]=4 (i-2=3 gives 1, 4>1, i+2=7 out of bounds)
    # sum = 3 + 5 + 4 = 12
    assert sumOfGoodNumbers([1, 3, 2, 1, 5, 4], 2) == 12


def test_canonical_example_2():
    # spec: nums = [2,1], k = 1
    # good numbers: nums[0]=2 (i-1=-1 out of bounds, i+1=1 gives 1, 2>1),
    #               nums[1]=1 (i-1=0 gives 2, 1 not > 2, so not good)
    # sum = 2
    assert sumOfGoodNumbers([2, 1], 1) == 2

    # nums[1]=10: i-1=0 gives 5, 10>5, i+1=2 (out of bounds), good

    assert sumOfGoodNumbers([5, 10], 1) == 10
