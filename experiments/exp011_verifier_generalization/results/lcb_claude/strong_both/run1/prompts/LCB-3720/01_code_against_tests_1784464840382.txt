```python
from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Create a graph adjacency list
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[u].append((v, w))

    # Binary search over the possible weights
    left, right = 1, 10**6  # Weights range from 1 to 10^6

    def can_form_graph(max_weight: int) -> bool:
        # Create a new graph with edges <= max_weight
        filtered_graph = [[] for _ in range(n)]
        for u in range(n):
            for v, w in graph[u]:
                if w <= max_weight:
                    filtered_graph[u].append(v)
        # Check reachability to node 0 from all nodes
        reachable = [0] * n
        stack = [0]
        reachable[0] = 1
        while stack:
            node = stack.pop()
            for neighbor in filtered_graph[node]:
                if not reachable[neighbor]:
                    reachable[neighbor] = 1
                    stack.append(neighbor)
        # Check outgoing edges constraint
        out_degree = [0] * n
        for u in range(n):
            for v in filtered_graph[u]:
                out_degree[u] += 1
                if out_degree[u] > threshold:
                    return False
        # Check if all nodes can reach node 0
        return all(reachable)

    answer = -1
    while left <= right:
        mid = (left + right) // 2
        if can_form_graph(mid):
            answer = mid
            right = mid - 1
        else:
            left = mid + 1

    return answer

```
