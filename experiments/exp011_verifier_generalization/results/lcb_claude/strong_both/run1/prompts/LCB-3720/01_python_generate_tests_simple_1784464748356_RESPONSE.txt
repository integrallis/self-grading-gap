from typing import List
import pytest
from solution import minMaxWeight


def test_canonical_example_1():
    # spec: n=5, edges=[[1,0,1],[2,0,2],[3,0,1],[4,3,1],[2,1,1]], threshold=2 -> 1
    # Remove edge 2->0. Max weight among remaining edges is 1.
    assert minMaxWeight(5, [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]], 2) == 1


def test_canonical_example_2():
    # spec: n=5, edges=[[0,1,1],[0,2,2],[0,3,1],[0,4,1],[1,2,1],[1,4,1]], threshold=1 -> -1
    # Impossible to reach node 0 from node 2.
    assert minMaxWeight(5, [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]], 1) == -1


def test_canonical_example_3():
    # spec: n=5, edges=[[1,2,1],[1,3,3],[1,4,5],[2,3,2],[3,4,2],[4,0,1]], threshold=1 -> 2
    # Remove edges 1->3 and 1->4. Max weight among remaining edges is 2.
    assert minMaxWeight(5, [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]], 1) == 2


def test_canonical_example_4():
    # spec: n=5, edges=[[1,2,1],[1,3,3],[1,4,5],[2,3,2],[4,0,1]], threshold=1 -> -1
    # Impossible to satisfy all conditions.
    assert minMaxWeight(5, [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]], 1) == -1


def test_two_nodes_single_edge():
    # n=2, one edge 1->0 with weight 5, threshold=1
    # Node 1 can reach node 0 directly. Max weight is 5.
    assert minMaxWeight(2, [[1, 0, 5]], 1) == 5


def test_simple_chain():
    # n=3, edges form chain 2->1->0, threshold=1
    # All nodes can reach 0. Max weight is max(w1, w2).
    assert minMaxWeight(3, [[2, 1, 3], [1, 0, 2]], 1) == 3


def test_threshold_forces_removal():
    # n=3, node 1 has two outgoing edges but threshold=1
    # Must remove one edge. Node 2 must still reach 0.
    # edges: 1->0 (w=10), 1->2 (w=5), 2->0 (w=3)
    # Keep 1->2 and 2->0 to satisfy threshold and reachability. Max weight = 5.
    assert minMaxWeight(3, [[1, 0, 10], [1, 2, 5], [2, 0, 3]], 1) == 5


def test_all_nodes_direct_to_zero():
    # n=4, all nodes have direct edge to 0
    # threshold allows all edges. Max weight is the max among all edge weights.
    assert minMaxWeight(4, [[1, 0, 2], [2, 0, 3], [3, 0, 1]], 3) == 3


def test_multiple_edges_same_pair_different_weights():
    # n=3, two edges from 1->0 with different weights (1 and 10)
    # Can choose the lighter edge. Node 2 also connects to 0.
    # Max weight minimized by choosing 1->0 with weight 1.
    assert minMaxWeight(3, [[1, 0, 1], [1, 0, 10], [2, 0, 2]], 2) == 2


def test_unreachable_node_zero():
    # n=3, node 1 has no path to 0
    # Only edges: 0->1, 2->0
    # Node 1 cannot reach node 0, return -1.
    assert minMaxWeight(3, [[0, 1, 1], [2, 0, 1]], 1) == -1


def test_threshold_zero_impossible():
    # n=3, threshold=1, but node needs edges to reach 0
    # If a non-zero node has no edges, it cannot reach 0.
    # edges: 2->0 only, node 1 has no outgoing edges
    assert minMaxWeight(3, [[2, 0, 5]], 1) == -1


def test_larger_threshold_allows_more_edges():
    # n=4, node 1 has 3 outgoing edges, threshold=3
    # All edges can be kept if they maintain reachability.
    assert minMaxWeight(4, [[1, 0, 1], [1, 2, 2], [1, 3, 3], [2, 0, 4], [3, 0, 5]], 3) == 5
