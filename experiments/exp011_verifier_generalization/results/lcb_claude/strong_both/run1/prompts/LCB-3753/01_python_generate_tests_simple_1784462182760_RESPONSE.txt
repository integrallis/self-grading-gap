from solution import maxDifference


def test_canonical_example_1():
    # spec: s = "aaaaabbc"
    # 'a' has frequency 5 (odd), 'b' has frequency 2 (even), 'c' has frequency 1 (odd)
    # max difference = 5 - 2 = 3
    assert maxDifference("aaaaabbc") == 3


def test_canonical_example_2():
    # spec: s = "abcabcab"
    # 'a' has frequency 3 (odd), 'b' has frequency 3 (odd), 'c' has frequency 2 (even)
    # max difference = 3 - 2 = 1
    assert maxDifference("abcabcab") == 1


def test_single_odd_single_even():
    # s = "aab"
    # 'a' has frequency 2 (even), 'b' has frequency 1 (odd)
    # max difference = 1 - 2 = -1
    assert maxDifference("aab") == -1


def test_multiple_odd_multiple_even():
    # s = "aaabbbccdd"
    # 'a' has frequency 3 (odd), 'b' has frequency 3 (odd), 'c' has frequency 2 (even), 'd' has frequency 2 (even)
    # max odd = 3, min even = 2
    # max difference = 3 - 2 = 1
    assert maxDifference("aaabbbccdd") == 1


def test_large_odd_small_even():
    # s = "aaaaaaaabb"
    # 'a' has frequency 8 (even), 'b' has frequency 2 (even)
    # Wait, we need at least one odd. Let's fix: s = "aaaaaaabbb"
    # 'a' has frequency 7 (odd), 'b' has frequency 3 (odd)
    # We need at least one even. Let's use: s = "aaaaaabbbb"
    # 'a' has frequency 6 (even), 'b' has frequency 4 (even)
    # Still no odd. Let's use: s = "aaaaaabbbbc"
    # 'a' has frequency 6 (even), 'b' has frequency 4 (even), 'c' has frequency 1 (odd)
    # max difference = 1 - 4 = -3 (not max, would be 1 - 6 = -5)
    # Actually max = max(odd) - min(even) = 1 - 4 = -3
    assert maxDifference("aaaaaabbbbc") == -3


def test_high_frequency_odd_low_frequency_even():
    # s = "aaaaaaaaaaabb"
    # 'a' has frequency 11 (odd), 'b' has frequency 2 (even)
    # max difference = 11 - 2 = 9
    assert maxDifference("aaaaaaaaaaabb") == 9


def test_low_frequency_odd_high_frequency_even():
    # s = "aaaaaaaaaaabbbbbbbbbbbbc"
    # 'a' has frequency 11 (odd), 'b' has frequency 12 (even), 'c' has frequency 1 (odd)
    # max odd = 11, min even = 12
    # max difference = 11 - 12 = -1
    assert maxDifference("aaaaaaaaaaabbbbbbbbbbbbc") == -1


def test_all_different_characters():
    # s = "abcde"
    # all have frequency 1 (odd)
    # We need at least one even. Let's use: s = "aabcd"
    # 'a' has frequency 2 (even), 'b', 'c', 'd' each have frequency 1 (odd)
    # max odd = 1, min even = 2
    # max difference = 1 - 2 = -1
    assert maxDifference("aabcd") == -1


def test_negative_difference():
    # s = "aabbbbbbbbbc"
    # 'a' has frequency 2 (even), 'b' has frequency 9 (odd), 'c' has frequency 1 (odd)
    # max odd = 9, min even = 2
    # max difference = 9 - 2 = 7
    assert maxDifference("aabbbbbbbbbc") == 7


def test_minimum_string_length():
    # s = "aab" (length 3, minimum per constraints)
    # 'a' has frequency 2 (even), 'b' has frequency 1 (odd)
    # max difference = 1 - 2 = -1
    assert maxDifference("aab") == -1
