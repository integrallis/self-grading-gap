from solution import hasSameDigits


def test_canonical_example_1():
    # spec: "3902" -> true
    # "3902" -> "292" -> "11" -> true (both digits are 1)
    assert hasSameDigits("3902") is True


def test_canonical_example_2():
    # spec: "34789" -> false
    # "34789" -> "7157" -> "862" -> "48" -> false (4 != 8)
    assert hasSameDigits("34789") is False


def test_string_with_three_digits():
    # "123" -> "35" (1+2=3, 2+3=5) -> false (3 != 5)
    assert hasSameDigits("123") is False


def test_string_with_three_digits_same_result():
    # "505" -> "55" (5+0=5, 0+5=5) -> true (5 == 5)
    assert hasSameDigits("505") is True


def test_string_with_four_digits():
    # "1111" -> "222" (1+1=2, 1+1=2, 1+1=2) -> "44" (2+2=4, 2+2=4) -> true
    assert hasSameDigits("1111") is True


def test_string_with_larger_sums():
    # "9999" -> "888" (9+9=18%10=8, 8, 8) -> "66" (8+8=16%10=6, 8+8=6) -> true
    assert hasSameDigits("9999") is True


def test_string_with_zeros():
    # "000" -> "00" (0+0=0, 0+0=0) -> true
    assert hasSameDigits("000") is True


def test_string_requiring_multiple_operations():
    # "12345" -> "3579" (1+2=3, 2+3=5, 3+4=7, 4+5=9)
    # -> "826" (3+5=8, 5+7=12%10=2, 7+9=16%10=6)
    # -> "08" (8+2=10%10=0, 2+6=8) -> false
    assert hasSameDigits("12345") is False


def test_alternating_high_low():
    # "9090" -> "909" (9+0=9, 0+9=9, 9+0=9) -> "88" (9+9=18%10=8, 9+9=8) -> true
    assert hasSameDigits("9090") is True
