def test_canonical_example_1():
    # spec: nums = [10,10,3,7,6] -> 4

    # [10,10], [3,7,6]: diff = 20 - 16 = 4 (even)
    # [10,10,3], [7,6]: diff = 23 - 13 = 10 (even)
    # [10,10,3,7], [6]: diff = 30 - 6 = 24 (even)
    assert countPartitions([10, 10, 3, 7, 6]) == 4
