import pytest
from solution import maxDistance


def test_canonical_example_1():
    # spec: s = "NWSE", k = 1 -> 3
    # Change s[2] from 'S' to 'N' to get "NWNE"
    # Positions: (0,1), (-1,1), (-1,2), (0,2)
    # Distances: 1, 2, 3, 2
    # Maximum: 3
    assert maxDistance("NWSE", 1) == 3


def test_canonical_example_2():
    # spec: s = "NSWWEW", k = 3 -> 6
    # Change s[1] from 'S' to 'N' and s[4] from 'E' to 'W' to get "NNWWWW"
    # Maximum Manhattan distance achievable is 6
    assert maxDistance("NSWWEW", 3) == 6


def test_single_move_north_no_changes():
    # s = "N", k = 0
    # Position after N: (0, 1), distance = 1
    assert maxDistance("N", 0) == 1


def test_single_move_south_no_changes():
    # s = "S", k = 0
    # Position after S: (0, -1), distance = 1
    assert maxDistance("S", 0) == 1


def test_single_move_east_no_changes():
    # s = "E", k = 0
    # Position after E: (1, 0), distance = 1
    assert maxDistance("E", 0) == 1


def test_single_move_west_no_changes():
    # s = "W", k = 0
    # Position after W: (-1, 0), distance = 1
    assert maxDistance("W", 0) == 1


def test_opposing_moves_no_changes():
    # s = "NS", k = 0
    # Position after N: (0, 1), distance = 1
    # Position after S: (0, 0), distance = 0
    # Maximum: 1
    assert maxDistance("NS", 0) == 1


def test_k_equals_string_length():
    # s = "SSSS", k = 4
    # Can change all moves to go in same direction, e.g., all N
    # Positions: (0,1), (0,2), (0,3), (0,4)
    # Maximum: 4
    assert maxDistance("SSSS", 4) == 4


def test_already_optimal_no_changes_needed():
    # s = "NNNN", k = 0
    # All moves going north
    # Positions: (0,1), (0,2), (0,3), (0,4)
    # Maximum: 4
    assert maxDistance("NNNN", 0) == 4


def test_two_moves_with_one_change():
    # s = "NS", k = 1
    # Change S to N to get "NN"
    # Positions: (0,1), (0,2)
    # Maximum: 2
    assert maxDistance("NS", 1) == 2


def test_diagonal_moves_no_changes():
    # s = "NE", k = 0
    # Position after N: (0, 1), distance = 1
    # Position after E: (1, 1), distance = 2
    # Maximum: 2
    assert maxDistance("NE", 0) == 2


def test_return_to_origin_with_changes():
    # s = "NSEW", k = 2
    # Can change S to N and W to E to get "NNEE"
    # Positions: (0,1), (0,2), (1,2), (2,2)
    # Distances: 1, 2, 3, 4
    # Maximum: 4
    assert maxDistance("NSEW", 2) == 4


def test_long_sequence_maximizing_distance():
    # s = "EEEE", k = 0
    # Positions: (1,0), (2,0), (3,0), (4,0)
    # Distances: 1, 2, 3, 4
    # Maximum: 4
    assert maxDistance("EEEE", 0) == 4


def test_k_zero_with_mixed_directions():
    # s = "NESW", k = 0
    # Positions: (0,1), (1,1), (1,0), (0,0)
    # Distances: 1, 2, 1, 0
    # Maximum: 2
    assert maxDistance("NESW", 0) == 2


def test_three_moves_maximize_y_axis():
    # s = "NSN", k = 1
    # Change S to N to get "NNN"
    # Positions: (0,1), (0,2), (0,3)
    # Maximum: 3
    assert maxDistance("NSN", 1) == 3


def test_three_moves_maximize_x_axis():
    # s = "EWE", k = 1
    # Change W to E to get "EEE"
    # Positions: (1,0), (2,0), (3,0)
    # Maximum: 3
    assert maxDistance("EWE", 1) == 3
