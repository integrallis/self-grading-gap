```python
def maxDistance(s: str, k: int) -> int:
    # Initialize position and maximum distance
    x, y = 0, 0
    max_distance = 0

    # Process each move
    for move in s:
        if move == 'N':
            y += 1
        elif move == 'S':
            y -= 1
        elif move == 'E':
            x += 1
        elif move == 'W':
            x -= 1

        # Calculate current Manhattan distance
        current_distance = abs(x) + abs(y)
        # Update maximum distance
        max_distance = max(max_distance, current_distance)

    # Calculate maximum distance with changes allowed
    # Required changes to optimize distance
    total_distance = abs(x) + abs(y)  
    # The maximum distance achievable by using k changes
    max_possible_distance = total_distance + k

    return max(max_distance, max_possible_distance)
```
