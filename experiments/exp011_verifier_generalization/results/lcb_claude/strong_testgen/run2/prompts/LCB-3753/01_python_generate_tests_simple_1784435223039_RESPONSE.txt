from solution import maxDifference


def test_canonical_example_1():
    # spec: s = "aaaaabbc"
    # 'a' has frequency 5 (odd), 'b' has frequency 2 (even), 'c' has frequency 1 (odd)
    # max difference = 5 - 2 = 3
    assert maxDifference("aaaaabbc") == 3


def test_canonical_example_2():
    # spec: s = "abcabcab"
    # 'a' has frequency 3 (odd), 'b' has frequency 3 (odd), 'c' has frequency 2 (even)
    # max difference = 3 - 2 = 1
    assert maxDifference("abcabcab") == 1


def test_single_odd_single_even():
    # s = "aabbb"
    # 'a' has frequency 2 (even), 'b' has frequency 3 (odd)
    # max difference = 3 - 2 = 1
    assert maxDifference("aabbb") == 1


def test_multiple_odd_and_even_frequencies():
    # s = "aaabbbcccc"
    # 'a' has frequency 3 (odd), 'b' has frequency 3 (odd), 'c' has frequency 4 (even)
    # max difference = 3 - 4 = -1 (not valid, we want odd - even maximized)
    # Actually: max odd is 3, min even is 4, so 3 - 4 = -1
    # But we want maximum difference, so we try all pairs
    # Pairs: (3, 4) -> 3 - 4 = -1
    # max difference = -1
    assert maxDifference("aaabbbcccc") == -1


def test_high_odd_low_even():
    # s = "aaaaaab"
    # 'a' has frequency 6 (even), 'b' has frequency 1 (odd)
    # max difference = 1 - 6 = -5

def test_maximum_odd_minimum_even():

    # Let's use "aaaaaaaaabc"
    # 'a' has frequency 9 (odd), 'b' has frequency 1 (odd), 'c' has frequency 1 (odd)
    # Still no even. Let's use "aaaaaaaabb"
    # 'a' has frequency 8 (even), 'b' has frequency 2 (even)
    # No odd frequency here. Let's use "aaaaaaaaabbc"
    # 'a' has frequency 9 (odd), 'b' has frequency 2 (even), 'c' has frequency 1 (odd)
    # max odd = 9, min even = 2
    # max difference = 9 - 2 = 7
    assert maxDifference("aaaaaaaaabbc") == 7
