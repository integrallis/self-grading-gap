import pytest
from typing import List
from solution import maxSum


def test_canonical_example_1():
    # spec: nums = [1,2,3,4,5] -> 15
    # Select the entire array without deleting any element
    assert maxSum([1, 2, 3, 4, 5]) == 15


def test_canonical_example_2():
    # spec: nums = [1,1,0,1,1] -> 1
    # Delete duplicates and select subarray with single element [1]
    assert maxSum([1, 1, 0, 1, 1]) == 1


def test_canonical_example_3():
    # spec: nums = [1,2,-1,-2,1,0,-1] -> 3
    # Delete negative elements and select subarray [2, 1] -> 2 + 1 = 3
    assert maxSum([1, 2, -1, -2, 1, 0, -1]) == 3


def test_single_element_positive():
    # Single element array with positive value
    # Can't delete all elements, so must return that element
    assert maxSum([5]) == 5


def test_single_element_negative():
    # Single element array with negative value
    # Must return that element as array can't be empty
    assert maxSum([-5]) == -5


def test_all_negative_elements():
    # When all elements are negative, pick the largest (least negative)
    # Best subarray would be single element with max value
    assert maxSum([-10, -5, -3, -8]) == -3


def test_all_same_positive():
    # All elements are the same positive number
    # Best subarray is single element (unique constraint)
    assert maxSum([7, 7, 7, 7]) == 7


def test_mixed_with_zero():
    # Array with positive, negative, and zero
    # Best might be just positive elements
    assert maxSum([3, 0, -2, 3, 5]) == 8  # [3, 5] after deleting duplicates


def test_two_elements_unique():
    # Two unique elements, both positive
    assert maxSum([10, 20]) == 30


def test_alternating_duplicates():
    # Pattern with alternating values
    # Need to handle duplicates spread throughout
    assert maxSum([5, 3, 5, 3, 5]) == 8  # [5, 3] = 8


def test_ascending_with_duplicates():
    # Ascending values with one duplicate
    assert maxSum([1, 2, 2, 3, 4]) == 10  # [1, 2, 3, 4] after removing one 2


def test_large_positive_small_negative():
    # Large positive values with small negatives in between
    assert maxSum([100, -1, 50]) == 150  # [100, 50] after deleting -1


def test_negative_then_positive():
    # Negative values followed by positive
    assert maxSum([-5, -10, 10, 20]) == 30  # [10, 20]


def test_positive_then_negative():
    # Positive values followed by negative
    assert maxSum([10, 20, -5, -10]) == 30  # [10, 20]


def test_single_zero():
    # Single element that is zero
    assert maxSum([0]) == 0


def test_multiple_zeros():
    # Multiple zeros in array
    assert maxSum([0, 0, 0]) == 0  # Single 0


def test_boundary_values():
    # Test with boundary values from constraints
    assert maxSum([100, -100, 100]) == 100  # Single 100 (unique constraint)
