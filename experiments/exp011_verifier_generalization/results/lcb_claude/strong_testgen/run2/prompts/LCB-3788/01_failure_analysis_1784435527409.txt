from typing import List

def maxSum(nums: List[int]) -> int:
    unique_nums = set()
    max_sum = float('-inf')  # Initialize to negative infinity to handle negative numbers
    current_sum = 0
    left = 0

    for right in range(len(nums)):
        while nums[right] in unique_nums:
            unique_nums.remove(nums[left])
            left += 1
        unique_nums.add(nums[right])
        # Calculate the sum of unique elements only
        current_sum = sum(unique_nums)
        max_sum = max(max_sum, current_sum)
    return max_sum
