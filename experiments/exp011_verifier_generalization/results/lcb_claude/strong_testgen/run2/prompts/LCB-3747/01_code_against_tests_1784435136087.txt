## Implementation

```python
from typing import List

def maxAdjacentDistance(nums: List[int]) -> int:
    n = len(nums)
    max_distance = 0
    for i in range(n):
        # Calculate the absolute difference between adjacent elements, considering circular nature
        distance = abs(nums[i] - nums[(i + 1) % n])
        if distance > max_distance:
            max_distance = distance
    return max_distance

```
