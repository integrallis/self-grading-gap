import pytest
from typing import List
from solution import maxAdjacentDistance


def test_canonical_example_1():
    # spec: nums = [1,2,4] -> 3
    # Adjacent pairs: (1,2), (2,4), (4,1) with differences 1, 2, 3
    # Maximum is 3 from |4 - 1| = 3
    assert maxAdjacentDistance([1, 2, 4]) == 3


def test_canonical_example_2():
    # spec: nums = [-5,-10,-5] -> 5
    # Adjacent pairs: (-5,-10), (-10,-5), (-5,-5) with differences 5, 5, 0
    # Maximum is 5 from |-5 - (-10)| = 5
    assert maxAdjacentDistance([-5, -10, -5]) == 5


def test_two_elements():
    # With 2 elements, only one pair is adjacent (circular)
    # nums = [1, 5]: difference is |1 - 5| = 4
    assert maxAdjacentDistance([1, 5]) == 4


def test_all_same_elements():
    # All elements same, all differences are 0
    # nums = [7, 7, 7, 7]: all differences are |7 - 7| = 0
    assert maxAdjacentDistance([7, 7, 7, 7]) == 0


def test_negative_numbers():
    # Test with all negative numbers
    # nums = [-1, -3, -2]: pairs (-1,-3), (-3,-2), (-2,-1) with differences 2, 1, 1
    # Maximum is 2 from |-1 - (-3)| = 2
    assert maxAdjacentDistance([-1, -3, -2]) == 2


def test_mixed_positive_negative():
    # Test with mix of positive and negative
    # nums = [-10, 20, -5]: pairs (-10,20), (20,-5), (-5,-10) with differences 30, 25, 5
    # Maximum is 30 from |-10 - 20| = 30
    assert maxAdjacentDistance([-10, 20, -5]) == 30


def test_increasing_sequence():
    # Monotonically increasing
    # nums = [1, 2, 3, 4]: pairs (1,2), (2,3), (3,4), (4,1) with differences 1, 1, 1, 3
    # Maximum is 3 from |4 - 1| = 3 (circular wrap)
    assert maxAdjacentDistance([1, 2, 3, 4]) == 3


def test_decreasing_sequence():
    # Monotonically decreasing
    # nums = [10, 8, 6, 4]: pairs (10,8), (8,6), (6,4), (4,10) with differences 2, 2, 2, 6
    # Maximum is 6 from |4 - 10| = 6 (circular wrap)
    assert maxAdjacentDistance([10, 8, 6, 4]) == 6


def test_zero_values():
    # Test with zeros
    # nums = [0, 5, 0]: pairs (0,5), (5,0), (0,0) with differences 5, 5, 0
    # Maximum is 5
    assert maxAdjacentDistance([0, 5, 0]) == 5


def test_large_differences():
    # Test with boundary values
    # nums = [-100, 100]: difference is |-100 - 100| = 200
    assert maxAdjacentDistance([-100, 100]) == 200
