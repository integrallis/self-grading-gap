import pytest
from solution import maxContainers


def test_canonical_example_1():
    # spec: n=2, w=3, maxWeight=15
    # deck has n*n = 2*2 = 4 cells
    # total weight of all containers = 4 * 3 = 12
    # 12 <= 15, so all 4 containers can be loaded
    assert maxContainers(2, 3, 15) == 4


def test_canonical_example_2():
    # spec: n=3, w=5, maxWeight=20
    # deck has n*n = 3*3 = 9 cells
    # maximum containers without exceeding maxWeight:
    # 4 containers weigh 4*5 = 20 <= 20
    # 5 containers weigh 5*5 = 25 > 20
    # so maximum is 4
    assert maxContainers(3, 5, 20) == 4


def test_all_containers_fit():
    # n=5, w=2, maxWeight=100
    # deck has 5*5 = 25 cells
    # total weight = 25 * 2 = 50 <= 100
    # all 25 containers fit
    assert maxContainers(5, 2, 100) == 25


def test_single_cell_deck_fits():
    # n=1, w=10, maxWeight=10
    # deck has 1*1 = 1 cell
    # weight = 1 * 10 = 10 <= 10
    # 1 container fits
    assert maxContainers(1, 10, 10) == 1


def test_single_cell_deck_does_not_fit():
    # n=1, w=10, maxWeight=9
    # deck has 1*1 = 1 cell
    # weight = 1 * 10 = 10 > 9
    # 0 containers fit
    assert maxContainers(1, 10, 9) == 0


def test_partial_loading():
    # n=4, w=7, maxWeight=50
    # deck has 4*4 = 16 cells
    # 7 containers weigh 7*7 = 49 <= 50
    # 8 containers weigh 8*7 = 56 > 50
    # maximum is 7
    assert maxContainers(4, 7, 50) == 7


def test_very_heavy_containers():
    # n=10, w=1000, maxWeight=5000
    # deck has 10*10 = 100 cells
    # 5 containers weigh 5*1000 = 5000 <= 5000
    # 6 containers weigh 6*1000 = 6000 > 5000
    # maximum is 5
    assert maxContainers(10, 1000, 5000) == 5


def test_exact_weight_match():
    # n=3, w=4, maxWeight=36
    # deck has 3*3 = 9 cells
    # 9 containers weigh 9*4 = 36 <= 36
    # all 9 containers fit exactly
    assert maxContainers(3, 4, 36) == 9


def test_large_deck_small_capacity():
    # n=100, w=1, maxWeight=50
    # deck has 100*100 = 10000 cells
    # 50 containers weigh 50*1 = 50 <= 50
    # 51 containers weigh 51*1 = 51 > 50
    # maximum is 50
    assert maxContainers(100, 1, 50) == 50


def test_minimum_inputs():
    # n=1, w=1, maxWeight=1
    # deck has 1*1 = 1 cell
    # 1 container weighs 1*1 = 1 <= 1
    # 1 container fits
    assert maxContainers(1, 1, 1) == 1
