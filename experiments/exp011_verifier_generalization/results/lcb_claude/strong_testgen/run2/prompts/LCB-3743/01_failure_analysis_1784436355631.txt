from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    meetings = list(zip(startTime, endTime))
    gaps = []
    
    # Calculate gaps before, between, and after meetings
    if meetings[0][0] > 0:
        gaps.append(meetings[0][0])  # gap before first meeting

    for i in range(n - 1):
        gap = meetings[i + 1][0] - meetings[i][1]
        if gap > 0:
            gaps.append(gap)  # gap between meetings

    if meetings[-1][1] < eventTime:
        gaps.append(eventTime - meetings[-1][1])  # gap after last meeting

    # Compute max free time based on existing gaps and possible rescheduling
    max_free_time = max(gaps) if gaps else 0 

    # Try to reschedule meetings within bounds of up to k reschedules
    for i in range(n):
        if i > 0 and k > 0:  # Reschedule before the i-th meeting
            new_start = max(0, meetings[i-1][1])
            if new_start + (meetings[i][1] - meetings[i][0]) <= eventTime:
                potential_free_time = new_start - (meetings[i-1][1] if i > 1 else 0)
                max_free_time = max(max_free_time, potential_free_time)

        if i < n - 1 and k > 0:  # Reschedule after the i-th meeting
            new_end = min(eventTime, meetings[i+1][0])
            if (new_end - (meetings[i][1] - meetings[i][0])) >= 0:
                potential_free_time = (new_end - (meetings[i][1] if i > 0 else 0))
                max_free_time = max(max_free_time, potential_free_time)

    return max(max_free_time, 0)
