from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    meetings = list(zip(startTime, endTime))
    gaps = []

    # Calculate gaps before, between, and after meetings
    if meetings[0][0] > 0:
        gaps.append(meetings[0][0])  # gap before first meeting

    for i in range(n - 1):
        gap = meetings[i + 1][0] - meetings[i][1]
        if gap > 0:
            gaps.append(gap)  # gap between meetings

    if meetings[-1][1] < eventTime:
        gaps.append(eventTime - meetings[-1][1])  # gap after last meeting

    # Calculate potential maximum free time with rescheduling
    max_free_time = 0
    for i in range(n):
        if i > 0:
            max_free_time = max(max_free_time, meetings[i][0] - meetings[i - 1][1])
        if k > 0 and i < n - 1:
            # Check potential free time if we reschedule the current meeting
            potential_free_time = (meetings[i + 1][0] - meetings[i - 1][1]) if i > 0 else (meetings[i + 1][0])
            max_free_time = max(max_free_time, potential_free_time)

    # Consider the case where we can reschedule all meetings
    if k >= n:
        max_free_time = max(max_free_time, eventTime)

    return max(max_free_time, 0)
```
