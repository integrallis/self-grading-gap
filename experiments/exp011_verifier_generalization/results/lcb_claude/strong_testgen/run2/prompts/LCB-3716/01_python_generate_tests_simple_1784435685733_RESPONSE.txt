from typing import List

def test_canonical_example_1():
    # spec: nums = [16,6,3] -> 3
    # The longest subsequence is [16, 6, 3] with absolute adjacent differences [10, 3]
    # 10 >= 3, so this is valid and has length 3
    assert longestSubsequence([16, 6, 3]) == 3


def test_canonical_example_2():
    # spec: nums = [6,5,3,4,2,1] -> 4
    # The longest subsequence is [6, 4, 2, 1] with absolute adjacent differences [2, 2, 1]
    # 2 >= 2 >= 1, so this is valid and has length 4
    assert longestSubsequence([6, 5, 3, 4, 2, 1]) == 4


def test_canonical_example_3():
    # spec: nums = [10,20,10,19,10,20] -> 5
    # The longest subsequence is [10, 20, 10, 19, 10] with absolute adjacent differences [10, 10, 9, 9]
    # 10 >= 10 >= 9 >= 9, so this is valid and has length 5
    assert longestSubsequence([10, 20, 10, 19, 10, 20]) == 5


def test_minimum_length_array():
    # spec: 2 <= nums.length, so minimum is 2 elements
    # Any two elements form a valid subsequence (single difference, trivially non-increasing)
    assert longestSubsequence([1, 2]) == 2


def test_all_same_elements():
    # All differences are 0, which forms a non-increasing sequence (0 >= 0 >= ...)
    assert longestSubsequence([5, 5, 5, 5]) == 4


def test_strictly_increasing_sequence():
    # For [1, 2, 3, 4], differences are [1, 1, 1] which is non-increasing (1 >= 1 >= 1)
    # So the whole sequence is valid
    assert longestSubsequence([1, 2, 3, 4]) == 4


def test_strictly_decreasing_sequence():
    # For [4, 3, 2, 1], differences are [1, 1, 1] which is non-increasing (1 >= 1 >= 1)
    # So the whole sequence is valid
    assert longestSubsequence([4, 3, 2, 1]) == 4


def test_increasing_differences():
    # For [1, 2, 4, 7], differences are [1, 2, 3] which is increasing (not non-increasing)
    # We need to find a subsequence with non-increasing differences
    # Subsequence [1, 2] has difference [1], length 2
    # Any valid subsequence would be at least 2
    assert longestSubsequence([1, 2, 4, 7]) >= 2


def test_single_large_then_small_differences():
    # [1, 10, 11] has differences [9, 1] which is non-increasing (9 >= 1)
    assert longestSubsequence([1, 10, 11]) == 3


def test_alternating_pattern():
    # [1, 5, 2, 6, 3] 
    # Subsequence [1, 5, 2] has differences [4, 3] which is non-increasing
    # Subsequence [1, 5, 6] has differences [4, 1] which is non-increasing
    # Need to verify the actual longest
    result = longestSubsequence([1, 5, 2, 6, 3])
