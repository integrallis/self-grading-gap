```python
from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n < 2:
        return n

    # Initialize dp array to store the lengths of longest subsequences
    dp = [1] * n
    max_length = 1

    # Iterate through each pair of elements
    for i in range(n):
        for j in range(i):
            # Compute the absolute difference
            diff1 = abs(nums[i] - nums[j])
            # Check if we can form a valid subsequence
            if j == 0 or diff1 <= abs(nums[j] - nums[j-1]):
                dp[i] = max(dp[i], dp[j] + 1)
                max_length = max(max_length, dp[i])

    return max_length
```
