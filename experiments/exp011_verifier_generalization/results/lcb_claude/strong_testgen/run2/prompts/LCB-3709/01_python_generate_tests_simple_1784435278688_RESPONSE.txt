import pytest
from solution import hasSpecialSubstring


def test_canonical_example_1():
    # spec: s = "aaabaaa", k = 3 -> true
    # substring s[4..6] = "aaa" satisfies all conditions
    assert hasSpecialSubstring("aaabaaa", 3) is True


def test_canonical_example_2():
    # spec: s = "abc", k = 2 -> false
    # no substring of length 2 with one distinct character satisfies conditions
    assert hasSpecialSubstring("abc", 2) is False


def test_substring_at_start_no_char_before():

    assert hasSpecialSubstring("aaaa", 2) is False

    # substring at start (no char before), different char after -> true
    assert hasSpecialSubstring("aab", 2) is True

    # substring at end (no char after), different char before -> true
    assert hasSpecialSubstring("baaa", 3) is True


def test_substring_at_end_same_char_before():
    # substring at end, same char before -> false
    assert hasSpecialSubstring("aaaa", 3) is False

    assert hasSpecialSubstring("abbbca", 3) is True

    # substring in middle, same char before -> false
    assert hasSpecialSubstring("aaabbb", 3) is False


def test_substring_in_middle_same_after():
    # substring in middle, same char after -> false
    assert hasSpecialSubstring("bbbaaaa", 3) is False

    # positions 1-3: "aaa" - char before is 'a', char after is 'a' -> false
    # positions 2-4: "aaa" - char before is 'a', no char after -> false
    assert hasSpecialSubstring("aaaaa", 3) is False


def test_run_exactly_k_isolated():
