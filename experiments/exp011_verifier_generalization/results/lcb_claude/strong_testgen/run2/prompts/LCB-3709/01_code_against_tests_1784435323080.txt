## Implementation

```python
def hasSpecialSubstring(s: str, k: int) -> bool:
    n = len(s)
    if k > n:
        return False
    for i in range(n - k + 1):
        substr = s[i:i + k]
        if len(set(substr)) == 1:
            char_before = s[i - 1] if i > 0 else None
            char_after = s[i + k] if i + k < n else None
            if (char_before is None or char_before != substr[0]) and (char_after is None or char_after != substr[0]):
                return True
    return False
```
