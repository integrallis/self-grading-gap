```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirror_map = {chr(i): chr(219 - i) for i in range(97, 123)}  # Mapping a-z to z-a

    for i in range(len(s)):
        if marked[i]:  # Skip if already marked
            continue
        mirror_char = mirror_map[s[i]]
        for j in range(i - 1, -1, -1):  # Look for the closest unmarked mirror
            if not marked[j] and s[j] == mirror_char:
                marked[i] = True
                marked[j] = True
                score += i - j
                break
    return score
```
