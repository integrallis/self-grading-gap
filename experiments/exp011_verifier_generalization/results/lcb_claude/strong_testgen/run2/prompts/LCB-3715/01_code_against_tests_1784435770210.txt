```python
from typing import List

def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Dictionary to hold the number of coins in each bag position
    coin_map = {}

    # Fill the coin_map with the coin counts for each bag
    for l, r, c in coins:
        for pos in range(l, r + 1):
            coin_map[pos] = coin_map.get(pos, 0) + c

    # Create a sorted list of unique positions
    positions = sorted(coin_map.keys())
    max_coins = 0

    # Iterate through possible starting points to find the maximum coins in k consecutive bags
    for start in range(len(positions)):
        end = start
        total_coins = 0
        # Count the total coins in the range of k bags
        while end < len(positions) and positions[end] < positions[start] + k:
            total_coins += coin_map[positions[end]]
            end += 1
        max_coins = max(max_coins, total_coins)

    return max_coins
```
