    # spec: [4,3,2,1] -> [0,0,1,1]

    # Sort: [0,0,1,1]
    assert transformArray([4, 3, 2, 1]) == [0, 0, 1, 1]


def test_canonical_example_2():
    # spec: [1,5,1,4,2] -> [0,0,1,1,1]
    # Replace even (4,2) with 0, odd (1,5,1) with 1: [1,1,1,0,0]
    # Sort: [0,0,1,1,1]
    assert transformArray([1, 5, 1, 4, 2]) == [0, 0, 1, 1, 1]

    # [1,3,5] -> replace all with 1: [1,1,1] -> sort: [1,1,1]
