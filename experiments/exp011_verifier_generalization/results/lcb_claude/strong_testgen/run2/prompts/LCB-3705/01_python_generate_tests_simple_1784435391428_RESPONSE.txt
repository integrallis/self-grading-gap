from typing import List
import pytest
from solution import largestInteger


def test_canonical_example_1():
    # spec: nums = [3,9,2,1,7], k = 3 -> 7
    # 3 appears in 1 subarray, 7 appears in 1 subarray, largest is 7
    assert largestInteger([3, 9, 2, 1, 7], 3) == 7


def test_canonical_example_2():
    # spec: nums = [3,9,7,2,1,7], k = 4 -> 3
    # 3 appears in 1 subarray of size 4
    assert largestInteger([3, 9, 7, 2, 1, 7], 4) == 3


def test_canonical_example_3():
    # spec: nums = [0,0], k = 1 -> -1
    # 0 appears in 2 subarrays of size 1: [0] and [0]
    assert largestInteger([0, 0], 1) == -1


def test_single_element_array():
    # nums = [5], k = 1 -> 5
    # Only one subarray [5], so 5 appears in exactly 1 subarray
    assert largestInteger([5], 1) == 5


def test_k_equals_array_length():
    # nums = [1, 2, 3], k = 3 -> 3
    # Only one subarray [1,2,3], all elements appear in exactly 1 subarray
    # Largest is 3
    assert largestInteger([1, 2, 3], 3) == 3


def test_all_elements_appear_in_multiple_subarrays():
    # nums = [1, 1, 1], k = 2 -> -1
    # Subarrays: [1,1], [1,1]
    # 1 appears in 2 subarrays, so no element appears in exactly 1
    assert largestInteger([1, 1, 1], 2) == -1


def test_multiple_almost_missing_integers():
    # nums = [5, 1, 2, 3, 4], k = 2
    # Subarrays: [5,1], [1,2], [2,3], [3,4]
    # 5 appears in 1 subarray: [5,1]
    # 1 appears in 2 subarrays: [5,1], [1,2]
    # 2 appears in 2 subarrays: [1,2], [2,3]
    # 3 appears in 2 subarrays: [2,3], [3,4]
    # 4 appears in 1 subarray: [3,4]
    # Almost missing: 5, 4; largest is 5
    assert largestInteger([5, 1, 2, 3, 4], 2) == 5


def test_element_at_boundary():
    # nums = [10, 2, 3], k = 2
    # Subarrays: [10,2], [2,3]
    # 10 appears in 1 subarray: [10,2]
    # 2 appears in 2 subarrays: [10,2], [2,3]
    # 3 appears in 1 subarray: [2,3]
    # Almost missing: 10, 3; largest is 10
    assert largestInteger([10, 2, 3], 2) == 10


def test_duplicate_values_different_positions():
    # nums = [7, 1, 2, 7, 3], k = 2
    # Subarrays: [7,1], [1,2], [2,7], [7,3]
    # 7 appears in 3 subarrays: [7,1], [2,7], [7,3]
    # 1 appears in 2 subarrays: [7,1], [1,2]
    # 2 appears in 2 subarrays: [1,2], [2,7]
    # 3 appears in 1 subarray: [7,3]
    # Almost missing: 3; largest is 3
    assert largestInteger([7, 1, 2, 7, 3], 2) == 3


def test_k_equals_one_all_unique():
    # nums = [3, 5, 2], k = 1
    # Subarrays: [3], [5], [2]
    # Each element appears in exactly 1 subarray
    # Largest is 5
    assert largestInteger([3, 5, 2], 1) == 5


def test_larger_values_within_constraints():

    # 50 appears in 1 subarray: [50,0]
    # 0 appears in 2 subarrays: [50,0], [0,25]
    # 25 appears in 1 subarray: [0,25]
    # Almost missing: 50, 25; largest is 50
    assert largestInteger([50, 0, 25], 2) == 50


def test_all_same_values():
    # nums = [3, 3, 3, 3], k = 2
    # Subarrays: [3,3], [3,3], [3,3]
    # 3 appears in 3 subarrays
    # No element appears in exactly 1 subarray
    assert largestInteger([3, 3, 3, 3], 2) == -1
