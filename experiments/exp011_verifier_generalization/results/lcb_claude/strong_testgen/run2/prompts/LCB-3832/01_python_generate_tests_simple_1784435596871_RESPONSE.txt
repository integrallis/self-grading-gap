import pytest
from typing import List
from solution import minCosts


def test_canonical_example_1():
    # spec: cost = [5,3,4,1,3,2] -> [5,3,3,1,1,1]
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]


def test_canonical_example_2():
    # spec: cost = [1,2,4,6,7] -> [1,1,1,1,1]
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]


def test_single_person():
    # n=1, only position 0 available, must pay cost[0] to reach it
    assert minCosts([5]) == [5]


def test_two_people_increasing():
    # n=2, positions 0,1
    # pos 0: pay cost[0]=3
    # pos 1: pay cost[1]=7
    assert minCosts([3, 7]) == [3, 7]


def test_two_people_decreasing():
    # n=2, positions 0,1
    # pos 0: pay cost[0]=7
    # pos 1: pay cost[1]=3 OR pay cost[0]=7 then swap free with person 1
    # minimum is 3
    assert minCosts([7, 3]) == [7, 3]


def test_all_same_cost():
    # n=3, all costs equal
    # pos 0: pay 2
    # pos 1: pay 2
    # pos 2: pay 2, then swap free OR pay cost[1]=2 then swap free
    assert minCosts([2, 2, 2]) == [2, 2, 2]


def test_decreasing_costs():
    # n=4, costs = [10, 8, 6, 4]
    # pos 0: 10
    # pos 1: 8
    # pos 2: 6
    # pos 3: 4
    assert minCosts([10, 8, 6, 4]) == [10, 8, 6, 4]


def test_minimum_at_start():
    # n=3, minimum cost at position 0
    # pos 0: pay 1
    # pos 1: pay 1, then swap free
    # pos 2: pay 1, then swap free
    assert minCosts([1, 5, 9]) == [1, 1, 1]


def test_minimum_in_middle():
    # n=5, minimum cost in middle
    # pos 0: 5
    # pos 1: 4
    # pos 2: 1
    # pos 3: 1 (pay cost[2]=1, then swap free)
    # pos 4: 1 (pay cost[2]=1, then swap free)
    assert minCosts([5, 4, 1, 6, 8]) == [5, 4, 1, 1, 1]


def test_minimum_at_end():
    # n=4, minimum cost at last position
    # pos 0: 9
    # pos 1: 7
    # pos 2: 5
    # pos 3: 1
    assert minCosts([9, 7, 5, 1]) == [9, 7, 5, 1]


def test_three_element_mixed():
    # n=3, costs = [3, 1, 2]
    # pos 0: 3
    # pos 1: 1
    # pos 2: min(2, 1) = 1 (pay cost[1]=1, swap free with person 2)
    assert minCosts([3, 1, 2]) == [3, 1, 1]
