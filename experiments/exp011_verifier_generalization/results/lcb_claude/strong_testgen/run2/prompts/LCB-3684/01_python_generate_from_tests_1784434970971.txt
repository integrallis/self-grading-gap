def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]
    for i in range(len(s) - len(prefix) + 1):
        if s[i:i + len(prefix)] == prefix:
            if s[i + len(prefix):].find(suffix) == 0:
                return True
    return False

```
