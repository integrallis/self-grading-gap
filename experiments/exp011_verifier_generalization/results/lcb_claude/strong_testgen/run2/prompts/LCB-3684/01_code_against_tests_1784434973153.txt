## Implementation

```python
def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]
    len_prefix = len(prefix)
    len_suffix = len(suffix)
    for i in range(len(s) - len_prefix + 1):
        if s[i:i + len_prefix] == prefix:
            if (len_suffix == 0) or (s[i + len_prefix:i + len_prefix + len_suffix].startswith(suffix)):
                return True
    return False

```
