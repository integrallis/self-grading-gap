import pytest
from solution import hasMatch


def test_canonical_example_1():
    # spec: s = "leetcode", p = "ee*e" -> true
    # "eetcode" is a substring of "leetcode" that matches pattern "ee*e" (with * = "tcod")
    assert hasMatch("leetcode", "ee*e") is True


def test_canonical_example_2():
    # spec: s = "car", p = "c*v" -> false
    # No substring of "car" matches pattern "c*v"
    assert hasMatch("car", "c*v") is False


def test_canonical_example_3():
    # spec: s = "luck", p = "u*" -> true
    # Substrings "u", "uc", "uck" all match pattern "u*"
    assert hasMatch("luck", "u*") is True


def test_star_at_beginning():
    # pattern "*a" should match any substring ending with 'a'
    assert hasMatch("bca", "*a") is True


def test_star_at_end():
    # pattern "a*" should match any substring starting with 'a'
    assert hasMatch("abc", "a*") is True


def test_star_in_middle():
    # pattern "a*c" should match substrings starting with 'a' and ending with 'c'
    assert hasMatch("abcd", "a*c") is True


def test_star_only():
    # pattern "*" should match any non-empty substring (always true for non-empty s)
    assert hasMatch("a", "*") is True


def test_no_match_prefix():
    # pattern "x*y" where 'x' is not in s
    assert hasMatch("abc", "x*y") is False


def test_no_match_suffix():
    # pattern "a*z" where 'z' is not in s
    assert hasMatch("abc", "a*z") is False


def test_exact_match_with_star_zero_chars():
    # pattern "ab*c" matches substring "abc" when * = ""
    assert hasMatch("abc", "ab*c") is True


def test_star_replaces_multiple_chars():
    # pattern "a*d" matches substring "abcd" when * = "bc"
    assert hasMatch("abcd", "a*d") is True


def test_pattern_longer_than_string():
    # pattern "ab*cd" might not match short string "ab"
    assert hasMatch("ab", "ab*cd") is False


def test_multiple_possible_matches():
    # pattern "a*a" can match "aa", "aba", "abca", etc.
    assert hasMatch("abaca", "a*a") is True


def test_single_char_string_and_pattern():
    # s = "a", p = "*" should match
    assert hasMatch("a", "*") is True


def test_pattern_prefix_not_in_string():
    # pattern "z*" where 'z' is not in s
    assert hasMatch("abc", "z*") is False


def test_pattern_suffix_not_in_string():
    # pattern "*z" where 'z' is not in s
    assert hasMatch("abc", "*z") is False


def test_both_parts_present_but_not_adjacent():
    # pattern "a*b" where 'a' and 'b' exist but need correct order
    assert hasMatch("acb", "a*b") is True


def test_both_parts_present_wrong_order():
    # pattern "b*a" in "abc" - 'b' must come before 'a' in the substring
