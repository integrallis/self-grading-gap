from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    
    # Sort the array to easily find min/max in subsequences
    nums.sort()
    
    # Calculate the contribution of each element in nums
    for i in range(n):
        # Contribution of nums[i] as min in subsequences of size up to k
        min_contrib = (1 << min(i, k)) - 1
        # Contribution of nums[i] as max in subsequences of size up to k
        max_contrib = (1 << min(n - 1 - i, k)) - 1
        # Total contribution for nums[i]
        total_sum += (min_contrib + max_contrib + 2) * nums[i]
        total_sum %= MOD

    return total_sum % MOD

```
