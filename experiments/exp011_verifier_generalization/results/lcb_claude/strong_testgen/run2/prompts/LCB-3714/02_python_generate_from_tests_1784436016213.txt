from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Calculate contributions of each number as min and max
    for i in range(n):
        # Count how many times nums[i] will be min and max
        min_count = 0
        max_count = 0

        for j in range(n):
            if j != i:
                # Count how many subsequences have nums[i] as min
                if nums[j] > nums[i]:
                    min_count += (1 << (n - 1 - j))
                # Count how many subsequences have nums[i] as max
                if nums[j] < nums[i]:
                    max_count += (1 << (n - 1 - j))

        # Add contributions of nums[i] as min
        total_sum += (min_count + 1) * nums[i]
        # Add contributions of nums[i] as max
        total_sum += (max_count + 1) * nums[i]

    return total_sum % MOD
