If coins[i][j] >= 0, the robot gains that many coins.
If coins[i][j] < 0, the robot encounters a robber, and the robber steals the absolute value of coins[i][j] coins.

The robot has a special ability to neutralize robbers in at most 2 cells on its path, preventing them from stealing coins in those cells.
Note: The robot's total coins can be negative.

Move to (1, 2), gaining 3 coins (total coins = 1 + 3 = 4).
Move to (2, 2), gaining 4 coins (total coins = 4 + 4 = 8).

```python
from typing import List
```

```python
def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[[-float('inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = max(coins[0][0], 0)
    if coins[0][0] < 0:
        dp[0][0][1] = 0
        dp[0][0][2] = 0

    for i in range(m):
        for j in range(n):
            for k in range(3):
                if i == 0 and j == 0:
                    continue
                current = coins[i][j]
                gain = max(current, 0)  # Gain is zero if negative unless neutralized
                if current < 0:
                    # If we encounter a robber and can neutralize
                    if k > 0:
                        # Neutralization effect
                        gain = 0
                    # Otherwise, we just lose this amount
                # Check from the top
                if i > 0:
                    dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k] + gain)
                # Check from the left
                if j > 0:
                    dp[i][j][k] = max(dp[i][j][k], dp[i][j-1][k] + gain)
                # Applying neutralization if there's a robber
                if current < 0 and k > 0:
                    if i > 0:
                        dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k-1])
                    if j > 0:
                        dp[i][j][k] = max(dp[i][j][k], dp[i][j-1][k-1])

    return max(dp[m-1][n-1][0], dp[m-1][n-1][1], dp[m-1][n-1][2])

```
