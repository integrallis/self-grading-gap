def test_canonical_example_1():
    # spec: Example 1
    # numberOfUsers = 2, events = [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","71","HERE"]]
    # At timestamp 10, id1 and id0 are mentioned. mentions = [1,1]
