from typing import List
import heapq


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Build the graph in reverse for reachability checks
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[v].append((u, w))

    # Check reachability from all nodes to node 0 using a BFS/DFS approach
    def can_reach_zero():
        visited = [False] * n
        stack = [0]
        while stack:
            node = stack.pop()
            for neighbor, _ in graph[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    stack.append(neighbor)
        return visited

    visited = can_reach_zero()
    # If any node cannot reach zero, return -1
    for i in range(n):
        if not visited[i]:
            return -1

    # Binary search for the minimum possible maximum edge weight
    left, right = 1, 10**6
    while left < right:
        mid = (left + right) // 2
        # Create a filtered graph with edges <= mid weight
        filtered_edges = [[] for _ in range(n)]
        for u, v, w in edges:
            if w <= mid:
                filtered_edges[u].append(v)

        # Check if we can reach node 0 from all nodes with the filtered graph
        def can_reach_zero_filtered():
            visited = [False] * n
            stack = [0]
            while stack:
                node = stack.pop()
                for neighbor in filtered_edges[node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        stack.append(neighbor)
            return visited

        visited_filtered = can_reach_zero_filtered()
        # Count outgoing edges for each node
        out_degree = [0] * n
        for u in range(n):
            out_degree[u] = len(filtered_edges[u])

        # Check if any node exceeds the threshold for outgoing edges
        if all(out <= threshold for out in out_degree) and all(visited_filtered):
            right = mid  # We can potentially lower the max weight
        else:
            left = mid + 1  # We need to increase the max weight

    return left

```
