from typing import List
import pytest
from solution import minMaxWeight


def test_canonical_example_1():
    # spec: n=5, edges=[[1,0,1],[2,0,2],[3,0,1],[4,3,1],[2,1,1]], threshold=2 -> 1
    # Remove edge 2->0. Maximum weight among remaining edges is 1.
    assert minMaxWeight(5, [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]], 2) == 1


def test_canonical_example_2():
    # spec: n=5, edges=[[0,1,1],[0,2,2],[0,3,1],[0,4,1],[1,2,1],[1,4,1]], threshold=1 -> -1
    # Impossible to reach node 0 from node 2.
    assert minMaxWeight(5, [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]], 1) == -1


def test_canonical_example_3():
    # spec: n=5, edges=[[1,2,1],[1,3,3],[1,4,5],[2,3,2],[3,4,2],[4,0,1]], threshold=1 -> 2
    # Remove edges 1->3 and 1->4. Maximum weight among remaining edges is 2.
    assert minMaxWeight(5, [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]], 1) == 2


def test_canonical_example_4():
    # spec: n=5, edges=[[1,2,1],[1,3,3],[1,4,5],[2,3,2],[4,0,1]], threshold=1 -> -1
    # Impossible to satisfy all conditions.
    assert minMaxWeight(5, [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]], 1) == -1


def test_simple_path_to_zero():
    # n=2, single edge 1->0 with weight 5, threshold=1
    # Node 1 can reach node 0. Max weight is 5.
    assert minMaxWeight(2, [[1, 0, 5]], 1) == 5


def test_direct_edges_to_zero():
    # n=3, edges [[1,0,3],[2,0,2]], threshold=1
    # Both nodes can reach 0 directly. Max weight is 3.
    assert minMaxWeight(3, [[1, 0, 3], [2, 0, 2]], 1) == 3


def test_threshold_forces_edge_removal():
    # n=3, edges [[1,0,1],[1,2,2],[2,0,3]], threshold=1
    # Node 1 has 2 outgoing edges, exceeds threshold=1.
    # Must remove one. Can remove 1->2 (weight 2), keeping max weight at 3.
    # Or remove 1->0 (weight 1), node 1 reaches 0 via 2, max weight 3.
    # Best: remove higher weight edge 1->2, max is 3. Or remove 2->0, node 2 goes via 1, max is 2.
    # Actually: keep 1->0 and 1->2, remove 2->0: node 2 can't reach 0. 
    # Keep 1->0 and 2->0, remove 1->2: max is 3, node 1 has 1 edge, node 2 has 1 edge.
    # Keep 1->2 and 2->0, remove 1->0: max is 3, node 1 reaches 0 via 2.
    # All valid solutions have max weight 3.
    assert minMaxWeight(3, [[1, 0, 1], [1, 2, 2], [2, 0, 3]], 1) == 3


def test_chain_to_zero():
    # n=4, edges [[3,2,1],[2,1,2],[1,0,3]], threshold=1
    # Node 3 reaches 0 via 3->2->1->0. Max weight is 3.
    assert minMaxWeight(4, [[3, 2, 1], [2, 1, 2], [1, 0, 3]], 1) == 3


def test_multiple_paths_choose_lower_weight():
    # n=3, edges [[1,0,10],[1,2,1],[2,0,2]], threshold=2
    # Node 1 can reach 0 directly (weight 10) or via 2 (max weight 2).
    # Can remove 1->0, max weight becomes 2.
    assert minMaxWeight(3, [[1, 0, 10], [1, 2, 1], [2, 0, 2]], 2) == 2


def test_node_zero_has_outgoing_edges():
    # n=3, edges [[0,1,5],[1,0,3],[2,1,2]], threshold=1
    # Node 0 can have outgoing edges (doesn't violate reachability to 0).
    # Node 1 reaches 0 via edge 1->0. Node 2 reaches 0 via 2->1->0.
    # Max weight is 5.
    assert minMaxWeight(3, [[0, 1, 5], [1, 0, 3], [2, 1, 2]], 1) == 5


def test_unreachable_node():
    # n=3, edges [[1,0,1]], threshold=1
    # Node 2 cannot reach node 0. Return -1.
    assert minMaxWeight(3, [[1, 0, 1]], 1) == -1


def test_all_nodes_reach_zero_different_weights():
    # n=4, edges [[1,0,1],[2,0,2],[3,0,3]], threshold=1
    # All nodes reach 0 directly. Max weight is 3.
    assert minMaxWeight(4, [[1, 0, 1], [2, 0, 2], [3, 0, 3]], 1) == 3


def test_threshold_higher_than_needed():
    # n=3, edges [[1,0,1],[2,0,2]], threshold=5
    # Threshold is high, no edges need removal. Max weight is 2.
    assert minMaxWeight(3, [[1, 0, 1], [2, 0, 2]], 5) == 2
