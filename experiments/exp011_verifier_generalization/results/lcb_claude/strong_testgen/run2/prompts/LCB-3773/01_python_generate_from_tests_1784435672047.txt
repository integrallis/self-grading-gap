from typing import List

def minimumPairRemoval(nums: List[int]) -> int:
    operations = 0
    while True:
        # Check if the array is non-decreasing
        if all(nums[i] <= nums[i + 1] for i in range(len(nums) - 1)):
            break
        # Find the pair with the minimum sum
        min_sum = float('inf')
        min_index = -1
        for i in range(len(nums) - 1):
            pair_sum = nums[i] + nums[i + 1]
            if pair_sum < min_sum:
                min_sum = pair_sum
                min_index = i
            elif pair_sum == min_sum and min_index == -1:
                min_index = i
        # Merge the found pair
        nums[min_index] = min_sum
        del nums[min_index + 1]
        operations += 1
    return operations

```
