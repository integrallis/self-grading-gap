import pytest
from solution import hasSameDigits


def test_canonical_example_1():
    # spec: s = "3902"
    # First operation: "3902" -> "292" (pairs: 3+9=12%10=2, 9+0=9, 0+2=2)
    # Second operation: "292" -> "11" (pairs: 2+9=11%10=1, 9+2=11%10=1)
    # Final: "11" -> both digits same -> true
    assert hasSameDigits("3902") is True


def test_canonical_example_2():

    # Second operation: "7157" -> "862" (7+1=8, 1+5=6, 5+7=12%10=2)
