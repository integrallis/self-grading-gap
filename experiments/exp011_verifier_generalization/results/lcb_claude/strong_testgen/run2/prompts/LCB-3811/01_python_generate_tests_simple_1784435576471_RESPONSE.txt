import pytest
from solution import reverseDegree


def test_canonical_example_1():
    # spec: "abc" -> 148
    # 'a' (26) * 1 + 'b' (25) * 2 + 'c' (24) * 3 = 26 + 50 + 72 = 148
    assert reverseDegree("abc") == 148


def test_canonical_example_2():
    # spec: "zaza" -> 160
    # 'z' (1) * 1 + 'a' (26) * 2 + 'z' (1) * 3 + 'a' (26) * 4 = 1 + 52 + 3 + 104 = 160
    assert reverseDegree("zaza") == 160


def test_single_character_a():
    # 'a' has reversed alphabet position 26, string position 1
    # 26 * 1 = 26
    assert reverseDegree("a") == 26


def test_single_character_z():
    # 'z' has reversed alphabet position 1, string position 1
    # 1 * 1 = 1
    assert reverseDegree("z") == 1


def test_single_character_m():
    # 'm' is the 13th letter, so reversed position is 26 - 13 + 1 = 14
    # 14 * 1 = 14
    assert reverseDegree("m") == 14


def test_all_same_character():
    # "aaa" -> 'a' (26) * 1 + 'a' (26) * 2 + 'a' (26) * 3 = 26 + 52 + 78 = 156
    assert reverseDegree("aaa") == 156


def test_reversed_alphabet_order():
    # "zyxw" -> 'z' (1) * 1 + 'y' (2) * 2 + 'x' (3) * 3 + 'w' (4) * 4
    # = 1 + 4 + 9 + 16 = 30
    assert reverseDegree("zyxw") == 30


def test_alphabet_order():
    # "abcd" -> 'a' (26) * 1 + 'b' (25) * 2 + 'c' (24) * 3 + 'd' (23) * 4
    # = 26 + 50 + 72 + 92 = 240
    assert reverseDegree("abcd") == 240


def test_longer_string():
    # "hello" -> 'h' (19) * 1 + 'e' (22) * 2 + 'l' (15) * 3 + 'l' (15) * 4 + 'o' (12) * 5
    # h: 26 - 8 + 1 = 19, e: 26 - 5 + 1 = 22, l: 26 - 12 + 1 = 15, o: 26 - 15 + 1 = 12
    # = 19 + 44 + 45 + 60 + 60 = 228
    assert reverseDegree("hello") == 228
