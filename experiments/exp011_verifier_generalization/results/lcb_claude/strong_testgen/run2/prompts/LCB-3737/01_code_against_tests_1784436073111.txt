```python
from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize a DP table to keep track of costs
    dp = [[float('inf')] * 3 for _ in range(n)]

    # Fill the first house costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the DP table while ensuring no two adjacent houses have the same color
    for i in range(1, n):
        for j in range(3):
            for k in range(3):
                if j != k:
                    dp[i][j] = min(dp[i][j], dp[i-1][k] + cost[i][j])

    # To handle the equidistant constraint, we will perform a second DP fill
    final_min_cost = float('inf')
    for j in range(3):
        # Calculate the cost while using the opposite house for equidistant checking
        dp_opposite = [[float('inf')] * 3 for _ in range(n)]
        for i in range(n):
            for k in range(3):
                if (i < n // 2 and j != k) or (i >= n // 2 and j != (2 - k)):
                    dp_opposite[n - 1 - i][k] = min(dp_opposite[n - 1 - i][k], dp[i][j])

        # Gather minimum cost from the second DP
        for k in range(3):
            if n // 2 >= 1 and j != (2 - k):
                final_min_cost = min(final_min_cost, dp_opposite[n-1][k])

    return final_min_cost

```
