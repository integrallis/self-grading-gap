from typing import List

from solution import minCost


def test_canonical_example_1():
    # Example 1: n=4, optimal sequence [1,2,3,2] with costs [3,2,1,3]
    # Total cost: 3 + 2 + 1 + 3 = 9
    n = 4
    cost = [[3, 5, 7], [6, 2, 9], [4, 8, 1], [7, 3, 5]]
    assert minCost(n, cost) == 9


def test_canonical_example_2():
    # Example 2: n=6, optimal sequence [1,3,2,3,1,2] with costs [2,8,1,2,3,2]
    # Total cost: 2 + 8 + 1 + 2 + 3 + 2 = 18
    n = 6
    cost = [[2, 4, 6], [5, 3, 8], [7, 1, 9], [4, 6, 2], [3, 5, 7], [8, 2, 4]]
    assert minCost(n, cost) == 18


def test_minimum_n_equals_2():
    # n=2: two houses at positions 0 and 1
    # They are adjacent (different colors) and equidistant from ends (different colors)
    # Must choose colors c1 != c2
    # If cost = [[1,2,3], [4,5,6]], optimal is color 1 for house 0 (cost 1) and color 2 for house 1 (cost 5)
    # Total: 1 + 5 = 6
    n = 2
    cost = [[1, 2, 3], [4, 5, 6]]
    assert minCost(n, cost) == 6
