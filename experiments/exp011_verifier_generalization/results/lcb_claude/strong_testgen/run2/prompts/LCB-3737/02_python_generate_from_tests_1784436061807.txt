from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize dp array for two sets of colors
    dp = [[0] * 3 for _ in range(2)]
    
    # First house costs
    for j in range(3):
        dp[0][j] = cost[0][j]
    
    for i in range(1, n):
        current = i % 2
        previous = (i - 1) % 2
        
        # Calculate costs for current house with respect to previous house
        for j in range(3):
            dp[current][j] = float('inf')
            for k in range(3):
                if j != k:
                    dp[current][j] = min(dp[current][j], dp[previous][k] + cost[i][j])
        
        # Handle the equidistant condition for the first half
        if i < n // 2:
            for j in range(3):
                for k in range(3):
                    if j != k:
                        dp[current][j] = min(dp[current][j], dp[previous][k] + cost[i][j])
    
    # For the last house, we need to consider the equidistant conditions
    result = float('inf')
    last_half = n // 2
    for j in range(3):
        for k in range(3):
            if j != k:
                result = min(result, dp[(n - 1) % 2][j])
    return result
```
