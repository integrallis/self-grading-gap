import pytest
from solution import hasSpecialSubstring


def test_canonical_example_1():
    # spec: s = "aaabaaa", k = 3 -> true
    # substring s[4..6] == "aaa" satisfies all conditions
    assert hasSpecialSubstring("aaabaaa", 3) is True


def test_canonical_example_2():
    # spec: s = "abc", k = 2 -> false
    # no substring of length 2 with one distinct character satisfies conditions
    assert hasSpecialSubstring("abc", 2) is False


def test_single_character_string_k_equals_1():
    # s = "a", k = 1
    # substring "a" at index 0 has length 1, one distinct char, no before/after chars
    assert hasSpecialSubstring("a", 1) is True


def test_all_same_characters_k_equals_length():
    # s = "aaaa", k = 4
    # entire string is one char, no before/after chars to violate conditions
    assert hasSpecialSubstring("aaaa", 4) is True


def test_all_same_characters_k_less_than_length():
    # s = "aaaa", k = 2
    # substring "aa" at index 0 works: no before char, after char is 'a' (same)
    # substring "aa" at index 1 fails: before is 'a', after is 'a'
    # substring "aa" at index 2 fails: before is 'a', no after char but before fails
    # Actually s[0:2] = "aa" has no before, but s[2] = 'a' which is same, so fails
    # s = "aaaa", substring at [0:2] = "aa", before=none, after='a' (same) -> fails
    # None of the substrings work because all are surrounded by same char
    assert hasSpecialSubstring("aaaa", 2) is False


def test_different_char_before_and_after():
    # s = "baaac", k = 3
    # substring s[1:4] = "aaa", before='b' (different), after='c' (different)
    assert hasSpecialSubstring("baaac", 3) is True


def test_same_char_after_fails():
    # s = "baaaa", k = 3
    # substring s[1:4] = "aaa", before='b' (different), after='a' (same) -> fails
    # substring s[2:5] = "aaa", before='a' (same) -> fails
    assert hasSpecialSubstring("baaaa", 3) is False


def test_same_char_before_fails():
    # s = "aaaab", k = 3
    # substring s[0:3] = "aaa", before=none, after='a' (same) -> fails
    # substring s[1:4] = "aaa", before='a' (same) -> fails
    assert hasSpecialSubstring("aaaab", 3) is False


def test_substring_at_start_no_before():
    # s = "aaab", k = 3
    # substring s[0:3] = "aaa", before=none, after='b' (different)
    assert hasSpecialSubstring("aaab", 3) is True


def test_substring_at_end_no_after():
    # s = "baaa", k = 3
    # substring s[1:4] = "aaa", before='b' (different), after=none
    assert hasSpecialSubstring("baaa", 3) is True


def test_multiple_groups_second_one_works():
    # s = "aaabbbcc", k = 3
    # substring s[0:3] = "aaa", before=none, after='b' (different) -> works
    assert hasSpecialSubstring("aaabbbcc", 3) is True


def test_k_equals_1_in_mixed_string():
    # s = "abc", k = 1
    # substring s[0:1] = "a", before=none, after='b' (different) -> works
    assert hasSpecialSubstring("abc", 1) is True


def test_alternating_characters():
    # s = "ababab", k = 1
    # substring s[0:1] = "a", before=none, after='b' (different) -> works
    assert hasSpecialSubstring("ababab", 1) is True


def test_no_valid_substring_different_chars():
    # s = "abcdef", k = 2
    # no substring of length 2 has all same characters
    assert hasSpecialSubstring("abcdef", 2) is False


def test_longer_run_k_in_middle():
    # s = "aaaaaa", k = 3
    # s[0:3] = "aaa", before=none, after='a' (same) -> fails
    # s[1:4] = "aaa", before='a' (same) -> fails
    # All substrings of length 3 have same char before or after
    assert hasSpecialSubstring("aaaaaa", 3) is False


def test_exact_run_length_isolated():
    # s = "abbbcd", k = 3
    # s[1:4] = "bbb", before='a' (different), after='c' (different)
    assert hasSpecialSubstring("abbbcd", 3) is True
