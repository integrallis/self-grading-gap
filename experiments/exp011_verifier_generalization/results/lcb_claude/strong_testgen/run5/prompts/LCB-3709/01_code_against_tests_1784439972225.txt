## Implementation

```python
def hasSpecialSubstring(s: str, k: int) -> bool:
    n = len(s)
    for i in range(n - k + 1):
        substring = s[i:i + k]
        if substring == substring[0] * k:
            before_is_different = i == 0 or s[i - 1] != substring[0]
            after_is_different = i + k == n or s[i + k] != substring[0]
            if before_is_different and after_is_different:
                return True
    return False

```
