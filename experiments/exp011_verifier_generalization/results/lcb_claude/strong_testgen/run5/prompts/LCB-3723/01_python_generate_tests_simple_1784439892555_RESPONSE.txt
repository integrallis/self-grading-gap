import pytest
from typing import List
from solution import sumOfGoodNumbers


def test_canonical_example_1():
    # spec: nums = [1,3,2,1,5,4], k = 2
    # Good elements:
    # - nums[1] = 3: i-k=-1 (doesn't exist), i+k=3, 3 > nums[3]=1 ✓
    # - nums[4] = 5: i-k=2, i+k=6 (doesn't exist), 5 > nums[2]=2 ✓
    # - nums[5] = 4: i-k=3, i+k=7 (doesn't exist), 4 > nums[3]=1 ✓
    # Sum = 3 + 5 + 4 = 12
    assert sumOfGoodNumbers([1, 3, 2, 1, 5, 4], 2) == 12


def test_canonical_example_2():
    # spec: nums = [2,1], k = 1
    # Good elements:
    # - nums[0] = 2: i-k=-1 (doesn't exist), i+k=1, 2 > nums[1]=1 ✓
    # - nums[1] = 1: i-k=0, i+k=2 (doesn't exist), 1 > nums[0]=2? No ✗
    # Sum = 2
    assert sumOfGoodNumbers([2, 1], 1) == 2


def test_all_elements_good_when_neither_index_exists():
    # nums = [5, 10], k = 2
    # For both indices, neither i-k nor i+k exist
    # Both elements are good
    # Sum = 5 + 10 = 15
    assert sumOfGoodNumbers([5, 10], 2) == 15


def test_no_good_elements():
    # nums = [5, 1, 10], k = 1
    # nums[0] = 5: i+k=1, 5 > nums[1]=1 ✓
    # nums[1] = 1: i-k=0, i+k=2, 1 > nums[0]=5? No, 1 > nums[2]=10? No ✗
    # nums[2] = 10: i-k=1, 10 > nums[1]=1 ✓
    # Sum = 5 + 10 = 15
    assert sumOfGoodNumbers([5, 1, 10], 1) == 15


def test_middle_element_only_when_not_good():
    # nums = [10, 5, 1], k = 1
    # nums[0] = 10: i+k=1, 10 > nums[1]=5 ✓
    # nums[1] = 5: i-k=0, i+k=2, 5 > nums[0]=10? No ✗
    # nums[2] = 1: i-k=1, 1 > nums[1]=5? No ✗
    # Sum = 10
    assert sumOfGoodNumbers([10, 5, 1], 1) == 10


def test_equal_values_not_good():
    # nums = [3, 3, 3], k = 1
    # nums[0] = 3: i+k=1, 3 > nums[1]=3? No (strictly greater) ✗
    # nums[1] = 3: i-k=0, i+k=2, 3 > nums[0]=3? No, 3 > nums[2]=3? No ✗
    # nums[2] = 3: i-k=1, 3 > nums[1]=3? No ✗
    # Sum = 0
    assert sumOfGoodNumbers([3, 3, 3], 1) == 0


def test_larger_k_value():
    # nums = [1, 2, 3, 4, 5, 6], k = 3
    # nums[0] = 1: i+k=3, 1 > nums[3]=4? No ✗
    # nums[1] = 2: i+k=4, 2 > nums[4]=5? No ✗
    # nums[2] = 3: i+k=5, 3 > nums[5]=6? No ✗
    # nums[3] = 4: i-k=0, 4 > nums[0]=1 ✓
    # nums[4] = 5: i-k=1, 5 > nums[1]=2 ✓
    # nums[5] = 6: i-k=2, 6 > nums[2]=3 ✓
    # Sum = 4 + 5 + 6 = 15
    assert sumOfGoodNumbers([1, 2, 3, 4, 5, 6], 3) == 15


def test_single_comparison_needed():
    # nums = [100, 1, 2, 3], k = 2
    # nums[0] = 100: i+k=2, 100 > nums[2]=2 ✓
    # nums[1] = 1: i+k=3, 1 > nums[3]=3? No ✗
    # nums[2] = 2: i-k=0, 2 > nums[0]=100? No ✗
    # nums[3] = 3: i-k=1, 3 > nums[1]=1 ✓
    # Sum = 100 + 3 = 103
    assert sumOfGoodNumbers([100, 1, 2, 3], 2) == 103
