```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # Create a mirror mapping

    for i in range(len(s)):
        if marked[i]:  # Skip if already marked
            continue
        for j in range(i - 1, -1, -1):  # Look for the closest unmarked mirror
            if not marked[j] and mirror[s[i]] == s[j]:
                marked[i] = marked[j] = True  # Mark both indices
                score += i - j  # Update score
                break  # Stop after the first found mirror
    return score
```
