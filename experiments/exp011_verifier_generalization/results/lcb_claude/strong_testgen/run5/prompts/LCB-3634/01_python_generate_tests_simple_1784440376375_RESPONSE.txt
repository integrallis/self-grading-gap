import pytest
from solution import calculateScore


def test_canonical_example_1():
    # spec: s = "aczzx" -> 5
    # i=0: 'a', no mirror 'z' before, skip
    # i=1: 'c', no mirror 'x' before, skip
    # i=2: 'z', mirror 'a' at j=0 (unmarked), mark both, score += 2-0 = 2
    # i=3: 'z', mirror 'a' at j=0 (marked), skip
    # i=4: 'x', mirror 'c' at j=1 (unmarked), mark both, score += 4-1 = 3
    # total score = 2 + 3 = 5
    assert calculateScore("aczzx") == 5


def test_canonical_example_2():
    # spec: s = "abcdef" -> 0
    # no character has its mirror before it
    # i=0: 'a', no mirror 'z' before
    # i=1: 'b', no mirror 'y' before
    # i=2: 'c', no mirror 'x' before
    # i=3: 'd', no mirror 'w' before
    # i=4: 'e', no mirror 'v' before
    # i=5: 'f', no mirror 'u' before
    # total score = 0
    assert calculateScore("abcdef") == 0


def test_single_character():
    # s = "a", length 1
    # i=0: no j < 0, skip
    # total score = 0
    assert calculateScore("a") == 0


def test_mirror_pair_adjacent():
    # s = "az"
    # i=0: 'a', no j < 0, skip
    # i=1: 'z', mirror 'a' at j=0, score += 1-0 = 1
    # total score = 1
    assert calculateScore("az") == 1


def test_mirror_pair_reversed():
    # s = "za"
    # i=0: 'z', no j < 0, skip
    # i=1: 'a', mirror 'z' at j=0, score += 1-0 = 1
    # total score = 1
    assert calculateScore("za") == 1


def test_multiple_same_character_no_mirror():
    # s = "aaa"
    # 'a' mirrors to 'z', none present
    # i=0,1,2: all skip
    # total score = 0
    assert calculateScore("aaa") == 0


def test_multiple_mirrors_closest_unmarked():
    # s = "azaz"
    # i=0: 'a', skip
    # i=1: 'z', mirror 'a' at j=0, mark 0,1, score += 1-0 = 1
    # i=2: 'a', mirror 'z' at j=1 (marked), skip
    # i=3: 'z', mirror 'a' at j=2 (unmarked), mark 2,3, score += 3-2 = 1
    # total score = 1 + 1 = 2
    assert calculateScore("azaz") == 2


def test_mirror_with_gap():
    # s = "abcxyz"
    # i=0: 'a', no 'z' before
    # i=1: 'b', no 'y' before
    # i=2: 'c', no 'x' before
    # i=3: 'x', mirror 'c' at j=2, score += 3-2 = 1
    # i=4: 'y', mirror 'b' at j=1, score += 4-1 = 3
    # i=5: 'z', mirror 'a' at j=0, score += 5-0 = 5
    # total score = 1 + 3 + 5 = 9
    assert calculateScore("abcxyz") == 9


def test_middle_letters():
    # s = "mn"
    # 'm' is 13th letter (index 12), mirror is 'n' (14th letter, index 13)
    # i=0: 'm', no 'n' before
    # i=1: 'n', mirror 'm' at j=0, score += 1-0 = 1
    # total score = 1
    assert calculateScore("mn") == 1


def test_multiple_candidates_choose_closest():
    # s = "azazaz"
    # i=0: 'a', skip
    # i=1: 'z', mirror 'a' at j=0, mark 0,1, score += 1
    # i=2: 'a', mirror 'z' at j=1 (marked), skip
    # i=3: 'z', mirror 'a' at j=2 (unmarked), mark 2,3, score += 1
    # i=4: 'a', mirror 'z' at j=3 (marked), skip
    # i=5: 'z', mirror 'a' at j=4 (unmarked), mark 4,5, score += 1
    # total score = 1 + 1 + 1 = 3
    assert calculateScore("azazaz") == 3


def test_no_matching_mirrors():
    # s = "abc"
    # i=0: 'a', no 'z' before
    # i=1: 'b', no 'y' before
    # i=2: 'c', no 'x' before
    # total score = 0
    assert calculateScore("abc") == 0


def test_same_letter_not_mirror():
    # s = "aa"
    # 'a' mirrors to 'z', not to itself
    # i=0: skip
    # i=1: 'a', mirror 'z', not found
    # total score = 0
    assert calculateScore("aa") == 0
