import pytest
from solution import hasSameDigits


def test_canonical_example_1():
    # spec: s = "3902"
    # First operation: "3902" -> "292" (12%10=2, 9%10=9, 2%10=2)
    # Second operation: "292" -> "11" (11%10=1, 11%10=1)
    # Since "11" has same digits, output is true
    assert hasSameDigits("3902") is True


def test_canonical_example_2():
    # spec: s = "34789"
    # After first operation: "7157" (7%10=7, 11%10=1, 15%10=5, 17%10=7)
    # After second operation: "862" (8%10=8, 6%10=6, 12%10=2)
    # After third operation: "48" (14%10=4, 8%10=8)
    # Since '4' != '8', output is false
    assert hasSameDigits("34789") is False
