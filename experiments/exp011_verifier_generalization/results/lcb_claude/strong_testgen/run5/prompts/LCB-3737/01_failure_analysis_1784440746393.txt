def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize a DP table to hold the minimum costs
    dp = [[0] * 3 for _ in range(n)]
    
    # Base case: fill the first house costs
    for j in range(3):
        dp[0][j] = cost[0][j]
    
    # Fill the dp table based on the constraints
    for i in range(1, n):
        for j in range(3):
            # Avoid adjacent houses having the same color
            dp[i][j] = cost[i][j] + min(
                dp[i-1][(j + 1) % 3],
                dp[i-1][(j + 2) % 3]
            )
    
    # Handle the equidistant constraint
    min_cost = float('inf')
    for j in range(3):
        # Calculate the total cost by considering the last house colors
        total_cost = dp[n-1][j]  # Cost with last house painted j
        for k in range(3):

                continue
