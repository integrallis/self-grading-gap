```python
from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize a DP table to hold the minimum costs
    dp = [[0] * 3 for _ in range(n)]
    
    # Base case: fill the first house costs
    for j in range(3):
        dp[0][j] = cost[0][j]
    
    # Fill the dp table based on the constraints
    for i in range(1, n):
        for j in range(3):
            # Avoid adjacent houses having the same color
            dp[i][j] = cost[i][j] + min(
                dp[i-1][(j + 1) % 3],
                dp[i-1][(j + 2) % 3]
            )

    # Handle the equidistant constraint
    min_cost = float('inf')
    for j in range(3):
        # For the first configuration (last house color = j)
        total_cost = dp[n-1][j]
        for k in range(3):
            if k != j:
                total_cost_k = total_cost + cost[n-1][k]
                # Ensure no conflict with the equidistant house
                equidistant_index = n - 1 - (n - 2 - j)
                if equidistant_index >= 0 and equidistant_index != n-1:
                    for m in range(3):
                        if m != j and m != k:
                            total_cost_k = min(total_cost_k, dp[n-2][m] + cost[n-1][k])
                min_cost = min(min_cost, total_cost_k)
    
    return min_cost
```
