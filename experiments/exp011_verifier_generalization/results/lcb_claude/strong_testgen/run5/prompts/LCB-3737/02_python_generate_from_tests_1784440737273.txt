from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize a DP table to hold the minimum costs
    dp = [[0] * 3 for _ in range(n)]
    
    # Base case: fill the first house costs
    for j in range(3):
        dp[0][j] = cost[0][j]
    
    # Fill the dp table based on the constraints
    for i in range(1, n):
        for j in range(3):
            # Avoid adjacent houses having the same color
            dp[i][j] = cost[i][j] + min(
                dp[i-1][(j + 1) % 3],
                dp[i-1][(j + 2) % 3]
            )
        
    # Handle the equidistant constraint
    # We have two cases: start with color 1 or color 2 for the last house
    min_cost = float('inf')
    for start_color in range(3):
        last_house_cost = 0
        for j in range(3):
            if j == start_color:
                continue
            last_house_cost = min(last_house_cost, dp[n-1][j])
        min_cost = min(min_cost, last_house_cost)
    
    return min_cost
```
