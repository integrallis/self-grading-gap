def test_canonical_example_1():
    # spec: nums = [2,3,1]
    # i=0: start=max(0,0-2)=0, subarray=[2], sum=2
    # i=1: start=max(0,1-3)=0, subarray=[2,3], sum=5
    # i=2: start=max(0,2-1)=1, subarray=[3,1], sum=4
    # total = 2 + 5 + 4 = 11
    assert subarraySum([2, 3, 1]) == 11
