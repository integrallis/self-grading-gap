from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Calculate contributions of each number in nums
    for i in range(n):
        # Each number nums[i] can be the minimum or maximum in subsequences
        # Count how many times it can be a minimum or maximum
        min_contributions = 0
        max_contributions = 0

        # Calculate contribution of nums[i] as min
        if k > 1:
            min_contributions = pow(2, min(k - 1, n - 1), MOD) * pow(2, n - k, MOD)
        else:
            min_contributions = 1

        # Calculate contribution of nums[i] as max
        if k > 1:
            max_contributions = pow(2, min(k - 1, n - 1), MOD) * pow(2, n - k, MOD)
        else:
            max_contributions = 1

        total_sum += (min_contributions + max_contributions) * nums[i] % MOD
        total_sum %= MOD

    return total_sum
