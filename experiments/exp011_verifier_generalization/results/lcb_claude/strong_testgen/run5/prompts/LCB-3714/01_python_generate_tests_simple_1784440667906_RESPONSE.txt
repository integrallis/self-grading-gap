from solution import minMaxSums
from typing import List


def test_canonical_example_1():
    # spec: nums = [1,2,3], k = 2 -> 24
    # Subsequences with at most 2 elements:
    # [1]: min=1, max=1, sum=2
    # [2]: min=2, max=2, sum=4
    # [3]: min=3, max=3, sum=6
    # [1,2]: min=1, max=2, sum=3
    # [1,3]: min=1, max=3, sum=4
    # [2,3]: min=2, max=3, sum=5
    # Total: 2+4+6+3+4+5 = 24
    assert minMaxSums([1, 2, 3], 2) == 24


def test_canonical_example_2():
    # spec: nums = [5,0,6], k = 1 -> 22
    # Subsequences with at most 1 element:
    # [5]: min=5, max=5, sum=10
    # [0]: min=0, max=0, sum=0
    # [6]: min=6, max=6, sum=12
    # Total: 10+0+12 = 22
    assert minMaxSums([5, 0, 6], 1) == 22


def test_canonical_example_3():
    # spec: nums = [1,1,1], k = 2 -> 12
    # Subsequences with at most 2 elements:
    # [1] appears 3 times: min=1, max=1, sum=2 each, total=6
    # [1,1] appears 3 times: min=1, max=1, sum=2 each, total=6
    # Total: 6+6 = 12
    assert minMaxSums([1, 1, 1], 2) == 12


def test_single_element_array():
    # nums = [5], k = 1
    # Only subsequence: [5], min=5, max=5, sum=10
    assert minMaxSums([5], 1) == 10


def test_k_equals_array_length():
    # nums = [1, 2], k = 2
    # Subsequences with at most 2 elements:
    # [1]: min=1, max=1, sum=2
    # [2]: min=2, max=2, sum=4
    # [1,2]: min=1, max=2, sum=3
    # Total: 2+4+3 = 9
    assert minMaxSums([1, 2], 2) == 9


def test_all_zeros():
    # nums = [0, 0], k = 2
    # Subsequences:
    # [0] appears 2 times: sum=0 each, total=0
    # [0,0]: min=0, max=0, sum=0
    # Total: 0
    assert minMaxSums([0, 0], 2) == 0


def test_large_values_modulo():
    # Test with values that might require modulo operation
    # nums = [1000000000, 1000000000], k = 1
    # [1000000000] appears 2 times: sum=2000000000 each, total=4000000000
    # 4000000000 % (10^9 + 7) = 4000000000 % 1000000007 = 2999999986
    assert minMaxSums([1000000000, 1000000000], 1) == 2999999986

    # nums = [3, 7], k = 1

    # Total: 20
    assert minMaxSums([3, 7], 1) == 20


def test_distinct_elements():
    # nums = [1, 2, 3, 4], k = 1
    # Each element forms a subsequence where min=max=element
    # [1]: sum=2, [2]: sum=4, [3]: sum=6, [4]: sum=8
    # Total: 20
    assert minMaxSums([1, 2, 3, 4], 1) == 20
