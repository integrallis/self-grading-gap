from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    # dp[neutralizations][i][j] represents the maximum coins collected to (i, j) with 'neutralizations' used
    dp = [[[-float('inf')] * n for _ in range(m)] for _ in range(3)]
    dp[0][0][0] = coins[0][0]  # Starting point

    for i in range(m):
        for j in range(n):
            for k in range(3):  # k represents the number of neutralizations used
                if i == 0 and j == 0:
                    continue  # Skip the starting cell as it's already initialized
                # Compute the maximum coins coming from the left and above
                for ni, nj in [(i - 1, j), (i, j - 1)]:
                    if 0 <= ni < m and 0 <= nj < n:
                        # Normal gain/loss from the cell
                        current = dp[k][ni][nj]
                        if current != -float('inf'):
                            if coins[i][j] >= 0:
                                dp[k][i][j] = max(dp[k][i][j], current + coins[i][j])
                            else:
                                # Robber scenario
                                if k > 0:
                                    # Use a neutralization
                                    dp[k][i][j] = max(dp[k][i][j], current)
                                # Without neutralization
                                dp[k][i][j] = max(dp[k][i][j], current + coins[i][j])

    # Find the maximum coins collected at the bottom-right corner with at most 2 neutralizations
    return max(dp[k][m - 1][n - 1] for k in range(3))
```
