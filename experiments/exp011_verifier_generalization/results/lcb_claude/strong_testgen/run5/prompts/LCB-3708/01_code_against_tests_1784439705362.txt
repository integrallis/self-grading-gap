## Implementation

```python
from typing import List

def zigzagTraversal(grid: List[List[int]]) -> List[int]:
    result = []
    rows = len(grid)
    for i in range(rows):
        if i % 2 == 0:
            # Traverse left to right
            result.extend(grid[i][j] for j in range(len(grid[i])) if j % 2 == 0)
        else:
            # Traverse right to left
            result.extend(grid[i][j] for j in range(len(grid[i])-1, -1, -1) if j % 2 == 0)
    return result
```
