import pytest
from solution import hasMatch


def test_canonical_example_1():
    # spec: s = "leetcode", p = "ee*e" -> true
    # By replacing '*' with "tcod", "eetcode" is a substring of "leetcode"
    assert hasMatch("leetcode", "ee*e") is True


def test_canonical_example_2():
    # spec: s = "car", p = "c*v" -> false
    # No substring matches the pattern
    assert hasMatch("car", "c*v") is False


def test_canonical_example_3():
    # spec: s = "luck", p = "u*" -> true
    # Substrings "u", "uc", "uck" match the pattern
    assert hasMatch("luck", "u*") is True


def test_star_at_beginning():
    # p = "*a" means any characters followed by 'a'
    # "ba" is substring of "abc" matching pattern
    assert hasMatch("abc", "*a") is True


def test_star_at_end():
    # p = "a*" means 'a' followed by any characters
    # "ab" is substring of "abc" matching pattern
    assert hasMatch("abc", "a*") is True


def test_star_in_middle():
    # p = "a*c" means 'a' followed by any characters followed by 'c'
    # "abc" is substring of "abc" matching pattern
    assert hasMatch("abc", "a*c") is True


def test_only_star():
    # p = "*" means any sequence of characters (including zero)
    # Any non-empty substring matches
    assert hasMatch("a", "*") is True


def test_star_zero_replacement():
    # p = "ab*cd" where '*' replaced with zero characters gives "abcd"
    # "abcd" is substring of "abcd"
    assert hasMatch("abcd", "ab*cd") is True


def test_star_multiple_chars_replacement():
    # p = "a*z" where '*' replaced with "bcdefgh" gives "abcdefghz"
    # "abcdefghz" is substring of "abcdefghz"
    assert hasMatch("abcdefghz", "a*z") is True


def test_pattern_longer_than_string_impossible():
    # p = "abc*def" requires at least "abcdef" (6 chars)
    # s = "ab" has only 2 chars, no substring can match
    assert hasMatch("ab", "abc*def") is False


def test_prefix_not_in_string():
    # p = "xy*" requires substring starting with "xy"
    # s = "abc" has no such substring
    assert hasMatch("abc", "xy*") is False


def test_suffix_not_in_string():
    # p = "*xy" requires substring ending with "xy"
    # s = "abc" has no such substring
    assert hasMatch("abc", "*xy") is False


def test_both_prefix_and_suffix_not_matching():
    # p = "x*y" requires substring starting with 'x' and ending with 'y'
    # s = "abc" has neither
    assert hasMatch("abc", "x*y") is False


def test_prefix_exists_but_suffix_not_after_it():
    # p = "a*x" requires 'a' followed by something ending with 'x'
    # s = "abc" has 'a' but no 'x' after it
    assert hasMatch("abc", "a*x") is False


def test_multiple_occurrences_first_works():
    # p = "e*e" can match "ee", "eetcode", etc.
    # s = "leetcode" contains "eetcode" matching pattern
    assert hasMatch("leetcode", "e*e") is True


def test_pattern_at_string_start():
    # p = "le*" matches substrings starting with "le"
    # s = "leetcode" starts with "le", so "le", "lee", etc. match
    assert hasMatch("leetcode", "le*") is True


def test_pattern_at_string_end():
    # p = "*de" matches substrings ending with "de"
    # s = "leetcode" ends with "de", so "code", "tcode", etc. match
    assert hasMatch("leetcode", "*de") is True


def test_exact_match_with_star_zero():
    # p = "a*b" with '*' as zero gives "ab"
    # s = "ab" has "ab" as substring
    assert hasMatch("ab", "a*b") is True


def test_single_char_string_single_char_pattern_with_star():
    # p = "a*" matches "a" (star replaced with zero chars)
    # s = "a" contains "a"
    assert hasMatch("a", "a*") is True


def test_single_char_pattern_star_only():
    # p = "*" can match any non-empty substring
    # s = "x" has substring "x"
