def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]
    prefix_length = len(prefix)
    suffix_length = len(suffix)

    # Iterate through possible starting positions for the prefix in s
    for i in range(len(s) - prefix_length - suffix_length + 1):
        # Check if the substring from s matches the prefix
        if s[i:i + prefix_length] == prefix:
            # Now check if the remaining part matches the suffix
            for j in range(i + prefix_length, len(s) + 1):
                if s[j - len(suffix):j] == suffix:
                    return True
    return False
```
