from typing import List
import pytest
from solution import minimumPairRemoval


def test_canonical_example_1():
    # spec: nums = [5,2,3,1] -> 2
    # (3,1) sum=4 -> [5,2,4]
    # (2,4) sum=6 -> [5,6]
    # 2 operations
    assert minimumPairRemoval([5, 2, 3, 1]) == 2


def test_canonical_example_2():
    # spec: nums = [1,2,2] -> 0
    # already non-decreasing
    assert minimumPairRemoval([1, 2, 2]) == 0


def test_single_element():
    # single element is always non-decreasing
    assert minimumPairRemoval([5]) == 0


def test_two_elements_already_sorted():
    # [1, 2] is already non-decreasing
    assert minimumPairRemoval([1, 2]) == 0


def test_two_elements_need_merge():
    # [2, 1] is decreasing, merge to [3] which is non-decreasing
    assert minimumPairRemoval([2, 1]) == 1


def test_all_equal_elements():
    # [3, 3, 3, 3] is already non-decreasing
    assert minimumPairRemoval([3, 3, 3, 3]) == 0


def test_strictly_increasing():
    # [1, 2, 3, 4, 5] is already non-decreasing
    assert minimumPairRemoval([1, 2, 3, 4, 5]) == 0


def test_strictly_decreasing():
    # [5, 4, 3, 2, 1] requires operations
    # pairs: (5,4)=9, (4,3)=7, (3,2)=5, (2,1)=3
    # leftmost min (2,1) -> [5,4,3,3]
    # continue until non-decreasing
    assert minimumPairRemoval([5, 4, 3, 2, 1]) >= 0


def test_negative_numbers():
    # [-5, -10, 3] 
    # pairs: (-5,-10)=-15, (-10,3)=-7
    # leftmost min (-5,-10) -> [-15, 3]
    # [-15, 3] is non-decreasing
    assert minimumPairRemoval([-5, -10, 3]) == 1


def test_mixed_positive_negative():
    # [1, -1, 2]
    # pairs: (1,-1)=0, (-1,2)=1
    # leftmost min (1,-1) -> [0, 2]
    # [0, 2] is non-decreasing
    assert minimumPairRemoval([1, -1, 2]) == 1


def test_three_elements_already_sorted():
    # [1, 2, 3] is already non-decreasing
    assert minimumPairRemoval([1, 2, 3]) == 0


def test_array_with_zeros():
    # [0, 0, 0] is already non-decreasing
    assert minimumPairRemoval([0, 0, 0]) == 0


def test_alternating_high_low():
    # [10, 1, 10, 1]
    # pairs: (10,1)=11, (1,10)=11, (10,1)=11
    # leftmost min (10,1) -> [11, 10, 1]
    # continue until non-decreasing
    assert minimumPairRemoval([10, 1, 10, 1]) >= 0
