```python
from typing import List
import heapq


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Create adjacency list with weights
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[u].append((v, w))

    # Function to check if we can reach node 0 with max edge weight allowed
    def can_reach_zero(max_weight):
        out_degree = [0] * n
        visited = [False] * n
        queue = []

        # Start from all nodes that can reach 0
        for node in range(n):
            for neighbor, weight in graph[node]:
                if weight <= max_weight:
                    out_degree[node] += 1
                    if out_degree[node] <= threshold:
                        queue.append(node)
                        visited[node] = True

        # BFS to check reachability
        while queue:
            current = queue.pop(0)
            for neighbor, weight in graph[current]:
                if weight <= max_weight and not visited[neighbor]:
                    visited[neighbor] = True
                    queue.append(neighbor)

        return visited[0]  # Check if node 0 is reachable

    # Binary search for the minimum maximum weight
    left, right = 1, 10**6
    result = -1
    while left <= right:
        mid = (left + right) // 2
        if can_reach_zero(mid):
            result = mid
            right = mid - 1
        else:
            left = mid + 1

    return result
```
