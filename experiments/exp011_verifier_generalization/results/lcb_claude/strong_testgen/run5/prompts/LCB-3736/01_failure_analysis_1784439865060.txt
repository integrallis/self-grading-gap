def findValidPair(s: str) -> str:
    # Count occurrences of each digit
    count = {str(i): 0 for i in range(1, 10)}
    for char in s:
        count[char] += 1

    # Traverse the string to find valid pairs
    for i in range(len(s) - 1):
        a, b = s[i], s[i + 1]
        if a != b and count[a] == int(a) and count[b] == int(b):
            return a + b

    return ""
