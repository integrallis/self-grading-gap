from solution import maxAdjacentDistance


def test_canonical_example_1():
    # spec: nums = [1,2,4] -> 3
    # Adjacent pairs: (1,2), (2,4), (4,1)
    # Differences: |1-2|=1, |2-4|=2, |4-1|=3
    # Maximum = 3
    assert maxAdjacentDistance([1, 2, 4]) == 3


def test_canonical_example_2():
    # spec: nums = [-5,-10,-5] -> 5
    # Adjacent pairs: (-5,-10), (-10,-5), (-5,-5)
    # Differences: |-5-(-10)|=5, |-10-(-5)|=5, |-5-(-5)|=0
    # Maximum = 5
    assert maxAdjacentDistance([-5, -10, -5]) == 5


def test_two_elements_positive():
    # nums = [1, 5]
    # Adjacent pairs: (1,5), (5,1)
    # Both give |1-5| = 4
    assert maxAdjacentDistance([1, 5]) == 4


def test_two_elements_negative():
    # nums = [-3, -7]
    # Adjacent pairs: (-3,-7), (-7,-3)
    # Both give |-3-(-7)| = 4
    assert maxAdjacentDistance([-3, -7]) == 4


def test_all_same_elements():
    # nums = [5, 5, 5, 5]
    # All adjacent differences are 0
    assert maxAdjacentDistance([5, 5, 5, 5]) == 0


def test_increasing_sequence():
    # nums = [1, 2, 3, 4]
    # Adjacent pairs: (1,2), (2,3), (3,4), (4,1)
    # Differences: 1, 1, 1, 3
    # Maximum = 3 (circular connection)
    assert maxAdjacentDistance([1, 2, 3, 4]) == 3


def test_mixed_positive_negative():
    # nums = [10, -10, 20]
    # Adjacent pairs: (10,-10), (-10,20), (20,10)
    # Differences: |10-(-10)|=20, |-10-20|=30, |20-10|=10
    # Maximum = 30
    assert maxAdjacentDistance([10, -10, 20]) == 30


def test_maximum_at_circular_boundary():
    # nums = [100, 1, 2, 3]
    # Adjacent pairs: (100,1), (1,2), (2,3), (3,100)
    # Differences: 99, 1, 1, 97
    # Maximum = 99
    assert maxAdjacentDistance([100, 1, 2, 3]) == 99


def test_negative_values_only():
    # nums = [-1, -50, -25, -75]
    # Adjacent pairs: (-1,-50), (-50,-25), (-25,-75), (-75,-1)
    # Differences: 49, 25, 50, 74
    # Maximum = 74
    assert maxAdjacentDistance([-1, -50, -25, -75]) == 74
