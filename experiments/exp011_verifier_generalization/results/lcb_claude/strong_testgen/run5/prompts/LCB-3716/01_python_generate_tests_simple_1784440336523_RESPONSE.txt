from typing import List
import pytest
from solution import longestSubsequence


def test_canonical_example_1():
    # spec: nums = [16,6,3] -> 3
    # The longest subsequence is [16, 6, 3] with absolute adjacent differences [10, 3]
    # 10 >= 3, so this is valid, length = 3
    assert longestSubsequence([16, 6, 3]) == 3


def test_canonical_example_2():
    # spec: nums = [6,5,3,4,2,1] -> 4
    # The longest subsequence is [6, 4, 2, 1] with absolute adjacent differences [2, 2, 1]
    # 2 >= 2 >= 1, so this is valid, length = 4
    assert longestSubsequence([6, 5, 3, 4, 2, 1]) == 4


def test_canonical_example_3():
    # spec: nums = [10,20,10,19,10,20] -> 5
    # The longest subsequence is [10, 20, 10, 19, 10] with absolute adjacent differences [10, 10, 9, 9]
    # 10 >= 10 >= 9 >= 9, so this is valid, length = 5
    assert longestSubsequence([10, 20, 10, 19, 10, 20]) == 5


def test_minimum_length_array():
    # spec: 2 <= nums.length, so minimum is 2 elements
    # Any 2-element array forms a valid subsequence of length at least 2
    # (single difference, trivially non-increasing)
    assert longestSubsequence([1, 2]) == 2


def test_all_same_elements():
    # All elements the same means all differences are 0
    # [5, 5, 5] -> differences [0, 0], which is non-increasing (0 >= 0)
    # Full subsequence of length 3
    assert longestSubsequence([5, 5, 5]) == 3


def test_strictly_increasing_array():
    # [1, 2, 3, 4] -> differences [1, 1, 1] if we take all
    # 1 >= 1 >= 1 is valid (non-increasing)
    # Should return at least 4
    assert longestSubsequence([1, 2, 3, 4]) == 4


def test_strictly_decreasing_array():
    # [4, 3, 2, 1] -> differences [1, 1, 1] if we take all
    # 1 >= 1 >= 1 is valid (non-increasing)
    # Should return at least 4
    assert longestSubsequence([4, 3, 2, 1]) == 4


def test_large_then_small_differences():
    # [1, 100, 101] -> differences [99, 1]
    # 99 >= 1, so full subsequence of length 3 is valid
    assert longestSubsequence([1, 100, 101]) == 3


def test_small_then_large_differences():
    # [100, 101, 1] -> differences [1, 100]
    # 1 >= 100 is false, so we can't take all three
    # Best might be just [100, 101] with difference [1], length 2
    # or [100, 1] with difference [99], length 2
    # or [101, 1] with difference [100], length 2
    assert longestSubsequence([100, 101, 1]) >= 2


def test_alternating_values():
    # [1, 10, 1, 10] -> if we take all, differences [9, 9, 9]
    # 9 >= 9 >= 9 is valid, length 4
    assert longestSubsequence([1, 10, 1, 10]) == 4


def test_boundary_values():
    # spec: 1 <= nums[i] <= 300
    # Test with boundary values
    assert longestSubsequence([1, 300]) == 2
    assert longestSubsequence([300, 1]) == 2
