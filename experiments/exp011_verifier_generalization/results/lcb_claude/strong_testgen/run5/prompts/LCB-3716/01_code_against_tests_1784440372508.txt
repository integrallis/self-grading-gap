from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    # Array to store the length of longest subsequence at each index
    dp = [1] * n
    # Variable to maintain the maximum length of valid subsequences
    max_length = 1

    # Iterate through the array from second element to the end
    for i in range(1, n):
        # We need to find the best subsequence length that can end with nums[i]
        for j in range(i):
            # Compute the absolute difference between current and previous element
            diff1 = abs(nums[i] - nums[j])
            if j > 0:
                # Previous difference in the valid subsequence
                diff2 = abs(nums[j] - nums[j - 1])
            else:
                # If this is the first element being compared, treat prev difference as inf
                diff2 = float('inf')
            # Check if the current difference is less than or equal to the previous difference
            if diff1 <= diff2:
                dp[i] = max(dp[i], dp[j] + 1)
        # Update the maximum length found
        max_length = max(max_length, dp[i])

    return max_length
