import pytest
from solution import maxDistance


def test_canonical_example_1():
    # spec: s = "NWSE", k = 1 -> 3
    # Change s[2] from 'S' to 'N' to get "NWNE"
    # Positions: (0,1), (-1,1), (-1,2), (0,2) with distances 1, 2, 3, 2
    # Maximum distance is 3
    assert maxDistance("NWSE", 1) == 3


def test_canonical_example_2():
    # spec: s = "NSWWEW", k = 3 -> 6
    # Change s[1] from 'S' to 'N' and s[4] from 'E' to 'W' to get "NNWWWW"
    # Maximum distance is 6
    assert maxDistance("NSWWEW", 3) == 6


def test_single_move_north_no_changes():
    # s = "N", k = 0
    # Position after move: (0, 1), distance = 1
    assert maxDistance("N", 0) == 1


def test_single_move_south_no_changes():
    # s = "S", k = 0
    # Position after move: (0, -1), distance = 1
    assert maxDistance("S", 0) == 1


def test_single_move_east_no_changes():
    # s = "E", k = 0
    # Position after move: (1, 0), distance = 1
    assert maxDistance("E", 0) == 1


def test_single_move_west_no_changes():
    # s = "W", k = 0
    # Position after move: (-1, 0), distance = 1
    assert maxDistance("W", 0) == 1


def test_opposite_moves_no_changes():
    # s = "NS", k = 0
    # Positions: (0,1), (0,0) with distances 1, 0
    # Maximum is 1
    assert maxDistance("NS", 0) == 1


def test_all_changes_allowed():
    # s = "SSSS", k = 4
    # Can change all to same direction to maximize distance
    # If all are N: (0,1), (0,2), (0,3), (0,4) -> max distance 4
    assert maxDistance("SSSS", 4) == 4


def test_no_changes_allowed():
    # s = "NNNN", k = 0
    # Positions: (0,1), (0,2), (0,3), (0,4) with distances 1, 2, 3, 4
    # Maximum is 4
    assert maxDistance("NNNN", 0) == 4


def test_return_to_origin():
    # s = "NESW", k = 0
    # Positions: (0,1), (1,1), (1,0), (0,0) with distances 1, 2, 1, 0
    # Maximum is 2
    assert maxDistance("NESW", 0) == 2


def test_k_equals_length():
    # s = "EW", k = 2
    # Can change all moves, best is to go in one direction
    # Change to "EE": (1,0), (2,0) -> max distance 2
    assert maxDistance("EW", 2) == 2


def test_zigzag_pattern():
    # s = "NESE", k = 0
    # Positions: (0,1), (1,1), (1,0), (2,0) with distances 1, 2, 1, 2
    # Maximum is 2
    assert maxDistance("NESE", 0) == 2


def test_one_change_extends_path():
    # s = "NNS", k = 1
    # Change S to N: "NNN"
    # Positions: (0,1), (0,2), (0,3) with distances 1, 2, 3
    # Maximum is 3
    assert maxDistance("NNS", 1) == 3


def test_large_k_with_short_string():
    # s = "N", k = 10
    # k > length but still only 1 move can be changed
    # Best is to keep or change to any direction, distance = 1
    assert maxDistance("N", 10) == 1


def test_two_moves_one_change():
    # s = "EW", k = 1
    # Change W to E: "EE"
    # Positions: (1,0), (2,0) with distances 1, 2
    # Maximum is 2
    assert maxDistance("EW", 1) == 2
