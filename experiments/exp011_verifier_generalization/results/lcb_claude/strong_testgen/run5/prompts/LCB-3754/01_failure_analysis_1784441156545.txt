```python
def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    moves = []
    for move in s:
        if move == 'N':
            y += 1
        elif move == 'S':
            y -= 1
        elif move == 'E':
            x += 1
        elif move == 'W':
            x -= 1
        moves.append((x, y))
    max_distance = max(abs(x), abs(y))  # Maximum distance without changes
    # Now consider using k changes optimally
    total_moves = len(s)
    distance_with_changes = max_distance + min(k, total_moves)  # Total moves possible + changes available
    return distance_with_changes
```
