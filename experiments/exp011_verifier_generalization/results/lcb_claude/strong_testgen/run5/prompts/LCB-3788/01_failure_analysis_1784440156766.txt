from typing import List

def maxSum(nums: List[int]) -> int:
    unique_nums = set()
    max_sum = float('-inf')
    current_sum = 0
    for num in nums:
        if num not in unique_nums:
            unique_nums.add(num)
            current_sum += num
        else:
            max_sum = max(max_sum, current_sum)
            while num in unique_nums:
                unique_nums.remove(nums[len(unique_nums) - 1])
                current_sum -= nums[len(unique_nums) - 1]
            unique_nums.add(num)
            current_sum += num
    return max(max_sum, current_sum) if max_sum != float('-inf') else current_sum
