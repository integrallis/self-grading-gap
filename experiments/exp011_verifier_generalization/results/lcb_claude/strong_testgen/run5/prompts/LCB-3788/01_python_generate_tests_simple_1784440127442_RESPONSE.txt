import pytest
from typing import List
from solution import maxSum


def test_canonical_example_1():
    # spec: nums = [1,2,3,4,5] -> 15
    # All elements are unique, sum = 1+2+3+4+5 = 15
    assert maxSum([1, 2, 3, 4, 5]) == 15


def test_canonical_example_2():
    # spec: nums = [1,1,0,1,1] -> 1
    # After deletions, select subarray [1] with sum = 1
    assert maxSum([1, 1, 0, 1, 1]) == 1


def test_canonical_example_3():
    # spec: nums = [1,2,-1,-2,1,0,-1] -> 3
    # Delete some elements, select subarray [2, 1] with sum = 2+1 = 3
    assert maxSum([1, 2, -1, -2, 1, 0, -1]) == 3


def test_single_element():
    # Single element array, must return that element
    # sum = 5
    assert maxSum([5]) == 5


def test_single_element_negative():
    # Single element array with negative value
    # sum = -10
    assert maxSum([-10]) == -10


def test_all_same_elements():
    # All elements are the same, can only select one
    # sum = 7
    assert maxSum([7, 7, 7, 7]) == 7


def test_two_unique_elements():
    # Two different elements, select both
    # sum = 3 + 5 = 8
    assert maxSum([3, 5]) == 8


def test_two_same_elements():
    # Two same elements, select one
    # sum = 4
    assert maxSum([4, 4]) == 4


def test_all_negative():
    # All negative elements, select the largest (least negative)
    # sum = -1
    assert maxSum([-5, -3, -1, -4]) == -1


def test_mix_positive_negative_with_duplicates():
    # Mix of positive and negative with duplicates
    # After deletions, can select [10, 20] with sum = 30
    assert maxSum([10, -5, 20, 10, -5]) == 30


def test_alternating_duplicates():
    # Pattern with alternating duplicates
    # Can select subarray [2, 3] with sum = 5
    assert maxSum([2, 2, 3, 3]) == 5


def test_negative_zero_positive():
    # Mix with zero
    # Can select [5, 10] with sum = 15
    assert maxSum([-5, 0, 5, 10]) == 15


def test_descending_order():
    # Descending unique elements
    # sum = 5+4+3+2+1 = 15
    assert maxSum([5, 4, 3, 2, 1]) == 15


def test_many_duplicates_single_unique():
    # Many duplicates of different values
    # Best is to select [100] with sum = 100
    assert maxSum([1, 1, 1, 100, 2, 2, 2]) == 100


def test_optimal_subarray_in_middle():
    # Optimal subarray is in the middle after deletions
    # Can select [10, 20, 30] with sum = 60
    assert maxSum([1, 10, 20, 30, 1]) == 60


def test_negative_before_positive():
    # Negative values before positive
    # Select [1, 2, 3] with sum = 6
    assert maxSum([-10, -20, 1, 2, 3]) == 6


def test_zero_in_middle():
    # Zero in the middle of positive values
    # Select [1, 2, 0, 3] if all unique or best subarray
    # sum = 1+2+3 = 6 (selecting [1,2,3] after deleting 0)
    assert maxSum([1, 2, 0, 3]) == 6


def test_large_range():
    # Elements at constraint boundaries
    # sum = 100 + 50 = 150
    assert maxSum([100, -100, 50]) == 150


def test_single_positive_among_negatives():
    # Single positive value among negatives
    # sum = 10
    assert maxSum([-1, -2, 10, -3, -4]) == 10


def test_duplicate_at_ends():
    # Duplicates at both ends
    # Select middle part [2, 3] with sum = 5
    assert maxSum([1, 2, 3, 1]) == 5
