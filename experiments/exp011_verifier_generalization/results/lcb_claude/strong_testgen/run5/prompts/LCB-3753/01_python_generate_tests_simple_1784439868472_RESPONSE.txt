import pytest
from solution import maxDifference


def test_canonical_example_1():
    # spec: s = "aaaaabbc"
    # 'a' has frequency 5 (odd), 'b' has frequency 2 (even), 'c' has frequency 1 (odd)
    # max difference = max(odd) - min(even) = 5 - 2 = 3
    assert maxDifference("aaaaabbc") == 3


def test_canonical_example_2():
    # spec: s = "abcabcab"
    # 'a' has frequency 3 (odd), 'b' has frequency 3 (odd), 'c' has frequency 2 (even)
    # max difference = max(odd) - min(even) = 3 - 2 = 1
    assert maxDifference("abcabcab") == 1


def test_single_char_odd_single_char_even():
    # s = "aabb"
    # 'a' has frequency 2 (even), 'b' has frequency 2 (even)
    # This violates constraint, but s = "aaab" works
    # 'a' has frequency 3 (odd), 'b' has frequency 1 (odd)
    # This also violates constraint. s = "aaabb" works
    # 'a' has frequency 3 (odd), 'b' has frequency 2 (even)
    # max difference = 3 - 2 = 1
    assert maxDifference("aaabb") == 1


def test_max_odd_min_even():
    # s = "aaaaaabb"
    # 'a' has frequency 6 (even), 'b' has frequency 2 (even)
    # Need odd, so s = "aaaaaaabb"
    # 'a' has frequency 7 (odd), 'b' has frequency 2 (even)
    # max difference = 7 - 2 = 5
    assert maxDifference("aaaaaaabb") == 5


def test_multiple_odd_multiple_even():
    # s = "aabbccc"
    # 'a' has frequency 2 (even), 'b' has frequency 2 (even), 'c' has frequency 3 (odd)
    # max difference = 3 - 2 = 1
    assert maxDifference("aabbccc") == 1


def test_minimum_length_string():
    # s = "aab"
    # 'a' has frequency 2 (even), 'b' has frequency 1 (odd)
    # max difference = 1 - 2 = -1
    assert maxDifference("aab") == -1


def test_large_odd_small_even():
    # s = "aaaaaaaaab"
    # 'a' has frequency 9 (odd), 'b' has frequency 1 (odd)
    # Need even, so s = "aaaaaaaabb"
    # 'a' has frequency 8 (even), 'b' has frequency 2 (even)
    # Need odd, so s = "aaaaaaabbb"
    # 'a' has frequency 7 (odd), 'b' has frequency 3 (odd)
    # Need even, so s = "aaaaaaaabbb" doesn't work. Try "aaaaaaaaabbc"
    # 'a' has frequency 9 (odd), 'b' has frequency 2 (even), 'c' has frequency 1 (odd)
    # max difference = 9 - 2 = 7
    assert maxDifference("aaaaaaaaabbc") == 7


def test_even_larger_than_odd():
    # s = "aaaaabbbbbbbbbbc"
    # 'a' has frequency 5 (odd), 'b' has frequency 10 (even), 'c' has frequency 1 (odd)
    # max difference = max(odd) - min(even) = 5 - 10 = -5
    assert maxDifference("aaaaabbbbbbbbbbc") == -5


def test_all_different_chars():
    # s = "abcdef"
    # All have frequency 1 (odd), need at least one even
    # s = "aabcdef"
    # 'a' has frequency 2 (even), rest have frequency 1 (odd)
    # max difference = 1 - 2 = -1
    assert maxDifference("aabcdef") == -1


def test_three_chars_varied_frequencies():
    # s = "aaabbbcc"
    # 'a' has frequency 3 (odd), 'b' has frequency 3 (odd), 'c' has frequency 2 (even)
    # max difference = 3 - 2 = 1
    assert maxDifference("aaabbbcc") == 1
