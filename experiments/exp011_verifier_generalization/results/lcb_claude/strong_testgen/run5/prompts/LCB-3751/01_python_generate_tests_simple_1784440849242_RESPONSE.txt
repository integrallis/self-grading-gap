from typing import List
import pytest
from solution import maxFrequency


def test_canonical_example_1():
    # spec: nums = [1,2,3,4,5,6], k = 1
    # After adding -5 to nums[2..5], 1 has a frequency of 2 in [1, 2, -2, -1, 0, 1]
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2


def test_canonical_example_2():
    # spec: nums = [10,2,3,4,5,5,4,3,2,2], k = 10
    # After adding 8 to nums[1..9], 10 has a frequency of 4 in [10, 10, 11, 12, 13, 13, 12, 11, 10, 10]
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4


def test_single_element_array_already_equal_to_k():
    # nums = [5], k = 5
    # Already has frequency 1 of value 5, no operation needed
    assert maxFrequency([5], 5) == 1


def test_single_element_array_needs_transformation():
    # nums = [3], k = 7
    # Add 4 to nums[0..0] to get [7], frequency of 7 is 1
    assert maxFrequency([3], 7) == 1


def test_all_elements_already_equal_to_k():
    # nums = [5, 5, 5], k = 5
    # Already has frequency 3 of value 5
    assert maxFrequency([5, 5, 5], 5) == 3


def test_transform_entire_array_to_k():
    # nums = [1, 1, 1], k = 10
    # Add 9 to nums[0..2] to get [10, 10, 10], frequency of 10 is 3
    assert maxFrequency([1, 1, 1], 10) == 3


def test_some_elements_already_k_transform_others():
    # nums = [5, 3, 5, 3], k = 5
    # Already has 2 occurrences of 5, add 2 to nums[1..1] to get [5, 5, 5, 3], frequency is 3
    # Or add 2 to nums[3..3] to get [5, 3, 5, 5], frequency is 3
    assert maxFrequency([5, 3, 5, 3], 5) == 3


def test_no_initial_k_transform_subarray():
    # nums = [2, 4, 6], k = 10
    # Add 8 to nums[0..0] to get [10, 4, 6], frequency of 10 is 1
    # Add 6 to nums[1..1] to get [2, 10, 6], frequency of 10 is 1
    # Add 4 to nums[2..2] to get [2, 4, 10], frequency of 10 is 1
    # Add 8 to nums[0..1] to get [10, 12, 6], frequency of 10 is 1
    # Add 6 to nums[1..2] to get [2, 10, 12], frequency of 10 is 1
    # Add 4 to nums[0..2] to get [6, 8, 10], frequency of 10 is 1
    assert maxFrequency([2, 4, 6], 10) == 1


def test_transform_middle_subarray():
    # nums = [10, 5, 10], k = 10
    # Already has 2 occurrences of 10, add 5 to nums[1..1] to get [10, 10, 10], frequency is 3
    assert maxFrequency([10, 5, 10], 10) == 3


def test_multiple_subarrays_possible():
    # nums = [1, 2, 1, 2], k = 1
    # Already has 2 occurrences of 1
    # Add -1 to nums[1..1] to get [1, 1, 1, 2], frequency is 3
    # Add -1 to nums[3..3] to get [1, 2, 1, 1], frequency is 3
    assert maxFrequency([1, 2, 1, 2], 1) == 3


def test_larger_array_with_mixed_values():
    # nums = [1, 1, 3, 3, 3, 5, 5], k = 3
    # Already has 3 occurrences of 3
    # Add 2 to nums[0..1] to get [3, 3, 3, 3, 3, 5, 5], frequency is 5
    assert maxFrequency([1, 1, 3, 3, 3, 5, 5], 3) == 5


def test_two_element_array():
    # nums = [1, 2], k = 1
    # Already has 1 occurrence of 1
    # Add -1 to nums[1..1] to get [1, 1], frequency is 2
    assert maxFrequency([1, 2], 1) == 2


def test_non_contiguous_k_values():
    # nums = [5, 1, 5, 1, 5], k = 5
    # Already has 3 occurrences of 5
    # Add 4 to nums[1..1] to get [5, 5, 5, 1, 5], frequency is 4
    # Add 4 to nums[3..3] to get [5, 1, 5, 5, 5], frequency is 4
    # Add 4 to nums[1..3] to get [5, 5, 9, 5, 5], frequency is 4
    assert maxFrequency([5, 1, 5, 1, 5], 5) == 4
