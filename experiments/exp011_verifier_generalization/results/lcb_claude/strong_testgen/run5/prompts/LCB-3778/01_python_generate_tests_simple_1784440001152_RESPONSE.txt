from typing import List
import pytest
from solution import transformArray


def test_canonical_example_1():
    # spec: [4,3,2,1] -> [0,0,1,1]
    # Even numbers (4,2) become 0, odd numbers (3,1) become 1 -> [0,1,0,1]
    # After sorting: [0,0,1,1]
    assert transformArray([4, 3, 2, 1]) == [0, 0, 1, 1]


def test_canonical_example_2():
    # spec: [1,5,1,4,2] -> [0,0,1,1,1]
    # Even numbers (4,2) become 0, odd numbers (1,5,1) become 1 -> [1,1,1,0,0]
    # After sorting: [0,0,1,1,1]
    assert transformArray([1, 5, 1, 4, 2]) == [0, 0, 1, 1, 1]


def test_single_element_even():
    # Single even number becomes 0
    assert transformArray([2]) == [0]


def test_single_element_odd():
    # Single odd number becomes 1
    assert transformArray([1]) == [1]


def test_all_even_numbers():
    # All even numbers [2,4,6] -> [0,0,0]
    assert transformArray([2, 4, 6]) == [0, 0, 0]


def test_all_odd_numbers():
    # All odd numbers [1,3,5] -> [1,1,1]
    assert transformArray([1, 3, 5]) == [1, 1, 1]


def test_mixed_already_sorted():
    # [2,3,4,5] -> [0,1,0,1] -> sorted [0,0,1,1]
    assert transformArray([2, 3, 4, 5]) == [0, 0, 1, 1]


def test_larger_numbers():
    # Testing with numbers in the constraint range (up to 1000)
    # [1000,999,500,501] -> [0,1,0,1] -> sorted [0,0,1,1]
    assert transformArray([1000, 999, 500, 501]) == [0, 0, 1, 1]


def test_multiple_same_values():
    # [3,3,3,2,2,2] -> [1,1,1,0,0,0] -> sorted [0,0,0,1,1,1]
    assert transformArray([3, 3, 3, 2, 2, 2]) == [0, 0, 0, 1, 1, 1]


def test_more_odds_than_evens():
    # [1,3,5,7,2] -> [1,1,1,1,0] -> sorted [0,1,1,1,1]
    assert transformArray([1, 3, 5, 7, 2]) == [0, 1, 1, 1, 1]


def test_more_evens_than_odds():
    # [2,4,6,8,1] -> [0,0,0,0,1] -> sorted [0,0,0,0,1]
    assert transformArray([2, 4, 6, 8, 1]) == [0, 0, 0, 0, 1]
