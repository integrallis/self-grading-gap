```python
from typing import List

def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers
    online_status = [True] * numberOfUsers
    user_offline_until = [0] * numberOfUsers

    for event in events:
        event_type = event[0]
        timestamp = int(event[1])

        if event_type == "OFFLINE":
            user_id = int(event[2][2:])
            online_status[user_id] = False
            user_offline_until[user_id] = timestamp + 60
        elif event_type == "MESSAGE":
            mentions_string = event[2]

            # Process online status before the message
            for user_id in range(numberOfUsers):
                if not online_status[user_id] and timestamp >= user_offline_until[user_id]:
                    online_status[user_id] = True

            if mentions_string == "ALL":
                for user_id in range(numberOfUsers):
                    mentions[user_id] += 1
            elif mentions_string == "HERE":
                for user_id in range(numberOfUsers):
                    if online_status[user_id]:
                        mentions[user_id] += 1
            else:
                mentioned_ids = mentions_string.split()
                for mentioned_id in mentioned_ids:
                    user_id = int(mentioned_id[2:])
                    mentions[user_id] += 1

    return mentions

```
