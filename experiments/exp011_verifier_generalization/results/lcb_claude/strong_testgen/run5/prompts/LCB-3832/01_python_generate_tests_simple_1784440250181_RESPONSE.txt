from typing import List
from solution import minCosts


def test_canonical_example_1():
    # spec: cost = [5,3,4,1,3,2] -> [5,3,3,1,1,1]
    # i=0: swap with person 0, cost=5
    # i=1: swap with person 1, cost=3
    # i=2: swap with person 1 (cost=3), then swap with person 2 (free), total=3
    # i=3: swap with person 3, cost=1
    # i=4: swap with person 3 (cost=1), then swap with person 4 (free), total=1
    # i=5: swap with person 3 (cost=1), then swap with person 5 (free), total=1
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]


def test_canonical_example_2():
    # spec: cost = [1,2,4,6,7] -> [1,1,1,1,1]
    # swap with person 0 for cost=1, then reach any position for free
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]


def test_single_person():
    # cost = [5], we are at position 1
    # to reach position 0, swap with person 0 for cost=5
    assert minCosts([5]) == [5]


def test_decreasing_costs():
    # cost = [10, 5, 1]
    # i=0: swap with person 0, cost=10
    # i=1: swap with person 1, cost=5
    # i=2: swap with person 2, cost=1
    assert minCosts([10, 5, 1]) == [10, 5, 1]


def test_increasing_costs():
    # cost = [1, 5, 10]
    # i=0: swap with person 0, cost=1
    # i=1: swap with person 0 (cost=1), then swap with person 1 (free), total=1
    # i=2: swap with person 0 (cost=1), then swap with person 2 (free), total=1
    assert minCosts([1, 5, 10]) == [1, 1, 1]


def test_all_same_costs():
    # cost = [3, 3, 3]
    # i=0: swap with person 0, cost=3
    # i=1: swap with person 1, cost=3
    # i=2: swap with person 1 (cost=3), then swap with person 2 (free), total=3
    assert minCosts([3, 3, 3]) == [3, 3, 3]


def test_two_people():
    # cost = [10, 1]
    # i=0: swap with person 0, cost=10
    # i=1: swap with person 1, cost=1
    assert minCosts([10, 1]) == [10, 1]


def test_two_people_reverse():
    # cost = [1, 10]
    # i=0: swap with person 0, cost=1
    # i=1: swap with person 0 (cost=1), then swap with person 1 (free), total=1
    assert minCosts([1, 10]) == [1, 1]

    # cost = [5, 7, 1, 9, 8]
    # i=0: swap with person 0, cost=5
    # i=1: swap with person 1, cost=7
    # i=2: swap with person 2, cost=1
    # i=3: swap with person 2 (cost=1), then swap with person 3 (free), total=1
    # i=4: swap with person 2 (cost=1), then swap with person 4 (free), total=1
    assert minCosts([5, 7, 1, 9, 8]) == [5, 7, 1, 1, 1]
