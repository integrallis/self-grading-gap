import pytest
from solution import maxContainers


def test_canonical_example_1():
    # spec: n=2, w=3, maxWeight=15
    # deck has 2*2=4 cells, each container weighs 3
    # total weight if all loaded: 4*3=12, which is <= 15
    # so all 4 containers can be loaded
    assert maxContainers(2, 3, 15) == 4


def test_canonical_example_2():
    # spec: n=3, w=5, maxWeight=20
    # deck has 3*3=9 cells, each container weighs 5
    # maximum containers without exceeding 20: floor(20/5)=4
    # (limited by weight, not deck size)
    assert maxContainers(3, 5, 20) == 4


def test_deck_size_is_limiting_factor():
    # n=2, deck has 2*2=4 cells, w=1, maxWeight=1000
    # weight allows 1000 containers, but deck only has 4 cells
    # so maximum is 4
    assert maxContainers(2, 1, 1000) == 4


def test_weight_is_limiting_factor():
    # n=10, deck has 10*10=100 cells, w=10, maxWeight=50
    # deck allows 100 containers, but weight allows only floor(50/10)=5
    # so maximum is 5
    assert maxContainers(10, 10, 50) == 5


def test_exact_weight_match():
    # n=3, deck has 3*3=9 cells, w=5, maxWeight=45
    # weight allows exactly floor(45/5)=9 containers
    # deck also allows 9 containers
    # so maximum is 9
    assert maxContainers(3, 5, 45) == 9


def test_single_cell_deck():
    # n=1, deck has 1*1=1 cell, w=100, maxWeight=100
    # weight allows floor(100/100)=1 container
    # deck allows 1 container
    # so maximum is 1
    assert maxContainers(1, 100, 100) == 1


def test_single_cell_deck_weight_too_small():
    # n=1, deck has 1*1=1 cell, w=100, maxWeight=99
    # weight allows floor(99/100)=0 containers
    # so maximum is 0
    assert maxContainers(1, 100, 99) == 0


def test_large_deck_small_weight_capacity():
    # n=100, deck has 100*100=10000 cells, w=1000, maxWeight=5000
    # weight allows floor(5000/1000)=5 containers
    # deck allows 10000 containers
    # so maximum is 5
    assert maxContainers(100, 1000, 5000) == 5


def test_minimal_inputs():
    # n=1, w=1, maxWeight=1
    # deck has 1*1=1 cell, weight allows floor(1/1)=1 container
    # so maximum is 1
    assert maxContainers(1, 1, 1) == 1


def test_weight_allows_zero_containers():
    # n=5, deck has 5*5=25 cells, w=10, maxWeight=5
    # weight allows floor(5/10)=0 containers
    # so maximum is 0
    assert maxContainers(5, 10, 5) == 0
