## Implementation

```python
from typing import List
from itertools import permutations

def totalNumbers(digits: List[int]) -> int:
    distinct_numbers = set()
    for last_digit in set(d for d in digits if d % 2 == 0):  # only even digits can be last
        for perm in permutations(digits, 3):
            if perm[2] == last_digit and perm[0] != 0:  # ensure no leading zero
                distinct_numbers.add(perm)
    return len(distinct_numbers)
```
