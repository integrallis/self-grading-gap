```python
from typing import List
from itertools import permutations

def totalNumbers(digits: List[int]) -> int:
    distinct_numbers = set()
    for last_digit in set(d for d in digits if d % 2 == 0):  # only even digits can be last
        for first_digit in set(d for d in digits if d != last_digit and d != 0):  # no leading zero
            for second_digit in digits:
                if second_digit != first_digit and second_digit != last_digit:
                    # Form the number and add it to the set
                    number = first_digit * 100 + second_digit * 10 + last_digit
                    distinct_numbers.add(number)
    return len(distinct_numbers)
```
