from typing import List

def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a dictionary to hold coin counts at each bag position
    coin_map = {}
    for l, r, c in coins:
        for i in range(l, r + 1):
            coin_map[i] = coin_map.get(i, 0) + c

    # Create a sorted list of positions
    positions = sorted(coin_map.keys())
    max_coins = 0
    current_coins = 0
    left = 0

    # Use the sliding window technique to find the maximum coins in k consecutive bags
    for right in range(len(positions)):
        current_coins += coin_map[positions[right]]

        # If we exceed k bags, move the left pointer to maintain the size
        while positions[right] - positions[left] + 1 > k:
            current_coins -= coin_map[positions[left]]
            left += 1

        # Update the maximum coins found
        max_coins = max(max_coins, current_coins)

    return max_coins
```
