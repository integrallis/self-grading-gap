import pytest
from typing import List
from solution import maximumCoins


def test_canonical_example_1():
    # spec: coins = [[8,10,1],[1,3,2],[5,6,4]], k = 4
    # Selecting bags at positions [3, 4, 5, 6] gives: 2 + 0 + 4 + 4 = 10
    assert maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) == 10


def test_canonical_example_2():
    # spec: coins = [[1,10,3]], k = 2
    # Selecting bags at positions [1, 2] gives: 3 + 3 = 6
    assert maximumCoins([[1, 10, 3]], 2) == 6


def test_single_segment_k_equals_segment_length():
    # Single segment [1, 5, 2] with k = 5
    # All 5 bags contain 2 coins each: 2 * 5 = 10
    assert maximumCoins([[1, 5, 2]], 5) == 10


def test_single_segment_k_less_than_segment_length():
    # Single segment [1, 10, 5] with k = 3
    # Any 3 consecutive bags in the segment give: 5 * 3 = 15
    assert maximumCoins([[1, 10, 5]], 3) == 15


def test_single_segment_k_greater_than_segment_length():
    # Single segment [5, 7, 3] (3 bags) with k = 5
    # Best window includes all 3 bags: 3 * 3 = 9
    assert maximumCoins([[5, 7, 3]], 5) == 9


def test_multiple_segments_no_overlap():
    # Segments [1, 2, 5] and [4, 5, 3], k = 3
    # Window [1, 2, 3]: 5 + 5 + 0 = 10
    # Window [2, 3, 4]: 5 + 0 + 3 = 8
    # Window [3, 4, 5]: 0 + 3 + 3 = 6
    assert maximumCoins([[1, 2, 5], [4, 5, 3]], 3) == 10


def test_k_equals_one():
    # k = 1, should return the maximum coin value in any single bag
    # Segments [1, 3, 2] and [5, 6, 7]
    # Max single bag has 7 coins
    assert maximumCoins([[1, 3, 2], [5, 6, 7]], 1) == 7


def test_gaps_between_segments():
    # Segments [1, 1, 10], [5, 5, 20], k = 2
    # Window [1, 2]: 10 + 0 = 10
    # Window [4, 5]: 0 + 20 = 20
    # Window [5, 6]: 20 + 0 = 20
    assert maximumCoins([[1, 1, 10], [5, 5, 20]], 2) == 20


def test_large_k_spanning_all_segments():
    # Segments [1, 2, 1], [4, 5, 2], k = 10
    # Best window includes all bags with coins: 1 + 1 + 0 + 2 + 2 = 6
    assert maximumCoins([[1, 2, 1], [4, 5, 2]], 10) == 6


def test_window_partially_overlapping_segments():
    # Segments [2, 4, 3], [6, 8, 5], k = 5
    # Window [2, 3, 4, 5, 6]: 3 + 3 + 3 + 0 + 5 = 14
    # Window [4, 5, 6, 7, 8]: 3 + 0 + 5 + 5 + 5 = 18
    assert maximumCoins([[2, 4, 3], [6, 8, 5]], 5) == 18


def test_three_segments_with_gaps():
    # Segments [1, 1, 1], [3, 3, 2], [5, 5, 3], k = 3
    # Window [1, 2, 3]: 1 + 0 + 2 = 3
    # Window [3, 4, 5]: 2 + 0 + 3 = 5
    assert maximumCoins([[1, 1, 1], [3, 3, 2], [5, 5, 3]], 3) == 5


def test_consecutive_segments():
    # Segments [1, 2, 5], [3, 4, 10], k = 3
    # Window [1, 2, 3]: 5 + 5 + 10 = 20
    # Window [2, 3, 4]: 5 + 10 + 10 = 25
    assert maximumCoins([[1, 2, 5], [3, 4, 10]], 3) == 25
