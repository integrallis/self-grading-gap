from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    
    # Calculate contribution of each element as maximum and minimum
    for i in range(n):
        # Each element can be chosen as min or max in combinations
        # Number of ways to choose k-1 elements from the remaining n-1 elements
        max_contrib = (pow(2, n - 1, MOD) - 1) * pow(2, min(k - 1, n - 1), MOD) % MOD
        min_contrib = (pow(2, n - 1, MOD) - 1) * pow(2, min(k - 1, n - 1), MOD) % MOD
        total_sum = (total_sum + nums[i] * (max_contrib + min_contrib)) % MOD
    
    return total_sum % MOD
```
