from typing import List
import pytest
from solution import minMaxSums


def test_canonical_example_1():
    # spec: nums = [1,2,3], k = 2
    # Subsequences with at most 2 elements:
    # [1]: min=1, max=1, sum=2
    # [2]: min=2, max=2, sum=4
    # [3]: min=3, max=3, sum=6
    # [1,2]: min=1, max=2, sum=3
    # [1,3]: min=1, max=3, sum=4
    # [2,3]: min=2, max=3, sum=5
    # Total: 2+4+6+3+4+5 = 24
    assert minMaxSums([1, 2, 3], 2) == 24


def test_canonical_example_2():
    # spec: nums = [5,0,6], k = 1
    # Subsequences with exactly 1 element (at most 1):
    # [5]: min=5, max=5, sum=10
    # [0]: min=0, max=0, sum=0
    # [6]: min=6, max=6, sum=12
    # Total: 10+0+12 = 22
    assert minMaxSums([5, 0, 6], 1) == 22


def test_canonical_example_3():
    # spec: nums = [1,1,1], k = 2
    # Subsequences with at most 2 elements:
    # [1] appears 3 times: each contributes min=1, max=1, sum=2, total=6
    # [1,1] appears 3 times: each contributes min=1, max=1, sum=2, total=6
    # Total: 6+6 = 12
    assert minMaxSums([1, 1, 1], 2) == 12


def test_single_element_array():
    # nums = [5], k = 1
    # Only subsequence: [5], min=5, max=5, sum=10
    assert minMaxSums([5], 1) == 10


def test_k_equals_array_length():
    # nums = [1,2], k = 2
    # Subsequences with at most 2 elements (all possible):
    # [1]: min=1, max=1, sum=2
    # [2]: min=2, max=2, sum=4
    # [1,2]: min=1, max=2, sum=3
    # Total: 2+4+3 = 9
    assert minMaxSums([1, 2], 2) == 9


def test_all_same_elements():
    # nums = [7,7], k = 2
    # Subsequences:
    # [7] appears 2 times: each contributes min=7, max=7, sum=14, total=28
    # [7,7] appears 1 time: contributes min=7, max=7, sum=14
    # Total: 28+14 = 42
    assert minMaxSums([7, 7], 2) == 42


def test_zero_in_array():
    # nums = [0], k = 1
    # Only subsequence: [0], min=0, max=0, sum=0
    assert minMaxSums([0], 1) == 0


def test_large_numbers():
    # nums = [1000000000], k = 1
    # Only subsequence: [1000000000], min=1000000000, max=1000000000, sum=2000000000
    assert minMaxSums([1000000000], 1) == 2000000000


def test_modulo_applied():
    # Test that the result is returned modulo 10^9 + 7
    # Using a case where the computation might be large
    # nums = [1000000000, 1000000000], k = 1
    # [1000000000] appears 2 times: each contributes 2*10^9, total=4*10^9
    # 4*10^9 mod (10^9+7) = 4000000000 mod 1000000007 = 2999999986
    assert minMaxSums([1000000000, 1000000000], 1) == 2999999986


def test_k_larger_than_needed():
    # nums = [1,2,3], k = 3 (k equals length)
    # All possible subsequences of size 1, 2, and 3:
    # Size 1: [1]=2, [2]=4, [3]=6, sum=12
    # Size 2: [1,2]=3, [1,3]=4, [2,3]=5, sum=12
    # Size 3: [1,2,3]: min=1, max=3, sum=4
    # Total: 12+12+4 = 28
    assert minMaxSums([1, 2, 3], 3) == 28


def test_two_elements_k_one():
    # nums = [3,7], k = 1
    # Subsequences with at most 1 element:
    # [3]: min=3, max=3, sum=6
    # [7]: min=7, max=7, sum=14
    # Total: 6+14 = 20
    assert minMaxSums([3, 7], 1) == 20
