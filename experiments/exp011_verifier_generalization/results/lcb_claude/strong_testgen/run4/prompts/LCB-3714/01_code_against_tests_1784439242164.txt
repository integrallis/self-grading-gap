```python
from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Calculate contributions for each element as min and max
    for num in nums:
        # Each number can be min or max in different combinations
        # Count how many times this number contributes as max and min
        max_contrib = pow(2, min(k, n) - 1, MOD) * pow(2, n - k, MOD) % MOD
        min_contrib = pow(2, min(k, n) - 1, MOD) * pow(2, n - k, MOD) % MOD
        total_sum = (total_sum + num * (max_contrib + min_contrib)) % MOD

    return total_sum % MOD
```
