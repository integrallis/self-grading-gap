from typing import List
import pytest
from solution import longestSubsequence


def test_canonical_example_1():
    # spec: nums = [16,6,3] -> 3
    # The longest subsequence is [16, 6, 3] with absolute adjacent differences [10, 3]
    # 10 >= 3, so this is valid and has length 3
    assert longestSubsequence([16, 6, 3]) == 3


def test_canonical_example_2():
    # spec: nums = [6,5,3,4,2,1] -> 4
    # The longest subsequence is [6, 4, 2, 1] with absolute adjacent differences [2, 2, 1]
    # 2 >= 2 >= 1, so this is valid and has length 4
    assert longestSubsequence([6, 5, 3, 4, 2, 1]) == 4


def test_canonical_example_3():
    # spec: nums = [10,20,10,19,10,20] -> 5
    # The longest subsequence is [10, 20, 10, 19, 10] with absolute adjacent differences [10, 10, 9, 9]
    # 10 >= 10 >= 9 >= 9, so this is valid and has length 5
    assert longestSubsequence([10, 20, 10, 19, 10, 20]) == 5


def test_minimum_length_array():
    # spec: 2 <= nums.length, so test with length 2
    # Any two-element array forms a valid subsequence of length 2
    assert longestSubsequence([1, 2]) == 2


def test_all_same_elements():
    # All elements are the same, so all differences are 0
    # Any subsequence will have differences [0, 0, ...] which is non-increasing
    # The entire array can be the subsequence
    assert longestSubsequence([5, 5, 5, 5]) == 4


def test_strictly_increasing():
    # nums = [1, 2, 3, 4, 5]
    # Differences between consecutive elements: [1, 1, 1, 1]
    # This is non-increasing (1 >= 1 >= 1 >= 1), so entire sequence works
    assert longestSubsequence([1, 2, 3, 4, 5]) == 5


def test_strictly_decreasing():
    # nums = [5, 4, 3, 2, 1]
    # Differences between consecutive elements: [1, 1, 1, 1]
    # This is non-increasing (1 >= 1 >= 1 >= 1), so entire sequence works
    assert longestSubsequence([5, 4, 3, 2, 1]) == 5


def test_increasing_differences():
    # nums = [1, 2, 4, 7]
    # Full sequence differences: [1, 2, 3] which is increasing, not non-increasing
    # Best might be [1, 2] with diff [1], or [1, 4] with diff [3], etc.
    # Any single pair has length 2
    assert longestSubsequence([1, 2, 4, 7]) == 2


def test_alternating_large_small():
    # nums = [1, 100, 2, 99]
    # Subsequence [1, 100, 2] has diffs [99, 98] which is non-increasing, length 3
    # Subsequence [1, 100, 99] has diffs [99, 1] which is non-increasing, length 3
    # Subsequence [1, 100, 2, 99] has diffs [99, 98, 97] which is non-increasing, length 4
    assert longestSubsequence([1, 100, 2, 99]) == 4


def test_single_peak():
    # nums = [1, 5, 2]
    # Full sequence diffs: [4, 3] which is non-increasing (4 >= 3)
    assert longestSubsequence([1, 5, 2]) == 3


def test_boundary_values():
    # spec: 1 <= nums[i] <= 300
    # Test with boundary values
    assert longestSubsequence([1, 300]) == 2
    assert longestSubsequence([300, 1]) == 2
