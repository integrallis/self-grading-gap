from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = [[1, 0]] * n  # dp[i][0] = length, dp[i][1] = last difference
    max_length = 0

    for i in range(n):
        for j in range(i):
            abs_diff = abs(nums[i] - nums[j])
            if abs_diff >= dp[j][1]:  # Check non-increasing condition
                dp[i][0] = max(dp[i][0], dp[j][0] + 1)
                dp[i][1] = abs_diff
        max_length = max(max_length, dp[i][0])

    return max_length
```
