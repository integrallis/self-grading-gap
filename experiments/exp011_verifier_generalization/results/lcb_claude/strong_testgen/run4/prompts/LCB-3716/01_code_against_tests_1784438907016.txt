```python
from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    # dp array to store the maximum length of subsequence ending at each index with corresponding last difference
    dp = [[1, 0] for _ in range(n)]  # dp[i][0] = length, dp[i][1] = last difference
    max_length = 1

    for i in range(n):
        for j in range(i):
            abs_diff = abs(nums[i] - nums[j])
            # Ensure the conditions for the subsequence are met
            if abs_diff >= dp[j][1]:  # Check if the absolute difference non-increasing condition holds
                dp[i][0] = max(dp[i][0], dp[j][0] + 1)  # Update length
                dp[i][1] = abs_diff  # Update the last difference
        max_length = max(max_length, dp[i][0])  # Track the maximum length found

    return max_length
```
