```python
from typing import List

def maximumCoins(coins: List[List[int]], k: int) -> int:
    # To store the number of coins at each position
    max_position = 0
    for l, r, c in coins:
        max_position = max(max_position, r)
    coin_count = [0] * (max_position + 2)  # +2 for easier boundary handling

    # Populate the coin_count array
    for l, r, c in coins:
        for i in range(l, r + 1):
            coin_count[i] += c

    # Sliding window to find the maximum coins in k consecutive bags
    max_coins = 0
    current_coins = sum(coin_count[1:k + 1])  # Initial sum for the first k positions
    max_coins = current_coins

    for i in range(2, max_position + 1):
        if i + k - 1 <= max_position:
            current_coins += coin_count[i + k - 1] - coin_count[i - 1]
            max_coins = max(max_coins, current_coins)

    return max_coins
```
