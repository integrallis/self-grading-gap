import pytest
from typing import List
from solution import maximumCoins


def test_canonical_example_1():
    # spec: coins = [[8,10,1],[1,3,2],[5,6,4]], k = 4 -> 10
    # Selecting bags at positions [3, 4, 5, 6] gives: 2 + 0 + 4 + 4 = 10
    assert maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) == 10


def test_canonical_example_2():
    # spec: coins = [[1,10,3]], k = 2 -> 6
    # Selecting bags at positions [1, 2] gives: 3 + 3 = 6
    assert maximumCoins([[1, 10, 3]], 2) == 6


def test_single_segment_k_equals_segment_length():
    # Single segment [1, 5, 2], k = 5 should give 2*5 = 10
    assert maximumCoins([[1, 5, 2]], 5) == 10


def test_single_segment_k_larger_than_segment():
    # Single segment [1, 3, 5], k = 10 should give 5*3 = 15 (only 3 bags have coins)
    assert maximumCoins([[1, 3, 5]], 10) == 15


def test_single_segment_k_smaller_than_segment():
    # Single segment [1, 10, 4], k = 3 should give 4*3 = 12
    assert maximumCoins([[1, 10, 4]], 3) == 12


def test_multiple_segments_no_overlap_k_spans_all():
    # segments [1,2,3], [4,5,2]; k = 5 spans positions [1,2,3,4,5] = 3+3+0+2+2 = 10
    assert maximumCoins([[1, 2, 3], [4, 5, 2]], 5) == 10


def test_multiple_segments_k_fits_in_gap():
    # segments [1,2,5], [10,12,3]; k = 3 in gap [3,4,5] gives 0
    # or starting at [1,2,3] gives 5+5+0 = 10
    # or starting at [10,11,12] gives 3+3+3 = 9
    assert maximumCoins([[1, 2, 5], [10, 12, 3]], 3) == 10


def test_k_equals_one():
    # k = 1, select single bag with maximum coins
    # segments [1,3,2], [5,7,10], [9,10,1]; max single bag has 10 coins
    assert maximumCoins([[1, 3, 2], [5, 7, 10], [9, 10, 1]], 1) == 10


def test_large_gap_between_segments():
    # segments [1,2,5], [1000,1001,7]; k = 10
    # Best window could be [1,2,3,...,10] = 5+5+0+0+... = 10
    # or [996,997,...,1005] = 0+0+...+7+7 = 14
    assert maximumCoins([[1, 2, 5], [1000, 1001, 7]], 10) == 14


def test_three_segments_k_spans_two():
    # segments [1,2,1], [4,5,2], [7,8,3]; k = 4
    # Window [2,3,4,5] gives 1+0+2+2 = 5
    # Window [4,5,6,7] gives 2+2+0+3 = 7
    # Window [5,6,7,8] gives 2+0+3+3 = 8
    assert maximumCoins([[1, 2, 1], [4, 5, 2], [7, 8, 3]], 4) == 8


def test_consecutive_segments():
    # segments [1,3,2], [4,6,5]; k = 4
    # Window [3,4,5,6] gives 2+5+5+5 = 17
    assert maximumCoins([[1, 3, 2], [4, 6, 5]], 4) == 17


def test_single_bag_segment():
    # segments [5,5,10], [10,10,20]; k = 3
    # Window [5,6,7] gives 10+0+0 = 10
    # Window [9,10,11] gives 0+20+0 = 20
    assert maximumCoins([[5, 5, 10], [10, 10, 20]], 3) == 20


def test_all_bags_empty_except_one_segment():
    # segment [100,102,7]; k = 5
    # Window [100,101,102,103,104] gives 7+7+7+0+0 = 21
    # Window [99,100,101,102,103] gives 0+7+7+7+0 = 21
    # Window [98,99,100,101,102] gives 0+0+7+7+7 = 21
    assert maximumCoins([[100, 102, 7]], 5) == 21


def test_k_very_large():
    # segments [1,2,3], [10,11,5]; k = 1000000
    # Only 4 bags have coins: 3+3+5+5 = 16
    assert maximumCoins([[1, 2, 3], [10, 11, 5]], 1000000) == 16
