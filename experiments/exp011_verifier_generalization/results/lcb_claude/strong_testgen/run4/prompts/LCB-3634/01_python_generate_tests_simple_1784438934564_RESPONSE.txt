import pytest
from solution import calculateScore


def test_canonical_example_1():
    # spec: s = "aczzx" -> 5
    # i=0: no j, skip
    # i=1: no j, skip
    # i=2: 'z' mirror is 'a', j=0 unmarked, mark both, score += 2-0=2
    # i=3: 'z' mirror is 'a', j=0 marked, no j, skip
    # i=4: 'x' mirror is 'c', j=1 unmarked, mark both, score += 4-1=3
    # total: 2+3=5
    assert calculateScore("aczzx") == 5


def test_canonical_example_2():
    # spec: s = "abcdef" -> 0
    # For each index i, there is no index j that satisfies the conditions
    assert calculateScore("abcdef") == 0


def test_single_character():
    # s = "a", i=0 has no j < 0, score = 0
    assert calculateScore("a") == 0


def test_two_mirror_characters():
    # s = "az", i=0 skip, i=1 'z' mirror is 'a' at j=0, score = 1-0 = 1
    assert calculateScore("az") == 1


def test_two_non_mirror_characters():
    # s = "ab", i=0 skip, i=1 'b' mirror is 'y', no match at j=0, score = 0
    assert calculateScore("ab") == 0


def test_multiple_same_character_with_mirror():
    # s = "aaazzz"
    # mirrors: a<->z
    # i=0: skip
    # i=1: 'a' mirror 'z', no j, skip
    # i=2: 'a' mirror 'z', no j, skip
    # i=3: 'z' mirror 'a', j=2 unmarked 'a', mark both, score += 3-2=1
    # i=4: 'z' mirror 'a', j=1 unmarked 'a', mark both, score += 4-1=3
    # i=5: 'z' mirror 'a', j=0 unmarked 'a', mark both, score += 5-0=5
    # total: 1+3+5=9
    assert calculateScore("aaazzz") == 9


def test_closest_unmarked_index():
    # s = "aazz"
    # mirrors: a<->z
    # i=0: skip
    # i=1: 'a' mirror 'z', no j, skip
    # i=2: 'z' mirror 'a', closest unmarked j with 'a' is j=1, mark both, score += 2-1=1
    # i=3: 'z' mirror 'a', closest unmarked j with 'a' is j=0, mark both, score += 3-0=3
    # total: 1+3=4
    assert calculateScore("aazz") == 4


def test_middle_alphabet_mirrors():
    # s = "mnop", mirrors: m<->n, o<->l, p<->k
    # i=0: skip
    # i=1: 'n' mirror 'm', j=0 has 'm', mark both, score += 1-0=1
    # i=2: 'o' mirror 'l', no j, skip
    # i=3: 'p' mirror 'k', no j, skip
    # total: 1
    assert calculateScore("mnop") == 1


def test_already_marked_indices_skipped():
    # s = "azza"
    # mirrors: a<->z
    # i=0: skip
    # i=1: 'z' mirror 'a', j=0 unmarked 'a', mark both, score += 1-0=1
    # i=2: 'z' mirror 'a', j=0 marked, no other j, skip
    # i=3: 'a' mirror 'z', j=1 marked, no other j, skip
    # total: 1
    assert calculateScore("azza") == 1


def test_mirror_pairs_at_ends():
    # s = "zyxabc", mirrors: z<->a, y<->b, x<->c
    # i=0: skip
    # i=1: skip
    # i=2: skip
    # i=3: 'a' mirror 'z', j=0 has 'z', mark both, score += 3-0=3
    # i=4: 'b' mirror 'y', j=1 has 'y', mark both, score += 4-1=3
    # i=5: 'c' mirror 'x', j=2 has 'x', mark both, score += 5-2=3
    # total: 3+3+3=9
    assert calculateScore("zyxabc") == 9


def test_no_mirrors_present():
    # s = "aaaaa", mirror of 'a' is 'z', no 'z' present
    assert calculateScore("aaaaa") == 0


def test_all_same_mirror_pair():
    # s = "aaaa", mirror of 'a' is 'z', never matches itself
    assert calculateScore("aaaa") == 0
