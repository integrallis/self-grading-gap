```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # a-z to z-a

    for i in range(len(s)):
        if marked[i]:  # Skip if already marked
            continue
        mirror_char = mirror[s[i]]
        # Find the closest unmarked index j
        for j in range(i - 1, -1, -1):
            if s[j] == mirror_char and not marked[j]:
                marked[i] = True
                marked[j] = True
                score += i - j
                break
    return score

```
