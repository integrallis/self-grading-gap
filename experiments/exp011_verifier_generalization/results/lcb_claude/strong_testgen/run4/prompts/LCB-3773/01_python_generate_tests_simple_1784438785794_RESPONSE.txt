import pytest
from typing import List
from solution import minimumPairRemoval


def test_canonical_example_1():
    # spec: [5,2,3,1] -> 2
    # (3,1) has min sum 4 -> [5,2,4]
    # (2,4) has min sum 6 -> [5,6]
    # 2 operations to make non-decreasing
    assert minimumPairRemoval([5, 2, 3, 1]) == 2


def test_canonical_example_2():
    # spec: [1,2,2] -> 0
    # already non-decreasing, no operations needed
    assert minimumPairRemoval([1, 2, 2]) == 0


def test_single_element_array():
    # single element is trivially non-decreasing
    assert minimumPairRemoval([5]) == 0


def test_two_element_ascending():
    # [1, 2] is already non-decreasing
    assert minimumPairRemoval([1, 2]) == 0


def test_two_element_descending():
    # [2, 1] needs one operation to become [3]
    assert minimumPairRemoval([2, 1]) == 1


def test_all_same_elements():
    # [3, 3, 3, 3] is already non-decreasing
    assert minimumPairRemoval([3, 3, 3, 3]) == 0


def test_strictly_increasing():
    # [1, 2, 3, 4, 5] is already non-decreasing
    assert minimumPairRemoval([1, 2, 3, 4, 5]) == 0


def test_strictly_decreasing():
    # [5, 4, 3, 2, 1] needs operations
    # min sum is (2,1)=3 -> [5,4,3,3]
    # min sum is (4,3)=7 -> [5,7,3]
    # min sum is (7,3)=10 -> [5,10]
    # 3 operations
    assert minimumPairRemoval([5, 4, 3, 2, 1]) == 3


def test_negative_numbers():
    # [-5, -3, -1] is already non-decreasing
    assert minimumPairRemoval([-5, -3, -1]) == 0


def test_mixed_negative_positive():
    # [-1, 2, 1] where (2,1) sum=3 -> [-1, 3]
    # already non-decreasing after 1 operation
    assert minimumPairRemoval([-1, 2, 1]) == 1


def test_with_zeros():
    # [0, 0, 0] is non-decreasing
    assert minimumPairRemoval([0, 0, 0]) == 0


def test_leftmost_pair_selection():
    # [3, 1, 3, 1] has two pairs with sum=4: (3,1) at index 0 and (3,1) at index 2
    # leftmost (3,1) at index 0 -> [4, 3, 1]
    # min sum is (3,1)=4 -> [4, 4]
    # 2 operations
    assert minimumPairRemoval([3, 1, 3, 1]) == 2


def test_large_numbers():
    # [1000, -1000] needs one operation -> [0]
    assert minimumPairRemoval([1000, -1000]) == 1


def test_already_sorted_longer():
    # [1, 2, 3, 4, 5, 6, 7, 8] is already non-decreasing
    assert minimumPairRemoval([1, 2, 3, 4, 5, 6, 7, 8]) == 0
