import pytest
from solution import hasSpecialSubstring


def test_canonical_example_1():
    # spec: s = "aaabaaa", k = 3 -> true
    # The substring s[4..6] == "aaa" satisfies the conditions
    assert hasSpecialSubstring("aaabaaa", 3) is True


def test_canonical_example_2():
    # spec: s = "abc", k = 2 -> false
    # No substring of length 2 that consists of one distinct character
    assert hasSpecialSubstring("abc", 2) is False


def test_substring_at_start_no_char_before():
    # substring at start: no character before, different char after
    # "aaa" at position 0, followed by 'b'
    assert hasSpecialSubstring("aaab", 3) is True


def test_substring_at_end_no_char_after():
    # substring at end: different char before, no character after
    # "bbb" at position 1, preceded by 'a'
    assert hasSpecialSubstring("abbb", 3) is True


def test_substring_in_middle():
    # substring in middle: different chars before and after
    # "ccc" at position 1, preceded by 'a', followed by 'b'
    assert hasSpecialSubstring("acccb", 3) is True


def test_entire_string_is_special_substring():
    # entire string is the substring with k = length
    # "aaa" with k = 3, no chars before or after
    assert hasSpecialSubstring("aaa", 3) is True


def test_k_equals_1_single_char():
    # k = 1, single character substring
    # any single char at start or end works, or surrounded by different chars
    assert hasSpecialSubstring("a", 1) is True


def test_k_equals_1_in_longer_string():
    # k = 1 in a longer string where char is isolated
    # "b" at position 1, surrounded by 'a'

    # alternating pattern, no runs of k same chars
    assert hasSpecialSubstring("ababab", 2) is False


def test_run_at_boundary_start():
    # run of exactly k at start, different after
    assert hasSpecialSubstring("aab", 2) is True


def test_run_at_boundary_end():
    # run of exactly k at end, different before
