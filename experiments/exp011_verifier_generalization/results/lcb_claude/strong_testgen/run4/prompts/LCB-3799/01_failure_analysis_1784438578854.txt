from typing import List
from itertools import permutations

def totalNumbers(digits: List[int]) -> int:
    even_digits = [d for d in set(digits) if d % 2 == 0]
    count = 0
    for even_digit in even_digits:
        for perm in permutations(set(digits), 3):
            if perm[2] == even_digit and perm[0] != 0:
                count += 1
    return count // 6  # Each valid permutation counted 6 times (3!)
