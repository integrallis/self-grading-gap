from typing import List
from itertools import permutations

def totalNumbers(digits: List[int]) -> int:
    count = 0
    unique_digits = list(set(digits))
    even_digits = [d for d in unique_digits if d % 2 == 0]
    for even_digit in even_digits:
        for perm in permutations(unique_digits, 3):
            if perm[2] == even_digit and perm[0] != 0:
                count += 1
    return count

```
