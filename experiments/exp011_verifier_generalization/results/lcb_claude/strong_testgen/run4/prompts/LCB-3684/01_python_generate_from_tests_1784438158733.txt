def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]
    for i in range(len(s) - len(prefix) - len(suffix) + 1):
        if s[i:i + len(prefix)] == prefix:
            if s[i + len(prefix):].startswith(suffix):
                return True
    # Check if the suffix can match the end of the string
    if s.endswith(suffix) and len(s) >= len(prefix):
        return True
    # Check if the prefix can match the start of the string
    if s.startswith(prefix) and len(s) >= len(suffix):
        return True
    return False

```
