import pytest
from solution import hasMatch


def test_canonical_example_1():
    # spec: s = "leetcode", p = "ee*e" -> true
    # By replacing the '*' with "tcod", the substring "eetcode" matches the pattern.
    assert hasMatch("leetcode", "ee*e") is True


def test_canonical_example_2():
    # spec: s = "car", p = "c*v" -> false
    # There is no substring matching the pattern.
    assert hasMatch("car", "c*v") is False


def test_canonical_example_3():
    # spec: s = "luck", p = "u*" -> true
    # The substrings "u", "uc", and "uck" match the pattern.
    assert hasMatch("luck", "u*") is True


def test_star_at_beginning():
    # Pattern starts with *, should match if suffix of pattern matches
    assert hasMatch("abcdef", "*def") is True


def test_star_at_end():
    # Pattern ends with *, should match if prefix of pattern matches
    assert hasMatch("abcdef", "abc*") is True


def test_star_in_middle():
    # Pattern has * in middle, need matching prefix and suffix
    assert hasMatch("abcdef", "ab*ef") is True


def test_star_matches_zero_characters():
    # * can match zero characters
    assert hasMatch("abc", "a*bc") is True


def test_star_matches_multiple_characters():
    # * can match multiple characters
    assert hasMatch("abcdefg", "ab*fg") is True


def test_no_match_prefix_missing():
    # Prefix before * doesn't exist in s
    assert hasMatch("abc", "xyz*") is False


def test_no_match_suffix_missing():
    # Suffix after * doesn't exist in s
    assert hasMatch("abc", "*xyz") is False


def test_no_match_prefix_and_suffix_dont_align():
    # Prefix and suffix exist but don't form valid substring
    assert hasMatch("abcdef", "bc*ab") is False


def test_single_character_pattern_with_star():
    # Pattern is just "*"
    assert hasMatch("abc", "*") is True


def test_pattern_longer_than_string_no_star_replacement():
    # Pattern would require * to match zero chars, but prefix+suffix too long
    assert hasMatch("ab", "a*bc") is False


def test_overlapping_prefix_and_suffix():
    # Prefix and suffix can overlap when * matches zero characters
    assert hasMatch("abab", "ab*ab") is True


def test_pattern_matches_entire_string():
    # Pattern with * can match the entire string
    assert hasMatch("hello", "he*lo") is True


def test_pattern_matches_at_end_of_string():
    # Pattern matches at the end of string
    assert hasMatch("abcdef", "de*f") is True


def test_pattern_matches_at_start_of_string():
    # Pattern matches at the start of string
    assert hasMatch("abcdef", "a*c") is True
