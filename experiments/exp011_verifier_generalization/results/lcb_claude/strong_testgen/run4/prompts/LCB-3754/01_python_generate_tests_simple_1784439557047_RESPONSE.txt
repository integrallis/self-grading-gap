import pytest
from solution import maxDistance


def test_canonical_example_1():
    # spec: s = "NWSE", k = 1 -> 3
    # Change s[2] from 'S' to 'N' to get "NWNE"
    # Positions: (0,1), (-1,1), (-1,2), (0,2)
    # Distances: 1, 2, 3, 2
    # Maximum: 3
    assert maxDistance("NWSE", 1) == 3


def test_canonical_example_2():
    # spec: s = "NSWWEW", k = 3 -> 6
    # Change s[1] from 'S' to 'N' and s[4] from 'E' to 'W' to get "NNWWWW"
    # Maximum Manhattan distance achievable is 6
    assert maxDistance("NSWWEW", 3) == 6


def test_single_move_north_no_changes():
    # s = "N", k = 0
    # Position after N: (0, 1), distance = 1
    assert maxDistance("N", 0) == 1


def test_single_move_south_no_changes():
    # s = "S", k = 0
    # Position after S: (0, -1), distance = 1
    assert maxDistance("S", 0) == 1


def test_single_move_east_no_changes():
    # s = "E", k = 0
    # Position after E: (1, 0), distance = 1
    assert maxDistance("E", 0) == 1


def test_single_move_west_no_changes():
    # s = "W", k = 0
    # Position after W: (-1, 0), distance = 1
    assert maxDistance("W", 0) == 1


def test_opposite_moves_cancel_no_changes():
    # s = "NS", k = 0
    # Positions: (0, 1), (0, 0)
    # Distances: 1, 0
    # Maximum: 1
    assert maxDistance("NS", 0) == 1


def test_all_moves_same_direction_no_changes():
    # s = "NNNN", k = 0
    # Positions: (0,1), (0,2), (0,3), (0,4)
    # Distances: 1, 2, 3, 4
    # Maximum: 4
    assert maxDistance("NNNN", 0) == 4


def test_can_change_all_moves():
    # s = "SSSS", k = 4
    # Can change all to N: positions (0,1), (0,2), (0,3), (0,4)
    # Maximum: 4
    assert maxDistance("SSSS", 4) == 4


def test_change_reverses_direction():
    # s = "NNS", k = 1
    # Change S to N: positions (0,1), (0,2), (0,3)
    # Maximum: 3
    assert maxDistance("NNS", 1) == 3


def test_k_is_zero():
    # s = "NESW", k = 0
    # No changes allowed, follow original path
    # Positions: (0,1), (1,1), (1,0), (0,0)
    # Distances: 1, 2, 1, 0
    # Maximum: 2
    assert maxDistance("NESW", 0) == 2


def test_path_returns_to_origin():
    # s = "ESWN", k = 0
    # Positions: (1,0), (1,-1), (0,-1), (0,0)
    # Distances: 1, 2, 1, 0
    # Maximum: 2
    assert maxDistance("ESWN", 0) == 2


def test_large_k_value():
    # s = "EW", k = 10
    # k > s.length, can change both moves
    # Change to "EE": positions (1,0), (2,0)
    # Maximum: 2
    assert maxDistance("EW", 10) == 2


def test_mixed_directions_with_changes():
    # s = "EENN", k = 2
    # Can optimize to maximize distance
    # Best: all move in same direction, e.g., "EEEE" or "NNNN"
    # Maximum: 4
    assert maxDistance("EENN", 2) == 4


def test_alternating_moves():
    # s = "NSNS", k = 0
    # Positions: (0,1), (0,0), (0,1), (0,0)
    # Distances: 1, 0, 1, 0
    # Maximum: 1
    assert maxDistance("NSNS", 0) == 1
