```python
from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # DP array for storing cumulative costs
    dp = [[0]*3 for _ in range(n)]

    # Initialize the first house's costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    for i in range(1, n):
        for j in range(3):
            # Calculate the minimum cost for painting house i with color j
            dp[i][j] = min(
                dp[i-1][(j+1) % 3],  # Previous house with color (j+1)
                dp[i-1][(j+2) % 3]   # Previous house with color (j+2)
            ) + cost[i][j]

    # Handle equidistant constraint
    min_cost = float('inf')
    for j in range(3):
        cost_option_1 = dp[n-1][j]  # Cost of using color j for the last house
        for j2 in range(3):
            if j != j2:  # Ensure colors are different
                cost_option_2 = dp[n//2-1][j2]  # Cost using color j2 for the middle house
                total_cost = cost_option_1 + cost_option_2
                min_cost = min(min_cost, total_cost)

    return min_cost

```
