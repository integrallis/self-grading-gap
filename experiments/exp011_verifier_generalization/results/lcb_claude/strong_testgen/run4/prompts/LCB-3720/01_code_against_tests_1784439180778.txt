```python
from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[u].append((v, w))

    def canReachZero(maxWeight):
        from collections import deque
        indegree = [0] * n
        outdegree = [0] * n
        for u in range(n):
            for v, w in graph[u]:
                if w <= maxWeight:
                    outdegree[u] += 1
                    indegree[v] += 1

        queue = deque()
        for i in range(n):
            if i != 0 and indegree[i] == 0:  # Start from all nodes except node 0
                queue.append(i)

        reachable = set()
        while queue:
            current = queue.popleft()
            if current != 0:  # Do not count node 0 as reachable for reachable set
                reachable.add(current)
            for neighbor, weight in graph[current]:
                if weight <= maxWeight:
                    indegree[neighbor] -= 1
                    if indegree[neighbor] == 0:
                        queue.append(neighbor)

        return len(reachable) == n - 1  # Check if all nodes except 0 are reachable

    left, right = 1, 10**6
    result = -1

    while left <= right:
        mid = (left + right) // 2
        if canReachZero(mid):
            result = mid
            right = mid - 1
        else:
            left = mid + 1

    return result

```
