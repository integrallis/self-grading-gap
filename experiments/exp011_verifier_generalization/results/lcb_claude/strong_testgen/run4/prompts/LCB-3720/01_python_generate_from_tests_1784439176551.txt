from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[u].append((v, w))

    def canReachZero(maxWeight):
        indegree = [0] * n
        outdegree = [0] * n
        for u in range(n):
            for v, w in graph[u]:
                if w <= maxWeight:
                    outdegree[u] += 1
                    indegree[v] += 1

        # Check if we can reach 0 from all nodes
        queue = []
        for i in range(1, n):
            if indegree[i] == 0:
                queue.append(i)

        reachable = set(queue)
        while queue:
            current = queue.pop(0)
            for neighbor, weight in graph[current]:
                if weight <= maxWeight:
                    indegree[neighbor] -= 1
                    if indegree[neighbor] == 0:
                        queue.append(neighbor)
                        reachable.add(neighbor)

        return len(reachable) == n - 1

    left, right = 1, 10**6
    result = -1

    while left <= right:
        mid = (left + right) // 2
        if canReachZero(mid):
            result = mid
            right = mid - 1
        else:
            left = mid + 1

    return result

```
