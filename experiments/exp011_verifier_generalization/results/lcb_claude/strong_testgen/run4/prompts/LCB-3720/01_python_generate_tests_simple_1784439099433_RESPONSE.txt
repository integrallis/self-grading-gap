from typing import List
import pytest
from solution import minMaxWeight


def test_canonical_example_1():
    # spec: n=5, edges=[[1,0,1],[2,0,2],[3,0,1],[4,3,1],[2,1,1]], threshold=2 -> 1
    # Remove edge 2->0 (weight 2). Remaining edges have max weight 1.
    assert minMaxWeight(5, [[1,0,1],[2,0,2],[3,0,1],[4,3,1],[2,1,1]], 2) == 1


def test_canonical_example_2():
    # spec: n=5, edges=[[0,1,1],[0,2,2],[0,3,1],[0,4,1],[1,2,1],[1,4,1]], threshold=1 -> -1
    # Impossible to reach node 0 from node 2
    assert minMaxWeight(5, [[0,1,1],[0,2,2],[0,3,1],[0,4,1],[1,2,1],[1,4,1]], 1) == -1


def test_canonical_example_3():
    # spec: n=5, edges=[[1,2,1],[1,3,3],[1,4,5],[2,3,2],[3,4,2],[4,0,1]], threshold=1 -> 2
    # Remove edges 1->3 and 1->4. Remaining edges have max weight 2.
    assert minMaxWeight(5, [[1,2,1],[1,3,3],[1,4,5],[2,3,2],[3,4,2],[4,0,1]], 1) == 2


def test_canonical_example_4():
    # spec: n=5, edges=[[1,2,1],[1,3,3],[1,4,5],[2,3,2],[4,0,1]], threshold=1 -> -1
    # Impossible to satisfy all conditions
    assert minMaxWeight(5, [[1,2,1],[1,3,3],[1,4,5],[2,3,2],[4,0,1]], 1) == -1


def test_simple_two_nodes_direct_connection():
    # n=2, single edge 1->0 with weight 5, threshold=1
    # Node 1 can reach node 0, so answer is 5
    assert minMaxWeight(2, [[1,0,5]], 1) == 5


def test_simple_two_nodes_no_connection():
    # n=2, no edges from node 1 to node 0, threshold=1
    # Node 1 cannot reach node 0, so answer is -1
    assert minMaxWeight(2, [[0,1,5]], 1) == -1


def test_all_nodes_directly_connect_to_zero():
    # n=3, edges 1->0 (weight 3), 2->0 (weight 7), threshold=1
    # All nodes can reach 0 directly, max weight is 7
    assert minMaxWeight(3, [[1,0,3],[2,0,7]], 1) == 7


def test_threshold_forces_edge_removal():
    # n=3, edges 1->0 (weight 10), 1->2 (weight 5), 2->0 (weight 3), threshold=1
    # Node 1 has 2 outgoing edges but threshold=1, must remove one
    # To minimize max weight, keep 1->2 (5) and 2->0 (3), max=5
    assert minMaxWeight(3, [[1,0,10],[1,2,5],[2,0,3]], 1) == 5


def test_multiple_edges_same_nodes_different_weights():
    # n=2, edges 1->0 with weights 10 and 5, threshold=1
    # Can choose the edge with weight 5
    assert minMaxWeight(2, [[1,0,10],[1,0,5]], 1) == 5


def test_chain_path_to_zero():
    # n=4, edges 3->2 (1), 2->1 (2), 1->0 (3), threshold=1
    # Chain from 3->2->1->0, max weight is 3
    assert minMaxWeight(4, [[3,2,1],[2,1,2],[1,0,3]], 1) == 3


def test_threshold_allows_multiple_outgoing():
    # n=4, edges 1->0 (5), 1->2 (3), 1->3 (7), 2->0 (2), 3->0 (4), threshold=3
    # Node 1 has 3 outgoing edges, threshold=3 allows all
    # All nodes can reach 0, max weight is 7
    assert minMaxWeight(4, [[1,0,5],[1,2,3],[1,3,7],[2,0,2],[3,0,4]], 3) == 7


def test_node_zero_has_outgoing_edges_irrelevant():
    # n=3, edges 0->1 (100), 0->2 (200), 1->0 (1), 2->0 (2), threshold=1
    # Nodes 1 and 2 can reach 0. Node 0's outgoing edges don't matter for reachability to 0
    # Max weight among edges needed is 2
    assert minMaxWeight(3, [[0,1,100],[0,2,200],[1,0,1],[2,0,2]], 1) == 2


def test_unreachable_node_with_other_edges():
    # n=3, edges 1->0 (5), 0->2 (10), threshold=1
    # Node 2 cannot reach node 0 (only edge is from 0 to 2)
    assert minMaxWeight(3, [[1,0,5],[0,2,10]], 1) == -1


def test_minimum_weight_selection_with_alternatives():
    # n=3, edges 1->0 (10), 2->1 (2), 2->0 (3), threshold=2
    # Node 1: direct edge to 0 (weight 10)
    # Node 2: can go 2->0 (3) or 2->1->0 (max 10). Choose 2->0.
    # Both 2->1 and 2->0 fit threshold=2. Max weight is 10 (from 1->0)
    assert minMaxWeight(3, [[1,0,10],[2,1,2],[2,0,3]], 2) == 10
