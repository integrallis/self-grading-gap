```python
from typing import List


def maxFrequency(nums: List[int], k: int) -> int:
    nums.sort()
    max_freq = 0
    left = 0
    total = 0
    total_times = 0

    for right in range(len(nums)):
        total += nums[right]
        total_times += nums[right] * (right - left + 1)

        while total + (right - left + 1) * k > total_times:
            total -= nums[left]
            left += 1
            total_times -= nums[left-1] * (right - left + 1)

        max_freq = max(max_freq, right - left + 1)

    return max_freq
```
