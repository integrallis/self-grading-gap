from typing import List
import pytest
from solution import maxFrequency


def test_canonical_example_1():
    # spec: nums = [1,2,3,4,5,6], k = 1
    # After adding -5 to nums[2..5], 1 has a frequency of 2 in [1, 2, -2, -1, 0, 1]
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2


def test_canonical_example_2():
    # spec: nums = [10,2,3,4,5,5,4,3,2,2], k = 10
    # After adding 8 to nums[1..9], 10 has a frequency of 4 in [10, 10, 11, 12, 13, 13, 12, 11, 10, 10]
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4


def test_single_element_already_k():
    # nums = [5], k = 5
    # Already has frequency 1, no operation needed (can select empty-like subarray or change nothing)
    assert maxFrequency([5], 5) == 1


def test_single_element_not_k():
    # nums = [3], k = 5
    # Can add 2 to nums[0..0] to get [5], frequency = 1
    assert maxFrequency([3], 5) == 1


def test_all_elements_same_as_k():
    # nums = [7, 7, 7], k = 7
    # All elements are already k, frequency = 3
    assert maxFrequency([7, 7, 7], 7) == 3


def test_convert_all_elements_to_k():
    # nums = [1, 1, 1], k = 5
    # Add 4 to nums[0..2] to get [5, 5, 5], frequency = 3
    assert maxFrequency([1, 1, 1], 5) == 3


def test_no_initial_k_convert_subarray():
    # nums = [2, 3, 4], k = 5
    # Add 2 to nums[1..2] to get [2, 5, 6], or add 3 to nums[0..0] to get [5, 3, 4]
    # Or add 1 to nums[2..2] to get [2, 3, 5]
    # Maximum frequency of k = 1
    assert maxFrequency([2, 3, 4], 5) == 1


def test_some_initial_k_values():
    # nums = [10, 5, 10], k = 10
    # Already has frequency 2, can also add 5 to nums[1..1] to get [10, 10, 10], frequency = 3
    assert maxFrequency([10, 5, 10], 10) == 3


def test_convert_middle_subarray():
    # nums = [1, 2, 2, 2, 1], k = 3
    # Add 1 to nums[1..3] to get [1, 3, 3, 3, 1], frequency = 3
    assert maxFrequency([1, 2, 2, 2, 1], 3) == 3


def test_initial_k_at_boundaries():
    # nums = [5, 1, 1, 5], k = 5
    # Already has frequency 2
    # Add 4 to nums[1..2] to get [5, 5, 5, 5], frequency = 4
    assert maxFrequency([5, 1, 1, 5], 5) == 4


def test_larger_array_with_repeating_values():
    # nums = [3, 3, 3, 7, 7, 7], k = 5
    # Add 2 to nums[0..2] to get [5, 5, 5, 7, 7, 7], frequency = 3
    # Or add -2 to nums[3..5] to get [3, 3, 3, 5, 5, 5], frequency = 3
    assert maxFrequency([3, 3, 3, 7, 7, 7], 5) == 3


def test_optimal_subarray_in_middle():
    # nums = [10, 8, 8, 8, 10], k = 10
    # Already has frequency 2
    # Add 2 to nums[1..3] to get [10, 10, 10, 10, 10], frequency = 5
    assert maxFrequency([10, 8, 8, 8, 10], 10) == 5


def test_two_element_array():
    # nums = [1, 3], k = 2
    # Add 1 to nums[0..0] to get [2, 3], or add -1 to nums[1..1] to get [1, 2]
    # Or add 1 to nums[0..1] to get [2, 4]
    # Maximum frequency of k = 1
    assert maxFrequency([1, 3], 2) == 1


def test_two_element_array_both_convertible():
    # nums = [2, 2], k = 5
    # Add 3 to nums[0..1] to get [5, 5], frequency = 2
    assert maxFrequency([2, 2], 5) == 2
