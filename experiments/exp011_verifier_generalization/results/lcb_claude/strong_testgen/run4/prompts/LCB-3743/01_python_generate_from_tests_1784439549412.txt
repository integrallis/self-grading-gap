from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_segments = []
    # Initialize free time before the first meeting
    if startTime[0] > 0:
        free_segments.append(startTime[0])
    # Calculate free time between meetings
    for i in range(1, n):
        gap = startTime[i] - endTime[i - 1]
        if gap > 0:
            free_segments.append(gap)
    # Initialize free time after the last meeting
    if endTime[-1] < eventTime:
        free_segments.append(eventTime - endTime[-1])
    # Sort free segments in descending order
    free_segments.sort(reverse=True)
    # Calculate maximum free time possible with k meetings rescheduled
    max_free_time = 0
    for i in range(min(k, len(free_segments))):
        max_free_time += free_segments[i]
    if k < len(free_segments):
        max_free_time += free_segments[k] if k < len(free_segments) else 0
    return max_free_time
```
