```python
from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_times = []

    # Calculate initial free time segments
    free_times.append(startTime[0])  # Free time until the first meeting
    for i in range(1, n):
        free_times.append(startTime[i] - endTime[i-1])  # Free time between meetings
    free_times.append(eventTime - endTime[-1])  # Free time after the last meeting

    # We can reschedule up to k meetings, which could create new free times
    free_times.sort(reverse=True)  # Sort free times in descending order

    # If k is larger than the number of free segments, we can only use k segments
    return sum(free_times[:k]) + (0 if k == 0 else free_times[k])

```
