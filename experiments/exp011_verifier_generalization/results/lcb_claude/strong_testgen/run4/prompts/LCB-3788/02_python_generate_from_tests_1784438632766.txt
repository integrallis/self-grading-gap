from typing import List

def maxSum(nums: List[int]) -> int:
    unique_elements = set()
    total_sum = 0
    max_sum = float('-inf')

    for num in nums:
        if num not in unique_elements:
            unique_elements.add(num)
            total_sum += num
        else:
            max_sum = max(max_sum, total_sum)
            # Reset for new unique subarray
            total_sum = num
            unique_elements = {num}

    # Final check for the last segment
    max_sum = max(max_sum, total_sum)
    return max_sum
```
