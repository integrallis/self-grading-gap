import pytest
from typing import List
from solution import transformArray


def test_canonical_example_1():
    # spec: [4,3,2,1] -> replace evens with 0, odds with 1 -> [0,1,0,1] -> sort -> [0,0,1,1]
    assert transformArray([4, 3, 2, 1]) == [0, 0, 1, 1]


def test_canonical_example_2():
    # spec: [1,5,1,4,2] -> replace evens with 0, odds with 1 -> [1,1,1,0,0] -> sort -> [0,0,1,1,1]
    assert transformArray([1, 5, 1, 4, 2]) == [0, 0, 1, 1, 1]


def test_single_even_element():
    # [2] -> [0] -> [0]
    assert transformArray([2]) == [0]


def test_single_odd_element():
    # [3] -> [1] -> [1]
    assert transformArray([3]) == [1]


def test_all_even_numbers():
    # [2,4,6,8] -> [0,0,0,0] -> [0,0,0,0]
    assert transformArray([2, 4, 6, 8]) == [0, 0, 0, 0]


def test_all_odd_numbers():
    # [1,3,5,7] -> [1,1,1,1] -> [1,1,1,1]
    assert transformArray([1, 3, 5, 7]) == [1, 1, 1, 1]


def test_mixed_numbers_unsorted():
    # [10,9,8,7] -> [0,1,0,1] -> [0,0,1,1]
    assert transformArray([10, 9, 8, 7]) == [0, 0, 1, 1]


def test_mixed_numbers_already_sorted():
    # [1,2,3,4] -> [1,0,1,0] -> [0,0,1,1]
    assert transformArray([1, 2, 3, 4]) == [0, 0, 1, 1]


def test_larger_array():
    # [100,99,50,51,1,2] -> [0,1,0,1,1,0] -> [0,0,0,1,1,1]
    assert transformArray([100, 99, 50, 51, 1, 2]) == [0, 0, 0, 1, 1, 1]


def test_maximum_constraint_values():
    # [1000,999,1,2] -> [0,1,1,0] -> [0,0,1,1]
    assert transformArray([1000, 999, 1, 2]) == [0, 0, 1, 1]
