import pytest
from solution import maxDifference


def test_canonical_example_1():
    # spec: "aaaaabbc" -> 3
    # 'a' has frequency 5 (odd), 'b' has frequency 2 (even)
    # max difference = 5 - 2 = 3
    assert maxDifference("aaaaabbc") == 3


def test_canonical_example_2():
    # spec: "abcabcab" -> 1
    # 'a' has frequency 3 (odd), 'c' has frequency 2 (even)
    # max difference = 3 - 2 = 1
    assert maxDifference("abcabcab") == 1


def test_single_odd_single_even():
    # "aaabbb" has 'a' with frequency 3 (odd) and 'b' with frequency 2 (even)
    # Difference = 3 - 2 = 1
    # Note: constraint guarantees at least one odd and one even
    assert maxDifference("aaabb") == 1


def test_max_odd_min_even():
    # "aaaaaabb" has 'a' with frequency 6 (even), 'b' with frequency 2 (even)
    # Need at least one odd, so let's try "aaaaaaabbc"
    # 'a' = 7 (odd), 'b' = 2 (even), 'c' = 1 (odd)
    # max odd = 7, min even = 2, difference = 7 - 2 = 5
    assert maxDifference("aaaaaaabbc") == 5


def test_multiple_characters_with_odd_and_even():
    # "aabbccc" has 'a' = 2 (even), 'b' = 2 (even), 'c' = 3 (odd)
    # max odd = 3, min even = 2, difference = 3 - 2 = 1
    assert maxDifference("aabbccc") == 1


def test_min_length_string():
    # "aab" has 'a' = 2 (even), 'b' = 1 (odd)
    # difference = 1 - 2 = -1
    # Wait, the spec says "odd - even", so we need max(odd) - min(even)
    # Actually looking at examples: 5 (odd) - 2 (even) = 3, 3 (odd) - 2 (even) = 1
    # So we want maximum odd frequency minus minimum even frequency
    assert maxDifference("aab") == -1


def test_large_odd_small_even():
    # "aaaaaaaaabbc" has 'a' = 9 (odd), 'b' = 2 (even), 'c' = 1 (odd)
    # max odd = 9, min even = 2, difference = 9 - 2 = 7
    assert maxDifference("aaaaaaaaabbc") == 7


def test_all_different_frequencies():
    # "aabbbcccc" has 'a' = 2 (even), 'b' = 3 (odd), 'c' = 4 (even)
    # max odd = 3, min even = 2, difference = 3 - 2 = 1
    assert maxDifference("aabbbcccc") == 1


def test_negative_difference():
    # "aabbbbbbbbbc" has 'a' = 2 (even), 'b' = 9 (odd), 'c' = 1 (odd)
    # max odd = 9, min even = 2, difference = 9 - 2 = 7
    # Let's try one where even is larger: "aaaaaaaabbc"
    # 'a' = 8 (even), 'b' = 2 (even), 'c' = 1 (odd)
    # max odd = 1, min even = 2, difference = 1 - 2 = -1
    assert maxDifference("aaaaaaaabbc") == -1
