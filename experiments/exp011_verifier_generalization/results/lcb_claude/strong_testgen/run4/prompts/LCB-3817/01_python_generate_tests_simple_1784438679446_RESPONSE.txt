import pytest
from solution import maxContainers


def test_canonical_example_1():
    # spec: n=2, w=3, maxWeight=15
    # deck has 2*2=4 cells, each container weighs 3
    # total weight if all loaded: 4*3=12, which is ≤15
    # so all 4 containers can be loaded
    assert maxContainers(2, 3, 15) == 4


def test_canonical_example_2():
    # spec: n=3, w=5, maxWeight=20
    # deck has 3*3=9 cells, each container weighs 5
    # maximum containers without exceeding 20: 20//5=4
    # (9 containers would weigh 45, exceeding 20)
    assert maxContainers(3, 5, 20) == 4


def test_single_cell_deck():
    # n=1 gives 1*1=1 cell
    # if w=10 and maxWeight=10, we can load 1 container (weight=10)
    assert maxContainers(1, 10, 10) == 1


def test_weight_capacity_limits_containers():
    # n=4 gives 4*4=16 cells
    # w=7, maxWeight=50
    # 50//7=7 containers can be loaded (7*7=49 ≤ 50)
    # even though deck has 16 cells
    assert maxContainers(4, 7, 50) == 7


def test_deck_size_limits_containers():
    # n=2 gives 2*2=4 cells
    # w=1, maxWeight=1000
    # maxWeight allows 1000 containers, but deck only has 4 cells
    # so only 4 containers can be loaded
    assert maxContainers(2, 1, 1000) == 4


def test_exact_fit_weight_capacity():
    # n=3 gives 3*3=9 cells
    # w=4, maxWeight=36
    # 36//4=9 containers, and deck has exactly 9 cells
    # all 9 containers fit perfectly
    assert maxContainers(3, 4, 36) == 9


def test_weight_capacity_less_than_one_container():
    # n=5 gives 5*5=25 cells
    # w=100, maxWeight=50
    # maxWeight < w, so 50//100=0 containers can be loaded
    assert maxContainers(5, 100, 50) == 0


def test_large_deck_small_capacity():
    # n=10 gives 10*10=100 cells
    # w=15, maxWeight=30
    # 30//15=2 containers can be loaded
    assert maxContainers(10, 15, 30) == 2


def test_minimum_inputs():
    # n=1, w=1, maxWeight=1 (all minimum per constraints)
    # deck has 1 cell, each container weighs 1, capacity is 1
    # can load 1 container
    assert maxContainers(1, 1, 1) == 1


def test_large_maxweight():
    # n=3 gives 3*3=9 cells
    # w=50, maxWeight=1000000000 (10^9, upper constraint)
    # maxWeight allows 1000000000//50=20000000 containers
    # but deck only has 9 cells
    assert maxContainers(3, 50, 1000000000) == 9
