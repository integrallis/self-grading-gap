import pytest
from solution import reverseDegree


def test_canonical_example_1():
    # spec: "abc" -> 148
    # 'a'(26)*1 + 'b'(25)*2 + 'c'(24)*3 = 26 + 50 + 72 = 148
    assert reverseDegree("abc") == 148


def test_canonical_example_2():
    # spec: "zaza" -> 160
    # 'z'(1)*1 + 'a'(26)*2 + 'z'(1)*3 + 'a'(26)*4 = 1 + 52 + 3 + 104 = 160
    assert reverseDegree("zaza") == 160


def test_single_character_a():
    # 'a' has reversed alphabet position 26, string position 1
    # 26 * 1 = 26
    assert reverseDegree("a") == 26


def test_single_character_z():
    # 'z' has reversed alphabet position 1, string position 1
    # 1 * 1 = 1
    assert reverseDegree("z") == 1


def test_single_character_m():
    # 'm' is the 13th letter, so reversed position is 27 - 13 = 14
    # 14 * 1 = 14
    assert reverseDegree("m") == 14


def test_repeated_character():
    # "aaa" -> 'a'(26)*1 + 'a'(26)*2 + 'a'(26)*3 = 26 + 52 + 78 = 156
    assert reverseDegree("aaa") == 156


def test_descending_alphabet():
    # "cba" -> 'c'(24)*1 + 'b'(25)*2 + 'a'(26)*3 = 24 + 50 + 78 = 152
    assert reverseDegree("cba") == 152


def test_all_z():
    # "zzz" -> 'z'(1)*1 + 'z'(1)*2 + 'z'(1)*3 = 1 + 2 + 3 = 6
    assert reverseDegree("zzz") == 6


def test_longer_string():
    # "abcdefghij" (10 characters)
    # a(26)*1 + b(25)*2 + c(24)*3 + d(23)*4 + e(22)*5 + f(21)*6 + g(20)*7 + h(19)*8 + i(18)*9 + j(17)*10
    # = 26 + 50 + 72 + 92 + 110 + 126 + 140 + 152 + 162 + 170 = 1100
    assert reverseDegree("abcdefghij") == 1100
