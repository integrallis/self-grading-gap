```python
from collections import Counter

def findValidPair(s: str) -> str:
    counts = Counter(s)
    for i in range(len(s) - 1):
        first = s[i]
        second = s[i + 1]
        if first != second:
            if counts[first] == int(first) and counts[second] == int(second):
                return first + second
    return ""
```
