from solution import maxAdjacentDistance
from typing import List


def test_canonical_example_1():
    # spec: nums = [1,2,4] -> 3
    # Adjacent pairs: (1,2)->|1-2|=1, (2,4)->|2-4|=2, (4,1)->|4-1|=3
    # Maximum is 3
    assert maxAdjacentDistance([1, 2, 4]) == 3


def test_canonical_example_2():
    # spec: nums = [-5,-10,-5] -> 5
    # Adjacent pairs: (-5,-10)->|-5-(-10)|=5, (-10,-5)->|-10-(-5)|=5, (-5,-5)->|-5-(-5)|=0
    # Maximum is 5
    assert maxAdjacentDistance([-5, -10, -5]) == 5


def test_two_element_array():
    # Minimum length is 2
    # Adjacent pairs: (1,5)->|1-5|=4, (5,1)->|5-1|=4 (circular)
    # Maximum is 4
    assert maxAdjacentDistance([1, 5]) == 4


def test_all_same_elements():
    # All elements are the same
    # Adjacent pairs: (3,3)->|3-3|=0 for all pairs
    # Maximum is 0
    assert maxAdjacentDistance([3, 3, 3, 3]) == 0


def test_positive_numbers():
    # nums = [10, 20, 15, 25]
    # Adjacent pairs: (10,20)->10, (20,15)->5, (15,25)->10, (25,10)->15
    # Maximum is 15
    assert maxAdjacentDistance([10, 20, 15, 25]) == 15


def test_negative_numbers():
    # nums = [-1, -10, -3]
    # Adjacent pairs: (-1,-10)->9, (-10,-3)->7, (-3,-1)->2
    # Maximum is 9
    assert maxAdjacentDistance([-1, -10, -3]) == 9


def test_mixed_positive_negative():
    # nums = [-10, 20, -5]
    # Adjacent pairs: (-10,20)->30, (20,-5)->25, (-5,-10)->5
    # Maximum is 30
    assert maxAdjacentDistance([-10, 20, -5]) == 30


def test_zero_included():
    # nums = [0, 10, -10]
    # Adjacent pairs: (0,10)->10, (10,-10)->20, (-10,0)->10
    # Maximum is 20
    assert maxAdjacentDistance([0, 10, -10]) == 20


def test_circular_has_max():
    # nums = [100, 1, 2, 3]
    # Adjacent pairs: (100,1)->99, (1,2)->1, (2,3)->1, (3,100)->97
    # Maximum is 99 (circular pair is not the max here, but first-second is)
    assert maxAdjacentDistance([100, 1, 2, 3]) == 99


def test_circular_pair_is_max():
    # nums = [50, 10, 20, 30]
    # Adjacent pairs: (50,10)->40, (10,20)->10, (20,30)->10, (30,50)->20
    # Maximum is 40
    assert maxAdjacentDistance([50, 10, 20, 30]) == 40
