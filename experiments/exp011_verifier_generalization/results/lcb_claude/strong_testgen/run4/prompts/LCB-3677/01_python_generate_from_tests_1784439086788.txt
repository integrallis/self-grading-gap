If coins[i][j] >= 0, the robot gains that many coins.
If coins[i][j] < 0, the robot encounters a robber, and the robber steals the absolute value of coins[i][j] coins.

The robot has a special ability to neutralize robbers in at most 2 cells on its path, preventing them from stealing coins in those cells.
Note: The robot's total coins can be negative.

```python
from typing import List
```


```python
def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[-float('inf')] * (n + 1) for _ in range(m + 1)]
    dp[0][1] = 0  # Start at (0,0) with 0 coins gained

    for i in range(m):
        for j in range(n):
            current = coins[i][j]
            for k in range(3):  # k is the number of neutralizations used
                if k == 0:
                    # No neutralization
                    dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j] + current)
                elif k == 1:
                    # One neutralization
                    if current < 0:
                        dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j])  # Neutralize this robber
                    else:
                        dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j] + current)
                elif k == 2:
                    # Two neutralizations
                    if current < 0:
                        dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j])  # Neutralize this robber
                    else:
                        dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j] + current)

            # Move to the right or down
            if i + 1 < m:
                dp[i + 2][j + 1] = max(dp[i + 2][j + 1], dp[i + 1][j + 1])
            if j + 1 < n:
                dp[i + 1][j + 2] = max(dp[i + 1][j + 2], dp[i + 1][j + 1])
    return dp[m][n]
```
