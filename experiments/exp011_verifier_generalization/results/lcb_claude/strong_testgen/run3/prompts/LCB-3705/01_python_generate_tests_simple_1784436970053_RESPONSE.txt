import pytest
from typing import List
from solution import largestInteger


def test_canonical_example_1():
    # spec: nums = [3,9,2,1,7], k = 3 -> 7
    # 7 appears in exactly one subarray of size 3: [2, 1, 7]
    assert largestInteger([3, 9, 2, 1, 7], 3) == 7


def test_canonical_example_2():
    # spec: nums = [3,9,7,2,1,7], k = 4 -> 3
    # 3 appears in exactly one subarray of size 4: [3, 9, 7, 2]
    assert largestInteger([3, 9, 7, 2, 1, 7], 4) == 3


def test_canonical_example_3():
    # spec: nums = [0,0], k = 1 -> -1
    # No integer appears in only one subarray of size 1
    # (0 appears in both [0] subarrays)
    assert largestInteger([0, 0], 1) == -1


def test_single_element_array():
    # nums = [5], k = 1
    # Only one subarray [5], 5 appears in exactly 1 subarray
    assert largestInteger([5], 1) == 5


def test_all_elements_appear_in_multiple_subarrays():
    # nums = [1, 1, 1], k = 2
    # Subarrays: [1, 1], [1, 1]
    # 1 appears in 2 subarrays, so no almost missing integer
    assert largestInteger([1, 1, 1], 2) == -1


def test_k_equals_array_length():
    # nums = [3, 5, 2], k = 3
    # Only one subarray of size 3: [3, 5, 2]
    # All elements appear in exactly 1 subarray
    # Return the largest: 5
    assert largestInteger([3, 5, 2], 3) == 5


def test_multiple_almost_missing_integers():
    # nums = [10, 20, 30, 40], k = 2
    # Subarrays: [10, 20], [20, 30], [30, 40]
    # 10 appears in 1 subarray: [10, 20]
    # 20 appears in 2 subarrays: [10, 20], [20, 30]
    # 30 appears in 2 subarrays: [20, 30], [30, 40]
    # 40 appears in 1 subarray: [30, 40]
    # Almost missing: 10, 40. Largest is 40
    assert largestInteger([10, 20, 30, 40], 2) == 40


def test_duplicate_values_in_different_subarrays():
    # nums = [5, 1, 5], k = 2
    # Subarrays: [5, 1], [1, 5]
    # 5 appears in 2 subarrays: [5, 1], [1, 5]
    # 1 appears in 2 subarrays: [5, 1], [1, 5]
    # No almost missing integer
    assert largestInteger([5, 1, 5], 2) == -1

    # nums = [1, 2, 3, 4, 5], k = 2

    # 2 appears in 2 subarrays: [1, 2], [2, 3]
    # 3 appears in 2 subarrays: [2, 3], [3, 4]
    # 4 appears in 2 subarrays: [3, 4], [4, 5]
    # 5 appears in 1 subarray: [4, 5]
    # Almost missing: 1, 5. Largest is 5
    assert largestInteger([1, 2, 3, 4, 5], 2) == 5


def test_zero_values():
    # nums = [0, 1, 2], k = 2
    # Subarrays: [0, 1], [1, 2]
    # 0 appears in 1 subarray: [0, 1]
    # 1 appears in 2 subarrays: [0, 1], [1, 2]
    # 2 appears in 1 subarray: [1, 2]
    # Almost missing: 0, 2. Largest is 2
    assert largestInteger([0, 1, 2], 2) == 2


def test_same_value_in_single_subarray():
    # nums = [7, 7, 7], k = 3
    # Only one subarray: [7, 7, 7]
    # 7 appears in 1 subarray
    assert largestInteger([7, 7, 7], 3) == 7


def test_larger_array_with_mixed_frequencies():
