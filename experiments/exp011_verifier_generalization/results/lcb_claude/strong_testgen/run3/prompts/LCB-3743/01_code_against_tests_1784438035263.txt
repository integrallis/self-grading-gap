```python
from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    # Calculate the gaps between meetings and the event boundaries
    gaps = []
    prev_end = 0
    for i in range(n):
        if startTime[i] > prev_end:
            gaps.append(startTime[i] - prev_end)
        prev_end = endTime[i]
    if prev_end < eventTime:
        gaps.append(eventTime - prev_end)

    # If we can move k meetings, we need to check how to maximize free time
    max_free_time = 0
    for i in range(len(gaps)):
        if i < len(gaps) - k:
            max_free_time = max(max_free_time, gaps[i])
        else:
            # Start merging gaps if we can move meetings
            merged_gap = sum(gaps[i:])
            max_free_time = max(max_free_time, merged_gap)

    return max_free_time
```
