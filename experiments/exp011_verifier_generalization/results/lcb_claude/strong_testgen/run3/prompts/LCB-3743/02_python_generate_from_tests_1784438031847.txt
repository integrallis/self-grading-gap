from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    # Calculate all gaps
    gaps = []
    prev_end = 0
    for i in range(n):
        if startTime[i] > prev_end:
            gaps.append(startTime[i] - prev_end)
        prev_end = endTime[i]
    if prev_end < eventTime:
        gaps.append(eventTime - prev_end)

    # If we can move k meetings, we can potentially combine the largest gaps
    gaps.sort(reverse=True)
    max_free_time = 0
    for i in range(min(k, len(gaps))):
        max_free_time += gaps[i]

    # The maximum free time is constrained by the largest free time gap
    return max(max_free_time, max(gaps) if gaps else 0)
```
