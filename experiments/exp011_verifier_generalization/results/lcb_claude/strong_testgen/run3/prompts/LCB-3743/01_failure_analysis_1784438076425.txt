from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    intervals = [(startTime[i], endTime[i]) for i in range(n)]

    # Calculate free time before the first meeting
    free_time_before_first = intervals[0][0]

    # Calculate gaps between meetings
    gaps = []
    for i in range(1, n):
        gaps.append(intervals[i][0] - intervals[i-1][1])

    # Calculate free time after the last meeting
    free_time_after_last = eventTime - intervals[-1][1]

    # If k is greater than or equal to n, we can move all meetings to maximize free time
    if k >= n:
        total_free_time = free_time_before_first + free_time_after_last + sum(gaps)
        return total_free_time

    # We will now evaluate how to maximize the largest possible gap
    max_free_time = max(free_time_before_first, free_time_after_last)  # Start with edges

    # Generate all possible gaps in the form of reachable gaps after moving k meetings
    for i in range(n - k):
        # If we can move meetings from 0 to i to before the current interval
        potential_gap = intervals[i+k][0] - intervals[0][1] if i+k < n else 0
        max_free_time = max(max_free_time, potential_gap)

    return max_free_time
