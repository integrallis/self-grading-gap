import pytest
from solution import hasSpecialSubstring


def test_canonical_example_1():
    # spec: s = "aaabaaa", k = 3 -> true
    # The substring s[4..6] == "aaa" satisfies the conditions
    assert hasSpecialSubstring("aaabaaa", 3) is True


def test_canonical_example_2():
    # spec: s = "abc", k = 2 -> false
    # No substring of length 2 consists of one distinct character
    assert hasSpecialSubstring("abc", 2) is False


def test_single_character_string_k_equals_1():
    # s = "a", k = 1
    # The entire string is a valid substring (no chars before/after)
    assert hasSpecialSubstring("a", 1) is True


def test_entire_string_same_character():
    # s = "aaaa", k = 4
    # The entire string is all 'a's and there are no chars before/after
    assert hasSpecialSubstring("aaaa", 4) is True


def test_substring_at_start_with_different_after():
    # s = "aaab", k = 3
    # s[0..2] == "aaa", no char before, char after is 'b' (different)
    assert hasSpecialSubstring("aaab", 3) is True


def test_substring_at_end_with_different_before():
    # s = "baaa", k = 3
    # s[1..3] == "aaa", char before is 'b' (different), no char after
    assert hasSpecialSubstring("baaa", 3) is True


def test_substring_in_middle_with_different_boundaries():
    # s = "baaac", k = 3
    # s[1..3] == "aaa", char before is 'b', char after is 'c'
    assert hasSpecialSubstring("baaac", 3) is True


def test_substring_same_char_before():
    # s = "aaaa", k = 2
    # s[0..1] == "aa" has no char before, but char after is 'a' (same)
    # s[1..2] == "aa" has char before 'a' (same)
    # s[2..3] == "aa" has char before 'a' (same), no char after
    # None satisfy the conditions
    assert hasSpecialSubstring("aaaa", 2) is False


def test_substring_same_char_after():
    # s = "aabaa", k = 2
    # s[0..1] == "aa" has no char before, char after is 'b' (different) -> valid!
    assert hasSpecialSubstring("aabaa", 2) is True


def test_no_repeating_characters():
    # s = "abcdef", k = 1
    # Each single character is a valid substring
    # e.g., s[1..1] == "b", char before is 'a' (different), char after is 'c' (different)
    assert hasSpecialSubstring("abcdef", 1) is True


def test_repeating_but_not_enough_length():
    # s = "aabbaabb", k = 3
    # No substring of 3 identical characters exists
    assert hasSpecialSubstring("aabbaabb", 3) is False


def test_exactly_k_length_run_isolated():
    # s = "abbba", k = 3
    # s[1..3] == "bbb", char before is 'a' (different), char after is 'a' (different)
    assert hasSpecialSubstring("abbba", 3) is True


def test_longer_run_contains_valid_k_substring():
    # s = "aaaaaa", k = 3
    # s[0..2] == "aaa", no char before, but char after is 'a' (same) -> invalid
    # s[1..3] == "aaa", char before is 'a' (same) -> invalid
    # All substrings of length 3 either have same char before or after
    assert hasSpecialSubstring("aaaaaa", 3) is False


def test_k_equals_string_length_all_same():
    # s = "bbb", k = 3
    # The entire string, no chars before/after
    assert hasSpecialSubstring("bbb", 3) is True


def test_k_equals_string_length_not_all_same():
    # s = "aba", k = 3
    # The entire string does not consist of one distinct character
    assert hasSpecialSubstring("aba", 3) is False


def test_multiple_valid_substrings():
    # s = "aabbbaa", k = 2
    # s[0..1] == "aa", no char before, char after is 'b' (different) -> valid!
    assert hasSpecialSubstring("aabbbaa", 2) is True


def test_run_exactly_k_at_boundaries():
    # s = "cccabbb", k = 3
    # s[0..2] == "ccc", no char before, char after is 'a' (different) -> valid!
    assert hasSpecialSubstring("cccabbb", 3) is True
