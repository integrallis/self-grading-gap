def findValidPair(s: str) -> str:
    from collections import Counter
    counts = Counter(s)
    for i in range(len(s) - 1):
        first = s[i]
        second = s[i + 1]
        if first != second:
            if counts[first] == int(first) and counts[second] == int(second):
                return first + second
    return ""
```
