from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    from collections import defaultdict, deque

    edges_by_weight = defaultdict(list)
    for u, v, w in edges:
        edges_by_weight[w].append((u, v))

    def can_reach_zero(max_weight: int) -> bool:
        graph = defaultdict(list)
        out_degree = [0] * n
        for weight in range(1, max_weight + 1):
            for u, v in edges_by_weight[weight]:
                if out_degree[u] < threshold:
                    graph[u].append(v)  # Build outgoing graph
                    out_degree[u] += 1
        visited = [False] * n
        queue = deque([0])
        visited[0] = True
        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    queue.append(neighbor)
        return all(visited[i] == True for i in range(n) if any(out_degree[i] > 0 for u in range(n) if (u, i) in edges))

    low, high = 1, 10**6
    answer = -1
    while low <= high:
        mid = (low + high) // 2
        if can_reach_zero(mid):
            answer = mid
            high = mid - 1
        else:
            low = mid + 1

    return answer
```
