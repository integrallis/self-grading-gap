import pytest
from solution import hasMatch


def test_canonical_example_1():
    # spec: s = "leetcode", p = "ee*e" -> true
    # By replacing '*' with "tcod", substring "eetcode" matches
    assert hasMatch("leetcode", "ee*e") is True


def test_canonical_example_2():
    # spec: s = "car", p = "c*v" -> false
    # No substring matching the pattern
    assert hasMatch("car", "c*v") is False


def test_canonical_example_3():
    # spec: s = "luck", p = "u*" -> true
    # Substrings "u", "uc", "uck" match the pattern
    assert hasMatch("luck", "u*") is True


def test_star_at_beginning():
    # p = "*abc" means any chars followed by "abc"
    # s = "xyzabc" has substring "xyzabc" matching "*abc" (replace * with "xyz")
    assert hasMatch("xyzabc", "*abc") is True


def test_star_at_end():
    # p = "abc*" means "abc" followed by any chars
    # s = "abcdef" has substring "abcdef" matching "abc*" (replace * with "def")
    assert hasMatch("abcdef", "abc*") is True


def test_star_in_middle():
    # p = "a*b" means "a" followed by any chars followed by "b"
    # s = "axyzb" has substring "axyzb" matching "a*b" (replace * with "xyz")
    assert hasMatch("axyzb", "a*b") is True


def test_star_replaced_with_empty():
    # p = "a*b" with * replaced by empty string gives "ab"
    # s = "cab" has substring "ab" matching pattern
    assert hasMatch("cab", "a*b") is True


def test_single_char_pattern_with_star():
    # p = "*" means any sequence of zero or more chars
    # Any non-empty string has at least one substring (itself or parts)
    # s = "a" has substring "a" matching "*" (replace * with "a")
    assert hasMatch("a", "*") is True


def test_no_match_prefix():
    # p = "x*y" requires "x" at start and "y" at end of substring
    # s = "abc" has no substring starting with "x"
    assert hasMatch("abc", "x*y") is False


def test_no_match_suffix():
    # p = "a*z" requires "a" at start and "z" at end of substring
    # s = "abcd" has substring starting with "a" but none ending with "z"
    assert hasMatch("abcd", "a*z") is False


def test_pattern_longer_than_string_prefix_suffix():
    # p = "ab*cd" has prefix "ab" and suffix "cd", minimum length 4
    # s = "abc" length 3, cannot contain a substring matching pattern
    assert hasMatch("abc", "ab*cd") is False


def test_exact_match_with_empty_star():
    # p = "abc*def" with * replaced by empty gives "abcdef"
    # s = "abcdef" has substring "abcdef" matching pattern
    assert hasMatch("abcdef", "abc*def") is True


def test_substring_in_middle():
    # p = "bc*de" 
    # s = "abcdefg" has substring "bcde" matching "bc*de" (* replaced with empty)
    assert hasMatch("abcdefg", "bc*de") is True


def test_repeated_chars():
    # p = "a*a"
    # s = "aaa" has substrings "aa", "aaa" matching "a*a" (* replaced with "" or "a")
    assert hasMatch("aaa", "a*a") is True


def test_no_prefix_match():
    # p = "xyz*"
    # s = "abc" has no substring starting with "xyz"
    assert hasMatch("abc", "xyz*") is False


def test_no_suffix_match():
    # p = "*xyz"
    # s = "abc" has no substring ending with "xyz"
    assert hasMatch("abc", "*xyz") is False


def test_star_only_pattern():
    # p = "*" can match any substring (replace * with any chars)
    # s = "hello" has many substrings like "h", "he", etc.
    assert hasMatch("hello", "*") is True


def test_pattern_parts_not_in_order():
    # p = "ba*cd" requires "ba" followed eventually by "cd"
    # s = "abcda" has no substring starting with "ba"
    assert hasMatch("abcda", "ba*cd") is False


def test_overlapping_prefix_suffix():
    # p = "ab*ab"
