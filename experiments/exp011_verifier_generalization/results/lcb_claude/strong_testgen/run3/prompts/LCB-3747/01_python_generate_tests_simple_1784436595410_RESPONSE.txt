from typing import List
import pytest
from solution import maxAdjacentDistance


def test_canonical_example_1():
    # spec: nums = [1,2,4] -> 3
    # Because nums is circular, nums[0] and nums[2] are adjacent.
    # They have the maximum absolute difference of |4 - 1| = 3.
    assert maxAdjacentDistance([1, 2, 4]) == 3


def test_canonical_example_2():
    # spec: nums = [-5,-10,-5] -> 5
    # The adjacent elements nums[0] and nums[1] have the maximum
    # absolute difference of |-5 - (-10)| = 5.
    assert maxAdjacentDistance([-5, -10, -5]) == 5


def test_two_elements_positive():
    # Minimum length array with two positive elements
    # Adjacent pairs: (1, 2)
    # Max difference: |2 - 1| = 1
    assert maxAdjacentDistance([1, 2]) == 1


def test_two_elements_negative():
    # Two negative elements
    # Adjacent pairs: (-3, -7)
    # Max difference: |-3 - (-7)| = |4| = 4
    assert maxAdjacentDistance([-3, -7]) == 4


def test_all_same_elements():
    # All elements are the same
    # Adjacent pairs: (5, 5), (5, 5), (5, 5)
    # Max difference: |5 - 5| = 0
    assert maxAdjacentDistance([5, 5, 5]) == 0


def test_ascending_sequence():
    # Ascending sequence
    # Adjacent pairs: (1, 2), (2, 3), (3, 4), (4, 1) [circular]
    # Max difference: max(|2-1|, |3-2|, |4-3|, |1-4|) = max(1, 1, 1, 3) = 3
    assert maxAdjacentDistance([1, 2, 3, 4]) == 3


def test_descending_sequence():
    # Descending sequence
    # Adjacent pairs: (10, 7), (7, 4), (4, 1), (1, 10) [circular]
    # Max difference: max(|7-10|, |4-7|, |1-4|, |10-1|) = max(3, 3, 3, 9) = 9
    assert maxAdjacentDistance([10, 7, 4, 1]) == 9


def test_mixed_positive_negative():
    # Mixed positive and negative values
    # Adjacent pairs: (10, -5), (-5, 3), (3, 10) [circular]
    # Max difference: max(|-5-10|, |3-(-5)|, |10-3|) = max(15, 8, 7) = 15
    assert maxAdjacentDistance([10, -5, 3]) == 15


def test_zero_included():
    # Array including zero
    # Adjacent pairs: (0, 5), (5, -3), (-3, 0) [circular]
    # Max difference: max(|5-0|, |-3-5|, |0-(-3)|) = max(5, 8, 3) = 8
    assert maxAdjacentDistance([0, 5, -3]) == 8


def test_maximum_in_middle():
    # Maximum difference is in the middle of the array
    # Adjacent pairs: (1, 1), (1, 10), (10, 2), (2, 1) [circular]
    # Max difference: max(|1-1|, |10-1|, |2-10|, |1-2|) = max(0, 9, 8, 1) = 9
    assert maxAdjacentDistance([1, 1, 10, 2]) == 9


def test_maximum_at_circular_boundary():
    # Maximum difference is at the circular boundary (last to first)
    # Adjacent pairs: (50, 51), (51, 52), (52, 1) [circular]
    # Max difference: max(|51-50|, |52-51|, |1-52|) = max(1, 1, 51) = 51
    assert maxAdjacentDistance([50, 51, 52, 1]) == 51
