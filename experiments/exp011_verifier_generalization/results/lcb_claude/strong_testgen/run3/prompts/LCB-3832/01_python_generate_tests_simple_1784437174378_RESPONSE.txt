from typing import List
import pytest
from solution import minCosts


def test_canonical_example_1():
    # spec: cost = [5,3,4,1,3,2] -> [5,3,3,1,1,1]
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]


def test_canonical_example_2():
    # spec: cost = [1,2,4,6,7] -> [1,1,1,1,1]
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]


def test_single_person():
    # n=1: only person 0, must pay cost[0] to reach position 0
    assert minCosts([10]) == [10]


def test_two_people_ascending():
    # n=2: cost = [1, 2]
    # pos 0: swap with person 0, cost 1
    # pos 1: swap with person 1, cost 2
    assert minCosts([1, 2]) == [1, 2]


def test_two_people_descending():
    # n=2: cost = [2, 1]
    # pos 0: swap with person 0, cost 2
    # pos 1: can swap with person 1 (cost 1) OR swap with person 0 (cost 2) then free swap with person 1
    # min(1, 2) = 1
    assert minCosts([2, 1]) == [2, 1]


def test_all_same_cost():
    # cost = [5, 5, 5]
    # pos 0: swap with person 0, cost 5
    # pos 1: swap with person 1, cost 5
    # pos 2: swap with person 2, cost 5
    assert minCosts([5, 5, 5]) == [5, 5, 5]


def test_decreasing_costs():
    # cost = [10, 5, 3, 1]
    # pos 0: swap with person 0, cost 10
    # pos 1: swap with person 1, cost 5
    # pos 2: swap with person 2, cost 3
    # pos 3: swap with person 3, cost 1
    # Once we swap with person 3 (cost 1), we can reach 0, 1, 2 for free
    # So pos 0, 1, 2 should be min(direct_cost, cost_to_reach_3)
    assert minCosts([10, 5, 3, 1]) == [1, 1, 1, 1]


def test_increasing_costs():
    # cost = [1, 3, 5, 10]
    # pos 0: swap with person 0, cost 1
    # After swapping with person 0, we're at pos 0 and can swap for free with anyone behind
    # pos 1: min(3, 1) = 1
    # pos 2: min(5, 1) = 1
    # pos 3: min(10, 1) = 1
    assert minCosts([1, 3, 5, 10]) == [1, 1, 1, 1]


def test_minimum_in_middle():
    # cost = [10, 8, 2, 9, 7]
    # pos 0: 10
    # pos 1: 8
    # pos 2: 2 (direct swap)
    # pos 3: min(9, 2) = 2 (swap with person 2, then free swap)
    # pos 4: min(7, 2) = 2 (swap with person 2, then free swap)
    assert minCosts([10, 8, 2, 9, 7]) == [2, 2, 2, 2, 2]


def test_constraint_minimum_n():
    # n=1 (minimum constraint): cost length 1
    assert minCosts([50]) == [50]


def test_constraint_maximum_cost():
    # cost values at maximum (100)
    assert minCosts([100, 100]) == [100, 100]


def test_constraint_minimum_cost():
    # cost values at minimum (1)
    assert minCosts([1, 1, 1]) == [1, 1, 1]
