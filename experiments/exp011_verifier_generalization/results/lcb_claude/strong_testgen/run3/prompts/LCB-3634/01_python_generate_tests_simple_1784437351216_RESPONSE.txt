from solution import calculateScore


def test_canonical_example_1():
    # spec: s = "aczzx" -> 5
    # i=0: 'a', no mirror before
    # i=1: 'c', no mirror before
    # i=2: 'z', mirror is 'a' at j=0, mark both, score += 2-0 = 2
    # i=3: 'z', mirror is 'a', but j=0 is marked, no valid j
    # i=4: 'x', mirror is 'c' at j=1, mark both, score += 4-1 = 3
    # total: 2 + 3 = 5
    assert calculateScore("aczzx") == 5


def test_canonical_example_2():
    # spec: s = "abcdef" -> 0
    # For each index i, there is no index j that satisfies the conditions
    assert calculateScore("abcdef") == 0


def test_single_character():
    # s = "a", only one character, no j < i exists
    assert calculateScore("a") == 0


def test_two_mirrors():
    # s = "az", i=0: 'a' no j<0, i=1: 'z' mirror is 'a' at j=0, score = 1-0 = 1
    assert calculateScore("az") == 1


def test_two_non_mirrors():
    # s = "ab", i=0: 'a' no j<0, i=1: 'b' mirror is 'y', not present
    assert calculateScore("ab") == 0


def test_mirror_same_letter():
    # s = "mm", mirror of 'm' (13th letter) is 'n' (14th letter), not 'm'
    # So no matches
    assert calculateScore("mm") == 0


def test_multiple_same_mirrors():
    # s = "azza", 
    # i=0: 'a' no j<0
    # i=1: 'z' mirror='a' at j=0, mark 0,1, score = 1
    # i=2: 'z' mirror='a', j=0 marked, no valid j
    # i=3: 'a' mirror='z', closest unmarked is j=2, mark 2,3, score += 3-2=1
    # total = 1 + 1 = 2
    assert calculateScore("azza") == 2


def test_closest_unmarked():
    # s = "azaz"
    # i=0: 'a' no j<0
    # i=1: 'z' mirror='a' at j=0, mark 0,1, score = 1
    # i=2: 'a' mirror='z', j=1 marked, no unmarked j
    # i=3: 'z' mirror='a', j=2 is 'a' and unmarked, mark 2,3, score += 3-2=1
    # total = 1 + 1 = 2
    assert calculateScore("azaz") == 2


def test_all_same_letter():
    # s = "aaaa", mirror of 'a' is 'z', no 'z' present
    assert calculateScore("aaaa") == 0


def test_reversed_alphabet_pair():
    # s = "za", i=0: 'z' no j<0, i=1: 'a' mirror='z' at j=0, score = 1-0 = 1
    assert calculateScore("za") == 1


def test_middle_letter_mirror():
    # 'm' is 13th letter (0-indexed: 12), 'n' is 14th (0-indexed: 13)
    # mirror of 'm' (position 12) is position 25-12=13 which is 'n'
    # s = "mn", i=0: 'm' no j<0, i=1: 'n' mirror='m' at j=0, score = 1
    assert calculateScore("mn") == 1


def test_longer_distance():
    # s = "a___z" (with other chars), distance should be larger
    # s = "abcdz", i=4: 'z' mirror='a' at j=0, score = 4
    assert calculateScore("abcdz") == 4


def test_multiple_pairs():
    # s = "azbycx"
    # i=0: 'a' no j<0
    # i=1: 'z' mirror='a' at j=0, mark 0,1, score = 1
    # i=2: 'b' mirror='y', not present yet
    # i=3: 'y' mirror='b' at j=2, mark 2,3, score += 3-2=1
    # i=4: 'c' mirror='x', not present yet
    # i=5: 'x' mirror='c' at j=4, mark 4,5, score += 5-4=1
    # total = 1 + 1 + 1 = 3
    assert calculateScore("azbycx") == 3
