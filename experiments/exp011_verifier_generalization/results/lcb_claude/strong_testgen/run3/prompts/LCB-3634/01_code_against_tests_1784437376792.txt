## Implementation

```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    for i in range(len(s)):
        if marked[i]:
            continue
        mirror = chr(ord('z') - (ord(s[i]) - ord('a')))
        for j in range(i - 1, -1, -1):
            if marked[j]:
                continue
            if s[j] == mirror:
                marked[i] = marked[j] = True
                score += i - j
                break
    return score

```
