from typing import List

def maxSum(nums: List[int]) -> int:
    unique_elements = set()
    max_sum = 0
    current_sum = 0

    for num in nums:
        if num not in unique_elements:
            unique_elements.add(num)
            current_sum += num
        else:
            # If we encounter a duplicate, we need to reset the current sum.
            # We should keep adding elements until we find the duplicate again.
            # This means we need to remove elements from current_sum until we can add this element.
            while nums and nums[0] != num:
                current_sum -= nums.pop(0)
            # Now we can remove the duplicate
            current_sum -= num
            unique_elements.remove(num)
            # Add the current number and reset max_sum if necessary
            current_sum += num
            unique_elements.add(num)

        max_sum = max(max_sum, current_sum)

    return max_sum
```
