import pytest
from typing import List
from solution import maxSum


def test_canonical_example_1():
    # spec: nums = [1,2,3,4,5] -> 15
    # Select the entire array without deleting any element
    assert maxSum([1, 2, 3, 4, 5]) == 15


def test_canonical_example_2():
    # spec: nums = [1,1,0,1,1] -> 1
    # Delete nums[0], nums[1], nums[2], nums[3], select [1]
    assert maxSum([1, 1, 0, 1, 1]) == 1


def test_canonical_example_3():
    # spec: nums = [1,2,-1,-2,1,0,-1] -> 3
    # Delete nums[2] == -1 and nums[3] == -2, select subarray [2, 1] from [1, 2, 1, 0, -1]
    assert maxSum([1, 2, -1, -2, 1, 0, -1]) == 3


def test_single_element():
    # Single element array, sum is the element itself
    assert maxSum([5]) == 5


def test_single_element_negative():
    # Single element negative array, sum is the negative element
    assert maxSum([-10]) == -10


def test_all_negative_elements():
    # All negative, best is to select single largest (least negative) element
    # [-5, -3, -8] -> select [-3] for sum of -3
    assert maxSum([-5, -3, -8]) == -3


def test_all_same_positive():
    # All same positive value, select one element
    # [7, 7, 7] -> select [7] for sum of 7
    assert maxSum([7, 7, 7]) == 7


def test_all_same_negative():
    # All same negative value, select one element
    # [-2, -2, -2] -> select [-2] for sum of -2
    assert maxSum([-2, -2, -2]) == -2


def test_mixed_with_duplicates():
    # Mixed positive and negative with duplicates
    # [3, -1, 3, 4] -> delete one 3, select subarray with unique elements
    # Best is [3, 4] or [-1, 3, 4] = 6
    assert maxSum([3, -1, 3, 4]) == 6


def test_two_elements_unique():
    # Two unique elements, select both
    # [10, 20] -> 30
    assert maxSum([10, 20]) == 30


def test_two_elements_duplicate():
    # Two duplicate elements, select one
    # [5, 5] -> 5
    assert maxSum([5, 5]) == 5


def test_all_zeros():
    # All zeros, sum is 0
    # [0, 0, 0] -> 0
    assert maxSum([0, 0, 0]) == 0


def test_with_zero_in_middle():
    # Array with zero in middle
    # [1, 0, 2] -> select [1, 0, 2] = 3 or skip zero [1, 2] after deletion = 3
    # After deleting 0: [1, 2] sum = 3
    assert maxSum([1, 0, 2]) == 3


def test_positive_negative_alternating():
    # Alternating positive and negative
    # [5, -2, 6, -1, 7] -> select positive subarrays
    # Best is [5, 6, 7] after deleting negatives = 18
    assert maxSum([5, -2, 6, -1, 7]) == 18


def test_large_values_at_boundary():
    # Test with boundary values
    # [100, -100, 100] -> select [100, -100, 100] = 100 or just [100] = 100
    # Delete -100: [100, 100] then select one [100] = 100
    assert maxSum([100, -100, 100]) == 100


def test_increasing_sequence():
    # Simple increasing sequence
    # [1, 2, 3] -> 6
    assert maxSum([1, 2, 3]) == 6


def test_decreasing_sequence():
    # Simple decreasing sequence
    # [3, 2, 1] -> 6
    assert maxSum([3, 2, 1]) == 6
