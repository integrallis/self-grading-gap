from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[[float('-inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0] if coins[0][0] >= 0 else 0

    for i in range(m):
        for j in range(n):
            for k in range(3):  # k is the number of neutralizations used
                if i == 0 and j == 0:
                    continue
                gain = coins[i][j]
                if gain < 0:
                    if k > 0:
                        gain = 0  # Neutralize robber
                    else:
                        gain = -gain  # Lose coins
                if i > 0:
                    dp[i][j][k] = max(dp[i][j][k], dp[i - 1][j][k] + gain)
                if j > 0:
                    dp[i][j][k] = max(dp[i][j][k], dp[i][j - 1][k] + gain)
                if gain < 0 and k > 0:
                    if i > 0:
                        dp[i][j][k] = max(dp[i][j][k], dp[i - 1][j][k - 1] + coins[i][j])
                    if j > 0:
                        dp[i][j][k] = max(dp[i][j][k], dp[i][j - 1][k - 1] + coins[i][j])

    return max(dp[m - 1][n - 1])
```
