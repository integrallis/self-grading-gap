import pytest
from typing import List
from solution import countMentions


def test_canonical_example_1():
    # spec: numberOfUsers=2, events=[["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","71","HERE"]]
    # At t=10: id1 and id0 mentioned -> [1,1]
    # At t=11: id0 goes offline (until t=71)
    # At t=71: id0 comes back online, "HERE" mentions all online users (both 0 and 1) -> [2,2]
    assert countMentions(2, [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "71", "HERE"]]) == [2, 2]


def test_canonical_example_2():
    # spec: numberOfUsers=2, events=[["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","12","ALL"]]
    # At t=10: id1 and id0 mentioned -> [1,1]
    # At t=11: id0 goes offline (until t=71)
    # At t=12: "ALL" mentions all users (including offline) -> [2,2]
    assert countMentions(2, [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "12", "ALL"]]) == [2, 2]


def test_canonical_example_3():
    # spec: numberOfUsers=2, events=[["OFFLINE","10","0"],["MESSAGE","12","HERE"]]
    # At t=10: id0 goes offline (until t=70)
    # At t=12: "HERE" mentions only online users (only id1) -> [0,1]
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "12", "HERE"]]) == [0, 1]


def test_single_user_single_mention():
    # Single user, single MESSAGE event mentioning that user
    # id0 mentioned once -> [1]
    assert countMentions(1, [["MESSAGE", "5", "id0"]]) == [1]


def test_multiple_mentions_same_message():
    # Spec: "a user can be mentioned multiple times in a single message event, and each mention should be counted separately"
    # id0 mentioned 3 times in one message -> [3]
    assert countMentions(1, [["MESSAGE", "5", "id0 id0 id0"]]) == [3]


def test_all_mentions_all_users():
    # "ALL" should mention all users
    # 3 users, "ALL" mentioned -> [1,1,1]
    assert countMentions(3, [["MESSAGE", "5", "ALL"]]) == [1, 1, 1]


def test_here_mentions_only_online():
    # "HERE" should mention only online users
    # 3 users, id1 offline, "HERE" at t=15 -> id0 and id2 mentioned -> [1,0,1]
    assert countMentions(3, [["OFFLINE", "10", "1"], ["MESSAGE", "15", "HERE"]]) == [1, 0, 1]


def test_offline_duration_exactly_60():
    # User goes offline at t=10, comes back at t=70
    # At t=69: user still offline, HERE doesn't mention them
    # At t=70: user back online, HERE mentions them
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "69", "HERE"]]) == [0, 1]
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "70", "HERE"]]) == [1, 1]


def test_multiple_offline_events():
    # Multiple users going offline at different times
    # id0 offline at t=10 (until t=70), id1 offline at t=20 (until t=80)
    # At t=30: HERE mentions no one offline -> [0,0]
    assert countMentions(2, [["OFFLINE", "10", "0"], ["OFFLINE", "20", "1"], ["MESSAGE", "30", "HERE"]]) == [0, 0]


def test_mixed_id_and_special_mentions():
    # Combination of id mentions and special keywords
    # id0 mentioned explicitly, then ALL mentioned
    # id0 gets 2 mentions, id1 gets 1 -> [2,1]
    assert countMentions(2, [["MESSAGE", "10", "id0"], ["MESSAGE", "20", "ALL"]]) == [2, 1]


def test_no_mentions_for_user():
    # Some users never mentioned
    # Only id1 mentioned -> [0,1,0]
    assert countMentions(3, [["MESSAGE", "10", "id1"]]) == [0, 1, 0]


def test_all_mentions_offline_users_too():
    # "ALL" mentions all users, even offline ones (per example 2)
    # id0 offline, "ALL" still mentions id0 -> [1,1]
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "20", "ALL"]]) == [1, 1]


def test_status_change_before_message_same_timestamp():
    # Spec: "if a user goes offline or comes back online, their status change is processed before handling any message event that occurs at the same timestamp"
    # User goes offline at t=10, message at t=10 with HERE -> user is offline, not mentioned
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "10", "HERE"]]) == [0, 1]


def test_user_comes_back_online_same_timestamp_as_message():
    # User offline at t=10 (comes back at t=70), message at t=70 with HERE
    # Status change processed first, so user is online and mentioned
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "70", "HERE"]]) == [1, 1]


def test_multiple_messages_accumulate_mentions():
    # Multiple MESSAGE events, mentions accumulate

    # "id0 id1 id0" -> id0 mentioned twice, id1 once -> [2,1]
    assert countMentions(2, [["MESSAGE", "10", "id0 id1 id0"]]) == [2, 1]
