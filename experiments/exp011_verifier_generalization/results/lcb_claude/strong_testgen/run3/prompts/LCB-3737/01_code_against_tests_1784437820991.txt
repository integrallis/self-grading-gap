from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize dp arrays for the minimum cost painting the houses with different color
    dp = [[0] * 3 for _ in range(n)]

    # Initialize the first house costs
dp[0] = cost[0]

    # Fill the dp table taking care of adjacent house constraints
    for i in range(1, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(dp[i - 1][(j + 1) % 3], dp[i - 1][(j + 2) % 3])

    # Calculate the total minimum cost considering equidistant pairs
    total_cost = float('inf')
    for j in range(3):
        # Must ensure the first and last houses are not painted the same
        # Calculate the cost with the first house in color j and find the best option for last house
        curr_cost = dp[n - 1][(j + 1) % 3] + cost[0][j]  # last house must have different color
        curr_cost2 = dp[n - 1][(j + 2) % 3] + cost[0][j]
        # Check against equidistant constraints
        total_cost = min(total_cost, curr_cost, curr_cost2)

    return total_cost
