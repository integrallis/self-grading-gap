from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # dp[i][j] will hold the minimum cost to paint up to house i with color j
    dp = [[0] * 3 for _ in range(n)]

    # Initialize the first house costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the dp table
    for i in range(1, n):
        for j in range(3):
            min_cost = float('inf')
            for k in range(3):
                if j != k:  # Ensure no adjacent houses have the same color
                    min_cost = min(min_cost, dp[i - 1][k])
            dp[i][j] = cost[i][j] + min_cost

    # Handle equidistant constraints
    min_cost = float('inf')
    for j in range(3):
        for k in range(3):
            if j != k:  # First and last house colors must differ
                total_cost = dp[n - 1][k] + cost[0][j]
                if k != (n // 2 + j) % 3:  # Ensure equidistant pairs differ
                    min_cost = min(min_cost, total_cost)

    return min_cost

```
