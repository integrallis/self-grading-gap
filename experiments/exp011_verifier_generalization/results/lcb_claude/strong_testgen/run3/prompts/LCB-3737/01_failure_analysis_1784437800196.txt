from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # dp[i][j] will hold the minimum cost to paint up to house i with color j
    dp = [[0] * 3 for _ in range(n)]

    # Initialize the first house costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the dp table
    for i in range(1, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(
                dp[i - 1][(j + 1) % 3],  # Not same color as previous
                dp[i - 1][(j + 2) % 3]   # Not same color as previous
            )

    # Handle equidistant constraints
    # For the last house, we need to consider the first house's color
    min_cost = float('inf')
    for j in range(3):
        for k in range(3):
            if j != k:  # First and last house colors must differ
                total_cost = dp[n - 1][k] + cost[0][j]
                min_cost = min(min_cost, total_cost)

    return min_cost
