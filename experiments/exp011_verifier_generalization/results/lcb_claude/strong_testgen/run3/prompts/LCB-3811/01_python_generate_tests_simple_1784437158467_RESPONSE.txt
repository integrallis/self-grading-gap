import pytest
from solution import reverseDegree


def test_canonical_example_1():
    # spec: "abc" -> 148
    # 'a'=26*1=26, 'b'=25*2=50, 'c'=24*3=72, sum=148
    assert reverseDegree("abc") == 148


def test_canonical_example_2():
    # spec: "zaza" -> 160
    # 'z'=1*1=1, 'a'=26*2=52, 'z'=1*3=3, 'a'=26*4=104, sum=160
    assert reverseDegree("zaza") == 160


def test_single_character_a():
    # 'a' at position 1: 26*1=26
    assert reverseDegree("a") == 26


def test_single_character_z():
    # 'z' at position 1: 1*1=1
    assert reverseDegree("z") == 1


def test_single_character_m():
    # 'm' is 13th letter, reversed position is 26-13+1=14
    # 14*1=14
    assert reverseDegree("m") == 14


def test_repeated_same_character():
    # "aaa": 'a'=26*1=26, 'a'=26*2=52, 'a'=26*3=78, sum=156
    assert reverseDegree("aaa") == 156


def test_all_z():
    # "zzz": 'z'=1*1=1, 'z'=1*2=2, 'z'=1*3=3, sum=6
    assert reverseDegree("zzz") == 6

    # "zyx": 'z'=1*1=1, 'y'=2*2=4, 'x'=3*3=9, sum=14
    assert reverseDegree("zyx") == 14


def test_longer_string():
    # "abcd": 'a'=26*1=26, 'b'=25*2=50, 'c'=24*3=72, 'd'=23*4=92, sum=240
    assert reverseDegree("abcd") == 240


def test_middle_of_alphabet():
