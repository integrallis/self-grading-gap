import pytest
from solution import maxDistance


def test_canonical_example_1():
    # spec: s = "NWSE", k = 1 -> 3
    # Change s[2] from 'S' to 'N' to get "NWNE"
    # Path: (0,0) -> N -> (0,1) [dist=1] -> W -> (-1,1) [dist=2] -> N -> (-1,2) [dist=3] -> E -> (0,2) [dist=2]
    # Maximum distance is 3
    assert maxDistance("NWSE", 1) == 3


def test_canonical_example_2():
    # spec: s = "NSWWEW", k = 3 -> 6
    # Change s[1] from 'S' to 'N' and s[4] from 'E' to 'W' to get "NNWWWW"
    # Maximum distance is 6
    assert maxDistance("NSWWEW", 3) == 6

    # Maximum distance is 2
    assert maxDistance("NESW", 0) == 2

    # Maximum distance is 3
    assert maxDistance("EEE", 0) == 3


def test_longer_string_with_changes():
    # s = "EEEE", k = 2
    # Can change 2 moves, e.g., keep all as E or change some to continue in same direction
    # Best: all moving east gives (4,0) with distance 4
    # Maximum distance is 4
    assert maxDistance("EEEE", 2) == 4


def test_changes_allow_longer_path():
    # s = "EW", k = 1
    # Original: (0,0) -> E -> (1,0) [dist=1] -> W -> (0,0) [dist=0], max=1
    # Change W to E: (0,0) -> E -> (1,0) [dist=1] -> E -> (2,0) [dist=2], max=2
    # Maximum distance is 2
    assert maxDistance("EW", 1) == 2


def test_mix_of_directions():

    # Path: (0,0) -> N -> (0,1) [dist=1] -> N -> (0,2) [dist=2] -> E -> (1,2) [dist=3] -> E -> (2,2) [dist=4] -> S -> (2,1) [dist=3] -> S -> (2,0) [dist=2]
    # Maximum distance is 4
    assert maxDistance("NNEESS", 0) == 4


def test_k_greater_than_needed():
    # s = "N", k = 10
    # Can only change 1 move but k=10, so effectively k=1
    # Best: keep as N or change to any direction, distance is 1
    # Maximum distance is 1
    assert maxDistance("N", 10) == 1
