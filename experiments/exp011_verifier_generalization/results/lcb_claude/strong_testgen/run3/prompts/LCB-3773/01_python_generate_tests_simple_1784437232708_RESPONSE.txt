from typing import List
import pytest
from solution import minimumPairRemoval


def test_canonical_example_1():
    # spec: nums = [5,2,3,1] -> 2
    # (3,1) sum=4 -> [5,2,4]
    # (2,4) sum=6 -> [5,6]
    # non-decreasing in 2 operations
    assert minimumPairRemoval([5, 2, 3, 1]) == 2


def test_canonical_example_2():
    # spec: nums = [1,2,2] -> 0
    # already non-decreasing
    assert minimumPairRemoval([1, 2, 2]) == 0


def test_single_element():
    # single element is already non-decreasing
    assert minimumPairRemoval([5]) == 0


def test_two_elements_non_decreasing():
    # [1, 2] is already non-decreasing
    assert minimumPairRemoval([1, 2]) == 0


def test_two_elements_decreasing():
    # [2, 1] needs one operation -> [3]
    assert minimumPairRemoval([2, 1]) == 1


def test_all_same_elements():
    # [3, 3, 3, 3] is already non-decreasing
    assert minimumPairRemoval([3, 3, 3, 3]) == 0


def test_strictly_decreasing():
    # [4, 3, 2, 1]
    # (2,1) sum=3 -> [4,3,3]
    # (3,3) sum=6 (leftmost among equal sums) -> [4,6]
    # non-decreasing in 2 operations
    assert minimumPairRemoval([4, 3, 2, 1]) == 2


def test_already_sorted_increasing():
    # [1, 2, 3, 4, 5] is already non-decreasing
    assert minimumPairRemoval([1, 2, 3, 4, 5]) == 0


def test_negative_numbers():
    # [-5, -2, -3, -1]
    # (-3,-1) sum=-4 -> [-5,-2,-4]
    # (-5,-2) sum=-7 -> [-7,-4]
    # non-decreasing in 2 operations
    assert minimumPairRemoval([-5, -2, -3, -1]) == 2


def test_mixed_positive_negative():
    # [-1, 2, 1]
    # (-1,2) sum=1, (2,1) sum=3, minimum is 1
    # -> [1, 1]
    # non-decreasing in 1 operation
    assert minimumPairRemoval([-1, 2, 1]) == 1


def test_three_elements_one_violation():
    # [1, 3, 2]
    # (3,2) sum=5 -> [1,5]
    # non-decreasing in 1 operation
    assert minimumPairRemoval([1, 3, 2]) == 1


def test_leftmost_selection():
    # [5, 1, 4, 1]
    # (5,1) sum=6, (1,4) sum=5, (4,1) sum=5
    # minimum is 5, leftmost is (1,4) -> [5,5,1]
    # (5,5) sum=10, (5,1) sum=6
    # minimum is 6 -> [5,6]
    # non-decreasing in 2 operations
    assert minimumPairRemoval([5, 1, 4, 1]) == 2
