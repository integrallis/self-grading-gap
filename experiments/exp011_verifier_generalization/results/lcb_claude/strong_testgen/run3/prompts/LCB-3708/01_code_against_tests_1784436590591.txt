```python
from typing import List

def zigzagTraversal(grid: List[List[int]]) -> List[int]:
    result = []
    rows = len(grid)
    for i in range(rows):
        if i % 2 == 0:
            # Traverse right
            for j in range(len(grid[i])):
                if j % 2 == 0:
                    result.append(grid[i][j])
        else:
            # Traverse left
            for j in range(len(grid[i]) - 1, -1, -1):
                if j % 2 == 1:
                    result.append(grid[i][j])
    return result
```
