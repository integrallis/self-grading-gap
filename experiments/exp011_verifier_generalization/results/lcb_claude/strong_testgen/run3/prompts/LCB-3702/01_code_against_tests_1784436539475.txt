```python
from typing import List
import math


def maxLength(nums: List[int]) -> int:
    n = len(nums)
    max_len = 0
    for start in range(n):
        prod = 1
        gcd = nums[start]
        lcm = nums[start]
        for end in range(start, n):
            prod *= nums[end]
            gcd = math.gcd(gcd, nums[end])
            lcm = (lcm * nums[end]) // math.gcd(lcm, nums[end])
            if prod == lcm * gcd:
                max_len = max(max_len, end - start + 1)
    return max_len
```
