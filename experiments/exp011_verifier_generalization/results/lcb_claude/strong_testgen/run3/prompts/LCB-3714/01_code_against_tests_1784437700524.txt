```python
from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total = 0

    # Calculate the contribution of each number in nums
    for i in range(n):
        # Each number can be a min or max in several subsequences
        # The number of subsequences that can include nums[i] as a min or max
        # is given by the combinations of the other elements
        count_as_min = (1 << (n - 1)) - (1 << (n - k - 1)) if n > k else (1 << (n - 1))
        count_as_max = (1 << (n - 1)) - (1 << (n - k - 1)) if n > k else (1 << (n - 1))
        total += (count_as_min * nums[i] + count_as_max * nums[i]) % MOD
        total %= MOD

    return total % MOD

```
