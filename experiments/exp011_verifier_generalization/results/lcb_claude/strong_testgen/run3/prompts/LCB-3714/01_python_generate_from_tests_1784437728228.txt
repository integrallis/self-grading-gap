from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total = 0

    # Pre-compute the contribution of each number in nums based on combinations
    for i in range(n):
        # Count how many times nums[i] is the minimum or maximum
        # Each element can be included in subsequences of size up to k
        count_as_min = 0
        count_as_max = 0

        # For counting contributions of nums[i] as min or max in subsequences
        # Total number of ways to choose (k-1) from (n-1) gives total subsequences of size k
        if n > k:
            count_as_min = pow(2, n - 1, MOD) - pow(2, n - k - 1, MOD)
            count_as_max = pow(2, n - 1, MOD) - pow(2, n - k - 1, MOD)
        else:
            count_as_min = pow(2, n - 1, MOD)
            count_as_max = pow(2, n - 1, MOD)

        total += (count_as_min * nums[i]) % MOD
        total += (count_as_max * nums[i]) % MOD
        total %= MOD

    return total % MOD

```
