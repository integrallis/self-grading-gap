from typing import List
import pytest
from solution import minMaxSums


def test_canonical_example_1():
    # spec: nums = [1,2,3], k = 2 -> 24
    # Subsequences with at most 2 elements:
    # [1]: min=1, max=1, sum=2
    # [2]: min=2, max=2, sum=4
    # [3]: min=3, max=3, sum=6
    # [1,2]: min=1, max=2, sum=3
    # [1,3]: min=1, max=3, sum=4
    # [2,3]: min=2, max=3, sum=5
    # Total: 2+4+6+3+4+5 = 24
    assert minMaxSums([1, 2, 3], 2) == 24


def test_canonical_example_2():
    # spec: nums = [5,0,6], k = 1 -> 22
    # Subsequences with exactly 1 element (at most 1):
    # [5]: min=5, max=5, sum=10
    # [0]: min=0, max=0, sum=0
    # [6]: min=6, max=6, sum=12
    # Total: 10+0+12 = 22
    assert minMaxSums([5, 0, 6], 1) == 22


def test_canonical_example_3():
    # spec: nums = [1,1,1], k = 2 -> 12
    # Subsequences with at most 2 elements:
    # [1] appears 3 times: min=1, max=1, sum=2 each -> 6
    # [1,1] appears 3 times: min=1, max=1, sum=2 each -> 6
    # Total: 6+6 = 12
    assert minMaxSums([1, 1, 1], 2) == 12


def test_single_element_array_k_1():
    # nums = [42], k = 1
    # Only subsequence: [42], min=42, max=42, sum=84
    assert minMaxSums([42], 1) == 84


def test_two_elements_k_2():
    # nums = [1, 5], k = 2
    # Subsequences: [1] (min=1,max=1,sum=2), [5] (min=5,max=5,sum=10), [1,5] (min=1,max=5,sum=6)
    # Total: 2+10+6 = 18
    assert minMaxSums([1, 5], 2) == 18


def test_k_equals_array_length():
    # nums = [2, 3, 4], k = 3
    # All subsequences with at most 3 elements (all possible subsequences)
    # [2]: 4, [3]: 6, [4]: 8
    # [2,3]: 5, [2,4]: 6, [3,4]: 7
    # [2,3,4]: 6
    # Total: 4+6+8+5+6+7+6 = 42
    assert minMaxSums([2, 3, 4], 3) == 42


def test_large_modulo():
    # Test that result is returned modulo 10^9 + 7
    # With large enough input, the sum should require modulo
    # Using a case where we can compute expected result
    nums = [1000000000] * 10
    k = 2
    # Each single element: min=10^9, max=10^9, sum=2*10^9
    # 10 such subsequences: 10 * 2*10^9 = 2*10^10
    # Each pair: min=10^9, max=10^9, sum=2*10^9
    # C(10,2) = 45 such pairs: 45 * 2*10^9 = 9*10^10
    # Total: 11*10^10
    # 11*10^10 mod (10^9+7) = 110000000000 mod 1000000007
    expected = (11 * 10**10) % (10**9 + 7)
    assert minMaxSums(nums, k) == expected


def test_all_zeros():
    # nums = [0, 0, 0], k = 2
    # All subsequences have min=0, max=0, sum=0
    # Total: 0
    assert minMaxSums([0, 0, 0], 2) == 0


def test_different_values_k_1():
    # nums = [10, 20, 30], k = 1
    # [10]: 20, [20]: 40, [30]: 60
    # Total: 120
    assert minMaxSums([10, 20, 30], 1) == 120


def test_four_elements_k_3():
    # nums = [1, 2, 3, 4], k = 3
    # Size 1: [1]:2, [2]:4, [3]:6, [4]:8 -> 20
    # Size 2: [1,2]:3, [1,3]:4, [1,4]:5, [2,3]:5, [2,4]:6, [3,4]:7 -> 30
    # Size 3: [1,2,3]:4, [1,2,4]:5, [1,3,4]:5, [2,3,4]:6 -> 20
    # Total: 70
    assert minMaxSums([1, 2, 3, 4], 3) == 70
