from typing import List
import pytest
from solution import transformArray


def test_canonical_example_1():
    # spec: [4,3,2,1] -> [0,0,1,1]
    # 4 even->0, 3 odd->1, 2 even->0, 1 odd->1 = [0,1,0,1]
    # sorted = [0,0,1,1]
    assert transformArray([4, 3, 2, 1]) == [0, 0, 1, 1]


def test_canonical_example_2():
    # spec: [1,5,1,4,2] -> [0,0,1,1,1]
    # 1 odd->1, 5 odd->1, 1 odd->1, 4 even->0, 2 even->0 = [1,1,1,0,0]
    # sorted = [0,0,1,1,1]
    assert transformArray([1, 5, 1, 4, 2]) == [0, 0, 1, 1, 1]


def test_single_even_number():
    # [2] -> 2 even->0 = [0], sorted = [0]
    assert transformArray([2]) == [0]


def test_single_odd_number():
    # [3] -> 3 odd->1 = [1], sorted = [1]
    assert transformArray([3]) == [1]


def test_all_even_numbers():
    # [2,4,6] -> all even->0 = [0,0,0], sorted = [0,0,0]
    assert transformArray([2, 4, 6]) == [0, 0, 0]


def test_all_odd_numbers():
    # [1,3,5] -> all odd->1 = [1,1,1], sorted = [1,1,1]
    assert transformArray([1, 3, 5]) == [1, 1, 1]


def test_mixed_numbers_already_sorted_result():
    # [2,4,1,3] -> [0,0,1,1], sorted = [0,0,1,1]
    assert transformArray([2, 4, 1, 3]) == [0, 0, 1, 1]


def test_large_numbers_within_constraint():
    # [1000,999,500,1] -> 1000 even->0, 999 odd->1, 500 even->0, 1 odd->1 = [0,1,0,1]
    # sorted = [0,0,1,1]
    assert transformArray([1000, 999, 500, 1]) == [0, 0, 1, 1]


def test_minimum_constraint_value():
    # [1] -> 1 odd->1 = [1], sorted = [1]
    assert transformArray([1]) == [1]


def test_array_length_two():
    # [2,3] -> 2 even->0, 3 odd->1 = [0,1], sorted = [0,1]
    assert transformArray([2, 3]) == [0, 1]


def test_array_length_two_reverse():
    # [3,2] -> 3 odd->1, 2 even->0 = [1,0], sorted = [0,1]
    assert transformArray([3, 2]) == [0, 1]
