import pytest
from solution import maxContainers


def test_canonical_example_1():
    # spec: n=2, w=3, maxWeight=15
    # deck has 2*2=4 cells, each container weighs 3
    # total weight if all loaded: 4*3=12, which is <= 15
    # so all 4 containers can be loaded
    assert maxContainers(2, 3, 15) == 4


def test_canonical_example_2():
    # spec: n=3, w=5, maxWeight=20
    # deck has 3*3=9 cells, each container weighs 5
    # max containers without exceeding maxWeight: 20//5 = 4
    # (5 containers would weigh 25 > 20)
    assert maxContainers(3, 5, 20) == 4


def test_single_cell_deck():
    # n=1 means 1x1 deck with 1 cell
    # if w <= maxWeight, 1 container can be loaded
    assert maxContainers(1, 5, 10) == 1


def test_weight_limit_less_than_one_container():
    # deck has 2*2=4 cells, but maxWeight only allows 0 containers
    # maxWeight=2, w=5: 2//5 = 0
    assert maxContainers(2, 5, 2) == 0


def test_weight_limit_exactly_one_container():
    # deck has 3*3=9 cells, but maxWeight allows exactly 1
    # maxWeight=7, w=7: 7//7 = 1
    assert maxContainers(3, 7, 7) == 1


def test_weight_limit_allows_all_containers():
    # deck has 2*2=4 cells, maxWeight allows all 4
    # 4 containers * 10 weight = 40 <= 50
    assert maxContainers(2, 10, 50) == 4


def test_weight_limit_constrains_below_deck_capacity():
    # deck has 5*5=25 cells, but weight only allows 10 containers
    # 10 containers * 3 weight = 30 <= 30
    assert maxContainers(5, 3, 30) == 10


def test_large_deck_small_weight_capacity():
    # deck has 10*10=100 cells, but weight allows only 5
    # 5 containers * 2 weight = 10 <= 10
    assert maxContainers(10, 2, 10) == 5


def test_minimum_inputs():
    # n=1, w=1, maxWeight=1: 1 cell, 1 weight, allows 1 container
    assert maxContainers(1, 1, 1) == 1


def test_large_deck_large_capacity():
    # deck has 100*100=10000 cells, weight allows all
    # 10000 containers * 50 weight = 500000 <= 1000000
    assert maxContainers(100, 50, 1000000) == 10000


def test_weight_limit_between_capacities():
    # deck has 4*4=16 cells, weight allows 7 containers
    # 7 containers * 5 weight = 35 <= 37
    # 8 containers * 5 weight = 40 > 37
    assert maxContainers(4, 5, 37) == 7
