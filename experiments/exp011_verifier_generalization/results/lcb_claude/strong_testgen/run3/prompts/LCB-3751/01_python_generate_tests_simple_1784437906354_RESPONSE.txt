from typing import List
import pytest
from solution import maxFrequency


def test_canonical_example_1():
    # spec: nums = [1,2,3,4,5,6], k = 1
    # After adding -5 to nums[2..5], array becomes [1, 2, -2, -1, 0, 1]
    # Frequency of 1 is 2
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2


def test_canonical_example_2():
    # spec: nums = [10,2,3,4,5,5,4,3,2,2], k = 10
    # After adding 8 to nums[1..9], array becomes [10, 10, 11, 12, 13, 13, 12, 11, 10, 10]
    # Frequency of 10 is 4
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4


def test_single_element_already_k():
    # If array has one element equal to k, frequency is 1 (no operation needed)
    assert maxFrequency([5], 5) == 1


def test_single_element_not_k():
    # If array has one element not equal to k, we can add to make it k, frequency is 1
    assert maxFrequency([3], 5) == 1


def test_all_elements_equal_to_k():
    # If all elements are already k, frequency is the length of the array
    assert maxFrequency([7, 7, 7, 7], 7) == 4


def test_no_elements_equal_to_k_but_can_convert():
    # Array [2, 2, 2], k = 5. Add 3 to entire array -> [5, 5, 5], frequency = 3
    assert maxFrequency([2, 2, 2], 5) == 3


def test_partial_conversion_of_subarray():
    # Array [1, 3, 3, 3], k = 5. Add 2 to nums[1..3] -> [1, 5, 5, 5], frequency = 3
    assert maxFrequency([1, 3, 3, 3], 5) == 3


def test_existing_k_plus_conversions():
    # Array [5, 2, 2], k = 5. Add 3 to nums[1..2] -> [5, 5, 5], frequency = 3
    assert maxFrequency([5, 2, 2], 5) == 3


def test_convert_middle_subarray():
    # Array [1, 2, 3, 2, 1], k = 5. Add 3 to nums[1..3] -> [1, 5, 6, 5, 1], frequency = 2
    assert maxFrequency([1, 2, 3, 2, 1], 5) == 2


def test_multiple_k_values_scattered():
    # Array [10, 5, 10, 5, 10], k = 10. Already has 3 tens, no operation increases it
    assert maxFrequency([10, 5, 10, 5, 10], 10) == 3


def test_convert_prefix():
    # Array [3, 3, 7], k = 5. Add 2 to nums[0..1] -> [5, 5, 7], frequency = 2
    assert maxFrequency([3, 3, 7], 5) == 2


def test_convert_suffix():
    # Array [7, 3, 3], k = 5. Add 2 to nums[1..2] -> [7, 5, 5], frequency = 2
    assert maxFrequency([7, 3, 3], 5) == 2


def test_two_elements_same_value():
    # Array [4, 4], k = 1. Add -3 to entire array -> [1, 1], frequency = 2
    assert maxFrequency([4, 4], 1) == 2


def test_two_elements_different_values():
    # Array [1, 2], k = 3. Add 2 to nums[1..1] -> [1, 4] or add 2 to nums[0..0] -> [3, 2], frequency = 1
    # Or add 1 to entire array -> [2, 3], frequency = 1
    assert maxFrequency([1, 2], 3) == 1
