## Implementation

```python
def findValidPair(s: str) -> str:
    from collections import Counter
    counts = Counter(s)
    for i in range(len(s) - 1):
        if s[i] != s[i + 1]:  # Check adjacent digits are not equal
            first_digit = int(s[i])
            second_digit = int(s[i + 1])
            # Check if both digits meet their counts requirement
            if counts[s[i]] == first_digit and counts[s[i + 1]] == second_digit:
                return s[i] + s[i + 1]
    return ""
```
