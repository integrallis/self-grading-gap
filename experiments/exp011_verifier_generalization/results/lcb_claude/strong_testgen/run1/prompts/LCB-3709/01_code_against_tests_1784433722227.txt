## Implementation

```python
def hasSpecialSubstring(s: str, k: int) -> bool:
    n = len(s)
    for i in range(n - k + 1):
        substring = s[i:i + k]
        if len(set(substring)) == 1:
            left_char = s[i - 1] if i > 0 else None
            right_char = s[i + k] if i + k < n else None
            if (left_char is None or left_char != substring[0]) and (right_char is None or right_char != substring[0]):
                return True
    return False

```
