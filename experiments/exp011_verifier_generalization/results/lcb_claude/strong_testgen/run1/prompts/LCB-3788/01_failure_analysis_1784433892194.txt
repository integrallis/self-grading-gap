from typing import List

def maxSum(nums: List[int]) -> int:
    unique_nums = set()
    current_sum = 0
    max_sum = float('-inf')

    for num in nums:
        if num not in unique_nums:
            unique_nums.add(num)
            # Only sum positive values to maximize the unique sum
            if num > 0:
                current_sum += num
            max_sum = max(max_sum, current_sum)
        else:
            # If we encounter a duplicate, reset the current sum
            current_sum = 0
            unique_nums.clear()
            if num > 0:
                unique_nums.add(num)
                current_sum += num
            max_sum = max(max_sum, current_sum)

    # If all elements are negative, we should return the max single element
    if max_sum == float('-inf'):
        return max(nums)

    return max(max_sum, current_sum)
