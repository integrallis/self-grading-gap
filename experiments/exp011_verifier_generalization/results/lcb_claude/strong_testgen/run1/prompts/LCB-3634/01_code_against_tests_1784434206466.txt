```python
def calculateScore(s: str) -> int:
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # Create a mirror mapping for 'a' to 'z'
    marked = [False] * len(s)  # Keep track of marked indices
    score = 0
    
    for i in range(len(s)):
        if marked[i]:  # Skip if already marked
            continue
        target = mirror[s[i]]  # Get the mirror character
        # Find the closest unmarked index j where s[j] is the mirror of s[i]
        for j in range(i):
            if not marked[j] and s[j] == target:
                marked[j] = True  # Only mark the matching character
                marked[i] = True  # Mark i
                score += i - j  # Update score
                break  # Found the closest match, break out
    return score

```
