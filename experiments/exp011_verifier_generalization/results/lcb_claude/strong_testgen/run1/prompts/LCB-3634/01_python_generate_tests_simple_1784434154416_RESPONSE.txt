import pytest
from solution import calculateScore


def test_canonical_example_1():
    # spec: s = "aczzx" -> 5
    # i=0 'a': no match, skip
    # i=1 'c': no match, skip
    # i=2 'z': matches j=0 'a' (mirror), mark both, score += 2-0 = 2
    # i=3 'z': no unmarked match, skip
    # i=4 'x': matches j=1 'c' (mirror), mark both, score += 4-1 = 3
    # total = 2 + 3 = 5
    assert calculateScore("aczzx") == 5


def test_canonical_example_2():
    # spec: s = "abcdef" -> 0
    # No character has a mirror match before it
    assert calculateScore("abcdef") == 0


def test_single_character():
    # Single character has no previous indices to match
    assert calculateScore("a") == 0


def test_two_characters_mirror():
    # s = "az": i=1, 'z' mirrors 'a' at j=0, score = 1-0 = 1
    assert calculateScore("az") == 1


def test_two_characters_not_mirror():
    # s = "ab": 'b' does not mirror 'a', no match
    assert calculateScore("ab") == 0


def test_mirror_pairs_consecutive():
    # s = "azby": 
    # i=1 'z' mirrors 'a' at j=0, score += 1
    # i=3 'y' mirrors 'b' at j=2, score += 1
    # total = 2
    assert calculateScore("azby") == 2


def test_multiple_same_character():
    # s = "aaa": no character mirrors 'a' (mirror of 'a' is 'z')
    assert calculateScore("aaa") == 0


def test_multiple_mirrors_closest_chosen():
    # s = "aaza": 
    # i=1 'a': no match (mirror is 'z'), skip
    # i=2 'z': mirrors 'a', closest unmarked is j=1, score += 2-1 = 1
    # i=3 'a': no match (j=0 'a' is not mirror, j=1 marked, j=2 marked), skip
    # total = 1
    assert calculateScore("aaza") == 1


def test_reverse_alphabet_all_pairs():
    # s = "abyz": 
    # i=2 'y' mirrors 'b' at j=1, score += 2-1 = 1
    # i=3 'z' mirrors 'a' at j=0, score += 3-0 = 3
    # total = 4
    assert calculateScore("abyz") == 4


def test_marked_indices_not_reused():
    # s = "azaz": 
    # i=1 'z' mirrors 'a' at j=0, mark 0 and 1, score += 1
    # i=2 'a': mirrors 'z', but j=1 is marked, no valid j, skip
    # i=3 'z': mirrors 'a', but j=0 and j=2 ('a') - j=2 is closest but unmarked, score += 3-2 = 1
    # total = 2
    assert calculateScore("azaz") == 2


def test_middle_of_alphabet():
    # s = "mo": 'm' is 13th (index 12), 'o' is 15th (index 14)
    # mirror of 'm' (index 12) is index 13 which is 'n'
    # mirror of 'o' (index 14) is index 11 which is 'l'
    # no matches
    assert calculateScore("mo") == 0


def test_middle_of_alphabet_mirror():
    # s = "mn": 'm' is 13th letter (0-indexed: 12), mirror is 'n' (0-indexed: 13)
    # i=1 'n' mirrors 'm' at j=0, score += 1
    assert calculateScore("mn") == 1


def test_long_string_with_mirrors():
    # s = "zyxabc":
    # i=3 'a' mirrors 'z' at j=0, score += 3
    # i=4 'b' mirrors 'y' at j=1, score += 3
    # i=5 'c' mirrors 'x' at j=2, score += 3
    # total = 9
    assert calculateScore("zyxabc") == 9
