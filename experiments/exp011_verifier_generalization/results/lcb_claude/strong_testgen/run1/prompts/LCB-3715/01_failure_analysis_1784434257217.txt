from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    events = []
    for l, r, c in coins:
        events.append((l, c))
        events.append((r + 1, -c))

    events.sort()
    max_coins = 0
    current_coins = 0
    current_position = 0
    active_coins = 0

    for position, change in events:
        if current_position < position:
            span = position - current_position
            if span > 0:
                if span >= k:
                    max_coins = max(max_coins, current_coins + active_coins * k)
                else:
                    max_coins = max(max_coins, current_coins + active_coins * span)
            current_position = position
        active_coins += change
        current_coins = active_coins * (min(k, (current_position + k - 1)) - current_position)

    return max_coins
