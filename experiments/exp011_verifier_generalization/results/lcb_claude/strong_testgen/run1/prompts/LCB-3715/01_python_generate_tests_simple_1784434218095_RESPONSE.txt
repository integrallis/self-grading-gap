from typing import List
import pytest
from solution import maximumCoins


def test_canonical_example_1():
    # spec: coins = [[8,10,1],[1,3,2],[5,6,4]], k = 4 -> 10
    # Selecting bags at positions [3, 4, 5, 6] gives: 2 + 0 + 4 + 4 = 10
    assert maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) == 10


def test_canonical_example_2():
    # spec: coins = [[1,10,3]], k = 2 -> 6
    # Selecting bags at positions [1, 2] gives: 3 + 3 = 6
    assert maximumCoins([[1, 10, 3]], 2) == 6


def test_single_segment_k_equals_segment_length():
    # Single segment [1, 5, 2] with k = 5 covers entire segment
    # Total: 2 * 5 = 10
    assert maximumCoins([[1, 5, 2]], 5) == 10


def test_single_segment_k_greater_than_segment_length():
    # Single segment [1, 3, 5] with k = 10
    # Best window of 10 consecutive bags includes [1, 2, 3] with coins
    # Total: 5 + 5 + 5 = 15 (rest are empty bags)
    assert maximumCoins([[1, 3, 5]], 10) == 15


def test_single_segment_k_less_than_segment_length():
    # Single segment [1, 10, 4] with k = 3
    # Any 3 consecutive bags within [1, 10] give: 4 * 3 = 12
    assert maximumCoins([[1, 10, 4]], 3) == 12


def test_multiple_segments_k_spans_gap():
    # Segments [1, 2, 3] and [5, 6, 2], k = 4
    # Window [2, 3, 4, 5]: 3 + 3 + 0 + 2 = 8
    # Window [3, 4, 5, 6]: 3 + 0 + 2 + 2 = 7
    assert maximumCoins([[1, 2, 3], [5, 6, 2]], 4) == 8


def test_k_equals_one():
    # k = 1 means select single bag with maximum coins
    # Segment [1, 1, 5] has 5 coins, [3, 3, 10] has 10 coins
    assert maximumCoins([[1, 1, 5], [3, 3, 10]], 1) == 10


def test_non_overlapping_segments_optimal_window_within_single_segment():
    # Segments [1, 5, 2], [10, 15, 3], k = 4
    # Best within first segment: 2 * 4 = 8
    # Best within second segment: 3 * 4 = 12
    assert maximumCoins([[1, 5, 2], [10, 15, 3]], 4) == 12


def test_window_starts_before_first_segment():
    # Segment [5, 7, 3], k = 6
    # Window [1, 2, 3, 4, 5, 6]: 0 + 0 + 0 + 0 + 3 + 3 = 6
    # Window [2, 3, 4, 5, 6, 7]: 0 + 0 + 0 + 3 + 3 + 3 = 9
    assert maximumCoins([[5, 7, 3]], 6) == 9


def test_window_ends_after_last_segment():
    # Segment [1, 3, 4], k = 5
    # Window [1, 2, 3, 4, 5]: 4 + 4 + 4 + 0 + 0 = 12
    assert maximumCoins([[1, 3, 4]], 5) == 12


def test_multiple_segments_optimal_crosses_boundaries():
    # Segments [1, 2, 1], [4, 5, 10], k = 3
    # Window [2, 3, 4]: 1 + 0 + 10 = 11
    # Window [3, 4, 5]: 0 + 10 + 10 = 20
    # Window [4, 5, 6]: 10 + 10 + 0 = 20
    assert maximumCoins([[1, 2, 1], [4, 5, 10]], 3) == 20


def test_large_k_single_small_segment():
    # Segment [100, 102, 7], k = 1000000000
    # Only bags [100, 101, 102] have coins: 7 * 3 = 21
    assert maximumCoins([[100, 102, 7]], 1000000000) == 21


def test_segments_at_large_positions():
    # Segments at large positions [1000000000, 1000000000, 5], k = 1
    assert maximumCoins([[1000000000, 1000000000, 5]], 1) == 5
