from typing import List
import pytest
from solution import minCosts


def test_canonical_example_1():
    # spec: cost = [5,3,4,1,3,2] -> [5,3,3,1,1,1]
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]


def test_canonical_example_2():
    # spec: cost = [1,2,4,6,7] -> [1,1,1,1,1]
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]


def test_single_person():
    # n=1: one person at position 0, we start at position 1
    # To reach position 0, we swap with person 0 for cost[0]
    assert minCosts([5]) == [5]


def test_two_people_decreasing():
    # cost = [10, 5]
    # position 0: swap with person 0 for cost 10
    # position 1: swap with person 1 for cost 5
    assert minCosts([10, 5]) == [10, 5]


def test_two_people_increasing():
    # cost = [3, 7]
    # position 0: swap with person 0 for cost 3
    # position 1: swap with person 0 for cost 3, then person 1 swaps for free (behind us)
    # OR swap with person 1 for cost 7
    # minimum is 3
    assert minCosts([3, 7]) == [3, 3]


def test_all_same_cost():
    # cost = [5, 5, 5]
    # position 0: cost 5
    # position 1: cost 5
    # position 2: min(cost[2]=5, cost[0]=5, cost[1]=5) = 5
    assert minCosts([5, 5, 5]) == [5, 5, 5]


def test_strictly_decreasing():
    # cost = [10, 8, 6, 4, 2]
    # position 0: cost 10
    # position 1: cost 8
    # position 2: cost 6
    # position 3: cost 4
    # position 4: cost 2
    # Each position is reached by direct swap (no benefit from free swaps)
    assert minCosts([10, 8, 6, 4, 2]) == [10, 8, 6, 4, 2]


def test_strictly_increasing():
    # cost = [1, 2, 3, 4, 5]
    # position 0: cost 1
    # position 1: min(2, 1) = 1 (swap with person 0, then free swap)
    # position 2: min(3, 1) = 1 (swap with person 0, then free swaps)
    # position 3: min(4, 1) = 1
    # position 4: min(5, 1) = 1
    assert minCosts([1, 2, 3, 4, 5]) == [1, 1, 1, 1, 1]


def test_minimum_cost_at_end():
    # cost = [10, 10, 10, 1]
    # position 0: cost 10
    # position 1: cost 10
    # position 2: cost 10
    # position 3: cost 1
    # Positions 0-3 can all be reached via person 3 then free swaps
    assert minCosts([10, 10, 10, 1]) == [10, 10, 10, 1]


def test_minimum_cost_in_middle():
    # cost = [5, 2, 7]
    # position 0: cost 5
    # position 1: cost 2
    # position 2: min(7, 2) = 2 (swap with person 1, then free swap)
    assert minCosts([5, 2, 7]) == [5, 2, 2]


def test_three_elements_with_middle_minimum():
    # cost = [10, 1, 20]
    # position 0: cost 10
    # position 1: cost 1
    # position 2: min(20, 1) = 1 (swap with person 1 for 1, then free swap)
    assert minCosts([10, 1, 20]) == [10, 1, 1]
