from typing import List
import pytest
from solution import minimumPairRemoval


def test_canonical_example_1():
    # spec: nums = [5,2,3,1] -> 2
    # (3,1) sum=4 -> [5,2,4]
    # (2,4) sum=6 -> [5,6]
    # 2 operations to make non-decreasing
    assert minimumPairRemoval([5, 2, 3, 1]) == 2


def test_canonical_example_2():
    # spec: nums = [1,2,2] -> 0
    # already non-decreasing
    assert minimumPairRemoval([1, 2, 2]) == 0


def test_single_element():
    # single element array is already non-decreasing
    assert minimumPairRemoval([5]) == 0


def test_two_elements_non_decreasing():
    # [1, 2] is already non-decreasing
    assert minimumPairRemoval([1, 2]) == 0


def test_two_elements_decreasing():
    # [2, 1] is decreasing, one merge makes it [3] which is non-decreasing
    assert minimumPairRemoval([2, 1]) == 1


def test_all_equal():
    # all equal elements form a non-decreasing array
    assert minimumPairRemoval([3, 3, 3, 3]) == 0


def test_strictly_increasing():
    # strictly increasing is non-decreasing
    assert minimumPairRemoval([1, 2, 3, 4, 5]) == 0


def test_strictly_decreasing():
    # [5, 4, 3, 2, 1]
    # minimum adjacent pair is (2,1) sum=3 -> [5,4,3,3]
    # minimum adjacent pair is (3,3) leftmost at index 2 -> [5,4,6]
    # minimum adjacent pair is (4,6) sum=10 -> [5,10]
    # minimum adjacent pair is (5,10) sum=15 -> [15]
    # 4 operations
    assert minimumPairRemoval([5, 4, 3, 2, 1]) == 4


def test_negative_numbers():
    # [-5, -3, -1] is already non-decreasing
    assert minimumPairRemoval([-5, -3, -1]) == 0


def test_negative_and_positive():
    # [-1, 2, 1] 
    # (2,1) sum=3 -> [-1,3] which is non-decreasing
    # 1 operation
    assert minimumPairRemoval([-1, 2, 1]) == 1


def test_multiple_pairs_with_same_minimum_sum():
    # [1, 2, 1, 2]
    # pairs: (1,2)=3, (2,1)=3, (1,2)=3
    # leftmost minimum is (1,2) at index 0 -> [3,1,2]
    # pairs: (3,1)=4, (1,2)=3
    # minimum is (1,2) -> [3,3] which is non-decreasing
    # 2 operations
    assert minimumPairRemoval([1, 2, 1, 2]) == 2


def test_already_sorted_with_duplicates():
    # [1, 1, 2, 2, 3, 3] is non-decreasing
    assert minimumPairRemoval([1, 1, 2, 2, 3, 3]) == 0


def test_three_elements_one_violation():
    # [1, 3, 2]
    # (3,2) sum=5 -> [1,5] which is non-decreasing
    # 1 operation
    assert minimumPairRemoval([1, 3, 2]) == 1
