```python
from typing import List

def subarraySum(nums: List[int]) -> int:
    total_sum = 0
    for i in range(len(nums)):
        start = max(0, i - nums[i])
        subarray_sum = sum(nums[start:i + 1])
        total_sum += subarray_sum
    return total_sum

```
