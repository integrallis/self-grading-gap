from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Calculate the contributions from all subsequences of size up to k
    for size in range(1, k + 1):
        # Number of ways to choose 'size' elements from 'n'
        from math import comb
        comb_size = comb(n, size)

        # For each chosen size, calculate contribution of each number
        for num in nums:
            # Each number contributes to 'size' subsequences as min and max
            contribution = (num * comb_size) % MOD
            total_sum = (total_sum + contribution) % MOD

            # Add contribution when this number is the max
            # For this, we need to consider how many sequences can be formed with the remaining numbers
            total_sum = (total_sum + (num * comb(n - 1, size - 1)) % MOD) % MOD

    return (total_sum * 2) % MOD
