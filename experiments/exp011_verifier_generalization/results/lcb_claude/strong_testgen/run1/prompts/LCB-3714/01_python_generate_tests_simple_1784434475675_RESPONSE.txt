from typing import List
import pytest
from solution import minMaxSums


def test_canonical_example_1():
    # spec: nums = [1,2,3], k = 2 -> 24
    # Subsequences with at most 2 elements:
    # [1]: min=1, max=1, sum=2
    # [2]: min=2, max=2, sum=4
    # [3]: min=3, max=3, sum=6
    # [1,2]: min=1, max=2, sum=3
    # [1,3]: min=1, max=3, sum=4
    # [2,3]: min=2, max=3, sum=5
    # Total: 2+4+6+3+4+5 = 24
    assert minMaxSums([1, 2, 3], 2) == 24


def test_canonical_example_2():
    # spec: nums = [5,0,6], k = 1 -> 22
    # Subsequences with at most 1 element (single elements only):
    # [5]: min=5, max=5, sum=10
    # [0]: min=0, max=0, sum=0
    # [6]: min=6, max=6, sum=12
    # Total: 10+0+12 = 22
    assert minMaxSums([5, 0, 6], 1) == 22


def test_canonical_example_3():
    # spec: nums = [1,1,1], k = 2 -> 12
    # Subsequences with at most 2 elements:
    # [1] appears 3 times (one for each position): min=1, max=1, sum=2 each -> 3*2=6
    # [1,1] appears 3 times (positions (0,1), (0,2), (1,2)): min=1, max=1, sum=2 each -> 3*2=6
    # Total: 6+6 = 12
    assert minMaxSums([1, 1, 1], 2) == 12


def test_single_element_array_k_one():
    # Single element array [5] with k=1
    # Only subsequence: [5] with min=5, max=5, sum=10
    assert minMaxSums([5], 1) == 10


def test_two_elements_k_one():
    # nums = [3, 7], k = 1
    # Subsequences: [3] -> 6, [7] -> 14
    # Total: 20
    assert minMaxSums([3, 7], 1) == 20


def test_two_elements_k_two():
    # nums = [3, 7], k = 2
    # Subsequences: [3] -> 6, [7] -> 14, [3,7] -> min=3, max=7, sum=10
    # Total: 6+14+10 = 30
    assert minMaxSums([3, 7], 2) == 30


def test_all_zeros():
    # nums = [0, 0, 0], k = 2
    # All subsequences have min=0, max=0, sum=0
    assert minMaxSums([0, 0, 0], 2) == 0


def test_k_equals_array_length():
    # nums = [1, 2], k = 2 (k equals length)
    # Subsequences: [1] -> 2, [2] -> 4, [1,2] -> 3
    # Total: 9
    assert minMaxSums([1, 2], 2) == 9


def test_large_values_modulo():
    # Test that modulo 10^9 + 7 is applied
    # nums = [10**9, 10**9], k = 1
    # Each element contributes 2 * 10^9
    # Total: 4 * 10^9
    # Modulo 10^9 + 7: 4000000000 % 1000000007 = 2999999986
    assert minMaxSums([10**9, 10**9], 1) == 2999999986


def test_three_elements_k_three():
    # nums = [1, 2, 3], k = 3
    # All subsequences of size 1, 2, and 3
    # Size 1: [1]->2, [2]->4, [3]->6
    # Size 2: [1,2]->3, [1,3]->4, [2,3]->5
    # Size 3: [1,2,3]->min=1,max=3,sum=4
    # Total: 2+4+6+3+4+5+4 = 28
    assert minMaxSums([1, 2, 3], 3) == 28


def test_distinct_values():
    # nums = [10, 20, 30, 40], k = 2
    # Size 1: [10]->20, [20]->40, [30]->60, [40]->80
    # Size 2: [10,20]->30, [10,30]->40, [10,40]->50, [20,30]->50, [20,40]->60, [30,40]->70
    # Total: 20+40+60+80+30+40+50+50+60+70 = 500
    assert minMaxSums([10, 20, 30, 40], 2) == 500
