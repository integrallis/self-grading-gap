from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Calculate total contributions for min and max separately
    for size in range(1, k + 1):
        # Number of ways to choose 'size' elements from 'n'
        from math import comb

        # For each chosen size, calculate contribution of each number as min
        for num in nums:
            # Each number can be the minimum in comb(n-1, size-1) subsequences
            min_contribution = (num * comb(n - 1, size - 1)) % MOD
            total_sum = (total_sum + min_contribution) % MOD

        # For each chosen size, calculate contribution of each number as max
        for num in nums:
            # Each number can be the maximum in comb(n-1, size-1) subsequences
            max_contribution = (num * comb(n - 1, size - 1)) % MOD
            total_sum = (total_sum + max_contribution) % MOD

    return total_sum % MOD
```
