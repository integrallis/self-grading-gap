from solution import maxDifference


def test_canonical_example_1():
    # spec: s = "aaaaabbc"
    # 'a' has frequency 5 (odd), 'b' has frequency 2 (even), 'c' has frequency 1 (odd)
    # max difference = 5 - 2 = 3
    assert maxDifference("aaaaabbc") == 3


def test_canonical_example_2():
    # spec: s = "abcabcab"
    # 'a' has frequency 3 (odd), 'b' has frequency 3 (odd), 'c' has frequency 2 (even)
    # max difference = 3 - 2 = 1
    assert maxDifference("abcabcab") == 1


def test_max_odd_min_even():
    # s = "aaaaaabbbbbbbc"
    # 'a' has frequency 6 (even), 'b' has frequency 7 (odd), 'c' has frequency 1 (odd)
    # max difference = 7 - 6 = 1
    assert maxDifference("aaaaaabbbbbbbc") == 1


def test_multiple_characters_odd_and_even():
    # s = "aabbccdddeee"
    # 'a' has frequency 2 (even), 'b' has frequency 2 (even), 'c' has frequency 2 (even),
    # 'd' has frequency 3 (odd), 'e' has frequency 3 (odd)
    # max odd = 3, min even = 2
    # max difference = 3 - 2 = 1
    assert maxDifference("aabbccdddeee") == 1


def test_large_odd_small_even():
    # s = "aaaaaaaaabb"
    # 'a' has frequency 9 (odd), 'b' has frequency 2 (even)
    # max difference = 9 - 2 = 7
    assert maxDifference("aaaaaaaaabb") == 7


def test_minimum_length():
    # Constraint: 3 <= s.length <= 100
    # s = "aab"
    # 'a' has frequency 2 (even), 'b' has frequency 1 (odd)
    # max difference = 1 - 2 = -1
    assert maxDifference("aab") == -1

    # s = "aabbcccccd"

    # 'd' has frequency 1 (odd)
    # max odd = 5, min even = 2

    assert maxDifference("aabbcccccd") == 3

    # s = "aaaaabb"
