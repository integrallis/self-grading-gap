from solution import maxDifference


def test_canonical_example_1():
    # spec: s = "aaaaabbc"
    # 'a' has frequency 5 (odd), 'b' has frequency 2 (even), 'c' has frequency 1 (odd)
    # max difference = 5 - 2 = 3
    assert maxDifference("aaaaabbc") == 3


def test_canonical_example_2():
    # spec: s = "abcabcab"
    # 'a' has frequency 3 (odd), 'b' has frequency 3 (odd), 'c' has frequency 2 (even)
    # max difference = 3 - 2 = 1
    assert maxDifference("abcabcab") == 1


def test_single_odd_single_even():
    # s = "aaabbb"
    # 'a' has frequency 3 (odd), 'b' has frequency 3 (odd)
    # No even frequency characters, but constraint guarantees at least one
    # s = "aabb"
    # 'a' has frequency 2 (even), 'b' has frequency 2 (even)
    # No odd frequency characters, but constraint guarantees at least one
    # s = "aaaabb"
    # 'a' has frequency 4 (even), 'b' has frequency 2 (even)
    # s = "aaaaabb"
    # 'a' has frequency 5 (odd), 'b' has frequency 2 (even)
    # max difference = 5 - 2 = 3
    assert maxDifference("aaaaabb") == 3


def test_max_odd_min_even():
    # s = "aaaaabc"
    # 'a' has frequency 5 (odd), 'b' has frequency 1 (odd), 'c' has frequency 1 (odd)
    # Need at least one even frequency
    # s = "aaaaaabbbc"
    # 'a' has frequency 6 (even), 'b' has frequency 3 (odd), 'c' has frequency 1 (odd)
    # max difference = 3 - 6 = -3 (negative, not maximum)
    # We want max(odd) - min(even)
    # s = "aaaaabbc"
    # 'a' has frequency 5 (odd), 'b' has frequency 2 (even), 'c' has frequency 1 (odd)
    # max difference = 5 - 2 = 3
    # s = "aaaaaabbbbbbbc"
    # 'a' has frequency 6 (even), 'b' has frequency 7 (odd), 'c' has frequency 1 (odd)
    # max difference = 7 - 6 = 1
    assert maxDifference("aaaaaabbbbbbbc") == 1


def test_multiple_characters_odd_and_even():
    # s = "aabbccdde"
    # 'a' has frequency 2 (even), 'b' has frequency 2 (even), 'c' has frequency 2 (even),
    # 'd' has frequency 2 (even), 'e' has frequency 1 (odd)
    # max difference = 1 - 2 = -1 (negative, not maximum)
    # Actually we need max(odd) - min(even) = 1 - 2 = -1
    # But we're looking for maximum difference, so we try all pairs
    # s = "aabbccdddeee"
    # 'a' has frequency 2 (even), 'b' has frequency 2 (even), 'c' has frequency 2 (even),
    # 'd' has frequency 3 (odd), 'e' has frequency 3 (odd)
    # max difference = 3 - 2 = 1
    assert maxDifference("aabbccdddeee") == 1


def test_large_odd_small_even():
    # s = "aaaaaaaaabb"
    # 'a' has frequency 9 (odd), 'b' has frequency 2 (even)
    # max difference = 9 - 2 = 7
    assert maxDifference("aaaaaaaaabb") == 7


def test_minimum_length():
    # Constraint: 3 <= s.length <= 100
    # s = "aab"
    # 'a' has frequency 2 (even), 'b' has frequency 1 (odd)
    # max difference = 1 - 2 = -1 (negative)
    # We need to maximize odd - even, so minimum even and maximum odd
    # max difference = 1 - 2 = -1
    assert maxDifference("aab") == -1


def test_all_same_frequency_except_one():
    # s = "aabbcccccd"
    # 'a' has frequency 2 (even), 'b' has frequency 2 (even), 'c' has frequency 5 (odd),
    # 'd' has frequency 1 (odd)
    # max difference = 5 - 2 = 3
