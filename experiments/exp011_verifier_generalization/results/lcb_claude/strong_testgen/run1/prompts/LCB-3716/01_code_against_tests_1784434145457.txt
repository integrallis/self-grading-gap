```python
from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n < 2:
        return n
    dp = [1] * n
    max_length = 1

    # Loop through the array to build the dp array
    for i in range(n):
        for j in range(i):
            # Calculate the absolute difference
            diff1 = abs(nums[i] - nums[j])
            if i > 0:
                diff2 = abs(nums[i] - nums[i-1])
            else:
                diff2 = float('inf')  # No previous element for the first element
            
            # Check if the current difference is less than or equal to the previous
            if diff1 >= diff2:
                dp[i] = max(dp[i], dp[j] + 1)
                max_length = max(max_length, dp[i])

    return max_length
```
