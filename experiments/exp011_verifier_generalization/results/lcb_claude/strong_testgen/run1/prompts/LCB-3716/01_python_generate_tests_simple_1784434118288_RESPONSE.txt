from typing import List
import pytest
from solution import longestSubsequence


def test_canonical_example_1():
    # Example 1: nums = [16,6,3]
    # Longest subsequence is [16, 6, 3] with absolute adjacent differences [10, 3]
    # 10 >= 3, so differences are non-increasing
    # Length = 3
    assert longestSubsequence([16, 6, 3]) == 3


def test_canonical_example_2():
    # Example 2: nums = [6,5,3,4,2,1]
    # Longest subsequence is [6, 4, 2, 1] with absolute adjacent differences [2, 2, 1]
    # 2 >= 2 >= 1, so differences are non-increasing
    # Length = 4
    assert longestSubsequence([6, 5, 3, 4, 2, 1]) == 4


def test_canonical_example_3():
    # Example 3: nums = [10,20,10,19,10,20]
    # Longest subsequence is [10, 20, 10, 19, 10] with absolute adjacent differences [10, 10, 9, 9]
    # 10 >= 10 >= 9 >= 9, so differences are non-increasing
    # Length = 5
    assert longestSubsequence([10, 20, 10, 19, 10, 20]) == 5


def test_minimum_length_array():
    # Constraint: 2 <= nums.length
    # Any array of length 2 forms a valid subsequence of length 2
    # The single difference is trivially non-increasing
    assert longestSubsequence([1, 2]) == 2


def test_all_same_elements():
    # If all elements are the same, all differences are 0
    # [5, 5, 5] -> differences [0, 0] which is non-increasing (0 >= 0)
    # Length = 3
    assert longestSubsequence([5, 5, 5]) == 3


def test_strictly_increasing_sequence():
    # nums = [1, 2, 3, 4, 5]
    # Subsequence [1, 2, 3, 4, 5] has differences [1, 1, 1, 1] (non-increasing: 1 >= 1 >= 1 >= 1)
    # Length = 5
    assert longestSubsequence([1, 2, 3, 4, 5]) == 5


def test_strictly_decreasing_sequence():
    # nums = [5, 4, 3, 2, 1]
    # Subsequence [5, 4, 3, 2, 1] has differences [1, 1, 1, 1] (non-increasing: 1 >= 1 >= 1 >= 1)
    # Length = 5
    assert longestSubsequence([5, 4, 3, 2, 1]) == 5


def test_large_then_small_differences():
    # nums = [1, 10, 11]
    # Subsequence [1, 10, 11] has differences [9, 1] (non-increasing: 9 >= 1)
    # Length = 3
    assert longestSubsequence([1, 10, 11]) == 3


def test_alternating_values():
    # nums = [1, 10, 1, 10]
    # Subsequence [1, 10, 1, 10] has differences [9, 9, 9] (non-increasing: 9 >= 9 >= 9)
    # Length = 4
    assert longestSubsequence([1, 10, 1, 10]) == 4


def test_single_large_jump():
    # nums = [1, 100, 101, 102]
    # Subsequence [1, 100, 101, 102] has differences [99, 1, 1] (non-increasing: 99 >= 1 >= 1)
    # Length = 4
    assert longestSubsequence([1, 100, 101, 102]) == 4
