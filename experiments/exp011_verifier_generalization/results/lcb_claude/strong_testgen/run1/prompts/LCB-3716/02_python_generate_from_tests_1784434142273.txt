from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = [1] * n
    diffs = [[0] * n for _ in range(n)]

    for i in range(n):
        for j in range(i + 1, n):
            diffs[i][j] = abs(nums[j] - nums[i])

    max_length = 1

    for i in range(n):
        for j in range(i):
            if diffs[j][i] >= diffs[j][i - 1]:
                dp[i] = max(dp[i], dp[j] + 1)
                max_length = max(max_length, dp[i])

    return max_length
```
