from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    last_end = 0
    free_intervals = []

    # Collect existing free intervals
    for start, end in zip(startTime, endTime):
        if start > last_end:
            free_intervals.append((last_end, start))
        last_end = end

    if last_end < eventTime:
        free_intervals.append((last_end, eventTime))

    # Initialize maximum free time from collected intervals
    max_free_time = 0
    for free_start, free_end in free_intervals:
        max_free_time = max(max_free_time, free_end - free_start)

    # To find max free time after rescheduling, we can calculate gaps to consider moving meetings
    moveable_meetings = []
    for i in range(n):
        meeting_duration = endTime[i] - startTime[i]
        if i > 0:
            gap_left = startTime[i] - endTime[i-1]
            if gap_left >= meeting_duration:
                moveable_meetings.append((gap_left, i, meeting_duration))
        else:
            if startTime[i] >= meeting_duration:
                moveable_meetings.append((startTime[i], i, meeting_duration))

    moveable_meetings.sort(reverse=True)  # Sort to prioritize larger gaps first

    # Reschedule meetings to create maximum free time
    for _ in range(k):
        if moveable_meetings:
            _, index, duration = moveable_meetings.pop(0)
            new_start = endTime[index - 1] if index > 0 else 0
            new_end = new_start + duration
            startTime[index] = new_start
            endTime[index] = new_end

            # Recalculate free intervals after potential meeting move
            last_end = 0
            new_free_intervals = []
            for start, end in zip(startTime, endTime):
                if start > last_end:
                    new_free_intervals.append((last_end, start))
                last_end = end
            if last_end < eventTime:
                new_free_intervals.append((last_end, eventTime))

            # Determine the maximum free time after rescheduling
            for free_start, free_end in new_free_intervals:
                max_free_time = max(max_free_time, free_end - free_start)

    return max_free_time
