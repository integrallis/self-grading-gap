from typing import List
import pytest
from solution import transformArray


def test_canonical_example_1():
    # spec: [4,3,2,1] -> [0,0,1,1]
    # Replace even (4,2) with 0, odd (3,1) with 1 -> [0,1,0,1]
    # Sort -> [0,0,1,1]
    assert transformArray([4, 3, 2, 1]) == [0, 0, 1, 1]


def test_canonical_example_2():
    # spec: [1,5,1,4,2] -> [0,0,1,1,1]
    # Replace even (4,2) with 0, odd (1,5,1) with 1 -> [1,1,1,0,0]
    # Sort -> [0,0,1,1,1]
    assert transformArray([1, 5, 1, 4, 2]) == [0, 0, 1, 1, 1]


def test_single_element_even():
    # Single even number: [2] -> [0]
    assert transformArray([2]) == [0]


def test_single_element_odd():
    # Single odd number: [3] -> [1]
    assert transformArray([3]) == [1]


def test_all_even_numbers():
    # All even: [2,4,6] -> [0,0,0] -> [0,0,0]
    assert transformArray([2, 4, 6]) == [0, 0, 0]


def test_all_odd_numbers():
    # All odd: [1,3,5] -> [1,1,1] -> [1,1,1]
    assert transformArray([1, 3, 5]) == [1, 1, 1]


def test_already_sorted_mixed():
    # Already sorted after transformation: [2,4,1,3] -> [0,0,1,1] -> [0,0,1,1]
    assert transformArray([2, 4, 1, 3]) == [0, 0, 1, 1]


def test_reverse_sorted_mixed():
    # Reverse sorted: [5,4,3,2] -> [1,0,1,0] -> [0,0,1,1]
    assert transformArray([5, 4, 3, 2]) == [0, 0, 1, 1]

    # Test with boundary values: [1000,999,1] -> [0,1,1] -> [0,1,1]
    assert transformArray([1000, 999, 1]) == [0, 1, 1]

    # Repeated values: [5,5,2,2] -> [1,1,0,0] -> [0,0,1,1]
