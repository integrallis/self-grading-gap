from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[[float('-inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0]  # Starting point

    for i in range(m):
        for j in range(n):
            for k in range(3):  # k is the neutralization count
                if i > 0:
                    # Coming from the top
                    new_coins = dp[i-1][j][k] + coins[i][j]
                    if coins[i][j] < 0:
                        new_coins = dp[i-1][j][k] - abs(coins[i][j]) if k == 0 else dp[i-1][j][k] + coins[i][j]
                    dp[i][j][k] = max(dp[i][j][k], new_coins)
                if j > 0:
                    # Coming from the left
                    new_coins = dp[i][j-1][k] + coins[i][j]
                    if coins[i][j] < 0:
                        new_coins = dp[i][j-1][k] - abs(coins[i][j]) if k == 0 else dp[i][j-1][k] + coins[i][j]
                    dp[i][j][k] = max(dp[i][j][k], new_coins)
                if coins[i][j] < 0 and k > 0:
                    # Neutralizing a robbery
                    dp[i][j][k] = max(dp[i][j][k], dp[i][j][k-1] + coins[i][j])

    return max(dp[m-1][n-1])
