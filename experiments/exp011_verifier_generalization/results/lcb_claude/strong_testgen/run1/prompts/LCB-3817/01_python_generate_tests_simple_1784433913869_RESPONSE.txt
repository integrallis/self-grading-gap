import pytest
from solution import maxContainers


def test_canonical_example_1():
    # spec: n=2, w=3, maxWeight=15
    # deck has n*n = 4 cells, each container weighs 3
    # total weight if all loaded = 4 * 3 = 12 <= 15
    # so all 4 containers can be loaded
    assert maxContainers(2, 3, 15) == 4


def test_canonical_example_2():
    # spec: n=3, w=5, maxWeight=20
    # deck has n*n = 9 cells, each container weighs 5
    # max containers without exceeding maxWeight = floor(20 / 5) = 4
    # (9 containers would weigh 45 > 20)
    assert maxContainers(3, 5, 20) == 4


def test_single_cell_deck():
    # n=1, w=10, maxWeight=10
    # deck has 1 cell, container weighs 10
    # 1 * 10 = 10 <= 10, so 1 container fits
    assert maxContainers(1, 10, 10) == 1


def test_weight_limit_less_than_one_container():
    # n=2, w=10, maxWeight=5
    # deck has 4 cells, but maxWeight only allows floor(5/10) = 0 containers
    assert maxContainers(2, 10, 5) == 0


def test_large_deck_small_weight_capacity():
    # n=10, w=7, maxWeight=50
    # deck has 100 cells, but maxWeight allows floor(50/7) = 7 containers
    assert maxContainers(10, 7, 50) == 7


def test_exact_capacity_match():
    # n=3, w=4, maxWeight=36
    # deck has 9 cells, total weight would be 9 * 4 = 36
