from typing import List
from solution import maxFrequency


def test_canonical_example_1():
    # spec: nums = [1,2,3,4,5,6], k = 1
    # After adding -5 to nums[2..5], 1 has a frequency of 2 in [1, 2, -2, -1, 0, 1]
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2


def test_canonical_example_2():
    # spec: nums = [10,2,3,4,5,5,4,3,2,2], k = 10
    # After adding 8 to nums[1..9], 10 has a frequency of 4 in [10, 10, 11, 12, 13, 13, 12, 11, 10, 10]
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4


def test_single_element_array_equals_k():
    # Single element that is already k, frequency should be 1
    assert maxFrequency([5], 5) == 1


def test_single_element_array_not_k():
    # Single element that is not k, can be converted to k with operation
    # Add (k - nums[0]) to nums[0..0] to get k
    assert maxFrequency([3], 5) == 1


def test_all_elements_already_k():
    # All elements are already k, frequency is len(nums)
    assert maxFrequency([7, 7, 7, 7], 7) == 4


def test_no_existing_k_convert_entire_array():
    # No element is k, convert entire array to k
    # Add (k - nums[i]) = (10 - 5) = 5 to all elements
    assert maxFrequency([5, 5, 5], 10) == 3


def test_some_elements_are_k():
    # Some elements are k, can increase frequency by converting a subarray
    # [1, 1, 2, 3] with k=1: already 2 ones, can convert [2,3] to get 4 ones
    assert maxFrequency([1, 1, 2, 3], 1) == 4


def test_convert_prefix_subarray():
    # Converting a prefix subarray nums[0..i]
    # [2, 3, 4, 1, 1] with k=1: convert [2,3,4] by adding -1, -2, -3 respectively
    # But we can only add one value x, so [2,3,4] all get x = -1 -> [1,2,3,1,1]
    # This gives frequency 3 (positions 0, 3, 4)
    assert maxFrequency([2, 3, 4, 1, 1], 1) == 3


def test_convert_suffix_subarray():
    # Converting a suffix subarray nums[i..n-1]
    # [5, 5, 10, 10] with k=5: already 2 fives, convert [10, 10] by adding -5 -> [5, 5, 5, 5]
    assert maxFrequency([5, 5, 10, 10], 5) == 4


def test_convert_middle_subarray():
    # Converting a middle subarray
    # [1, 5, 5, 5, 1] with k=1: already 2 ones, convert middle [5,5,5] by adding -4 -> [1,1,1,1,1]
    assert maxFrequency([1, 5, 5, 5, 1], 1) == 5


def test_optimal_subarray_not_all():
    # Sometimes converting fewer elements gives better result
    # [1, 2, 2, 2, 10] with k=2: already 3 twos, converting [1] gives 4 twos
    # vs converting all 5 elements gives 5 twos
    assert maxFrequency([1, 2, 2, 2, 10], 2) == 5


def test_two_element_array():
    # Two elements, different strategies
    # [3, 7] with k=3: already 1 three, convert [7] to get 2 threes
    assert maxFrequency([3, 7], 3) == 2


def test_k_not_in_array():
    # k doesn't appear in original array
    # [5, 6, 7, 8] with k=10: convert all to 10
    assert maxFrequency([5, 6, 7, 8], 10) == 4


def test_multiple_occurrences_scattered():
    # k appears multiple times scattered through array
    # [1, 5, 1, 5, 1] with k=1: already 3 ones, convert [5,5] separately or together
    # Converting [5] at index 1 gives [1,1,1,5,1] = 4 ones
    # Converting [5,5] at indices 1,3 can't be done in one subarray unless also converting the 1 in between
    # Converting [5,1,5] gives [1,1,1] at positions 1,2,3 plus existing 1s at 0,4 = 5 ones
    assert maxFrequency([1, 5, 1, 5, 1], 1) == 5
