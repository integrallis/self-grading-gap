from typing import List

def maxFrequency(nums: List[int], k: int) -> int:
    count = 0
    n = len(nums)
    nums.sort()
    left = 0
    total = 0
    for right in range(n):
        total += nums[right]
        while nums[right] * (right - left + 1) > total + k:
            total -= nums[left]
            left += 1
        count = max(count, right - left + 1)
    return count
```
