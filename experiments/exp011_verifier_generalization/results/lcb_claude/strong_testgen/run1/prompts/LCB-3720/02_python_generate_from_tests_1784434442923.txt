from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    from collections import defaultdict, deque
    import sys

    def canReachZero(maxWeight):
        graph = defaultdict(list)
        indegree = [0] * n

        for u, v, w in edges:
            if w <= maxWeight:
                graph[v].append(u)
                indegree[u] += 1

        queue = deque([0])
        visited = set([0])

        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if neighbor not in visited:
                    visited.add(neighbor)
                    queue.append(neighbor)

        return len(visited) == n

    def countOutgoingEdges(maxWeight):
        graph = defaultdict(list)
        for u, v, w in edges:
            if w <= maxWeight:
                graph[u].append((v, w))

        outgoingCount = [0] * n

        for u in range(n):
            for v, _ in graph[u]:
                outgoingCount[u] += 1
                if outgoingCount[u] > threshold:
                    return False
        return True

    left, right = 1, 10**6
    bestWeight = sys.maxsize

    while left <= right:
        mid = (left + right) // 2

        if canReachZero(mid) and countOutgoingEdges(mid):
            bestWeight = mid
            right = mid - 1
        else:
            left = mid + 1

    return bestWeight if bestWeight != sys.maxsize else -1

```
