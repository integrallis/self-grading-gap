import pytest
from typing import List
from solution import minMaxWeight


def test_canonical_example_1():
    # spec: n=5, edges=[[1,0,1],[2,0,2],[3,0,1],[4,3,1],[2,1,1]], threshold=2 -> 1
    # Remove edge 2->0 (weight 2). Remaining edges have max weight 1.
    assert minMaxWeight(5, [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]], 2) == 1


def test_canonical_example_2():
    # spec: n=5, edges=[[0,1,1],[0,2,2],[0,3,1],[0,4,1],[1,2,1],[1,4,1]], threshold=1 -> -1
    # Impossible to reach node 0 from node 2.
    assert minMaxWeight(5, [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]], 1) == -1


def test_canonical_example_3():
    # spec: n=5, edges=[[1,2,1],[1,3,3],[1,4,5],[2,3,2],[3,4,2],[4,0,1]], threshold=1 -> 2
    # Remove edges 1->3 and 1->4. Remaining edges have max weight 2.
    assert minMaxWeight(5, [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]], 1) == 2


def test_canonical_example_4():
    # spec: n=5, edges=[[1,2,1],[1,3,3],[1,4,5],[2,3,2],[4,0,1]], threshold=1 -> -1
    # Impossible to satisfy all conditions.
    assert minMaxWeight(5, [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]], 1) == -1


def test_simple_two_nodes_direct_path():
    # n=2, node 1 can reach node 0 directly with edge weight 5
    assert minMaxWeight(2, [[1, 0, 5]], 1) == 5


def test_simple_two_nodes_no_path():
    # n=2, node 0 has outgoing edge but node 1 cannot reach node 0
    assert minMaxWeight(2, [[0, 1, 1]], 1) == -1


def test_all_nodes_direct_to_zero():
    # n=4, all nodes have direct edges to node 0 with varying weights
    # threshold=1 means each node can have at most 1 outgoing edge
    assert minMaxWeight(4, [[1, 0, 3], [2, 0, 5], [3, 0, 2]], 1) == 5


def test_threshold_constraint_forces_removal():
    # n=3, node 1 has 2 outgoing edges but threshold=1
    # Must choose one edge, select the lighter one
    assert minMaxWeight(3, [[1, 0, 2], [1, 2, 3], [2, 0, 1]], 1) == 2


def test_chain_path_to_zero():
    # n=4, nodes form chain: 3->2->1->0
    # All edges weight 1, threshold allows all edges
    assert minMaxWeight(4, [[1, 0, 1], [2, 1, 1], [3, 2, 1]], 3) == 1


def test_multiple_edges_same_nodes():
    # Multiple edges between same pair of nodes with different weights
    # Choose the lighter edge
    assert minMaxWeight(3, [[1, 0, 10], [1, 0, 2], [2, 1, 1]], 2) == 2


def test_larger_threshold_allows_all_edges():
    # threshold is large enough to allow all edges
    assert minMaxWeight(4, [[1, 0, 5], [2, 0, 3], [3, 2, 1]], 5) == 5


def test_node_zero_has_no_incoming_edges():
    # Node 0 has outgoing edges but no incoming edges, impossible
    assert minMaxWeight(3, [[0, 1, 1], [0, 2, 1]], 1) == -1


def test_single_edge_weight_one():
    # Minimal case: 2 nodes, single edge with weight 1
    assert minMaxWeight(2, [[1, 0, 1]], 1) == 1


def test_disconnected_component():
    # Node 1 can reach 0, but node 2 is isolated
    assert minMaxWeight(3, [[1, 0, 1]], 1) == -1


def test_cycle_not_reaching_zero():
    # Nodes 1 and 2 form a cycle but neither reaches node 0
    assert minMaxWeight(3, [[1, 2, 1], [2, 1, 1]], 1) == -1


def test_optimal_path_requires_heavier_edge():
    # n=4, must use heavier edge to satisfy reachability
    # 3->2 (weight 5), 2->1 (weight 1), 1->0 (weight 1)
    # Also 3->0 (weight 10) available but 3->2->1->0 is better
    assert minMaxWeight(4, [[3, 2, 5], [2, 1, 1], [1, 0, 1], [3, 0, 10]], 1) == 5
