```python
from typing import List

def zigzagTraversal(grid: List[List[int]]) -> List[int]:
    result = []
    rows = len(grid)
    cols = len(grid[0])
    for i in range(rows):
        if i % 2 == 0:  # even index row: left to right
            for j in range(0, cols, 2):  # skip every alternate cell
                result.append(grid[i][j])
        else:  # odd index row: right to left
            for j in range(cols - 1, -1, -2):  # skip every alternate cell
                result.append(grid[i][j])
    return result
```
