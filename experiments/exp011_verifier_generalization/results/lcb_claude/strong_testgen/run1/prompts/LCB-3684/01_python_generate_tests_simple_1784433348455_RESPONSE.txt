import pytest
from solution import hasMatch


def test_canonical_example_1():
    # spec: s = "leetcode", p = "ee*e" -> true
    # By replacing the '*' with "tcod", the substring "eetcode" matches the pattern.
    assert hasMatch("leetcode", "ee*e") is True


def test_canonical_example_2():
    # spec: s = "car", p = "c*v" -> false
    # There is no substring matching the pattern.
    assert hasMatch("car", "c*v") is False


def test_canonical_example_3():
    # spec: s = "luck", p = "u*" -> true
    # The substrings "u", "uc", and "uck" match the pattern.
    assert hasMatch("luck", "u*") is True


def test_star_at_beginning():
    # Pattern "*abc" should match if any substring ends with "abc"
    assert hasMatch("xyzabc", "*abc") is True


def test_star_at_beginning_no_match():
    # Pattern "*abc" should not match if no substring ends with "abc"
    assert hasMatch("xyzab", "*abc") is False


def test_star_in_middle():
    # Pattern "a*b" should match if there's a substring starting with "a" and ending with "b"
    assert hasMatch("axxb", "a*b") is True


def test_star_replaces_zero_characters():
    # Pattern "ab*cd" where * is replaced by zero characters gives "abcd"
    assert hasMatch("abcd", "ab*cd") is True


def test_star_replaces_multiple_characters():
    # Pattern "a*z" where * is replaced by multiple characters
    assert hasMatch("abcdefghiz", "a*z") is True


def test_pattern_longer_than_string():
    # Pattern "abc*def" where even with * as empty, pattern is too long
    assert hasMatch("ab", "abc*def") is False


def test_exact_match_with_star_as_empty():
    # Pattern equals string when * is replaced by empty string
    assert hasMatch("hello", "hel*lo") is True


def test_prefix_only_match():
    # Pattern "ab*" matches substring starting with "ab"
    assert hasMatch("abc", "ab*") is True


def test_suffix_only_match():
    # Pattern "*cd" matches substring ending with "cd"
    assert hasMatch("abcd", "*cd") is True


def test_single_char_pattern_with_star():
    # Pattern "*" with star only
    assert hasMatch("a", "*") is True


def test_no_prefix_match():
    # Pattern "x*y" where prefix "x" doesn't exist in string
    assert hasMatch("abc", "x*y") is False


def test_no_suffix_match():
    # Pattern "a*z" where suffix "z" doesn't exist after any "a"
    assert hasMatch("abc", "a*z") is False


def test_multiple_possible_matches():
    # Pattern "a*a" could match "aa", "aba", "abca", etc.
    assert hasMatch("abaca", "a*a") is True


def test_pattern_at_end_of_string():
    # Pattern matches at the very end of the string
    assert hasMatch("hello", "l*o") is True


def test_pattern_at_start_of_string():
    # Pattern matches at the very start of the string
    assert hasMatch("hello", "h*l") is True
