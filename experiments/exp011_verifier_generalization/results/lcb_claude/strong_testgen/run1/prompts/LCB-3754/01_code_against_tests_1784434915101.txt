```python
def maxDistance(s: str, k: int) -> int:
    # Initialize position and max distance
    x, y = 0, 0
    max_distance = 0

    # Iterate through the string to track the movements
    for move in s:
        if move == 'N':
            y += 1
        elif move == 'S':
            y -= 1
        elif move == 'E':
            x += 1
        elif move == 'W':
            x -= 1

        # Calculate the current Manhattan distance
        current_distance = abs(x) + abs(y)
        max_distance = max(max_distance, current_distance)

    # Calculate the total moves available
    remaining_moves = k
    max_distance += min(remaining_moves, abs(x) + abs(y))

    # Return the final maximum distance
    return max_distance

```
