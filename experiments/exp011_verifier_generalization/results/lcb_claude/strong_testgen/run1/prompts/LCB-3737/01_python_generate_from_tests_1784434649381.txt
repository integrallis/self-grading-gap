from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize a DP array to store the minimum cost for painting up to each house.
    dp = [[0] * 3 for _ in range(n)]
    
    # Base case: for the first house, the cost is just the painting costs.
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the DP table for each house from the second to the last.
    for i in range(1, n):
        for j in range(3):
            # Cost for painting house i with color j
            # Ensure to not paint the previous house (i-1) the same color
            dp[i][j] = cost[i][j] + min(dp[i-1][(j+1) % 3], dp[i-1][(j+2) % 3])

    # Handle equidistant constraints (0 with n-1, 1 with n-2, ...)
    min_cost = float('inf')
    for first_color in range(3):
        for second_color in range(3):
            if second_color != first_color:
                # Calculate costs for the first half and the second half
                cost_first_half = dp[n//2 - 1][first_color]
                cost_second_half = dp[n - 1][second_color]
                total_cost = cost_first_half + cost_second_half
                min_cost = min(min_cost, total_cost)

    return min_cost
```
