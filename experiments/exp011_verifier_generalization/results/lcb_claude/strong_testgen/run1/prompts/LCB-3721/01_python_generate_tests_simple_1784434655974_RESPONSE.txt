from typing import List

def test_canonical_example_1():
    # spec: numberOfUsers = 2, events = [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","71","HERE"]]
    # At timestamp 10: id1 and id0 mentioned -> mentions = [1,1]
    # At timestamp 11: id0 goes offline
    # At timestamp 71: id0 comes back online (11 + 60 = 71), "HERE" mentions all online users -> mentions = [2,2]
    numberOfUsers = 2
    events = [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "71", "HERE"]]
    assert countMentions(numberOfUsers, events) == [2, 2]
