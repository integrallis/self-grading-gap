import pytest
from solution import maxDifference


def test_canonical_example_1():
    """
    Tests the first example from the specification.
    s = "aaaaabbc"
    """
    # spec: s = "aaaaabbc" -> 3
    # Frequencies: 'a': 5 (odd), 'b': 2 (even), 'c': 1 (odd)
    # Odd frequencies: {5, 1}. Max odd freq = 5.
    # Even frequencies: {2}. Min even freq = 2.
    # Max difference = 5 - 2 = 3.
    assert maxDifference("aaaaabbc") == 3


def test_canonical_example_2():
    """
    Tests the second example from the specification.
    s = "abcabcab"
    """
    # spec: s = "abcabcab" -> 1
    # Frequencies: 'a': 3 (odd), 'b': 3 (odd), 'c': 2 (even)
    # Odd frequencies: {3, 3}. Max odd freq = 3.
    # Even frequencies: {2}. Min even freq = 2.
    # Max difference = 3 - 2 = 1.
    assert maxDifference("abcabcab") == 1


def test_multiple_odd_and_even_frequencies():
    """
    Tests a case with multiple characters having odd frequencies and multiple
    characters having even frequencies to ensure correct max/min selection.
    """
    # s = "aabbbccccdddddeeeee"
    # Frequencies: a=2(E), b=3(O), c=4(E), d=5(O), e=5(O)
    # Odd frequencies: {3, 5, 5}. Max odd freq = 5.
    # Even frequencies: {2, 4}. Min even freq = 2.
    # Max difference = 5 - 2 = 3.
    assert maxDifference("aabbbccccdddddeeeee") == 3


def test_max_difference_is_negative():
    """
    Tests a case where the maximum possible difference is a negative number.
    This uses the minimum length string allowed by constraints.
    """
    # s = "aab", length 3.
    # Frequencies: a=2(E), b=1(O)
    # Odd frequencies: {1}. Max odd freq = 1.
    # Even frequencies: {2}. Min even freq = 2.
    # Max difference = 1 - 2 = -1.
    assert maxDifference("aab") == -1


def test_odd_freq_char_is_not_overall_most_frequent():
    """
    Tests that the character with the maximum odd frequency is chosen, even if it's
    not the most frequent character in the string overall.
    """
    # s = "aaabbcccccc"
    # Frequencies: a=3(O), b=2(E), c=6(E)
    # The most frequent char 'c' (6) has an even frequency.
    # Odd frequencies: {3}. Max odd freq = 3.
    # Even frequencies: {2, 6}. Min even freq = 2.
    # Max difference = 3 - 2 = 1.
    assert maxDifference("aaabbcccccc") == 1


def test_even_freq_char_is_not_overall_least_frequent():
    """
    Tests that the character with the minimum even frequency is chosen, even if it's
    not the least frequent character in the string overall.
    """
    # s = "abbbbcccccc"
    # Frequencies: a=1(O), b=4(E), c=6(E)
    # The least frequent char 'a' (1) has an odd frequency.
    # We must subtract the minimum EVEN frequency (4), not the overall min freq.
    # Odd frequencies: {1}. Max odd freq = 1.
    # Even frequencies: {4, 6}. Min even freq = 4.
    # Max difference = 1 - 4 = -3.
    assert maxDifference("abbbbcccccc") == -3
