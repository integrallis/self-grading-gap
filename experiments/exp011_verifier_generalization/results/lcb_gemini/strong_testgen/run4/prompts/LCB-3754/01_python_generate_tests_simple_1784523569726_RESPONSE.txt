import pytest

def test_canonical_example_1():
    """
    Test case from Example 1 in the problem description.
    s = "NWSE", k = 1
    By changing s[2] ('S') to 'N', the string becomes "NWNE".
    The path is:
    - N: (0, 1), Manhattan distance = 1
    - W: (-1, 1), Manhattan distance = 2
    - N: (-1, 2), Manhattan distance = 3
    - E: (0, 2), Manhattan distance = 2
    The maximum Manhattan distance achieved is 3.
    """
    assert maxDistance("NWSE", 1) == 3

def test_canonical_example_2():
    """
    Test case from Example 2 in the problem description.
    s = "NSWWEW", k = 3
    By changing s[1] ('S') to 'N' and s[4] ('E') to 'W', the string becomes "NNWWWW".
    This uses 2 out of 3 available changes.
    The path is:
    - N: (0, 1), dist=1
    - N: (0, 2), dist=2
    - W: (-1, 2), dist=3
    - W: (-2, 2), dist=4
    - W: (-3, 2), dist=5
    - W: (-4, 2), dist=6
    The maximum Manhattan distance achieved is 6.
    """
    assert maxDistance("NSWWEW", 3) == 6

def test_no_changes():
    """
    Test with k=0, so no changes are allowed. The path is fixed.
    s = "NWS"
    - N: (0, 1), dist=1
    - W: (-1, 1), dist=2
    - S: (-1, 0), dist=1
    The maximum distance reached is 2.
    """
    assert maxDistance("NWS", 0) == 2

def test_unlimited_changes():
    """
    Test with k >= len(s), allowing all characters to be changed.
    The maximum distance is always len(s), as we can make all moves in one direction.
    s = "NSWWEW", k = 6, len(s) = 6.
    Change to "NNNNNN". The max distance reached is 6 at the end.
    """
    assert maxDistance("NSWWEW", 6) == 6

def test_more_changes_than_length():
    """
    Test with k > len(s). This is equivalent to k = len(s).
    The maximum distance is len(s).
    s = "WE", k = 5, len(s) = 2.
    Change to "NN". The max distance is 2.
    """
    assert maxDistance("WE", 5) == 2

    """

    assert maxDistance("N", 0) == 1

    The max distance is 1.

    assert maxDistance("E", 1) == 1

    - S: (1, 0), dist=1
    - W: (0, 0), dist=0

def test_canceling_moves_with_changes():
    """
    Test a path with canceling moves, but with enough k to optimize.
    s = "NESW", k = 2.
    With 2 changes, we can make the path unidirectional.
    For example, change 'S'->'N' and 'W'->'E'. Path becomes "NENE".
    - N: (0, 1), dist=1
    - E: (1, 1), dist=2
    - N: (1, 2), dist=3
    - E: (2, 2), dist=4
    The maximum distance is 4.

    assert maxDistance("NESW", 2) == 4

def test_all_same_direction():
    """
    Test a path where all moves are already in the same direction.
    s = "WWWW", k = 2.
    The path is already optimal for maximizing the -x coordinate.
    Changes do not help increase the maximum Manhattan distance.
    The max distance is len(s) = 4.
    """
    assert maxDistance("WWWW", 2) == 4

def test_long_alternating_path():
    """
    Test a longer, alternating path to check a more complex case.
    s = "WE" * 10 = "WEWEWEWEWEWEWEWEWEWE", len=20, k=5.
    The maximum distance can be found by considering prefixes and optimizing them.
    Consider the prefix of length 11, s[0..10] = "WEWEWEWEWEW".
    Original path has 6 'W's, 5 'E's, pos: (-1, 0).
    To maximize |-x+y|, we change 'E' moves to 'W' or 'N'.
    The prefix has 5 'E's. With k=5, we can change all of them.
    The optimized 11 moves are all 'W' or 'N'.
    The distance becomes 11.
    This is the maximum achievable for any prefix.
    """
    assert maxDistance("WE" * 10, 5) == 11

def test_zero_k_but_optimal_path():

    s = "E" * 10, k = 0

    The distance increases by 1 at each step, reaching a max of 10.
    """
    assert maxDistance("E" * 10, 0) == 10
