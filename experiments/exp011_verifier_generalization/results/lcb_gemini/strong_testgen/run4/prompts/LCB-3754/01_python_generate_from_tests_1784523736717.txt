def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0  # Starting at the origin (0, 0)
    max_distance = 0
    for move in s:
        if move == 'N':
            y += 1
        elif move == 'S':
            y -= 1
        elif move == 'E':
            x += 1
        elif move == 'W':
            x -= 1
        # Calculate current Manhattan distance
        current_distance = abs(x) + abs(y)
        # Update maximum distance
        max_distance = max(max_distance, current_distance)
    # Considering k changes, the maximum distance is limited by total moves + k
    return max_distance + k

```
