def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0  # Starting at the origin (0, 0)
    max_distance = 0

    # Count movements in each direction
    n_count = 0
    s_count = 0
    e_count = 0
    w_count = 0

    for move in s:
        if move == 'N':
            y += 1
            n_count += 1
        elif move == 'S':
            y -= 1
            s_count += 1
        elif move == 'E':
            x += 1
            e_count += 1
        elif move == 'W':
            x -= 1
            w_count += 1

        # Calculate current Manhattan distance
        current_distance = abs(x) + abs(y)
        # Update maximum distance
        max_distance = max(max_distance, current_distance)

    # Calculate possible max distance with k changes
    total_moves = len(s)
    effective_distance = total_moves + k

    return max(max_distance, effective_distance)  # Return the maximum of the two calculations
```
