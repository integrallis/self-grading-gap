## Implementation

```python
def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]

    # Check all possible starting positions in s for the prefix
    for i in range(len(s) - len(prefix) + 1):
        # Check if the prefix matches
        if s.startswith(prefix, i):
            # Check if the suffix can be found after the prefix's end
            if s.find(suffix, i + len(prefix)) != -1:
                return True
    return False
```
