    """
    Tests a string where all characters are the same.
    s = "bbb"
    'b': reverse alphabet index 25
    'b' @ index 1: 25 * 1 = 25
    'b' @ index 2: 25 * 2 = 50
    'b' @ index 3: 25 * 3 = 75
    Total reverse degree: 25 + 50 + 75 = 150
    """

def test_string_with_alphabet_in_order():
    """
    Tests a string containing the entire alphabet in order.
    s = "abcdefghijklmnopqrstuvwxyz"
    The calculation is sum_{i=1 to 26} (27-i) * i.
    This expands to 27 * sum(i) - sum(i^2).
    sum(i) for i=1..26 is 26*27/2 = 351.
    sum(i^2) for i=1..26 is 26*27*(2*26+1)/6 = 6201.
    Total: 27 * 351 - 6201 = 9477 - 6201 = 3276.
    """
    s = "abcdefghijklmnopqrstuvwxyz"
    assert reverseDegree(s) == 3276

def test_string_with_alphabet_in_reverse_order():
    """
    Tests a string containing the entire alphabet in reverse order.
    s = "zyxwvutsrqponmlkjihgfedcba"
    The calculation is sum_{i=1 to 26} i * i = sum(i^2).
    sum(i^2) for i=1..26 is 26*(26+1)*(2*26+1)/6 = 26*27*53/6 = 6201.
    """
    s = "zyxwvutsrqponmlkjihgfedcba"
    assert reverseDegree(s) == 6201

def test_max_length_string():
    """
    Tests a string of maximum possible length (1000) with a single repeating character 'z'.
    s = 'z' * 1000
    The reverse alphabet index for 'z' is 1.
    The calculation is sum_{i=1 to 1000} 1 * i.
    This is the sum of the first 1000 integers: 1000 * 1001 / 2 = 500500.
    """
    s = 'z' * 1000
    assert reverseDegree(s) == 500500
