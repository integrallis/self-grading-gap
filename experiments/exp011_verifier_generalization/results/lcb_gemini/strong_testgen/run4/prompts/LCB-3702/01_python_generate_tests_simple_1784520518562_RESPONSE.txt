def test_canonical_example_1():
    """
    Test with the first canonical example from the specification.
    spec: maxLength([1, 2, 1, 2, 1, 1, 1]) -> 5
    """
    # The longest product equivalent subarray is [1, 2, 1, 1, 1].
    # For arr = [1, 2, 1, 1, 1]:
    # prod(arr) = 1 * 2 * 1 * 1 * 1 = 2
    # gcd(arr) = gcd(1, 2, 1, 1, 1) = 1
    # lcm(arr) = lcm(1, 2, 1, 1, 1) = 2
    # prod == lcm * gcd -> 2 == 2 * 1, which is true.
    # Its length is 5.
    # No subarray of length 6 or 7 is product equivalent.
    # e.g., for arr = [1, 2, 1, 2, 1, 1]: prod=4, gcd=1, lcm=2. 4 != 1*2.
    assert maxLength([1, 2, 1, 2, 1, 1, 1]) == 5

def test_minimum_length_input():
    """
    Tests an input at the minimum specified length (2).
    Any subarray of length 2 is product equivalent.
    """
    # Input has length 2. The only non-empty subarrays are [7], [8], [7, 8].
    # [7]: prod=7, gcd=7, lcm=7. 7 != 7*7. Not PE.
    # [8]: prod=8, gcd=8, lcm=8. 8 != 8*8. Not PE.
    # [7, 8]: prod=56. gcd=1, lcm=56. 56 == 1*56. PE. Length 2.
    # Max length is 2.
    assert maxLength([7, 8]) == 2

    """
    Tests a case where no subarray of length > 2 is product equivalent.
    The answer must be 2, as all subarrays of length 2 are product equivalent.
    """
    # The input is [6, 8, 4, 9].
    # All subarrays of length 2 are product equivalent, so the answer is at least 2.
    # Check length 3 subarrays:
    # [6, 8, 4]: prod=192, gcd=2, lcm=24. 192 != 2*24.
    # [8, 4, 9]: prod=288, gcd=1, lcm=72. 288 != 1*72.
    # Check length 4 subarray:
    # [6, 8, 4, 9]: prod=1728, gcd=1, lcm=72. 1728 != 1*72.
    # The longest product equivalent subarray has length 2.
    assert maxLength([6, 8, 4, 9]) == 2
