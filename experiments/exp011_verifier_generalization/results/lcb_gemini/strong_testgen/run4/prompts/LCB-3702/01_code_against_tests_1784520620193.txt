```python
from typing import List
import math
from functools import reduce

def maxLength(nums: List[int]) -> int:
    def prod(arr):
        return reduce(lambda x, y: x * y, arr)

    def gcd_arr(arr):
        return reduce(math.gcd, arr)

    def lcm(a, b):
        return (a * b) // math.gcd(a, b) if a and b else 0

    def lcm_arr(arr):
        return reduce(lcm, arr)

    n = len(nums)
    max_length = 0

    for start in range(n):
        for end in range(start + 1, n + 1):
            subarray = nums[start:end]
            p = prod(subarray)
            g = gcd_arr(subarray)
            l = lcm_arr(subarray)
            if p == g * l:
                max_length = max(max_length, end - start)

    return max_length
```
