```python
from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n < 2:
        return n
    dp = [1] * n  # Initialize dp array, where dp[i] is the length of longest subsequence ending at index i
    for i in range(n):
        for j in range(i):
            diff_i = abs(nums[i] - nums[j])
            if j > 0:
                prev_diff = abs(nums[j] - nums[j-1])
                if diff_i <= prev_diff:
                    dp[i] = max(dp[i], dp[j] + 1)
            else:
                dp[i] = max(dp[i], 2)  # At least two elements can form a valid subsequence
    return max(dp)
```
