from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = [1] * n  # Initialize dp array
    max_length = 0
    for i in range(n):
        for j in range(i):
            diff_i = abs(nums[i] - nums[j])
            if j == 0 or (abs(nums[j] - nums[j-1]) >= diff_i):
                dp[i] = max(dp[i], dp[j] + 1)
        max_length = max(max_length, dp[i])
    return max_length
```
