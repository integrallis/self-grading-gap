import pytest
from solution import hasSpecialSubstring


def test_canonical_example_1():
    # spec: s = "aaabaaa", k = 3 -> true
    # The substring "aaa" at index 4 (s[4:7]) satisfies the conditions:
    # 1. Length is 3 (k).
    # 2. Consists of one character ('a').
    # 3. The character before (s[3]) is 'b', which is different.
    # 4. There is no character after.
    assert hasSpecialSubstring("aaabaaa", 3) is True


def test_canonical_example_2():
    # spec: s = "abc", k = 2 -> false
    # Substrings of length 2 are "ab" and "bc".
    # Neither consists of only one distinct character.
    assert hasSpecialSubstring("abc", 2) is False


def test_special_substring_in_middle():
    # A valid substring "ccc" is present in the middle of the string.
    # s = "abcccde", k = 3
    # Substring "ccc" at index 2 (s[2:5]):
    # Before: s[1] = 'b' (different). After: s[5] = 'd' (different).
    assert hasSpecialSubstring("abcccde", 3) is True


def test_special_substring_at_start():
    # A valid substring "aaa" is at the start of the string.
    # s = "aaab", k = 3
    # Substring "aaa" at index 0 (s[0:3]):
    # Before: none. After: s[3] = 'b' (different).
    assert hasSpecialSubstring("aaab", 3) is True


def test_special_substring_at_end():
    # A valid substring "aaa" is at the end of the string.
    # s = "baaa", k = 3
    # Substring "aaa" at index 1 (s[1:4]):
    # Before: s[0] = 'b' (different). After: none.
    assert hasSpecialSubstring("baaa", 3) is True


def test_special_substring_is_entire_string():
    # The entire string is a valid special substring.

    assert hasSpecialSubstring("zzzz", 4) is True

    # Before: none. After: s[2] = 'b' (same, fails).

    assert hasSpecialSubstring("abbb", 2) is False

    # The run of 'c's is of length 4, which is > k.
    # This creates two monochrome substrings of length 3:
    # s = "abccccde", k = 3
    # Substring "ccc" at index 2: Before is 'b' (ok), but after is 'c' (fail).
    # Substring "ccc" at index 3: Before is 'c' (fail), but after is 'd' (ok).
    # Neither is a special substring.
    assert hasSpecialSubstring("abccccde", 3) is False

    # s[2:100] = "a"*98. before='a'(fail), after='a'(fail).
