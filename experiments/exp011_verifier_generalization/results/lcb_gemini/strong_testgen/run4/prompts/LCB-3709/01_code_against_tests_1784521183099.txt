```python
def hasSpecialSubstring(s: str, k: int) -> bool:
    n = len(s)
    for i in range(n - k + 1):
        substring = s[i:i + k]
        if len(set(substring)) == 1:
            # Check the character before the substring
            before_ok = (i == 0 or s[i - 1] != substring[0])
            # Check the character after the substring
            after_ok = (i + k == n or s[i + k] != substring[0])
            if before_ok and after_ok:
                return True
    return False
```
