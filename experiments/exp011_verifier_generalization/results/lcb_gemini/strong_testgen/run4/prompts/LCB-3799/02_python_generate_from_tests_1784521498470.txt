from typing import List


def totalNumbers(digits: List[int]) -> int:
    from itertools import permutations
    even_digits = {d for d in digits if d % 2 == 0}
    count = 0

    for last_digit in even_digits:
        remaining_digits = digits.copy()
        remaining_digits.remove(last_digit)
        for first_digit in (d for d in remaining_digits if d != 0):
            # Create permutations of the remaining digits
            for perm in permutations(remaining_digits, 1):
                count += 1
            # Add the last digit to the count
            count += len(remaining_digits) - 1

    return count
```
