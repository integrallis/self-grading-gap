from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize a DP array to store the minimum cost up to each house, considering the constraints.
    dp = [[0] * 3 for _ in range(n)]

    # Fill in the base case for the first house.
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Iterate through each house starting from the second one.
    for i in range(1, n):
        for j in range(3):
            # Calculate the minimum cost for painting house i with color j.
            # We cannot use the same color as the previous house.
            dp[i][j] = min(dp[i - 1][(j + 1) % 3], dp[i - 1][(j + 2) % 3]) + cost[i][j]

            # If we are in the second half of the houses, we need to consider the equidistant constraint.
            if i < n // 2:
                continue
            # For the second half, we also need to prevent equidistant houses from having the same color.
            opposite_color = 2 - j  # The color of the opposite house in the second half
            dp[i][j] = min(dp[i][j], dp[i - 1][(j + 1) % 3] + cost[i][j], dp[i - 1][(j + 2) % 3] + cost[i][j])

    # The result will be the minimum cost of painting the last house with any of the three colors.
    return min(dp[n - 1])
