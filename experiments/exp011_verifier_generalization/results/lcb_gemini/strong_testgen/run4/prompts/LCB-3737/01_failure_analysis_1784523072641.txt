    # Fill in the base case for the first house.
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Iterate through each house starting from the second one.
    for i in range(1, n):
        for j in range(3):
            # Calculate the minimum cost for painting house i with color j.
            # We cannot use the same color as the previous house.
