from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    from collections import defaultdict, deque

    # Build the graph with edges directed away from node 0
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))

    # Function to check if we can reach node 0 with a max weight limit
    def canReachWithMaxWeight(max_weight: int) -> bool:
        visited = [False] * n
        queue = deque()
        for i in range(n):
            visited[i] = (len([1 for v, w in graph[i] if w <= max_weight]) <= threshold)
            if visited[i]:
                queue.append(i)
        visited[0] = False  # node 0 can never have outgoing edges

        while queue:
            node = queue.popleft()
            for neighbor, weight in graph[node]:
                if not visited[neighbor] and weight <= max_weight:
                    visited[neighbor] = True
                    queue.append(neighbor)
        return all(visited[i] for i in range(n))

    # Binary search for the minimum possible maximum edge weight
    left, right = 1, max(w for _, _, w in edges)
    answer = -1
    while left <= right:
        mid = (left + right) // 2
        if canReachWithMaxWeight(mid):
            answer = mid
            right = mid - 1
        else:
            left = mid + 1
    return answer
