# Note: The problem asks to maximize the LONGEST continuous period of free time.
# The core idea is that we can create a large free time slot by consolidating
# smaller free time slots. This is done by moving meetings to be adjacent to
# each other. With `m` moves, we can make `m+1` meetings contiguous.
#
# A maximal free slot can be created in one of three locations:
# 1. At the beginning of the event: [0, T]
# 2. At the end of the event: [T, eventTime]
# 3. In the middle, between two meetings.
#
# The strategy is to fix two "anchors" and pack all meetings between them
# together, leaving one large free slot. The anchors can be the event
# boundaries (t=0, t=eventTime) or any two meetings that are not moved.
#
# If we fix meetings `i` and `j` as anchors, we can move the `j-i-1` meetings
# between them. This must be at most `k` moves. The free space we can create
# is the original space between `i` and `j` minus the total duration of the
# meetings we pack.
#
# Let D[m] be the duration of meeting m.
# Let PS[m] be the prefix sum of durations up to m-1.
# - To create a gap at the start, anchored by meeting `j`, we move `j` meetings.
#   Requires `j <= k`. Free time = `startTime[j] - sum(D[0]...D[j-1])`.
# - To create a gap at the end, anchored by meeting `i`, we move `n-1-i` meetings.
#   Requires `n-1-i <= k`. Free time = `(eventTime - endTime[i]) - sum(D[i+1]...D[n-1])`.
# - To create a middle gap, anchored by `i` and `j`, we move `j-i-1` meetings.
#   Requires `j-i-1 <= k`. Free time = `(startTime[j] - endTime[i]) - sum(D[i+1]...D[j-1])`.
#
# The tests below calculate the expected output based on this logic, trying all valid
# combinations of anchors.

from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    durations = [endTime[i] - startTime[i] for i in range(n)]
    total_duration = sum(durations)
    max_free_time = 0

    # Check free time before the first meeting
    max_free_time = max(max_free_time, startTime[0])

    # Check free time after the last meeting
    max_free_time = max(max_free_time, eventTime - endTime[-1])

    # Check free time between meetings
    for i in range(n - 1):
        # Initial gap between end of meeting i and start of meeting i+1
        gap = startTime[i + 1] - endTime[i]
        max_free_time = max(max_free_time, gap)

    # Try to create larger free time by moving meetings
    for i in range(n):
        for j in range(i + 1, n):
            # Meetings between i and j can be moved
            if j - i - 1 <= k:
                # Calculate the space and duration of the meetings moved
                space = (startTime[j] - endTime[i]) - sum(durations[i + 1:j])
                if space >= 0:
                    max_free_time = max(max_free_time, space)

    # Total free time possible
    return max(max_free_time, eventTime - total_duration)
