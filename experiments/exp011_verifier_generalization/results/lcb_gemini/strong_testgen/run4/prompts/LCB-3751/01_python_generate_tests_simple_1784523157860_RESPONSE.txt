import pytest
from typing import List

# This is the test suite for a function to be implemented in `solution.py`.

# The function signature is:
# def maxFrequency(nums: List[int], k: int) -> int:

from solution import maxFrequency


def test_canonical_example_1():
    """
    spec: maxFrequency([1,2,3,4,5,6], 1) -> 2
    Explanation: nums has one '1'. We can select a subarray and add an integer 'x'
    to make more '1's. If we select subarray nums[5..5] which is [6], and add x=-5,
    it becomes [1]. The array becomes [1,2,3,4,5,1], and the frequency of 1 is 2.
    This represents a gain of 1. The maximum possible gain is 1.
    Total frequency = 1 (initial) + 1 (gain) = 2.
    The example's explanation mentions adding -5 to nums[2..5] which makes nums[5] (6) become 1,
    also resulting in a frequency of 2.
    """
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2


def test_canonical_example_2():
    """
    spec: maxFrequency([10,2,3,4,5,5,4,3,2,2], 10) -> 4
    Explanation: The initial frequency of 10 is 1 (at index 0).
    We can convert the three '2's into '10's by adding 8.
    This is done by selecting a subarray that contains all three '2's (at indices 1, 8, 9)
    but not the original '10'. For subarray nums[1..9] ([2,3,4,5,5,4,3,2,2]), we add x=8.
    The three '2's become '10's.
    The final array is [10, 10, 11, 12, 13, 13, 12, 11, 10, 10].
    The frequency of 10 is 4. This is a gain of 3.
    Total frequency = 1 (initial) + 3 (gain) = 4.
    """
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4


def test_single_element_array_is_k():
    """
    Test with a single-element array where the element is already k.
    """
    # spec: maxFrequency([5], 5) -> 1
    # Initial frequency is 1. No other elements to convert. No operation can increase the count.
    assert maxFrequency([5], 5) == 1


def test_single_element_array_not_k():
    """
    Test with a single-element array that can be converted to k.
    """
    # spec: maxFrequency([5], 10) -> 1
    # Initial frequency is 0. We can convert the 5 to 10 by adding 5.
    # The new frequency is 1.
    assert maxFrequency([5], 10) == 1


def test_all_elements_are_k():
    """

    assert maxFrequency([7, 7, 7, 7], 7) == 4

    """

    """

    # Initial frequency is 0. We can convert any single element to 10.
    # For example, convert 3 to 10. The max frequency is 1.
    assert maxFrequency([1, 2, 3], 10) == 1


def test_all_elements_same_but_not_k():
    """
    Test where all elements can be converted to k.
    """
    # spec: maxFrequency([5, 5, 5, 5], 1) -> 4
    # Initial frequency is 0. We can convert all four 5s to 1s by choosing the
    # whole array as the subarray and adding x = -4.
    # The new frequency is 4.
    assert maxFrequency([5, 5, 5, 5], 1) == 4


def test_best_gain_requires_choosing_best_value_to_convert():
    """
    Test a case where multiple values could be converted, and we must choose the best one.
    """
    # spec: maxFrequency([4, 2, 4, 2, 4, 3, 3], 4) -> 5
    # Initial frequency of 4 is 3.
    # We could convert the two '2's. This provides a gain of at most 2.
    # We could convert the two '3's. This provides a gain of at most 2.
    # Let's analyze gains more carefully.
    # To convert '3's: choose subarray nums[5..6] = [3, 3]. It contains no '4's. Gain is 2.
    # To convert '2's: choose subarray nums[1..3] = [2, 4, 2]. Contains two '2's, one '4'. Gain is 2-1=1.
    # The best strategy is to convert the '3's.
    # Total frequency = 3 (initial) + 2 (gain) = 5.
    assert maxFrequency([4, 2, 4, 2, 4, 3, 3], 4) == 5


def test_best_gain_from_subarray_with_no_k():
    """
    Test where the optimal subarray to modify contains no k's.
    """
    # spec: maxFrequency([1, 1, 1, 5, 5], 5) -> 5
    # Initial frequency of 5 is 2.
    # We can convert the three '1's into '5's by adding 4.
    # This requires a subarray covering indices 0, 1, 2, e.g., nums[0..2].
    # This subarray contains three '1's and zero '5's. The gain is 3.
    # Total frequency = 2 (initial) + 3 (gain) = 5.
    assert maxFrequency([1, 1, 1, 5, 5], 5) == 5


def test_interspersed_k_and_value_to_convert():
    """
    Test where the value to be converted (v) and k are interspersed, limiting the gain.
    """
    # spec: maxFrequency([1, 5, 1, 5, 1], 1) -> 4
    # Initial frequency of 1 is 3.
    # We want to convert 5's to 1's. There are two 5's.
    # If we pick a subarray covering both 5's (e.g., nums[1..3]), it also contains a 1.
    # This subarray [5, 1, 5] would give a gain of count(5)-count(1) = 2-1=1.
    # A simpler choice is a subarray with one 5, e.g. nums[1..1]=[5]. Gain is 1.
    # The max gain is 1. Total frequency = 3 (initial) + 1 (gain) = 4.
    assert maxFrequency([1, 5, 1, 5, 1], 1) == 4
