from typing import List

def maxFrequency(nums: List[int], k: int) -> int:
    nums.sort()
    left = 0
    total = 0
    max_freq = 0

    for right in range(len(nums)):
        total += nums[right]
        # Increase frequency of k while total adjustment with x is valid
        while (total + (right - left + 1) * k) < nums[right] * (right - left + 1):
            total -= nums[left]
            left += 1
        max_freq = max(max_freq, right - left + 1)

    return max_freq
```
