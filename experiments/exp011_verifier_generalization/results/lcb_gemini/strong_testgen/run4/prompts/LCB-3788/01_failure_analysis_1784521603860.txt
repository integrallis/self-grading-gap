```python
from typing import List

def maxSum(nums: List[int]) -> int:
    seen = set()
    max_sum = float('-inf')
    current_sum = 0

    for num in nums:
        if num not in seen:
            seen.add(num)
            current_sum += num
        else:
            # If we see a duplicate, we need to reset the current sum
            # and start again from the next unique element
            current_sum = num
            seen = {num}
        max_sum = max(max_sum, current_sum)

    return max_sum
```
