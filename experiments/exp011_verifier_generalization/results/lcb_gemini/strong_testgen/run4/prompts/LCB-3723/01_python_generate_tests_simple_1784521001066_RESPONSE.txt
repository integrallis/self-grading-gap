def test_canonical_example_1():
    """
    Tests the first example from the specification.
    """
    nums = [1, 3, 2, 1, 5, 4]
    k = 2
    # Calculation:
    # i=0, nums[0]=1: check nums[2]=2. 1 > 2 is False.
    # i=1, nums[1]=3: check nums[3]=1. 3 > 1 is True. Good. (i-k < 0)
    # i=2, nums[2]=2: check nums[0]=1, nums[4]=5. 2 > 1 is True, but 2 > 5 is False.
    # i=3, nums[3]=1: check nums[1]=3, nums[5]=4. 1 > 3 is False.
    # i=4, nums[4]=5: check nums[2]=2. 5 > 2 is True. Good. (i+k >= len)
    # i=5, nums[5]=4: check nums[3]=1. 4 > 1 is True. Good. (i+k >= len)
    # Sum = 3 + 5 + 4 = 12
    expected = 12
    assert sumOfGoodNumbers(nums, k) == expected
