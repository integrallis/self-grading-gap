from typing import List


def largestInteger(nums: List[int], k: int) -> int:
    from collections import defaultdict
    count = defaultdict(int)
    n = len(nums)

    # Count occurrences of each number in all subarrays of size k
    for i in range(n - k + 1):
        seen = set()
        for j in range(k):
            if nums[i + j] not in seen:
                count[nums[i + j]] += 1
                seen.add(nums[i + j])

    # Find the largest number that appears in exactly one subarray
    largest_missing = -1
    for num, cnt in count.items():
        if cnt == 1:
            largest_missing = max(largest_missing, num)

    # Check for the condition on largest_missing
    if largest_missing == -1 and count:
        largest_missing = max(num for num, cnt in count.items() if cnt == 1)

    return largest_missing
