from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[[float('-inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0] if coins[0][0] >= 0 else 0  # Start with the top-left corner

    for i in range(m):
        for j in range(n):
            for k in range(3):  # k is the count of neutralizations used
                if i == 0 and j == 0:
                    continue
                current_value = coins[i][j]
                if current_value < 0:
                    if k > 0:
                        # Neutralize the robber
                        dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k-1] if i > 0 else float('-inf'), dp[i][j-1][k-1] if j > 0 else float('-inf')))
                    # If not neutralized
                    dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k] - abs(current_value) if i > 0 else float('-inf'), dp[i][j-1][k] - abs(current_value) if j > 0 else float('-inf')))
                else:
                    dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k] + current_value if i > 0 else float('-inf'), dp[i][j-1][k] + current_value if j > 0 else float('-inf')))

    max_coins = float('-inf')
    for k in range(3):
        max_coins = max(max_coins, dp[m-1][n-1][k])
    return max_coins
