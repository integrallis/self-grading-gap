from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    power_of_2 = [1] * (n + 1)  # Precompute powers of 2 modulo MOD

    for i in range(1, n + 1):
        power_of_2[i] = (power_of_2[i - 1] * 2) % MOD

    for i in range(n):
        # Contribution of nums[i] as maximum
        max_contributions = (power_of_2[i] * power_of_2[n - 1 - i]) % MOD
        total_sum = (total_sum + nums[i] * max_contributions) % MOD

        # Contribution of nums[i] as minimum
        min_contributions = (power_of_2[n - 1 - i] * power_of_2[i]) % MOD
        total_sum = (total_sum + nums[i] * min_contributions) % MOD

    return total_sum % MOD
```
