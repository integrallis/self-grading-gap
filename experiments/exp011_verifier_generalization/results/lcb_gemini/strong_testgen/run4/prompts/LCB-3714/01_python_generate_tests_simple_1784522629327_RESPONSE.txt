def test_large_element_values_modulo():
    """
    Tests with large element values that require modulo arithmetic.
    """

    nums = [10**9, 10**9 - 1]
    k = 2
    # Sorted nums: [10^9 - 1, 10^9].
    # Subsequences: [10^9-1], [10^9], [10^9-1, 10^9]
    # Sums of min+max:
    # (10^9-1 + 10^9-1) + (10^9 + 10^9) + (10^9-1 + 10^9)
    # = (2*10^9-2) + (2*10^9) + (2*10^9-1) = 6*10^9 - 3
    # Modulo 10^9+7:
    # 6 * (10^9+7 - 7) - 3 = 6*(MOD-7) - 3 = -42 - 3 = -45
    # -45 mod MOD = MOD - 45 = 1000000007 - 45 = 999999962

    # With n=70, sum = 71 * (2^70 - 1)
