## Implementation

```python
from typing import List

def countPartitions(nums: List[int]) -> int:
    total_sum = sum(nums)
    current_left_sum = 0
    count = 0
    for i in range(len(nums) - 1):  # Iterate until the second to last element
        current_left_sum += nums[i]
        current_right_sum = total_sum - current_left_sum
        if (current_left_sum - current_right_sum) % 2 == 0:
            count += 1
    return count
```
