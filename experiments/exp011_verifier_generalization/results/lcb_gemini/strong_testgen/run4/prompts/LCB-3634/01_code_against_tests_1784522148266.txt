```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    for i in range(len(s)):
        if marked[i]:
            continue
        mirror = chr(219 - ord(s[i]))  # Calculate mirror character
        for j in range(i - 1, -1, -1):  # Look for the closest unmarked mirror
            if not marked[j] and s[j] == mirror:
                score += i - j
                marked[i] = marked[j] = True
                break
    return score

```
