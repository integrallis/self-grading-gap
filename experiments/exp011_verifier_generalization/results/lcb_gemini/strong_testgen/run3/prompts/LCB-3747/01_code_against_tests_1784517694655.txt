```python
from typing import List

def maxAdjacentDistance(nums: List[int]) -> int:
    max_diff = 0
    n = len(nums)
    for i in range(n):
        # Calculate the absolute difference between adjacent elements, including circular
        diff = abs(nums[i] - nums[(i + 1) % n])
        max_diff = max(max_diff, diff)
    return max_diff
```
