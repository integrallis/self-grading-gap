from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[[float('-inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0] if coins[0][0] >= 0 else 0

    for i in range(m):
        for j in range(n):
            for k in range(3):
                if i == 0 and j == 0:
                    continue
                max_coins = float('-inf')
                if i > 0:
                    max_coins = max(max_coins, dp[i - 1][j][k])
                if j > 0:
                    max_coins = max(max_coins, dp[i][j - 1][k])
                if coins[i][j] >= 0:
                    dp[i][j][k] = max(dp[i][j][k], max_coins + coins[i][j])
                else:
                    robber_value = -coins[i][j]
                    if k > 0:
                        dp[i][j][k] = max(dp[i][j][k], max_coins)
                    if k < 2:
                        dp[i][j][k + 1] = max(dp[i][j][k + 1], max_coins - robber_value)
    return max(dp[m - 1][n - 1])
