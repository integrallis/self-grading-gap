def test_canonical_example_1():
    # spec: findValidPair("2523533") -> "23"
    # In "2523533", count('2')=2 and count('3')=3.
    # The first pair "25" is invalid because count('5') is 2, not 5.
    # The next distinct pair is "23", where count('2')=2 and count('3')=3. This is the first valid pair.
    assert findValidPair("2523533") == "23"

    # The next pair "21" is valid because count('2')=2 and count('1')=1.
