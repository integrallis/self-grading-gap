```python
from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    left, right = 1, max(edge[2] for edge in edges)
    
    def can_form_graph(max_weight: int) -> bool:
        from collections import defaultdict, deque
        graph = defaultdict(list)
        out_degrees = [0] * n
        
        for u, v, w in edges:
            if w <= max_weight and out_degrees[u] < threshold:
                graph[v].append(u)
                out_degrees[u] += 1
        
        reachable = [False] * n
        queue = deque([0])
        reachable[0] = True
        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if not reachable[neighbor]:
                    reachable[neighbor] = True
                    queue.append(neighbor)
        
        return all(reachable)
    
    while left < right:
        mid = (left + right) // 2
        if can_form_graph(mid):
            right = mid
        else:
            left = mid + 1
    
    return left if can_form_graph(left) else -1
```
