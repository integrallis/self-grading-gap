from typing import List
import pytest

from solution import minCosts


def test_canonical_example_1():
    """
    Tests the first example from the specification.
    """
    cost = [5, 3, 4, 1, 3, 2]
    # To reach position i, we must pay to get past at least one person j <= i.
    # The minimum cost to reach i is the minimum cost of paying any person j <= i.
    # Once we pay cost[j], we are at position j and can move to any k > j for free.
    # Expected calculation:
    # answer[0] = min(cost[0]) = min(5) = 5
    # answer[1] = min(cost[0], cost[1]) = min(5, 3) = 3
    # answer[2] = min(cost[0..2]) = min(5, 3, 4) = 3
    # answer[3] = min(cost[0..3]) = min(5, 3, 4, 1) = 1
    # answer[4] = min(cost[0..4]) = min(5, 3, 4, 1, 3) = 1
    # answer[5] = min(cost[0..5]) = min(5, 3, 4, 1, 3, 2) = 1
    expected = [5, 3, 3, 1, 1, 1]
    assert minCosts(cost) == expected

def test_strictly_decreasing_costs():
    """
    Tests with a list of strictly decreasing costs.
    The result should be the input list itself, as each new cost is a new minimum.
    """
    cost = [100, 50, 25, 10]
    # Expected calculation:
    # answer[0] = min(100) = 100
    # answer[1] = min(100, 50) = 50
    # answer[2] = min(100, 50, 25) = 25

    expected = [100, 50, 25, 10]
