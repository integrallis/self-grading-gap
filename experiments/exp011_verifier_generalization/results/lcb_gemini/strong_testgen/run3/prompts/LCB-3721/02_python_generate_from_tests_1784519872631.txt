```python
from typing import List

def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers
    online_status = [True] * numberOfUsers
    offline_until = [0] * numberOfUsers

    # Sort events based on timestamp
    events.sort(key=lambda x: int(x[1]))

    for event in events:
        event_type, timestamp, data = event
        timestamp = int(timestamp)

        # Process offline events
        if event_type == "OFFLINE":
            user_id = int(data)
            online_status[user_id] = False
            offline_until[user_id] = timestamp + 60

        # Process message events
        elif event_type == "MESSAGE":
            # Update online status for users going offline before this message
            for user_id in range(numberOfUsers):
                if not online_status[user_id] and timestamp >= offline_until[user_id]:
                    online_status[user_id] = True

            mentions_string = data.split()
            mentioned_users = set()

            for mention in mentions_string:
                if mention.startswith("id"):
                    user_id = int(mention[2:])
                    mentions[user_id] += 1
                    mentioned_users.add(user_id)
                elif mention == "ALL":
                    for user_id in range(numberOfUsers):
                        mentions[user_id] += 1
                elif mention == "HERE":
                    for user_id in range(numberOfUsers):
                        if online_status[user_id]:
                            mentions[user_id] += 1

    return mentions
```
