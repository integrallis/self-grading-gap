from typing import List

def maxSum(nums: List[int]) -> int:
    unique_elements = set()
    current_sum = 0
    max_sum = float('-inf')

    for num in nums:
        if num not in unique_elements:
            unique_elements.add(num)
            current_sum += num
        else:
            max_sum = max(max_sum, current_sum)
            # Reset for the next unique sequence
            unique_elements.clear()
            unique_elements.add(num)
            current_sum = num

    # Final check for the last sequence
    max_sum = max(max_sum, current_sum)

    if max_sum == float('-inf'):
        return max(nums)
    return max(max_sum, 0)
