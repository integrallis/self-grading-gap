from typing import List

def maxSum(nums: List[int]) -> int:
    unique_elements = set()
    current_sum = 0
    max_sum = float('-inf')

    for num in nums:
        if num not in unique_elements:
            unique_elements.add(num)
            current_sum += num
        else:
            max_sum = max(max_sum, current_sum)
            # Reset for the next unique sequence
            unique_elements.clear()
            unique_elements.add(num)
            current_sum = num

    # Final check for the last sequence
    max_sum = max(max_sum, current_sum)

    return max(max_sum, current_sum, 0)  # Ensure we return at least 0 for non-negative sums.
```
