from typing import List

def maxSum(nums: List[int]) -> int:
    unique_elements = set()
    max_sum = float('-inf')
    current_sum = 0

    for num in nums:
        if num not in unique_elements:
            unique_elements.add(num)
            current_sum += num
        else:
            max_sum = max(max_sum, current_sum)
            # Reset the current sum and unique elements
            unique_elements.clear()
            unique_elements.add(num)
            current_sum = num

    return max(max_sum, 0)  # Ensure we return at least 0 for non-negative sums.
