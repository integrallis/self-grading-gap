import pytest
from solution import reverseDegree


def test_canonical_example_1():
    """
    Tests the first example from the specification.
    s = "abc"
    'a' (value 26) at index 1: 26 * 1 = 26
    'b' (value 25) at index 2: 25 * 2 = 50
    'c' (value 24) at index 3: 24 * 3 = 72
    Total = 26 + 50 + 72 = 148
    """
    assert reverseDegree("abc") == 148


def test_canonical_example_2():
    """
    Tests the second example from the specification.
    s = "zaza"
    'z' (value 1) at index 1: 1 * 1 = 1
    'a' (value 26) at index 2: 26 * 2 = 52
    'z' (value 1) at index 3: 1 * 3 = 3
    'a' (value 26) at index 4: 26 * 4 = 104
    Total = 1 + 52 + 3 + 104 = 160
    """
    assert reverseDegree("zaza") == 160


def test_min_length_string_start_of_alphabet():
    """
    Tests a string of minimum possible length (1) with a character from the start of the alphabet.
    s = "a"
    'a' (value 26) at index 1: 26 * 1 = 26
    Total = 26
    """

def test_min_length_string_end_of_alphabet():
    """

    'z' (value 1) at index 1: 1 * 1 = 1
    Total = 1

def test_string_with_all_same_character():
    """
    Tests a string where all characters are the same to verify positional multiplication.
    s = "ccc"
    'c' (value 24) at index 1: 24 * 1 = 24
    'c' (value 24) at index 2: 24 * 2 = 48
    'c' (value 24) at index 3: 24 * 3 = 72
    Total = 24 + 48 + 72 = 144
    """
    assert reverseDegree("ccc") == 144


def test_full_alphabet_string():
    """
    Tests a string containing all characters of the alphabet once.
    s = "abcdefghijklmnopqrstuvwxyz"
    'a' (26)*1 + 'b' (25)*2 + ... + 'z' (1)*26
    The sum is sum((27-i) * i for i in range(1, 27))
    = 27 * sum(i) - sum(i*i)
    = 27 * (26*27/2) - (26*27*53/6)
    = 27 * 351 - 6201 = 9477 - 6201 = 3276
    """
    assert reverseDegree("abcdefghijklmnopqrstuvwxyz") == 3276


def test_max_length_string_with_same_char():
    """
    Tests a string at the maximum specified length (1000).
    s = 'a' * 1000
    The calculation is the sum of (26 * i) for i from 1 to 1000.
    = 26 * sum(i for i=1..1000)
    = 26 * (1000 * 1001 / 2)
    = 26 * 500500 = 13013000
    """
    assert reverseDegree('a' * 1000) == 13013000
