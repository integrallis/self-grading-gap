```python
from typing import List

def countPartitions(nums: List[int]) -> int:
    total_sum = sum(nums)
    if total_sum % 2 != 0:
        return 0
    left_sum = 0
    count = 0
    for i in range(len(nums) - 1):
        left_sum += nums[i]
        right_sum = total_sum - left_sum
        if (left_sum - right_sum) % 2 == 0:
            count += 1
    return count
```
