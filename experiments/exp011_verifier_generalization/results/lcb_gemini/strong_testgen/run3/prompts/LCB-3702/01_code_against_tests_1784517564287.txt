```python
from typing import List
import math

def maxLength(nums: List[int]) -> int:
    def gcd(a: int, b: int) -> int:
        while b:
            a, b = b, a % b
        return a

    def lcm(a: int, b: int) -> int:
        return abs(a * b) // gcd(a, b)

    def gcd_list(arr: List[int]) -> int:
        result = arr[0]
        for num in arr[1:]:
            result = gcd(result, num)
        return result

    def lcm_list(arr: List[int]) -> int:
        result = arr[0]
        for num in arr[1:]:
            result = lcm(result, num)
        return result

    max_len = 0
    n = len(nums)
    for i in range(n):
        for j in range(i, n):
            subarray = nums[i:j + 1]
            product = math.prod(subarray)
            g = gcd_list(subarray)
            l = lcm_list(subarray)
            if product == l * g:
                max_len = max(max_len, j - i + 1)
    return max_len
```
