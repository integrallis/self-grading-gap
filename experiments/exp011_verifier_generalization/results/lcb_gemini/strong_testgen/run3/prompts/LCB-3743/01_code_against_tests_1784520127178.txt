```python
from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    gaps = []

    # Calculate gaps before the first meeting, between meetings, and after the last meeting
    gaps.append(startTime[0])  # Gap from 0 to the start of the first meeting
    for i in range(1, n):
        gaps.append(startTime[i] - endTime[i - 1])  # Gap between end of previous meeting and start of current meeting
    gaps.append(eventTime - endTime[-1])  # Gap from end of the last meeting to eventTime

    # Find the maximum sum of k+1 gaps
    max_free_time = 0
    current_sum = sum(gaps[:k + 1])
    max_free_time = current_sum

    for i in range(k + 1, len(gaps)):
        current_sum += gaps[i] - gaps[i - (k + 1)]  # slide the window
        max_free_time = max(max_free_time, current_sum)

    return max_free_time
```
