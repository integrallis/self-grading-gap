from typing import List

def maxFrequency(nums: List[int], k: int) -> int:
    count = 0
    nums.sort()
    left = 0
    total = 0
    for right in range(len(nums)):
        total += nums[right]
        while total + (right - left + 1) * k > (right - left + 1) * nums[right]:
            total -= nums[left]
            left += 1
        count = max(count, right - left + 1)
    return count
```
