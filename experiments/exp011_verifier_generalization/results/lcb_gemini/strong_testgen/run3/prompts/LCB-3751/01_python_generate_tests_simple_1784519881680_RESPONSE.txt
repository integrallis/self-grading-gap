from typing import List

def test_canonical_example_1():
    """
    Tests the first example from the specification.
    """
    # spec: After adding -5 to nums[2..5], 1 has a frequency of 2 in [1, 2, -2, -1, 0, 1].
    # Initial nums: [1, 2, 3, 4, 5, 6], k = 1.
    # The original 1 at index 0 remains.
    # The 6 at index 5 is converted to 1 by adding -5. This requires an operation on a
    # subarray that includes index 5, e.g., nums[2..5].
    # Final count of 1 is 2.
    assert maxFrequency(nums=[1, 2, 3, 4, 5, 6], k=1) == 2


def test_canonical_example_2():
    """
    Tests the second example from the specification.
    """
    # spec: After adding 8 to nums[1..9], 10 has a frequency of 4 in [10, 10, 11, 12, 13, 13, 12, 11, 10, 10].
    # Initial nums: [10, 2, 3, 4, 5, 5, 4, 3, 2, 2], k = 10.
    # The original 10 at index 0 is not in the modified subarray and remains 10.
    # The three 2s at indices 1, 8, and 9 are converted to 10 by adding 8 to the subarray nums[1..9].
    # Final count of 10 is 1 (original) + 3 (converted) = 4.
    assert maxFrequency(nums=[10, 2, 3, 4, 5, 5, 4, 3, 2, 2], k=10) == 4


def test_no_operation_is_optimal():
    """
    Tests a case where the initial frequency is already the maximum possible.
    """
    # nums = [5, 5, 5], k = 5. Initial frequency is 3.
    # Any operation on a subarray [i..j] would involve adding a non-zero x,
    # changing the 5s to something else and reducing the frequency.
    # Adding x=0 changes nothing. The maximum frequency is 3.
    assert maxFrequency(nums=[5, 5, 5], k=5) == 3


def test_convert_isolated_element():
    """
    Tests converting a single element to join an existing group of k.

    # Convert nums[0]=1 to 5 by choosing subarray [0..0] and adding x=4.
    # The array becomes [5, 5, 5], for a final frequency of 3.
    assert maxFrequency(nums=[1, 5, 5], k=5) == 3


def test_convert_entire_array():
    """
    Tests converting all elements in the array to k when they are all the same.
    """

    # Choose subarray [0..3] (the whole array) and add x=9.
    # The array becomes [10, 10, 10, 10], for a final frequency of 4.
    assert maxFrequency(nums=[1, 1, 1, 1], k=10) == 4

    """

    # We can convert any single element to 10.
    # e.g., choose subarray [0..0] and add x=9 to make nums[0] into 10.
    # The maximum frequency is 1.
    assert maxFrequency(nums=[1, 2, 3], k=10) == 1


def test_interspersed_values():
    """
    Tests converting values that are interspersed with k.
    """
    # nums = [2, 1, 2, 1, 2], k = 2. Initial frequency of 2 is 3.
    # We can convert one of the 1s to a 2.
    # e.g., choose subarray [1..1] and add x=1. Array becomes [2, 2, 2, 1, 2].
    # Frequency of 2 becomes 4. This is a gain of 1 without sacrificing existing 2s.
    assert maxFrequency(nums=[2, 1, 2, 1, 2], k=2) == 4


def test_single_element_is_k():
    """
    Tests an array with a single element which is already k.
    """
    # nums = [50], k = 50. Initial and maximum frequency is 1.
    assert maxFrequency(nums=[50], k=50) == 1


def test_single_element_not_k():
    """
    Tests an array with a single element that can be converted to k.
    """
    # nums = [1], k = 50. Initial frequency is 0.

    # Maximum frequency is 1.
    assert maxFrequency(nums=[1], k=50) == 1


def test_sacrifice_k_for_net_gain():
    """
    Tests a scenario where it's optimal to convert a subarray
    containing k, because the number of other elements converted is greater.
    """
    # nums = [1, 1, 5, 1, 1], k = 5. Initial frequency of 5 is 1.
    # To maximize, convert all 1s to 5s. This requires choosing x=4.
    # The only way to convert all four 1s is to select the whole array [0..4].
    # This operation converts four 1s to 5s, but also the one 5 to a 9.
    # Net change in frequency of 5: +4 (from 1s) - 1 (from original 5) = +3.
    # Final frequency = 1 (initial) + 3 (net gain) = 4.
    assert maxFrequency(nums=[1, 1, 5, 1, 1], k=5) == 4


def test_choose_best_candidate_value():
    """
    Tests choosing the best value to convert when multiple options exist.
    """
    # nums = [1, 2, 2, 3, 3, 3], k = 10. Initial frequency is 0.
    # Options for conversion to 10:
    # - Convert 1s (count 1): max freq = 1.
    # - Convert 2s (count 2): max freq = 2 by operating on subarray [1..2].
    # - Convert 3s (count 3): max freq = 3 by operating on subarray [3..5].
    # The best option is to convert the three 3s.
    assert maxFrequency(nums=[1, 2, 2, 3, 3, 3], k=10) == 3
