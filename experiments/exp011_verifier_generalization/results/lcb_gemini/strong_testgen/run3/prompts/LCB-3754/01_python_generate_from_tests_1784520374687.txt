def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_dist = 0
    for i in range(len(s)):
        if s[i] == 'N':
            y += 1
        elif s[i] == 'S':
            y -= 1
        elif s[i] == 'E':
            x += 1
        elif s[i] == 'W':
            x -= 1
        L = i + 1
        c = min(k, L)
        # Calculate maximum distance for four target diagonals
        max_dist = max(max_dist, abs(x) + abs(y))  # original distance
        max_dist = max(max_dist, (x + y) + 2 * c)  # maximized x+y
        max_dist = max(max_dist, (x - y) + 2 * c)  # maximized x-y
        max_dist = max(max_dist, (-x + y) + 2 * c)  # maximized -x+y
        max_dist = max(max_dist, (-x - y) + 2 * c)  # maximized -x-y
    return max_dist
