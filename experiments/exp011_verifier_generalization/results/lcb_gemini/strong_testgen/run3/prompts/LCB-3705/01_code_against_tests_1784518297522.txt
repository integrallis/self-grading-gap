```python
from typing import List

def largestInteger(nums: List[int], k: int) -> int:
    count = {}
    n = len(nums)

    # Count the occurrences of each number in each subarray of size k
    for i in range(n - k + 1):
        subarray = nums[i:i + k]
        unique_numbers = set(subarray)
        for num in unique_numbers:
            if num in count:
                count[num] += 1
            else:
                count[num] = 1

    # Find the largest integer that appears exactly once
    largest_missing = -1
    for num, occurrences in count.items():
        if occurrences == 1:
            largest_missing = max(largest_missing, num)

    return largest_missing
```
