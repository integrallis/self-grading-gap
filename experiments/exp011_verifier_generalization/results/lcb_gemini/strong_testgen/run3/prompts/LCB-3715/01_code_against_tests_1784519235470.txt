from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a dictionary to represent the coins in bags
    coin_count = {}

    # Fill in the coin counts based on the segments provided
    for l, r, c in coins:
        for i in range(l, r + 1):
            coin_count[i] = coin_count.get(i, 0) + c

    max_coins = 0
    current_coins = 0

    # We will use an ordered list to allow for efficient sliding window calculation
    bags = sorted(coin_count.items())

    # Create a sliding window to calculate the sum of k consecutive bags
    left = 0
    for right in range(len(bags)):
        # Add coins from the right end of the window
        current_coins += bags[right][1]

        # If we exceed the window size, move the left end
        while bags[right][0] - bags[left][0] + 1 > k:
            current_coins -= bags[left][1]
            left += 1

        # Check if the current window is valid and update max_coins
        if bags[right][0] - bags[left][0] + 1 <= k:
            max_coins = max(max_coins, current_coins)

    return max_coins
