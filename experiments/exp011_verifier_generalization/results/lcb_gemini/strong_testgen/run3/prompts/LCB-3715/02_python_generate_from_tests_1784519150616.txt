from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a list to represent the coins in bags based on the maximum right position
    max_right = max(r for _, r, _ in coins)
    coin_count = [0] * (max_right + 1)

    # Fill in the coin counts based on the segments provided
    for l, r, c in coins:
        for i in range(l, r + 1):
            coin_count[i] += c

    # Now we can find the maximum coins that can be collected in k consecutive bags
    max_coins = 0
    current_coins = sum(coin_count[1:k + 1])  # Initial sum for the first k bags
    max_coins = current_coins

    # Slide the window of size k across the range
    for i in range(1, max_right - k + 2):
        current_coins = current_coins - coin_count[i] + coin_count[i + k - 1]
        max_coins = max(max_coins, current_coins)

    return max_coins
```
