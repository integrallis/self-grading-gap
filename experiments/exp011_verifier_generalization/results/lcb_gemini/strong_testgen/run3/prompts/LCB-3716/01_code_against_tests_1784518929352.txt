```python
from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    # dp[i] will hold the longest subsequence length with last two differences
    dp = [[1] * 301 for _ in range(n)]  # We have maximum nums[i] <= 300
    max_length = 1

    for i in range(n):
        for j in range(i):
            abs_diff = abs(nums[i] - nums[j])  # Current absolute difference
            for k in range(j):
                if abs_diff <= abs(nums[j] - nums[k]):  # Maintain non-increasing order of differences
                    dp[i][abs_diff] = max(dp[i][abs_diff], dp[j][abs(nums[j] - nums[k])] + 1)
                    max_length = max(max_length, dp[i][abs_diff])

    return max_length
```
