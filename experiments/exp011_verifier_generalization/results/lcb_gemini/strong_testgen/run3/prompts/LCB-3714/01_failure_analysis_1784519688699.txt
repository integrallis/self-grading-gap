from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    nums.sort()
    max_sum = 0
    min_sum = 0

    # Calculate contribution of each element as max
    for i in range(n):
        max_contrib = pow(2, min(i, k - 1), MOD) * pow(2, n - 1 - i, MOD) % MOD
        max_sum = (max_sum + nums[i] * max_contrib) % MOD

    # Calculate contribution of each element as min
    for i in range(n):
        min_contrib = pow(2, min(n - 1 - i, k - 1), MOD) * pow(2, i, MOD) % MOD
        min_sum = (min_sum + nums[i] * min_contrib) % MOD

    return (max_sum + min_sum) % MOD
