import pytest
from solution import calculateScore


def test_canonical_example_1():
    # spec: input s = "aczzx" -> output 5
    # i=0, 'a': skip
    # i=1, 'c': skip
    # i=2, 'z': mirror 'a'. Closest unmarked is j=0. Mark 0, 2. score = 2 - 0 = 2.
    # i=3, 'z': mirror 'a'. No unmarked 'a' before. Skip.
    # i=4, 'x': mirror 'c'. Closest unmarked is j=1. Mark 1, 4. score = 2 + (4 - 1) = 5.
    assert calculateScore("aczzx") == 5


def test_canonical_example_2():
    # spec: input s = "abcdef" -> output 0
    # No character has its mirror present in the string before it.
    # The score remains 0 throughout the process.
    assert calculateScore("abcdef") == 0


def test_single_character():
    # A single character string has no preceding characters to match with.
    # i=0: no j < i.
    # score should be 0.
    assert calculateScore("a") == 0


def test_no_mirror_pairs():
    # The string contains characters, but none are mirrors of each other.
    # 'a' mirror is 'z', 'b' is 'y', 'c' is 'x'. None are present.
    assert calculateScore("abcabc") == 0


def test_simple_pair_at_start():
    # A simple string with one mirror pair.
    # s = "az"
    # i=0, 'a': skip
    # i=1, 'z': mirror 'a'. Closest unmarked j is 0. Score = 1 - 0 = 1.
    assert calculateScore("az") == 1


def test_multiple_disjoint_pairs():
    # Multiple pairs that do not overlap or interleave.
    # s = "azby"
    # i=1, 'z': mirror 'a' at j=0. score = 1 - 0 = 1. Mark 0, 1.
    # i=3, 'y': mirror 'b' at j=2. score = 1 + (3 - 2) = 2. Mark 2, 3.
    # Oh wait, my math was off. It's 1 + 1 = 2.
    # Let me re-calculate "azby":
    # i=0 'a': skip
    # i=1 'z': mirror 'a' at j=0. score += 1-0=1. mark 0,1. score=1.
    # i=2 'b': skip
    # i=3 'y': mirror 'b' at j=2. score += 3-2=1. mark 2,3. score=2.
    # It should be 2. Let me re-verify my previous test case "azby".
    # azby:
    #   i=1, s[1]='z', mirror='a'. Unmarked j<1 is j=0. s[0]='a'. Match. Mark 0,1. score=1-0=1.
    #   i=3, s[3]='y', mirror='b'. Unmarked j<3 are j=2. s[2]='b'. Match. Mark 2,3. score+=3-2=1.
    #   Total score = 1+1=2.
    # The prompt for "azby" was previously thought to be 3. The correct value is 2.
    # The prompt for "abzy" was 4.
    #   i=2, s[2]='z', mirror='a'. Unmarked j<2 are 0,1. s[0]='a'. j=0. Match. Mark 0,2. score=2-0=2.
    #   i=3, s[3]='y', mirror='b'. Unmarked j<3 is 1. s[1]='b'. Match. Mark 1,3. score+=3-1=2.
    #   Total score = 2+2=4.
    # Let's use abzy, it is more interesting.
    # s = "abzy"
    # i=2, 'z': mirror 'a' at j=0. score = 2 - 0 = 2. Mark 0, 2.
    # i=3, 'y': mirror 'b' at j=1. score = 2 + (3 - 1) = 4. Mark 1, 3.
    assert calculateScore("abzy") == 4


def test_chooses_closest_of_multiple_candidates():
    # When multiple unmarked candidates exist, the one with the largest index j is chosen.
    # s = "aaz"
    # i=2, 'z': mirror 'a'. Unmarked candidates at j=0 and j=1.
    # Closest is j=1. Score = 2 - 1 = 1.
    assert calculateScore("aaz") == 1


def test_ignores_already_marked_characters():
    # A character that has been part of a previous match cannot be used again.
    # s = "zaza"
    # i=1, 'a': mirror 'z' at j=0. Score = 1 - 0 = 1. Mark 0, 1.
    # i=2, 'z': mirror 'a'. Candidate at j=1 is marked. No other candidate. Skip.
    # i=3, 'a': mirror 'z'. Unmarked candidate at j=2. Score = 1 + (3 - 2) = 2. Mark 2, 3.
    assert calculateScore("zaza") == 2


def test_interleaved_pairs_complex():
    # A more complex case with interleaved pairs.
    # s = "abzzy"
    # i=2, 'z': mirror 'a' at j=0. score = 2 - 0 = 2. Mark 0, 2.
    # i=3, 'z': mirror 'a'. No unmarked 'a' before it.
    # i=4, 'y': mirror 'b' at j=1 (unmarked). score = 2 + (4 - 1) = 5. Mark 1, 4.
    assert calculateScore("abzzy") == 5


def test_all_same_character():
    # A string of all same characters will have no mirror pairs unless a character is its own mirror
    # (which is not possible for the English alphabet).
    # 'a' mirror is 'z', which is not in the string.
    assert calculateScore("aaaaa") == 0


def test_long_string_with_no_matches():
    # A longer string that should result in a score of 0.
    # This checks for correctness, not performance, but on a non-trivial length.
    s = "abcdefghijklmnopqrstuvwxyz"
    # No character's mirror appears before it in the alphabet.
    assert calculateScore(s) == 0


def test_palindrome_of_mirrors():
    # A palindromic string made of mirror pairs.
    # s = "mnnm"
    # i=1, 'n': mirror 'm' at j=0. score = 1 - 0 = 1. Mark 0, 1.
    # i=2, 'n': mirror 'm'. No unmarked 'm's before.
    # i=3, 'm': mirror 'n' at j=2. score = 1 + (3 - 2) = 2. Mark 2, 3.
    assert calculateScore("mnnm") == 2
