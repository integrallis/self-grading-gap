## Implementation

```python
def hasSpecialSubstring(s: str, k: int) -> bool:
    n = len(s)
    for i in range(n - k + 1):
        substring = s[i:i + k]
        if len(set(substring)) == 1:
            left_diff = (i == 0 or s[i - 1] != substring[0])
            right_diff = (i + k == n or s[i + k] != substring[0])
            if left_diff and right_diff:
                return True
    return False

```
