from typing import List
import pytest

from solution import sumOfGoodNumbers


def test_canonical_example_1():
    """
    Tests the first example from the specification.
    """
    # spec: sumOfGoodNumbers([1,3,2,1,5,4], 2) -> 12
    # Explanation:
    # nums[1] = 3: i-k=-1 (OOB), i+k=3. nums[1] > nums[3] (3 > 1) -> Good
    # nums[4] = 5: i-k=2, i+k=6 (OOB). nums[4] > nums[2] (5 > 2) -> Good
    # nums[5] = 4: i-k=3, i+k=7 (OOB). nums[5] > nums[3] (4 > 1) -> Good
    # Other numbers are not good:
    # nums[0] = 1: vs nums[2]=2. 1 > 2 is false.
    # nums[2] = 2: vs nums[0]=1, nums[4]=5. 2 > 5 is false.
    # nums[3] = 1: vs nums[1]=3, nums[5]=4. 1 > 3 is false.
    # Sum of good numbers: 3 + 5 + 4 = 12
    assert sumOfGoodNumbers([1, 3, 2, 1, 5, 4], 2) == 12


def test_canonical_example_2():
    """
    Tests the second example from the specification.
    """
    # spec: sumOfGoodNumbers([2,1], 1) -> 2
    # Explanation:
    # nums[0] = 2: i-k=-1 (OOB), i+k=1. nums[0] > nums[1] (2 > 1) -> Good
    # nums[1] = 1: i-k=0, i+k=2 (OOB). nums[1] > nums[0] (1 > 2) is false.
    # Sum of good numbers: 2
    assert sumOfGoodNumbers([2, 1], 1) == 2


def test_no_good_numbers():
    """
    Tests a case where no numbers are good due to strict inequality.
    """
    # spec: input nums = [10, 10, 10, 10], k = 1
    # For any i, nums[i] is not strictly greater than its neighbors.
    # For example, nums[0] > nums[1] (10 > 10) is false.
    # Expected output: 0
    assert sumOfGoodNumbers([10, 10, 10, 10], 1) == 0


def test_middle_element_is_good():
    """
    Tests a case where a middle element is good, requiring checks on both sides.
    """
    # spec: input nums = [1, 1, 10, 1, 1], k = 2
    # nums[2] = 10: i-k=0, i+k=4. nums[2] > nums[0] (10 > 1) and nums[2] > nums[4] (10 > 1). Both true -> Good.
    # All other elements are 1 and cannot be greater than their neighbors.
    # Expected output: 10
    assert sumOfGoodNumbers([1, 1, 10, 1, 1], 2) == 10


def test_max_k_value():
    """
    Tests a case with the maximum allowed value for k (floor(nums.length / 2)).
    """
    # spec: input nums = [10, 20, 30, 5, 15, 25], k = 3 (since floor(6/2)=3)
    # nums[0]=10: vs nums[3]=5 (10 > 5) -> Good
    # nums[1]=20: vs nums[4]=15 (20 > 15) -> Good
    # nums[2]=30: vs nums[5]=25 (30 > 25) -> Good
    # nums[3]=5: vs nums[0]=10 (5 > 10) is false
    # nums[4]=15: vs nums[1]=20 (15 > 20) is false
    # nums[5]=25: vs nums[2]=30 (25 > 30) is false
    # Expected output: 10 + 20 + 30 = 60
    assert sumOfGoodNumbers([10, 20, 30, 5, 15, 25], 3) == 60


def test_mixed_good_numbers():
    """
    Tests a varied list where good numbers appear at the beginning, middle, and end.
    """
    # spec: input nums = [10, 2, 3, 4, 11, 6, 7, 8, 12], k = 1
    # nums[0]=10: vs nums[1]=2 (10 > 2) -> Good
    # nums[4]=11: vs nums[3]=4 (11 > 4) and nums[5]=6 (11 > 6) -> Good
    # nums[8]=12: vs nums[7]=8 (12 > 8) -> Good
    # All other numbers are not strictly greater than at least one of their neighbors.
    # Expected output: 10 + 11 + 12 = 33
    assert sumOfGoodNumbers([10, 2, 3, 4, 11, 6, 7, 8, 12], 1) == 33


def test_ascending_array():
    """

    # nums[2]=3: vs nums[1]=2 (ok), but vs nums[3]=4 (3 > 4) is false
    # nums[3]=4: vs nums[2]=3 (ok), but vs nums[4]=5 (4 > 5) is false
    # nums[4]=5: vs nums[3]=4 (5 > 4) -> Good
    # Expected output: 5
    assert sumOfGoodNumbers([1, 2, 3, 4, 5], 1) == 5
