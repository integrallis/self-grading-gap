```python
from typing import List


def sumOfGoodNumbers(nums: List[int], k: int) -> int:
    good_sum = 0
    n = len(nums)
    for i in range(n):
        is_good = True
        if i - k >= 0:
            is_good = is_good and (nums[i] > nums[i - k])
        if i + k < n:
            is_good = is_good and (nums[i] > nums[i + k])
        if is_good:
            good_sum += nums[i]
    return good_sum

```
