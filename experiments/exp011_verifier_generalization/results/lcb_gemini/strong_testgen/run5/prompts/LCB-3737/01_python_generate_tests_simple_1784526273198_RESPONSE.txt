def test_canonical_example_1():
    """
    Test with the first example from the specification.
    n = 4, cost = [[3,5,7],[6,2,9],[4,8,1],[7,3,5]] -> 9
    The optimal painting sequence is [color 1, color 2, color 3, color 2] with
    corresponding costs [3, 2, 1, 3].
    Total cost: 3 (house 0, color 1) + 2 (house 1, color 2) + 1 (house 2, color 3) + 3 (house 3, color 2) = 9
    - Adjacency: color(0)!=color(1), color(1)!=color(2), color(2)!=color(3). OK.
    - Symmetry:
      - (0,3): color 1 != color 2. OK.
      - (1,2): color 2 != color 3. OK.

def test_min_n_is_2():
    """

    # (color 2, color 1): 10 + 2 = 12

    # (color 3, color 1): 100 + 2 = 102

def test_uniform_costs():
    """
    Test where all painting costs are the same.
    The total cost will be n * cost_per_house if a valid coloring exists.
    """
    n = 6
    cost_per_paint = 10
    cost = [[cost_per_paint] * 3 for _ in range(n)]
    # A valid coloring exists. For example, [color 1, color 2, color 1, color 2, color 1, color 2].
    # - Adjacency: All OK.
    # - Symmetry: (0,5) -> 1!=2; (1,4) -> 2!=1; (2,3) -> 1!=2. All OK.
    # Since any paint choice costs 10, the total minimum cost is 6 * 10 = 60.
    expected = 60
    assert minCost(n, cost) == expected

def test_biased_costs_forcing_non_greedy_choice():
    """
    Test where one color is significantly cheaper, but constraints prevent its overuse.
    """
    n = 4
    cost = [[1, 100, 100], [1, 100, 100], [1, 100, 100], [1, 100, 100]]
    # To minimize cost, we want to use color 1 (cost 1) as much as possible.
    # A simple greedy approach for each house would pick color 1, but this violates constraints.
    # Let's analyze a valid coloring: [color 1, color 2, color 1, color 2]
    # house 0: color 1 -> cost 1
    # house 1: color 2 -> cost 100
    # house 2: color 1 -> cost 1
    # house 3: color 2 -> cost 100
    # Total cost: 1 + 100 + 1 + 100 = 202
    # Constraints check for [1,2,1,2]:
    # - Adjacency: 1!=2, 2!=1, 1!=2. OK.
    # - Symmetry: (0,3) -> 1!=2. (1,2) -> 2!=1. OK.
    # This coloring is valid and appears optimal, as it requires two cheap and two expensive paints.
    expected = 202
    assert minCost(n, cost) == expected
