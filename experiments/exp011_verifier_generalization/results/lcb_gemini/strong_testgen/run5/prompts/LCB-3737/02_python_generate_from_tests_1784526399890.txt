from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # Dynamic programming table to store the minimum costs
    dp = [[0] * 3 for _ in range(n)]

    # Base case: first house costs directly from the cost array
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the dp table for each house
    for i in range(1, n):
        for j in range(3):
            # Ensure no adjacent houses have the same color
            dp[i][j] = min(
                dp[i - 1][(j + 1) % 3],
                dp[i - 1][(j + 2) % 3]
            ) + cost[i][j]

    # Now handle the symmetry constraint
    min_cost = float('inf')

    # Check for all possible colorings for the first half of the houses
    for first_color in range(3):
        for second_color in range(3):
            if first_color != second_color:
                # Calculate the cost if the first house is of 'first_color'
                cost_with_first_color = dp[n // 2 - 1][first_color]
                # Calculate the cost for the second half based on symmetry
                cost_with_second_color = dp[n - 1][second_color]
                total_cost = cost_with_first_color + cost_with_second_color
                min_cost = min(min_cost, total_cost)

    return min_cost

```
