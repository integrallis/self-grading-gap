import pytest
from typing import List

# The function to be tested is assumed to be in a file named `solution.py`.
# As per TDD (RED phase), this test suite is written before the implementation.
from solution import maxFrequency

# Helper explanation for test comments:
#
# The goal is to maximize the frequency of `k`. We can perform one operation:
# pick a subarray `nums[i..j]` and add an integer `x` to all its elements.
# To make some elements `v` inside the subarray become `k`, we must add `x = k - v`.
# This operation changes all occurrences of `v` in the subarray to `k`, but also changes
# all occurrences of `k` in the subarray to `k + (k - v)`, which is not `k` (unless v=k).
#
# The final frequency of `k` will be:
#   (initial frequency of k)
# - (number of k's in the chosen subarray `nums[i..j]`)
# + (number of v's in the chosen subarray `nums[i..j]`)
#
# This can be restated as:
#   (initial frequency of k) + gain
# where gain = count(v in sub) - count(k in sub).
#
# We need to find the maximum possible gain over all choices of `v`, `i`, and `j`.
# For a fixed `v`, this is equivalent to finding the maximum subarray sum on a
# "gain array" where elements are 1 if `nums[i]==v`, -1 if `nums[i]==k`, and 0 otherwise.
# The overall max frequency is `initial_freq_k + max(0, max_gain_over_all_v)`.


def test_canonical_example_1():
    # spec: Input: nums = [1,2,3,4,5,6], k = 1 -> Output: 2
    # Explanation: After adding -5 to nums[2..5], 1 has a frequency of 2
    # in [1, 2, -2, -1, 0, 1].
    # Initial freq of k=1 is 1 (at index 0).
    # The operation converts v=6 to 1 with x=-5 in subarray nums[2..5].
    # Gain for v=6, subarray [2..5]: count(v=6)=1, count(k=1)=0. Gain=1.
    # Total freq = 1 (initial) + 1 (gain) = 2.
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2


def test_canonical_example_2():
    # spec: Input: nums = [10,2,3,4,5,5,4,3,2,2], k = 10 -> Output: 4
    # Explanation: After adding 8 to nums[1..9], 10 has a frequency of 4 in
    # [10, 10, 11, 12, 13, 13, 12, 11, 10, 10].
    # Initial freq of k=10 is 1 (at index 0).
    # The operation converts v=2 to 10 (x=8) in subarray nums[1..9].
    # In this subarray, there are three 2s and zero 10s.
    # Gain = count(v=2 in sub) - count(k=10 in sub) = 3 - 0 = 3.
    # Total freq = 1 (initial) + 3 (gain) = 4.
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4


def test_single_element_is_k():
    # A single element array that is already k. No operation can improve the frequency.
    # Initial freq of k=10 is 1. Max gain is 0. Total freq = 1.
    assert maxFrequency([10], 10) == 1


def test_single_element_not_k():
    # A single element array. The operation can change it to k.
    # Initial freq of k=10 is 0.
    # Convert v=20 to k=10 with x=-10. Subarray is the whole array.
    # Gain = count(v=20)=1 - count(k=10)=0 = 1.
    # Total freq = 0 + 1 = 1.
    assert maxFrequency([20], 10) == 1


def test_all_elements_are_k():
    # All elements are already k. No other values to convert.
    # Initial freq of k=5 is 3. Max gain is 0.
    # Total freq = 3.
    assert maxFrequency([5, 5, 5], 5) == 3


def test_k_not_in_array():
    # k is not initially present. We must create it.
    # Initial freq of k=4 is 0.
    # We can convert any single element to 4. e.g., v=1.
    # For v=1, max gain is 1 (from subarray [1]).
    # For v=2, max gain is 1 (from subarray [2]).
    # For v=3, max gain is 1 (from subarray [3]).
    # Overall max gain is 1. Total freq = 0 + 1 = 1.
    assert maxFrequency([1, 2, 3], 4) == 1


def test_best_gain_from_single_element_surrounded_by_k():
    # The best strategy is to convert a single `v` that is surrounded by `k`s.
    # nums = [1, 5, 1, 5, 1], k = 1. Initial freq of k=1 is 3.
    # Consider v=5. The gain array is [-1, 1, -1, 1, -1].
    # Max subarray sum is 1 (from subarray [1] at index 1 or 3).
    # Max gain is 1. Total freq = 3 + 1 = 4.
    # e.g., change nums[1]=5 to 1. New array [1,1,1,5,1], freq=4.
    assert maxFrequency([1, 5, 1, 5, 1], 1) == 4


def test_best_gain_from_consecutive_block_of_v():
    # A block of convertible elements `v` gives a large gain.
    # nums = [1, 5, 5, 5, 1], k = 1. Initial freq of k=1 is 2.
    # Consider v=5. Gain array is [-1, 1, 1, 1, -1].
    # Max subarray sum is 3 (from [1, 1, 1]).
    # Max gain is 3. Total freq = 2 + 3 = 5.
    # Change subarray [5,5,5] to [1,1,1]. New array [1,1,1,1,1], freq=5.
    assert maxFrequency([1, 5, 5, 5, 1], 1) == 5


def test_best_gain_from_sparse_elements_in_subarray():
    # The best subarray may contain other elements besides v and k.
    # nums = [10, 2, 0, 2, 0, 2], k = 10. Initial freq of k=10 is 1.
    # Consider v=2. Gain array is [-1, 1, 0, 1, 0, 1].
    # Max subarray sum on this is 3 (for subarray [1, 0, 1, 0, 1]).
    # Max gain is 3. Total freq = 1 + 3 = 4.
    # Change nums[1..5]=[2,0,2,0,2] with x=8 (to convert 2s to 10s).
    # New array: [10, 10, 8, 10, 8, 10]. Freq of 10 is 4.
    assert maxFrequency([10, 2, 0, 2, 0, 2], 10) == 4


def test_multiple_candidate_values():
    # There are multiple values that can be converted to k, need to find the best one.
    # nums = [1, 2, 2, 3, 3, 3], k = 4. Initial freq is 0.
    # v=1: max gain = 1.
    # v=2: max gain = 2.
    # v=3: max gain = 3.
    # Overall max gain is 3. Total freq = 0 + 3 = 3.
    assert maxFrequency([1, 2, 2, 3, 3, 3], 4) == 3

def test_k_and_v_at_ends_of_array():
    # Check if the algorithm correctly handles subarrays starting at 0 or ending at n-1.
    # nums = [2, 1, 1, 1, 2], k = 1. Initial freq=3.
    # v=2. gain array = [1, -1, -1, -1, 1].
