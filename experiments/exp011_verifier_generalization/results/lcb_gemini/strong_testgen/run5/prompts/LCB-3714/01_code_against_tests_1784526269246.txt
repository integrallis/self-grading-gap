```python
from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    nums.sort()
    total_min = 0
    total_max = 0
    
    for i in range(n):
        # Count how many times `nums[i]` is the minimum of subsequences
        count_min = comb(i, 0)  # subsequences of length 0
        for j in range(1, min(k, i + 1)):
            count_min += comb(i, j)
        total_min += nums[i] * count_min
        
        # Count how many times `nums[i]` is the maximum of subsequences
        count_max = comb(n - 1 - i, 0)  # subsequences of length 0
        for j in range(1, min(k, n - i)):
            count_max += comb(n - 1 - i, j)
        total_max += nums[i] * count_max

    return (total_min + total_max) % MOD


def comb(n: int, k: int) -> int:
    if k < 0 or k > n:
        return 0
    if k == 0 or k == n:
        return 1
    k = min(k, n - k)  # Take advantage of symmetry
    c = 1
    for i in range(k):
        c = c * (n - i) // (i + 1)
    return c

```
