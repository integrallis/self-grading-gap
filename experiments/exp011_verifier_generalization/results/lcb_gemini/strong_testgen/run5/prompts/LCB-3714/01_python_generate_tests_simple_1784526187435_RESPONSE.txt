import pytest
from typing import List
from solution import minMaxSums


def test_canonical_example_1():
    """
    Tests the first example from the specification.
    spec: nums = [1,2,3], k = 2 -> 24
    """
    # Subsequences with length <= 2:
    # [1]: min=1, max=1, min+max=2
    # [2]: min=2, max=2, min+max=4
    # [3]: min=3, max=3, min+max=6
    # [1, 2]: min=1, max=2, min+max=3
    # [1, 3]: min=1, max=3, min+max=4
    # [2, 3]: min=2, max=3, min+max=5
    # Total sum = 2 + 4 + 6 + 3 + 4 + 5 = 24
    assert minMaxSums([1, 2, 3], 2) == 24


def test_canonical_example_2():
    """
    Tests the second example from the specification.
    spec: nums = [5,0,6], k = 1 -> 22
    """
    # Subsequences with length <= 1 (i.e., length 1):
    # [5]: min=5, max=5, min+max=10
    # [0]: min=0, max=0, min+max=0
    # [6]: min=6, max=6, min+max=12
    # Total sum = 10 + 0 + 12 = 22
    assert minMaxSums([5, 0, 6], 1) == 22


def test_canonical_example_3():
    """
    Tests the third example from the specification with duplicate elements.
    spec: nums = [1,1,1], k = 2 -> 12
    """
    # Let nums be [1a, 1b, 1c]
    # Subsequences of length 1 (3 of them): [1a], [1b], [1c]
    # Each has min=1, max=1, sum=2. Total: 3 * 2 = 6
    # Subsequences of length 2 (3 of them): [1a,1b], [1a,1c], [1b,1c]
    # Each has min=1, max=1, sum=2. Total: 3 * 2 = 6
    # Grand total = 6 + 6 = 12
    assert minMaxSums([1, 1, 1], 2) == 12

    # nums = [3, 1, 2], k=2 should be the same as nums = [1, 2, 3], k=2.
    # Expected result is 24, as per canonical example 1.
    assert minMaxSums([3, 1, 2], 2) == 24

def test_k_equals_n():
    """

    # [1]: min=1, max=1, min+max=2
    # [5]: min=5, max=5, min+max=10
    # [1, 5]: min=1, max=5, min+max=6
    # Total sum = 2 + 10 + 6 = 18
    assert minMaxSums([1, 5], 2) == 18

    # nums = [10, 20], k = 10. Effective k is 2.
    # This is equivalent to nums=[10, 20], k=2
    # [10]: min=10, max=10, min+max=20
    # [20]: min=20, max=20, min+max=40
    # [10, 20]: min=10, max=20, min+max=30
    # Total sum = 20 + 40 + 30 = 90
    assert minMaxSums([10, 20], 10) == 90


def test_large_numbers_and_modulo():
    """
    Tests if the modulo arithmetic is handled correctly for large numbers.
    """
    MOD = 10**9 + 7
    nums = [MOD - 2, MOD - 1]
    k = 2
    # Subsequences and their min+max sums:
    # [MOD - 2]: 2 * (MOD - 2)
    # [MOD - 1]: 2 * (MOD - 1)
    # [MOD - 2, MOD - 1]: (MOD - 2) + (MOD - 1) = 2*MOD - 3
    # Total sum = (2*MOD-4) + (2*MOD-2) + (2*MOD-3) = 6*MOD - 9
    # (6*MOD - 9) % MOD = -9 % MOD = MOD - 9
    expected = MOD - 9
    assert minMaxSums(nums, k) == expected


def test_k_gt_2():
    """
    Tests a case with k greater than 2 to exercise combination calculations.
    """
    # nums = [1,2,3,4], k = 3
    # Subsequences:
    # len 1: [1],[2],[3],[4] -> sums: 2,4,6,8. Total: 20
    # len 2: [1,2],[1,3],[1,4],[2,3],[2,4],[3,4] -> sums: 3,4,5,5,6,7. Total: 30
    # len 3: [1,2,3],[1,2,4],[1,3,4],[2,3,4] -> sums: 4,5,5,7. Total: 21
    # Grand total = 20 + 30 + 21 = 71
    # Note: Mistake in manual calculation above.
    # Correct calculation using combinatorial method:
    # Sorted nums=[1,2,3,4], n=4, k=3
    # Sum of Mins:
    # 1 * (C(3,0)+C(3,1)+C(3,2)) = 1 * (1+3+3) = 7
    # 2 * (C(2,0)+C(2,1)+C(2,2)) = 2 * (1+2+1) = 8
    # 3 * (C(1,0)+C(1,1))       = 3 * (1+1)   = 6
    # 4 * (C(0,0))             = 4 * 1       = 4
    # Total Mins = 7+8+6+4 = 25
    # Sum of Maxs:
    # 1 * (C(0,0))             = 1 * 1       = 1
    # 2 * (C(1,0)+C(1,1))       = 2 * (1+1)   = 4
    # 3 * (C(2,0)+C(2,1)+C(2,2)) = 3 * (1+2+1) = 12
    # 4 * (C(3,0)+C(3,1)+C(3,2)) = 4 * (1+3+3) = 28
    # Total Maxs = 1+4+12+28 = 45
    # Total Sum = 25 + 45 = 70
    assert minMaxSums([1, 2, 3, 4], 3) == 70
