```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a list to hold the number of coins at each position
    # We will use a dictionary to handle large coordinate values efficiently
    coin_map = {}

    # Populate the coin_map with the coin counts for each range
    for l, r, c in coins:
        for position in range(l, r + 1):
            coin_map[position] = coin_map.get(position, 0) + c

    # Create a sorted list of unique positions
    positions = sorted(coin_map.keys())
    n = len(positions)

    # Calculate the prefix sums for quick range sum calculation
    prefix_sum = [0] * (n + 1)
    for i in range(n):
        prefix_sum[i + 1] = prefix_sum[i] + coin_map[positions[i]]

    max_coins = 0

    # Iterate through each position as the start of the window
    for i in range(n):
        start_position = positions[i]
        end_position = start_position + k - 1

        # Find the last position within the range [start_position, end_position]
        j = i
        while j < n and positions[j] <= end_position:
            j += 1

        # Calculate the sum of coins in the window from i to j-1
        total_coins = prefix_sum[j] - prefix_sum[i]
        max_coins = max(max_coins, total_coins)

    return max_coins

```
