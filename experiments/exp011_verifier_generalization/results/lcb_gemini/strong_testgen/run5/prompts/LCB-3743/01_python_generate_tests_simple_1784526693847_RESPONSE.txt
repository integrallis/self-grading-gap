import pytest
from typing import List
from solution import maxFreeTime

# Helper function to calculate expected result based on the sliding window on gaps model
def _calculate_expected(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    if n == 0:
        return eventTime
        
    gaps = []
    # Gap at the beginning of the event
    gaps.append(startTime[0])
    # Gaps between meetings
    for i in range(1, n):
        gaps.append(startTime[i] - endTime[i-1])
    # Gap at the end of the event
    gaps.append(eventTime - endTime[n-1])

    window_size = k + 1
    if window_size > len(gaps):
        window_size = len(gaps)

    current_sum = sum(gaps[0:window_size])
    max_sum = current_sum

    for i in range(window_size, len(gaps)):
        current_sum += gaps[i] - gaps[i - window_size]
        if current_sum > max_sum:
            max_sum = current_sum
            
    return max_sum


def test_canonical_example_1():
    """
    Tests the first example from the specification.
    spec: eventTime = 5, k = 1, startTime = [1,3], endTime = [2,5] -> 2
    """
    eventTime = 5
    k = 1
    startTime = [1, 3]
    endTime = [2, 5]
    
    # Gaps:
    # Before first meeting: startTime[0] - 0 = 1
    # Between meetings: startTime[1] - endTime[0] = 3 - 2 = 1
    # After last meeting: eventTime - endTime[1] = 5 - 5 = 0
    # gaps = [1, 1, 0]
    # With k=1, we can merge k+1=2 gaps.
    # Windows of size 2:
    # [1, 1] -> sum = 2
    # [1, 0] -> sum = 1
    # Max sum is 2.
    expected = 2
    assert maxFreeTime(eventTime, k, startTime, endTime) == expected

def test_no_moves_allowed_k_is_zero():
    """
    Tests the case where k=0, meaning no meetings can be rescheduled.
    The result should be the largest initial free time gap.
    """
    eventTime = 20
    k = 0
    startTime = [2, 8, 15]
    endTime = [5, 10, 18]

    # Gaps:
    # Before first: 2
    # Between 0 and 1: 8 - 5 = 3
    # Between 1 and 2: 15 - 10 = 5
    # After last: 20 - 18 = 2
    # gaps = [2, 3, 5, 2]
    # With k=0, we can "merge" k+1=1 gap. This means taking the max of individual gaps.
    # Max of [2, 3, 5, 2] is 5.
    expected = 5
    assert maxFreeTime(eventTime, k, startTime, endTime) == expected

def test_all_meetings_movable_k_is_n():
    """
    Tests the case where k=n, meaning all meetings can be rescheduled.
    The result should be the total free time available.
    """
    eventTime = 20
    k = 3
    startTime = [2, 8, 15]
    endTime = [5, 10, 18]
    n = len(startTime) # n=3

    # Gaps are [2, 3, 5, 2].
    # With k=n=3, we can merge k+1=4 gaps.
    # There are n+1=4 gaps in total.
    # The result is the sum of all gaps: 2 + 3 + 5 + 2 = 12.
    # This is equivalent to total free time: 20 - (3+2+3) = 20 - 8 = 12.
    expected = 12
    assert maxFreeTime(eventTime, k, startTime, endTime) == expected
