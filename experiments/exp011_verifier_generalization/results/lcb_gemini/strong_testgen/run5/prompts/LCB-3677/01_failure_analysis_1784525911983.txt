from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[-float('inf')] * (n + 1) for _ in range(m + 1)]
    dp[0][0] = 0  # Start at (0, 0) with 0 coins

    for i in range(m):
        for j in range(n):
            current = coins[i][j]

            # Store possible profits for one and two neutralizations
            for neutralizations in range(3):  # 0, 1, or 2 neutralizations
                if dp[i][j] != -float('inf'):
                    # Case 1: Without neutralization
                    dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j] + current)
                    # Case 2: With a neutralization if current is negative
                    if current < 0 and neutralizations > 0:
                        dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j] - current)

            # Move right
            if j < n - 1:
                dp[i][j + 1] = max(dp[i][j + 1], dp[i][j])
            # Move down
            if i < m - 1:
                dp[i + 1][j] = max(dp[i + 1][j], dp[i][j])

    # Return the best result at the bottom-right corner, allowing two neutralizations
    return dp[m][n] if dp[m][n] != -float('inf') else -1
