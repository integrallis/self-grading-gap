from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[-float('inf')] * (n + 1) for _ in range(m + 1)]
    dp[0][0] = 0  # Start at (0, 0) with 0 coins

    for i in range(m):
        for j in range(n):
            current = coins[i][j]
            # Update without neutralization
            if dp[i][j] != -float('inf'):
                dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j] + current)

            # Update with one neutralization (if current is negative, negate it to treat the neutralization)
            if current < 0:
                if dp[i][j] != -float('inf'):
                    dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j] - current)

            # Update for the two neutralization scenarios
            if j > 0 and dp[i][j + 1] != -float('inf'):
                dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j + 1] + current)
            if i > 0 and dp[i + 1][j] != -float('inf'):
                dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i + 1][j] + current)

    # Return the best result at the bottom-right corner, allowing two neutralizations
    return dp[m][n] if dp[m][n] != -float('inf') else -1
