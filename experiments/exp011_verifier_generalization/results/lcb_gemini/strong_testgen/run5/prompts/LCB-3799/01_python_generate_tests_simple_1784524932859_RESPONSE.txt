from typing import List
from solution import totalNumbers


def test_canonical_example_1():
    """
    Tests with a list of unique digits containing even numbers.
    """
    # spec: totalNumbers([1, 2, 3, 4]) -> 12
    # Explanation: The 12 distinct 3-digit even numbers are:
    # 124, 132, 134, 142, 214, 234, 312, 314, 324, 342, 412, and 432.
    # For a number to be even, the last digit must be 2 or 4.
    # If last digit is 2, the first two are permutations of 2 from {1, 3, 4}, which is 3*2=6.
    # If last digit is 4, the first two are permutations of 2 from {1, 2, 3}, which is 3*2=6.
    # Total = 6 + 6 = 12.
    assert totalNumbers([1, 2, 3, 4]) == 12


def test_canonical_example_2():
    """
    Tests with duplicate digits and a zero.
    """
    # spec: totalNumbers([0, 2, 2]) -> 2
    # Explanation: The only 3-digit even numbers are 202 and 220.
    # - 202: uses one '2' for the hundreds, '0' for tens, and the other '2' for units.
    # - 220: uses one '2' for hundreds, the other '2' for tens, and '0' for units.
    # Note that 022 is not a 3-digit number.
    assert totalNumbers([0, 2, 2]) == 2


def test_canonical_example_3():
    """
    Tests with all digits being the same and even.
    """
    # spec: totalNumbers([6, 6, 6]) -> 1
    # Explanation: Only the number 666 can be formed.
    # The digits are all the same, so only one distinct number is possible.
    # 666 is a 3-digit even number.
    assert totalNumbers([6, 6, 6]) == 1


def test_canonical_example_4():
    """
    Tests with a list of digits containing no even numbers.

    # spec: totalNumbers([1, 3, 5]) -> 0
    # Explanation: No even 3-digit numbers can be formed.
    # To form an even number, the last digit must be even.

    assert totalNumbers([1, 3, 5]) == 0
