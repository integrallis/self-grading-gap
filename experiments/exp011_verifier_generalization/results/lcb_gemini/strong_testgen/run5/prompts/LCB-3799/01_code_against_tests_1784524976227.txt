## Implementation

```python
from typing import List


def totalNumbers(digits: List[int]) -> int:
    from itertools import permutations
    even_numbers = set()
    # Generate all permutations of the digits taken 3 at a time
    for perm in permutations(digits, 3):
        # Check if the last digit is even and the first is not zero
        if perm[2] % 2 == 0 and perm[0] != 0:
            even_numbers.add(perm)
    return len(even_numbers)
```
