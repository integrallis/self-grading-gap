import pytest
from typing import List

# Import the function to be tested
from solution import minCosts

#
# Canonical examples from the specification
#

def test_canonical_example_1():
    """
    Tests the first example from the specification.
    """
    cost = [5, 3, 4, 1, 3, 2]
    # The minimum costs to reach each position are:
    # i=0: Pay cost[0]=5. Cost = 5.
    # i=1: Pay cost[1]=3. Cost = 3.
    # i=2: Pay cost[1]=3, then free swap to 2. Cost = 3.
    # i=3: Pay cost[3]=1. Cost = 1.
    # i=4: Pay cost[3]=1, then free swap to 4. Cost = 1.
    # i=5: Pay cost[3]=1, then free swap to 5. Cost = 1.
    # This corresponds to a running minimum of the cost array.
    # min(5) = 5
    # min(5, 3) = 3
    # min(5, 3, 4) = 3
    # min(5, 3, 4, 1) = 1
    # min(5, 3, 4, 1, 3) = 1
    # min(5, 3, 4, 1, 3, 2) = 1
    expected = [5, 3, 3, 1, 1, 1]
    assert minCosts(cost) == expected

def test_canonical_example_2():
    """
    Tests the second example from the specification.
    """
    cost = [1, 2, 4, 6, 7]
    # The minimum cost is always achieved by paying cost[0]=1.
    # Then all other positions can be reached for free.
    # min(1) = 1
    # min(1, 2) = 1
    # min(1, 2, 4) = 1
    # min(1, 2, 4, 6) = 1
    # min(1, 2, 4, 6, 7) = 1
    expected = [1, 1, 1, 1, 1]
    assert minCosts(cost) == expected

def test_single_element_cost_list():
    """
    Tests with the minimum possible size of cost array (n=1).
    """
    cost = [42]
    # To reach position 0, the only option is to pay cost[0].
    # min(42) = 42
    expected = [42]
    assert minCosts(cost) == expected

def test_strictly_decreasing_costs():
    """
    Tests a scenario with strictly decreasing costs. The minimum cost available
    changes at each step.
    """
    cost = [30, 20, 10]
    # min(30) = 30
    # min(30, 20) = 20
    # min(30, 20, 10) = 10
    expected = [30, 20, 10]
    assert minCosts(cost) == expected

def test_all_costs_equal():
    """
    Tests a scenario where all costs are the same.
    """
    cost = [7, 7, 7, 7]
    # The minimum cost is always 7.
    # min(7) = 7
    # min(7, 7) = 7
    # min(7, 7, 7) = 7
    # min(7, 7, 7, 7) = 7
    expected = [7, 7, 7, 7]
    assert minCosts(cost) == expected

def test_valley_shaped_costs():
    """
    Tests costs that decrease and then increase.
    """
    cost = [5, 4, 1, 2, 3]
    # The minimum cost is found at index 2 and remains the minimum.
    # min(5) = 5
    # min(5, 4) = 4
    # min(5, 4, 1) = 1
    # min(5, 4, 1, 2) = 1
    # min(5, 4, 1, 2, 3) = 1
    expected = [5, 4, 1, 1, 1]
    assert minCosts(cost) == expected

def test_max_n_and_max_cost_values():
    """
    Tests with the maximum number of elements (n=100) and high cost values.
    """
    cost = list(range(100, 0, -1)) # [100, 99, 98, ..., 1]
    # This is the same pattern as the strictly decreasing test.
    # answer[i] = min(cost[0]...cost[i]) = cost[i]
    expected = list(range(100, 0, -1))
    assert minCosts(cost) == expected

def test_min_cost_values():
    """
    Tests with all costs being the minimum possible value (1).
    """
    cost = [1] * 50 # A list of 50 ones.
    # The minimum cost is always 1.
    expected = [1] * 50
    assert minCosts(cost) == expected
