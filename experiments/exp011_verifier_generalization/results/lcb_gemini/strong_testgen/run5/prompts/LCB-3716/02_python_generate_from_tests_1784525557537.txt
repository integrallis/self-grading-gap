from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n < 2:
        return n
    dp = [1] * n  # dp[i] will hold the length of the longest subsequence ending at index i
    max_len = 1

    for i in range(n):
        for j in range(i):
            diff1 = abs(nums[i] - nums[j])
            if j > 0:
                diff2 = abs(nums[j] - nums[j - 1])
                if diff1 >= diff2:
                    dp[i] = max(dp[i], dp[j] + 1)
            else:
                dp[i] = max(dp[i], 2)  # At least it can form a subsequence of length 2
        max_len = max(max_len, dp[i])

    return max_len
```
