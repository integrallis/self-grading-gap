from typing import List
import heapq


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Create adjacency list and weights map
    graph = {i: [] for i in range(n)}
    for u, v, w in edges:
        graph[u].append((v, w))

    # Binary search on the maximum weight
    def canReach(max_weight: int) -> bool:
        out_degrees = [0] * n
        reachable = [False] * n
        queue = []

        # Build a filtered graph by weight
        for u in range(n):
            for v, w in graph[u]:
                if w <= max_weight:
                    out_degrees[u] += 1
                    if out_degrees[u] > threshold:
                        return False
                    reachable[v] = True
                    queue.append(u)

        # BFS to check if all nodes can reach 0
        visited = [False] * n
        while queue:
            current = queue.pop(0)
            if visited[current]:
                continue
            visited[current] = True
            for v, w in graph[current]:
                if w <= max_weight and not visited[v]:
                    queue.append(v)

        return all(visited[i] for i in range(1, n))

    left, right = 1, 10**6
    best = -1
    while left <= right:
        mid = (left + right) // 2
        if canReach(mid):
            best = mid
            right = mid - 1
        else:
            left = mid + 1

    return best

```
