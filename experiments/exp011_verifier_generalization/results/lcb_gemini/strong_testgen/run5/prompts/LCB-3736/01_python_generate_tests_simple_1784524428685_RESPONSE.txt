def test_no_valid_pair_when_all_adjacent_digits_are_same():
    # s = "333"
    # The count of '3' is 3, which is valid.
    # However, all adjacent pairs are "33", which are invalid as the digits are not different.
    assert findValidPair("333") == ""

def test_no_valid_pair_when_digit_counts_are_incorrect():
    # s = "41414"
    # counts: '4' appears 3 times (value 4), '1' appears 2 times (value 1).
    # Neither digit has a count matching its value. No pair can be valid.
    assert findValidPair("41414") == ""

def test_returns_first_valid_pair_when_multiple_exist():
    # s = "221333"
    # counts: '2': 2, '1': 1, '3': 3. All are valid counts.
    # pair "22": invalid (equal).
    # pair "21": valid. '2' and '1' have correct counts. This is the first valid pair.
    # pair "13": also valid, but not the first.
    assert findValidPair("221333") == "21"

def test_finds_valid_pair_at_end_of_string():
    # s = "44441"
    # counts: '4': 4, '1': 1. Both are valid counts.
    # All pairs "44" are invalid (equal).
    # The last pair "41" is valid.
    assert findValidPair("44441") == "41"

def test_handles_large_digits_correctly():
    # s = "1999999999"
    # counts: '1': 1, '9': 9. Both are valid counts.
    # The first pair "19" is valid.
    # Other pairs "99" are invalid.
    assert findValidPair("1999999999") == "19"
