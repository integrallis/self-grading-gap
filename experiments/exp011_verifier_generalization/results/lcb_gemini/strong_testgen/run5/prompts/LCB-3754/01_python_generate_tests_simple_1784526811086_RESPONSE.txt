import pytest
from solution import maxDistance

#
# Canonical Examples from the Specification
#

def test_canonical_example_1():
    """
    Test with the first example from the specification.
    Input: s = "NWSE", k = 1
    Output: 3
    Explanation: Change s[2] from 'S' to 'N'. The string s becomes "NWNE".
    Path:
    s[0] = 'N': (0, 1), dist=1
    s[1] = 'W': (-1, 1), dist=2
    s[2] = 'N': (-1, 2), dist=3
    s[3] = 'E': (0, 2), dist=2
    The maximum Manhattan distance at any point is 3.
    """
    assert maxDistance("NWSE", 1) == 3

def test_canonical_example_2():
    """
    Test with the second example from the specification.
    Input: s = "NSWWEW", k = 3
    Output: 6
    Explanation: Change s[1] from 'S' to 'N', and s[4] from 'E' to 'W'.
    s becomes "NNWWWW".
    Path:
    'N': (0, 1), dist=1
    'N': (0, 2), dist=2
    'W': (-1, 2), dist=3
    'W': (-2, 2), dist=4
    'W': (-3, 2), dist=5
    'W': (-4, 2), dist=6
    The maximum Manhattan distance is 6.
    """
    assert maxDistance("NSWWEW", 3) == 6

def test_no_changes_allowed():
    """
    Test with k=0, where no changes are possible.
    The result should be the maximum distance of the original path.
    s = "NESW"
    Path:
    'N': (0, 1), dist=1
    'E': (1, 1), dist=2
    'S': (1, 0), dist=1
    'W': (0, 0), dist=0
    Max distance during traversal is 2.
    """
    assert maxDistance("NESW", 0) == 2

def test_all_changes_allowed():
    """
    Test with k=len(s), allowing all characters to be changed.
    The maximum possible distance is len(s).
    s = "EW", k = 2
    Can be changed to "NN".
    Path:
    'N': (0, 1), dist=1
    'N': (0, 2), dist=2
    Max distance is 2.
    """
    assert maxDistance("EW", 2) == 2

def test_single_move_no_change():
    """
    Test with a single move and no changes.
    s = "W", k = 0
    Path: 'W': (-1, 0), dist=1.
    Max distance is 1.
    """
    assert maxDistance("W", 0) == 1

def test_single_move_with_change():
    """
    Test with a single move and k=1.
    Changing the move doesn't increase the distance.
    s = "W", k = 1. Can be "N", "S", or "E".
    Path for 'N': (0, 1), dist=1.
    Max distance is 1.
    """
    assert maxDistance("W", 1) == 1

def test_all_same_direction():
    """
    Test with a path already going in one direction.
    Changes cannot improve the distance.
    s = "NNNN", k = 1
    Path: (0,1), (0,2), (0,3), (0,4).
    Max distance is 4. Changing one 'N' to another direction will not increase this.
    """
    assert maxDistance("NNNN", 1) == 4

def test_cancelling_moves_no_change():
    """
    Test with moves that cancel each other out, with k=0.
    s = "NSNS"
    Path:
    'N': (0, 1), dist=1
    'S': (0, 0), dist=0
    'N': (0, 1), dist=1
    'S': (0, 0), dist=0
    Max distance is 1.
    """
    assert maxDistance("NSNS", 0) == 1

def test_cancelling_moves_with_one_change():
    """
    Test with cancelling moves, where one change can significantly improve distance.
    s = "NSNS", k = 1
    Change s[1] 'S' to 'N' -> "NNNS".
    Path for "NNNS":
    'N': (0, 1), dist=1
    'N': (0, 2), dist=2
    'N': (0, 3), dist=3
    'S': (0, 2), dist=2
    Max distance is 3.
    """
    assert maxDistance("NSNS", 1) == 3

def test_cancelling_moves_with_enough_changes():
    """
    Test with cancelling moves, with enough changes to make the path optimal.
    s = "NSNS", k = 2
    Change both 'S' to 'N' -> "NNNN".
    Path: (0,1), (0,2), (0,3), (0,4).
    Max distance is 4.
    """
    assert maxDistance("NSNS", 2) == 4

def test_max_dist_achieved_mid_path():
    """
    Test a case where the max distance is achieved mid-path.
    s = "NWS", k = 1
    Change s[2] 'S' to 'N' -> "NWN".
    Path for "NWN":
    'N': (0, 1), dist=1
    'W': (-1, 1), dist=2
    'N': (-1, 2), dist=3
    The path ends at (-1, 2), but the max distance is 3, achieved at the end of this short path.
    """
    assert maxDistance("NWS", 1) == 3

def test_k_is_just_enough():
    """
    Test a case where k is exactly the number of changes needed.
    s = "EESW", k=1
    Original path: (1,0) d=1 -> (2,0) d=2 -> (2,-1) d=3 -> (1,-1) d=2. Max dist=3.
    With k=1, change 'W' to 'E' -> "EESE".
    Path for "EESE":
    'E': (1,0), d=1
    'E': (2,0), d=2
    'S': (2,-1), d=3
    'E': (3,-1), d=4
    Max distance is 4.
    """
    assert maxDistance("EESW", 1) == 4

def test_k_not_enough_for_full_length():
    """
    Test a case where k is not enough to achieve len(s) as distance.
    s = "SESE", k=1
    Original path: (0,-1) d=1 -> (1,-1) d=2 -> (1,-2) d=3 -> (2,-2) d=4. Max dist=4.
    With k=1, we can't improve this.
    Let's use s="NSWE", k=1
    Original path: (0,1)d=1 -> (0,0)d=0 -> (-1,0)d=1 -> (0,0)d=0. Max dist=1.
    With k=1, change 'S' to 'N' -> "NNWE".
    Path for "NNWE":
    'N': (0,1), d=1
    'N': (0,2), d=2
    'W': (-1,2), d=3
    'E': (0,2), d=2
    Max distance is 3.
    """
    assert maxDistance("NSWE", 1) == 3
