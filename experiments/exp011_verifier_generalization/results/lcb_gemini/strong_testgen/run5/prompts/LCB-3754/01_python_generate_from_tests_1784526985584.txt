```python
def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    # Count movements
    n_count = s.count('N')
    s_count = s.count('S')
    e_count = s.count('E')
    w_count = s.count('W')

    # Calculate the net movements
    net_north = n_count - s_count
    net_east = e_count - w_count

    # Calculate the current distance from origin
    position = (net_east, net_north)
    max_distance = abs(position[0]) + abs(position[1])

    # Walk through each move and calculate maximum distance
    for move in s:
        if move == 'N':
            y += 1
        elif move == 'S':
            y -= 1
        elif move == 'E':
            x += 1
        elif move == 'W':
            x -= 1

        # Calculate current distance
        current_distance = abs(x) + abs(y)
        max_distance = max(max_distance, current_distance)

        # Check if we can utilize one of the k changes
        if k > 0:
            # Calculate prospective distance if we could change current move
            prospective_distance = abs(x) + abs(y) + 2
            max_distance = max(max_distance, prospective_distance)
            k -= 1

    # Return the maximum possible Manhattan distance
    return max_distance

```
