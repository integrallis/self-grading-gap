from typing import List
from itertools import permutations


def totalNumbers(digits: List[int]) -> int:
    distinct_numbers = set()
    even_digits = [d for d in digits if d % 2 == 0]
    for even_digit in even_digits:
        # We will form numbers ending with the current even digit
        remaining_digits = digits.copy()
        remaining_digits.remove(even_digit)
        for first_digit in remaining_digits:
            if first_digit == 0:
                continue  # Skip leading zero
            for second_digit in remaining_digits:
                if second_digit == first_digit:
                    continue  # Only use distinct digits
                distinct_numbers.add((first_digit, second_digit, even_digit))
    return len(distinct_numbers)
```
