from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    gaps = []

    # Calculate the gaps before, between, and after meetings
    gaps.append(startTime[0])  # Gap before the first meeting
    for i in range(1, n):
        gaps.append(startTime[i] - endTime[i - 1])  # Gaps between meetings
    gaps.append(eventTime - endTime[-1])  # Gap after the last meeting

    # If k is greater than or equal to number of gaps - 1, we can merge all gaps
    if k >= len(gaps) - 1:
        return sum(gaps)

    # Sort gaps to maximize free time by merging the largest gaps
    gaps.sort(reverse=True)
    # Sum the largest k + 1 gaps
    return sum(gaps[:k + 1])
