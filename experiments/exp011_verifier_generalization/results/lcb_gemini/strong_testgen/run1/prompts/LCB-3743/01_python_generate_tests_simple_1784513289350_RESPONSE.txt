def test_k_equals_n():
    """
    Tests the case where all meetings can be rescheduled (k=n).
    The max free time should be the total free time available in the event.
    eventTime = 20, k=3, n=3, startTime = [2,8,15], endTime = [5,10,18]
    Gaps: [2, 8-5, 15-10, 20-18] -> [2, 3, 5, 2]
    With k=3 (n=3), we can merge k+1=4 gaps (all gaps).
    Sum of all gaps: 2 + 3 + 5 + 2 = 12.
    Alternatively, total meeting duration = (5-2)+(10-8)+(18-15) = 3+2+3=8.
    Total free time = eventTime - total duration = 20 - 8 = 12.
    """
    assert maxFreeTime(eventTime=20, k=3, startTime=[2, 8, 15], endTime=[5, 10, 18]) == 12
