from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize a 2D DP array
    dp = [[0] * 3 for _ in range(n)]

    # Base case: the cost of painting the first house
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the dp array based on the recurrence relations
    for i in range(1, n):
        for j in range(3):
            # Choose the minimum cost from previous house avoiding the same color
            dp[i][j] = min(
                dp[i-1][(j+1)%3],
                dp[i-1][(j+2)%3]
            ) + cost[i][j]

    # The last house must also satisfy equidistant constraint
    min_cost = float('inf')
    for j in range(3):
        for k in range(3):
            if j != k:
                # Calculate cost for the last house while considering equidistant pairs
                if n > 2:
                    min_cost = min(min_cost, dp[n-2][j] + cost[n-1][k])
                else:
                    min_cost = min(min_cost, dp[n-1][j] + cost[n-2][k])

    return min_cost
