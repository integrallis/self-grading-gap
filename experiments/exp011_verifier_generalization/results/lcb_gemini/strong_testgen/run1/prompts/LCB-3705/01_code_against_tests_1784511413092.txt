```python
from typing import List

def largestInteger(nums: List[int], k: int) -> int:
    from collections import defaultdict

    count = defaultdict(int)
    n = len(nums)

    # Count occurrences of each number in each subarray of size k
    for i in range(n - k + 1):
        unique_numbers = set()
        for j in range(k):
            unique_numbers.add(nums[i + j])
        for number in unique_numbers:
            count[number] += 1

    # Find the largest integer that appears in exactly one subarray
    largest_almost_missing = -1
    for number, occurrences in count.items():
        if occurrences == 1:
            largest_almost_missing = max(largest_almost_missing, number)

    return largest_almost_missing
```
