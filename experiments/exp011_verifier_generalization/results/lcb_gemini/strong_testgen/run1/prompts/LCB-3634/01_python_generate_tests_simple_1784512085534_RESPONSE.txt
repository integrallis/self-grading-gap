import pytest
from solution import calculateScore

# Note: The problem defines mirror letters like 'a' <-> 'z', 'b' <-> 'y'.

def test_canonical_example_1():
    # spec: calculateScore("aczzx") -> 5
    # i=2, s[2]='z', mirror='a'. Closest unmarked 'a' is at j=0. score += 2-0=2. Mark 0,2.
    # i=4, s[4]='x', mirror='c'. Closest unmarked 'c' is at j=1. score += 4-1=3. Mark 1,4.
    # Total score = 2 + 3 = 5.
    assert calculateScore("aczzx") == 5

def test_canonical_example_2():
    # spec: calculateScore("abcdef") -> 0
    # No character's mirror appears earlier in the string.
    # For example, at i=5, s[5]='f', mirror='u'. 'u' is not in s[0:5].
    # No pairs are formed, so score remains 0.
    assert calculateScore("abcdef") == 0

def test_min_length_string():
    # Test with a single-character string, which is the minimum length.
    # s = "a"
    # i=0: no j < i, so nothing happens.
    # Score is 0.
    assert calculateScore("a") == 0

def test_no_mirror_pairs_exist():
    # Test a string where no character's mirror is present.
    # s = "programming"
    # The mirrors of 'p','r','o','g','a','m','i','n' are not in the string.
    # Score is 0.
    assert calculateScore("programming") == 0

def test_simple_adjacent_pair():
    # s = "az"
    # i=0: 'a', skip.
    # i=1: 'z', mirror is 'a'. j=0 is 'a' and unmarked. score += 1-0=1.
    # Total score = 1.
    assert calculateScore("az") == 1

def test_simple_distant_pair():
    # s = "ahelloz"
    # i=0..5: no pairs formed.
    # i=6: 'z', mirror 'a'. j=0 is 'a' and unmarked. score += 6-0=6.
    # Total score = 6.
    assert calculateScore("ahelloz") == 6

def test_pair_order_matters():
    # s = "za"
    # i=0: 'z', skip.
    # i=1: 'a', mirror 'z'. j=0 is 'z' and unmarked. score += 1-0=1.
    # Wait, the spec says j < i. Let's re-verify this.
    # i=0: s[0]='z', no j < 0. Skip.
    # i=1: s[1]='a', mirror 'z'. s[0] is 'z'. j=0. Unmarked.
    # score += 1-0 = 1.
    # My initial assumption was wrong.
    # Let's test a case where this matters more clearly.
    # s = "bazy"
    # i=0, 'b': skip
    # i=1, 'a': skip
    # i=2, 'z': mirror 'a'. j=1 is 'a'. score += 2-1 = 1
    # i=3, 'y': mirror 'b'. j=0 is 'b'. score += 3-0 = 3
    # Total = 4
    assert calculateScore("bazy") == 4
    
    # Let's add a case where order is more obvious.
    # s = "za", mirror of 'a' at i=1 is 'z' at j=0. This is a valid pair.
    # score += 1-0=1
    assert calculateScore("za") == 1
    # The key is `j < i`, so a later character can pair with an earlier one.

def test_multiple_partners_choose_closest():
    # s = "abazy"
    # i=0..2: skip
    # i=3: 'z', mirror 'a'. Potential partners at j=0 and j=2.
    # Closest is j=2. score += 3-2=1. Mark 2,3.
    # i=4: 'y', mirror 'b'. Partner at j=1. It's unmarked. score += 4-1=3. Mark 1,4.
    # Total score = 1 + 3 = 4.
    assert calculateScore("abazy") == 4

def test_partner_already_marked():
    # s = "azaz"
    # i=0: 'a', skip
    # i=1: 'z', mirror 'a'. j=0. score += 1-0=1. Mark 0,1.
    # i=2: 'a', mirror 'z'. j=1 is 'z', but is marked. No other unmarked 'z' before. Skip.
    # i=3: 'z', mirror 'a'. j=0 is 'a', but is marked. j=2 is 'a' and is unmarked.
    # score += 3-2=1. Mark 2,3.
    # Total score = 1 + 1 = 2.
    assert calculateScore("azaz") == 2

def test_nested_pairs():
    # s = "abzy"
    # i=0,1: skip
    # i=2: 'z', mirror 'a'. j=0. score += 2-0=2. Mark 0,2.
    # i=3: 'y', mirror 'b'. j=1. score += 3-1=2. Mark 1,3.
    # Total score = 2 + 2 = 4.
    assert calculateScore("abzy") == 4

def test_overlapping_pairs():
    # s = "abzya"
    # i=0,1: skip
    # i=2: 'z', mirror 'a'. j=0. score += 2-0=2. Mark 0,2.
    # i=3: 'y', mirror 'b'. j=1. score += 3-1=2. Mark 1,3.
    # i=4: 'a', mirror 'z'. j=2 is 'z', but is marked. Skip.
    # Total score = 2 + 2 = 4.
    assert calculateScore("abzya") == 4

def test_self_mirroring_pair():
    # 'm' and 'n' are mirrors of each other.
    # s = "mnmn"
    # i=0: 'm', skip.
    # i=1: 'n', mirror 'm'. j=0. score += 1-0=1. Mark 0,1.
    # i=2: 'm', mirror 'n'. j=1 is 'n', but marked. Skip.
    # i=3: 'n', mirror 'm'. j=0 is 'm', marked. j=2 is 'm', unmarked. score += 3-2=1. Mark 2,3.
    # Total score = 1 + 1 = 2.
    assert calculateScore("mnmn") == 2

def test_long_string_alternating_pairs():
    # s = "az" * 50, length 100
    # For each 'z' at index 2k+1, it pairs with 'a' at 2k.
    # The score for each pair is (2k+1) - 2k = 1.
    # There are 50 such pairs.
    # Total score = 50 * 1 = 50.
    s = "az" * 50
    assert calculateScore(s) == 50

def test_long_string_separated_halves():
    # s = "a"*50 + "z"*50, length 100
    # The first 'z' is at i=50. It pairs with the last 'a' at j=49. score += 50-49=1.
    # The second 'z' is at i=51. It pairs with the second-to-last 'a' at j=48. score += 51-48=3.
    # The k-th 'z' (0-indexed, k=0..49) is at i=50+k. It pairs with a at j=49-k.
    # The score for this pair is (50+k) - (49-k) = 1 + 2k.
    # Total score is sum_{k=0}^{49} (1 + 2k) = 50*1 + 2*sum_{k=0}^{49}(k)
    # = 50 + 2 * (49*50/2) = 50 + 49*50 = 50 + 2450 = 2500.
    s = "a" * 50 + "z" * 50
    assert calculateScore(s) == 2500
