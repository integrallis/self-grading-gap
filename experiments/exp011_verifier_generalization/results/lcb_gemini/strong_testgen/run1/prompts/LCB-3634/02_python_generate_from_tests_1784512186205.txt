def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # Create a mapping for mirrors
    for i in range(len(s)):
        if marked[i]:  # Skip if the current character is already marked
            continue
        current_char = s[i]
        mirror_char = mirror[current_char]
        for j in range(i - 1, -1, -1):  # Find the closest unmarked index j
            if not marked[j] and s[j] == mirror_char:
                score += i - j
                marked[i] = marked[j] = True  # Mark both indices
                break
    return score

```
