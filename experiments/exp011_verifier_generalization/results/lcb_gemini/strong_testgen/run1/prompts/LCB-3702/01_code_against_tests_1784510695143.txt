```python
from typing import List
import math
from functools import reduce

def maxLength(nums: List[int]) -> int:
    def _gcd(a, b):
        return math.gcd(a, b)

    def _gcd_list(numbers):
        return reduce(_gcd, numbers)

    def _lcm(a, b):
        if a == 0 or b == 0:
            return 0
        return abs(a * b) // _gcd(a, b) if a != 0 and b != 0 else 0

    def _lcm_list(numbers):
        return reduce(_lcm, numbers, 1)

    n = len(nums)
    max_length = 0

    for start in range(n):
        product = 1
        current_gcd = nums[start]
        for end in range(start, n):
            product *= nums[end]
            current_gcd = _gcd(current_gcd, nums[end])
            current_lcm = _lcm_list(nums[start:end + 1])
            if product == current_lcm * current_gcd:
                max_length = max(max_length, end - start + 1)

    return max_length
```
