def test_large_numbers_and_modulo():
    """
    Tests the modulo arithmetic with large numbers that would otherwise overflow.
    """
    # nums = [10^9, 10^9 - 1], k = 2. MOD = 10^9 + 7.
    # Let A = 10^9, B = 10^9 - 1.
    # Subsequences and their min/max sums:
    # [A]: sum = A + A = 2A
    # [B]: sum = B + B = 2B
    # [A, B]: sum = B + A
    # Total = 2A + 2B + (A+B) = 3A + 3B = 3(A+B).
    # A+B = 10^9 + 10^9 - 1 = 2*10^9 - 1
    # Total = 3 * (2*10^9 - 1) = 6*10^9 - 3.
    # Modulo calculation:
    # 10^9 ≡ -7 (mod 10^9 + 7)
    # Total ≡ 6 * (-7) - 3 (mod 10^9 + 7)
    # Total ≡ -42 - 3 ≡ -45 (mod 10^9 + 7)
    # Total ≡ 10^9 + 7 - 45 = 999999962
    large_num_1 = 10**9
    large_num_2 = 10**9 - 1
    assert minMaxSums(nums=[large_num_1, large_num_2], k=2) == 999999962

def test_larger_n_k_combination():

    Tests a slightly more complex case to verify the general algorithm.
    """

    # as a min and max over all valid subsequences.
    # Contribution of 1: min in 7 subs, max in 1 sub. -> 1*7 + 1*1 = 8
    # Contribution of 2: min in 4 subs, max in 2 subs. -> 2*4 + 2*2 = 12
    # Contribution of 3: min in 2 subs, max in 4 subs. -> 3*2 + 3*4 = 18
    # Contribution of 4: min in 1 sub, max in 7 subs. -> 4*1 + 4*7 = 32
    # Total = 8 + 12 + 18 + 32 = 70
    assert minMaxSums(nums=[1, 2, 3, 4], k=3) == 70
