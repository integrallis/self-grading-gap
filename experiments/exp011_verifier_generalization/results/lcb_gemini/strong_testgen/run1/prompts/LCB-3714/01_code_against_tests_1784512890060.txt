```python
from typing import List

MOD = 10**9 + 7


def factorial(n: int) -> int:
    result = 1
    for i in range(2, n + 1):
        result = (result * i) % MOD
    return result


def mod_inverse(a: int, p: int) -> int:
    return pow(a, p - 2, p)


def choose(n: int, k: int) -> int:
    if k > n or k < 0:
        return 0
    return (factorial(n) * mod_inverse(factorial(k), MOD) % MOD * mod_inverse(factorial(n - k), MOD) % MOD) % MOD


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Iterate through each number in nums
    for i in range(n):
        # For each nums[i], calculate how many subsequences will treat it as min and as max
        if k < n:
            min_contributions = choose(n - 1, k - 1)
            max_contributions = choose(n - 1, k - 1)
        else:
            # If k == n, every number is both a min and max in all subsequences
            min_contributions = 1
            max_contributions = 1

        total_sum = (total_sum + (nums[i] * (min_contributions + max_contributions)) % MOD) % MOD

    return total_sum
```
