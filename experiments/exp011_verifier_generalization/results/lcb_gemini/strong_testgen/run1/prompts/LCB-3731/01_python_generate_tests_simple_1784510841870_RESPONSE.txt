from typing import List
import pytest
from solution import subarraySum


def test_canonical_example_1():
    """
    Tests the first example from the specification.
    Input: nums = [2,3,1]
    Output: 11
    """
    nums = [2, 3, 1]
    # Calculation:
    # i=0: start=max(0, 0-2)=0. subarray=nums[0:1]=[2]. Sum=2
    # i=1: start=max(0, 1-3)=0. subarray=nums[0:2]=[2,3]. Sum=5
    # i=2: start=max(0, 2-1)=1. subarray=nums[1:3]=[3,1]. Sum=4
    # Total sum = 2 + 5 + 4 = 11
    expected = 11
    assert subarraySum(nums) == expected
