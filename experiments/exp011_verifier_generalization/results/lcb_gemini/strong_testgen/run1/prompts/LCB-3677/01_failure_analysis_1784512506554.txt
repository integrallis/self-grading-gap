from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    # DP table to store maximum profit with at most 2 neutralizations
    dp = [[[-float('inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0]  # Starting point

    for i in range(m):
        for j in range(n):
            for k in range(3):  # k is the number of neutralizations used
                if i == 0 and j == 0:
                    continue  # Starting cell already accounted for
                current = coins[i][j]

                if i > 0:
                    # Coming from above
                    previous = dp[i - 1][j][k]
                    if current < 0:
                        dp[i][j][k] = max(dp[i][j][k], previous + (current if k < 2 else 0))  # Without neutralizing
                        if k < 2:
                            dp[i][j][k + 1] = max(dp[i][j][k + 1], previous)
                    else:
                        dp[i][j][k] = max(dp[i][j][k], previous + current)

                if j > 0:
                    # Coming from the left
                    previous = dp[i][j - 1][k]
                    if current < 0:

                            dp[i][j][k + 1] = max(dp[i][j][k + 1], previous)
                    else:
                        dp[i][j][k] = max(dp[i][j][k], previous + current)

    # The final answer is the maximum profit obtainable at the bottom-right corner.
    return max(max(dp[m-1][n-1]), 0)  # Ensure non-negative profit
