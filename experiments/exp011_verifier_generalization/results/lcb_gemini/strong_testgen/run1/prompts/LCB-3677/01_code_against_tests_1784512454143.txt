```python
from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[float('-inf')] * (n + 1) for _ in range(m + 1)]
    dp[0][1] = 0  # Starting point

    for i in range(m):
        for j in range(n):
            current = coins[i][j]
            for k in range(3):  # k is the number of neutralizations used
                if i > 0:
                    profit_from_top = dp[i][j + 1] + (current if current >= 0 else 0) + (0 if current < 0 and k < 2 else -current)
                    dp[i + 1][j + 1] = max(dp[i + 1][j + 1], profit_from_top)
                if j > 0:
                    profit_from_left = dp[i + 1][j] + (current if current >= 0 else 0) + (0 if current < 0 and k < 2 else -current)
                    dp[i + 1][j + 1] = max(dp[i + 1][j + 1], profit_from_left)

    return max(0, dp[m][n])  # Return the maximum profit at the bottom-right corner, ensuring non-negative profit
```
