def test_threshold_forces_path_choice():
    """
    A node has multiple outgoing edges, but the threshold forces it to drop some.
    The correct answer depends on a path that remains viable.
    """
    n = 3
    edges = [[1, 0, 10], [1, 2, 5], [2, 0, 8]]
    threshold = 1
    # For node 1, out-degree is 2, threshold is 1. It must drop one edge.
    # If max_weight is 8, edges (1,2,5) and (2,0,8) are available.
    # Node 1 can drop (1,0,10) and use path 1->2->0.
    # If max_weight is 7, edge (2,0,8) is gone. 2 cannot reach 0, so 1 cannot either.
    # So, minimum max_weight is 8.
    assert minMaxWeight(n, edges, threshold) == 8

def test_threshold_forces_choice_amongst_paths():
    """
    A node has an out-degree > threshold. Some of its destination nodes have better paths
    to 0 than others. The solution depends on selecting the right edges to keep.
    """
    n = 5
    edges = [[1, 2, 10], [1, 3, 10], [1, 4, 10], [2, 0, 5], [3, 0, 5], [4, 0, 20]]
    threshold = 2
    # Node 1 has out-degree 3, but threshold is 2. It must drop one edge.
    # To ensure all nodes reach 0, the max weight must be at least 20 (for node 4).
    # With max_weight=20, node 1 can reach 0 via 2, 3, or 4.
    # It can drop any of its 3 outgoing edges and still find a path.
    # If max_weight=19, edge (4,0,20) is gone. Node 4 can't reach 0.
    # So the answer is 20.
    assert minMaxWeight(n, edges, threshold) == 20

def test_multiple_edges_between_nodes():
    """
    The graph has multiple edges between the same two nodes, with different weights.
    """
    n = 3
    edges = [[1, 2, 10], [1, 2, 20], [2, 0, 5]]
    threshold = 1
    # Node 1 has two edges to node 2. Total out-degree is 2. Threshold is 1.
    # To reach 0, 1 needs to go via 2.
    # With max weight 10, 1 can use (1,2,10) and 2 can use (2,0,5).
    # Node 1 must drop one edge. It can drop (1,2,20) and keep (1,2,10).
    assert minMaxWeight(n, edges, threshold) == 10
