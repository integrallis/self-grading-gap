from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Helper function to check reachability using BFS
    def bfs_check(max_weight: int) -> bool:
        from collections import defaultdict, deque
        graph = defaultdict(list)
        out_degree = [0] * n

        # Build a graph with filtered edges
        for u, v, w in edges:
            if w <= max_weight:
                graph[u].append(v)
                out_degree[u] += 1

        # Check if out-degree exceeds threshold
        if any(deg > threshold for deg in out_degree):
            return False

        # BFS to check if all nodes can reach node 0
        queue = deque([0])
        visited = set([0])

        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if neighbor not in visited:
                    visited.add(neighbor)
                    queue.append(neighbor)

        return len(visited) == n

    # Binary search for the minimum maximum weight
    left, right = 1, 10**6
    answer = -1

    while left <= right:
        mid = (left + right) // 2
        if bfs_check(mid):
            answer = mid
            right = mid - 1
        else:
            left = mid + 1

    return answer
