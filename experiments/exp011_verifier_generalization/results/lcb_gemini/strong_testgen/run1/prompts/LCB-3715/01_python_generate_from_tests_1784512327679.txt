from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Dictionary to hold the count of coins at each position
    coin_count = {}

    # Fill the coin counts based on the segments
    for l, r, c in coins:
        for i in range(l, r + 1):
            if i in coin_count:
                coin_count[i] += c
            else:
                coin_count[i] = c

    # Compute the maximum coins that can be collected in k consecutive bags
    max_coins = 0
    current_sum = 0

    # Calculate the initial sum for the first k bags
    for i in range(1, k + 1):
        current_sum += coin_count.get(i, 0)
    max_coins = current_sum

    # Use a sliding window to calculate the max coins
    for i in range(1, max(coin_count.keys()) + 1 - k):
        current_sum = current_sum - coin_count.get(i, 0) + coin_count.get(i + k, 0)
        max_coins = max(max_coins, current_sum)

    return max_coins
```
