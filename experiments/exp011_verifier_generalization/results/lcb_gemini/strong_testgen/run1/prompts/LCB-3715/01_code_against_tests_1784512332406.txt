```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Dictionary to hold the count of coins at each position
    from collections import defaultdict
    coin_count = defaultdict(int)

    # Fill the coin counts based on the segments
    for l, r, c in coins:
        coin_count[l] += c
        coin_count[r + 1] -= c

    # Create a list of positions and their cumulative coin totals
    sorted_keys = sorted(coin_count.keys())
    current_coins = 0
    position_coins = []
    for key in sorted_keys:
        current_coins += coin_count[key]
        position_coins.append((key, current_coins))

    # Using a sliding window to find the max coins for k consecutive bags
    max_coins = 0
    start = 0
    for end in range(len(position_coins)):
        while position_coins[end][0] - position_coins[start][0] >= k:
            start += 1
        num_bags = position_coins[end][0] - position_coins[start][0] + 1
        coins_in_window = position_coins[start][1] + (k - num_bags) * (position_coins[end][1] // num_bags)
        max_coins = max(max_coins, coins_in_window)

    return max_coins
```
