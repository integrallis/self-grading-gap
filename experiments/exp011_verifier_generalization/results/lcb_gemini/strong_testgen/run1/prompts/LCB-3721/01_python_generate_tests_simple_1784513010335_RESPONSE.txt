def test_canonical_example_1():
    """
    Tests the first example from the problem description.

        ["MESSAGE", "71", "HERE"]
    ]

    # t=10 (MESSAGE "id1 id0"): users 0 and 1 mentioned. mentions = [1, 1].
    # t=11 (OFFLINE "0"): user 0 goes offline until t=71.
    # t=71 (Status update): user 0 comes back online. Both are online.
    # t=71 (MESSAGE "HERE"): online users (0, 1) are mentioned. mentions = [1+1, 1+1] = [2, 2].
    expected = [2, 2]
