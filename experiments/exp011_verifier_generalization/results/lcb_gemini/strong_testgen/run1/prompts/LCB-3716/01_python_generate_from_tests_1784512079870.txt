from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = [1] * n  # dp[i] will be the length of the longest subsequence ending with nums[i]
    for i in range(n):
        for j in range(i):
            if abs(nums[i] - nums[j]) >= (dp[j] - 1):
                dp[i] = max(dp[i], dp[j] + 1)
    return max(dp)
```
