```python
from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = [1] * n  # dp[i] will be the length of the longest subsequence ending with nums[i]
    diffs = [[0] * n for _ in range(n)]  # Store the absolute differences

    for i in range(n):
        for j in range(i):
            diffs[i][j] = abs(nums[i] - nums[j])

    for i in range(n):
        for j in range(i):
            if diffs[i][j] > 0:
                k = j
                while k >= 0:
                    if diffs[j][k] >= diffs[i][j]:
                        dp[i] = max(dp[i], dp[j] + 1)
                    k -= 1

    return max(dp)
```
