```python
from typing import List

def zigzagTraversal(grid: List[List[int]]) -> List[int]:
    result = []
    for i in range(len(grid)):
        row = grid[i]
        if i % 2 == 0:
            # Traverse right
            for j in range(0, len(row)):
                if j % 2 == 0:
                    result.append(row[j])
        else:
            # Traverse left
            for j in range(len(row) - 1, -1, -1):
                if j % 2 == 0:
                    result.append(row[j])
    return result

```
