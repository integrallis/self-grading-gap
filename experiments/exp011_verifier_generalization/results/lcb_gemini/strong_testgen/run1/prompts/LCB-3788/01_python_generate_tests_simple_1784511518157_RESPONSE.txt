import pytest
from typing import List
from solution import maxSum

# ## Task
#
# **Implement maxSum()**
#
# You are given an integer array nums.
# You are allowed to delete any number of elements from nums without making it empty. After performing the deletions, select a subarray of nums such that:
#
# All elements in the subarray are unique.
# The sum of the elements in the subarray is maximized.
#
# Return the maximum sum of such a subarray.
#
# Constraints:
#
# 1 <= nums.length <= 100
# -100 <= nums[i] <= 100

# The problem can be interpreted as finding a subsequence of unique elements with the maximum sum.
# This simplifies to summing all unique positive numbers. If there are no positive numbers,
# the result is the largest number in the array (which will be zero or negative).

# MANDATORY: Canonical Examples from Specification

def test_canonical_example_1():
    """
    Test with input from Example 1: nums = [1,2,3,4,5] -> 15
    """
    # spec: maxSum([1,2,3,4,5]) -> 15
    # The unique positive numbers are 1, 2, 3, 4, 5. Their sum is 15.
    # This can be achieved by selecting the entire array, which has unique elements and sums to 15.
    nums = [1, 2, 3, 4, 5]
    expected = 15
    assert maxSum(nums) == expected

def test_canonical_example_2():
    """
    Test with input from Example 2: nums = [1,1,0,1,1] -> 1
    """
    # spec: maxSum([1,1,0,1,1]) -> 1
    # The only unique positive number is 1. The maximum sum is 1.
    # This can be achieved by forming a subsequence of just [1].
    nums = [1, 1, 0, 1, 1]
    expected = 1
    assert maxSum(nums) == expected

def test_canonical_example_3():
    """
    Test with input from Example 3: nums = [1,2,-1,-2,1,0,-1] -> 3
    """
    # spec: maxSum([1,2,-1,-2,1,0,-1]) -> 3
    # The unique positive numbers are 1 and 2. Their sum is 3.
    # A subsequence like [2, 1] can be formed, with unique elements and sum 3.
    nums = [1, 2, -1, -2, 1, 0, -1]
    expected = 3
    assert maxSum(nums) == expected
