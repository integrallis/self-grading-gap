Test suite for the reverseDegree function.

This test suite is written in the RED phase of Test-Driven Development (TDD).

import pytest
from solution import reverseDegree

# The problem specification defines the reverse alphabet value for a character 'c'
# as 27 minus its 1-indexed position in the standard alphabet.
# This can be calculated as: 26 - (ord(c) - ord('a')).
# 'a' -> 26 - (0) = 26
# 'b' -> 26 - (1) = 25
# ...
# 'z' -> 26 - (25) = 1


def test_canonical_example_1():
    """Tests the first example from the specification: s = "abc"."""
    # 'a' (val=26) at pos 1: 26 * 1 = 26
    # 'b' (val=25) at pos 2: 25 * 2 = 50
    # 'c' (val=24) at pos 3: 24 * 3 = 72
    # Total = 26 + 50 + 72 = 148
    assert reverseDegree("abc") == 148


def test_canonical_example_2():
    """Tests the second example from the specification: s = "zaza"."""
    # 'z' (val=1) at pos 1: 1 * 1 = 1
    # 'a' (val=26) at pos 2: 26 * 2 = 52
    # 'z' (val=1) at pos 3: 1 * 3 = 3
    # 'a' (val=26) at pos 4: 26 * 4 = 104
    # Total = 1 + 52 + 3 + 104 = 160
    assert reverseDegree("zaza") == 160


def test_single_character_at_start_of_alphabet():
    """Tests a single character string ('a'), which is the minimum length and start of alphabet."""
    # 'a' (val=26) at pos 1: 26 * 1 = 26
    # Total = 26
    assert reverseDegree("a") == 26


def test_single_character_at_end_of_alphabet():
    """Tests a single character string ('z'), which is the minimum length and end of alphabet."""
