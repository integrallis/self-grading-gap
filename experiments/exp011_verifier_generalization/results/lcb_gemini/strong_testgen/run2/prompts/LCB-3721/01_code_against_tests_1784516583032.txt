```python
from typing import List

def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    # Initialize mentions count for each user
    mentions = [0] * numberOfUsers
    # Track online status of users
    online_status = [True] * numberOfUsers
    # Track when users will come back online
    offline_until = [0] * numberOfUsers

    # Sort events by timestamp
    events.sort(key=lambda x: int(x[1]))

    for event in events:
        event_type, timestamp_str, detail = event
        timestamp = int(timestamp_str)

        # Process offline events first
        if event_type == "OFFLINE":
            user_id = int(detail)
            online_status[user_id] = False
            offline_until[user_id] = timestamp + 60

        # Now process message events
        elif event_type == "MESSAGE":
            # Update online status for users
            for i in range(numberOfUsers):
                if not online_status[i] and timestamp >= offline_until[i]:
                    online_status[i] = True

            mentions_in_message = set()

            if detail == "ALL":
                mentions_in_message.update(range(numberOfUsers))
            elif detail == "HERE":
                # Only add online users
                mentions_in_message.update(i for i in range(numberOfUsers) if online_status[i])
            else:
                # Parse individual user mentions
                for mention in detail.split():
                    if mention.startswith("id"):
                        user_id = int(mention[2:])
                        mentions_in_message.add(user_id)

            # Count mentions
            for user_id in mentions_in_message:
                mentions[user_id] += 1

    return mentions
```
