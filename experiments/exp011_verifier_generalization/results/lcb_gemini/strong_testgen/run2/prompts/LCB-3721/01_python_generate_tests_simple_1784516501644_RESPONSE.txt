def test_complex_scenario_multiple_users_and_offline_periods():
    """
    Tests a more complex scenario with multiple users going offline and coming back online.
    """
    numberOfUsers = 3
    events = [
        ["MESSAGE", "10", "id0 id1"],
        ["OFFLINE", "11", "0"],
        ["OFFLINE", "12", "1"],
        ["MESSAGE", "20", "HERE"],
        ["MESSAGE", "71", "HERE"],
        ["MESSAGE", "72", "HERE"],
    ]
    # t=10: "id0 id1", all online. mentions=[1, 1, 0]
    # t=11: user 0 offline until 71.
    # t=12: user 1 offline until 72.
    # t=20: "HERE", only user 2 online. mentions=[1, 1, 1]
    # t=71: user 0 back online. "HERE", users 0, 2 online. mentions=[1+1, 1, 1+1]=[2, 1, 2]
    # t=72: user 1 back online. "HERE", all users 0,1,2 online. mentions=[2+1, 1+1, 2+1]=[3, 2, 3]
    expected = [3, 2, 3]
    assert countMentions(numberOfUsers, events) == expected
