def test_canonical_example_1():
    """
    Tests the first example from the specification.
    Input: cost = [5, 3, 4, 1, 3, 2]
    """
    cost = [5, 3, 4, 1, 3, 2]
    # The minimum cost to reach position i is the minimum of cost[0]...cost[i].
    # ans[0] = min(5) = 5
    # ans[1] = min(5, 3) = 3
    # ans[2] = min(5, 3, 4) = 3
    # ans[3] = min(5, 3, 4, 1) = 1
    # ans[4] = min(5, 3, 4, 1, 3) = 1
    # ans[5] = min(5, 3, 4, 1, 3, 2) = 1
    expected = [5, 3, 3, 1, 1, 1]
    assert minCosts(cost) == expected

    """
    Tests the second example from the specification.
    Input: cost = [1, 2, 4, 6, 7]

    # The minimum cost to reach any position is dominated by the cost of swapping
    # with person 0, which is 1.
    # ans[0] = min(1) = 1
    # ans[1] = min(1, 2) = 1
    # ans[2] = min(1, 2, 4) = 1

    expected = [1, 1, 1, 1, 1]

def test_strictly_increasing_costs():
    """
    Tests a scenario where costs are always increasing.
    """
    cost = [10, 20, 30, 40, 50]
    # When costs are increasing, the minimum cost to reach any position i

    # ans[0] = min(10) = 10
    # ans[1] = min(10, 20) = 10
    # ans[2] = min(10, 20, 30) = 10
    # ans[3] = min(10, 20, 30, 40) = 10

    assert minCosts(cost) == expected
