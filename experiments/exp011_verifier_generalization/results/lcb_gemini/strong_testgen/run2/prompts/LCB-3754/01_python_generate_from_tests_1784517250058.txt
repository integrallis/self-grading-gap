def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    for char in s:
        if char == 'N':
            y += 1
        elif char == 'S':
            y -= 1
        elif char == 'E':
            x += 1
        elif char == 'W':
            x -= 1
        # Calculate current distance
        current_distance = abs(x) + abs(y)
        max_distance = max(max_distance, current_distance)
    # To maximize distance, we can change k steps to either N/S or E/W
    # The furthest we can go in one direction is by changing steps in the other direction
    return max_distance + k
```
