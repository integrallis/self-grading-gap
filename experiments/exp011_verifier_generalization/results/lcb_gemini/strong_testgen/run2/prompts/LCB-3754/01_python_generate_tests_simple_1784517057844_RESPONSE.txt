from solution import maxDistance


def test_canonical_example_1():
    """
    From spec example 1: s = "NWSE", k = 1 -> 3
    - Optimal change: s[2] from 'S' to 'N'. s becomes "NWNE".
    - Path:
      s[0] = 'N': (0, 1), dist 1
      s[1] = 'W': (-1, 1), dist 2
      s[2] = 'N': (-1, 2), dist 3
      s[3] = 'E': (0, 2), dist 2
    - The maximum Manhattan distance is 3.
    """
    assert maxDistance("NWSE", 1) == 3

def test_canonical_example_2():
    """
    From spec example 2: s = "NSWWEW", k = 3 -> 6
    - A possible optimal change is s[1] 'S'->'N', s[4] 'E'->'W'. s becomes "NNWWWW".
    - Path for "NNWWWW":
      'N' -> (0, 1), dist 1
      'N' -> (0, 2), dist 2
      'W' -> (-1, 2), dist 3
      'W' -> (-2, 2), dist 4
      'W' -> (-3, 2), dist 5
      'W' -> (-4, 2), dist 6
    - The maximum Manhattan distance is 6.
    """
    assert maxDistance("NSWWEW", 3) == 6

def test_no_changes_allowed():
    """
    Test with k=0, where no changes are possible.
    The result is the maximum distance achievable with the original path.
    s = "NESW", k = 0
    - Path:
      'N' -> (0, 1), dist 1
      'E' -> (1, 1), dist 2
      'S' -> (1, 0), dist 1
      'W' -> (0, 0), dist 0
    - Maximum distance is 2.
    """
    assert maxDistance("NESW", 0) == 2

def test_all_changes_allowed():
    """
    Test where k is equal to the length of s.
    All characters can be changed. To maximize distance, change all to one direction.
    s = "EWNS", k = 4, length is 4.
    - Change to "NNNN".
    - Path: (0,1), (0,2), (0,3), (0,4).
    - Distances: 1, 2, 3, 4. Maximum is 4.
    """
    assert maxDistance("EWNS", 4) == 4

def test_single_move_with_change():
    """
    Test with a single character string and one change allowed.
    s = "S", k = 1
    - Original path: (0, -1), dist 1.
    - With one change, e.g., to 'N', path is (0, 1), dist 1.
    - The maximum distance is 1.
    """
    assert maxDistance("S", 1) == 1

def test_uniform_direction_no_benefit_from_k():
    """
    Test a path that already moves consistently in one direction.
    Changes should not improve the maximum distance.
    s = "EEEE", k = 2
    - Original path: (1,0), (2,0), (3,0), (4,0). Dists: 1, 2, 3, 4. Max is 4.
    - Any change would disrupt this linear progression.
    - For example, changing to "ENEE": path (1,0),(1,1),(2,1),(3,1). Dists: 1,2,3,4. Max is still 4.
    - The best strategy is to make no changes.
    """
    assert maxDistance("EEEE", 2) == 4

def test_cancelling_path_with_changes():
    """
    Test a path where moves cancel each other out. Changes can significantly improve the distance.
    s = "NSNS", k = 1
    - Original path: (0,1), (0,0), (0,1), (0,0). Max dist is 1.
    - Change s[1] 'S' to 'N'. s becomes "NNNS".
    - Path for "NNNS":
      'N' -> (0, 1), dist 1
      'N' -> (0, 2), dist 2
      'N' -> (0, 3), dist 3
      'S' -> (0, 2), dist 2
    - Maximum distance is 3.
    """
    assert maxDistance("NSNS", 1) == 3

def test_peak_distance_not_at_end():
    """
    Test a case where the maximum distance is achieved before the end of the path.
    s = "NWS", k = 1
    - Change s[2] 'S' to 'N'. s becomes "NWN".
    - Path for "NWN":
      'N' -> (0, 1), dist 1
      'W' -> (-1, 1), dist 2
      'N' -> (-1, 2), dist 3
    - Maximum distance is 3.
    """
    assert maxDistance("NWS", 1) == 3
