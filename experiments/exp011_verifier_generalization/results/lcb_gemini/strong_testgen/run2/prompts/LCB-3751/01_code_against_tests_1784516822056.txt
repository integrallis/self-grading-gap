```python
from typing import List


def maxFrequency(nums: List[int], k: int) -> int:
    count = 0
    max_freq = 0
    left = 0
    total_addition = 0

    for right in range(len(nums)):
        total_addition += nums[right]
        while total_addition + (right - left + 1) * k > sum(nums[left:right + 1]):
            total_addition -= nums[left]
            left += 1
        max_freq = max(max_freq, right - left + 1)

    return max_freq

```
