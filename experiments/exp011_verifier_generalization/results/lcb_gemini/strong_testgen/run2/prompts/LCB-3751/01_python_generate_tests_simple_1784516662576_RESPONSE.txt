import pytest

from solution import maxFrequency


def test_canonical_example_1():
    """
    Test with the first canonical example from the problem description.
    nums = [1, 2, 3, 4, 5, 6], k = 1 -> 2
    Explanation: Add -5 to nums[2..5]. The array becomes [1, 2, -2, -1, 0, 1].
    The frequency of 1 is 2.
    This corresponds to choosing x=-5. The original number converted to 1 was 6.
    Initial count of 1 is 1 (at index 0). The operation on nums[5] adds one more 1.
    Total frequency: 1 (original) + 1 (new) = 2.
    """
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2


def test_canonical_example_2():
    """

    Explanation: Add 8 to nums[1..9].
    - Original nums[0]=10 is preserved.
    - nums[1]=2 becomes 10 (2+8=10).
    - nums[8]=2 becomes 10 (2+8=10).
    - nums[9]=2 becomes 10 (2+8=10).
    Total frequency: 1 (original) + 3 (new) = 4.
    """
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4


def test_k_not_in_nums():
    """
    Test a case where the target k is not initially present in the array.
    The best operation should convert a block of identical numbers into k.
    nums = [1, 1, 2, 2, 1], k = 3
    Initial frequency of 3 is 0.
    Option 1: Convert 1s to 3s (add x=2). Best subarray is [0..4] which contains all three 1s.
    The new array becomes [3, 3, 4, 4, 3]. Frequency of 3 is 3.
    Option 2: Convert 2s to 3s (add x=1). Best subarray is [2..3] which contains both 2s.
    The new array becomes [1, 1, 3, 3, 1]. Frequency of 3 is 2.
    The maximum frequency is 3.
    """
    assert maxFrequency([1, 1, 2, 2, 1], 3) == 3


def test_all_elements_are_k():
    """
    Test a case where all elements are already equal to k.
    No operation can increase the frequency.
    nums = [7, 7, 7], k = 7
    Initial frequency is 3. Any operation on a non-empty subarray will
    change some 7s to something else, decreasing the frequency.
    The maximum frequency is the initial frequency, 3.
    """
    assert maxFrequency([7, 7, 7], 7) == 3


def test_all_elements_same_but_not_k():
    """
    Test a case where all elements are identical, but not k.
    The entire array can be converted to k.
    nums = [5, 5, 5, 5], k = 10
    Add x=5 to the whole array [0..3].
    The new array becomes [10, 10, 10, 10]. Frequency of 10 is 4.
    """
    assert maxFrequency([5, 5, 5, 5], 10) == 4


def test_single_element_array_is_k():
    """
    Test with a single-element array that is already k.
    nums = [10], k = 10
    Frequency is 1. No operation can improve this.
    """
    assert maxFrequency([10], 10) == 1


def test_single_element_array_not_k():
    """
    Test with a single-element array that can be converted to k.
    nums = [10], k = 20
    Add x=10 to nums[0..0]. The array becomes [20]. Frequency is 1.
    """
    assert maxFrequency([10], 20) == 1


def test_operation_can_increase_count():
    """
    Test a case where an operation can increase the count by converting
    some elements while leaving existing k's untouched.
    nums = [1, 10, 1, 10, 1], k = 10
    Initial frequency of 10 is 2.
    We can convert one of the 1s to 10 by adding x=9.
    E.g., add 9 to subarray nums[0..0]. Array becomes [10, 10, 1, 10, 1]. Freq=3.
    The max frequency is 3.
    """
    assert maxFrequency([1, 10, 1, 10, 1], 10) == 3


def test_multiple_candidate_values_to_convert():
    """

    nums = [1, 1, 10, 2, 2], k = 3
    Initial frequency of 3 is 0.
    Option 1: Convert 1s to 3s (x=2). Subarray [0..1] contains two 1s, giving freq=2.
    Option 2: Convert 10s to 3s (x=-7). Subarray [2..2] contains one 10, giving freq=1.
    Option 3: Convert 2s to 3s (x=1). Subarray [3..4] contains two 2s, giving freq=2.
    Maximum frequency is 2.
    """
    assert maxFrequency([1, 1, 10, 2, 2], 3) == 2


def test_interspersed_k_and_v():
    """
    A case where the optimal subarray might need to span over
    some k's, incurring a cost, but still yielding a net gain.
    nums = [1, 10, 1, 10, 1], k = 1
    Initial frequency of 1 is 3.
    We can convert 10s to 1s by adding x=-9.
    Choosing subarray [1..3] converts nums[1]=10 and nums[3]=10 to 1s,
    but also converts nums[2]=1 to -8.
    Net gain from this operation: 2 new 1s, 1 lost 1. Total gain = 1.
    New freq: 3 (initial) + 1 (gain) = 4.
    """
    assert maxFrequency([1, 10, 1, 10, 1], 1) == 4
