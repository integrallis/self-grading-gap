def test_large_numbers_and_modulo():
    """
    Test with large numbers to ensure modulo arithmetic is handled correctly.
    """
    nums = [MOD - 1, MOD - 1]
    k = 2
    # Let X = MOD - 1. In modulo MOD, X is -1.
    # Subsequences:
    # [X] from index 0: min=X, max=X. Sum = 2X = -2 (mod MOD).
    # [X] from index 1: min=X, max=X. Sum = 2X = -2 (mod MOD).
    # [X, X]: min=X, max=X. Sum = 2X = -2 (mod MOD).
    # Total sum = -2 + -2 + -2 = -6 (mod MOD).
    # -6 mod MOD is MOD - 6.
    expected = MOD - 6
    assert minMaxSums(nums, k) == expected
