from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    # Calculate contribution of each number as minimum and maximum
    for i in range(n):
        # Calculate combinations for max and min counts
        for j in range(1, k + 1):
            # Count how many times nums[i] can be the min
            count_min = (1 << (n - i - 1)) - 1  # 2^(n-i-1) - 1
            count_max = (1 << i) - 1  # 2^i - 1
            total_sum += (nums[i] * (count_min + count_max)) % MOD
            total_sum %= MOD
    return (total_sum * 2) % MOD

```
