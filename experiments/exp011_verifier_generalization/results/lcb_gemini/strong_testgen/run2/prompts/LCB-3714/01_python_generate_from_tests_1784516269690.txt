from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Calculate counts of subsequences contribution
    for i in range(n):
        # Determine contribution of nums[i] as min and max
        # Counts of how often nums[i] can be considered
        count_as_min = 0
        count_as_max = 0

        # Contribution when considering as minimum
        for j in range(1, k + 1):
            count_as_min += combination(n-1, j-1)

        # Contribution when considering as maximum
        for j in range(1, k + 1):
            count_as_max += combination(n-1, j-1)

        total_sum += (nums[i] * (count_as_min + count_as_max)) % MOD
        total_sum %= MOD

    return total_sum % MOD


def combination(n: int, k: int) -> int:
    if k > n:
        return 0
    if k == 0 or k == n:
        return 1
    k = min(k, n - k)  # Take advantage of symmetry

    numer = 1
    denom = 1
    for i in range(k):
        numer = (numer * (n - i)) % MOD
        denom = (denom * (i + 1)) % MOD

    return (numer * pow(denom, MOD - 2, MOD)) % MOD
