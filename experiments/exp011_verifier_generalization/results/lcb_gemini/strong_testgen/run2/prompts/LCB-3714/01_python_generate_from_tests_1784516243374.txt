from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    total_subsequences = 0

    # Calculate counts of how many times each num can be the min and max in subsequences
    for i in range(n):
        # Calculate contributions for nums[i] as min
        count_min = 0
        count_max = 0

        for j in range(1, k + 1):
            if i - j + 1 >= 0:
                count_min += 1
            if n - (i + 1) >= j - 1:
                count_max += 1

        total_sum += (nums[i] * (count_min + count_max)) % MOD
        total_sum %= MOD

    total_subsequences = 0
    for j in range(1, k + 1):
        total_subsequences += combination(n, j)
        total_subsequences %= MOD

    return total_sum * 2 % MOD


def combination(n: int, k: int) -> int:
    if k > n:
        return 0
    if k == 0 or k == n:
        return 1
    k = min(k, n - k)  # Take advantage of symmetry

    numer = 1
    denom = 1
    for i in range(k):
        numer = (numer * (n - i)) % MOD
        denom = (denom * (i + 1)) % MOD

    return (numer * pow(denom, MOD - 2, MOD)) % MOD
