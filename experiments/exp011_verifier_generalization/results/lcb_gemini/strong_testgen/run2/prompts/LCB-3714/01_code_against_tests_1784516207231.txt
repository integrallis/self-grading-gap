```python
from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Count how many times each number can be the min and max
    for i in range(n):
        count_min = 0
        count_max = 0

        # For each number, calculate how many times it can be the min
        for j in range(1, k + 1):
            if i - j + 1 >= 0:
                count_min += 1
            if n - (i + 1) >= j - 1:
                count_max += 1

        total_sum += (nums[i] * (count_min + count_max)) % MOD
        total_sum %= MOD

    return total_sum * 2 % MOD

```
