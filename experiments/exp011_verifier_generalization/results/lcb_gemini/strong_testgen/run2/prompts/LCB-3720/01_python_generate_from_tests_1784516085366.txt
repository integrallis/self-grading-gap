from typing import List

def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    edges.sort(key=lambda x: x[2])
    left, right = 1, 10**6
    
    def can_reach_zero(max_weight: int) -> bool:
        graph = [[] for _ in range(n)]
        out_degree = [0] * n
        
        for u, v, w in edges:
            if w <= max_weight:
                graph[u].append(v)
                out_degree[u] += 1
                if out_degree[u] > threshold:
                    return False
        
        reachable = [False] * n
        stack = [0]
        while stack:
            node = stack.pop()
            if reachable[node]:
                continue
            reachable[node] = True
            for neighbor in graph[node]:
                if not reachable[neighbor]:
                    stack.append(neighbor)
        
        return all(reachable[i] for i in range(1, n))

    while left < right:
        mid = (left + right) // 2
        if can_reach_zero(mid):
            right = mid
        else:
            left = mid + 1

    return left if can_reach_zero(left) else -1

```
