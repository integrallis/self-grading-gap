```python
from typing import List

def maxSum(nums: List[int]) -> int:
    unique_elements = set()
    max_sum = float('-inf')
    current_sum = 0
    for num in nums:
        if num in unique_elements:
            # Reset the current sum if we encounter a duplicate
            current_sum = num
            unique_elements = {num}
        else:
            current_sum += num
            unique_elements.add(num)
        max_sum = max(max_sum, current_sum)
    return max(max_sum, max(nums))
```
