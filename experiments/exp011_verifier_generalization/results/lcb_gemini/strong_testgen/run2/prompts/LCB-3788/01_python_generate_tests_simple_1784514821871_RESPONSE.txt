def test_canonical_example_1():
    """
    Tests the first example from the specification.
    Input: nums = [1,2,3,4,5] -> Output: 15
    """
    # spec: maxSum([1,2,3,4,5]) -> 15
    # Explanation: Select the entire array without deleting any element to obtain the maximum sum.
    # The subarray [1,2,3,4,5] has unique elements and sums to 15. All elements are positive and unique.
    # The sum of unique positive numbers is 1+2+3+4+5 = 15.
    assert maxSum([1, 2, 3, 4, 5]) == 15


def test_canonical_example_2():
    """
    Tests the second example from the specification.
    Input: nums = [1,1,0,1,1] -> Output: 1
    """
    # spec: maxSum([1,1,0,1,1]) -> 1
    # Explanation: Delete elements to leave [1]. The maximum sum from a subarray with unique elements is 1.
    # The unique numbers are {0, 1}. The unique positive number is {1}. The sum is 1.
    assert maxSum([1, 1, 0, 1, 1]) == 1


def test_canonical_example_3():
    """
    Tests the third example from the specification.
    Input: nums = [1,2,-1,-2,1,0,-1] -> Output: 3

    # spec: maxSum([1,2,-1,-2,1,0,-1]) -> 3
    # Explanation: After deletions, we could have a sequence like [2, 1], which has unique elements and sums to 3.
    # The unique numbers are {-2, -1, 0, 1, 2}. The unique positive numbers are {1, 2}. Their sum is 1 + 2 = 3.
