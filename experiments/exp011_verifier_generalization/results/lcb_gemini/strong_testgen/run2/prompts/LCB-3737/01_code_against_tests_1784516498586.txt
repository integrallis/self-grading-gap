```python
from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # Dynamic programming table for storing minimum costs
    dp = [[0] * 3 for _ in range(n)]
    # Initialize the first house costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the dp table
    for i in range(1, n):
        for j in range(3):
            # Choose the minimum cost for painting house i with color j
            # Ensure we do not choose the same color as the previous house
            dp[i][j] = cost[i][j] + min(dp[i-1][(j+1) % 3], dp[i-1][(j+2) % 3])

    # Handle the equidistant condition
    min_cost = float('inf')
    for j in range(3):
        cost_with_last_color = dp[n-1][j]
        # Avoid color conflicts for equidistant houses
        for k in range(3):
            if j != k:  # ensure no two adjacent are the same
                cost_with_last_color += min(dp[n-2][(k+1) % 3], dp[n-2][(k+2) % 3])
        min_cost = min(min_cost, cost_with_last_color)

    return min_cost
```
