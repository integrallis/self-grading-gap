from typing import List
import itertools


def totalNumbers(digits: List[int]) -> int:
    even_digits = [d for d in digits if d % 2 == 0]
    if not even_digits:
        return 0
    distinct_numbers = set()
    for last_digit in even_digits:
        for first_digit in set(digits) - {0}:
            if first_digit != last_digit:
                for middle_digit in set(digits) - {first_digit, last_digit}:
                    number = first_digit * 100 + middle_digit * 10 + last_digit
                    distinct_numbers.add(number)
                if digits.count(last_digit) > 1:
                    middle_digit = last_digit
                    number = first_digit * 100 + middle_digit * 10 + last_digit
                    distinct_numbers.add(number)
    return len(distinct_numbers)
```
