from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    gaps = []

    # Calculate initial gaps
    if startTime[0] > 0:
        gaps.append(startTime[0])  # Free time before the first meeting
    for i in range(1, n):
        gaps.append(startTime[i] - endTime[i - 1])  # Free time between meetings
    if endTime[-1] < eventTime:
        gaps.append(eventTime - endTime[-1])  # Free time after the last meeting

    # If no meetings can be rescheduled, return the maximum initial free time
    if k == 0:
        return max(gaps) if gaps else eventTime

    # If k is large enough to merge all meetings, calculate total duration
    total_duration = sum(endTime[i] - startTime[i] for i in range(n))
    if k >= n - 1:
        return eventTime - total_duration

    # Sort gaps to find the largest possible free time after rescheduling
    gaps.sort(reverse=True)
    # Take the largest k gaps and sum them
    max_free_time = sum(gaps[:k]) if k <= len(gaps) else sum(gaps)
    return max(max_free_time, max(gaps))
