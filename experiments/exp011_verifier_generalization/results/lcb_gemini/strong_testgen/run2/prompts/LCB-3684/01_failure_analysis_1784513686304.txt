def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]
    prefix_length = len(prefix)
    suffix_length = len(suffix)
    # Iterate through the string to find a valid contiguous match
    for i in range(len(s) - prefix_length + 1):
        if s[i:i + prefix_length] == prefix:
            # Check if a valid suffix can be found after the prefix
            if s[i + prefix_length:].startswith(suffix):
                return True
            # Check if the suffix can be found anywhere after the prefix
            for j in range(i + prefix_length, len(s) - suffix_length + 1):
                if s[j:j + suffix_length] == suffix:
                    return True
    return False
