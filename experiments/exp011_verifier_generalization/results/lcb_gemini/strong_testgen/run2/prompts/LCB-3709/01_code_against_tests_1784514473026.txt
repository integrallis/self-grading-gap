```python
def hasSpecialSubstring(s: str, k: int) -> bool:
    n = len(s)
    if k > n:
        return False
    for i in range(n - k + 1):
        substring = s[i:i + k]
        if len(set(substring)) == 1:
            left_neighbor = s[i - 1] if i > 0 else None
            right_neighbor = s[i + k] if i + k < n else None
            if (left_neighbor is not None and left_neighbor == substring[0]):
                continue
            if (right_neighbor is not None and right_neighbor == substring[0]):
                continue
            return True
    return False
```
