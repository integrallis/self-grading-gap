def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a dictionary to hold coin counts at each bag position
    bag_coins = {}
    for l, r, c in coins:
        for pos in range(l, r + 1):
            bag_coins[pos] = bag_coins.get(pos, 0) + c

    # Create a sorted list of positions
    positions = sorted(bag_coins.keys())
    total_bags = len(positions)
    max_coins = 0

    # Sliding window to find the maximum coins in k consecutive bags
    left = 0
    current_coins = 0

    for right in range(total_bags):
        current_coins += bag_coins[positions[right]]

        # Move the left end of the window if the size exceeds k
        while positions[right] - positions[left] + 1 > k:
            current_coins -= bag_coins[positions[left]]
            left += 1

        # Update maximum coins
        max_coins = max(max_coins, current_coins)
