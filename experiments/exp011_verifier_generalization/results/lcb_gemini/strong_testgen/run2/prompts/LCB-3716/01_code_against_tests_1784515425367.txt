```python
from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = [1] * n  # dp[i] will be the length of the longest subsequence ending at index i
    for i in range(n):
        for j in range(i):
            # Check for valid subsequences and maintain non-increasing absolute differences
            if dp[j] + 1 > dp[i]:
                if not (len(dp[:j]) > 1 and abs(nums[j] - nums[j-1]) < abs(nums[i] - nums[j])):
                    dp[i] = max(dp[i], dp[j] + 1)
    return max(dp)
```
