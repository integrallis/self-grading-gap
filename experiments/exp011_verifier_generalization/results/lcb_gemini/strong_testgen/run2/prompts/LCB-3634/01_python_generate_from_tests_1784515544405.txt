```python
def calculateScore(s: str) -> int:
    score = 0
    marked = set()
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # Create a mirror mapping for 'a' to 'z'

    for i in range(len(s)):
        if i in marked:
            continue
        current_char = s[i]
        mirror_char = mirror[current_char]

        # Find the closest unmarked index j for the mirror character
        for j in range(i - 1, -1, -1):
            if j in marked:
                continue
            if s[j] == mirror_char:
                score += i - j
                marked.add(i)
                marked.add(j)
                break

    return score

```
