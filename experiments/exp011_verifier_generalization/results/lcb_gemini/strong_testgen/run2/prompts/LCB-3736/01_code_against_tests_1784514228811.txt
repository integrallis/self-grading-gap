```python
from collections import Counter

def findValidPair(s: str) -> str:
    counts = Counter(s)
    for i in range(len(s) - 1):
        if s[i] != s[i + 1]:  # Check if adjacent digits are different
            first_digit = int(s[i])
            second_digit = int(s[i + 1])
            if counts[s[i]] == first_digit and counts[s[i + 1]] == second_digit:
                return s[i] + s[i + 1]  # Return the valid pair
    return ""  # Return empty string if no valid pair is found
```
