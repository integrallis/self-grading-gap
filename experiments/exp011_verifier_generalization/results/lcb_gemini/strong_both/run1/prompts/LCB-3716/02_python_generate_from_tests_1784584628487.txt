from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = [1] * n
    # To keep track of the differences
    for i in range(n):
        for j in range(i):
            diff1 = abs(nums[i] - nums[j])
            # Check all previous subsequences ending at j
            if dp[j] > 1:
                for k in range(j):
                    diff2 = abs(nums[j] - nums[k])
                    if diff1 >= diff2:
                        dp[i] = max(dp[i], dp[j] + 1)
                    
    return max(dp)
```
