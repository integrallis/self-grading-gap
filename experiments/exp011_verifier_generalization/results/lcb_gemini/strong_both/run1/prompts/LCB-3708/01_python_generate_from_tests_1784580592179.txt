from typing import List

def zigzagTraversal(grid: List[List[int]]) -> List[int]:
    result = []
    for i in range(len(grid)):
        if i % 2 == 0:
            # Traverse right in even rows
            result.extend(grid[i][::2])  # Keep cells at even indices
        else:
            # Traverse left in odd rows
            result.extend(grid[i][::-1][::2])  # Keep cells at even indices of the reversed row
    return result

```
