```python
from typing import List

def maxAdjacentDistance(nums: List[int]) -> int:
    max_distance = 0
    n = len(nums)
    for i in range(n):
        # Calculate the absolute difference between adjacent elements
        # Include the circular difference between the last and first elements
        diff = abs(nums[i] - nums[(i + 1) % n])
        # Update max_distance if the current difference is greater
        if diff > max_distance:
            max_distance = diff
    return max_distance

```
