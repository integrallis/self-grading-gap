## Implementation

```python
def hasMatch(s: str, p: str) -> bool:
    # Split the pattern at the '*'
    prefix, suffix = p.split('*')
    # Check if there exists a substring in 's' that starts with 'prefix' and ends with 'suffix'
    for i in range(len(s) - len(prefix) - len(suffix) + 1):
        if s[i:i + len(prefix)] == prefix:
            # Check if the suffix can be found after the prefix
            if s[i + len(prefix):].endswith(suffix):
                return True
    return False

```
