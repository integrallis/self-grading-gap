"""
Test suite for the hasMatch function.

The hasMatch function checks if a pattern `p` with a single wildcard '*'
can be formed into a substring of a string `s`.

Specification:
- hasMatch(s: str, p: str) -> bool
- `s` and `p` are strings.
- `p` contains exactly one '*' character.
- The '*' in `p` can be replaced with any sequence of zero or more characters.
- Returns `true` if the modified `p` can be made a substring of `s`, `false` otherwise.
- A substring is a contiguous non-empty sequence of characters.

Constraints:
- 1 <= s.length <= 50
- 1 <= p.length <= 50
- `s` contains only lowercase English letters.
- `p` contains only lowercase English letters and exactly one '*'.
"""

from typing import List
import pytest
from solution import hasMatch


def test_canonical_example_1():
    """
    Tests the first example from the specification.
    s = "leetcode", p = "ee*e" -> true
    """
    # spec: "By replacing the '*' with "tcod", the substring "eetcode" matches the pattern."
    # The substring "eetcode" is in "leetcode", starts with "ee", and ends with "e".
    assert hasMatch("leetcode", "ee*e") is True


def test_canonical_example_2():
    """
    Tests the second example from the specification.
    s = "car", p = "c*v" -> false
    """
    # spec: "There is no substring matching the pattern."
    # No substring of "car" starts with 'c' and ends with 'v'.
    assert hasMatch("car", "c*v") is False


def test_canonical_example_3():
    """
    Tests the third example from the specification.
    s = "luck", p = "u*" -> true
    """
    # spec: "The substrings "u", "uc", and "uck" match the pattern."
    # p has prefix "u" and an empty suffix. Substrings "u", "uc", "uck" all start with "u".
    assert hasMatch("luck", "u*") is True
