import pytest
from typing import List

# Import the function to be tested
from solution import sumOfGoodNumbers


def test_canonical_example_1():
    """
    Tests the first example from the specification.
    """
    nums = [1, 3, 2, 1, 5, 4]
    k = 2
    # Explanation from spec:
    # nums[1]=3: i=1, k=2. i-k is out of bounds. i+k=3. nums[1]>nums[3] (3>1). Good.
    # nums[4]=5: i=4, k=2. i-k=2. i+k is out of bounds. nums[4]>nums[2] (5>2). Good.
    # nums[5]=4: i=5, k=2. i-k=3. i+k is out of bounds. nums[5]>nums[3] (4>1). Good.
    # Others are not good:
    # nums[0]=1: i=0, k=2. i+k=2. nums[0]>nums[2] (1>2) is False.
    # nums[2]=2: i=2, k=2. i-k=0, i+k=4. nums[2]>nums[4] (2>5) is False.
    # nums[3]=1: i=3, k=2. i-k=1, i+k=5. nums[3]>nums[1] (1>3) is False.
    # Sum of good numbers = 3 + 5 + 4 = 12.
    assert sumOfGoodNumbers(nums, k) == 12


def test_canonical_example_2():

    # nums[1]=1: i=1, k=1. i-k=0. i+k is out of bounds. nums[1]>nums[0] (1>2) is False.

def test_no_good_numbers_due_to_equality():
    """
    Tests a case where no numbers are good because they are not strictly greater.
    """
    nums = [5, 5, 5, 5, 5]
    k = 2
    # For every element nums[i], its neighbors at i-k or i+k (if they exist)
    # have the same value, 5. The condition is "strictly greater", so nums[i] > 5
    # is always false.
    # Sum = 0.
    assert sumOfGoodNumbers(nums, k) == 0


def test_no_good_numbers_in_increasing_sequence():
    """
    Tests a case where most numbers are not good because they are smaller than their right neighbor.
    """
    nums = [10, 20, 30, 40, 50]
    k = 1
    # nums[0]=10: check nums[1]=20. 10 > 20 is False.
    # nums[1]=20: check nums[0]=10, nums[2]=30. 20 > 30 is False.
    # nums[2]=30: check nums[1]=20, nums[3]=40. 30 > 40 is False.
    # nums[3]=40: check nums[2]=30, nums[4]=50. 40 > 50 is False.
    # nums[4]=50: check nums[3]=40. 50 > 40 is True. Good.
    # Sum = 50.
    assert sumOfGoodNumbers(nums, k) == 50


def test_internal_peaks_are_good():
    """
    Tests elements that are good because they are greater than both neighbors at distance k.
    """
    nums = [1, 10, 2, 20, 3]
    k = 1
    # nums[0]=1: check nums[1]=10. 1 > 10 is False.
    # nums[1]=10: check nums[0]=1 and nums[2]=2. 10 > 1 and 10 > 2. Good.
    # nums[2]=2: check nums[1]=10 and nums[3]=20. 2 > 10 is False.
    # nums[3]=20: check nums[2]=2 and nums[4]=3. 20 > 2 and 20 > 3. Good.
    # nums[4]=3: check nums[3]=20. 3 > 20 is False.
    # Sum = 10 + 20 = 30.
    assert sumOfGoodNumbers(nums, k) == 30


def test_max_k_for_even_length_array():
    """
    Tests the behavior at the maximum allowed value of k for an even-length array.
    k = floor(nums.length / 2)
    """
    nums = [10, 1, 2, 3, 4, 5]
    k = 3  # floor(6 / 2) = 3
    # With k=3, no element has both i-k and i+k neighbors.
    # nums[0]=10: check nums[3]=3. 10 > 3. Good.
    # nums[1]=1: check nums[4]=4. 1 > 4 is False.
    # nums[2]=2: check nums[5]=5. 2 > 5 is False.
    # nums[3]=3: check nums[0]=10. 3 > 10 is False.
    # nums[4]=4: check nums[1]=1. 4 > 1. Good.
    # nums[5]=5: check nums[2]=2. 5 > 2. Good.
    # Sum = 10 + 4 + 5 = 19.
    assert sumOfGoodNumbers(nums, k) == 19


def test_max_k_for_odd_length_array():
    """
    Tests the behavior at the maximum allowed value of k for an odd-length array.
    k = floor(nums.length / 2)
    """
    nums = [10, 2, 8, 4, 9, 1, 12]
    k = 3  # floor(7 / 2) = 3
    # nums[0]=10: check nums[3]=4. 10 > 4. Good.
    # nums[1]=2: check nums[4]=9. 2 > 9 is False.
    # nums[2]=8: check nums[5]=1. 8 > 1. Good.
    # nums[3]=4: check nums[0]=10 and nums[6]=12. 4 > 10 is False.
    # nums[4]=9: check nums[1]=2. 9 > 2. Good.
    # nums[5]=1: check nums[2]=8. 1 > 8 is False.
    # nums[6]=12: check nums[3]=4. 12 > 4. Good.
    # Sum = 10 + 8 + 9 + 12 = 39.
    assert sumOfGoodNumbers(nums, k) == 39
