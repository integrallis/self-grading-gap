from typing import List

MOD = 10**9 + 7

from math import comb

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    nums.sort()
    total_sum = 0

    for i in range(n):
        # Calculate contributions as min
        min_coeff_sum = sum(comb(n - 1 - i, j - 1) for j in range(1, k + 1))
        # Calculate contributions as max
        max_coeff_sum = sum(comb(i, j - 1) for j in range(1, k + 1))
        
        term = (nums[i] * (min_coeff_sum + max_coeff_sum)) % MOD
        total_sum = (total_sum + term) % MOD

    return total_sum % MOD

```
