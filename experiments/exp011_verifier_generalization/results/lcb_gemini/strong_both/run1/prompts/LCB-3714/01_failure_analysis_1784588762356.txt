from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    nums.sort()
    total_sum = 0
    
    # Precompute the sums of combinations
    sums_of_combs = [[0] * (k + 1) for _ in range(n)]
    for i in range(n):
        sums_of_combs[i][0] = 1  # C(i, 0)
    for j in range(1, k + 1):
        sums_of_combs[0][j] = 1  # C(0, j) = 0 except C(0, 0)
    
    for i in range(1, n):
        for j in range(1, k + 1):
            sums_of_combs[i][j] = (sums_of_combs[i-1][j] + sums_of_combs[i-1][j-1]) % MOD

    for i in range(n):
        # Calculate contributions as min
        min_coeff_sum = sums_of_combs[n - 1 - i][k - 1]
        # Calculate contributions as max
        max_coeff_sum = sums_of_combs[i][k - 1]
        
        term = (nums[i] * (min_coeff_sum + max_coeff_sum)) % MOD
        total_sum = (total_sum + term) % MOD

    return total_sum % MOD
