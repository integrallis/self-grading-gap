from typing import List

MOD = 10**9 + 7

from math import comb

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    nums.sort()
    total_sum = 0

    for i in range(n):
        # Calculate contributions as min
        for j in range(1, k + 1):
            total_sum += nums[i] * comb(n - 1 - i, j - 1)
            total_sum %= MOD
        # Calculate contributions as max
        for j in range(1, k + 1):
            total_sum += nums[i] * comb(i, j - 1)
            total_sum %= MOD

    return total_sum % MOD

```
