```python
from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize two DP arrays for the two halves of the houses
    dp1 = [[0] * 3 for _ in range(n // 2)]
    dp2 = [[0] * 3 for _ in range(n // 2)]

    # Fill the first half
    for j in range(3):
        dp1[0][j] = cost[0][j]

    for i in range(1, n // 2):
        for j in range(3):
            dp1[i][j] = cost[i][j] + min(dp1[i - 1][(j + 1) % 3], dp1[i - 1][(j + 2) % 3])

    # Fill the second half
    for j in range(3):
        dp2[0][j] = cost[n // 2][j]

    for i in range(1, n // 2):
        for j in range(3):
            dp2[i][j] = cost[n // 2 + i][j] + min(dp2[i - 1][(j + 1) % 3], dp2[i - 1][(j + 2) % 3])

    # Calculate the total minimum cost
    result = float('inf')
    for j1 in range(3):
        for j2 in range(3):
            if j1 != j2:  # Ensure colors from both halves are different
                cost_combination = (dp1[n // 2 - 1][j1] + dp2[n // 2 - 1][j2])
                result = min(result, cost_combination)

    return result
```
