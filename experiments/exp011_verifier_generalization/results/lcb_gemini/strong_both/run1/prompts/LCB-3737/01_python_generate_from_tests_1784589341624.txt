from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    dp = [[0] * 3 for _ in range(n)]

    # Fill the dp array for the first house
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the dp array for the remaining houses
    for i in range(1, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(dp[i - 1][(j + 1) % 3], dp[i - 1][(j + 2) % 3])

    # Calculate the total minimum cost ensuring equidistant pairs are handled
    result = float('inf')
    for j1 in range(3):
        for j2 in range(3):
            if j1 != j2:
                cost_combination = dp[n - 1][j1] + dp[n - 2][j2]
                result = min(result, cost_combination)

    return result
```
