from typing import List

def minimumPairRemoval(nums: List[int]) -> int:
    operations = 0
    while True:
        # Find the pairs with their sums
        pairs = [(nums[i] + nums[i + 1], i) for i in range(len(nums) - 1)]
        # Find the pair with the minimum sum
        min_sum, min_index = min(pairs)
        # If the array is already non-decreasing, break
        if all(nums[i] <= nums[i + 1] for i in range(len(nums) - 1)):
            break
        # Merge the leftmost pair with the minimum sum
        nums[min_index] = min_sum
        del nums[min_index + 1]  # Remove the second element of the pair
        operations += 1
    return operations
```
