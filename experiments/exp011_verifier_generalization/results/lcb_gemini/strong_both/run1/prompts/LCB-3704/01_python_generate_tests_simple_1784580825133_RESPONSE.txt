def test_max_length_array_even_sum():
    """
    spec: For nums=[2]*100 (n=100), there are 99 possible partitions.
    The total sum is 200 (even).
    For any partition i, left_sum is sum of even numbers, so it's even.
    right_sum is also sum of even numbers, so it's even.
    The difference of two even numbers is always even.
    Thus, all 99 partitions result in an even sum difference.
    """
    nums = [2] * 100
    assert countPartitions(nums) == 99
