```python
def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    for i in range(len(s)):
        if s[i] == 'N':
            y += 1
        elif s[i] == 'S':
            y -= 1
        elif s[i] == 'E':
            x += 1
        elif s[i] == 'W':
            x -= 1
        current_distance = abs(x) + abs(y)
        max_distance = max(max_distance, current_distance)
        distance_with_changes = min(i + 1, current_distance + 2 * k)
        max_distance = max(max_distance, distance_with_changes)
    return max_distance

```
