```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a mapping of bags to coin counts
    coin_map = {}
    for l, r, c in coins:
        for i in range(l, r + 1):
            coin_map[i] = coin_map.get(i, 0) + c

    # List of coin counts arranged by bag position
    bags = sorted(coin_map.keys())
    max_coins = 0
    current_coins = 0
    left = 0

    # Sliding window to find max coins in k consecutive bags
    for right in range(len(bags)):
        current_coins += coin_map[bags[right]]
        while bags[right] - bags[left] + 1 > k:
            current_coins -= coin_map[bags[left]]
            left += 1
        max_coins = max(max_coins, current_coins)

    return max_coins

```
