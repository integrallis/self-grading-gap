def test_window_fully_within_one_long_segment():
    # A single segment that is longer than k.
    # The optimal window will be entirely within this segment.
    coins = [[5, 15, 10]]
    k = 3
    # A window of size 3, e.g., [5, 7], collects 10 + 10 + 10 = 30 coins.
    expected = 30
    assert maximumCoins(coins, k) == expected


def test_window_straddles_gap_between_segments():
    # The optimal window bridges a gap between two segments.
    coins = [[1, 2, 10], [10, 11, 20]]
    k = 5
    # The window [1, 5] would collect 10 (at 1) + 10 (at 2) = 20.
    # The window [7, 11] would collect 20 (at 10) + 20 (at 11) = 40.
    # Any other window yields a smaller or equal sum.
    expected = 40
    assert maximumCoins(coins, k) == expected


def test_k_is_one_max_coin_value():
    # When k=1, the result is simply the maximum coin value in any single bag.
    coins = [[1, 2, 10], [100, 101, 20]]
    k = 1
    # The maximum value in a single bag is 20.
    expected = 20
    assert maximumCoins(coins, k) == expected


def test_k_is_large_covers_all_coins():
    # When k is large enough to span all segments, the result is the sum of all available coins.
    coins = [[1, 2, 10], [100, 101, 20]]
    k = 200
    # Total coins = (2 coins of value 10) + (2 coins of value 20) = 2*10 + 2*20 = 60.
    # A window of size 200, e.g., [1, 200], can collect all of these.
    expected = 60
    assert maximumCoins(coins, k) == expected


def test_point_segments():
    # Segments can be single points (l_i == r_i).
    coins = [[1, 1, 100], [5, 5, 200]]
    k = 3
    # A window of size 3, like [3, 5], captures bag 5, which has 200 coins.
    # Window [1, 3] captures bag 1 (100 coins).
    # The maximum is 200.
    expected = 200
    assert maximumCoins(coins, k) == expected


def test_adjacent_segments():
    # Test with segments that are next to each other.
    coins = [[1, 3, 5], [4, 6, 10]]
    k = 4
    # The window [3, 6] collects:
    # bag 3 (5 coins) + bag 4 (10 coins) + bag 5 (10 coins) + bag 6 (10 coins) = 35.
    expected = 35
    assert maximumCoins(coins, k) == expected


def test_single_segment_k_equals_length():
    # k is the exact length of the single coin segment.
    coins = [[10, 14, 8]]
    k = 5
    # The segment length is 14 - 10 + 1 = 5.
    # The optimal window [10, 14] collects all coins: 5 bags * 8 coins/bag = 40.
    expected = 40
    assert maximumCoins(coins, k) == expected


def test_single_segment_k_larger_than_length():
    # k is larger than the single coin segment's length.
    coins = [[10, 14, 8]]
    k = 7
    # The segment length is 5. Any window of size 7 containing [10, 14]
    # will collect all coins from the segment. Total is 5 * 8 = 40.
    expected = 40
    assert maximumCoins(coins, k) == expected
