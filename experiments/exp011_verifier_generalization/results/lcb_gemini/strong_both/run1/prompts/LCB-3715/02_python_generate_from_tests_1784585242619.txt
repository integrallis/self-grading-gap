from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a coins_count array to map coin values to a range of bags
    max_pos = 0
    for l, r, c in coins:
        max_pos = max(max_pos, r)

    # Initialize the counts for each position on the number line
    coins_count = [0] * (max_pos + 2)  # +2 to handle r+1 indexing safely

    # Fill the coins_count based on the input segments
    for l, r, c in coins:
        coins_count[l] += c
        coins_count[r + 1] -= c

    # Compute the prefix sum to get the actual number of coins at each bag position
    for i in range(1, len(coins_count)):
        coins_count[i] += coins_count[i - 1]

    # Find the maximum coins obtainable in any k-length window
    max_coins = 0
    current_coins = sum(coins_count[1:k + 1])  # Initial sum for the first window
    max_coins = current_coins

    # Slide the window across the coins_count array
    for i in range(2, max_pos + 2):
        current_coins += coins_count[i] - coins_count[i - k]
        max_coins = max(max_coins, current_coins)

    return max_coins
```
