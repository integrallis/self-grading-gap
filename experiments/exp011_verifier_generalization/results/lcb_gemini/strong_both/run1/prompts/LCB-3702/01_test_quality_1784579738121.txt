def _is_product_equivalent(arr: List[int]) -> bool:
    """
    Checks if an array is product equivalent based on the prime factorization property.
    This helper is for reasoning about test cases, not for the implementation.
    A subarray `arr` is product equivalent if for each prime `p`, at most one
    number in `arr` is divisible by `p`.
    """
    if not arr:
        return False
    
    primes = [2, 3, 5, 7]
    for p in primes:
        count = 0
        for x in arr:
            if x > 1 and x % p == 0:
                count += 1
        if count > 1:
            return False
    return True
