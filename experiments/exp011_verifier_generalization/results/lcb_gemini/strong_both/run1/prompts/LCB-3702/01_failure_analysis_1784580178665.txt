# This property holds for ANY array of length 1 or 2.
# For an array `arr` of length > 2, it holds if and only if for every prime `p`,
# at most one number in `arr` is divisible by `p`.

    # spec: maxLength([1,2,1,2,1,1,1]) -> 5
    # The longest PE subarray is [1, 2, 1, 1, 1], length 5.
    # It is PE because it has length > 2, and for each prime, at most one element is divisible by it (only '2' is divisible by 2).
    # Any longer subarray (e.g., [1,2,1,2,1,1]) contains two '2's, so it is not PE.
    assert maxLength([1,2,1,2,1,1,1]) == 5

def test_canonical_example_2():
    # spec: maxLength([2,3,4,5,6]) -> 3
    # The longest PE subarray is [3, 4, 5], length 3.
    # It is PE: p=2 only divides 4, p=3 only divides 3, p=5 only divides 5.
    # Subarrays of length 4 are not PE:
    # [2,3,4,5]: not PE (2 and 4 are divisible by 2).

    assert maxLength([2,3,4,5,6]) == 3

from typing import List
import math


def gcd(a: int, b: int) -> int:
    while b:
        a, b = b, a % b
    return a


def lcm(a: int, b: int) -> int:
    return a * b // gcd(a, b)


def maxLength(nums: List[int]) -> int:
    max_length = 0
    n = len(nums)

    for start in range(n):
        prime_count = {}
        for end in range(start, n):
            num = nums[end]
            for p in [2, 3, 5, 7]:
                if num % p == 0:
                    if p in prime_count:
                        prime_count[p] += 1
                    else:
                        prime_count[p] = 1

            if len(prime_count) > 0 and any(count > 1 for count in prime_count.values()):
                break

            max_length = max(max_length, end - start + 1)

    return max_length
