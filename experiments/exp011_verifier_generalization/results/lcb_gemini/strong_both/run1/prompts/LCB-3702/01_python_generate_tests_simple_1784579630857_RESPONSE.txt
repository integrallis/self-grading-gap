from solution import maxLength

# Helper function for test case reasoning. This is NOT part of the solution.
# It's used here in comments to justify expected results.
def _is_product_equivalent(arr: List[int]) -> bool:
    """
    Checks if an array is product equivalent based on the prime factorization property.
    This helper is for reasoning about test cases, not for the implementation.
    A subarray `arr` is product equivalent if for each prime `p`, at most one
    number in `arr` is divisible by `p`.
    """
    if not arr:
        return False
    
    primes = [2, 3, 5, 7]
    for p in primes:
        count = 0
        for x in arr:
            if x > 1 and x % p == 0:
                count += 1
        if count > 1:
            return False
    return True

def test_canonical_example_1():
    # spec: maxLength([1,2,1,2,1,1,1]) -> 5
    # The longest product equivalent subarray is [1, 2, 1, 1, 1].
    # In [1, 2, 1, 1, 1], for any prime p, at most one number is divisible by p.
    # The whole array [1,2,1,2,1,1,1] is not product equivalent because two '2's are divisible by 2.
    # Subarrays of length 6 like [1,2,1,2,1,1] also contain two '2's.
    assert maxLength([1,2,1,2,1,1,1]) == 5

def test_canonical_example_2():
    # spec: maxLength([2,3,4,5,6]) -> 3
    # The longest product equivalent subarray is [3, 4, 5].
    # For [3, 4, 5]: p=2 only divides 4, p=3 only divides 3, p=5 only divides 5.
    # Subarrays of length 4:
    # [2,3,4,5]: not PE (2,4 are div by 2)
    # [3,4,5,6]: not PE (4,6 are div by 2; 3,6 are div by 3)
    assert maxLength([2,3,4,5,6]) == 3

def test_canonical_example_3():
    # spec: maxLength([1,2,3,1,4,5,1]) -> 5
    # The longest product equivalent subarray is [3, 1, 4, 5, 1].
    # For [3,1,4,5,1]: p=2 only divides 4, p=3 only divides 3, p=5 only divides 5.
    # Any longer subarray includes 2, e.g. [1,2,3,1,4,5], which is not PE (2,4 are div by 2).
    assert maxLength([1,2,3,1,4,5,1]) == 5

def test_all_ones():
    # An array of only 1s is always product equivalent.
    # prod([1,1,1]) = 1, gcd = 1, lcm = 1. 1 == 1 * 1.
    # The longest subarray is the array itself.
    nums = [1, 1, 1, 1, 1, 1, 1]
    assert maxLength(nums) == 7

def test_distinct_primes():
    # An array of distinct primes (and 1) is product equivalent.
    # For [2, 3, 5, 7], no two numbers share a prime factor.
    # The longest subarray is the array itself.
    nums = [2, 3, 5, 7]
    assert maxLength(nums) == 4

def test_all_incompatible_even_numbers():
    # An array where all numbers are divisible by 2.
    # Any subarray of length > 1 will have at least two numbers divisible by 2.
    # The longest product equivalent subarrays are single elements.
    nums = [2, 4, 6, 8, 10, 2]
    assert maxLength(nums) == 1

def test_all_incompatible_multiples_of_3():
    # An array where all numbers are divisible by 3.
    # Similar to the even numbers case, longest PE subarray has length 1.
    nums = [3, 6, 9, 3]
    assert maxLength(nums) == 1

def test_longest_at_start():
    # The longest PE subarray is at the beginning.
    # [2, 3, 5] is PE. Length 3.
    # [2, 3, 5, 6] is not PE (2,6 div by 2; 3,6 div by 3).
    nums = [2, 3, 5, 6, 8]
    assert maxLength(nums) == 3

def test_longest_at_end():
    # The longest PE subarray is at the end.
    # [5, 3, 2] is PE. Length 3.

    nums = [8, 6, 5, 3, 2]

def test_long_pe_subarray_in_middle():
    # A case with a long PE subarray in the middle, surrounded by incompatible numbers.
    # Subarray [8, 1, 3, 5, 7, 1] is PE. Length 6.
    # _is_product_equivalent([8, 1, 3, 5, 7, 1]) is True.
    # [6, 8, 1, 3, 5, 7, 1] is not PE (6,8 div by 2).
    # [8, 1, 3, 5, 7, 1, 9] is not PE (3,9 div by 3).
    nums = [6, 8, 1, 3, 5, 7, 1, 9, 4]
    assert maxLength(nums) == 6

    # Minimum length input (2) which is a PE array.

    nums = [7, 8]
    assert maxLength(nums) == 2

def test_minimum_length_input_not_pe():
    # Minimum length input (2) which is not a PE array.
    # [6, 9] is not PE: both are divisible by 3.
    # Longest PE subarrays are [6] and [9], length 1.
    nums = [6, 9]
    assert maxLength(nums) == 1

def test_complex_case_with_composite_numbers():
    # A complex case demonstrating the prime factor rule.
    # The longest PE subarray is [4, 1, 9, 5, 1, 7]. Length 6.
    # In this subarray: only 4 is div by 2, only 9 by 3, only 5 by 5, only 7 by 7.
    # Any extension breaks the property:
    # [3, 4, 1, 9, 5, 1, 7] is not PE (3,9 div by 3).
    # [4, 1, 9, 5, 1, 7, 10] is not PE (4,10 div by 2; 5,10 div by 5).
    nums = [3, 4, 1, 9, 5, 1, 7, 10]
    assert maxLength(nums) == 6

    nums = [10, 1, 4, 9, 1, 6, 8]
