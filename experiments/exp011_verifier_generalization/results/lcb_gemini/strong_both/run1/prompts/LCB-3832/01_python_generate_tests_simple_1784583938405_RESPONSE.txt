from typing import List
import pytest
from solution import minCosts


def test_canonical_example_1():
    """
    Tests the first example from the specification.
    """
    cost = [5, 3, 4, 1, 3, 2]
    # spec: minCosts([5, 3, 4, 1, 3, 2]) -> [5, 3, 3, 1, 1, 1]
    # The minimum cost to reach position i is the minimum of cost[0...i].
    # i=0: min(5) = 5
    # i=1: min(5, 3) = 3
    # i=2: min(5, 3, 4) = 3
    # i=3: min(5, 3, 4, 1) = 1
    # i=4: min(5, 3, 4, 1, 3) = 1
    # i=5: min(5, 3, 4, 1, 3, 2) = 1
    expected = [5, 3, 3, 1, 1, 1]
    assert minCosts(cost) == expected

def test_canonical_example_2():
    """

    # Since cost[0] is 1, the minimum prefix cost is always 1.
    # i=0: min(1) = 1
