from typing import List


def maxFrequency(nums: List[int], k: int) -> int:
    count_k = nums.count(k)
    max_freq = count_k
    gain = [0] * len(nums)

    for i in range(len(nums)):
        gain[i] = (k - nums[i])

    max_gain = 0
    current_gain = 0
    for g in gain:
        current_gain += g
        if current_gain < 0:
            current_gain = 0
        max_gain = max(max_gain, current_gain)

    return max_freq + max_gain

```
