def test_canonical_example_1():
    """
    Tests the first example from the specification.
    spec: nums = [1,2,3,4,5,6], k = 1 -> 2
    Explanation: Add x=-5 to nums[2..5] ([3,4,5,6]).
    The array becomes [1, 2, -2, -1, 0, 1]. The value 1 now appears twice.
    This is achieved by targeting nums[5]=6 to become 1.
    """
    nums = [1, 2, 3, 4, 5, 6]
    k = 1

    assert maxFrequency(nums, k) == expected


def test_canonical_example_2():
    """

    The original nums[1], nums[8], nums[9] were all 2. They become 10.

    The value 10 now appears four times (original at index 0, plus three new ones).

    expected = 4

def test_single_element_array_is_k():
    """
    Tests an array with a single element which is already k.
    The frequency is 1, and no operation can increase it.
    """
    nums = [10]
    k = 10
    # The only element is k, frequency is 1. No operation can improve this.
    expected = 1
    assert maxFrequency(nums, k) == expected


def test_single_element_array_not_k():
    """
    Tests an array with a single element that can be changed to k.
    The operation is to add (k - nums[0]) to the whole array (nums[0..0]).
    """
    nums = [10]
    k = 20
    # Add 10 to nums[0..0] to make it [20]. Frequency of 20 becomes 1.
    expected = 1
    assert maxFrequency(nums, k) == expected


def test_all_elements_are_k():
    """
    Tests a case where all elements are already k.
    Any operation on a non-empty subarray will change some elements from k to something else,
    reducing the frequency. The best option is a no-op (e.g., add x=0).
    """
    nums = [7, 7, 7, 7]
    k = 7
    # All elements are k. Initial frequency is 4. No operation can increase it.
    expected = 4
    assert maxFrequency(nums, k) == expected


def test_all_elements_same_can_become_k():
    """
    Tests a case where all elements are the same, and can be changed to k.
    The optimal operation is to change the whole array.
    """
    nums = [8, 8, 8, 8]
    k = 7
    # Add -1 to the whole array [8,8,8,8]. It becomes [7,7,7,7]. Frequency of 7 is 4.
    expected = 4
    assert maxFrequency(nums, k) == expected


def test_k_not_present_in_array():
    """
    Tests a case where k is not in the original array.
    The maximum frequency is found by converting the most numerous other value.
    """
    nums = [1, 2, 2, 3, 3]
    k = 10
    # k is not present. We can change either the two 2s or the two 3s to 10.
    # e.g., add 8 to nums[1..2]=[2,2] -> [10,10]. Array becomes [1,10,10,3,3]. Freq of 10 is 2.
    expected = 2
    assert maxFrequency(nums, k) == expected


def test_increase_count_by_conversion():
    """
    Tests a case where k is present, and its frequency can be increased by converting other elements.
    """
    nums = [1, 1, 1, 2, 2]
    k = 1
    # Initial frequency of 1 is 3.
    # Convert the two 2s to 1s by adding -1 to nums[3..4]=[2,2].
    # The array becomes [1,1,1,1,1]. Frequency of 1 is 5.
    expected = 5
    assert maxFrequency(nums, k) == expected


def test_optimal_subarray_avoids_k():
    """
    Tests that the best choice of subarray may be one that avoids including k,
    even if a larger subarray containing k is available.
    """
    nums = [1, 2, 1, 2, 1]
    k = 1
    # Initial freq of 1 is 3. To convert 2s to 1s, add x=-1.
    # Choosing subarray nums[1..1]=[2] changes it to [1], making the array [1,1,1,2,1]. Freq=4.
    # Choosing nums[1..2]=[2,1] would change it to [1,0], making array [1,1,0,2,1]. Freq=3.
    # The best is to target a subarray with a net gain, so we get 3 (initial) + 1 (gain) = 4.
    expected = 4
    assert maxFrequency(nums, k) == expected


def test_multiple_candidates_for_conversion():
    """
    Tests a case with multiple different values that could be converted to k.
    The algorithm should choose the one yielding the highest frequency.
    """
    nums = [10, 1, 1, 2, 2, 2]
    k = 10
    # Initial freq of 10 is 1.
    # Option 1: Convert two 1s to 10s (gain=2). Total freq = 1 + 2 = 3.
    # Option 2: Convert three 2s to 10s (gain=3). Total freq = 1 + 3 = 4.
    # The best choice is to convert the 2s.
    expected = 4
    assert maxFrequency(nums, k) == expected
