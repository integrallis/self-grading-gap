from typing import List


def maxFrequency(nums: List[int], k: int) -> int:
    nums.sort()
    n = len(nums)
    left = 0
    max_freq = 0
    total = 0

    for right in range(n):
        total += nums[right]
        while (right - left + 1) * k > total + (right - left + 1) * nums[left]:
            total -= nums[left]
            left += 1
        max_freq = max(max_freq, right - left + 1)

    return max_freq

```
