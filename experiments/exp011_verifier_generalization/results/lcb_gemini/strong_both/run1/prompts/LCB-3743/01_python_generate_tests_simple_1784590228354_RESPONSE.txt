import pytest
from typing import List
from solution import maxFreeTime

# Note: The problem asks to find the longest continuous period of free time.
# This can be achieved by creating a large gap. Gaps exist at the start of the event,
# at the end, and between any two consecutive meetings.
#
# The key insight is that rescheduling a meeting `i` allows merging the free time gap
# before it (`startTime[i] - endTime[i-1]`) and the gap after it
# (`startTime[i+1] - endTime[i]`). By moving meeting `i`, we can effectively
# combine these two gaps into a single, larger one.
#
# With `k` movable meetings, we can make `k` such merges. This is equivalent to
# merging a contiguous block of `k+1` original gaps. The problem then becomes
# finding the maximum sum of a contiguous sub-array of gaps, where the sub-array
# length is at most `k+1`.
#
# The gaps are:
# 1. Before the first meeting: `startTime[0]`
# 2. Between meetings `i` and `i+1`: `startTime[i+1] - endTime[i]`
# 3. After the last meeting: `eventTime - endTime[n-1]`

def test_canonical_example_1():
    """
    Tests Example 1 from the specification.
    Input: eventTime = 5, k = 1, startTime = [1,3], endTime = [2,5]
    Output: 2
    """
    eventTime = 5
    k = 1
    startTime = [1, 3]
    endTime = [2, 5]
    
    # n=2 meetings. There are n+1=3 gaps.
    # Gap 0 (before meeting 0): startTime[0] - 0 = 1
    # Gap 1 (between meeting 0 and 1): startTime[1] - endTime[0] = 3 - 2 = 1
    # Gap 2 (after meeting 1): eventTime - endTime[1] = 5 - 5 = 0
    # Gaps = [1, 1, 0]
    # We can merge k+1 = 2 consecutive gaps.
    # Possible merged sums:
    #   - Gaps[0] = 1
    #   - Gaps[1] = 1
    #   - Gaps[2] = 0
    #   - Gaps[0]+Gaps[1] = 1 + 1 = 2
    #   - Gaps[1]+Gaps[2] = 1 + 0 = 1
    # The maximum sum is 2.
    assert maxFreeTime(eventTime, k, startTime, endTime) == 2

def test_canonical_example_2():
    """
    Tests Example 2 from the specification.
    Input: eventTime = 10, k = 1, startTime = [0,2,9], endTime = [1,4,10]
    Output: 6
    """
    eventTime = 10
    k = 1
    startTime = [0, 2, 9]
    endTime = [1, 4, 10]
    
    # n=3 meetings. There are n+1=4 gaps.
    # Gap 0: startTime[0] = 0
    # Gap 1: startTime[1] - endTime[0] = 2 - 1 = 1
    # Gap 2: startTime[2] - endTime[1] = 9 - 4 = 5
    # Gap 3: eventTime - endTime[2] = 10 - 10 = 0
    # Gaps = [0, 1, 5, 0]
    # We can merge k+1 = 2 consecutive gaps.
    # Possible sums of <= 2 consecutive gaps:
    # Sums of 1: max(0, 1, 5, 0) = 5
    # Sums of 2: 0+1=1, 1+5=6, 5+0=5. Max is 6.
    # The overall maximum is 6.
    assert maxFreeTime(eventTime, k, startTime, endTime) == 6

def test_canonical_example_3():
    """
    Tests Example 3 from the specification.
    Input: eventTime = 5, k = 2, startTime = [0,1,2,3,4], endTime = [1,2,3,4,5]
    Output: 0
    """
    eventTime = 5
    k = 2
    startTime = [0, 1, 2, 3, 4]
    endTime = [1, 2, 3, 4, 5]
    
    # n=5 meetings. There are n+1=6 gaps.
    # Gap 0: startTime[0] = 0
    # Gap 1: startTime[1] - endTime[0] = 1 - 1 = 0
    # Gap 2: startTime[2] - endTime[1] = 2 - 2 = 0
    # Gap 3: startTime[3] - endTime[2] = 3 - 3 = 0
    # Gap 4: startTime[4] - endTime[3] = 4 - 4 = 0
    # Gap 5: eventTime - endTime[4] = 5 - 5 = 0
    # Gaps = [0, 0, 0, 0, 0, 0]
    # We can merge k+1 = 3 consecutive gaps.
    # The sum of any number of these gaps is 0.
    assert maxFreeTime(eventTime, k, startTime, endTime) == 0

def test_merge_all_gaps_when_k_is_n():
    """
    When k=n, we can move all meetings and merge all n+1 gaps.
    The max free time is the total event time minus the total duration of all meetings.
    """
    eventTime = 20
    k = 3
    startTime = [2, 8, 15]
    endTime = [5, 10, 18] # durations: 3, 2, 3. Total duration = 8
    
    # n=3, k=3. We can merge k+1=4 gaps. There are n+1=4 gaps in total.
    # We can merge all gaps.
    # Total meeting duration = (5-2) + (10-8) + (18-15) = 3 + 2 + 3 = 8.
    # Max free time = eventTime - total meeting duration = 20 - 8 = 12.
    assert maxFreeTime(eventTime, k, startTime, endTime) == 12

def test_large_single_gap_is_optimal():
    """
    Tests a case where the largest free time is an existing gap, and merging does not yield a better result.
    """
    eventTime = 100
    k = 1
    startTime = [0, 90]
    endTime = [10, 100]

    # n=2, k=1. Merge up to 2 gaps.
    # Gaps = [startTime[0], startTime[1]-endTime[0], eventTime-endTime[1]]
    #      = [0, 90-10, 100-100] = [0, 80, 0]
    # Sums of <=2 consecutive gaps:
    #   - Singles: max(0, 80, 0) = 80
    #   - Pairs: 0+80=80, 80+0=80. Max is 80.
    # Overall max is 80.
    assert maxFreeTime(eventTime, k, startTime, endTime) == 80

def test_min_n_and_max_k():
    """
    Tests the minimum n=2, with k=n=2.
    """
    eventTime = 10
    k = 2
    startTime = [2, 8]
    endTime = [4, 9]

    # n=2, k=2. Merge up to k+1=3 gaps. There are n+1=3 gaps.
    # We can merge all gaps.
    # Total meeting duration = (4-2) + (9-8) = 2 + 1 = 3.
    # Max free time = eventTime - total duration = 10 - 3 = 7.
    assert maxFreeTime(eventTime, k, startTime, endTime) == 7

def test_multiple_gaps_merge():
    """
    Tests a scenario where merging a few non-trivial gaps is required.
    """
    eventTime = 30
    k = 1
    startTime = [0, 5, 15, 28]
    endTime = [2, 10, 25, 30]

    # n=4, k=1. Merge up to k+1=2 gaps.
    # Gaps:
    # g0 = 0
    # g1 = 5-2 = 3
    # g2 = 15-10 = 5
    # g3 = 28-25 = 3
    # g4 = 30-30 = 0
    # Gaps = [0, 3, 5, 3, 0]
    # Sums of <=2 consecutive gaps:
    #   - Singles: max is 5.
    #   - Pairs: 0+3=3, 3+5=8, 5+3=8, 3+0=3. Max is 8.
    # Overall max is 8.
    assert maxFreeTime(eventTime, k, startTime, endTime) == 8
