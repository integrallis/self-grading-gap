# Note: The problem asks to find the longest continuous period of free time.
# This can be achieved by creating a large gap. Gaps exist at the start of the event,
# at the end, and between any two consecutive meetings.
#
# The key insight is that rescheduling a meeting `i` allows merging the free time gap
# before it (`startTime[i] - endTime[i-1]`) and the gap after it
# (`startTime[i+1] - endTime[i]`). By moving meeting `i` to be adjacent to meeting `i-1`,
# the gap between them becomes 0, and the space it previously occupied is effectively
# added to the gap after it. This merges the two gaps.
#
# With `k` movable meetings, we can make `k` such merges. This allows us to merge a
# contiguous block of `k+1` original gaps. The problem then becomes finding the maximum
# sum of a contiguous sub-array of gaps, where the sub-array length is at most `k+1`.
#
# The gaps are:
# 1. Before the first meeting: `startTime[0]`
# 2. Between meetings `i` and `i+1`: `startTime[i+1] - endTime[i]`
# 3. After the last meeting: `eventTime - endTime[n-1]`
# There are n+1 such gaps in total.

def test_canonical_example_1():
    """
    Tests Example 1 from the specification.
    Input: eventTime = 5, k = 1, startTime = [1,3], endTime = [2,5]
    Output: 2
    """
    eventTime = 5
    k = 1
    startTime = [1, 3]
    endTime = [2, 5]
    
    # n=2 meetings. There are n+1=3 gaps.
    # Gap 0 (before meeting 0): startTime[0] - 0 = 1
    # Gap 1 (between meeting 0 and 1): startTime[1] - endTime[0] = 3 - 2 = 1
    # Gap 2 (after meeting 1): eventTime - endTime[1] = 5 - 5 = 0
    # Gaps = [1, 1, 0]
    # We can merge k+1 = 2 consecutive gaps.
    # Possible merged sums:
    #   - Gaps[0] = 1
    #   - Gaps[1] = 1
    #   - Gaps[2] = 0
    #   - Gaps[0]+Gaps[1] = 1 + 1 = 2
    #   - Gaps[1]+Gaps[2] = 1 + 0 = 1
    # The maximum sum is 2.
    assert maxFreeTime(eventTime, k, startTime, endTime) == 2

def test_large_n_performance():
    """
    Tests with a large number of meetings (n=10^5) to ensure the solution is efficient (e.g., O(n)).
    """
    n = 10**5
    k = 100
    
    startTime = []
    endTime = []
    
    currentTime = 0
    
    # Initial gap
    start_gap = 1
    currentTime += start_gap
    
    # Generate meetings and inter-meeting gaps
    for i in range(n):
        s = currentTime
        e = s + 1  # All meetings have duration 1
        startTime.append(s)
        endTime.append(e)
        currentTime = e
        
        # Define gap after this meeting
        if i == n // 2:
            gap_after = 1000
        else:
            gap_after = 1
        
        currentTime += gap_after
        
    eventTime = currentTime
    
    # The gaps are [start_gap, gap_after(0), ..., gap_after(n-1)].
    # gaps = [1, 1, ..., 1 (at i=n//2-1), 1000 (at i=n//2), 1, ..., 1].
    # The large gap of 1000 is between meeting n//2 and n//2 + 1.
    # k=100 allows merging k+1=101 gaps.
    # The max sum will be from a window of 101 gaps centered on the 1000-gap.
    # This window includes the 1000-gap and 100 surrounding 1-gaps.
    # Expected sum = 1000 + 100 * 1 = 1100.
    assert maxFreeTime(eventTime, k, startTime, endTime) == 1100
