# MANDATORY: Canonical Examples from Specification

def test_canonical_example_1():
    """
    Test with the first example from the specification.
    s = "aczzx"
    - i=0, s[0]='a': no j < i.
    - i=1, s[1]='c': no j < i with mirror 'x'.
    - i=2, s[2]='z': mirror 'a'. Closest unmarked j is 0. Mark 0, 2. score = 2-0 = 2.
    - i=3, s[3]='z': mirror 'a'. Index 0 is marked. No other unmarked 'a'.
    - i=4, s[4]='x': mirror 'c'. Closest unmarked j is 1. Mark 1, 4. score = 2 + (4-1) = 5.
    Expected: 5
    """
    assert calculateScore("aczzx") == 5
