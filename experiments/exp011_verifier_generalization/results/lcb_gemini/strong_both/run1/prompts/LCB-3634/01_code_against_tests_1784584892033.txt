```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # Create a mirror mapping for 'a' to 'z'

    for i in range(len(s)):
        if marked[i]:  # If already marked, skip
            continue
        for j in range(i - 1, -1, -1):  # Look for the closest unmarked index
            if marked[j]:  # If marked, skip
                continue
            if mirror[s[i]] == s[j]:  # Check if s[j] is the mirror of s[i]
                marked[i] = marked[j] = True  # Mark both indices
                score += (i - j)  # Update score
                break  # Move to the next index

    return score
```
