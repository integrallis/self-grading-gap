## Implementation

```python
def hasSpecialSubstring(s: str, k: int) -> bool:
    n = len(s)
    for i in range(n - k + 1):
        substring = s[i:i + k]
        if len(set(substring)) == 1:
            before_condition = (i == 0 or s[i - 1] != substring[0])
            after_condition = (i + k == n or s[i + k - 1] != substring[0])
            if before_condition and after_condition:
                return True
    return False

```
