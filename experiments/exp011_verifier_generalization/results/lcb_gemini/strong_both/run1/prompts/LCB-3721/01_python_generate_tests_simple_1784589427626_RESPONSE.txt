def test_canonical_example_1():
    """
    Test case from the first example in the problem description.
    A user goes offline, then comes back online just in time for a 'HERE' mention.
    """
    numberOfUsers = 2
    events = [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "71", "HERE"]]
    # Explanation:
    # t=10: MESSAGE "id1 id0". Both users online. mentions=[1, 1].
    # t=11: OFFLINE "0". User 0 goes offline. Back online at 11+60=71.
    # t=71: MESSAGE "HERE". User 0 is back online. Both users are online.
    #       User 0 and 1 are mentioned.
    # Final mentions: User 0: 1 (t=10) + 1 (t=71) = 2. User 1: 1 (t=10) + 1 (t=71) = 2.
    expected = [2, 2]
    assert countMentions(numberOfUsers, events) == expected

def test_complex_scenario_with_mixed_mentions():
    """
    A more complex test case with multiple offline/online events and mixed mention types.
    """
    numberOfUsers = 4
    events = [
        ["OFFLINE", "10", "1"],        # U1 offline [10, 69]. Online at 70.
        ["MESSAGE", "15", "id0 id1"],  # U0 mentioned, U1 mentioned (id mentions offline). mentions: [1,1,0,0]
        ["OFFLINE", "20", "3"],        # U3 offline [20, 79]. Online at 80.
        ["MESSAGE", "25", "HERE"],     # U0, U2 are online. mentions: [2,1,1,0]
        ["MESSAGE", "30", "ALL"],      # All 4 users mentioned. mentions: [3,2,2,1]
        ["OFFLINE", "75", "0"],        # U0 offline [75, 134]. U1 is online now (since 70).
        ["MESSAGE", "80", "HERE"],     # At t=80, U3 is back online. Online: U1, U2, U3. U0 is offline. mentions: [3,3,3,2]
    ]
    # Step-by-step breakdown:
    # Initially: online={0,1,2,3}, mentions=[0,0,0,0]
    # t=10: OFFLINE "1". online={0,2,3}
    # t=15: MESSAGE "id0 id1". U0 mentioned, U1 mentioned. mentions=[1,1,0,0]
    # t=20: OFFLINE "3". online={0,2}
    # t=25: MESSAGE "HERE". U0, U2 mentioned. mentions=[1+1, 1, 0+1, 0] -> [2,1,1,0]
    # t=30: MESSAGE "ALL". U0,U1,U2,U3 mentioned. mentions=[2+1, 1+1, 1+1, 0+1] -> [3,2,2,1]
    # t=70: U1 comes back online. online={0,1,2}
    # t=75: OFFLINE "0". online={1,2}
    # t=80: U3 comes back online. online={1,2,3}
    # t=80: MESSAGE "HERE". U1,U2,U3 mentioned. mentions=[3, 2+1, 2+1, 1+1] -> [3,3,3,2]
    expected = [3, 3, 3, 2]
    assert countMentions(numberOfUsers, events) == expected
