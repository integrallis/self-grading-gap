from typing import List

def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers
    online_status = [True] * numberOfUsers
    offline_until = [-1] * numberOfUsers
    events.sort(key=lambda x: int(x[1]))  # Sort events by timestamp

    for event in events:
        event_type, timestamp, arg = event
        timestamp = int(timestamp)

        # Process offline events first
        if event_type == "OFFLINE":
            user_id = int(arg)
            online_status[user_id] = False
            offline_until[user_id] = timestamp + 60

        # Process message events
        elif event_type == "MESSAGE":
            # Update online status before processing messages
            for i in range(numberOfUsers):
                if timestamp >= offline_until[i]:
                    online_status[i] = True

            if arg == "ALL":
                for i in range(numberOfUsers):
                    mentions[i] += 1  # Count every user
            elif arg == "HERE":
                for i in range(numberOfUsers):
                    if online_status[i]:
                        mentions[i] += 1  # Count only online users
            else:
                mentioned_ids = arg.split()
                for user_id in mentioned_ids:
                    if user_id.startswith("id"):
                        user_index = int(user_id[2:])
                        mentions[user_index] += 1  # Count specific user mentions

    return mentions
