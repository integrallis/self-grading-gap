def test_canonical_example_1():
    """
    Example 1 from the specification.
    n = 5, edges = [[1,0,1],[2,0,2],[3,0,1],[4,3,1],[2,1,1]], threshold = 2
    Output: 1
    Explanation: If we allow a max weight of 1, we can use edges:
    [1,0,1], [3,0,1], [4,3,1], [2,1,1].
    - Paths to 0: 1->0, 2->1->0, 3->0, 4->3->0. All nodes can reach 0.
    - Out-degrees: out(1)=1, out(2)=1, out(3)=1, out(4)=1. All are <= threshold=2.
    - Max weight is 1. This is the minimal possible.
    """
    n = 5
    edges = [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]]
    threshold = 2
    assert minMaxWeight(n, edges, threshold) == 1

def test_threshold_forces_higher_weight_solution():
    """
    Tests a scenario where the threshold constraint forces the selection of a solution
    with a higher maximum weight, even though a path with a lower max weight exists
    but would violate the threshold.
    """
    n = 4
    threshold = 1
    edges = [[1, 2, 10], [2, 0, 20], [1, 3, 5], [3, 0, 30]]
    # With max_weight < 30, the edge (3,0,30) is not available, and node 3 cannot reach 0.
    # With max_weight = 30, all edges are potentially available. We need to select a subgraph.
    # A valid subgraph is {(1,2,10), (2,0,20), (3,0,30)}.
    # Out-degrees in this subgraph: out(1)=1, out(2)=1, out(3)=1. All <= threshold=1.
    # Reachability: 1->2->0, 2->0, 3->0. All nodes can reach 0.
    # The max weight of this subgraph is 30. This is the minimum possible.
    assert minMaxWeight(n, edges, threshold) == 30


def test_simple_line_graph():

    threshold = 1
    # The only way to connect all nodes to 0 is to use all edges.
    # The max weight in this set is 20.
    assert minMaxWeight(n, edges, threshold) == 20


def test_choice_of_path_matters():
    """
    Tests that the algorithm correctly chooses a path with a lower max weight.
    """
    n = 3
    edges = [[1, 0, 100], [1, 2, 10], [2, 0, 20]]
    threshold = 1
    # To connect node 1 to 0, there are two options:
    # 1. Direct edge (1,0) with weight 100.
    # 2. Path 1 -> 2 -> 0 with max edge weight of max(10, 20) = 20.
    # The minimal maximum weight is 20.
    assert minMaxWeight(n, edges, threshold) == 20

def test_high_threshold():
    """
    Tests a case where the threshold is large enough not to be a constraint.
    """
    n = 5
    edges = [[1, 0, 10], [1, 2, 5], [2, 0, 8], [3, 0, 12], [4, 0, 100]]
    threshold = 4  # threshold = n - 1
    # We need to connect all nodes 1, 2, 3, 4.
    # To connect node 4, we must use the edge (4,0) with weight 100.
    # So the minimum max weight is at least 100. At this weight, all edges are available
    # and it's possible to connect everyone, as threshold is not a limiting factor.
    assert minMaxWeight(n, edges, threshold) == 100


def test_multiple_edges_between_nodes():
    """
    Tests a graph with multiple edges between the same pair of nodes.
    The algorithm should choose the cheaper edge to build a path if possible.
    """
    n = 3
    edges = [[1, 2, 10], [2, 0, 30], [1, 2, 20]]  # Two edges from 1 to 2
    threshold = 1
    # Path 1->2->0 is required.
    # To minimize max weight, we should use the (1,2) edge with weight 10.
    # The max weight on this path is max(10, 30) = 30.
    assert minMaxWeight(n, edges, threshold) == 30


def test_all_weights_are_the_same():
    """
    Tests a case where all edges have the same weight and a solution exists.

    threshold = 1
    # A path 1->2->3->0 exists. All edges have weight 10.
    # The max weight is 10. If we tried a smaller weight, no edges would be available.
    assert minMaxWeight(n, edges, threshold) == 10
