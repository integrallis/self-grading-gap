from typing import List
import heapq


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    def canReachAll(max_weight: int) -> bool:
        graph = [[] for _ in range(n)]
        for a, b, w in edges:
            if w <= max_weight:
                graph[b].append(a)
        # BFS to see if we can reach 0 from all other nodes
        reachable = [False] * n
        reachable[0] = True
        queue = [0]
        while queue:
            current = queue.pop(0)
            for neighbor in graph[current]:
                if not reachable[neighbor]:
                    reachable[neighbor] = True
                    queue.append(neighbor)
        return all(reachable)

    def outDegreeCheck(max_weight: int) -> bool:
        out_degree = [0] * n
        for a, b, w in edges:
            if w <= max_weight:
                out_degree[a] += 1
                if out_degree[a] > threshold:
                    return False
        return True

    # Binary search for the minimum possible max weight
    left, right = 1, max(weight for _, _, weight in edges)
    result = -1
    while left <= right:
        mid = (left + right) // 2
        if canReachAll(mid) and outDegreeCheck(mid):
            result = mid
            right = mid - 1
        else:
            left = mid + 1
    return result

def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:

        reachable = [False] * n
        reachable[0] = True
        queue = [0]

            current = queue.pop(0)
            for neighbor in graph[current]:
                if not reachable[neighbor]:
                    reachable[neighbor] = True
                    queue.append(neighbor)

        out_degree = [0] * n
        for a, b, w in edges:
            if w <= max_weight:
                out_degree[a] += 1
                if out_degree[a] > threshold:
                    return False
        return True

    result = -1

        mid = (left + right) // 2
        if canReachAll(mid) and outDegreeCheck(mid):
            result = mid
            right = mid - 1
        else:
            left = mid + 1
