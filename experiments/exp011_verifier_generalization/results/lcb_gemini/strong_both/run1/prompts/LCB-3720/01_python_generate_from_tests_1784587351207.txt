from typing import List
import heapq


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    if not edges:
        return -1 if n > 1 else 0

    def is_possible(max_weight: int) -> bool:
        candidate_edges = [e for e in edges if e[2] <= max_weight]
        rev_adj = [[] for _ in range(n)]
        for u, v, _ in candidate_edges:
            rev_adj[v].append(u)

        reachable = [False] * n
        reachable[0] = True
        queue = [0]
        while queue:
            current = queue.pop(0)
            for neighbor in rev_adj[current]:
                if not reachable[neighbor]:
                    reachable[neighbor] = True
                    queue.append(neighbor)
        if not all(reachable):
            return False

        out_degree = [0] * n
        for u, v, _ in candidate_edges:
            out_degree[u] += 1
            if out_degree[u] > threshold:
                return False

        return True

    left, right = 1, max(weight for _, _, weight in edges)
    result = -1
    while left <= right:
        mid = (left + right) // 2
        if is_possible(mid):
            result = mid
            right = mid - 1
        else:
            left = mid + 1
    return result

```
