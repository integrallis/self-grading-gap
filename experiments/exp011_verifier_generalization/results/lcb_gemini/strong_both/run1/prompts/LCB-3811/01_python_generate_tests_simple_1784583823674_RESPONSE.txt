import pytest
from solution import reverseDegree


def test_canonical_example_1():
    """
    Tests the first example from the specification: s = "abc".
    """
    # spec: input '"abc"' -> output '148'
    # Calculation:
    # 'a' (pos 1, rev_alpha 26): 26 * 1 = 26
    # 'b' (pos 2, rev_alpha 25): 25 * 2 = 50
    # 'c' (pos 3, rev_alpha 24): 24 * 3 = 72
    # Total = 26 + 50 + 72 = 148
    assert reverseDegree("abc") == 148
