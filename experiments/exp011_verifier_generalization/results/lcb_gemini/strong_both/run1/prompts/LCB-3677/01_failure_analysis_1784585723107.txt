from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m = len(coins)
    n = len(coins[0]) if m > 0 else 0

    # dp[i][j][k] will be the max coins collected when reaching cell (i,j) with k neutralizations used
    dp = [[[-float('inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = max(0, coins[0][0]) if coins[0][0] >= 0 else 0  # Initialize the starting point

    for i in range(m):
        for j in range(n):
            for k in range(3):
                if i == 0 and j == 0:
                    continue
                # Collect coins from the top cell
                if i > 0:
                    if coins[i][j] >= 0:
                        dp[i][j][k] = max(dp[i][j][k], dp[i - 1][j][k] + coins[i][j])
                    elif k > 0:
                        dp[i][j][k] = max(dp[i][j][k], dp[i - 1][j][k - 1])
                # Collect coins from the left cell
                if j > 0:
                    if coins[i][j] >= 0:
                        dp[i][j][k] = max(dp[i][j][k], dp[i][j - 1][k] + coins[i][j])
                    elif k > 0:
                        dp[i][j][k] = max(dp[i][j][k], dp[i][j - 1][k - 1])

    result = -float('inf')
    for k in range(3):
        result = max(result, dp[m - 1][n - 1][k])
    return result
