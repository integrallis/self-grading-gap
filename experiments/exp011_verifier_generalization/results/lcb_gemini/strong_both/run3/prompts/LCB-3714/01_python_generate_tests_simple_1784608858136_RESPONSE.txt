import pytest

def test_canonical_example_1():
    """
    Tests the first example from the problem description.
    nums = [1, 2, 3], k = 2
    """
    # Subsequences of length at most 2:
    # [1]: min=1, max=1, sum=2
    # [2]: min=2, max=2, sum=4
    # [3]: min=3, max=3, sum=6
    # [1, 2]: min=1, max=2, sum=3
    # [1, 3]: min=1, max=3, sum=4
    # [2, 3]: min=2, max=3, sum=5
    # Total sum = 2 + 4 + 6 + 3 + 4 + 5 = 24
    assert minMaxSums(nums=[1, 2, 3], k=2) == 24


def test_canonical_example_2():
    """
    Tests the second example from the problem description.
    nums = [5, 0, 6], k = 1
    """
    # Only subsequences of length 1 are considered (at most k=1 elements).
    # For a single-element subsequence, min and max are the element itself.
    # [5]: min=5, max=5, sum=10
    # [0]: min=0, max=0, sum=0
    # [6]: min=6, max=6, sum=12
    # Total sum = 10 + 0 + 12 = 22
    assert minMaxSums(nums=[5, 0, 6], k=1) == 22


def test_canonical_example_3():
    """
    Tests the third example from the problem description, which includes duplicate numbers.
    nums = [1, 1, 1], k = 2
    """
    # The array has 3 elements, all are 1.
    # Subsequences of length 1: There are 3 such subsequences: [1], [1], [1].
    # For each, min=1, max=1, sum=2. Total for length 1 = 3 * 2 = 6.
    # Subsequences of length 2: There are C(3,2)=3 such subsequences: [1,1], [1,1], [1,1].
    # For each, min=1, max=1, sum=2. Total for length 2 = 3 * 2 = 6.
    # Total sum = 6 + 6 = 12.
    assert minMaxSums(nums=[1, 1, 1], k=2) == 12

def test_k_equals_n():
    """

    # Length 1: [10](20), [20](40), [30](60). Sum = 120.
    # Length 2: [10,20](30), [10,30](40), [20,30](50). Sum = 120.
    # Length 3: [10,20,30](40). Sum = 40.
    # Total sum = 120 + 120 + 40 = 280.
    assert minMaxSums(nums=[10, 20, 30], k=3) == 280

def test_with_zero_and_duplicates():
    """
    Tests a case with both a zero and duplicate non-zero values.

    # nums = [8, 0, 8], k = 2. Sorted: [0, 8a, 8b]
    # Length 1: [0](0), [8a](16), [8b](16). Sum = 32.
    # Length 2: [0, 8a](8), [0, 8b](8), [8a, 8b](16). Sum = 32.

def test_large_numbers_with_modulo():
    """
    Tests that the modulo arithmetic is handled correctly with large numbers.
    """
    # nums = [10^9, 10^9, 10^9], k = 2
    # This has the same combinatorial structure as [1,1,1] with k=2, which yielded 12.
    # The result should be (12 * 10^9) mod (10^9 + 7).
    # 12 * 10^9 = 12 * (10^9 + 7 - 7) = 12*(10^9+7) - 84
    # (12 * 10^9) mod (10^9 + 7) = -84 mod (10^9 + 7) = 10^9 + 7 - 84
    # 1000000007 - 84 = 999999923
    large_val = 10**9
    assert minMaxSums(nums=[large_val, large_val, large_val], k=2) == 999999923
