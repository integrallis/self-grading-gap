from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    nums.sort()
    total_sum = 0

    # Calculate the total contribution of minimums and maximums
    for i in range(n):
        # Contribution as minimum
        min_contribution = nums[i] * (pow(2, min(k, n - 1), MOD) - 1) % MOD
        # Contribution as maximum
        max_contribution = nums[i] * (pow(2, min(k, n - 1), MOD) - 1) % MOD
        total_sum = (total_sum + min_contribution + max_contribution) % MOD

    return total_sum

```
