from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    if k > n:
        k = n

    total_sum = 0

    # Calculate contributions from subsequences of length 1 to k
    for length in range(1, k + 1):
        # Calculate the number of combinations for choosing `length` elements from `n`
        from math import comb
        comb_count = comb(n, length)

        # For each number in the nums, calculate its contribution as min and max
        for num in nums:
            # Each number contributes to both min and max of its subsequences
            # The contribution count is the number of ways to select (length-1) other elements
            contribution = (comb_count * num) % MOD
            total_sum = (total_sum + contribution * 2) % MOD

    return total_sum

```
