from typing import List
from itertools import permutations

def totalNumbers(digits: List[int]) -> int:
    distinct_numbers = set()
    even_digits = [d for d in digits if d % 2 == 0]
    for even_digit in even_digits:
        # Generate permutations with the even digit at the last position
        for p in permutations(digits):
            if p[2] == even_digit and p[0] != 0:  # Last digit is even and no leading zero
                distinct_numbers.add(p)
    return len(distinct_numbers)
