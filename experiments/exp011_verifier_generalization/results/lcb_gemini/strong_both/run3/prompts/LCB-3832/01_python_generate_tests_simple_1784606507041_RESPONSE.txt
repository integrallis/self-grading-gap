import pytest
from typing import List
from solution import minCosts


def test_canonical_example_1():
    """
    Tests the first example from the specification.
    Input: cost = [5, 3, 4, 1, 3, 2]
    """
    cost = [5, 3, 4, 1, 3, 2]
    # The minimum cost to reach position i is the minimum cost to swap with any person j <= i.
    # After swapping to position j, moving to a later position i (i > j) is free.
    # answer[i] = min(cost[0]...cost[i])
    #
    # answer[0] = min(cost[0]) = 5
    # answer[1] = min(cost[0], cost[1]) = min(5, 3) = 3
    # answer[2] = min(cost[0..2]) = min(5, 3, 4) = 3
    # answer[3] = min(cost[0..3]) = min(5, 3, 4, 1) = 1
    # answer[4] = min(cost[0..4]) = min(5, 3, 4, 1, 3) = 1
    # answer[5] = min(cost[0..5]) = min(5, 3, 4, 1, 3, 2) = 1
    expected = [5, 3, 3, 1, 1, 1]
    assert minCosts(cost) == expected


def test_canonical_example_2():
    """
    Tests the second example from the specification.
    Input: cost = [1, 2, 4, 6, 7]
    """
    cost = [1, 2, 4, 6, 7]
    # The minimum cost to reach position i is the running minimum of the cost array up to i.
    #
    # answer[0] = min(cost[0]) = 1
    # answer[1] = min(cost[0..1]) = min(1, 2) = 1
    # answer[2] = min(cost[0..2]) = min(1, 2, 4) = 1
    # answer[3] = min(cost[0..3]) = min(1, 2, 4, 6) = 1
    # answer[4] = min(cost[0..4]) = min(1, 2, 4, 6, 7) = 1
    expected = [1, 1, 1, 1, 1]
    assert minCosts(cost) == expected

def test_strictly_decreasing_costs():
    """
    Tests a case where costs are strictly decreasing. The minimum cost
    to reach position i is always cost[i] itself.

    cost = [100, 50, 25, 10, 1]
    # answer[0] = min(100) = 100
    # answer[1] = min(100, 50) = 50
    # answer[2] = min(100, 50, 25) = 25

    # answer[4] = min(100, 50, 25, 10, 1) = 1
    expected = [100, 50, 25, 10, 1]
    assert minCosts(cost) == expected
