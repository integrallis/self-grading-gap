```python
from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n == 0:
        return 0
    # Create a dp array to store the maximum length of subsequence ending at each index
    dp = [1] * n
    # Create an array to store the differences
    diff = [0] * (n - 1)

    for i in range(n - 1):
        diff[i] = abs(nums[i] - nums[i + 1])

    # Iterate to find the longest subsequence
    for i in range(n):
        for j in range(i + 1, n):
            if (j > i + 1) and (diff[j - 1] > diff[j - 2]):
                dp[j] = max(dp[j], dp[i] + 1)

    return max(dp)
```
