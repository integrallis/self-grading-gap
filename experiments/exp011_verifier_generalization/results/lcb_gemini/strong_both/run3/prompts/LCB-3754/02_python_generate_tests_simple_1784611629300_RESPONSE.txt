import pytest
from solution import maxDistance

# MANDATORY: Canonical Examples from Specification

def test_canonical_example_1():
    # spec: maxDistance("NWSE", 1) -> 3
    # Explanation: Change s[2] from 'S' to 'N'. The string s becomes "NWNE".
    # Path of "NWNE":
    #   s[0] == 'N': pos=(0, 1), dist = 1
    #   s[1] == 'W': pos=(-1, 1), dist = 2
    #   s[2] == 'N': pos=(-1, 2), dist = 3
    #   s[3] == 'E': pos=(0, 2), dist = 2
    # The maximum Manhattan distance achieved is 3.
    assert maxDistance("NWSE", 1) == 3

def test_canonical_example_2():
    # spec: maxDistance("NSWWEW", 3) -> 6
    # Explanation: Change s[1] from 'S' to 'N', and s[4] from 'E' to 'W'. s becomes "NNWWWW".
    # This uses 2 of the 3 available changes.
    # Path of "NNWWWW":
    #   s[0] == 'N': pos=(0, 1), dist = 1
    #   s[1] == 'N': pos=(0, 2), dist = 2
    #   s[2] == 'W': pos=(-1, 2), dist = 3
    #   s[3] == 'W': pos=(-2, 2), dist = 4
    #   s[4] == 'W': pos=(-3, 2), dist = 5
    #   s[5] == 'W': pos=(-4, 2), dist = 6
    # The maximum Manhattan distance achieved is 6.
    assert maxDistance("NSWWEW", 3) == 6

def test_no_changes_k_is_zero():
    # spec: s="NWS", k=0. No changes are allowed.
    # Path: N -> (0, 1), dist=1. W -> (-1, 1), dist=2. S -> (-1, 0), dist=1.
    # The maximum distance reached during this path is 2.
    assert maxDistance("NWS", 0) == 2

def test_k_is_zero_and_path_returns_to_origin():
    # spec: s="NESW", k=0. No changes are allowed.
    # Path: N -> (0,1) d=1; E -> (1,1) d=2; S -> (1,0) d=1; W -> (0,0) d=0.
    # The maximum distance reached is 2.
    assert maxDistance("NESW", 0) == 2

def test_tradeoff_between_axes_with_limited_changes():
    # spec: s="NESW", k=1.
    # The max distance can be achieved at an intermediate step.
    # Consider the prefix "NES". Original path: N(0,1) d=1, E(1,1) d=2, S(1,0) d=1. Max dist=2.
    # With k=1, we can change 'S' to 'N', making the prefix path "NEN".
    # Path for "NEN": N(0,1) d=1, E(1,1) d=2, N(1,2) d=3.
    # The maximum distance achieved is 3.
    # For the full string "NESW", the path could be "NENW", which also has a max distance of 3.
    assert maxDistance("NESW", 1) == 3

def test_single_char_string_with_change():
    # spec: s="W", k=1.
    # Original path is W -> (-1, 0), dist=1.
    # With one change, e.g., W -> N, path is N -> (0, 1), dist=1.
    # The maximum achievable distance remains 1.
    assert maxDistance("W", 1) == 1

def test_all_one_direction_no_gain_from_changes():
    # spec: s="EEEE", k=2.
    # Original path: E->(1,0) d=1, E->(2,0) d=2, E->(3,0) d=3, E->(4,0) d=4. Max dist=4.
    # Changes do not increase the max distance, as the path is already a straight line.
    # E.g., change to "ENEE": path E(1,0)d=1, N(1,1)d=2, E(2,1)d=3, E(3,1)d=4. Max is 4.
    # The maximum possible distance is len(s)=4, which is already achieved.
    assert maxDistance("EEEE", 2) == 4

def test_back_and_forth_path():
    # spec: s="NSNS", k=1.
    # Original max distance is 1.
    # With k=1, we can change s[1] 'S' to 'N'. s becomes "NNNS".
    # Path: N->(0,1) d=1, N->(0,2) d=2, N->(0,3) d=3, S->(0,2) d=2.
    # The maximum distance on this new path is 3.
    assert maxDistance("NSNS", 1) == 3

def test_long_alternating_path():
    # spec: s="NSNSNS", k=2. len=6.
    # The maximum distance is found by considering prefixes.
    # At step i=5, the prefix is "NSNSN". It has 3 'N's and 2 'S's.
    # With k=2, we can change both 'S's to 'N's, making the path for the first 5 steps "NNNNN".
    # Path: (0,1)d=1, (0,2)d=2, (0,3)d=3, (0,4)d=4, (0,5)d=5.
    # Max distance of 5 is achieved at step 5. This is the highest possible.
    assert maxDistance("NSNSNS", 2) == 5

def test_minimal_constraints():
    # spec: s="N", len=1. k=0.
    # The only move is N, to (0,1). Distance is 1.
    assert maxDistance("N", 0) == 1

def test_max_k_is_length():
    # spec: s="WE", k=2. k=len(s).
    # All characters can be changed. Change to "NN" for a straight path.
    # Path: N -> (0,1) d=1, N -> (0,2) d=2. Max distance is 2.
    assert maxDistance("WE", 2) == 2
