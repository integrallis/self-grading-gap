import pytest
from solution import maxDifference

#
# MANDATORY: Canonical Examples from Specification
#

def test_canonical_example_1():
    # spec: maxDifference("aaaaabbc") -> 3
    # Frequencies: 'a': 5 (odd), 'b': 2 (even), 'c': 1 (odd)
    # The pairs of (odd_freq, even_freq) are (5, 2) and (1, 2).
    # Differences: 5 - 2 = 3, and 1 - 2 = -1.
    # The maximum difference is 3.
    assert maxDifference("aaaaabbc") == 3

def test_canonical_example_2():
    # spec: maxDifference("abcabcab") -> 1
    # Frequencies: 'a': 3 (odd), 'b': 3 (odd), 'c': 2 (even)
    # The pairs of (odd_freq, even_freq) are (3, 2).
    # Difference: 3 - 2 = 1.
    # The maximum difference is 1.
    assert maxDifference("abcabcab") == 1

def test_multiple_odd_and_even_frequencies():
    # s = "aaaaaaaaabbccdde"
    # Frequencies: 'a': 9 (odd), 'b': 2 (even), 'c': 2 (even), 'd': 2 (even), 'e': 1 (odd)
    # Odd frequencies: {9, 1}
    # Even frequencies: {2}
    # To maximize (odd_freq - even_freq), we need the largest odd frequency and the smallest even frequency.
    # Max odd freq = 9 (from 'a')
    # Min even freq = 2 (from 'b', 'c', or 'd')
    # Difference = 9 - 2 = 7.
    assert maxDifference("aaaaaaaaabbccdde") == 7

def test_negative_difference_is_max():
    # s = "aab"
    # Frequencies: 'a': 2 (even), 'b': 1 (odd)
    # There is only one character with an odd frequency ('b') and one with an even frequency ('a').
    # Odd frequency: 1
    # Even frequency: 2
    # The only possible difference is 1 - 2 = -1.
    assert maxDifference("aab") == -1

def test_min_length_string_another_case():
    # s = "abb" (length 3, the minimum allowed)
    # Frequencies: 'a': 1 (odd), 'b': 2 (even)
    # Max odd frequency = 1
    # Min even frequency = 2
    # Difference = 1 - 2 = -1.
    assert maxDifference("abb") == -1

def test_max_length_string_positive_result():
    # s = 'a' * 97 + 'b' * 2 + 'c', length is 97 + 2 + 1 = 100 (maximum allowed)
    # Frequencies: 'a': 97 (odd), 'b': 2 (even), 'c': 1 (odd)
    # Odd frequencies: {97, 1}
    # Even frequencies: {2}
    # Max odd freq = 97
    # Min even freq = 2
    # Difference = 97 - 2 = 95.
    assert maxDifference('a' * 97 + 'b' * 2 + 'c') == 95

def test_max_length_string_negative_result():
    # s = 'a' * 98 + 'b' + 'c', length is 98 + 1 + 1 = 100 (maximum allowed)
    # Frequencies: 'a': 98 (even), 'b': 1 (odd), 'c': 1 (odd)
    # Odd frequencies: {1}
    # Even frequencies: {98}
    # Max odd freq = 1
    # Min even freq = 98
    # Difference = 1 - 98 = -97.
    assert maxDifference('a' * 98 + 'b' + 'c') == -97

def test_single_value_for_odd_and_even_freqs():
    # s = "aabbccddeefffggg"
    # Frequencies: a:2, b:2, c:2, d:2, e:2 (all even), f:3, g:3 (all odd)
    # Odd freqs: {3}, Even freqs: {2}
    # Max odd freq = 3
    # Min even freq = 2
    # Difference = 3 - 2 = 1
    assert maxDifference("aabbccddeefffggg") == 1
