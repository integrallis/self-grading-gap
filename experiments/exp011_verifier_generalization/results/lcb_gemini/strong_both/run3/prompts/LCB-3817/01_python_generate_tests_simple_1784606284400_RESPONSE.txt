def test_canonical_example_1():
    """
    Tests the first example from the specification.
    The deck size (4) is smaller than the number of containers allowed by weight (5),
    so the deck size is the limiting factor.
    """
    # spec: n = 2, w = 3, maxWeight = 15 -> 4
    # Max containers by deck size = n * n = 2 * 2 = 4
    # Max containers by weight = floor(maxWeight / w) = floor(15 / 3) = 5
    # Result = min(4, 5) = 4
    assert maxContainers(n=2, w=3, maxWeight=15) == 4
