from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    # dp[i][j][k] will store the maximum coins collected to reach (i, j) using k neutralizations
    dp = [[[-float('inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0] if coins[0][0] >= 0 else 0  # Starting point

    for i in range(m):
        for j in range(n):
            for k in range(3):  # k is the number of neutralizations used
                if i == 0 and j == 0:
                    continue  # Skip the starting cell
                # If moving from above
                if i > 0:
                    value = coins[i][j]
                    if value < 0 and k > 0:
                        dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k-1])  # Neutralize robber
                    else:
                        dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k] + value)
                # If moving from the left
                if j > 0:
                    value = coins[i][j]
                    if value < 0 and k > 0:
                        dp[i][j][k] = max(dp[i][j][k], dp[i][j-1][k-1])  # Neutralize robber
                    else:
                        dp[i][j][k] = max(dp[i][j][k], dp[i][j-1][k] + value)

    # The maximum coins collected at the bottom right corner with at most 2 neutralizations
    return max(dp[m-1][n-1][0], dp[m-1][n-1][1], dp[m-1][n-1][2])
