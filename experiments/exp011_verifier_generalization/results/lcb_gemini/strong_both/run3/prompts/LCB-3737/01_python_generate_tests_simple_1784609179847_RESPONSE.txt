def test_canonical_example_1():
    """
    Tests the first example provided in the problem specification.

    # The optimal painting sequence is colors [1, 2, 3, 2] (1-indexed).
    # This corresponds to costs: cost[0][0] + cost[1][1] + cost[2][2] + cost[3][1]

    """
    Tests the behavior for the smallest allowed value of n (n=2).
    """

    cost = [[1, 10, 100], [2, 20, 200]]
    # For n=2, houses 0 and 1 are both adjacent and equidistant.
    # Therefore, they must have different colors. We find the minimum cost pair.
    # cost(c0,c1) | c1=0(2) | c1=1(20) | c1=2(200)

    # c0=0(1)     |   -     |   21     |   201

    # c0=2(100)   |   102   |   120    |   -

def test_identical_costs_across_houses():
    """
    Tests with identical costs for each house, which forces suboptimal color choices
    due to the constraints.
    """
    n = 4
    cost = [[10, 1, 20], [10, 1, 20], [10, 1, 20], [10, 1, 20]]
    # The cheap color (index 1, cost 1) is preferred, but constraints limit its use.
    # We can use the cheap color at most twice to satisfy the constraints.
    # For example, paint houses 0 and 2 with the cheap color (cost 1+1).
    # Houses 1 and 3 must use other colors, e.g., the next cheapest (cost 10+10).
    # A valid painting is colors [2, 1, 2, 1] (1-indexed), or [1,0,1,0] (0-indexed).
    # Total cost = cost[0][1] + cost[1][0] + cost[2][1] + cost[3][0] = 1 + 10 + 1 + 10 = 22.
    assert minCost(n, cost) == 22
