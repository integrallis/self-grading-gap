```python
from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # DP table to store minimum costs
    dp = [[0] * 3 for _ in range(n)]
    # Initialize the first house costs
    for j in range(3):
        dp[0][j] = cost[0][j]
    # Fill the DP table
    for i in range(1, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(
                dp[i-1][(j+1) % 3],
                dp[i-1][(j+2) % 3]
            )
    # Handle the equidistant constraint by considering the first half and the second half
    # The houses must be painted such that i and n-1-i aren't of the same color
    min_cost = float('inf')
    for color in range(3):
        for mid_color in range(3):
            if color != mid_color:
                min_cost = min(min_cost, dp[n//2 - 1][color] + dp[n - 1][mid_color])
    return min_cost
```
