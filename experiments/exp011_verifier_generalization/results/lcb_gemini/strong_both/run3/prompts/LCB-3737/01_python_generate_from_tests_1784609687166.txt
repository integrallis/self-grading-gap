from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # Dynamic programming array to store minimum costs
    dp = [[0] * 3 for _ in range(n)]

    # Initialize the first house costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    for i in range(1, n):
        for j in range(3):
            # Minimum cost to paint the i-th house with color j
            dp[i][j] = cost[i][j] + min(
                dp[i-1][(j+1) % 3], 
                dp[i-1][(j+2) % 3], 
                dp[n-1-i][(j+1) % 3], 
                dp[n-1-i][(j+2) % 3] 
            )

    # Calculate the minimum costs for the last house considering equidistant constraints
    return min(dp[n-1])
```
