from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # Dynamic programming arrays for the two potential colorings
    dp1 = [[0] * 3 for _ in range(n)]  # Case 1: First house can be any color
    dp2 = [[0] * 3 for _ in range(n)]  # Case 2: First house is color 0

    # Initialize the first house costs
    for j in range(3):
        dp1[0][j] = cost[0][j]
    dp2[0][0] = cost[0][0]  # only color 0 for the second case
    dp2[0][1] = float('inf')  # invalid
    dp2[0][2] = float('inf')  # invalid

    for i in range(1, n):
        for j in range(3):
            # Coloring the i-th house in case 1
            dp1[i][j] = min(dp1[i-1][(j+1)%3], dp1[i-1][(j+2)%3]) + cost[i][j]
            # Coloring the i-th house in case 2
            if i < n // 2:
                dp2[i][0] = min(dp2[i-1][1], dp2[i-1][2]) + cost[i][0]
                dp2[i][1] = float('inf')
                dp2[i][2] = float('inf')
            else:
                dp2[i][0] = float('inf')
                dp2[i][1] = min(dp2[i-1][0], dp2[i-1][2]) + cost[i][1]
                dp2[i][2] = min(dp2[i-1][0], dp2[i-1][1]) + cost[i][2]

    # Calculate the minimum costs for the two cases
    total_cost_case1 = min(dp1[n-1])
    total_cost_case2 = min(dp2[n-1])

    return min(total_cost_case1, total_cost_case2)
```
