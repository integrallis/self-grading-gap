## Implementation

```python
from typing import List


def largestInteger(nums: List[int], k: int) -> int:
    from collections import defaultdict
    subarray_count = defaultdict(int)
    n = len(nums)

    for i in range(n - k + 1):
        unique_numbers = set(nums[i:i + k])
        for number in unique_numbers:
            subarray_count[number] += 1

    almost_missing = [num for num, count in subarray_count.items() if count == 1]
    return max(almost_missing) if almost_missing else -1
```
