from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    durations = [endTime[i] - startTime[i] for i in range(n)]

    # Calculate initial free times between meetings
    free_times = []
    free_times.append(startTime[0])  # Free time before the first meeting

    for i in range(1, n):
        free_times.append(startTime[i] - endTime[i - 1])  # Free time between meetings

    free_times.append(eventTime - endTime[-1])  # Free time after the last meeting

    # Initial maximum free time without any moves
    max_free_time = max(free_times)

    # If k >= n, we can move all meetings to create one big free time
    if k >= n:
        total_duration = sum(durations)
        return eventTime - total_duration

    # We will consider moving up to k meetings to maximize the free time
    max_possible_free_time = 0
    for i in range(n - k):  # Iterate through initial gaps
        current_free_time = free_times[i]
        for j in range(1, k + 1):  # Combine with next k gaps
            if i + j < len(free_times):
                current_free_time += free_times[i + j]
        max_possible_free_time = max(max_possible_free_time, current_free_time)

    return max(max_free_time, max_possible_free_time)
