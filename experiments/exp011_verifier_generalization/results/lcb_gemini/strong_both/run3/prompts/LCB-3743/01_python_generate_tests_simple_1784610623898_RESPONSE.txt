import pytest
from typing import List

from solution import maxFreeTime


def test_canonical_example_1():
    """
    Test with the first example from the specification.
    Input: eventTime = 5, k = 1, startTime = [1,3], endTime = [2,5]
    Output: 2
    Explanation: Reschedule the meeting at [1, 2] to [2, 3], leaving no meetings during the time [0, 2].
    """
    eventTime = 5
    k = 1
    startTime = [1, 3]
    endTime = [2, 5]
    
    # Calculation:
    # n = 2, k = 1. Meetings: M0=[1,2] (d=1), M1=[3,5] (d=2).
    # Initial free times: [0,1] (len 1), [2,3] (len 1). Max initial free time = 1.
    # We can move k=1 meeting.
    # Option 1: Move M0. The space available for M0 is between the event start and M1's start: [0, startTime[1]] = [0,3].
    #   Duration of M0 is 1. We can pack M0 at the end of its available space: [2,3].
    #   This leaves free time [0,2], length 2. M1 is not affected. Total moves = 1 <= k.
    #   New free time: max(len([0,2])) = 2.
    # Option 2: Move M1. The space available for M1 is between M0's end and the event end: [endTime[0], eventTime] = [2,5].
    #   Duration of M1 is 2. Pack M1 at start of this space: [2,4]. Leaves free time [4,5], length 1.
    # Maximum free time is 2.
    assert maxFreeTime(eventTime, k, startTime, endTime) == 2

    # We can move k=1 meeting.

    # Option 2: Move M1. Space: [endTime[0], startTime[2]]=[1,9]. M1 duration=2. Move to [1,3]. Free time [3,9], len 6.
    # Option 3: Move M2. Space: [endTime[1], eventTime]=[4,10]. M2 duration=1. Move to [4,5]. Free time [5,10], len 5.

    assert maxFreeTime(eventTime, k, startTime, endTime) == 6

def test_canonical_example_3():
    """
    Test with the third example from the specification.
    Input: eventTime = 5, k = 2, startTime = [0,1,2,3,4], endTime = [1,2,3,4,5]
    Output: 0
    Explanation: There is no time during the event not occupied by meetings.
    """
    eventTime = 5
    k = 2
    startTime = [0, 1, 2, 3, 4]
    endTime = [1, 2, 3, 4, 5]

    # Calculation:
    # n = 5. Meetings are perfectly packed, filling the entire [0,5] event time.
    # Total duration of meetings is 5. Event time is 5.
    # No matter how we reschedule, the total time occupied by meetings is 5.
    # The maximum free time is eventTime - total_meeting_duration = 5 - 5 = 0.
    assert maxFreeTime(eventTime, k, startTime, endTime) == 0

def test_all_meetings_movable():
    """
    Test case where k=n, so all meetings can be moved.
    The maximum free time should be the eventTime minus the total duration of all meetings.
    """
    eventTime = 100
    k = 3
    startTime = [10, 20, 30]
    endTime = [15, 25, 35]
    n = 3

    # Calculation:
    # Durations: d0=5, d1=5, d2=5. Total duration = 15.
    # Since all meetings can be moved (k=n=3), they can be packed contiguously.
    # For example, at the beginning: [0,5], [5,10], [10,15].
    # This leaves the interval [15, 100] free, which has a length of 85.
    # Expected: eventTime - total_duration = 100 - 15 = 85.
    assert maxFreeTime(eventTime, k, startTime, endTime) == 85

def test_no_gaps_between_some_meetings():
    """
    Test where some meetings are already packed together, leaving no gap.
    """
    eventTime = 20
    k = 1
    startTime = [5, 10, 15]
    endTime = [10, 15, 19]
    n = 3

    # Calculation:
    # Meetings: M0=[5,10](d=5), M1=[10,15](d=5), M2=[15,19](d=4).
    # Initial free times: [0,5](len 5), [10,10](len 0), [15,15](len 0), [19,20](len 1). Max initial = 5.
    # k=1.
    # Move M0: space [0, s1]=[0,10]. d0=5. Free time = 10 - 0 - 5 = 5.
    # Move M1: space [e0, s2]=[10,15]. d1=5. Free time = 15 - 10 - 5 = 0.

    assert maxFreeTime(eventTime, k, startTime, endTime) == 5

def test_large_k_but_not_all():
    """
    Test with k being a large portion of n, but less than n. The best strategy might involve
    moving a block of meetings smaller than k.
    """
    eventTime = 100
    k = 3
    startTime = [10, 20, 30, 90]
    endTime = [15, 25, 35, 95]
    n = 4

    # Calculation:
    # d=[5,5,5,5]. Initial max free time: gap [35,90] has length 55.
    # We can move a contiguous block of j meetings (1<=j<=k).
    # The max free time is the maximum of free times created by moving blocks of size j=1, j=2, and j=3.
    # j=1: Max free time by moving M3 is 100 - e2 - d3 = 100 - 35 - 5 = 60.
    # j=2: Max free time by moving M1,M2 is s3-e0-(d1+d2) = 90-15-10 = 65.
    #      Max free time by moving M2,M3 is 100-e1-(d2+d3) = 100-25-10 = 65.
    # j=3: Max free time by moving M0,M1,M2 is s3-0-(d0+d1+d2) = 90-0-15 = 75.
    # The overall max is max(initial_55, 60, 65, 75) = 75.
    assert maxFreeTime(eventTime, k, startTime, endTime) == 75
