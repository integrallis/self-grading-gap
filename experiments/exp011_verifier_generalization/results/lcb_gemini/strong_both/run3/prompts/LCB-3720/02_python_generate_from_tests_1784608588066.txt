from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Sort the edges based on weight.
    edges.sort(key=lambda x: x[2])
    left, right = 1, 10**6
    result = -1

    while left <= right:
        mid = (left + right) // 2
        # Create a graph with max weight <= mid
        graph = [[] for _ in range(n)]
        for u, v, w in edges:
            if w <= mid:
                graph[u].append((v, w))

        # Check if we can reach node 0 from all nodes with a limited number of outgoing edges
        out_degrees = [0] * n
        reachable = set()

        # Perform DFS from node 0
        def dfs(node):
            if node in reachable:
                return
            reachable.add(node)
            for neighbor, _ in graph[node]:
                if out_degrees[node] < threshold:
                    out_degrees[node] += 1
                    dfs(neighbor)

        dfs(0)

        # Check if all nodes can reach node 0
        if len(reachable) == n:
            result = mid
            right = mid - 1
        else:
            left = mid + 1

    return result
```
