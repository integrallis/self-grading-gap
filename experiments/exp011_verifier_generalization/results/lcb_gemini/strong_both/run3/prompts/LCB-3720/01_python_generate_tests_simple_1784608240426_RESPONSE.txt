def test_threshold_forces_higher_weight():
    # spec: A lower weight option is available but creates an impossible-to-resolve subproblem, forcing a higher weight solution.
    n = 4
    edges = [[1, 2, 10], [1, 3, 1], [2, 0, 1], [3, 0, 1]]
    threshold = 1
    # Max weight 1: edges [1,3,1], [2,0,1], [3,0,1]. Node 1 cannot reach node 2.
    # With these edges, it's impossible for node 1 to have a path to 0. (Path check fails).
    # Max weight 10: all edges included. All nodes can reach 0.
    # With threshold=1, node 1 can pick either 1->2 or 1->3. Both choices lead to a valid solution.
    # Minimum max weight is therefore 10.
    assert minMaxWeight(n, edges, threshold) == 10
