from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    edges.sort(key=lambda x: x[2])
    left, right = 1, 10**6
    result = -1

    while left <= right:
        mid = (left + right) // 2
        # Create a graph with max weight <= mid
        graph = [[] for _ in range(n)]
        for u, v, w in edges:
            if w <= mid:
                graph[u].append((v, w))

        # Check if all nodes can reach node 0 with a limited number of outgoing edges
        out_degrees = [0] * n
        reachable = set()

        # Perform DFS to check reachability of node 0
        def dfs(node):
            if node in reachable:
                return
            reachable.add(node)
            for neighbor, _ in graph[node]:
                if out_degrees[node] < threshold:
                    out_degrees[node] += 1
                    dfs(neighbor)

        # Start DFS from all nodes except 0 to check if they can reach 0
        for node in range(1, n):
            if node not in reachable:
                dfs(node)

        # Check if all nodes can reach node 0
        if len(reachable) == n - 1:
            result = mid
            right = mid - 1
        else:
            left = mid + 1

    return result

```
