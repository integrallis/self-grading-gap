## Implementation

```python
from typing import List

def maxAdjacentDistance(nums: List[int]) -> int:
    max_distance = 0
    length = len(nums)
    for i in range(length):
        # Get the adjacent indices in a circular manner
        next_index = (i + 1) % length
        # Calculate the absolute difference
        distance = abs(nums[i] - nums[next_index])
        # Update the maximum distance found
        max_distance = max(max_distance, distance)
    return max_distance
```
