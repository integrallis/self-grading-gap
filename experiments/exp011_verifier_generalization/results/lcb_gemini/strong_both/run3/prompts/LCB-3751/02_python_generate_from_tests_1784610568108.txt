from typing import List


def maxFrequency(nums: List[int], k: int) -> int:
    # Frequency map to count occurrences of each number
    freq = [0] * 51  # Since 1 <= nums[i], k <= 50
    for num in nums:
        freq[num] += 1

    initial_freq = freq[k]  # Initial frequency of k
    max_gain = 0

    # Now we need to find the best value to convert to k
    for v in range(1, 51):  # Check all possible values from 1 to 50
        if freq[v] > 0:
            # Calculate gain using a sliding window approach
            gain = [0] * len(nums)
            for i in range(len(nums)):
                if nums[i] == v:
                    gain[i] = 1  # Gain for converting v to k
                elif nums[i] == k:
                    gain[i] = -1  # Loss for converting k to k

            # Find maximum subarray sum using Kadane's algorithm
            current_sum = 0
            for g in gain:
                current_sum += g
                if current_sum < 0:
                    current_sum = 0
                max_gain = max(max_gain, current_sum)

    return initial_freq + max_gain

```
