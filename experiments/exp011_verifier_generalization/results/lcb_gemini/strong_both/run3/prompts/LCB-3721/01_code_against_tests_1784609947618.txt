from typing import List

def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers
    online_status = [True] * numberOfUsers
    event_queue = []

    # Process events and sort by timestamp
    for event in events:
        event_queue.append((int(event[1]), event))
    event_queue.sort()

    # Process sorted events
    for timestamp, event in event_queue:
        if event[0] == "OFFLINE":
            user_id = int(event[2])
            online_status[user_id] = False
            # User comes back online after 60 time units
            return_time = timestamp + 60
            # Schedule the user to come back online
            if return_time < 100000:
                event_queue.append((return_time, ["ONLINE", return_time, user_id]))
        elif event[0] == "MESSAGE":
            mentions_string = event[2]
            if mentions_string == "ALL":
                # Mention all users, regardless of status
                for i in range(numberOfUsers):
                    mentions[i] += 1
            elif mentions_string == "HERE":
                # Mention only online users
                for i in range(numberOfUsers):
                    if online_status[i]:
                        mentions[i] += 1
            else:
                # Mention specific users
                mentioned_ids = mentions_string.split()
                for mentioned_id in mentioned_ids:
                    if mentioned_id.startswith("id"):
                        user_id = int(mentioned_id[2:])
                        mentions[user_id] += 1

        # Process any scheduled online events
        for i in range(len(event_queue)):  
            if event_queue[i][0] <= timestamp and event_queue[i][1][0] == "ONLINE":
                user_id = int(event_queue[i][1][2])
                online_status[user_id] = True

    return mentions
