from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a dictionary to store coin counts at each position.
    coin_map = {}

    # Fill the coin_map with the given segments.
    for l, r, c in coins:
        for position in range(l, r + 1):
            coin_map[position] = coin_map.get(position, 0) + c

    # Create a sorted list of positions.
    positions = sorted(coin_map.keys())
    max_coins = 0
    n = len(positions)

    # We will use a sliding window approach to find the maximum coins in k consecutive bags.
    left = 0
    current_sum = 0

    for right in range(n):
        current_sum += coin_map[positions[right]]

        # Move left pointer to maintain the window size of k.
        while positions[right] - positions[left] + 1 > k:
            current_sum -= coin_map[positions[left]]
            left += 1

        # Only consider the sum if we have exactly k bags.
        if positions[right] - positions[left] + 1 == k:
            max_coins = max(max_coins, current_sum)

    return max_coins
```
