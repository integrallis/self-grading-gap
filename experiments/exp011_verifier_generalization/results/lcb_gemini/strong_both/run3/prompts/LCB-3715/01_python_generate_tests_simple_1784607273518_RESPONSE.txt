def test_canonical_example_1():
    """
    Tests the first example from the specification.
    The optimal window [3, 6] spans a gap and parts of two segments.
    """
    coins = [[8, 10, 1], [1, 3, 2], [5, 6, 4]]
    k = 4
    # Explanation:
    # The coin distribution is:
    # - Bags 1, 2, 3 have 2 coins each.
    # - Bags 5, 6 have 4 coins each.
    # - Bags 8, 9, 10 have 1 coin each.
    # - Other bags have 0 coins.
    # A window of size 4 from position 3 to 6 ([3, 4, 5, 6]) yields:
    # Bag 3: 2 coins (from [1, 3, 2])
    # Bag 4: 0 coins
    # Bag 5: 4 coins (from [5, 6, 4])
    # Bag 6: 4 coins (from [5, 6, 4])
    # Total: 2 + 0 + 4 + 4 = 10.
    assert maximumCoins(coins, k) == 10
