```python
from typing import List

def minimumPairRemoval(nums: List[int]) -> int:
    operations = 0
    while True:
        # Find pairs of adjacent elements
        pairs = [(nums[i] + nums[i + 1], i) for i in range(len(nums) - 1)]
        # Get the minimum sum pair and its index
        min_pair_sum, min_index = min(pairs, default=(float('inf'), -1))
        # If the array is non-decreasing, break
        if all(nums[i] <= nums[i + 1] for i in range(len(nums) - 1)):
            break
        # Replace the minimum sum pair with their sum
        nums[min_index] = min_pair_sum
        del nums[min_index + 1]  # Remove the next element in the pair
        operations += 1
    return operations
```
