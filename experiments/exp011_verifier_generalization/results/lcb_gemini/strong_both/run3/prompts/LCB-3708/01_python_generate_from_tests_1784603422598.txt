from typing import List

def zigzagTraversal(grid: List[List[int]]) -> List[int]:
    result = []
    for i in range(len(grid)):
        if i % 2 == 0:
            # Traverse right
            row = grid[i]
        else:
            # Traverse left
            row = grid[i][::-1]
        # Skip every alternate element by index from the continuous list
        start_index = 0 if i % 2 == 0 else 1
        result.extend(row[j] for j in range(start_index, len(row), 2))
    return result
```
