from typing import List

def zigzagTraversal(grid: List[List[int]]) -> List[int]:
    result = []
    for i in range(len(grid)):
        if i % 2 == 0:
            # Traverse right
            row = grid[i]
        else:
            # Traverse left
            row = grid[i][::-1]
        # Skip every alternate element
        result.extend(row[j] for j in range(len(row)) if j % 2 == 0)
    return result

```
