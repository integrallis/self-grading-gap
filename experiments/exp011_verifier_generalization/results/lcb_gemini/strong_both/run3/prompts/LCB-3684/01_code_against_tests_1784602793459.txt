## Implementation

```python
def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]
    # Iterate through the string to find a matching substring
    for i in range(len(s) - len(prefix) - len(suffix) + 1):
        # Check if the prefix matches
        if s[i:i + len(prefix)] == prefix:
            # Check if the suffix matches starting from the end of the prefix
            if s[i + len(prefix):].find(suffix) != -1:
                return True
    return False
```
