from typing import List

def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers
    online_status = [True] * numberOfUsers
    offline_until = [0] * numberOfUsers

    events.sort(key=lambda x: int(x[1]))  # Sort events by timestamp

    for event in events:
        event_type, timestamp, detail = event
        timestamp = int(timestamp)

        if event_type == "OFFLINE":
            user_id = int(detail)
            online_status[user_id] = False
            offline_until[user_id] = timestamp + 60

        elif event_type == "MESSAGE":
            # Process offline users coming back online
            for user_id in range(numberOfUsers):
                if offline_until[user_id] <= timestamp:
                    online_status[user_id] = True

            mentions_to_count = set()
            if detail == "HERE":
                mentions_to_count = {i for i in range(numberOfUsers) if online_status[i]}
            elif detail == "ALL":
                mentions_to_count = {i for i in range(numberOfUsers)}
            else:
                for mention in detail.split():
                    if mention.startswith("id"):
                        user_id = int(mention[2:])
                        mentions_to_count.add(user_id)

            for user_id in mentions_to_count:
                mentions[user_id] += 1

    return mentions
```
