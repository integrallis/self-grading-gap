def test_canonical_example_1():
    """

    """
    numberOfUsers = 2
    events = [
        ["MESSAGE", "71", "HERE"],  # Should be processed last
        ["OFFLINE", "11", "0"],     # Should be processed second
        ["MESSAGE", "10", "id1 id0"], # Should be processed first
    ]
    # Expected chronological processing:
    # T=10: MESSAGE "id1 id0". counts: [1, 1]
    # T=11: OFFLINE "0". User 0 offline until T=71.
    # T=71: User 0 back online, then MESSAGE "HERE" mentions both. counts: [2, 2]
    expected = [2, 2]
    assert countMentions(numberOfUsers, events) == expected


def test_simultaneous_offline_and_here_message():
    """

    expected = [1, 0, 1]
    assert countMentions(numberOfUsers, events) == expected


def test_simultaneous_online_and_here_message():
    """
    Tests that a user coming back online at timestamp T is included in a "HERE" mention at T.
    """
    numberOfUsers = 2
    events = [["OFFLINE", "10", "0"], ["MESSAGE", "70", "HERE"]]
    # At T=10, user 0 goes offline. Comes back online at T=10+60=70.
    # At T=70, the status change (user 0 back online) happens first.
    # Both users are online.
    # Then, MESSAGE "HERE" mentions both users 0 and 1.
    expected = [1, 1]
    assert countMentions(numberOfUsers, events) == expected

def test_complex_scenario():
    """
    A more complex test with multiple users, multiple event types, and interleaved statuses.
    """
    numberOfUsers = 4
    events = [
        ["MESSAGE", "10", "id0 id1"],         # T=10: [1,1,0,0]
        ["OFFLINE", "15", "0"],               # u0 offline until 75
        ["OFFLINE", "20", "2"],               # u2 offline until 80
        ["MESSAGE", "25", "HERE"],            # T=25: HERE mentions u1, u3. -> [1,2,0,1]
        ["MESSAGE", "30", "ALL"],             # T=30: ALL mentions all. -> [2,3,1,2]
        ["MESSAGE", "75", "id0 HERE"],        # T=75: u0 online. HERE=u0,u1,u3. id0. -> [4,4,1,3]
        ["OFFLINE", "76", "1"],               # u1 offline until 136
        ["MESSAGE", "80", "HERE"],            # T=80: u2 online. HERE=u0,u2,u3. -> [5,4,2,4]
    ]
    # Tracing the counts:
    # Initial: [0,0,0,0]
    # T=10: msg(id0, id1) -> [1,1,0,0]
    # T=15: u0 offline
    # T=20: u2 offline
    # T=25: msg(HERE -> u1,u3) -> [1, 1+1, 0, 0+1] = [1,2,0,1]
    # T=30: msg(ALL -> u0,u1,u2,u3) -> [1+1, 2+1, 0+1, 1+1] = [2,3,1,2]
    # T=75: u0 online. msg(id0, HERE->u0,u1,u3) -> [2+2, 3+1, 1, 2+1] = [4,4,1,3]
    # T=76: u1 offline
    # T=80: u2 online. msg(HERE->u0,u2,u3) -> [4+1, 4, 1+1, 3+1] = [5,4,2,4]
    expected = [5, 4, 2, 4]
    assert countMentions(numberOfUsers, events) == expected
