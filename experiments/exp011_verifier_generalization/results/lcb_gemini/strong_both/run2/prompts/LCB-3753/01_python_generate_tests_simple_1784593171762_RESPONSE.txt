    s = "aaaaabbc"

    # Frequencies: 'a': 5 (odd), 'b': 2 (even), 'c': 1 (odd)
    # Odd frequencies: {5, 1}
    # Even frequencies: {2}
    # To maximize (odd_freq - even_freq), we take max(odd_freqs) - min(even_freqs).
