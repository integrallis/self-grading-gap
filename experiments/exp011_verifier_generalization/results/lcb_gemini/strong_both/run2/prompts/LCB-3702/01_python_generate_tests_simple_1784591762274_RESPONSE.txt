from typing import List

from solution import maxLength


def test_canonical_example_1():
    """
    Test with the first example from the specification.
    Input: nums = [1,2,1,2,1,1,1]
    Output: 5
    Explanation: The longest product equivalent subarray is [1, 2, 1, 1, 1].
    For this subarray, prod=2, gcd=1, lcm=2. The condition 2 == 1 * 2 holds.
    It has length 5. Any longer subarray is not product equivalent.
    """
    nums = [1, 2, 1, 2, 1, 1, 1]
    expected = 5
    assert maxLength(nums) == expected

def test_all_ones():
    """
    Test with an array containing only 1s.
    For any subarray of [1, 1, 1, 1, 1], prod=1, gcd=1, and lcm=1.
    The condition 1 == 1 * 1 always holds.
    Therefore, any subarray is product equivalent. The longest is the entire array.
    """
    nums = [1, 1, 1, 1, 1]
    expected = 5
    assert maxLength(nums) == expected


def test_no_subarray_longer_than_two():
    """
    Test with an array where no subarray of length > 2 is product equivalent.
    Any subarray of length 2 is product equivalent. For [2, 4, 6, 8],
    subarrays of length > 2 like [2,4,6] have a gcd > 1 (gcd=2), which makes them
    not product equivalent.
    """
    nums = [2, 4, 6, 8]
    expected = 2
    assert maxLength(nums) == expected


def test_minimum_input_length():
    """
    Test with an array of the minimum allowed length (2).
    Any subarray of length 2 is product equivalent.
    The longest possible subarray is the array itself, length 2.
    """
    nums = [9, 10]
    expected = 2
    assert maxLength(nums) == expected


def test_maximum_input_length_all_ones():
    """
    Test with an array of the maximum allowed length (100) containing only 1s.
    As with the all_ones test, any subarray is product equivalent.

    assert maxLength(nums) == expected


def test_long_product_equivalent_subarray():
    """
    Test with an array containing a long product equivalent subarray.
    The subarray [7, 1, 5, 1, 9, 1, 8] has length 7.
    Its gcd=1, and its elements greater than 1 ({7, 5, 9, 8}) are pairwise coprime.
    This makes it product equivalent. No longer subarray is product equivalent.
    """
    nums = [10, 7, 1, 5, 1, 9, 1, 8, 2]
    expected = 7
    assert maxLength(nums) == expected


def test_mixed_numbers_short_result():
    """
    Test with a mixed array where the longest product equivalent subarray is short.
    Subarrays like [3, 4, 5], [3, 2, 1], and [1, 2, 3] are product equivalent
    and have length 3. No subarray of length 4 or more is product equivalent.
    For example, for [5,4,3,2], prod=120, gcd=1, lcm=60. 120 != 60 * 1.
    """
    nums = [6, 5, 4, 3, 2, 1, 2, 3, 4, 5, 6]
    expected = 3
    assert maxLength(nums) == expected


def test_subarray_with_only_primes_and_one():
    """
    Test a subarray that is product equivalent due to being composed of primes.
    The subarray [1, 2, 3, 5, 7] has gcd=1, and its elements > 1 are distinct primes,
    making them pairwise coprime. This is a product equivalent subarray of length 5.
    """
    nums = [1, 2, 3, 5, 7, 4]
    # The subarray [1, 2, 3, 5, 7] is length 5 and product equivalent.
    # The subarray [2, 3, 5, 7, 4] is not, as 2 and 4 share a prime factor.
    expected = 5
    assert maxLength(nums) == expected
