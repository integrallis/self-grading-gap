```python
from typing import List


def maxFrequency(nums: List[int], k: int) -> int:
    nums.sort()  # Sort the array to facilitate the sliding window approach
    left = 0  # Left pointer for the sliding window
    total = 0  # Total number of additions needed to make elements equal to k
    max_freq = 0  # Maximum frequency of k possible

    for right in range(len(nums)):
        total += nums[right]  # Sum of elements in the current window
        # Calculate the cost to convert all elements in the current window to k
        while nums[right] * (right - left + 1) - total > k:
            total -= nums[left]  # Remove the leftmost element from the total
            left += 1  # Move the left pointer right
        # Update the maximum frequency of k
        max_freq = max(max_freq, right - left + 1)

    return max_freq

```
