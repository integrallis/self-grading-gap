```python
from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    nums.sort()
    total_sum = 0

    # Calculate contributions for each element as min and max
    for i in range(n):
        # How many subsequences can be formed with nums[i] as max
        for j in range(1, k + 1):
            if i >= j - 1:
                max_contrib = nums[i] * (pow(2, j - 1, MOD)) % MOD
                total_sum = (total_sum + max_contrib) % MOD

        # How many subsequences can be formed with nums[i] as min
        for j in range(1, k + 1):
            if n - i - 1 >= j - 1:
                min_contrib = nums[i] * (pow(2, j - 1, MOD)) % MOD
                total_sum = (total_sum + min_contrib) % MOD

    return total_sum % MOD

```
