from typing import List
import collections


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    edges.sort(key=lambda x: x[2])  # Sort edges by weight
    low, high = 1, 10**6  # The weights are in the range [1, 10^6]

    def canAchieve(max_weight: int) -> bool:
        graph = collections.defaultdict(list)
        out_degree = [0] * n
        for u, v, w in edges:
            if w <= max_weight:
                graph[u].append(v)
                out_degree[u] += 1
                if out_degree[u] > threshold:
                    return False
        visited = [False] * n
        queue = collections.deque([0])
        visited[0] = True
        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    queue.append(neighbor)
        return all(visited[i] for i in range(1, n))

    while low < high:
        mid = (low + high) // 2
        if canAchieve(mid):
            high = mid
        else:
            low = mid + 1

    return -1 if low > 10**6 else low
