```python
from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    durations = [endTime[i] - startTime[i] for i in range(n)]

    # Calculate initial gaps
    gaps = [startTime[0]] + [startTime[i] - endTime[i - 1] for i in range(1, n)] + [eventTime - endTime[-1]]
    max_gap = max(gaps)

    # Create pairs of (duration, gap) to consider moves
    moves = sorted([(durations[i], gaps[i + 1]) for i in range(n - 1)], key=lambda x: x[0])

    # We can use k moves to reduce the largest gaps
    total_duration = sum(durations)
    for i in range(min(k, n - 1)):
        total_duration -= moves[i][0]
        max_gap = max(max_gap, moves[i][1] + total_duration)

    return max_gap
```
