import pytest
from solution import hasMatch

# Note: The problem statement is the sole source of truth.
# The tests below are designed to codify the specification's behavior.
# "A substring is a contiguous non-empty sequence of characters within a string."

# ==============================================================================
# Canonical Examples from Specification
# ==============================================================================


def test_canonical_example_1():
    # spec: hasMatch("leetcode", "ee*e") -> true
    # Explanation: By replacing the '*' with "tcod", p becomes "eetcode".
    # "eetcode" is a substring of s (s[1:]).
    assert hasMatch("leetcode", "ee*e") is True


def test_canonical_example_2():
    # spec: hasMatch("car", "c*v") -> false
    # Explanation: There is no substring matching the pattern.
    # The prefix "c" is at s[0], but the suffix "v" is not found in s[1:].
    assert hasMatch("car", "c*v") is False


def test_canonical_example_3():
    # spec: hasMatch("luck", "u*") -> true
    # Explanation: The substrings "u", "uc", and "uck" match the pattern.
    # The substring "u" from s matches p="u*" where "*" is an empty sequence.
    assert hasMatch("luck", "u*") is True


# ==============================================================================
# Additional Tests for Star Position
# ==============================================================================


def test_star_at_beginning_match():
    # p = "*end", s = "theend". The substring "theend" matches with *="the".
    assert hasMatch("theend", "*end") is True


def test_star_at_beginning_empty_wildcard():
    # p = "*end", s = "end". The substring "end" matches with *="".
    assert hasMatch("end", "*end") is True


def test_star_at_end_match():
    # p = "start*", s = "startoftheshow". The substring "startoftheshow" matches with *="oftheshow".
    assert hasMatch("startoftheshow", "start*") is True


def test_star_at_end_empty_wildcard():
    # p = "start*", s = "start". The substring "start" matches with *="".
    assert hasMatch("start", "start*") is True


def test_star_in_middle_empty_wildcard():
    # p = "a*c", s = "ac". The substring "ac" matches with *="".
    assert hasMatch("ac", "a*c") is True


def test_star_in_middle_single_char_wildcard():
    # p = "a*c", s = "abc". The substring "abc" matches with *="b".
    assert hasMatch("abc", "a*c") is True


def test_star_in_middle_long_wildcard():
    # p = "a*z", s = "abcdefghijklmnopqrstuvwxyz". The substring is the whole of s. *="bc...y".
    assert hasMatch("abcdefghijklmnopqrstuvwxyz", "a*z") is True


# ==============================================================================
# Tests for No-Match Scenarios
# ==============================================================================


def test_no_match_prefix_not_found():
    # The prefix "x" of p="x*a" is not in s="banana".
    assert hasMatch("banana", "x*a") is False


def test_no_match_suffix_not_found():
    # The suffix "x" of p="b*x" is not in s="banana".
    assert hasMatch("banana", "b*x") is False


def test_no_match_suffix_before_prefix():
    # In s="cbda", the prefix "a" is at index 3. The suffix "b" must be found after s[3].
    # The "b" at index 1 is not part of a valid match.
    assert hasMatch("cbda", "a*b") is False


def test_no_match_pattern_longer_than_string():
    # The non-wildcard part of p ("abcd") is longer than s ("abc").
    assert hasMatch("abc", "ab*cd") is False


def test_no_match_s_has_chars_but_not_in_order():
    # s has the letters 'z' and 'a', but 'z' does not appear after 'a' in any substring.
    assert hasMatch("zyxw", "a*z") is False


# ==============================================================================
# Edge Cases and Other Scenarios
# ==============================================================================


def test_pattern_is_only_a_star():
    # p="*". '*' can be replaced by any non-empty substring of s.
    # Since s is non-empty, we can always find one (e.g., s itself).
    assert hasMatch("anything", "*") is True


def test_minimal_strings_match():
    # s="a", p="a*". The substring "a" matches with *="".
    assert hasMatch("a", "a*") is True


def test_minimal_strings_no_match():
    # s="a", p="b*". The prefix "b" is not in s.
    assert hasMatch("a", "b*") is False


def test_overlapping_prefix_and_suffix_in_pattern():
    # s="aa", p="a*a". The substring "aa" matches with *="".
    assert hasMatch("aa", "a*a") is True


def test_non_overlapping_prefix_and_suffix_but_close():
    # s="aba", p="a*a". The substring "aba" matches with *="b".
    assert hasMatch("aba", "a*a") is True


def test_no_match_with_single_char_and_overlapping_pattern():
    # s="a", p="a*a". p_prefix="a", p_suffix="a".
    # The prefix "a" is at s[0]. The suffix "a" must be found at or after index 1, but is not.
    assert hasMatch("a", "a*a") is False


def test_multiple_prefix_candidates_in_s():
    # s="mississippi", p="issi*i"
    # p_prefix="issi", p_suffix="i"
    # The first "issi" is at index 1. We search for "i" from index 1+4=5. Found at index 7. Match.
    assert hasMatch("mississippi", "issi*i") is True


def test_long_string_and_pattern_match():
    # s has y at index 24 and b at index 27. The substring s[24:28] is "yzab", which matches "y*b".
    s = "abcdefghijklmnopqrstuvwxyz" * 2
    p = "y*b"
    assert hasMatch(s, p) is True


def test_long_string_and_pattern_no_match():
    # In s, z is at the end and a is at the start. They are not in the correct order in any substring.
    s = "abcdefghijklmnopqrstuvwxyz"
    p = "z*a"
    assert hasMatch(s, p) is False
