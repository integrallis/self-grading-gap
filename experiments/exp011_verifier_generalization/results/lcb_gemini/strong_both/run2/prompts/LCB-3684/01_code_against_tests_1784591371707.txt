## Implementation

```python
def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]
    prefix_length = len(prefix)
    suffix_length = len(suffix)
    for i in range(len(s) - prefix_length - suffix_length + 1):
        if s[i:i + prefix_length] == prefix:
            if s[i + prefix_length:i + prefix_length + suffix_length] == suffix:
                return True
            if s[i + prefix_length:].endswith(suffix):
                return True
    return False
```
