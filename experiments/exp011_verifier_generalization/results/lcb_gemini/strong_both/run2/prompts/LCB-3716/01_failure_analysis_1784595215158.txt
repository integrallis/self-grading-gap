from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = [1] * n
    max_length = 1

    for i in range(n):
        for j in range(i):
            diff1 = abs(nums[i] - nums[j])
            valid = True
            for k in range(j):
                diff2 = abs(nums[j] - nums[k])
                if diff1 > diff2:
                    valid = False
                    break
            if valid:
                dp[i] = max(dp[i], dp[j] + 1)
                max_length = max(max_length, dp[i])

    return max_length
