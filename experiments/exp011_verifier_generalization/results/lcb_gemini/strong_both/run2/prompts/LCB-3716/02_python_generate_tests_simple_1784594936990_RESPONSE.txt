def test_canonical_example_1():
    # spec: longestSubsequence([16, 6, 3]) -> 3
    # The longest subsequence is [16, 6, 3] with the absolute adjacent differences [10, 3].
    # The sequence of differences [10, 3] is non-increasing.
    nums = [16, 6, 3]
    expected = 3
    assert longestSubsequence(nums) == expected
