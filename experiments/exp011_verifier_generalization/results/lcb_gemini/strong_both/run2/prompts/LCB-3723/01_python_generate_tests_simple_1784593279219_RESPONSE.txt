def test_canonical_example_1():

    """

    # - nums[1] = 3: i-k=-1 (out of bounds), i+k=3. nums[3]=1. 3 > 1. Good.
    # - nums[4] = 5: i-k=2, i+k=6 (out of bounds). nums[2]=2. 5 > 2. Good.

    # - Condition is 2 > 1, which is true. All 50 are good.
