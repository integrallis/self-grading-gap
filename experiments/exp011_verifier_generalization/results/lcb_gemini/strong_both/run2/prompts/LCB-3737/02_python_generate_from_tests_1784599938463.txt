from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # Use dynamic programming to calculate the minimum cost
    dp = [[float('inf')] * 3 for _ in range(n)]

    # Initialize the first house costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    for i in range(1, n):
        for j in range(3):
            # Ensure no adjacent houses have the same color
            for k in range(3):
                if j != k:
                    dp[i][j] = min(dp[i][j], dp[i-1][k] + cost[i][j])

    # Handle equidistant constraints
    if n > 2:
        for i in range(2, n):
            for j in range(3):
                dp[i][j] = min(dp[i][j], dp[i-2][(j+1)%3] + cost[i][j], dp[i-2][(j+2)%3] + cost[i][j])

    # The answer is the minimum of the last house costs
    return min(dp[n-1])

```
