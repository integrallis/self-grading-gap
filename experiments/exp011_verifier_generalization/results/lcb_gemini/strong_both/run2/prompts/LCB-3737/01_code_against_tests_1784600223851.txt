from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # DP table
    dp = [[float('inf')] * 3 for _ in range(3)]

    # Initial fill based on the first house
    for j in range(3):
        dp[0][j] = cost[0][j]

    for i in range(1, n // 2 + 1):
        new_dp = [[float('inf')] * 3 for _ in range(3)]
        for c1 in range(3):
            for c2 in range(3):
                if c1 != c2:  # Ensure the equidistant colors are different
                    for prev_c1 in range(3):
                        if c1 != prev_c1:  # No adjacent same colors
                            new_dp[c1][c2] = min(new_dp[c1][c2], dp[prev_c1][(prev_c1 + c2) % 3] + cost[i][c1] + cost[n - 1 - i][c2])
        dp = new_dp

    # Finally, we fetch the minimum cost from the last DP table
    return min(min(dp[c1][c2] for c2 in range(3)) for c1 in range(3))
