## Implementation

```python
from typing import List

def zigzagTraversal(grid: List[List[int]]) -> List[int]:
    full_path = []
    for i in range(len(grid)):
        if i % 2 == 0:
            full_path.extend(grid[i])  # Even row: traverse left to right
        else:
            full_path.extend(reversed(grid[i]))  # Odd row: traverse right to left
    return full_path[::2]  # Return every alternate element
```
