from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[[-float('inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0]  # Initialize starting point
    if coins[0][0] < 0:
        dp[0][0][1] = 0  # Neutralize if starting cell is a robber

    for i in range(m):
        for j in range(n):
            for k in range(3):  # k indicates number of robbers neutralized
                if i == 0 and j == 0:
                    continue  # Skip starting cell
                if i > 0:
                    value = coins[i][j]  
                    # Option 1: Arrive at (i,j) and do NOT neutralize it.
                    dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k] + value)
                    # Option 2: Arrive at (i,j) and neutralize it.
                    if value < 0 and k > 0:
                        dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k-1])  # Neutralize the robber
                if j > 0:
                    value = coins[i][j]
                    # Option 1: Arrive at (i,j) and do NOT neutralize it.
                    dp[i][j][k] = max(dp[i][j][k], dp[i][j-1][k] + value)
                    # Option 2: Arrive at (i,j) and neutralize it.
                    if value < 0 and k > 0:
                        dp[i][j][k] = max(dp[i][j][k], dp[i][j-1][k-1])  # Neutralize the robber

    return max(dp[m-1][n-1])  # Return the maximum profit at the bottom-right corner
