import pytest
from solution import calculateScore

# No typing imports are needed for the function signature `def calculateScore(s: str) -> int`
# as `str` and `int` are built-in types.

def test_canonical_example_1():
    """
    Tests the first example from the specification.
    s = "aczzx"
    i=0 ('a'): No j<0. Skip.
    i=1 ('c'): No j<1 with mirror 'x'. Skip.
    i=2 ('z'): Mirror 'a'. Unmarked 'a' at j=0. Mark 0, 2. score += (2 - 0) = 2.
    i=3 ('z'): Mirror 'a'. 'a' at j=0 is marked. No other unmarked 'a'. Skip.
    i=4 ('x'): Mirror 'c'. Unmarked 'c' at j=1. Mark 1, 4. score += (4 - 1) = 3.
    Total score = 2 + 3 = 5.
    """
    assert calculateScore("aczzx") == 5

def test_canonical_example_2():
    """
    Tests the second example from the specification.
    s = "abcdef"
    No character's mirror ('z', 'y', 'x', 'w', 'v', 'u') is present in the string.
    Thus, no pairs are ever found.
    The score remains 0.
    """
    assert calculateScore("abcdef") == 0

def test_single_character():
    """
    Tests a string with a single character.
    s = "m"
    The loop for i starts at 0. The condition for finding a partner is j < i.
    No such j exists. The score is 0.
    """
    assert calculateScore("m") == 0

def test_no_mirror_pairs_present():
    """
    Tests a string where characters are present but no mirror pairs can be formed.
    s = "abcabc"
    Mirrors are 'z', 'y', 'x'. None are in the string.
    The score remains 0.
    """
    assert calculateScore("abcabc") == 0
    
def test_simple_adjacent_pair():
    """
    Tests a single, adjacent mirror pair.
    s = "az"
    i=0 ('a'): Skip.
    i=1 ('z'): Mirror 'a' at j=0. Mark 0, 1. score += (1 - 0) = 1.
    Total score = 1.
    """
    assert calculateScore("az") == 1

def test_simple_distant_pair():
    """
    Tests a single, non-adjacent mirror pair.
    s = "a...z" (e.g., "abcz")
    i=0,1,2: Skip.
    i=3 ('z'): Mirror 'a' at j=0. Mark 0, 3. score += (3 - 0) = 3.
    Total score = 3.
    """
    assert calculateScore("abcz") == 3

def test_multiple_separate_pairs():
    """
    Tests a string with multiple, non-overlapping pairs.
    s = "azby"
    i=1 ('z'): Mirror 'a' at j=0. score += (1 - 0) = 1. Marked: 0, 1.
    i=3 ('y'): Mirror 'b' at j=2. score += (3 - 2) = 1. Marked: 0, 1, 2, 3.
    Total score = 1 + 1 = 2.
    """
    assert calculateScore("azby") == 2

def test_chooses_closest_of_multiple_partners():
    """
    Tests that when multiple potential partners exist, the one with the largest index j is chosen.
    s = "abazy"
    i=3 ('z'): Mirror 'a'. Potential partners at j=0 and j=2. Closest is j=2.
               score += (3 - 2) = 1. Marked: 2, 3.
    i=4 ('y'): Mirror 'b'. Partner at j=1.
               score += (4 - 1) = 3. Marked: 1, 2, 3, 4.
    Total score = 1 + 3 = 4.
    """
    assert calculateScore("abazy") == 4

def test_ignores_already_marked_partners():
    """
    Tests that an index, once marked, cannot be used as a partner again.
    s = "azaz"
    i=1 ('z'): Mirror 'a' at j=0. score += (1 - 0) = 1. Marked: 0, 1.
    i=3 ('z'): Mirror 'a'. Partner 'a' at j=0 is marked. Unmarked partner 'a' is at j=2.
               score += (3 - 2) = 1. Marked: 0, 1, 2, 3.
    Total score = 1 + 1 = 2.
    """
    assert calculateScore("azaz") == 2

def test_interleaved_long_distance_pairs():
    """
    Tests pairs that are interleaved over longer distances.
    s = "abczyx"
    i=3 ('y'): Mirror 'b' at j=1. score += (3 - 1) = 2. Marked: 1, 3.
    i=4 ('x'): Mirror 'c' at j=2. score += (4 - 2) = 2. Marked: 1, 2, 3, 4.
    i=5 ('z'): Mirror 'a' at j=0. score += (5 - 0) = 5. Marked: 0, 1, 2, 3, 4, 5.
    Total score = 2 + 2 + 5 = 9.
    """
    assert calculateScore("abczyx") == 9

def test_interleaved_adjacent_pairs():
    """
    Tests pairs that are interleaved but adjacent to their triggers.
    s = "azbycx"
    i=1 ('z'): Mirror 'a' at j=0. score += (1-0)=1. Marked: 0, 1.
    i=3 ('y'): Mirror 'b' at j=2. score += (3-2)=1. Marked: 0, 1, 2, 3.
    i=5 ('x'): Mirror 'c' at j=4. score += (5-4)=1. Marked: 0, 1, 2, 3, 4, 5.
    Total score = 1 + 1 + 1 = 3.
    """
    assert calculateScore("azbycx") == 3
    
def test_long_string_with_no_matches():
    """
    Tests a longer string with no mirror pairs.
    s = "aaaaaaaaaaaaaaaaaaaa" (20 'a's)
    Mirror of 'a' is 'z', which is not in the string. Score is 0.
    """
    assert calculateScore("a" * 20) == 0

def test_long_string_with_one_match_at_end():
    """
    Tests a longer string with a single match at the very end.
    s = "aaaaaaaaaaz" (10 'a's, 1 'z')
    i=10 ('z'): Mirror 'a'. The closest unmarked 'a' is at j=9.
                score += (10 - 9) = 1.
    Total score = 1.
    """
    assert calculateScore("a" * 10 + "z") == 1
