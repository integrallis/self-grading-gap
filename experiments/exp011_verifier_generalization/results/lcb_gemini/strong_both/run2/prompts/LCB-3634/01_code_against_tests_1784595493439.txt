```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    for i in range(len(s)):
        if marked[i]:
            continue
        mirror = chr(ord('a') + (ord('z') - ord(s[i])))
        for j in range(i - 1, -1, -1):
            if s[j] == mirror and not marked[j]:
                marked[i] = True
                marked[j] = True
                score += i - j
                break
    return score

```
