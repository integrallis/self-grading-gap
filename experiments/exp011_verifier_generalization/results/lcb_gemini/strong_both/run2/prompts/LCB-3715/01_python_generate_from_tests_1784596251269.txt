from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    events = []
    # Create events for each coin segment
    for l, r, c in coins:
        events.append((l, c))
        events.append((r + 1, -c))

    # Sort events by position
    events.sort()

    max_coins = 0
    current_coins = 0
    last_position = None

    # Iterate through sorted events
    for position, change in events:
        if last_position is not None and last_position < position:
            # Calculate the number of bags in the range
            length = position - last_position
            if length > 0:
                # If we have a valid range, compute the max coins for window size k
                max_coins = max(max_coins, current_coins * min(length, k))

        # Update current coins and last position
        current_coins += change
        last_position = position

    return max_coins
```
