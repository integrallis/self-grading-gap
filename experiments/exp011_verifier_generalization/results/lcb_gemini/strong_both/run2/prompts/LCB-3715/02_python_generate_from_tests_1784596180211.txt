from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a dictionary to map each position to its coin count
    coin_map = {}
    for l, r, c in coins:
        for pos in range(l, r + 1):
            coin_map[pos] = coin_map.get(pos, 0) + c

    # Create a sorted list of positions
    positions = sorted(coin_map.keys())
    max_coins = 0

    # Iterate over all possible starting points for k-length windows
    for start in range(len(positions)):
        end = start
        total_coins = 0
        # Move the end pointer to find the end of the k-length window
        while end < len(positions) and positions[end] < positions[start] + k:
            total_coins += coin_map[positions[end]]
            end += 1
        # Update maximum coins found
        max_coins = max(max_coins, total_coins)

    return max_coins

```
