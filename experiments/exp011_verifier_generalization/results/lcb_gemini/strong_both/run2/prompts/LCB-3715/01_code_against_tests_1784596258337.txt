```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    events = []
    # Create events for each coin segment
    for l, r, c in coins:
        events.append((l, c))
        events.append((r + 1, -c))

    # Sort events by position
    events.sort()

    max_coins = 0
    current_coins = 0
    last_position = 0

    # Iterate through sorted events
    for position, change in events:
        # The number of bags fully covered from the last position to the current position
        length = position - last_position
        if length > 0: 
            # Total coins obtainable in the range of bags
            max_coins += (length * current_coins) if length <= k else (k * current_coins)
        # Update current coins and last position
        current_coins += change
        last_position = position

    return max_coins
```
