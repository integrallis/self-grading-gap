def maximumCoins(coins: List[List[int]], k: int) -> int:

        for pos in range(l, r + 1):
            coin_map[pos] = coin_map.get(pos, 0) + c

    # Create a sorted list of positions

    max_coins = 0

    # Iterate over all possible starting points for k-length windows
