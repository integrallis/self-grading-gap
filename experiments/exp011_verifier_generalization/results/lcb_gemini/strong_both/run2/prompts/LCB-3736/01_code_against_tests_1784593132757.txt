## Implementation

```python
from collections import Counter

def findValidPair(s: str) -> str:
    counts = Counter(s)
    for i in range(len(s) - 1):
        if s[i] != s[i + 1]:  # Check if adjacent digits are different
            d1, d2 = s[i], s[i + 1]
            if counts[d1] == int(d1) and counts[d2] == int(d2):
                return d1 + d2
    return ""
```
