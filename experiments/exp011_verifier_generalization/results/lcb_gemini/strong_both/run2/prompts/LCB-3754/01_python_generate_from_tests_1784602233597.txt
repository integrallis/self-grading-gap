from solution import maxDistance

def test_canonical_example_1():
    # spec: s = "NWSE", k = 1 -> 3

    # The max distance with this path is at the end: pos(-4, 2), dist 6.
    assert maxDistance("NSWWEW", 3) == 6

def test_no_changes_allowed():
    # s = "NESW", k = 0
    # The path is N -> (0, 1), E -> (1, 1), S -> (1, 0), W -> (0, 0).
    # The Manhattan distances are 1, 2, 1, 0. The maximum is 2.
    assert maxDistance("NESW", 0) == 2

def test_all_changes_allowed():
    # s = "WESN", k = 4. length is 4.
    # We can change all characters. To maximize distance, change to "NNNN".
    # The path becomes (0,1), (0,2), (0,3), (0,4).
    # The maximum distance is 4.
    assert maxDistance("WESN", 4) == 4

def test_k_greater_than_length():
    # s = "EW", k = 3. length is 2.
    # We can change all characters. Optimal is "EE" or "WW" etc.
    # For "EE", path is (1,0), (2,0). Max distance is 2.
    assert maxDistance("EW", 3) == 2

def test_single_move_no_change():
    # s = "S", k = 0.
    # The path is just one step to (0, -1).
    # The distance is 1.
    assert maxDistance("S", 0) == 1

def test_single_move_with_change():
    # s = "S", k = 1.
    # We can change 'S' to 'N', 'E', or 'W'.
    # The position after one move is always 1 unit from the origin.
    # The maximum distance is 1.
    assert maxDistance("S", 1) == 1

def test_unidirectional_path_no_changes_needed():
    # s = "EEEE", k = 2.
    # The original path already moves optimally in one direction.
    # Path: (1,0), (2,0), (3,0), (4,0). Distances 1, 2, 3, 4.
    # Any change would be suboptimal. The maximum distance is 4.
    assert maxDistance("EEEE", 2) == 4

def test_simple_mixed_path_with_one_change():
    # s = "EES", k = 1.
    # Change 'S' to 'E'. The string becomes "EEE".
    # Path for "EEE": (1,0), (2,0), (3,0). Distances 1, 2, 3.
    # The maximum distance is 3.
    assert maxDistance("EES", 1) == 3

def test_alternating_path_with_one_change():
    # s = "NSNS", k = 1.
    # Change s[1]='S' to 'N'. The string becomes "NNNS".
    # Path: N -> (0,1), dist 1
    #       N -> (0,2), dist 2
    #       N -> (0,3), dist 3
    #       S -> (0,2), dist 2
    # The maximum distance is 3.
    assert maxDistance("NSNS", 1) == 3

def test_cancelling_path_with_multiple_changes():
    # s = "NESWNESW", k = 2.
    # With 2 changes, we can make the path more direct.
    # E.g., change s[2]='S' to 'N' and s[3]='W' to 'E'.
    # The modified string starts with "NENEN...".
    # The path for the first 6 moves "NENENE":
    # N(0,1), E(1,1), N(1,2), E(2,2), N(2,3), E(3,3).
    # The distances are 1, 2, 3, 4, 5, 6. The max distance is 6.
    assert maxDistance("NESWNESW", 2) == 6

def test_long_string_full_changes():
    # s = "W" * 100, k = 100.
    # We can change all moves. Optimal is to make them all the same, e.g., "E"*100.
    # The final position will be (100, 0), and max distance is 100.
    # This also works if we don't change anything, since "W"*100 also yields max distance 100.
    assert maxDistance("W" * 100, 100) == 100
