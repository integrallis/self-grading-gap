import pytest
from solution import hasSameDigits


def test_canonical_example_1():
    # spec: hasSameDigits("3902") -> true
    # Step 1: "3902" -> (3+9)%10=2, (9+0)%10=9, (0+2)%10=2 -> "292"
    # Step 2: "292" -> (2+9)%10=1, (9+2)%10=1 -> "11"
    # The final two digits '1' and '1' are the same.
    assert hasSameDigits("3902") is True
