```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # Creating a mirror mapping for 'a' to 'z'

    for i in range(len(s)):
        if marked[i]:  # Skip if already marked
            continue
        for j in range(i - 1, -1, -1):  # Look for unmarked indices before i
            if not marked[j] and s[j] == mirror[s[i]]:  # Check for mirror pair
                score += i - j  # Calculate score
                marked[i] = marked[j] = True  # Mark both indices
                break  # Exit the inner loop after finding the closest pair

    return score
```
