def test_canonical_example_1():
    """
    Tests the first example from the specification.
    spec: minCosts([5, 3, 4, 1, 3, 2]) -> [5, 3, 3, 1, 1, 1]
    """
    cost = [5, 3, 4, 1, 3, 2]
    # To reach pos i, we can pay cost[k] for any k <= i and then move to i for free.
    # The minimum cost is therefore the minimum of all costs up to index i.
    # ans[0] = min(cost[0]) = 5
    # ans[1] = min(cost[0], cost[1]) = min(5, 3) = 3
    # ans[2] = min(cost[0], cost[1], cost[2]) = min(5, 3, 4) = 3
    # ans[3] = min(cost[:4]) = min(5, 3, 4, 1) = 1
    # ans[4] = min(cost[:5]) = min(5, 3, 4, 1, 3) = 1
    # ans[5] = min(cost[:6]) = min(5, 3, 4, 1, 3, 2) = 1
    expected = [5, 3, 3, 1, 1, 1]
    assert minCosts(cost) == expected
