from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    # dp[i] will hold the length of the longest subsequence ending at index i
    dp = [1] * n

    # Iterate through each element in nums
    for i in range(n):
        for j in range(i):
            # Calculate the absolute difference
            diff1 = abs(nums[i] - nums[j])
            # Check the condition for non-increasing differences
            if dp[j] > 1:
                # Find the maximum length of subsequence ending at i
                for k in range(j):
                    diff2 = abs(nums[j] - nums[k])
                    if diff1 <= diff2:
                        dp[i] = max(dp[i], dp[j] + 1)
    return max(dp)
```
