from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[-float('inf')] * (n + 1) for _ in range(m + 1)]
    dp[0][0] = 0  # Starting point, no coins gained or lost

    for i in range(m):
        for j in range(n):
            current_coins = coins[i][j]
            for neutralized in range(3):  # 0 to 2 neutralizations
                if current_coins >= 0:
                    dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j] + current_coins)
                else:
                    if neutralized > 0:
                        dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j])  # Neutralize this robber
                    else:
                        dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j] + current_coins)  # Take the loss

            if j + 1 < n:
                dp[i][j + 1] = max(dp[i][j + 1], dp[i][j])  # Move right
            if i + 1 < m:
                dp[i + 1][j] = max(dp[i + 1][j], dp[i][j])  # Move down

    return max(dp[m][n], 0)  # Ensure non-negative return if all paths lead to loss
```
