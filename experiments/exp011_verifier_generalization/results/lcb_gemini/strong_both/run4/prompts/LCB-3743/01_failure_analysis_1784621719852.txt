from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    # Calculate gaps before the first meeting, between meetings, and after the last meeting
    gaps = []
    gaps.append(startTime[0])  # Free time before the first meeting
    for i in range(1, len(startTime)):
        gaps.append(startTime[i] - endTime[i - 1])  # Free time between meetings
    gaps.append(eventTime - endTime[-1])  # Free time after the last meeting

    # If k >= len(gaps) - 1, we can merge all gaps
    if k >= len(gaps) - 1:
        return sum(gaps)

    # Sort gaps to find the largest possible merges
    gaps.sort()  # Sort gaps in ascending order

    # Calculate the maximum free time by merging the largest k+1 gaps
    # We create a sliding window sum of the largest k+1 gaps
    return sum(gaps[-(k + 1):])
