# test_max_sum.py
from typing import List
from solution import maxSum

# Note: The problem statement is complex: "delete any number of elements... select a
# subarray of the result with unique elements to maximize the sum". This process is
# equivalent to finding a subsequence of the original array that has unique elements
# and the maximum possible sum. This simpler model is used to derive expected results.
# The algorithm is:
# 1. Find the set of unique numbers in the input array.
# 2. Sum all the positive unique numbers.
# 3. If this sum is > 0, it is the answer.
# 4. If the sum is 0 (i.e., no positive unique numbers), check if 0 is a unique number.
#    - If yes, the answer is 0.
#    - If no (all unique numbers are negative), the answer is the largest (least negative)
#      of these unique numbers, as we must select at least one element.

def test_canonical_example_1():
    # spec: input '[1, 2, 3, 4, 5]' -> output '15'
    # The unique elements are {1, 2, 3, 4, 5}. All are positive.
    # The maximum sum is the sum of all unique positive elements: 1 + 2 + 3 + 4 + 5 = 15.
    assert maxSum([1, 2, 3, 4, 5]) == 15

def test_canonical_example_2():
    # spec: input '[1, 1, 0, 1, 1]' -> output '1'
    # The unique elements are {0, 1}.
    # The only positive unique element is 1. The maximum sum is 1.
    assert maxSum([1, 1, 0, 1, 1]) == 1

def test_canonical_example_3():
    # spec: input '[1, 2, -1, -2, 1, 0, -1]' -> output '3'
    # The unique elements are {1, 2, -1, -2, 0}.
    # The positive unique elements are {1, 2}.
    # The maximum sum is 1 + 2 = 3.
    assert maxSum([1, 2, -1, -2, 1, 0, -1]) == 3

def test_all_positives_with_duplicates():
    # spec: input with positive numbers and duplicates
    # The unique elements are {2, 3, 5}. All are positive.
    # The maximum sum is 2 + 3 + 5 = 10.

def test_all_negatives():
    # spec: input with only negative numbers
    # The unique elements are {-1, -2, -3}. There are no positive unique elements or zero.
    # To maximize the sum, we must choose at least one element. We choose the largest one.
    # The maximum sum is -1.
    assert maxSum([-1, -2, -3, -1, -2]) == -1
