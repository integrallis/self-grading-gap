from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Prepare a list to hold the number of coins at each position
    max_position = 0
    for l, r, c in coins:
        max_position = max(max_position, r)

    # Create an array to store the number of coins at each position
    coin_count = [0] * (max_position + 1)
    for l, r, c in coins:
        for i in range(l, r + 1):
            coin_count[i] += c

    # Now find the maximum coins we can collect from any k consecutive bags
    max_coins = 0
    current_coins = sum(coin_count[1:k + 1])  # Initial sum for the first k bags
    max_coins = current_coins

    for i in range(1, max_position - k + 2):  # Sliding window
        current_coins = current_coins - coin_count[i] + coin_count[i + k]
        max_coins = max(max_coins, current_coins)

    return max_coins
```
