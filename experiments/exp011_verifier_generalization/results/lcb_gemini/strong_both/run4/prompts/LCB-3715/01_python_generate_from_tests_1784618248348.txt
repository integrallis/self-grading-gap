from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Prepare a list to hold the number of coins at each position
    max_position = 0
    for l, r, c in coins:
        max_position = max(max_position, r)

    # Create a sparse dictionary to store the number of coins at each position
    coin_count = {}
    for l, r, c in coins:
        for i in range(l, r + 1):
            coin_count[i] = coin_count.get(i, 0) + c

    # Prepare for sliding window
    max_coins = 0
    current_coins = 0
    # Calculate initial sum for the first k bags
    for i in range(1, k + 1):
        current_coins += coin_count.get(i, 0)
    max_coins = current_coins

    # Sliding window over the range of bags
    for i in range(1, max_position - k + 2):
        current_coins -= coin_count.get(i, 0)
        current_coins += coin_count.get(i + k, 0)
        max_coins = max(max_coins, current_coins)

    return max_coins
```
