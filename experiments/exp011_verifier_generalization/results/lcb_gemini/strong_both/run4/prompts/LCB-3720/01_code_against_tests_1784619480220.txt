from typing import List
import heapq

def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Build the reversed adjacency list
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[v].append((u, w))  # Reverse the graph for reachability check

    # Function to check if we can achieve a max weight limit
    def canAchieve(max_weight: int) -> bool:
        out_degree = [0] * n
        reachable = [False] * n
        reachable[0] = True  # mark node 0 as reachable

        # Min-heap for exploring the nodes
        heap = [(0, 0)]  # (current node, current weight)
        while heap:
            node, _ = heapq.heappop(heap)
            # Go through neighbors
            for neighbor, weight in graph[node]:
                if weight <= max_weight:
                    if out_degree[neighbor] < threshold:
                        out_degree[neighbor] += 1
                        if not reachable[u]:
                            reachable[u] = True
                            heapq.heappush(heap, (neighbor, weight))

        return all(reachable)  # Check if all nodes can reach node 0

    # Binary search for the minimum possible max weight
    left, right = 1, 10**6
    result = -1
    while left <= right:
        mid = (left + right) // 2
        if canAchieve(mid):
            result = mid
            right = mid - 1
        else:
            left = mid + 1

    return result
