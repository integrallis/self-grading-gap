from typing import List

import pytest

from solution import minMaxWeight


def test_canonical_example_1():
    # spec: n=5, edges=[[1,0,1],[2,0,2],[3,0,1],[4,3,1],[2,1,1]], threshold=2 -> 1
    # A max weight of 1 is possible. Edges with weight > 1 are removed.
    # The remaining edges ([1,0,1], [3,0,1], [4,3,1], [2,1,1]) form a graph where:
    # 1. All nodes can reach 0 (e.g., 2->1->0, 4->3->0).
    # 2. All node out-degrees are <= 2.
    # Since 1 is a possible max weight and it's the minimum weight in the valid set, it's the answer.
    n = 5
    edges = [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]]
    threshold = 2
    assert minMaxWeight(n, edges, threshold) == 1


def test_canonical_example_2():
    # spec: n=5, edges=[[0,1,1],[0,2,2],[0,3,1],[0,4,1],[1,2,1],[1,4,1]], threshold=1 -> -1
    # In the given graph, there are no incoming edges to node 0.
    # Therefore, no other node can ever reach node 0, making it impossible to satisfy the first condition.
    n = 5
    edges = [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]]
    threshold = 1
    assert minMaxWeight(n, edges, threshold) == -1


def test_canonical_example_3():
    # spec: n=5, edges=[[1,2,1],[1,3,3],[1,4,5],[2,3,2],[3,4,2],[4,0,1]], threshold=1 -> 2
    # If max_weight=1, only edges [1,2,1] and [4,0,1] are kept. Node 3 cannot reach 0.
    # If max_weight=2, edges [1,2,1], [2,3,2], [3,4,2], [4,0,1] are kept.
    # In this subgraph, all out-degrees are <= 1, and all nodes can reach 0 (e.g., via path 1->2->3->4->0).
    # Since conditions fail for 1 but pass for 2, 2 is the minimal maximum weight.
    n = 5
    edges = [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]]
    threshold = 1
    assert minMaxWeight(n, edges, threshold) == 2


def test_canonical_example_4():
    # spec: n=5, edges=[[1,2,1],[1,3,3],[1,4,5],[2,3,2],[4,0,1]], threshold=1 -> -1
