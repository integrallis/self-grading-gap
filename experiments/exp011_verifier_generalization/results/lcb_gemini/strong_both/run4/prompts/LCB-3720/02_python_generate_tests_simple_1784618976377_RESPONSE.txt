from typing import List

def test_canonical_example_1():
    # spec: n=5, edges=[[1,0,1],[2,0,2],[3,0,1],[4,3,1],[2,1,1]], threshold=2 -> 1
    # Check if a max weight of 1 is possible.
    # Edges with weight <= 1 are: [1,0,1], [3,0,1], [4,3,1], [2,1,1].
    # In the graph with these edges:
    # - Reachability: 1->0, 3->0, 2->1->0, 4->3->0. All nodes can reach 0.
    # - Out-degree: out(1)=1, out(2)=1, out(3)=1, out(4)=1. All are <= threshold 2.
    # All conditions are met. Since 1 is the lowest possible positive weight, it is the minimal max weight.
    n = 5
    edges = [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]]
    threshold = 2
    assert minMaxWeight(n, edges, threshold) == 1

    # Therefore, no other node can ever reach node 0, making it impossible to satisfy the first condition.

    edges = [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]]

def test_canonical_example_3():

    # Check max_weight=1: Edges kept are [1,2,1] and [4,0,1]. Node 3 has no path to 0. Fails.
    # Check max_weight=2: Edges kept are [1,2,1], [2,3,2], [3,4,2], [4,0,1].
    # - Reachability: 1->2->3->4->0. All nodes can reach 0.

    # All conditions are met. Since 1 fails and 2 succeeds, 2 is the minimal maximum weight.

def test_canonical_example_4():
    # spec: n=5, edges=[[1,2,1],[1,3,3],[1,4,5],[2,3,2],[4,0,1]], threshold=1 -> -1
    # Node 1 has three outgoing edges ([1,2,1], [1,3,3], [1,4,5]), but the threshold is 1.
    # We must choose exactly one edge from node 1.

    # - Keep [1,3,3]: Node 1's path would be 1->3, but 3 has no outgoing edge. 1 cannot reach 0.
    # In all valid subgraphs respecting the threshold, some nodes cannot reach 0. It's impossible.
    n = 5
    edges = [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]]
    threshold = 1
    assert minMaxWeight(n, edges, threshold) == -1


def test_min_n_is_2():
    # spec constraint: 2 <= n <= 10^5
    # Test with n=2. The only other node is 1, which must reach 0.
    # The only available edge is 1->0 with weight 10.
    # We must keep this edge.
    # - Reachability: 1 can reach 0.
    # - Out-degree: out(1)=1, which is <= threshold 1.
    # The conditions are met. The maximum weight in the resulting graph is 10.
    n = 2
    edges = [[1, 0, 10]]
    threshold = 1
    assert minMaxWeight(n, edges, threshold) == 10


def test_threshold_is_max_n_minus_1():
    # spec constraint: 1 <= threshold <= n - 1
    # Test with n=3 and threshold=2 (which is n-1).
    # Nodes 1 and 2 must reach 0.
    # The only path for node 1 to reach 0 is 1->2->0.
    # We must keep both edges [1,2,5] and [2,0,10].
    # - Reachability: 1->2->0, 2->0. OK.

    # The conditions are met. The max weight is max(5, 10) = 10.
    n = 3
    edges = [[1, 2, 5], [2, 0, 10]]
    threshold = 2
    assert minMaxWeight(n, edges, threshold) == 10


def test_multiple_edges_between_same_nodes():
    # spec: "There may be multiple edges between a pair of nodes"
    # To satisfy reachability, node 2 must use edge [2,0,20].
    # Node 1 must use an edge to node 2. Two options exist: [1,2,5] and [1,2,10].
    # Node 1 has an out-degree threshold of 1, so we must pick only one of these.
    # Let's try to achieve the minimum possible max weight. The edge weights are 5, 10, 20.
    # - Try max_weight < 20: Impossible, as node 2 cannot reach 0.
    # - Try max_weight = 20:
    #   - Edges with weight <= 20: [1,2,5], [1,2,10], [2,0,20].
    #   - Select a subgraph:
    #     - Must take [2,0,20]. out(2)=1.
    #     - Must take one edge from 1 to 2. Let's pick [1,2,5]. out(1)=1.
    #   - Resulting subgraph has edges {[1,2,5], [2,0,20]}.
    #   - Reachability and out-degree are satisfied. The max weight is 20.
    # Since any max weight < 20 is impossible, 20 is the minimal max weight.
    n = 3
    edges = [[1, 2, 5], [1, 2, 10], [2, 0, 20]]
    threshold = 1
    assert minMaxWeight(n, edges, threshold) == 20
