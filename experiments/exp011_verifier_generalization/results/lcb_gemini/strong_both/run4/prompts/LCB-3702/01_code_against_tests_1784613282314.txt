```python
from typing import List
import math


def maxLength(nums: List[int]) -> int:
    n = len(nums)
    max_length = 2  # Minimum length of product equivalent subarray
    for start in range(n):
        for end in range(start + 2, n + 1):  # Start from length 2
            subarray = nums[start:end]
            prod = math.prod(subarray)
            gcd_value = subarray[0]
            lcm_value = subarray[0]
            for num in subarray[1:]:
                gcd_value = math.gcd(gcd_value, num)
                lcm_value = (lcm_value * num) // math.gcd(lcm_value, num)
            if prod == lcm_value * gcd_value:
                max_length = max(max_length, end - start)
    return max_length

```
