# Note: The problem defines a "product equivalent" array `arr` as one where
# prod(arr) == lcm(arr) * gcd(arr).
# For any two numbers `a, b`, it is a mathematical identity that
# a * b == lcm(a, b) * gcd(a, b).
# Therefore, any subarray of length 2 is always product equivalent.
# Since the input array `nums` has a minimum length of 2, the result of
# maxLength will always be at least 2.

def test_canonical_example_1():
    """
    Tests the first example from the specification.
    spec: maxLength([1, 2, 1, 2, 1, 1, 1]) -> 5
    """
    # The longest product equivalent subarray is [1, 2, 1, 1, 1].
    # prod([1, 2, 1, 1, 1]) = 2
    # gcd([1, 2, 1, 1, 1]) = 1
    # lcm([1, 2, 1, 1, 1]) = 2
    # prod == gcd * lcm (2 == 1 * 2). Length is 5.
    # A longer subarray like [1, 2, 1, 2, 1] is not product equivalent
    # (prod=4, gcd=1, lcm=2; 4 != 1 * 2).
    nums = [1, 2, 1, 2, 1, 1, 1]
    assert maxLength(nums) == 5
