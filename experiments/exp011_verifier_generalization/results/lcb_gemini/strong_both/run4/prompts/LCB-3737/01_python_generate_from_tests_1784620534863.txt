from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    dp = [[0] * 3 for _ in range(n)]

    # Initialize the first house costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    for i in range(1, n):
        for j in range(3):
            # For the current house, calculate the minimum cost for each color
            dp[i][j] = cost[i][j] + min(
                dp[i-1][(j+1) % 3],  # Not the same as previous house
                dp[i-1][(j+2) % 3]   # Not the same as previous house
            )

    # Ensure to adhere to the equidistant condition
    for i in range(n // 2, n):
        for j in range(3):
            opposite_color = (j + 1) % 3
            dp[i][j] = min(dp[i][j], dp[i - (n // 2)][opposite_color])

    # The answer is the minimum cost to paint the last house with any color
    return min(dp[n-1])
```
