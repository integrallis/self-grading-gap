from typing import List
from itertools import permutations

def totalNumbers(digits: List[int]) -> int:
    even_digits = [d for d in digits if d % 2 == 0]
    count = 0
    for last in set(even_digits):  # last digit must be even
        remaining = digits.copy()
        remaining.remove(last)  # remove one occurrence of the last digit
        for first in set(remaining):  # first digit must not be zero
            if first == 0:
                continue
            remaining.remove(first)  # remove the first digit
            count += len(remaining)  # the rest can be any of the remaining digits
            remaining.append(first)  # add it back for the next iteration
        remaining.append(last)  # add the last digit back for next iteration
    return count
```
