from typing import List


def maxFrequency(nums: List[int], k: int) -> int:
    from collections import Counter
    count = Counter(nums)
    initial_count = count[k]
    max_gain = 0

    # Iterate through all possible values in nums to consider as v
    for v in count:
        if v == k:
            continue
        gain = 0
        current_gain = []

        # Calculate the gain from converting v to k
        for num in nums:
            if num == v:
                current_gain.append(1)
            elif num == k:
                current_gain.append(-1)
            else:
                current_gain.append(0)

        # Find the maximum subarray sum using Kadane's algorithm
        current_sum = 0
        for value in current_gain:
            current_sum += value
            if current_sum < 0:
                current_sum = 0
            max_gain = max(max_gain, current_sum)

    return initial_count + max_gain
