import pytest
from typing import List
from solution import minMaxSums

# The problem statement specifies the modulo for the answer.
MOD = 10**9 + 7


def test_canonical_example_1():
    # spec: nums = [1, 2, 3], k = 2 -> 24
    # Explanation from spec:
    # Subsequence | Minimum | Maximum | Sum (Min+Max)
    # ---------------------------------------------
    # [1]         | 1       | 1       | 2
    # [2]         | 2       | 2       | 4
    # [3]         | 3       | 3       | 6
    # [1, 2]      | 1       | 2       | 3
    # [1, 3]      | 1       | 3       | 4
    # [2, 3]      | 2       | 3       | 5
    # ---------------------------------------------
    # Total Sum: 2 + 4 + 6 + 3 + 4 + 5 = 24
    assert minMaxSums(nums=[1, 2, 3], k=2) == 24


def test_canonical_example_2():
    # spec: nums = [5, 0, 6], k = 1 -> 22
    # Explanation: k=1 means only subsequences of length 1 are considered.
    # For a subsequence of one element, min and max are the element itself.
    # [5]: min=5, max=5, sum=10
    # [0]: min=0, max=0, sum=0
    # [6]: min=6, max=6, sum=12
    # Total: 10 + 0 + 12 = 22
    assert minMaxSums(nums=[5, 0, 6], k=1) == 22


def test_canonical_example_3():
    # spec: nums = [1, 1, 1], k = 2 -> 12
    # Explanation: Subsequences are distinct based on the indices of their elements.
    # Subsequences of length 1 (3 of them, from indices 0, 1, and 2):
    #   [1]: min=1, max=1 -> sum=2. Total for length 1: 3 * 2 = 6.
    # Subsequences of length 2 (3 of them, from indices (0,1), (0,2), and (1,2)):
    #   [1, 1]: min=1, max=1 -> sum=2. Total for length 2: 3 * 2 = 6.
    # Grand total = 6 + 6 = 12.
    assert minMaxSums(nums=[1, 1, 1], k=2) == 12


def test_large_numbers_trigger_modulo():
    # spec: Test modulo behavior with large numbers
    # nums = [10**9, 10**9 - 1], k = 2
    # Let n1 = 10**9 and n2 = 10**9 - 1. MOD = 10**9 + 7.
    # Subsequences of length at most 2:
    # [n1]: min=n1, max=n1. Sum = 2 * n1
    # [n2]: min=n2, max=n2. Sum = 2 * n2
    # [n1, n2]: min=n2, max=n1. Sum = n1 + n2
    # Total sum = (2 * n1) + (2 * n2) + (n1 + n2)
    #           = 3 * n1 + 3 * n2 = 3 * (2 * 10**9 - 1)
    #           = 6 * 10**9 - 3
    # Calculation modulo 10**9 + 7:
    # 6 * 1_000_000_000 - 3 = 5_999_999_997
    # 5_999_999_997 % 1_000_000_007 = 999_999_962
    # Alternatively: 10^9 = MOD - 7.
    # Total = 6 * (MOD - 7) - 3 = 6*MOD - 42 - 3 = 6*MOD - 45
    # (-45) % MOD = MOD - 45 = 999999962.
    assert minMaxSums(nums=[10**9, 10**9 - 1], k=2) == 999999962


def test_single_element_list():
    # spec: Test minimum list length constraint, nums.length = 1.
    # nums = [100], k = 1
    # The only subsequence with at most 1 element is [100].
    # For [100], min=100, max=100. Sum = 100 + 100 = 200.
    # Total sum is 200.
    assert minMaxSums(nums=[100], k=1) == 200


def test_k_equals_list_length():
    # spec: Test boundary where k is equal to nums.length.
    # nums = [1, 10], k = 2
    # Since k=2=nums.length, all non-empty subsequences are considered.
    # Subsequence | Minimum | Maximum | Sum (Min+Max)
    # ---------------------------------------------
    # [1]         | 1       | 1       | 2
    # [10]        | 10      | 10      | 20
    # [1, 10]     | 1       | 10      | 11
    # ---------------------------------------------
    # Total Sum: 2 + 20 + 11 = 33.
    assert minMaxSums(nums=[1, 10], k=2) == 33
