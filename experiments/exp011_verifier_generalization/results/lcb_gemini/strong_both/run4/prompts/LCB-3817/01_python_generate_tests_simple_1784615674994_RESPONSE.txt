#
# Canonical examples from the specification
#

def test_canonical_example_1():
    """
    Tests the first example from the specification.
    The deck capacity is the limiting factor.
    """
    n, w, maxWeight = 2, 3, 15
    # Deck capacity: n*n = 2*2 = 4
    # Weight capacity: maxWeight // w = 15 // 3 = 5
    # Expected: min(4, 5) = 4
    assert maxContainers(n, w, maxWeight) == 4
