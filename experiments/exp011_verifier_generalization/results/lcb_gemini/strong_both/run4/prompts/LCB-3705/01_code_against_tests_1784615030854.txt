```python
from typing import List

def largestInteger(nums: List[int], k: int) -> int:
    from collections import defaultdict

    count_map = defaultdict(int)
    n = len(nums)
    
    # Iterate through all possible subarrays of size k
    for i in range(n - k + 1):
        subarray = nums[i:i+k]
        unique_elements = set(subarray)
        
        # Count each unique element in the current subarray
        for num in unique_elements:
            count_map[num] += 1

    # Filter numbers that appear exactly once
    almost_missing = [num for num, count in count_map.items() if count == 1]
    
    # Return the largest almost missing integer or -1 if none exists
    return max(almost_missing) if almost_missing else -1

```
