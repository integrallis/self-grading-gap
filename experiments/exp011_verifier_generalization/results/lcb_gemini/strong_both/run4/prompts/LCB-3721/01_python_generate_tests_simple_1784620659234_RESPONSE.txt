def create_expected_for_large_test(size, modifications):

        lst[index] = value
    return lst


def test_canonical_example_1():
    """
    Tests the first example from the specification.
    - t=10: MESSAGE "id1 id0". All online. mentions: [1, 1].

    - t=71: User 0 comes back online. MESSAGE "HERE". Both online. mentions: [1+1, 1+1] = [2, 2].

    # t=10: User 50 goes offline until t=70.
    # t=20: MESSAGE "ALL". All 100 users get 1 mention.
    # t=30: MESSAGE "HERE". User 50 is offline. The other 99 online users get 1 mention.
