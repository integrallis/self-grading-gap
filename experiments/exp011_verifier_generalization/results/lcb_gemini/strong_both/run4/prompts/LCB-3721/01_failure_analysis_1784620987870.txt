from typing import List

def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers
    online_status = [True] * numberOfUsers
    offline_till = [0] * numberOfUsers

    # First, sort events by timestamp.
    events.sort(key=lambda x: int(x[1]))

    for event in events:
        event_type, timestamp, *details = event
        timestamp = int(timestamp)

        # Update online status before processing MESSAGE events
        for user_id in range(numberOfUsers):
            if not online_status[user_id] and timestamp >= offline_till[user_id]:
                online_status[user_id] = True
        
        if event_type == "OFFLINE":
            user_id = int(details[0])
            online_status[user_id] = False
            offline_till[user_id] = timestamp + 60
        elif event_type == "MESSAGE":
            mentions_string = details[0]
            if mentions_string == "ALL":
                for i in range(numberOfUsers):
                    mentions[i] += 1
            elif mentions_string == "HERE":
                for i in range(numberOfUsers):
                    if online_status[i]:
                        mentions[i] += 1
            else:
                user_ids = mentions_string.split()
                for user_id_str in user_ids:
                    if user_id_str.startswith("id"):
                        user_id = int(user_id_str[2:])
                        mentions[user_id] += 1  # Count mentions regardless of online status

    return mentions
