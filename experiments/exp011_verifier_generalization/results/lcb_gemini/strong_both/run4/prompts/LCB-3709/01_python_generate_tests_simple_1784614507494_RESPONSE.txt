import pytest
from solution import hasSpecialSubstring

# Note: The problem statement uses 1-based inclusive indexing (e.g., s[4..6]) for examples,
# while Python uses 0-based exclusive slicing (e.g., s[4:7]). The tests below use
# Python conventions but map to the logic of the examples.

def test_canonical_example_1():
    # spec: s = "aaabaaa", k = 3 -> true
    # The substring s[4:7] == "aaa" satisfies the conditions:
    # - Length is 3 (k).
    # - All characters are the same ('a').
    # - The character before is s[3] ('b'), which is different.
    # - There is no character after.
    assert hasSpecialSubstring("aaabaaa", 3) is True

def test_canonical_example_2():
    # spec: s = "abc", k = 2 -> false
    # Substrings of length 2 are "ab" and "bc".
    # Neither consists of one distinct character.
    assert hasSpecialSubstring("abc", 2) is False

def test_special_substring_in_middle():
    # The substring "ccc" at s[2:5] is special.

def test_special_substring_at_start():
    # The substring "aaa" at s[0:3] is special.
    # - It has no preceding character.
    # - It is followed by 'b'.
    assert hasSpecialSubstring("aaab", 3) is True

def test_entire_string_is_special():
    # The substring "aaaa" (the whole string) is special.
    # - It has no preceding or following characters to check against.
    assert hasSpecialSubstring("aaaa", 4) is True
