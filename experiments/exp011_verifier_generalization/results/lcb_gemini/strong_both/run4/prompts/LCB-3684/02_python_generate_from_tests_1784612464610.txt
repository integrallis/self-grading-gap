def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]
    prefix_length = len(prefix)
    suffix_length = len(suffix)
    # Check if there is enough length in s for prefix and suffix
    if prefix_length + suffix_length > len(s):
        return False
    # Check for all possible starting points for the prefix in s
    for i in range(len(s) - prefix_length + 1):
        if s[i:i + prefix_length] == prefix:
            # Now check if there's a valid suffix after the prefix
            if s[i + prefix_length:i + prefix_length + suffix_length] == suffix:
                return True
            # We can also allow for the '*' to represent zero characters
            if i + prefix_length + suffix_length <= len(s):
                if s[i + prefix_length:i + prefix_length + suffix_length] == '':
                    return True
    return False

```
