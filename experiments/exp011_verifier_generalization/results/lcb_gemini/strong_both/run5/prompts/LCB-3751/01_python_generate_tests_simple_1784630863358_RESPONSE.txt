import pytest
from typing import List
from solution import maxFrequency

def test_canonical_example_1():
    """
    Test with the first canonical example from the problem description.
    nums = [1, 2, 3, 4, 5, 6], k = 1
    """
    # initial count of k=1 is 1.
    # To maximize, we need to find max(c(v,i,j) - c(k,i,j)) over all subarrays and values v.
    # Let v=6. Let subarray be nums[5..5] = [6].
    # Gain = c(6,5,5) - c(1,5,5) = 1 - 0 = 1.
    # Another way to get gain 1 is the example's subarray nums[2..5] = [3,4,5,6].
    # The mode is any element (freq 1), and k does not appear. Gain = 1 - 0 = 1.
    # The maximum possible gain is 1.
    # Final frequency = initial_count + max_gain = 1 + 1 = 2.
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2

def test_canonical_example_2():
    """
    Test with the second canonical example from the problem description.
    nums = [10, 2, 3, 4, 5, 5, 4, 3, 2, 2], k = 10
    """
    # initial count of k=10 is 1 (at index 0).
    # Consider v=2. To find max gain for v=2, we use max subarray sum on an array A
    # where A[i]=1 if nums[i]==2, A[i]=-1 if nums[i]==10, else 0.
    # A = [-1, 1, 0, 0, 0, 0, 0, 0, 1, 1]
    # Max subarray sum of A is for [1, 0, ..., 1, 1], which is 3. This corresponds to nums[1..9].
    # Gain for v=2 is c(2,1,9) - c(10,1,9) = 3 - 0 = 3.
    # Gains for other values v are smaller (e.g., for v=3, gain is 2).
    # Max gain is 3.
    # Final frequency = initial_count + max_gain = 1 + 3 = 4.
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4
