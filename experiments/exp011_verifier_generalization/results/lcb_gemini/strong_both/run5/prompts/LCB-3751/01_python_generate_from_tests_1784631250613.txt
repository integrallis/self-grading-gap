from typing import List


def maxFrequency(nums: List[int], k: int) -> int:
    from collections import Counter
    from itertools import accumulate

    count = Counter(nums)
    initial_count = count[k]
    max_gain = 0

    # Create a transformation array based on the difference from k
    A = [0] * len(nums)
    for i in range(len(nums)):
        if nums[i] < k:
            A[i] = 1  # we can increase this to k
        elif nums[i] > k:
            A[i] = -1  # we need to decrease this to k

    # We need to find the maximum subarray sum of A
    current_sum = 0
    for a in A:
        current_sum += a
        if current_sum < 0:
            current_sum = 0
        max_gain = max(max_gain, current_sum)

    return initial_count + max_gain
```
