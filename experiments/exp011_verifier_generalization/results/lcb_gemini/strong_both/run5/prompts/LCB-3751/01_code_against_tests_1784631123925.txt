## Implementation

```python
from typing import List


def maxFrequency(nums: List[int], k: int) -> int:
    from collections import Counter

    count = Counter(nums)
    initial_count = count[k]
    max_gain = 0

    for v in count.keys():
        if v == k:
            continue
        gain = count[v] - count[k]
        max_gain = max(max_gain, gain)

    return initial_count + max_gain

```
