def test_canonical_example_1():
    """
    Test with the first canonical example from the problem description.
    nums = [1, 2, 3, 4, 5, 6], k = 1

    # initial count of k=1 is 1.
    # To maximize, we need to find max(c(v,i,j) - c(k,i,j)) over all subarrays and values v.
    # Let v=6. Let subarray be nums[5..5] = [6].
    # Gain = c(6,5,5) - c(1,5,5) = 1 - 0 = 1.
    # Another way to get gain 1 is the example's subarray nums[2..5] = [3,4,5,6].
    # The mode is any element (freq 1), and k does not appear. Gain = 1 - 0 = 1.
