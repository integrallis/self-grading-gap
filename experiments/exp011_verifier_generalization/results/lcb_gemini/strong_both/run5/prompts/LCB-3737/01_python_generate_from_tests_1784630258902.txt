from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize a DP array with size 2 (for current and previous)
    dp = [[0] * 3 for _ in range(2)]

    # Fill the first house costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Iterate through houses 1 to n-1
    for i in range(1, n):
        current = i % 2
        previous = (i - 1) % 2
        for j in range(3):
            # Calculate the minimum cost for painting house i with color j
            dp[current][j] = float('inf')
            for k in range(3):
                if j != k:  # Adjacent houses cannot have the same color
                    dp[current][j] = min(dp[current][j], dp[previous][k] + cost[i][j])

        # Handle the equidistant condition by checking the previous set of houses
        if i >= 2:
            for j in range(3):
                for k in range(3):
                    if j != k:
                        dp[current][j] = min(dp[current][j], dp[previous][k] + cost[i][j])

        # For the last two houses, ensure equidistant constraints are respected
        if i == n - 1:
            for j in range(3):
                for k in range(3):
                    if j != k:
                        dp[current][j] = min(dp[current][j], dp[previous][k] + cost[i][j])

    # Calculate the minimum cost from the last house
    final_cost = float('inf')
    for j in range(3):
        final_cost = min(final_cost, dp[(n - 1) % 2][j])

    return final_cost
