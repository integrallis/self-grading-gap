from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[[float('-inf')] * 3 for _ in range(n)] for _ in range(m)]  # dp[i][j][k]: max coins using k neutralizations

    # Initialize the starting point
    if coins[0][0] >= 0:
        dp[0][0][0] = coins[0][0]
    else:
        dp[0][0][0] = coins[0][0]  # Take the hit
        dp[0][0][1] = 0  # Neutralize it

    for i in range(m):
        for j in range(n):
            if i == 0 and j == 0:
                continue  # Skip the starting point
            for k in range(3):  # For each neutralization count
                val = coins[i][j]  # Value at current cell
                prev_profit = float('-inf')
                if i > 0:
                    prev_profit = max(prev_profit, dp[i-1][j][k])
                if j > 0:
                    prev_profit = max(prev_profit, dp[i][j-1][k])

                if prev_profit == float('-inf'):
                    continue

                if val >= 0:
                    dp[i][j][k] = max(dp[i][j][k], prev_profit + val)
                else:  # It's a robber
                    # Take the hit
                    dp[i][j][k] = max(dp[i][j][k], prev_profit + val)
                    # Neutralize if we can
                    if k > 0:
                        prev_profit_k_minus_1 = float('-inf')
                        if i > 0:
                            prev_profit_k_minus_1 = max(prev_profit_k_minus_1, dp[i-1][j][k-1])
                        if j > 0:
                            prev_profit_k_minus_1 = max(prev_profit_k_minus_1, dp[i][j-1][k-1])
                        if prev_profit_k_minus_1 != float('-inf'):
                            dp[i][j][k] = max(dp[i][j][k], prev_profit_k_minus_1)  # Neutralizing means not losing any coins

    # Return the maximum profit possible at the bottom-right corner
    return max(dp[m-1][n-1])
