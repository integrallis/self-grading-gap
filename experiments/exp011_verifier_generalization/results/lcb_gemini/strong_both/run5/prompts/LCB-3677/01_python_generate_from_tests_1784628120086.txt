from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[[float('-inf'), float('-inf')] for _ in range(n)] for _ in range(m)]  # dp[i][j] = [max coins, neutralizations used]
    dp[0][0] = [coins[0][0], 0] if coins[0][0] >= 0 else [0, 1]  # Use one neutralization if needed

    for i in range(m):
        for j in range(n):
            current_coins, neutralizations = dp[i][j]  # Current state
            if current_coins == float('-inf'):
                continue  # Skip unreachable states

            # Move right
            if j + 1 < n:
                next_coins = coins[i][j + 1]
                if next_coins >= 0:
                    dp[i][j + 1] = max(dp[i][j + 1], [current_coins + next_coins, neutralizations])
                else:
                    # If it's a robber
                    if neutralizations < 2:
                        dp[i][j + 1] = max(dp[i][j + 1], [current_coins, neutralizations + 1])  # Neutralize this robber

            # Move down
            if i + 1 < m:
                next_coins = coins[i + 1][j]
                if next_coins >= 0:
                    dp[i + 1][j] = max(dp[i + 1][j], [current_coins + next_coins, neutralizations])
                else:
                    # If it's a robber
                    if neutralizations < 2:
                        dp[i + 1][j] = max(dp[i + 1][j], [current_coins, neutralizations + 1])  # Neutralize this robber

    # Maximum profit at the bottom-right corner considering neutralizations
    max_coins = dp[m - 1][n - 1][0]
    return max_coins if max_coins != float('-inf') else 0
