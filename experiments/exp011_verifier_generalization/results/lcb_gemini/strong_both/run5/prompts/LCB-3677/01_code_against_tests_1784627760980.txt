from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[float('-inf')] * (n + 1) for _ in range(m + 1)]
    dp[0][0] = coins[0][0]  # Starting point (0,0) with its coins

    for i in range(m):
        for j in range(n):
            if dp[i][j] == float('-inf'):
                continue  # Skip unreachable states

            # Current coins before considering robbers
            current_coins = dp[i][j]

            # Move right
            if j + 1 < n:
                dp[i][j + 1] = max(dp[i][j + 1], current_coins + coins[i][j + 1])

            # Move down
            if i + 1 < m:
                dp[i + 1][j] = max(dp[i + 1][j], current_coins + coins[i + 1][j])

            # Handle robbers in the next steps
            if j + 1 < n and coins[i][j + 1] < 0:
                # Neutralize if possible
dp[i][j + 1] = max(dp[i][j + 1], current_coins)
            if i + 1 < m and coins[i + 1][j] < 0:
                # Neutralize if possible
                dp[i + 1][j] = max(dp[i + 1][j], current_coins)

    # Maximum profit at the bottom-right corner considering neutralizations
    return dp[m - 1][n - 1]
