from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a dictionary to hold the number of coins in each bag position
    coin_map = {}
    for l, r, c in coins:
        for i in range(l, r + 1):
            coin_map[i] = coin_map.get(i, 0) + c

    # Create a list of bag positions sorted in ascending order
    positions = sorted(coin_map.keys())
    n = len(positions)
    max_coins = 0

    # Calculate the maximum coins in any window of size k
    left = 0
    current_coins = 0
    for right in range(n):
        current_coins += coin_map[positions[right]]

        # Ensure the window size does not exceed k
        while positions[right] - positions[left] + 1 > k:
            current_coins -= coin_map[positions[left]]
            left += 1

        # Update the max_coins if the current window is valid
        max_coins = max(max_coins, current_coins)

    return max_coins
