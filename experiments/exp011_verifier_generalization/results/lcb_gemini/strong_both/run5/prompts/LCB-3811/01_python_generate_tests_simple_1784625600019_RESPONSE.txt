# MANDATORY: Canonical Examples from Specification
def test_canonical_example_1():
    """
    Tests the first example from the specification.
    Input: s = "abc"

    'b' (value=25) at pos 2: 25 * 2 = 50
    'c' (value=24) at pos 3: 24 * 3 = 72
    Total = 26 + 50 + 72 = 148
    """
    assert reverseDegree("abc") == 148
