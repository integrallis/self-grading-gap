from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    
    # Calculate the total number of subsequences of size 1 to k
    for size in range(1, k + 1):
        # The number of ways to choose 'size' elements from 'n'
        num_subsequences = combination(n, size)
        
        # Each subsequence will contribute its min + max
        for num in nums:
            # Each number will be the min or max in num_subsequences//2
            contribution = (2 * num) % MOD * num_subsequences % MOD
            total_sum = (total_sum + contribution) % MOD
    
    return total_sum


def combination(n: int, k: int) -> int:
    if k > n:
        return 0
    if k == 0 or k == n:
        return 1
    k = min(k, n - k)  # Take advantage of symmetry
    c = 1
    for i in range(k):
        c = c * (n - i) // (i + 1)
    return c
```
