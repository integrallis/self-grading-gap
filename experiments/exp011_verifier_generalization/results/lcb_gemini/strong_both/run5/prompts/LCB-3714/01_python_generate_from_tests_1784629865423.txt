from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Precompute powers of 2 up to k
    power_of_2 = [1] * (k + 1)
    for i in range(1, k + 1):
        power_of_2[i] = (power_of_2[i - 1] * 2) % MOD

    # Calculate contribution of each element as min and max
    for i in range(n):
        # Each element nums[i] can be the max in (2^(count of elements chosen from nums))
        max_contrib = (power_of_2[min(k, n - 1)]) % MOD
        # Each element nums[i] can be the min in (2^(count of elements chosen from nums))
        min_contrib = (power_of_2[min(k, n - 1)]) % MOD
        # Total contribution for this element
        total_sum = (total_sum + (max_contrib + min_contrib) * nums[i]) % MOD

    return total_sum
