```python
def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    # We will track the number of moves in each direction
    move_count = {'N': 0, 'S': 0, 'E': 0, 'W': 0}

    for char in s:
        if char == 'N':
            y += 1
            move_count['N'] += 1
        elif char == 'S':
            y -= 1
            move_count['S'] += 1
        elif char == 'E':
            x += 1
            move_count['E'] += 1
        elif char == 'W':
            x -= 1
            move_count['W'] += 1
        # Calculate current distance
        current_distance = abs(x) + abs(y)
        max_distance = max(max_distance, current_distance)

    # Calculate the maximum possible distance after making k changes
    # We can change moves to increase the distance by 2 units for each change
    total_moves = move_count['N'] + move_count['S'] + move_count['E'] + move_count['W']
    # The maximum distance we can achieve is the initial distance + k (since we can change k moves)
    max_possible_distance = current_distance + min(k, total_moves) * 2
    return max(max_distance, max_possible_distance)
```
