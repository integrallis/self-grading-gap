from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n == 0:
        return 0
    # 2D dp array where dp[i][j] is length of longest valid subsequence ending at nums[j] and nums[i]
    dp = [[1] * n for _ in range(n)]
    max_length = 1
    for i in range(1, n):  # Start from the second element
        for j in range(i):  # Check all previous elements
            dp[i][j] = 2  # Base case: subsequence with nums[j] and nums[i]
            for k in range(j):  # Finding a valid predecessor
                # Check if we can extend the sequence
                if abs(nums[i] - nums[j]) <= abs(nums[j] - nums[k]):
                    dp[i][j] = max(dp[i][j], 1 + dp[j][k])
                    max_length = max(max_length, dp[i][j])
    return max_length
