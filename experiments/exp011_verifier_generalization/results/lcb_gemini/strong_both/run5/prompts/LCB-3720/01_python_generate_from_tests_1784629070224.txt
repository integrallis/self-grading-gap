from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Binary search for the minimum possible maximum weight
    left, right = 1, 10**6
    answer = -1

    while left <= right:
        mid = (left + right) // 2
        if canReachZero(n, edges, threshold, mid):
            answer = mid
            right = mid - 1
        else:
            left = mid + 1

    return answer


def canReachZero(n: int, edges: List[List[int]], threshold: int, max_weight: int) -> bool:
    from collections import defaultdict, deque

    graph = defaultdict(list)
    out_degree = [0] * n

    # Build the graph with edges <= max_weight
    for u, v, w in edges:
        if w <= max_weight:
            graph[u].append(v)
            out_degree[u] += 1

    # Check threshold on the outgoing edges
    for node in range(n):
        if out_degree[node] > threshold:
            return False

    # Use BFS or DFS to check reachability to node 0 from all nodes
    reachable = [False] * n
    queue = deque([0])
    reachable[0] = True

    while queue:
        curr = queue.popleft()
        for neighbor in graph[curr]:
            if not reachable[neighbor]:
                reachable[neighbor] = True
                queue.append(neighbor)

    # Check if all nodes can reach node 0
    for i in range(n):
        if not reachable[i]:
            return False

    return True

```
