from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Binary search for the minimum possible maximum weight
    left, right = 1, 10**6
    answer = -1

    while left <= right:
        mid = (left + right) // 2
        if canReachZero(n, edges, threshold, mid):
            answer = mid
            right = mid - 1
        else:
            left = mid + 1

    return answer


def canReachZero(n: int, edges: List[List[int]], threshold: int, max_weight: int) -> bool:
    from collections import defaultdict, deque

    graph = defaultdict(list)
    in_degree = [0] * n

    # Build the graph with edges <= max_weight and calculate in-degrees
    for u, v, w in edges:
        if w <= max_weight:
            graph[v].append(u)  # Reverse the graph for reachability to 0
            in_degree[u] += 1

    # Check that no node exceeds the threshold on the outgoing edges
    out_degree = [0] * n
    for u, v, w in edges:
        if w <= max_weight:
            out_degree[u] += 1
            if out_degree[u] > threshold:
                return False

    # Use BFS or DFS to check reachability to node 0 from all nodes
    reachable = [False] * n
    queue = deque([0])
    reachable[0] = True

    while queue:
        curr = queue.popleft()
        for neighbor in graph[curr]:
            if not reachable[neighbor]:
                reachable[neighbor] = True
                queue.append(neighbor)
    
    # Check if all nodes can reach node 0
    for i in range(n):
        if not reachable[i]:
            return False

    return True

    def canReachZero(n: int, edges: List[List[int]], threshold: int, max_weight: int) -> bool:
        from collections import defaultdict, deque

        # 1. Build a reverse graph of edges with weight <= max_weight.
        rev_adj = defaultdict(list)
        for u, v, w in edges:
            if w <= max_weight:
                rev_adj[v].append(u)

        # 2. Traverse backwards from node 0 to find all nodes that can form a valid path.
        q = deque([0])
        reachable_nodes = {0}
        out_degree_used = [0] * n

        while q:
            v = q.popleft()
            for u in rev_adj[v]:
                # If node u has a free out-degree slot, it can use the edge (u,v)
                # to connect to the already-reachable component containing v.
                if out_degree_used[u] < threshold:
                    # Use one out-degree slot for u.
                    out_degree_used[u] += 1
                    
                    # If this is the first valid path found for u, mark it as reachable
                    # and add it to the queue to find its predecessors.
                    if u not in reachable_nodes:
                        reachable_nodes.add(u)
                        q.append(u)
        
        # 3. If all n nodes are in the set of reachable nodes, a valid subgraph exists.
        return len(reachable_nodes) == n
