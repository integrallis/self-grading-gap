def test_threshold_forces_higher_weight():
    # A case where a lower max_weight is possible but violates the threshold.
    # max_weight=10: Edges are [1,0,10], [2,0,5]. Node 3 has path 3->1->0 and 3->2->0.
    # But node 3 has out-degree 2, which violates threshold=1.
    # To satisfy the threshold, we must choose one, e.g., 3->1->0. This works. max_weight is 10.
    # Let's check a lower weight. max_weight=5: only edge [2,0,5] is available. Node 1 and 3 cannot reach 0.
    # The smallest max_weight is 10.
    n = 4
    edges = [[1, 0, 10], [2, 0, 5], [3, 1, 8], [3, 2, 9]]
    threshold = 1
    assert minMaxWeight(n, edges, threshold) == 10
