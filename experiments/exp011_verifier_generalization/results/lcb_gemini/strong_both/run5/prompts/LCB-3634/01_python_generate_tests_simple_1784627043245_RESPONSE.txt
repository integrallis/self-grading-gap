import pytest
from solution import calculateScore

# The problem statement defines the function signature as `calculateScore(s: str) -> int`.
# This signature does not require any imports from the `typing` module.


def test_canonical_example_1():
    # spec: input 'aczzx' -> output 5
    # i=2, s[2]='z'. Mirror is 'a'. Closest unmarked j<2 is j=0 (s[0]='a'). Score += 2-0=2. Mark 0,2.
    # i=4, s[4]='x'. Mirror is 'c'. Closest unmarked j<4 is j=1 (s[1]='c'). Score += 4-1=3. Mark 1,4.
    # Total score = 2 + 3 = 5.
    assert calculateScore("aczzx") == 5


def test_canonical_example_2():
    # spec: input 'abcdef' -> output 0
    # No character has a preceding mirror in the string.
    # For example, at i=5, s[5]='f', mirror is 'u'. No 'u' appears at j<5.
    # Total score = 0.
    assert calculateScore("abcdef") == 0


def test_single_character_string():
    # A single character string has no preceding indices (j < i).
    # Therefore, no matches can be found, and the score is 0.
    assert calculateScore("m") == 0


def test_no_mirror_pairs_exist():
    # The string "abcabc" contains no mirror characters (e.g., 'z' for 'a').

def test_mirrors_exist_but_not_preceding():
    # The string "zyxw" contains characters that are mirrors of others,
    # but for any index i, its mirror is never at a preceding index j < i.
    assert calculateScore("zyxw") == 0


def test_simple_adjacent_pairs():
    # s = "azbycx"
    # i=1, 'z' finds mirror 'a' at j=0. score += 1-0=1.
    # i=3, 'y' finds mirror 'b' at j=2. score += 3-2=1.

    # Total score = 1 + 1 + 1 = 3.

    # i=4, 'y' finds mirror 'b' at j=1. score += 4-1=3. Mark 1,4.
    # i=5, 'x' finds mirror 'c' at j=2. score += 5-2=3. Mark 2,5.
    # Total score = 3 + 3 + 3 = 9.
    assert calculateScore("abczyx") == 9


def test_unmarked_rule_prevents_rematching():
    # s = "azaz"
    # i=1, 'z' finds mirror 'a' at j=0. score += 1-0=1. Mark 0,1.

    # i=3, 'z' finds mirror 'a' at j=2 (unmarked). score += 3-2=1. Mark 2,3.
    # Total score = 1 + 1 = 2.
    assert calculateScore("azaz") == 2


def test_closest_unmarked_rule_is_applied():
    # s = "abazy"
    # i=3, 'z'. Mirror is 'a'. Candidates: j=0, j=2. Both are unmarked.
    # Closest is j=2. score += 3-2=1. Mark 2,3.
    # i=4, 'y'. Mirror is 'b'. Only candidate is j=1 (unmarked).
    # score += 4-1=3. Mark 1,4.
    # Total score = 1 + 3 = 4.
    assert calculateScore("abazy") == 4


def test_self_mirroring_characters():
    # 'm' and 'n' are mirrors of each other.
    # s = "mnnm"
    # i=1, 'n' finds mirror 'm' at j=0. score += 1-0=1. Mark 0,1.
    # i=2, 'n'. Mirror is 'm'. s[0] is 'm' but index 0 is marked. No match.
    # i=3, 'm' finds mirror 'n' at j=2 (unmarked). score += 3-2=1. Mark 2,3.
    # Total score = 1 + 1 = 2.
    assert calculateScore("mnnm") == 2
