```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # Create a mirror mapping a-z to z-a

    for i in range(len(s)):
        if marked[i]:  # Skip marked indices
            continue
        mirror_char = mirror[s[i]]
        # Find the closest unmarked index j
        for j in range(i - 1, -1, -1):
            if marked[j]:  # Skip marked indices
                continue
            if s[j] == mirror_char:
                score += i - j
                marked[i] = marked[j] = True  # Mark both indices
                break
    return score

```
