## Implementation

```python
from collections import Counter

def maxDifference(s: str) -> int:
    frequencies = Counter(s)
    odd_frequencies = [freq for freq in frequencies.values() if freq % 2 != 0]
    even_frequencies = [freq for freq in frequencies.values() if freq % 2 == 0]
    max_odd = max(odd_frequencies)
    min_even = min(even_frequencies)
    return max_odd - min_even


```
