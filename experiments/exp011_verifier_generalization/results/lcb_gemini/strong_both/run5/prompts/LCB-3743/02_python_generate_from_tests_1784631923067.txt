from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    total_duration = sum(endTime[i] - startTime[i] for i in range(n))

    if k >= n - 1:
        return eventTime - total_duration

    gaps = []
    last_end = 0

    for i in range(n):
        if i == 0:
            gaps.append(startTime[i])  # gap before the first meeting
        else:
            gaps.append(startTime[i] - endTime[i - 1])  # gap between meetings
        last_end = endTime[i]

    gaps.append(eventTime - last_end)  # gap after the last meeting

    # Sort gaps in descending order to maximize the free time
    gaps.sort(reverse=True)

    # Combine the largest k gaps with the largest available gap
    max_free_time = max(gaps)
    for i in range(min(k, len(gaps) - 1)):
        max_free_time += gaps[i + 1]

    return max_free_time
```
