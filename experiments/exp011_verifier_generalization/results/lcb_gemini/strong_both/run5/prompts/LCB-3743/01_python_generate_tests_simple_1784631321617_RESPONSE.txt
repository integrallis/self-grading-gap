import pytest
from typing import List

from solution import maxFreeTime


def test_canonical_example_1():
    # spec: Example 1
    # Reschedule the meeting at [1, 2] to [2, 3].
    # The meetings become [2, 3] and [3, 5], which are adjacent.
    # This leaves the time [0, 2] free. The duration is 2.
    assert maxFreeTime(eventTime=5, k=1, startTime=[1, 3], endTime=[2, 5]) == 2


def test_canonical_example_2():
    # spec: Example 2
    # Reschedule the meeting at [2, 4] to [1, 3].
    # Meetings become [0, 1], [1, 3], [9, 10].
    # The free time between the second and third meeting is [3, 9]. The duration is 6.
    assert maxFreeTime(eventTime=10, k=1, startTime=[0, 2, 9], endTime=[1, 4, 10]) == 6


def test_canonical_example_3():
    # spec: Example 3
    # The meetings [0,1], [1,2], [2,3], [3,4], [4,5] cover the entire event time [0,5].
    # Total duration of meetings is 5, eventTime is 5.
    # No free time can be created.
    assert maxFreeTime(eventTime=5, k=2, startTime=[0, 1, 2, 3, 4], endTime=[1, 2, 3, 4, 5]) == 0


def test_all_meetings_movable():
    # When k=n, all meetings can be moved.
    # They can be packed together to create one large free time slot.
    # Meetings: [2,5] (d=3), [8,10] (d=2), [15,18] (d=3). Total duration = 8.
    # Max free time = eventTime - total duration = 20 - 8 = 12.
    assert maxFreeTime(eventTime=20, k=3, startTime=[2, 8, 15], endTime=[5, 10, 18]) == 12


def test_almost_all_meetings_movable():
    # When k >= n-1, all meetings can be made contiguous.
    # The maximum free time is the total event time minus the total duration of all meetings.
    # Meetings: [2,5] (d=3), [8,10] (d=2), [15,18] (d=3). Total duration = 8.
    # Max free time = eventTime - total duration = 20 - 8 = 12.
    assert maxFreeTime(eventTime=20, k=2, startTime=[2, 8, 15], endTime=[5, 10, 18]) == 12


def test_middle_gap_maximization():
    # The best strategy is to create a large gap in the middle.
    # Meetings: [0,1] (d=1), [10,15] (d=5), [29,30] (d=1).
    # k=1, we can move one meeting. Move [10,15].
    # Initial gaps: [1,10] (9), [15,29] (14). Max initial free time is 14.
    # Move meeting [10,15] to be adjacent to [0,1]. New position: [1,6].
    # Meetings become [0,1], [1,6], [29,30].
    # The new large gap is [6, 29], with duration 23.
    assert maxFreeTime(eventTime=30, k=1, startTime=[0, 10, 29], endTime=[1, 15, 30]) == 23


def test_start_gap_maximization():
    # The best strategy is to create a large gap at the beginning.
    # Meetings: [10,12] (d=2), [16,18] (d=2). k=1.
    # Initial gaps: [0,10] (10), [12,16] (4), [18,20] (2). Max initial free time is 10.
    # Move meeting [10,12] to be adjacent to [16,18]. New position: [14,16].
    # Meetings become [14,16], [16,18].
    # The new large gap is [0, 14], with duration 14.
    assert maxFreeTime(eventTime=20, k=1, startTime=[10, 16], endTime=[12, 18]) == 14


def test_end_gap_maximization():
    # The best strategy is to create a large gap at the end.
    # Meetings: [2,4] (d=2), [8,10] (d=2). k=1.
    # Initial gaps: [0,2] (2), [4,8] (4), [10,20] (10). Max initial free time is 10.
    # Move meeting [8,10] to be adjacent to [2,4]. New position: [4,6].
    # Meetings become [2,4], [4,6].
    # The new large gap is [6, 20], with duration 14.
    assert maxFreeTime(eventTime=20, k=1, startTime=[2, 8], endTime=[4, 10]) == 14


def test_no_improvement_with_moves():
    # Moving meetings does not always improve the maximum free time.
    # Meetings: [0,1] (d=1), [18,20] (d=2). k=1.
    # Initial gap: [1,18], duration 17.
    # Move [0,1] right against [18,20]: new pos [17,18]. Gap [0,17].
    # Move [18,20] left against [0,1]: new pos [1,3]. Gap [3,20], duration 17.
    # The maximum free time remains 17.
    assert maxFreeTime(eventTime=20, k=1, startTime=[0, 18], endTime=[1, 20]) == 17

def test_minimal_meetings_and_k():
    # Test with minimum n=2 and k=1.
    # This is similar to canonical example 1 but with different values.
    # Meetings: [5,10] (d=5), [15,20] (d=5). k=1.
    # Initial gaps: [0,5] (5), [10,15] (5), [20,30] (10). Max initial free time is 10.
    # Move [5,10] to [10,15]. Gap at start becomes [0,10].
    # Move [15,20] to [10,15]. Gap at end becomes [15,30] (15).
    # Max free time is 15.
    assert maxFreeTime(eventTime=30, k=1, startTime=[5, 15], endTime=[10, 20]) == 15
