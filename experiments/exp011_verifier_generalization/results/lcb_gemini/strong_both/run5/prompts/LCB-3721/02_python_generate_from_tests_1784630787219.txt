from typing import List

def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers  # Initialize mentions for each user
    online_status = [True] * numberOfUsers  # Track online status of each user
    offline_until = [0] * numberOfUsers  # Track when each user comes back online

    # Sort events by timestamp to process in order
    events.sort(key=lambda x: int(x[1]))

    for event in events:
        event_type, timestamp, detail = event
        timestamp = int(timestamp)

        if event_type == "OFFLINE":
            user_id = int(detail)
            # Set user offline
            online_status[user_id] = False
            offline_until[user_id] = timestamp + 60

        elif event_type == "MESSAGE":
            # Process user statuses that change before this message
            for user_id in range(numberOfUsers):
                if not online_status[user_id] and timestamp >= offline_until[user_id]:
                    online_status[user_id] = True

            # Handle mentions in message
            mentions_string = detail.split()  # Split mentions string into tokens
            mentioned_users = set()  # To avoid double counting

            for token in mentions_string:
                if token.startswith("id"):
                    user_id = int(token[2:])
                    if user_id not in mentioned_users:
                        mentions[user_id] += 1  # Count mention
                        mentioned_users.add(user_id)
                elif token == "ALL":
                    for user_id in range(numberOfUsers):
                        if user_id not in mentioned_users:
                            mentions[user_id] += 1  # Count mention for all users
                            mentioned_users.add(user_id)
                elif token == "HERE":
                    for user_id in range(numberOfUsers):
                        if online_status[user_id] and user_id not in mentioned_users:
                            mentions[user_id] += 1  # Count mention only for online users
                            mentioned_users.add(user_id)

    return mentions
```
