from typing import List
from solution import maxLength

# Note: The problem relies on a property of gcd and lcm.
# For two integers a, b:
#   prod([a, b]) = a * b
#   gcd([a, b]) = gcd(a, b)
#   lcm([a, b]) = (a * b) // gcd(a, b)
#   Therefore, lcm([a, b]) * gcd([a, b]) = a * b = prod([a, b])
# This means any subarray of length 2 is always "product equivalent".
# Since the input array length is guaranteed to be at least 2,
# the minimum possible answer is 2.


def test_canonical_example_1():
    """
    Tests the first example from the specification.
    Input: nums = [1, 2, 1, 2, 1, 1, 1]
    Output: 5
    """
    # spec: maxLength([1,2,1,2,1,1,1]) -> 5
    # The longest product equivalent subarray is [1, 2, 1, 1, 1] (length 5).
    # For this subarray:
    # prod([1, 2, 1, 1, 1]) = 2
    # gcd([1, 2, 1, 1, 1]) = 1
    # lcm([1, 2, 1, 1, 1]) = 2
    # The condition prod == gcd * lcm holds (2 == 1 * 2).
    # No subarray of length 6 or 7 is product equivalent.
    assert maxLength([1, 2, 1, 2, 1, 1, 1]) == 5


def test_canonical_example_2():
    """
    Tests the second example from the specification.
    Input: nums = [2, 3, 4, 5, 6]
    Output: 3
    """
    # spec: maxLength([2,3,4,5,6]) -> 3
    # The longest product equivalent subarray is [3, 4, 5] (length 3).
    # For this subarray:
    # prod([3, 4, 5]) = 60
    # gcd([3, 4, 5]) = 1
    # lcm([3, 4, 5]) = 60
    # The condition prod == gcd * lcm holds (60 == 1 * 60).
    # No subarray of length 4 or 5 is product equivalent.
    assert maxLength([2, 3, 4, 5, 6]) == 3


def test_canonical_example_3():
    """
    Tests the third example from the specification.
    Input: nums = [1, 2, 3, 1, 4, 5, 1]
    Output: 5
    """
    # spec: maxLength([1,2,3,1,4,5,1]) -> 5
    # A product equivalent subarray of length 5 is [3, 1, 4, 5, 1].
    # For this subarray:
    # prod([3, 1, 4, 5, 1]) = 60
    # gcd([3, 1, 4, 5, 1]) = 1
    # lcm([3, 1, 4, 5, 1]) = 60
    # The condition prod == gcd * lcm holds (60 == 1 * 60).
    # No longer subarray is product equivalent.
    assert maxLength([1, 2, 3, 1, 4, 5, 1]) == 5


def test_entire_array_is_solution():
    """
    Tests a case where the entire input array is the longest product-equivalent subarray.
    This often happens with numbers that are pairwise coprime.
    """
    # spec: maxLength([2, 3, 5]) -> 3
    # The longest possible subarray is the array itself, [2, 3, 5].
    # For this subarray:
    # prod([2, 3, 5]) = 30
    # gcd([2, 3, 5]) = 1
    # lcm([2, 3, 5]) = 30
    # The condition prod == gcd * lcm holds (30 == 1 * 30).
    # Therefore, the longest product equivalent subarray is the array itself, length 3.
    assert maxLength([2, 3, 5]) == 3


def test_only_length_two_subarrays_are_equivalent():
    """
    Tests a case where the longest product-equivalent subarray has length 2,
    which is the minimum possible answer for any valid input.
    """
    # spec: maxLength([6, 8, 9]) -> 2
    # Any subarray of length 2 is product equivalent. Max length is at least 2.
    # Let's check the subarray of length 3: [6, 8, 9].
    # prod([6, 8, 9]) = 6 * 8 * 9 = 432
    # gcd([6, 8, 9]) = gcd(gcd(6, 8), 9) = gcd(2, 9) = 1
    # lcm([6, 8, 9]) = lcm(lcm(6, 8), 9) = lcm(24, 9) = 72
    # prod != gcd * lcm (432 != 72).
    # Since no subarray of length 3 is product equivalent, the max length is 2.
    assert maxLength([6, 8, 9]) == 2


def test_minimum_input_length():
    """
    Tests an input at the minimum boundary for length (nums.length == 2).
    """
    # spec: maxLength([7, 10]) -> 2
    # The input has the minimum allowed length of 2.
    # The longest subarray is [7, 10] itself.
    # As established, any subarray of length 2 is product equivalent.
    # Therefore, the answer must be 2.
    assert maxLength([7, 10]) == 2
