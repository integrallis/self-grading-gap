from solution import maxLength


def test_canonical_example_1():
    """
    Tests the first example from the specification.
    Input: nums = [1, 2, 1, 2, 1, 1, 1]
    Output: 5
    """
    # spec: maxLength([1,2,1,2,1,1,1]) -> 5
    # The longest product equivalent subarray is [1, 2, 1, 1, 1] (length 5).
    # For this subarray:
    # prod([1, 2, 1, 1, 1]) = 2
    # gcd([1, 2, 1, 1, 1]) = 1
    # lcm([1, 2, 1, 1, 1]) = 2
    # The condition prod == gcd * lcm holds (2 == 1 * 2).
    # No subarray of length 6 or 7 is product equivalent.
    assert maxLength([1, 2, 1, 2, 1, 1, 1]) == 5
