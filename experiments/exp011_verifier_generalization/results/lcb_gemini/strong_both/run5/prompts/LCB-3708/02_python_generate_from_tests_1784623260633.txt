from typing import List

def zigzagTraversal(grid: List[List[int]]) -> List[int]:
    result = []
    for i in range(len(grid)):
        if i % 2 == 0:
            # Even row: traverse right
            row = grid[i]
        else:
            # Odd row: traverse left
            row = grid[i][::-1]
        # Append every alternate cell
        result.extend(row[j] for j in range(len(row)) if j % 2 == 0)
    return result
```
