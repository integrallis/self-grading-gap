## Implementation

```python
def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]
    current_pos = 0
    while True:
        prefix_idx = s.find(prefix, current_pos)
        if prefix_idx == -1:
            return False  # No more prefixes found
        suffix_idx = s.find(suffix, prefix_idx + len(prefix))
        if suffix_idx != -1:
            return True  # Found a valid prefix-suffix pair
        current_pos = prefix_idx + 1
    return False
```
