def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]
    # Check if there exists a substring in s that starts with prefix and ends with suffix.
    for i in range(len(s) - len(suffix) - len(prefix) + 1):
        if s.startswith(prefix, i) and s.endswith(suffix, i + len(prefix)):
            return True
    return False

```
