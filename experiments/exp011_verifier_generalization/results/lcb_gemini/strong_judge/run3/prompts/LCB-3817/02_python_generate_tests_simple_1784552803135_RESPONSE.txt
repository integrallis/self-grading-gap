from solution import maxContainers

def test_canonical_example_1():
    # n = 2, w = 3, maxWeight = 15
    # The deck has 4 cells (2x2), and each container weighs 3. 
    # Total weight = 4 * 3 = 12, which does not exceed maxWeight of 15.
    assert maxContainers(2, 3, 15) == 4

def test_canonical_example_2():
    # n = 3, w = 5, maxWeight = 20
    # The deck has 9 cells (3x3), and each container weighs 5. 
    # Maximum weight we can load = 20; hence we can load 4 containers (4 * 5 = 20).
    assert maxContainers(3, 5, 20) == 4

def test_single_container_heavier_than_max_weight():
    # n = 10, w = 100, maxWeight = 99
    # The deck has 100 cells (10x10), but each container weighs 100, which exceeds maxWeight of 99.
    # Hence, we cannot load any containers.
    assert maxContainers(10, 100, 99) == 0

def test_minimum_constraint_values():
    # n = 1, w = 1, maxWeight = 1
    # The deck has 1 cell (1x1), and each container weighs 1. 
    # Total weight = 1 * 1 = 1, which does not exceed maxWeight of 1.
    assert maxContainers(1, 1, 1) == 1

def test_maximum_constraint_values():
    # n = 1000, w = 1000, maxWeight = 10**9
    # The deck has 1,000,000 cells (1000x1000), and each container weighs 1000.
    # Total weight we can load = 10^9; hence we can load 1000 containers (1000 * 1000 = 10^6).
    assert maxContainers(1000, 1000, 10**9) == 1000000
