def maxDistance(s: str, k: int) -> int:
    # Initial position at the origin
    x, y = 0, 0
    # Track the maximum distance
    max_distance = 0

    # Iterate over each character in the string
    for char in s:
        if char == 'N':
            y += 1
        elif char == 'S':
            y -= 1
        elif char == 'E':
            x += 1
        elif char == 'W':
            x -= 1
        # Calculate the current Manhattan distance
        current_distance = abs(x) + abs(y)
        # Update max_distance considering k changes
        max_distance = max(max_distance, current_distance + k)

    # The maximum distance we can reach
    return max_distance
```
