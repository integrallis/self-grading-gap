from solution import maxDistance

def test_canonical_example_1():
    # Given input: s = "NWSE", k = 1
    # Change s[2] from 'S' to 'N' to maximize distance
    # Final string: "NWNE" -> Positions: (0,1) -> (-1,1) -> (-1,2) -> (0,2)
    # Maximum Manhattan distance = 3
    assert maxDistance("NWSE", 1) == 3

def test_canonical_example_2():
    # Given input: s = "NSWWEW", k = 3
    # Change s[1] from 'S' to 'N', and s[4] from 'E' to 'W' to maximize distance
    # Final string: "NNWWWW" -> Positions: (0,1) -> (0,2) -> (0,3) -> (-1,3) -> (-2,3) -> (-3,3)
    # Maximum Manhattan distance = 6
    assert maxDistance("NSWWEW", 3) == 6
