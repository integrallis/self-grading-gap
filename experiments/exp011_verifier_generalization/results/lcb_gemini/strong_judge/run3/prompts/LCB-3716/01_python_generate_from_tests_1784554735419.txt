from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n < 2:
        return n
    dp = [{} for _ in range(n)]  # Track lengths of subsequences with last differences

    for i in range(n):
        for j in range(i):
            diff = abs(nums[i] - nums[j])
            # Base case: a subsequence with two elements
            best_prev_len = 1
            for prev_diff, length in dp[j].items():
                if prev_diff >= diff:
                    best_prev_len = max(best_prev_len, length)
            dp[i][diff] = max(dp[i].get(diff, 0), 1 + best_prev_len)

    return max(max(lengths.values(), default=0) for lengths in dp)
```
