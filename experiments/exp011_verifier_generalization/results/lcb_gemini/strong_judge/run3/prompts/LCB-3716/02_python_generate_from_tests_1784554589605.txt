from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n < 2:
        return n
    dp = [1] * n  # dp[i] will store the length of the longest subsequence ending at index i
    differences = []

    for i in range(n):
        for j in range(i):
            diff1 = abs(nums[i] - nums[j])
            if j == 0 or (len(differences) > 0 and diff1 <= differences[-1]):
                dp[i] = max(dp[i], dp[j] + 1)
                if len(differences) == 0 or diff1 != differences[-1]:
                    differences.append(diff1)

    return max(dp)
```
