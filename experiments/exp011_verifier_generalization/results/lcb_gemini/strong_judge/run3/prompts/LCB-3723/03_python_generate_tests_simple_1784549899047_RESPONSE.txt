from typing import List
from solution import sumOfGoodNumbers

def test_canonical_example_1():
    # spec: input [1, 3, 2, 1, 5, 4], k = 2
    # good numbers: nums[1] = 3 (greater than nums[-1] and nums[3]), nums[4] = 5 (greater than nums[2] and nums[6]), nums[5] = 4 (greater than nums[3] and nums[7])
    # sum = 3 + 5 + 4 = 12
    assert sumOfGoodNumbers([1, 3, 2, 1, 5, 4], 2) == 12

def test_canonical_example_2():
    # spec: input [2, 1], k = 1
    # good number: nums[0] = 2 (greater than nums[1])
    # sum = 2
    assert sumOfGoodNumbers([2, 1], 1) == 2

def test_good_number_greater_than_both_neighbors():
    # spec: input [2, 10, 3], k = 1
    # good number: nums[1] = 10 (greater than nums[0] and nums[2])
    # sum = 10
    assert sumOfGoodNumbers([2, 10, 3], 1) == 10

def test_not_good_due_to_equal_neighbor():
    # spec: input [10, 5, 5], k = 1
    # good number: nums[0] = 10 (greater than nums[1])
    # nums[1] and nums[2] are not good because they are not greater than their neighbors
    # sum = 10
    assert sumOfGoodNumbers([10, 5, 5], 1) == 10

def test_no_good_numbers():
    # spec: input [5, 5, 5, 5], k = 1
    # no good numbers since all are equal
    # sum = 0
    assert sumOfGoodNumbers([5, 5, 5, 5], 1) == 0

def test_max_length_good_numbers():
    # spec: input with maximum constraint length 100
    # good numbers: all numbers from 1 to 100 are greater than their neighbors
    # sum = 1 + 2 + 3 + ... + 100 = 5050
    assert sumOfGoodNumbers(list(range(1, 101)), 1) == 5050

def test_max_value_good_numbers():
    # spec: input with maximum value 1000
    # good number: nums[0] = 1000 (greater than nums[1])
    # sum = 1000
    assert sumOfGoodNumbers([1000, 500], 1) == 1000
