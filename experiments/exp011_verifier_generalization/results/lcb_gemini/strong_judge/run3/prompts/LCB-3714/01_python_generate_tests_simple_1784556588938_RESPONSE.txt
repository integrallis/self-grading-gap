from solution import minMaxSums

def test_canonical_example_1():
    # spec: minMaxSums([1, 2, 3], 2) -> 24
    # All subsequences with at most 2 elements are:
    # [1], [2], [3], [1, 2], [1, 3], [2, 3]
    # Their min/max sums give 24 in total.
    assert minMaxSums([1, 2, 3], 2) == 24

def test_canonical_example_2():
    # spec: minMaxSums([5, 0, 6], 1) -> 22
    # All subsequences with exactly 1 element are:
    # [5], [0], [6]
    # Their min/max sums give 22 in total.
    assert minMaxSums([5, 0, 6], 1) == 22

def test_canonical_example_3():
    # spec: minMaxSums([1, 1, 1], 2) -> 12
    # All subsequences with at most 2 elements (with duplicates) are:
    # [1], [1], [1], [1, 1], [1, 1], [1, 1]
    # Their min/max sums give 12 in total.
