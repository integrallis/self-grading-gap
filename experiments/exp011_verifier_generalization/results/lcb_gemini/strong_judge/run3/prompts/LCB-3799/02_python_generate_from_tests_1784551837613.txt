from typing import List


def totalNumbers(digits: List[int]) -> int:
    from collections import Counter
    count = Counter(digits)
    even_digits = [d for d in count if d % 2 == 0]
    total = 0

    for last_digit in even_digits:
        if last_digit == 0:
            continue  # Skip if last digit is 0, as it can't be a leading digit
        count[last_digit] -= 1  # Use this digit as the last digit

        # Count valid leading digits
        for first_digit in count:
            if count[first_digit] > 0 and first_digit != 0:
                # Select first digit
                count[first_digit] -= 1  # Use this digit as the first digit

                # Count the remaining digits for the middle position
                total += sum(count.values())

                count[first_digit] += 1  # Restore the count for first digit

        count[last_digit] += 1  # Restore the count for last digit
    return total

```
