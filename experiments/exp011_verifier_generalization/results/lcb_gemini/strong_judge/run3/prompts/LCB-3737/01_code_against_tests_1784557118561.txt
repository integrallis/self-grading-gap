```python
from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize a DP table where dp[i][j] is the minimum cost to paint up to house i with color j
    dp = [[0] * 3 for _ in range(n)]

    # Base cases: Cost of painting the first house with each color
    for j in range(3):
        dp[0][j] = cost[0][j]

    for i in range(1, n):
        for j in range(3):
            # Calculate the minimum cost for painting house i with color j
            dp[i][j] = cost[i][j] + min(
                dp[i-1][(j+1) % 3],  # Previous house with different color
                dp[i-1][(j+2) % 3]   # Previous house with different color
            )

    # Minimum cost considering the restrictions
    min_cost = float('inf')
    for j in range(3):
        for k in range(3):
            if j != k:  # Ensure opposite house has a different color
                # Combine the costs of the last house and its opposite
                min_cost = min(min_cost, dp[n-1][j] + dp[n//2 - 1][k])

    return min_cost

```
