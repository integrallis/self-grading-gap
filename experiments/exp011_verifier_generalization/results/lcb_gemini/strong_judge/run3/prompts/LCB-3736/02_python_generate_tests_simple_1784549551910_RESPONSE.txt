from solution import findValidPair

def test_canonical_example_1():
    # Input: "2523533"
    # Digit '2' appears 2 times and digit '3' appears 3 times.
    # Valid pair is "23".
    assert findValidPair("2523533") == "23"

def test_canonical_example_2():
    # Input: "221"
    # Digit '2' appears 2 times and digit '1' appears 1 time.
    # Valid pair is "21".
    assert findValidPair("221") == "21"

def test_canonical_example_3():
    # Input: "22"
    # Both digits are '2' and they are equal, no valid pairs.
    assert findValidPair("22") == ""

def test_multiple_potential_valid_pairs():
    # Input: "122333"
    # Valid pairs are "12" (count '1' is 1, count '2' is 2) and "23" (count '2' is 2, count '3' is 3).
    # The first valid pair is "12".
    assert findValidPair("122333") == "12"

def test_non_equal_adjacent_invalid_due_to_counts():
    # Input: "223"
    # Digit '2' appears 2 times and digit '3' appears 1 time.
    # The pair "22" is invalid (equal digits), and "23" is also invalid (count '2' is 2, but count '3' is 1).
    # Hence, the output should be "".
    assert findValidPair("223") == ""
