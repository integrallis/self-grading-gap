from typing import List
from solution import countPartitions

def test_canonical_example_1():
    # spec: input: [10, 10, 3, 7, 6]
    # partitions are: [10], [10, 3, 7, 6] -> 10 - 26 = -16 (even)
    # [10, 10], [3, 7, 6] -> 20 - 16 = 4 (even)
    # [10, 10, 3], [7, 6] -> 23 - 13 = 10 (even)
    # [10, 10, 3, 7], [6] -> 30 - 6 = 24 (even)
    # Total even differences: 4
    assert countPartitions([10, 10, 3, 7, 6]) == 4

def test_canonical_example_2():
    # spec: input: [1, 2, 2]
    # partitions are: [1], [2, 2] -> 1 - 4 = -3 (odd)
    # [1, 2], [2] -> 3 - 2 = 1 (odd)
    # Total even differences: 0
    assert countPartitions([1, 2, 2]) == 0

def test_canonical_example_3():
    # spec: input: [2, 4, 6, 8]

    # [2, 4], [6, 8] -> 6 - 14 = -8 (even)
    # [2, 4, 6], [8] -> 12 - 8 = 4 (even)

def test_minimum_array_length():
    # spec: input: [5, 5]

    # spec: input: [1] * 100

    # ... (repeated for each partition) -> total of 99 even differences
    assert countPartitions([1] * 100) == 99
