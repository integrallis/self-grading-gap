```python
from typing import List

def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Function to check if we can achieve a certain max weight
    def canAchieve(max_weight):
        out_degree_used = [0] * n
        reachable = [False] * n
        reachable[0] = True
        
        # Iteratively find nodes that can reach the already-reachable set
        # until no more nodes can be made reachable.
        while True:
            nodes_added_this_pass = False
            for u, v, w in edges:
                # Check for an edge from an unreachable node 'u' to a reachable node 'v'
                if not reachable[u] and reachable[v] and w <= max_weight:
                    # Check if 'u' has available out-degree capacity
                    if out_degree_used[u] < threshold:
                        reachable[u] = True
                        out_degree_used[u] += 1
                        nodes_added_this_pass = True
            
            if not nodes_added_this_pass:
                break
        
        return all(reachable)

    # Binary search over the edge weights
    low, high = 1, 10**6
    answer = -1
    while low <= high:
        mid = (low + high) // 2
        if canAchieve(mid):
            answer = mid
            high = mid - 1
        else:
            low = mid + 1

    return answer

```
