from typing import List
import heapq


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Create a graph representation
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[v].append((u, w))  # Reversed for reachability from 0

    # Function to check if we can achieve a certain max weight
    def canAchieve(max_weight):
        out_edges_count = [0] * n
        reachable = [False] * n
        queue = []
        # Start with node 0
        reachable[0] = True
        for v in range(n):
            for u, w in graph[v]:
                if w <= max_weight:
                    out_edges_count[v] += 1
                    if out_edges_count[v] <= threshold:
                        heapq.heappush(queue, (w, v, u))

        while queue:
            _, v, u = heapq.heappop(queue)
            if not reachable[u]:
                reachable[u] = True
                for next_u, w in graph[u]:
                    if w <= max_weight:
                        out_edges_count[u] += 1
                        if out_edges_count[u] <= threshold:
                            heapq.heappush(queue, (w, u, next_u))

        return all(reachable)

    # Binary search over the edge weights
    low, high = 1, 10**6
    answer = -1
    while low <= high:
        mid = (low + high) // 2
        if canAchieve(mid):
            answer = mid
            high = mid - 1
        else:
            low = mid + 1

    return answer

```
