def test_minimal_graph_n_equals_2():
    # Tests the lower bound of n=2.
    assert minMaxWeight(n=2, edges=[[1, 0, 100]], threshold=1) == 100

def test_max_threshold_is_not_a_constraint():
    # Tests the upper bound of threshold=n-1, which should not limit edge choices.
    # The result should be determined by connectivity and weight minimization.
    # min max-weight path from 1 to 0 is 1->2->0 (max edge weight 8).
    # min max-weight path from 2 to 0 is 2->0 (max edge weight 8).
    assert minMaxWeight(n=3, edges=[[1, 0, 10], [1, 2, 5], [2, 0, 8]], threshold=2) == 8

def test_multiple_edges_between_nodes():
    # Tests that the algorithm correctly handles multiple edges between two nodes.
    # Path 2->1->0 is required. Edge 2->1 has weight 5.
    # To complete the path, edge 1->0 with weight 10 must be chosen over 1->0 with weight 20.
    # This makes the minimum max-weight 10.
    assert minMaxWeight(n=3, edges=[[1, 0, 10], [2, 1, 5], [1, 0, 20]], threshold=1) == 10
