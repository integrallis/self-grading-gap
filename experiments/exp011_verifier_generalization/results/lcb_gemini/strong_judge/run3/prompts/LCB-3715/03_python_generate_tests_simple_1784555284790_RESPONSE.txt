from typing import List
from solution import maximumCoins

def test_canonical_example_1():
    # coins = [[8,10,1],[1,3,2],[5,6,4]], k = 4
    # Selecting bags at positions [3, 4, 5, 6] gives 2 + 0 + 4 + 4 = 10
    assert maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) == 10

def test_canonical_example_2():
    # coins = [[1,10,3]], k = 2
    # Selecting bags at positions [1, 2] gives 3 + 3 = 6
    assert maximumCoins([[1, 10, 3]], 2) == 6

def test_boundary_k_equals_1():
    # coins = [[1,2,10], [5,6,20]], k = 1
    # The maximum c_i is 20
    assert maximumCoins([[1, 2, 10], [5, 6, 20]], 1) == 20

def test_large_k_covering_all_segments():
    # coins = [[1,2,10], [100,100,20]], k = 100
    # All coins can be collected: (2-1+1)*10 + (100-100+1)*20 = 10 + 20 = 40
    assert maximumCoins([[1, 2, 10], [100, 100, 20]], 100) == 40

def test_segments_of_length_1():
    # coins = [[1,1,10], [3,5,2], [7,7,10]], k = 3
    # The maximum for a consecutive window of size 3 is from bags [1, 2, 3]
    # which yields 10 + 0 + 2 = 12
    assert maximumCoins([[1, 1, 10], [3, 5, 2], [7, 7, 10]], 3) == 12

def test_large_coordinate_values():
    # coins = [[10**9 - 2, 10**9 - 1, 100], [10**9 + 1, 10**9 + 1, 200]], k = 3
    # The optimal window of size 3 is [10**9-1, 10**9, 10**9+1]
    # which spans the gap and collects 100 + 0 + 200 = 300
    assert maximumCoins([[10**9 - 2, 10**9 - 1, 100], [10**9 + 1, 10**9 + 1, 200]], 3) == 300
