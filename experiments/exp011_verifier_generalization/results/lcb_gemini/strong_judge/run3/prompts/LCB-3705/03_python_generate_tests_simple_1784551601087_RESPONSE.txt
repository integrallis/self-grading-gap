from typing import List
from solution import largestInteger

def test_canonical_example_1():
    # spec: largestInteger([3, 9, 2, 1, 7], 3) -> 7
    # Explanation:
    # 1 appears in 2 subarrays of size 3: [9, 2, 1] and [2, 1, 7].
    # 2 appears in 3 subarrays of size 3: [3, 9, 2], [9, 2, 1], [2, 1, 7].
    # 3 appears in 1 subarray of size 3: [3, 9, 2].
    # 7 appears in 1 subarray of size 3: [2, 1, 7].
    # 9 appears in 2 subarrays of size 3: [3, 9, 2], and [9, 2, 1].
    # The largest integer that appears in exactly one subarray of size k is 7.
    assert largestInteger([3, 9, 2, 1, 7], 3) == 7

def test_canonical_example_2():
    # spec: largestInteger([3, 9, 7, 2, 1, 7], 4) -> 3
    # Explanation:

    # 2 appears in 3 subarrays of size 4: [3, 9, 7, 2], [9, 7, 2, 1], [7, 2, 1, 7].

def test_boundary_case_minimum_length():
    # spec: largestInteger([42], 1) -> 42

def test_maximum_length():
    # spec: largestInteger([i for i in range(51)], 50) -> 50
    # Explanation:
    # The only subarray of size 50 is [0, 1, 2, ..., 49].
    # Each number from 0 to 49 appears in exactly one subarray of size 50.
    # The largest integer that appears in exactly one subarray of size k is 50.
    assert largestInteger([i for i in range(51)], 50) == 50

def test_maximum_element_value():

    assert largestInteger([50] * 50, 1) == -1
