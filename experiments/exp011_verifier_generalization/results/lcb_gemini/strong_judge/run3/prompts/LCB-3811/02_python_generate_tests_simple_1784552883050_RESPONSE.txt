from solution import reverseDegree

def test_canonical_example_1():
    # For input "abc":
    # 'a' -> 26 * 1 = 26
    # 'b' -> 25 * 2 = 50
    # 'c' -> 24 * 3 = 72
    # Total = 26 + 50 + 72 = 148
    assert reverseDegree("abc") == 148

def test_canonical_example_2():
    # For input "zaza":
    # 'z' -> 1 * 1 = 1
    # 'a' -> 26 * 2 = 52
    # 'z' -> 1 * 3 = 3
    # 'a' -> 26 * 4 = 104
    # Total = 1 + 52 + 3 + 104 = 160
    assert reverseDegree("zaza") == 160

def test_minimum_length():
    # For input "a":
    # 'a' -> 26 * 1 = 26
    assert reverseDegree("a") == 26

def test_maximum_length():
    # For input "z" * 1000:
    # Each 'z' contributes 1 * index (1 to 1000)
    # Total = 1 * (1 + 2 + ... + 1000) = 1 * (1000 * 1001 / 2) = 500500
    assert reverseDegree("z" * 1000) == 500500
