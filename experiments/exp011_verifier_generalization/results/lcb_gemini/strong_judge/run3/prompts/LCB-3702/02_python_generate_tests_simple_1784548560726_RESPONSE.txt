from typing import List
from solution import maxLength

def test_canonical_example_1():
    # spec: longest product equivalent subarray in [1,2,1,2,1,1,1] is [1, 2, 1, 1, 1]
    # prod([1, 2, 1, 1, 1]) = 2, gcd([1, 2, 1, 1, 1]) = 1, lcm([1, 2, 1, 1, 1]) = 2
    assert maxLength([1, 2, 1, 2, 1, 1, 1]) == 5

def test_canonical_example_2():
    # spec: longest product equivalent subarray in [2,3,4,5,6] is [3, 4, 5]
    # prod([3, 4, 5]) = 60, gcd([3, 4, 5]) = 1, lcm([3, 4, 5]) = 60
    assert maxLength([2, 3, 4, 5, 6]) == 3

def test_canonical_example_3():
    # spec: longest product equivalent subarray in [1,2,3,1,4,5,1] is [3, 1, 4, 5, 1]
    # prod([3, 1, 4, 5, 1]) = 60, gcd([3, 1, 4, 5, 1]) = 1, lcm([3, 1, 4, 5, 1]) = 60
    assert maxLength([1, 2, 3, 1, 4, 5, 1]) == 5

def test_length_one_subarray():
    # spec: longest product equivalent subarray in [2, 4, 6] is [2]
    # prod([2]) = 2, gcd([2]) = 2, lcm([2]) = 2
    assert maxLength([2, 4, 6]) == 1

def test_entire_array_product_equivalent():
    # spec: longest product equivalent subarray in [2, 3, 5, 7] is the entire array
    # prod([2, 3, 5, 7]) = 210, gcd([2, 3, 5, 7]) = 1, lcm([2, 3, 5, 7]) = 210
    assert maxLength([2, 3, 5, 7]) == 4
