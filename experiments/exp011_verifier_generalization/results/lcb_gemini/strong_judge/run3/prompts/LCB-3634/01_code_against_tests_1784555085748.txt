## Implementation

```python
def calculateScore(s: str) -> int:
    score = 0
    marked = set()
    n = len(s)
    for i in range(n):
        if i in marked:
            continue
        for j in range(i - 1, -1, -1):
            if j in marked:
                continue
            if ord(s[i]) + ord(s[j]) == 219:  # mirror check: a+z=219, b+y=219, ...
                score += i - j
                marked.add(i)
                marked.add(j)
                break
    return score
```
