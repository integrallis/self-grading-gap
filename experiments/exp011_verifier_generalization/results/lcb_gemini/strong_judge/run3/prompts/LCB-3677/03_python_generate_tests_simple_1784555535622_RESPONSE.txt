from typing import List
from solution import maximumAmount

def test_canonical_example_1():
    # spec: coins = [[0, 1, -1], [1, -2, 3], [2, -3, 4]]
    # Explanation: 
    # Start at (0, 0) with 0 coins (total coins = 0).
    # Move to (0, 1), gaining 1 coin (total = 1).
    # Move to (1, 1), where there's a robber stealing 2 coins. 
    # Neutralize here (total = 1).
    # Move to (1, 2), gaining 3 coins (total = 4).
    # Move to (2, 2), gaining 4 coins (total = 8).
    assert maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) == 8

def test_canonical_example_2():
    # spec: coins = [[10, 10, 10], [10, 10, 10]]
    # Explanation:

    # Move to (0, 1), gaining 10 coins (total = 20).
    # Move to (0, 2), gaining another 10 coins (total = 30).
    # Move to (1, 2), gaining the final 10 coins (total = 40).
    assert maximumAmount([[10, 10, 10], [10, 10, 10]]) == 40
