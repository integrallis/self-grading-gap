from typing import List

def zigzagTraversal(grid: List[List[int]]) -> List[int]:
    result = []
    for i in range(len(grid)):
        row = grid[i]
        if i % 2 == 0:
            # Traverse right
            result.extend(row[j] for j in range(0, len(row), 2))
        else:
            # Traverse left
            result.extend(row[j] for j in range(len(row) - 1, -1, -2))
    return result
