from typing import List

def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers
    online_status = [True] * numberOfUsers
    event_dict = []

    for event in events:
        event_dict.append((int(event[1]), event[0], event[2]))

    # Sort events by timestamp
    event_dict.sort(key=lambda x: x[0])

    for timestamp, event_type, detail in event_dict:
        if event_type == "OFFLINE":
            user_id = int(detail)
            online_status[user_id] = False
            # User will come back online after 60 time units
        elif event_type == "MESSAGE":
            mentions_in_message = []
            if detail == "ALL":
                mentions_in_message = range(numberOfUsers)
            elif detail == "HERE":
                mentions_in_message = [i for i in range(numberOfUsers) if online_status[i]]
            else:
                mentions_in_message = [int(id[2:]) for id in detail.split()]

            for user_id in mentions_in_message:
                mentions[user_id] += 1

            # Process any offline users coming back online before the next message
            for i in range(numberOfUsers):
                if not online_status[i] and timestamp >= event_dict[-1][0] - 60:
                    online_status[i] = True

    return mentions
