from solution import maxDifference

def test_canonical_example_1():
    # spec: maxDifference("aaaaabbc") -> 3
    # 'a' has frequency 5 (odd), 'b' has frequency 2 (even); 5 - 2 = 3
    assert maxDifference("aaaaabbc") == 3

def test_canonical_example_2():
    # spec: maxDifference("abcabcab") -> 1
    # 'a' has frequency 3 (odd), 'c' has frequency 2 (even); 3 - 2 = 1
    assert maxDifference("abcabcab") == 1

def test_multiple_even_frequencies():
    # spec: maxDifference("aaaaabbbbcc") -> 3
    # 'a' has frequency 5 (odd), 'b' has frequency 4 (even), 'c' has frequency 2 (even);
    # max(odd_freqs) = 5, min(even_freqs) = 2; 5 - 2 = 3
    assert maxDifference("aaaaabbbbcc") == 3

def test_minimum_length():
    # spec: maxDifference("aab") -> -1
    # 'a' has frequency 2 (even), 'b' has frequency 1 (odd);
    # 1 - 2 = -1
    assert maxDifference("aab") == -1
