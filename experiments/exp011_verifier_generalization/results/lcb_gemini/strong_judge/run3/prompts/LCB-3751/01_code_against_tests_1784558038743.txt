## Implementation

```python
from typing import List


def maxFrequency(nums: List[int], k: int) -> int:
    nums.sort()
    max_freq = 0
    l = 0
    total_sum = 0

    for r in range(len(nums)):
        total_sum += nums[r]
        while (r - l + 1) * nums[r] - total_sum > k:
            total_sum -= nums[l]
            l += 1
        max_freq = max(max_freq, r - l + 1)

    return max_freq

```
