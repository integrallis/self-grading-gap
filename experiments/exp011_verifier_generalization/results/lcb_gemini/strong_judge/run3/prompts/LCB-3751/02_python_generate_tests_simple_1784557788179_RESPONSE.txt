from typing import List
from solution import maxFrequency

def test_canonical_example_1():
    # spec: maxFrequency([1, 2, 3, 4, 5, 6], 1) -> 2
    # Explanation: After adding -5 to nums[2..5], 1 has a frequency of 2 in [1, 2, -2, -1, 0, 1].
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2

def test_canonical_example_2():
    # spec: maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) -> 4
    # Explanation: After adding 8 to nums[1..9], 10 has a frequency of 4 in [10, 10, 11, 12, 13, 13, 12, 11, 10, 10].
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4

def test_strategy_sacrificing_k():
    # Test where the optimal strategy involves sacrificing an existing k
    # Initial array: [1, 5, 1, 5, 5, 5]; k = 1
    # By adding -4 to nums[1..4], we can convert the five 5's to 1's
    # Resulting array: [1, 1, 1, 1, 1, 1]; frequency of 1 is 5
    assert maxFrequency([1, 5, 1, 5, 5, 5], 1) == 5

def test_no_operation_optimal():
    # Test where no operation is optimal because all elements are already k
    # Initial array: [7, 7, 7]; k = 7
    # Frequency of 7 is already 3
    assert maxFrequency([7, 7, 7], 7) == 3

def test_k_not_present():
    # Test where k is not present in the initial array
    # Initial array: [2, 3, 4]; k = 1
    # We can only add -1 to all elements or any other number,
    # but the maximum frequency we can achieve for 1 is 1 (by adding -1 to all).
    assert maxFrequency([2, 3, 4], 1) == 1

def test_single_element_array():
    # Test for the n=1 boundary condition
    # Initial array: [42]; k = 10
    # The only element is 42; we can add -32 to make it 10
    # The frequency of 10 will be 1
    assert maxFrequency([42], 10) == 1
