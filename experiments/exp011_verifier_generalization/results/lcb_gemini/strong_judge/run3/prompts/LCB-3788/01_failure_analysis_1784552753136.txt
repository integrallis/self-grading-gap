from typing import List
from collections import Counter

def maxSum(nums: List[int]) -> int:
    counts = Counter(nums)
    if all(c > 1 for c in counts.values()):
        return max(nums)
    unique_nums = set()
    for num in nums:
        if num > 0:
            unique_nums.add(num)
    return sum(unique_nums) if unique_nums else max(nums)
