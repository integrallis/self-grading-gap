from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    # dp[i][j][k] represents the maximum coins we can collect to reach (i, j) with k neutralizations used.
    dp = [[[-float('inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0]  # Starting point

    for i in range(m):
        for j in range(n):
            for k in range(3):
                if i == 0 and j == 0:
                    continue  # Skip the starting point
                # Collect coins from the upper cell
                if i > 0:
                    coin_value = coins[i][j]
                    if coin_value < 0 and k > 0:
                        dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k-1])  # Neutralize robbery
                    else:
                        dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k] + coin_value)
                # Collect coins from the left cell
                if j > 0:
                    coin_value = coins[i][j]

                        dp[i][j][k] = max(dp[i][j][k], dp[i][j-1][k] + coin_value)

    # The maximum coins we can collect to reach (m-1, n-1) with at most 2 neutralizations
