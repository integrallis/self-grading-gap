## Implementation

```python
def findValidPair(s: str) -> str:
    from collections import Counter

    # Count occurrences of each digit in the string
    counts = Counter(s)

    # Iterate over the string to find valid pairs
    for i in range(len(s) - 1):
        first_digit = s[i]
        second_digit = s[i + 1]

        # Check if the digits are different
        if first_digit != second_digit:
            # Check the count condition
            if counts[first_digit] == int(first_digit) and counts[second_digit] == int(second_digit):
                return first_digit + second_digit

    return ""
```
