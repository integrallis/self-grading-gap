from solution import findValidPair

def test_canonical_example_1():
    # spec: input "2523533" -> digit '2' appears 2 times and digit '3' appears 3 times.
    # Therefore, the output should be "23".
    assert findValidPair("2523533") == "23"

def test_canonical_example_2():
    # spec: input "221" -> digit '2' appears 2 times and digit '1' appears 1 time.
    # Therefore, the output should be "21".
    assert findValidPair("221") == "21"

def test_canonical_example_3():
    # spec: input "22" -> both digits are '2' and they are equal.
    # Therefore, there are no valid adjacent pairs, so the output should be "".
    assert findValidPair("22") == ""

def test_first_valid_pair_multiple_exists():
    # input "2244441" -> digit '2' appears 2 times and digit '4' appears 4 times.
    # The first non-equal adjacent pair is "24", which is valid.
    assert findValidPair("2244441") == "24"

def test_maximum_length_input():
    # input "1" + "22" + "3" * 97 -> digit '1' appears 1 time, digit '2' appears 2 times, digit '3' appears 97 times.
    # The first non-equal adjacent pair is "12", which is valid.
    assert findValidPair("1" + "22" + "3" * 97) == "12"
