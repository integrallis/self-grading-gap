from solution import calculateScore

def test_canonical_example_1():
    # spec: calculateScore("aczzx") -> 5
    # Explanation: 
    # i = 0: no j, skip
    # i = 1: no j, skip
    # i = 2: j = 0, score += (2 - 0) = 2
    # i = 3: no j, skip
    # i = 4: j = 1, score += (4 - 1) = 3
    # Total score = 2 + 3 = 5
    assert calculateScore("aczzx") == 5

def test_canonical_example_2():
    # spec: calculateScore("abcdef") -> 0
    # Explanation: 
    # For every index i, there is no index j that satisfies the conditions.
    # Thus, the total score remains 0.
    assert calculateScore("abcdef") == 0
