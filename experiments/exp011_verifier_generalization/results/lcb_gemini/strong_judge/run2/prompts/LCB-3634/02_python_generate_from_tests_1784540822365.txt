```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # Create mirror mapping

    for i in range(len(s)):
        if marked[i]:  # Skip if already marked
            continue
        target = mirror[s[i]]
        # Find the closest unmarked index j
        for j in range(i - 1, -1, -1):
            if not marked[j] and s[j] == target:
                marked[i] = marked[j] = True  # Mark both indices
                score += i - j  # Update score
                break
    return score

```
