from collections import defaultdict

def calculateScore(s: str) -> int:
    score = 0
    available_indices = defaultdict(list)
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # Create mirror mapping

    for i, char in enumerate(s):
        target = mirror[char]
        if available_indices[target]:
            j = available_indices[target].pop()
            score += i - j
        else:
            available_indices[char].append(i)

    return score

```
