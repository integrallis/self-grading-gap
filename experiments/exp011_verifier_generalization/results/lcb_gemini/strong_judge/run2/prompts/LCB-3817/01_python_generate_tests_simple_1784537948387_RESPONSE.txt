from solution import maxContainers

def test_canonical_example_1():
    # Input: n = 2, w = 3, maxWeight = 15
    # The deck can hold 2 * 2 = 4 containers, and total weight is 4 * 3 = 12 which is <= 15
    assert maxContainers(2, 3, 15) == 4

def test_canonical_example_2():
    # Input: n = 3, w = 5, maxWeight = 20
    # The deck can hold 3 * 3 = 9 containers, but total weight is 9 * 5 = 45 which exceeds 20
    # Maximum number of containers = floor(20 / 5) = 4
    assert maxContainers(3, 5, 20) == 4
