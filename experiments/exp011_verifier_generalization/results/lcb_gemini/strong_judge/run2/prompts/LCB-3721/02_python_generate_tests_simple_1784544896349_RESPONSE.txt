from typing import List
from solution import countMentions

def test_canonical_example_1():
    # spec: countMentions(2, [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "71", "HERE"]]) 
    # mentions at timestamp 10: id1 and id0 -> [1, 1]
    # id0 goes offline at timestamp 11
    # mentions at timestamp 71: id0 is offline, id1 is online -> [2, 2]
    assert countMentions(2, [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "71", "HERE"]]) == [2, 2]

def test_canonical_example_2():
    # spec: countMentions(2, [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "12", "ALL"]]) 
    # mentions at timestamp 10: id1 and id0 -> [1, 1]
    # id0 goes offline at timestamp 11
    # mentions at timestamp 12: ALL mentions both id0 and id1 -> [2, 2]
    assert countMentions(2, [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "12", "ALL"]]) == [2, 2]

def test_canonical_example_3():
    # spec: countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "12", "HERE"]]) 
    # id0 goes offline at timestamp 10
    # mentions at timestamp 12: HERE mentions only online users -> [0 (id0 offline), 1 (id1 online)]
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "12", "HERE"]]) == [0, 1]

def test_duplicate_mentions():
    # Testing duplicate mentions in a single MESSAGE event
    # spec: countMentions(2, [["MESSAGE", "10", "id0 id0"], ["MESSAGE", "12", "id1"]]) 
    # mentions at timestamp 10: id0 is mentioned twice -> [2, 0]
    # mentions at timestamp 12: id1 is mentioned once -> [2, 1]
    assert countMentions(2, [["MESSAGE", "10", "id0 id0"], ["MESSAGE", "12", "id1"]]) == [2, 1]

def test_unsorted_events():
    # Testing with events not sorted by timestamp
    # spec: countMentions(2, [["MESSAGE", "12", "id0"], ["OFFLINE", "10", "0"], ["MESSAGE", "14", "id1"]])
    # mentions at timestamp 10: id0 goes offline
    # mentions at timestamp 12: id0 is mentioned but is offline -> [0, 0]
    # mentions at timestamp 14: id1 is mentioned while id0 is still offline -> [0, 1]
    assert countMentions(2, [["MESSAGE", "12", "id0"], ["OFFLINE", "10", "0"], ["MESSAGE", "14", "id1"]]) == [0, 1]

def test_offline_user_mention():
    # Testing mentioning an offline user
    # spec: countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "12", "id0 id1"]]) 
    # id0 goes offline at timestamp 10
    # mentions at timestamp 12: id0 is offline but mentioned, id1 is online -> [1, 1]
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "12", "id0 id1"]]) == [1, 1]
