from typing import List
from solution import maximumCoins

def test_canonical_example_1():
    # coins = [[8,10,1],[1,3,2],[5,6,4]], k = 4
    # Selecting bags at positions [3, 4, 5, 6] gives: 2 + 0 + 4 + 4 = 10
    assert maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) == 10

def test_canonical_example_2():
    # coins = [[1,10,3]], k = 2
    # Selecting bags at positions [1, 2] gives: 3 + 3 = 6
    assert maximumCoins([[1, 10, 3]], 2) == 6

def test_single_point_segment():
    # coins = [[5,5,100]], k = 3
    # Selecting bags at position [5, 5, 5] gives: 100 + 100 + 100 = 100
    assert maximumCoins([[5, 5, 100]], 3) == 100

def test_window_covers_one_of_two_distant_segments():
    # coins = [[1,2,10], [100,100,5]], k = 10
    # The best window is [1, 2, ..., 10] covering: 10 + 10 = 20
    assert maximumCoins([[1, 2, 10], [100, 100, 5]], 10) == 20

def test_large_coordinate_values():
    # coins = [[10**9, 10**9, 50]], k = 2
    # Selecting bags at position [10**9, 10**9] gives: 50 + 50 = 50
    assert maximumCoins([[10**9, 10**9, 50]], 2) == 50

def test_k_spanning_all_segments():
    # coins = [[10,20,5], [100,101,8]], k = 91
    # The optimal window is [10, ..., 100] collecting: 11*5 + 2*8 = 60
    assert maximumCoins([[10, 20, 5], [100, 101, 8]], 91) == 60
