from solution import subarraySum

def test_canonical_example_1():
    # Input: [2, 3, 1]
    # Subarray sums:
    # i=0: nums[0] = [2] -> 2
    # i=1: nums[0...1] = [2, 3] -> 5
    # i=2: nums[1...2] = [3, 1] -> 4
    # Total: 2 + 5 + 4 = 11
    assert subarraySum([2, 3, 1]) == 11

def test_canonical_example_2():
    # Input: [3, 1, 1, 2]
    # Subarray sums:
    # i=0: nums[0] = [3] -> 3
    # i=1: nums[0...1] = [3, 1] -> 4
    # i=2: nums[1...2] = [1, 1] -> 2
    # i=3: nums[1...3] = [1, 1, 2] -> 4
    # Total: 3 + 4 + 2 + 4 = 13
    assert subarraySum([3, 1, 1, 2]) == 13
