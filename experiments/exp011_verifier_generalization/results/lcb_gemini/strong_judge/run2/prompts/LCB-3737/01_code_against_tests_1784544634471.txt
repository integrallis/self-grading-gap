from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # Define a 2D DP array to hold the minimum costs
    dp = [[0] * 3 for _ in range(n)]
    # Initialize the first house with the given costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the dp array for the first half of the houses
    for i in range(1, n // 2):
        for j in range(3):
            dp[i][j] = min(dp[i-1][(j+1)%3], dp[i-1][(j+2)%3]) + cost[i][j]

    # For the second half, we need to consider the equidistant restrictions
    for i in range(n // 2, n):
        for j in range(3):
            dp[i][j] = min(dp[i-1][(j+1)%3], dp[i-1][(j+2)%3]) + cost[i][j]
            # Ensure equidistant houses are not painted the same
            if i - n // 2 >= 0:
                dp[i][j] = min(dp[i][j], dp[i - n // 2][(j + 1) % 3] + cost[i][j])
                dp[i][j] = min(dp[i][j], dp[i - n // 2][(j + 2) % 3] + cost[i][j])

    # The answer will be the minimum cost of painting the last house
    return min(dp[n-1][0], dp[n-1][1], dp[n-1][2])
