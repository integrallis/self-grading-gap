```python
from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    # Calculate contribution of each number as min and max
    for i in range(n):
        # Calculate how many subsequences can include nums[i] as min
        min_contrib = (1 << min(k, n - 1)) - 1
        # Calculate how many subsequences can include nums[i] as max
        max_contrib = (1 << min(k, n - 1)) - 1
        if k > 1:
            min_contrib = (min_contrib * (k - 1)) % MOD
            max_contrib = (max_contrib * (k - 1)) % MOD
        total_sum += (nums[i] * (min_contrib + max_contrib)) % MOD
        total_sum %= MOD
    return total_sum
```
