from typing import List

def zigzagTraversal(grid: List[List[int]]) -> List[int]:
    result = []
    for i in range(len(grid)):
        if i % 2 == 0:
            # Traverse right on even rows
            for j in range(0, len(grid[i]), 2):
                result.append(grid[i][j])
        else:
            # Traverse left on odd rows
            for j in range(1, len(grid[i]), 2):
                result.append(grid[i][len(grid[i]) - j])
    return result
```
