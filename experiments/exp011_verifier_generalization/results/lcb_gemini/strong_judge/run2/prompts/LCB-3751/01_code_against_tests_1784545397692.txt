```python
from typing import List

def maxFrequency(nums: List[int], k: int) -> int:
    nums.sort()  # Sort the array to use the sliding window approach
    l = 0  # Left pointer for the window
    total = 0  # Total sum of the current window
    max_freq = 0  # Maximum frequency of k

    for r in range(len(nums)):
        total += nums[r]  # Add the current number to the total
        while nums[r] * (r - l + 1) - total > k:
            total -= nums[l]  # Remove the leftmost element from the total
            l += 1  # Move the left pointer to the right
        max_freq = max(max_freq, r - l + 1)  # Update max frequency

    return max_freq + (k - total) // (r - l + 1) if total < k else max_freq

```
