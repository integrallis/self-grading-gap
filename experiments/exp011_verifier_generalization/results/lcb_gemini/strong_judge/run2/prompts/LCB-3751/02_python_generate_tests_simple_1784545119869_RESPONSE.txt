from typing import List
from solution import maxFrequency

def test_canonical_example_1():
    # spec: maxFrequency([1, 2, 3, 4, 5, 6], 1) -> 2
    # Explanation: After adding -5 to nums[2..5], 1 has a frequency of 2 in [1, 2, -2, -1, 0, 1].
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2

def test_canonical_example_2():
    # spec: maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) -> 4
    # Explanation: After adding 8 to nums[1..9], 10 has a frequency of 4 in [10, 10, 11, 12, 13, 13, 12, 11, 10, 10].
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4

def test_no_operation_increases_frequency():
    # spec: maxFrequency([5, 5, 5], 5) -> 3
    # Explanation: There are already 3 occurrences of 5; no operation can increase this.
    assert maxFrequency([5, 5, 5], 5) == 3

def test_target_value_not_present():
    # spec: maxFrequency([1, 2, 3], 4) -> 1
    # Explanation: The only way to achieve 4 is to convert one of the existing values.
    assert maxFrequency([1, 2, 3], 4) == 1

def test_minimum_array_length():
    # spec: maxFrequency([10], 1) -> 1
    # Explanation: Only one element exists; frequency of 1 can be achieved by converting 10.
    assert maxFrequency([10], 1) == 1

def test_optimal_subarray_includes_k():
    # spec: maxFrequency([2, 1, 2], 1) -> 2
    # Explanation: The optimal operation is to convert one of the 2s to 1, resulting in two 1s.
    assert maxFrequency([2, 1, 2], 1) == 2

def test_optimal_subarray_does_not_include_all_instances():
    # spec: maxFrequency([2, 1, 1, 2], 1) -> 3
    # Explanation: Convert the two 2s to 1, resulting in three 1s.
    assert maxFrequency([2, 1, 1, 2], 1) == 3
