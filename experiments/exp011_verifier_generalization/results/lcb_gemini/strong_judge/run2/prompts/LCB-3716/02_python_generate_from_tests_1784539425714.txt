from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = [1] * n
    max_length = 1
    for i in range(n):
        for j in range(i):
            diff1 = abs(nums[i] - nums[j])
            if dp[j] > 1:
                # Check the last difference
                last_diff = abs(nums[j] - nums[j - 1])
                if diff1 <= last_diff:
                    dp[i] = max(dp[i], dp[j] + 1)
            else:
                dp[i] = max(dp[i], 2)
        max_length = max(max_length, dp[i])
    return max_length
```
