from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    edges.sort(key=lambda x: x[2])  # Sort edges by weight

    def can_construct(max_weight: int) -> bool:
        from collections import defaultdict, deque
        graph = defaultdict(list)
        out_degree = [0] * n

        # Build the graph with edges under max_weight constraint
        for u, v, w in edges:
            if w <= max_weight and out_degree[u] < threshold:
                graph[u].append(v)
                out_degree[u] += 1

        # BFS to check if all nodes can reach node 0
        reachable = set()
        for start in range(n):
            if start != 0 and out_degree[start] > 0:
                continue
            queue = deque([start])
            while queue:
                node = queue.popleft()
                if node not in reachable:
                    reachable.add(node)
                    for neighbor in graph[node]:
                        queue.append(neighbor)

        return len(reachable) == n

    left, right = 1, 10**6  # Minimum and maximum possible edge weights
    result = -1

    while left <= right:
        mid = (left + right) // 2
        if can_construct(mid):
            result = mid
            right = mid - 1  # Try for a smaller max weight
        else:
            left = mid + 1  # Increase max weight

    return result

```
