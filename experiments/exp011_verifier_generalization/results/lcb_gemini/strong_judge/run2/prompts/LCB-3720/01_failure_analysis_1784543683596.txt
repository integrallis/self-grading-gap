                graph[u].append(v)
                # Count outgoing edges considering threshold
                if out_degree[u] < threshold:
                    out_degree[u] += 1
                else:
                    graph[u].pop()  # Remove edge if threshold exceeded

        # Check reachability from node 0 using reversed graph
        reachable = set()
        queue = deque([0])  # Start BFS from node 0
        while queue:
            node = queue.popleft()
            if node not in reachable:
                reachable.add(node)
                for neighbor in graph[node]:  # Explore neighbors
                    if neighbor not in reachable:
                        queue.append(neighbor)

        # Ensure all nodes can reach node 0
        if len(reachable) != n:
            return False

        # Now check that we can select edges to satisfy the out_degree constraint
        selected_out_degree = [0] * n

            if out_degree[u] > threshold:
                return False
