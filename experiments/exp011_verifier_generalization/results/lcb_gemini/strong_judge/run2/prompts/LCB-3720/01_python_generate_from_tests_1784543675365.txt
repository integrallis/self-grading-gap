from typing import List
from collections import defaultdict, deque


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    edges.sort(key=lambda x: x[2])  # Sort edges by weight

    def can_construct(max_weight: int) -> bool:
        graph = defaultdict(list)
        out_degree = [0] * n

        # Build the graph with edges under max_weight constraint
        for u, v, w in edges:
            if w <= max_weight and out_degree[u] < threshold:
                graph[u].append(v)
                out_degree[u] += 1

        # Check reachability from node 0 using reversed graph
        rev_graph = defaultdict(list)
        for u in graph:
            for v in graph[u]:
                rev_graph[v].append(u)  # Building reverse graph

        # BFS to find all nodes that can reach node 0
        reachable = set()
        queue = deque([0])  # Start BFS from node 0
        while queue:
            node = queue.popleft()
            if node not in reachable:
                reachable.add(node)
                for neighbor in rev_graph[node]:
                    if neighbor not in reachable:
                        queue.append(neighbor)

        return len(reachable) == n  # Check if all nodes can reach 0

    left, right = 1, 10**6  # Minimum and maximum possible edge weights
    result = -1

    while left <= right:
        mid = (left + right) // 2
        if can_construct(mid):
            result = mid
            right = mid - 1  # Try for a smaller max weight
        else:
            left = mid + 1  # Increase max weight

    return result
```
