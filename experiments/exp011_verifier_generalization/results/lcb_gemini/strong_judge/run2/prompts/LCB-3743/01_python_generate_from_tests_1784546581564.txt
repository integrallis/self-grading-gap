from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    if n == 0: return eventTime
    meetings = sorted(zip(startTime, endTime))
    free_chunks = [meetings[0][0] - 0]  # Free time before the first meeting
    for i in range(1, n):
        free_chunks.append(meetings[i][0] - meetings[i-1][1])  # Gaps between meetings
    free_chunks.append(eventTime - meetings[-1][1])  # Free time after the last meeting
    if k >= len(free_chunks) - 1:
        return sum(free_chunks)
    # Sliding window to find the maximum sum of k + 1 chunks
    window_sum = sum(free_chunks[:k + 1])
    max_free_time = window_sum
    for i in range(k + 1, len(free_chunks)):
        window_sum += free_chunks[i] - free_chunks[i - (k + 1)]
        max_free_time = max(max_free_time, window_sum)
    return max_free_time

```
