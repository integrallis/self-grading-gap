from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    total_meeting_duration = sum(end - start for start, end in zip(startTime, endTime))
    if k >= n - 1:
        return eventTime - total_meeting_duration
    gaps = []
    for i in range(1, n):
        gaps.append(startTime[i] - endTime[i - 1])
    gaps.sort()
    max_free_time = 0
    for i in range(k):
        max_free_time += gaps[i]
    max_free_time += (startTime[0] - 0) + (eventTime - endTime[-1])
    return max(max_free_time, 0)
```
