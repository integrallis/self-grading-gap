from typing import List
from solution import minCosts

def test_canonical_example_1():
    # spec: input [5, 3, 4, 1, 3, 2] 
    # - to position 0: cost[0] = 5
    # - to position 1: cost[1] = 3
    # - to position 2: swap with person 1 (cost 3), then person 2 for free = 3
    # - to position 3: cost[3] = 1
    # - to position 4: swap with person 3 (cost 1), then person 4 for free = 1
    # - to position 5: swap with person 3 (cost 1), then person 5 for free = 1
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]

def test_canonical_example_2():
    # spec: input [1, 2, 4, 6, 7]
    # - to position 0: cost[0] = 1
    # - to position 1: cost[0] = 1 (can pay 1 to swap with person 0)
    # - to position 2: cost[0] = 1 (can pay 1 to swap with person 0)
    # - to position 3: cost[0] = 1 (can pay 1 to swap with person 0)
    # - to position 4: cost[0] = 1 (can pay 1 to swap with person 0)
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]
