from solution import maxDistance

def test_canonical_example_1():
    # spec: maxDistance("NWSE", 1) -> 3
    # Explanation: Change s[2] from 'S' to 'N'. The string s becomes "NWNE".
    # Maximum Manhattan distance achieved is 3.
    assert maxDistance("NWSE", 1) == 3

def test_canonical_example_2():
    # spec: maxDistance("NSWWEW", 3) -> 6
    # Explanation: Change s[1] from 'S' to 'N', and s[4] from 'E' to 'W'.
    # The string s becomes "NNWWWW". Maximum Manhattan distance achieved is 6.
    assert maxDistance("NSWWEW", 3) == 6

def test_boundary_case_k_zero():
    # spec: maxDistance("NESW", 0) -> 2
    # Explanation: No changes allowed. The maximum Manhattan distance achieved is 2.
    assert maxDistance("NESW", 0) == 2

def test_boundary_case_k_equals_length():
    # spec: maxDistance("SW", 2) -> 2
    # Explanation: All characters can be changed. The maximum Manhattan distance achieved is 2.
    assert maxDistance("SW", 2) == 2

def test_boundary_case_length_one():
    # spec: maxDistance("N", 0) -> 1
    # Explanation: Only one move available, which is 'N'. Maximum Manhattan distance achieved is 1.
    assert maxDistance("N", 0) == 1

def test_upper_bound_length():
    # spec: maxDistance("N" * 10**5, 0) -> 100000
    # Explanation: No changes allowed. The path goes straight north, reaching (0, 100000).
    # Maximum Manhattan distance achieved is 100000.
    assert maxDistance("N" * 10**5, 0) == 100000

def test_intermediate_max_distance():
    # spec: maxDistance("NESW", 1) -> 3
    # Explanation: Change one direction, e.g., change 'S' to 'E'.
    # The path can be (0,0) -> (0,1) -> (1,1) -> (1,0) -> (2,0), 
    # achieving a maximum Manhattan distance of 3 at (2,0).
    assert maxDistance("NESW", 1) == 3
