```python
def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    for move in s:
        if move == 'N':
            y += 1
        elif move == 'S':
            y -= 1
        elif move == 'E':
            x += 1
        elif move == 'W':
            x -= 1
        # Calculate the current Manhattan distance
        current_distance = abs(x) + abs(y)
        # Calculate the number of changes needed to maximize distance
        min_changes = min(abs(x) + abs(y), len(s) - (abs(x) + abs(y)))
        if min_changes <= k:
            max_distance = max(max_distance, current_distance + 2 * (k - min_changes))
        else:
            max_distance = max(max_distance, current_distance)
    return max_distance
```
