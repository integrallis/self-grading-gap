from solution import maxDifference

def test_canonical_example_1():
    # Input: "aaaaabbc"
    # Frequencies: a=5 (odd), b=2 (even) -> max difference = 5 - 2 = 3
    assert maxDifference("aaaaabbc") == 3

def test_canonical_example_2():
    # Input: "abcabcab"
    # Frequencies: a=3 (odd), b=3 (odd), c=2 (even) -> max difference = 3 - 2 = 1
    assert maxDifference("abcabcab") == 1

def test_multiple_distinct_frequencies():
    # Input: "aaabbbbcccccdddddd"
    # Frequencies: a=3 (odd), b=4 (even), c=5 (odd), d=6 (even)
    # max difference = max(odd) - min(even) = 5 - 4 = 1
    assert maxDifference("aaabbbbcccccdddddd") == 1

def test_min_length_boundary_negative_result():
    # Input: "aab"
    # Frequencies: a=2 (even), b=1 (odd) -> max difference = 1 - 2 = -1
    assert maxDifference("aab") == -1
