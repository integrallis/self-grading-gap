from typing import List
from solution import minCosts

def test_canonical_example_1():
    # spec: cost = [5,3,4,1,3,2] 
    # answer = [5,3,3,1,1,1] 
    # Explanation:
    # i = 0. Swap with person 0 for a cost of 5.
    # i = 1. Swap with person 1 for a cost of 3.
    # i = 2. Swap with person 1 for a cost of 3, then swap with person 2 for free.
    # i = 3. Swap with person 3 for a cost of 1.
    # i = 4. Swap with person 3 for a cost of 1, then swap with person 4 for free.
    # i = 5. Swap with person 3 for a cost of 1, then swap with person 5 for free.
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]

def test_canonical_example_2():
    # spec: cost = [1,2,4,6,7]
    # answer = [1,1,1,1,1]
    # Explanation:
    # Swap with person 0 for a cost of 1, then we will be able to reach any position i for free.
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]
