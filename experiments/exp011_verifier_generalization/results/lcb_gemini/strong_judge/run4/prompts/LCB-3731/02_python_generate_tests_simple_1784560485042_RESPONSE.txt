def test_canonical_example_1():
    # spec: nums = [2, 3, 1]
    # Subarray sums are:
    # i=0: nums[0] = [2] -> 2
    # i=1: nums[0...1] = [2, 3] -> 5
    # i=2: nums[1...2] = [3, 1] -> 4
    # Total sum = 2 + 5 + 4 = 11
    assert subarraySum([2, 3, 1]) == 11
