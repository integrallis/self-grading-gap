from solution import minMaxWeight

def test_canonical_example_1():
    # spec: minMaxWeight(5, [[1,0,1],[2,0,2],[3,0,1],[4,3,1],[2,1,1]], 2) -> 1
    assert minMaxWeight(5, [[1,0,1],[2,0,2],[3,0,1],[4,3,1],[2,1,1]], 2) == 1

def test_canonical_example_2():
    # spec: minMaxWeight(5, [[0,1,1],[0,2,2],[0,3,1],[0,4,1],[1,2,1],[1,4,1]], 1) -> -1
    assert minMaxWeight(5, [[0,1,1],[0,2,2],[0,3,1],[0,4,1],[1,2,1],[1,4,1]], 1) == -1

def test_canonical_example_3():
    # spec: minMaxWeight(5, [[1,2,1],[1,3,3],[1,4,5],[2,3,2],[3,4,2],[4,0,1]], 1) -> 2
    assert minMaxWeight(5, [[1,2,1],[1,3,3],[1,4,5],[2,3,2],[3,4,2],[4,0,1]], 1) == 2

def test_canonical_example_4():

    assert minMaxWeight(5, [[1,2,1],[1,3,3],[1,4,5],[2,3,2],[4,0,1]], 1) == -1

# Additional test cases
def test_multiple_edges():
    # n=4, edges=[[1,0,10], [2,1,5], [2,1,8], [3,2,2]], threshold=1
    # Only edge 2->1 can be kept, max weight = 10. 
    assert minMaxWeight(4, [[1,0,10], [2,1,5], [2,1,8], [3,2,2]], 1) == 10

def test_high_threshold():
    # n=4, edges=[[1,0,20], [2,0,10], [3,0,5]], threshold=3
    # All edges can be kept, max weight = 20.
    assert minMaxWeight(4, [[1,0,20], [2,0,10], [3,0,5]], 3) == 20

def test_minimum_nodes():
    # n=2, edges=[[1,0,100]], threshold=1
    # Only one edge exists, max weight = 100.
    assert minMaxWeight(2, [[1,0,100]], 1) == 100

def test_graph_with_cycle():
    # n=4, edges=[[1,2,5], [2,1,5], [1,0,10]], threshold=2
    # Edge 1->0 is required for reachability, max weight = 10.
    assert minMaxWeight(4, [[1,2,5], [2,1,5], [1,0,10]], 2) == 10

def test_pruning_choice():
    # n=4, edges=[[1,2,5], [1,3,10], [2,0,1], [3,0,1]], threshold=1
    # Optimal path is 1->2->0, max weight = 5, must prune 1->3.
    assert minMaxWeight(4, [[1,2,5], [1,3,10], [2,0,1], [3,0,1]], 1) == 5
