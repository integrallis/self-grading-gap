from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    from collections import defaultdict, deque

    # Build the graph with weights
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))

    # Binary search for the minimum possible maximum edge weight
    def canAchieve(maxWeight):
        # Create a modified graph with edges <= maxWeight
        new_graph = defaultdict(list)
        for u in range(n):
            count = 0
            for v, w in graph[u]:
                if w <= maxWeight and count < threshold:
                    new_graph[u].append(v)
                    count += 1

        # Check if all nodes can reach node 0
        reachable = set()
        queue = deque([0])
        while queue:
            node = queue.popleft()
            if node not in reachable:
                reachable.add(node)
                queue.extend(new_graph[node])

        return len(reachable) == n

    left, right = 1, 10**6
    result = -1
    while left <= right:
        mid = (left + right) // 2
        if canAchieve(mid):
            result = mid
            right = mid - 1
        else:
            left = mid + 1

    return result

```
