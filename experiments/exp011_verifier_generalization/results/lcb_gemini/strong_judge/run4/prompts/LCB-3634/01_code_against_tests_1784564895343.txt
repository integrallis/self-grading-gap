## Implementation

```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirror = lambda c: chr(ord('z') - (ord(c) - ord('a')))

    for i in range(len(s)):
        if marked[i]:
            continue
        for j in range(i - 1, -1, -1):
            if not marked[j] and s[j] == mirror(s[i]):
                marked[i] = True
                marked[j] = True
                score += i - j
                break
    return score

```
