from solution import calculateScore

def test_canonical_example_1():
    # spec: calculateScore("aczzx") -> 5
    # Explanation:
    # i = 0. No j found, skip.
    # i = 1. No j found, skip.
    # i = 2. j = 0 (mirror of 'c' is 'x'), score += 2 - 0 = 2.
    # i = 3. No j found, skip.
    # i = 4. j = 1 (mirror of 'x' is 'c'), score += 4 - 1 = 3.
    # Total score = 2 + 3 = 5
    assert calculateScore("aczzx") == 5

def test_canonical_example_2():
    # spec: calculateScore("abcdef") -> 0
    # Explanation:
    # No j found for any i, so score remains 0.
    assert calculateScore("abcdef") == 0

def test_multiple_mirror_pairs():
    # spec: calculateScore("abazy") -> 4
    # Explanation:
    # i = 0. No j found, skip.
    # i = 1. No j found, skip.
    # i = 2. j = 0 (mirror of 'a' is 'z'), score += 2 - 0 = 2.
    # i = 3. j = 1 (mirror of 'b' is 'y'), score += 3 - 1 = 2.
    # Total score = 2 + 2 = 4
    assert calculateScore("abazy") == 4

def test_minimum_input_length():
    # spec: calculateScore("a") -> 0
    # Explanation:
    # No j found for i = 0, so score remains 0.
    assert calculateScore("a") == 0
