from typing import List


def maxFrequency(nums: List[int], k: int) -> int:
    n = len(nums)
    count = 0
    sum_in_window = 0
    left = 0

    # Count how many k's are already in nums
    current_count = nums.count(k)

    # Sort the nums to apply the sliding window approach
    nums.sort()

    # Iterate over the sorted array with a sliding window
    for right in range(n):
        sum_in_window += nums[right]
        # Check if we can convert the current window to k
        while (right - left + 1) * k > sum_in_window + current_count:
            sum_in_window -= nums[left]
            left += 1
        count = max(count, right - left + 1)

    return count + current_count

```
