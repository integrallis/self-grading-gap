from typing import List


def maxFrequency(nums: List[int], k: int) -> int:
    n = len(nums)
    nums.sort()
    left = 0
    total_add = 0
    max_freq = 0

    for right in range(n):
        total_add += nums[right]
        while (right - left + 1) * k > total_add + (k * (right - left + 1)):
            total_add -= nums[left]
            left += 1
        max_freq = max(max_freq, right - left + 1)

    return max_freq
```
