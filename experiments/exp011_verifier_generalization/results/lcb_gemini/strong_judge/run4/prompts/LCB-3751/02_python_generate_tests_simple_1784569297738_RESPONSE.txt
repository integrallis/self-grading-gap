from typing import List
from solution import maxFrequency

def test_canonical_example_1():
    # spec: maxFrequency([1, 2, 3, 4, 5, 6], 1) -> 2
    # Explanation: After adding -5 to nums[2..5], 1 has a frequency of 2 in [1, 2, -2, -1, 0, 1].
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2

def test_canonical_example_2():
    # spec: maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) -> 4
    # Explanation: After adding 8 to nums[1..9], 10 has a frequency of 4 in [10, 10, 11, 12, 13, 13, 12, 11, 10, 10].
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4

def test_k_not_present():
    # spec: maxFrequency([2, 3, 2], 10) -> 2
    # Explanation: After adding 8 to nums[0..2], 10 has a frequency of 2 in [10, 11, 10].
    assert maxFrequency([2, 3, 2], 10) == 2

def test_all_elements_same_not_k():
    # spec: maxFrequency([7, 7, 7], 1) -> 3
    # Explanation: We can't create any 1s, thus the frequency remains 3.
    assert maxFrequency([7, 7, 7], 1) == 3

def test_no_operation_improves_count():
    # spec: maxFrequency([5, 5, 5], 5) -> 3
    # Explanation: All elements are already 5, so frequency remains 3.
    assert maxFrequency([5, 5, 5], 5) == 3

def test_interspersed_values():
    # spec: maxFrequency([2, 1, 2, 1, 2], 1) -> 3
    # Explanation: After adding -1 to nums[1..3], we convert 2s to 1, giving us 3 instances of 1.
    assert maxFrequency([2, 1, 2, 1, 2], 1) == 3

def test_single_element_array():
    # spec: maxFrequency([10], 20) -> 1
    # Explanation: With only one element, the frequency of 20 can only be 1.
    assert maxFrequency([10], 20) == 1
