from typing import List
from solution import maximumCoins

def test_canonical_example_1():
    # coins = [[8,10,1],[1,3,2],[5,6,4]], k = 4
    # Selecting bags at positions [3, 4, 5, 6] gives 2 + 0 + 4 + 4 = 10
    assert maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) == 10

def test_canonical_example_2():
    # coins = [[1,10,3]], k = 2
    # Selecting bags at positions [1, 2] gives 3 + 3 = 6
    assert maximumCoins([[1, 10, 3]], 2) == 6

def test_k_equals_one():
    # coins = [[1, 5, 10], [6, 10, 5]], k = 1
    # Maximum coins in any single bag is 10 (from bag at position 1)
    assert maximumCoins([[1, 5, 10], [6, 10, 5]], 1) == 10

def test_large_coordinates():
    # coins = [[999999900, 1000000000, 5]], k = 5
    # Only one bag at positions [999999900, 999999901, 999999902, 999999903, 999999904] gives 5 coins
    assert maximumCoins([[999999900, 1000000000, 5]], 5) == 5

def test_large_k_exceeding_span():
    # coins = [[1, 2, 3], [4, 5, 6]], k = 10
    # Total span of all bags is 5, so k exceeds the total span. The maximum is 3 + 6 = 9
    assert maximumCoins([[1, 2, 3], [4, 5, 6]], 10) == 9
