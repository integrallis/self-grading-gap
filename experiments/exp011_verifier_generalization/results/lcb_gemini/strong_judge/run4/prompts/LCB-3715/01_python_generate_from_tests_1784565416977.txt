from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a map to store the total coins in each bag position
    coin_map = {}

    # Fill the map with coins for each range
    for l, r, c in coins:
        for i in range(l, r + 1):
            if i in coin_map:
                coin_map[i] += c
            else:
                coin_map[i] = c

    # Create a sorted list of unique bag positions
    positions = sorted(coin_map.keys())
    max_coins = 0

    # Evaluate the maximum coins in k consecutive bags
    for i in range(len(positions)):
        current_coins = 0
        # Calculate the sum of coins for k consecutive bags
        for j in range(k):
            if i + j < len(positions):
                current_coins += coin_map[positions[i + j]]
        max_coins = max(max_coins, current_coins)

    return max_coins
```
