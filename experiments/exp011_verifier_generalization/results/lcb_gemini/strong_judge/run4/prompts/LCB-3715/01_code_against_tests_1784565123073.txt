```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Initialize an array to hold the coin counts at each bag position
    max_position = 0
    for l, r, c in coins:
        max_position = max(max_position, r)
    coin_count = [0] * (max_position + 1)

    # Fill the coin_count array based on the input segments
    for l, r, c in coins:
        for i in range(l, r + 1):
            coin_count[i] += c

    # Use a sliding window to find the maximum coins in k consecutive bags
    max_coins = 0
    current_coins = sum(coin_count[1:k + 1])  # Initial sum of first k bags
    max_coins = current_coins

    for i in range(2, max_position + 1):
        current_coins += coin_count[i] - (coin_count[i - k] if i - k >= 1 else 0)
        max_coins = max(max_coins, current_coins)

    return max_coins

```
