```python
from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    # dp[i][j][k] will hold the maximum coins collected to reach (i, j) using k neutralizations
    dp = [[[-float('inf')] * 3 for _ in range(n)] for _ in range(m)]
    if coins[0][0] >= 0:
        dp[0][0][0] = coins[0][0]
    else:
        dp[0][0][0] = coins[0][0]
        dp[0][0][1] = 0

    for i in range(m):
        for j in range(n):
            for k in range(3):  # k = number of neutralizations used
                if i == 0 and j == 0:
                    continue
                # Coins from the top cell
                if i > 0:
                    from_top = -float('inf')
                    if coins[i][j] >= 0:
                        from_top = dp[i-1][j][k] + coins[i][j]
                    elif k > 0:
                        from_top = dp[i-1][j][k-1]  # Neutralize the robber
                    dp[i][j][k] = max(dp[i][j][k], from_top)
                # Coins from the left cell
                if j > 0:
                    from_left = -float('inf')
                    if coins[i][j] >= 0:
                        from_left = dp[i][j-1][k] + coins[i][j]
                    elif k > 0:
                        from_left = dp[i][j-1][k-1]  # Neutralize the robber
                    dp[i][j][k] = max(dp[i][j][k], from_left)

    # Get the maximum coins collected at the bottom-right corner with any neutralization count
    return max(dp[m-1][n-1])
```
