from typing import List
from solution import maximumAmount

def test_canonical_example_1():
    # spec: input [[0, 1, -1], [1, -2, 3], [2, -3, 4]] 
    # Explanation of path:
    # Start at (0, 0) with 0 coins (total coins = 0).
    # Move to (0, 1), gaining 1 coin (total coins = 1).
    # Move to (1, 1), where there's a robber stealing 2 coins. The robot uses one neutralization here (total coins = 1).
    # Move to (1, 2), gaining 3 coins (total coins = 4).
    # Move to (2, 2), gaining 4 coins (total coins = 8).
    assert maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) == 8

def test_canonical_example_2():
    # spec: input [[10, 10, 10], [10, 10, 10]]
    # Explanation of path:
    # Start at (0, 0) with 10 coins (total coins = 10).
    # Move to (0, 1), gaining 10 coins (total coins = 20).
    # Move to (0, 2), gaining another 10 coins (total coins = 30).
    # Move to (1, 2), gaining the final 10 coins (total coins = 40).
    assert maximumAmount([[10, 10, 10], [10, 10, 10]]) == 40

def test_single_cell_grid():
    # spec: input [[-10]]
    # The only cell has -10, but the robot starts with 0 coins.
    # Total coins = 0 - 10 = -10.
    assert maximumAmount([[-10]]) == -10

def test_single_row_grid():
    # spec: input [[1, -10, 1, -20, 1]]
    # Path: Start at (0,0) with 1, move to (0,1) using neutralization, move to (0,2) with 1, move to (0,3) using neutralization, move to (0,4) with 1.
    # Total coins = 1 + 0 + 1 + 0 + 1 = 3.
    assert maximumAmount([[1, -10, 1, -20, 1]]) == 3

def test_single_column_grid():
    # spec: input [[1], [-10], [1], [-20], [1]]
    # Path: Start at (0,0) with 1, move to (1,0) using neutralization, move to (2,0) with 1, move to (3,0) using neutralization, move to (4,0) with 1.
    # Total coins = 1 + 0 + 1 + 0 + 1 = 3.
    assert maximumAmount([[1], [-10], [1], [-20], [1]]) == 3

def test_negative_profit():
    # spec: input [[-1, -1], [-1, -1]]
    # The robot starts with 0, encounters robbers at every cell.
    # Total coins = 0 - 1 - 1 - 1 - 1 = -4 (but only 2 can be neutralized).
    assert maximumAmount([[-1, -1], [-1, -1]]) == -1

def test_exactly_two_neutralizations():
    # spec: input [[1, -1000, 10], [1, -1000, 10], [1, 1, 1]]
    # Optimal path: Use neutralizations on the two -1000s.
    # Total coins = 1 + 0 + 10 + 1 + 1 + 1 = 12.
    assert maximumAmount([[1, -1000, 10], [1, -1000, 10], [1, 1, 1]]) == 12

def test_more_than_two_robbers():
    # spec: input [[1, -1, -10, -100], [1000, 1000, 1000, 1000]]
    # Optimal path: Neutralize -10 and -100.
    # Total coins = 1 + 0 + 0 + 0 + 1000 + 1000 + 1000 = 1000.
    assert maximumAmount([[1, -1, -10, -100], [1000, 1000, 1000, 1000]]) == 1000
