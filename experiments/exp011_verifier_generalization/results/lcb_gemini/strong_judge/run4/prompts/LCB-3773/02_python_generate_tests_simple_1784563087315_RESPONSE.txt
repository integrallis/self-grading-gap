from typing import List
from solution import minimumPairRemoval

def test_canonical_example_1():
    # spec: input [5, 2, 3, 1] -> output 2
    # Explanation: 
    # Pair (3,1) has the minimum sum of 4. After replacement, nums = [5,2,4].
    # Pair (2,4) has the minimum sum of 6. After replacement, nums = [5,6].
    # The array became non-decreasing in two operations.
    assert minimumPairRemoval([5, 2, 3, 1]) == 2

def test_canonical_example_2():
    # spec: input [1, 2, 2] -> output 0
    # Explanation: 
    # The array is already sorted (non-decreasing).
    assert minimumPairRemoval([1, 2, 2]) == 0

def test_tie_breaking_rule():
    # spec: input [1, 5, 2, 4] -> output 2
    # Explanation:
    # Pair (1,5) has the minimum sum of 6, then (2,4) has a minimum sum of 6, 
    # but we choose the leftmost pair (1,5) first resulting in [6,2,4].
    # Next, we replace (2,4) resulting in [6,6], which is non-decreasing.
    assert minimumPairRemoval([1, 5, 2, 4]) == 2

def test_with_negative_numbers():
    # spec: input [3, -5, 4, 1] -> output 3
    # Explanation:
    # Pair (3, -5) has the minimum sum of -2, resulting in [-2, 4, 1].
    # Next, pair (4, 1) has a minimum sum of 5, resulting in [-2, 5].
    # Finally, we can replace (5, -2) -> [3].
    # The array becomes non-decreasing in three operations.
    assert minimumPairRemoval([3, -5, 4, 1]) == 3

def test_minimum_length_array():
    # spec: input [42] -> output 0
    # Explanation:
    # An array of length 1 is always non-decreasing.
    assert minimumPairRemoval([42]) == 0
