```python
from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    for i in range(1, k + 1):  # for subsequence lengths from 1 to k
        contribution = 0
        for num in nums:
            contribution += num * (comb(n - 1, i - 1))
            contribution %= MOD
        total_sum += contribution * (2 * i) % MOD  # each contributes as min and max
        total_sum %= MOD
    return total_sum

```
