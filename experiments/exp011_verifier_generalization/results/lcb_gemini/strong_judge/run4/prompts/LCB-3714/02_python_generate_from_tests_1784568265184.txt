from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    # Calculate the number of ways to choose i elements from n elements
    # This will be used to calculate the contribution of each element
    from math import comb
    for i in range(1, k + 1):  # for subsequence lengths from 1 to k
        for num in nums:
            # Each number contributes its value as min and max to the subsequences
            total_sum += num * (comb(n - 1, i - 1) * (1 + (i - 1))) % MOD
            total_sum %= MOD
    return (total_sum * 2) % MOD
```
