from typing import List
from solution import maxSum

def test_canonical_example_1():
    # spec: maxSum([1, 2, 3, 4, 5]) -> 15 (sum of all elements)

def test_canonical_example_2():
    # spec: maxSum([1, 1, 0, 1, 1]) -> 1 (maximum unique element is 1)
    assert maxSum([1, 1, 0, 1, 1]) == 1

def test_canonical_example_3():
    # spec: maxSum([1, 2, -1, -2, 1, 0, -1]) -> 3 (select subarray [2, 1])
    assert maxSum([1, 2, -1, -2, 1, 0, -1]) == 3

    # spec: maxSum([-5, -2, 0, -5]) -> 0 (no positive numbers, max is 0)
    assert maxSum([-5, -2, 0, -5]) == 0

def test_single_element_negative():
    # spec: maxSum([-42]) -> -42 (only one element)
    assert maxSum([-42]) == -42

def test_element_value_boundaries():
    # spec: maxSum([100, -100, 50, 100]) -> 150 (unique values 100 and 50)
    assert maxSum([100, -100, 50, 100]) == 150
