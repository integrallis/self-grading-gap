def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    length = len(s)

    for i in range(length):
        move = s[i]
        if move == 'N':
            y += 1
        elif move == 'S':
            y -= 1
        elif move == 'E':
            x += 1
        elif move == 'W':
            x -= 1

        # Current manhattan distance
        current_distance = abs(x) + abs(y)
        max_distance = max(max_distance, current_distance)

        # Calculate the potential maximum distance if we can change moves
        remaining_moves = k - (i + 1)
        if remaining_moves >= 0:
            distance_with_changes = current_distance + min(remaining_moves, 2 * (length - i - 1))
            max_distance = max(max_distance, distance_with_changes)

    return max_distance
```
