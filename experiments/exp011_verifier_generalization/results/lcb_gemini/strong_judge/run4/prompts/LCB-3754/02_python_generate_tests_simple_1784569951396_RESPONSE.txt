from solution import maxDistance

def test_canonical_example_1():
    # spec: maxDistance("NWSE", 1) should return 3
    # Changing 'S' to 'N' gives the path "NWNE" which has maximum Manhattan distance 3.
    assert maxDistance("NWSE", 1) == 3

def test_canonical_example_2():
    # spec: maxDistance("NSWWEW", 3) should return 6
    # Changing 'S' to 'N' and 'E' to 'W' gives the path "NNWWWW" which has maximum Manhattan distance 6.
    assert maxDistance("NSWWEW", 3) == 6

def test_k_zero_changes():
    # spec: maxDistance("NWSE", 0) should return 2
    # The path is "NWSE" which has maximum Manhattan distance 2.
    assert maxDistance("NWSE", 0) == 2

def test_k_equals_length():
    # spec: maxDistance("NESW", 4) should return 4
    # Changing all characters to one direction gives maximum Manhattan distance 4.
    assert maxDistance("NESW", 4) == 4

def test_minimum_length_one():
    # spec: maxDistance("W", 0) should return 1
    # The path is just one step west, giving maximum Manhattan distance 1.
    assert maxDistance("W", 0) == 1
