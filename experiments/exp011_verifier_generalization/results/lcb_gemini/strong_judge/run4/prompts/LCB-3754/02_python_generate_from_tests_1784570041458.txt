def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0

    for move in s:
        if move == 'N':
            y += 1
        elif move == 'S':
            y -= 1
        elif move == 'E':
            x += 1
        elif move == 'W':
            x -= 1

        # Current manhattan distance
        current_distance = abs(x) + abs(y)
        max_distance = max(max_distance, current_distance)

    # The maximum distance can be increased by changing up to k moves
    return max_distance + k
```
