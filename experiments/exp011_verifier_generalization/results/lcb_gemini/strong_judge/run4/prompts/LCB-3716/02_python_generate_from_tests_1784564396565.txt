from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = [1] * n  # dp[i] stores the length of longest subsequence ending at i
    for i in range(n):
        for j in range(i):
            diff1 = abs(nums[i] - nums[j])
            if j == 0:
                continue
            # Check for all preceding indices to form a non-increasing sequence of absolute differences
            for k in range(j):
                diff2 = abs(nums[j] - nums[k])
                if diff1 <= diff2:
                    dp[i] = max(dp[i], dp[j] + 1)
    return max(dp)
```
