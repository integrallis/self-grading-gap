from solution import maxContainers

def test_canonical_example_1():
    # spec: maxContainers(2, 3, 15) -> 4
    # The deck can hold 4 containers (2x2 = 4) and the total weight is 4 * 3 = 12, which is <= 15
    assert maxContainers(2, 3, 15) == 4

def test_canonical_example_2():
    # spec: maxContainers(3, 5, 20) -> 4
    # The deck can hold 9 containers (3x3 = 9), but the total weight of 4 containers is 4 * 5 = 20, which is <= 20
    assert maxContainers(3, 5, 20) == 4

def test_lower_boundary_conditions():
    # spec: maxContainers(1, 1, 1) -> 1
    # The deck can hold 1 container (1x1 = 1) and the total weight is 1 * 1 = 1, which is <= 1
    assert maxContainers(1, 1, 1) == 1

def test_upper_boundary_conditions():
    # spec: maxContainers(1000, 1000, 10**9) -> 1000000
    # The deck can hold 1000000 containers (1000x1000 = 1000000) and the total weight of 1000 containers is 1000 * 1000 = 1000000, which is <= 10^9
    # maxWeight allows all containers to be loaded since 10^9 / 1000 = 1000000
    assert maxContainers(1000, 1000, 10**9) == 1000000

    # spec: maxContainers(10, 3, 29) -> 9

    assert maxContainers(10, 3, 29) == 9

def test_zero_containers_case():
    # spec: maxContainers(10, 10, 9) -> 0

    assert maxContainers(10, 10, 9) == 0
