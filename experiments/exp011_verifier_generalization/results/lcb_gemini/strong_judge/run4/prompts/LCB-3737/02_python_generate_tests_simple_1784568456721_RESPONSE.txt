from typing import List
from solution import minCost

def test_canonical_example_1():
    # spec: input '4\n[[3, 5, 7], [6, 2, 9], [4, 8, 1], [7, 3, 5]]' should return 9
    assert minCost(4, [[3, 5, 7], [6, 2, 9], [4, 8, 1], [7, 3, 5]]) == 9

def test_canonical_example_2():
    # spec: input '6\n[[2, 4, 6], [5, 3, 8], [7, 1, 9], [4, 6, 2], [3, 5, 7], [8, 2, 4]]' should return 18
    assert minCost(6, [[2, 4, 6], [5, 3, 8], [7, 1, 9], [4, 6, 2], [3, 5, 7], [8, 2, 4]]) == 18

def test_minimum_n():
    # For n=2, cost = [[1, 2, 3], [4, 5, 6]], the minimum cost is min(1+5, 1+6, 2+4, 2+6, 3+4, 3+5) = 6
    assert minCost(2, [[1, 2, 3], [4, 5, 6]]) == 6

def test_minimum_cost_value():
    # For n=2, cost = [[10, 0, 10], [10, 10, 0]], the minimum cost is 0 + 0 = 0
    assert minCost(2, [[10, 0, 10], [10, 10, 0]]) == 0
