from typing import List

def zigzagTraversal(grid: List[List[int]]) -> List[int]:
    result = []
    rows = len(grid)
    cols = len(grid[0])
    for r in range(rows):
        if r % 2 == 0:
            # Traverse left to right
            for c in range(0, cols, 2):
                result.append(grid[r][c])
        else:
            # Traverse right to left
            for c in range(1, cols, 2):
                result.append(grid[r][c])
    return result
