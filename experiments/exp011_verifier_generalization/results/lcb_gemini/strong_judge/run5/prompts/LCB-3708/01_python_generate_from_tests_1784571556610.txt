from typing import List

def zigzagTraversal(grid: List[List[int]]) -> List[int]:
    result = []
    rows = len(grid)
    cols = len(grid[0])
    for r in range(rows):
        if r % 2 == 0:
            # Traverse left to right
            for c in range(0, cols, 2):
                result.append(grid[r][c])
        else:
            # Traverse right to left
            start_col = cols - 1
            if start_col % 2 == 0:
                start_col -= 1
            for c in range(start_col, -1, -2):
                result.append(grid[r][c])
    return result

```
