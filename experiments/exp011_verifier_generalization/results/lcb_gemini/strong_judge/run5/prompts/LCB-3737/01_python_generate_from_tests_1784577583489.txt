from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize a 2D list to store the minimum costs
    dp = [[0] * 3 for _ in range(n)]

    # Base case: the cost of painting the first house
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the dp table
    for i in range(1, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(
                dp[i-1][(j+1) % 3],  # not the same color as the previous house
                dp[i-1][(j+2) % 3]   # not the same color as the previous house
            )

    # Handle the equidistance constraint
    min_cost = float('inf')
    for j in range(3):
        # Calculate the total cost by considering the last house colors
        total_cost = dp[n-1][j] + min(
            dp[n-2][(j+1) % 3],  # ensure the second last house is different
            dp[n-2][(j+2) % 3]   # ensure the second last house is different
        )
        min_cost = min(min_cost, total_cost)

    return min_cost

```
