from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Binary search for the minimum possible maximum weight
    left, right = 1, 10**6
    result = -1

    while left <= right:
        mid = (left + right) // 2
        # Filter edges with weight <= mid
        filtered_edges = [(a, b) for a, b, w in edges if w <= mid]

        # Create adjacency list with outgoing edges limited by threshold
        from collections import defaultdict, deque
        graph = defaultdict(list)
        outgoing_count = defaultdict(int)

        for a, b in filtered_edges:
            if outgoing_count[a] < threshold:
                graph[a].append(b)
                outgoing_count[a] += 1

        # Check if node 0 is reachable from all other nodes
        reachable = [False] * n
        queue = deque()
        for i in range(n):
            if i != 0 and outgoing_count[i] > 0:
                queue.append(i)
                reachable[i] = True

        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if not reachable[neighbor]:
                    reachable[neighbor] = True
                    queue.append(neighbor)

        # Check if all nodes can reach node 0
        if all(reachable[i] == (i == 0) for i in range(n)):
            result = mid
            right = mid - 1  # Try for a smaller maximum weight
        else:
            left = mid + 1  # Increase the maximum weight

    return result

```
