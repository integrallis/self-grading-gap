from solution import findValidPair

def test_canonical_example_1():
    # spec: input "2523533" -> digit '2' appears 2 times and digit '3' appears 3 times.
    # valid pair is "23"

def test_canonical_example_2():
    # spec: input "221" -> digit '2' appears 2 times and digit '1' appears 1 time.
    # valid pair is "21"
    assert findValidPair("221") == "21"

def test_canonical_example_3():
    # spec: input "22" -> there are no valid adjacent pairs.
    assert findValidPair("22") == ""

def test_first_valid_pair_among_multiple_pairs():
    # input "122333" -> digit '1' appears 1 time, '2' appears 2 times, '3' appears 3 times.

    assert findValidPair("122333") == "12"

    # input "122" -> digit '1' appears 1 time, '2' appears 2 times.

    # input "3331" -> digit '3' appears 3 times, '1' appears 1 time.

    assert findValidPair("3331") == "31"

    # input "1999999999" -> digit '1' appears 1 time, '9' appears 9 times.

    # input "1" + "2"*99 -> digit '1' appears 1 time, '2' appears 99 times.
