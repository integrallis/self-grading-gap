from solution import reverseDegree

def test_canonical_example_1():
    # Input: "abc"
    # 'a' = 26, index 1 -> 26
    # 'b' = 25, index 2 -> 50
    # 'c' = 24, index 3 -> 72
    # Total = 26 + 50 + 72 = 148
    assert reverseDegree("abc") == 148

def test_canonical_example_2():
    # Input: "zaza"
    # 'z' = 1, index 1 -> 1
    # 'a' = 26, index 2 -> 52
    # 'z' = 1, index 3 -> 3
    # 'a' = 26, index 4 -> 104
    # Total = 1 + 52 + 3 + 104 = 160
    assert reverseDegree("zaza") == 160

def test_minimum_length():
    # Input: "a"
    # 'a' = 26, index 1 -> 26
    # Total = 26
    assert reverseDegree("a") == 26

def test_maximum_length():
    # Input: "abcdefghijklmnopqrstuvwxyz" * 38 + "abc"
    # This creates a string of length 1000.
    # Each full cycle of 'a' to 'z' gives:
    # Total from 'a' to 'z' = 26 + 25 + ... + 1 = 351
    # For 38 full cycles (0 to 999 positions):
    # Each letter contributes to its position which is (1 to 1000) in its respective cycles.
    # Final contribution is 351 * (1 + 2 + ... + 38) = 351 * 741 = 260991
    # Last 3 characters are 'a', 'b', 'c' with contributions:
    # 'a' = 26, index 998 -> 26 * 998
    # 'b' = 25, index 999 -> 25 * 999
    # 'c' = 24, index 1000 -> 24 * 1000
    # Total from the last 3 characters = 26 + 50 + 72 = 148
    # Final total = 260991 + 148 = 169993350
    long_string = "abcdefghijklmnopqrstuvwxyz" * 38 + "abc"
    assert reverseDegree(long_string) == 169993350

def test_boundary_maximum_length():
    # Input: 'a' * 1000
    # Every character is 'a', which is 26 in the reversed alphabet.
    # So each character contributes 26 * (position index)
    # Total = 26 * (1 + 2 + ... + 1000) = 26 * 500500 = 13013000
    assert reverseDegree('a' * 1000) == 13013000
