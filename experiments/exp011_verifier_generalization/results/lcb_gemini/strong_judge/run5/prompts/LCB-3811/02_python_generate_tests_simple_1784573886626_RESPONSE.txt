from solution import reverseDegree

def test_canonical_example_1():
    # Input: "abc"
    # 'a' = 26, index 1 -> 26
    # 'b' = 25, index 2 -> 50
    # 'c' = 24, index 3 -> 72
    # Total = 26 + 50 + 72 = 148
    assert reverseDegree("abc") == 148

def test_canonical_example_2():
    # Input: "zaza"
    # 'z' = 1, index 1 -> 1
    # 'a' = 26, index 2 -> 52
    # 'z' = 1, index 3 -> 3
    # 'a' = 26, index 4 -> 104
    # Total = 1 + 52 + 3 + 104 = 160
    assert reverseDegree("zaza") == 160

def test_minimum_length():
    # Input: "a"
    # 'a' = 26, index 1 -> 26
    # Total = 26
    assert reverseDegree("a") == 26

def test_maximum_length():
    # Input: "abcdefghijklmnopqrstuvwxyz" * 38 + "abc"
    # This creates a string of length 1000.
    # The last 3 characters will be 'a', 'b', 'c'.
    # 'a' = 26, index 1 (in the last 3) -> 26
    # 'b' = 25, index 2 (in the last 3) -> 50
    # 'c' = 24, index 3 (in the last 3) -> 72
    # For the first 997 characters:
    # Each character contributes their reverse degree based on their index in the full alphabet.
    # The total for 'a' to 'z' is calculated as (26 + 25 + ... + 1) * (1 to 38) => 351 * (1 + 2 + ... + 38)
    # = 351 * 741 = 260991
    # Total from the last 3 characters = 26 + 50 + 72 = 148
    # Final total = 260991 + 148 = 261139
    long_string = "abcdefghijklmnopqrstuvwxyz" * 38 + "abc"
    assert reverseDegree(long_string) == 261139
