def test_canonical_example_1():
    # Input: "abc"
    # 'a' = 26, index 1 -> 26
    # 'b' = 25, index 2 -> 50
    # 'c' = 24, index 3 -> 72

    assert reverseDegree("abc") == 148

def test_canonical_example_2():
    # Input: "zaza"
    # 'z' = 1, index 1 -> 1
    # 'a' = 26, index 2 -> 52
    # 'z' = 1, index 3 -> 3
    # 'a' = 26, index 4 -> 104
    # Total = 1 + 52 + 3 + 104 = 160
