```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a dictionary to hold the coin counts for each bag position
    bags = {}

    # Fill the dictionary with the coins from the segments
    for l, r, c in coins:
        for pos in range(l, r + 1):
            bags[pos] = bags.get(pos, 0) + c

    # Create a sorted list of positions
    positions = sorted(bags.keys())
    max_coins = 0
    current_coins = 0
    left = 0

    # Use a sliding window to find the maximum coins in up to k consecutive bags
    for right in range(len(positions)):
        current_coins += bags[positions[right]]

        # If the window exceeds k, move the left pointer
        while positions[right] - positions[left] + 1 > k:
            current_coins -= bags[positions[left]]
            left += 1

        # Update the maximum coins found 
        max_coins = max(max_coins, current_coins)

    return max_coins

```
