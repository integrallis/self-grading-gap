from solution import maxDifference

def test_canonical_example_1():
    # Input: "aaaaabbc"
    # Frequencies: a: 5 (odd), b: 2 (even) -> max difference = 5 - 2 = 3
    assert maxDifference("aaaaabbc") == 3

def test_canonical_example_2():
    # Input: "abcabcab"
    # Frequencies: a: 3 (odd), c: 2 (even) -> max difference = 3 - 2 = 1
    assert maxDifference("abcabcab") == 1

def test_minimum_length_case():
    # Input: "abb"
    # Frequencies: a: 1 (odd), b: 2 (even) -> max difference = 1 - 2 = -1
    assert maxDifference("abb") == -1

def test_maximum_length_case():
    # Input: "a" * 98 + "bc" (98 'a's, 1 'b', 1 'c')
    # Frequencies: a: 98 (even), b: 1 (odd), c: 1 (odd) 
    # Possible differences: 1 - 98 = -97 (only considering odd 'b' with even 'a')
    assert maxDifference("a" * 98 + "bc") == -97
