```python
from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n == 0:
        return 0
    # dp[i] will hold the maximum length of subsequence ending at index i
    dp = [1] * n
    # max_length will store the maximum length of valid subsequence found
    max_length = 1
    for i in range(1, n):
        for j in range(i):
            diff1 = abs(nums[i] - nums[j])
            if dp[j] > 0:
                # Check if the current element can be added to the subsequence
                valid = True
                for k in range(j):
                    diff2 = abs(nums[j] - nums[k])
                    if diff2 < diff1:
                        valid = False
                        break
                if valid:
                    dp[i] = max(dp[i], dp[j] + 1)
                    max_length = max(max_length, dp[i])
    return max_length

```
