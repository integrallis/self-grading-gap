```python
def calculateScore(s: str) -> int:
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # Create a mirror dictionary for 'a' to 'z'
    marked = set()  # To track marked indices
    score = 0
    for i in range(len(s)):
        if i in marked:
            continue  # Skip if already marked
        for j in range(i - 1, -1, -1):  # Search for the closest unmarked index j
            if j in marked:
                continue  # Skip marked indices
            if mirror[s[i]] == s[j]:  # Check if s[j] is the mirror of s[i]
                marked.add(i)
                marked.add(j)
                score += i - j  # Update score
                break  # Stop searching after finding the closest match
    return score
```
