from solution import calculateScore

def test_canonical_example_1():
    # Given input "aczzx":
    # i = 0: No j found -> score = 0
    # i = 1: No j found -> score = 0
    # i = 2: j = 0 (mirror of 'c' is 'x'), score += 2 - 0 = 2 -> score = 2
    # i = 3: No j found -> score = 2
    # i = 4: j = 1 (mirror of 'f' is 'u'), score += 4 - 1 = 3 -> score = 5
    # Final score = 5
    assert calculateScore("aczzx") == 5
