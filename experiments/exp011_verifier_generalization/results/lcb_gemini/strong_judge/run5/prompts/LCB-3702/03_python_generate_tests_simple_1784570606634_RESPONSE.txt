from typing import List
from solution import maxLength

def test_canonical_example_1():
    # Input: [1, 2, 1, 2, 1, 1, 1]
    # Longest product equivalent subarray is [1, 2, 1, 1, 1]
    # prod([1, 2, 1, 1, 1]) = 2
    # gcd([1, 2, 1, 1, 1]) = 1
    # lcm([1, 2, 1, 1, 1]) = 2
    # Thus, prod(arr) == lcm(arr) * gcd(arr) holds true
    assert maxLength([1, 2, 1, 2, 1, 1, 1]) == 5

def test_canonical_example_2():
    # Input: [2, 3, 4, 5, 6]
    # Longest product equivalent subarray is [3, 4, 5]
    # prod([3, 4, 5]) = 60
    # gcd([3, 4, 5]) = 1
    # lcm([3, 4, 5]) = 60
    # Thus, prod(arr) == lcm(arr) * gcd(arr) holds true
    assert maxLength([2, 3, 4, 5, 6]) == 3

def test_canonical_example_3():
    # Input: [1, 2, 3, 1, 4, 5, 1]
    # Longest product equivalent subarray is [3, 1, 4, 5, 1]
    # prod([3, 1, 4, 5, 1]) = 60
    # gcd([3, 1, 4, 5, 1]) = 1
    # lcm([3, 1, 4, 5, 1]) = 60
    # Thus, prod(arr) == lcm(arr) * gcd(arr) holds true
    assert maxLength([1, 2, 3, 1, 4, 5, 1]) == 5

def test_lower_bound_length():
    # Input: [3, 7]
    # Any two-element array is product equivalent
    assert maxLength([3, 7]) == 2

def test_minimum_product_equivalent_subarray():
    # Input: [8, 9, 10]
    # Longest product equivalent subarray is [8, 9] or [9, 10]
    # prod([8, 9]) = 72, gcd([8, 9]) = 1, lcm([8, 9]) = 72
    # Thus, prod(arr) == lcm(arr) * gcd(arr) holds true
    assert maxLength([8, 9, 10]) == 2

def test_upper_bound_length():
    # Input: [1] * 100
    # The whole array is product equivalent since all elements are 1
    assert maxLength([1] * 100) == 100

def test_identical_non_one_elements():
    # Input: [3, 3, 3, 3]
