from solution import hasSpecialSubstring

def test_canonical_example_1():
    # spec: hasSpecialSubstring("aaabaaa", 3) -> true
    assert hasSpecialSubstring("aaabaaa", 3) is True

def test_canonical_example_2():
    # spec: hasSpecialSubstring("abc", 2) -> false
    assert hasSpecialSubstring("abc", 2) is False

def test_valid_substring_at_beginning():
    # valid substring "bbb" at the beginning with different adjacent character
    # spec: hasSpecialSubstring("bbba", 3) -> true
    assert hasSpecialSubstring("bbba", 3) is True

def test_invalid_substring_due_to_adjacent_character():
    # sequence of 3 identical characters invalidated by adjacent identical character
    # spec: hasSpecialSubstring("aaaa", 3) -> false
    assert hasSpecialSubstring("aaaa", 3) is False

def test_valid_substring_at_end():
    # valid substring "ccc" at the end with different adjacent character
    # spec: hasSpecialSubstring("aabbbccc", 3) -> true
    assert hasSpecialSubstring("aabbbccc", 3) is True

def test_entire_string_is_valid_substring():
    # the whole string is a valid substring of length 2
    # spec: hasSpecialSubstring("zz", 2) -> true
    assert hasSpecialSubstring("zz", 2) is True

def test_edge_case_k_equals_1():
    # single character can be valid if surrounded by different characters
    # spec: hasSpecialSubstring("aba", 1) -> true
    assert hasSpecialSubstring("aba", 1) is True

def test_edge_case_k_equals_1_invalid():
    # single character cannot be valid if it has the same character adjacent
    # spec: hasSpecialSubstring("aaa", 1) -> false
    assert hasSpecialSubstring("aaa", 1) is False
