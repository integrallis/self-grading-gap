from solution import maxDistance

def test_canonical_example_1():
    # spec: maxDistance("NWSE", 1) -> 3
    assert maxDistance("NWSE", 1) == 3

def test_canonical_example_2():
    # spec: maxDistance("NSWWEW", 3) -> 6
    assert maxDistance("NSWWEW", 3) == 6

def test_k_equals_zero():
    # spec: maxDistance("NESW", 0) -> 2
    # No changes allowed, maximum distance is achieved by following the path (0,0) -> (1,0) -> (1,1) -> (0,1) -> (0,0)
    # The furthest from origin at (1,1), distance = 1 + 1 = 2
    assert maxDistance("NESW", 0) == 2

def test_k_equals_length():
    # spec: maxDistance("EWNS", 4) -> 4
    # All characters can be changed to one direction, maximum distance is achieved by changing to "NNNN" or "EEEE", which is 4.
    assert maxDistance("EWNS", 4) == 4

def test_single_character():
    # spec: maxDistance("N", 0) -> 1
    # With no changes, the maximum distance at (0,1) from origin is 1.
    assert maxDistance("N", 0) == 1
