from typing import List


def maxFrequency(nums: List[int], k: int) -> int:
    count_k = nums.count(k)
    max_freq = count_k
    nums.sort()
    left = 0
    total = 0

    for right in range(len(nums)):
        total += nums[right]
        while total + (right - left + 1) * k > sum(nums[left:right + 1]):
            total -= nums[left]
            left += 1
        max_freq = max(max_freq, right - left + 1 + count_k)

    return max_freq
```
