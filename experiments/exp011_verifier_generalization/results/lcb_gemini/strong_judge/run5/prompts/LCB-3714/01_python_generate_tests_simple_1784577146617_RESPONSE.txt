from typing import List
from solution import minMaxSums

def test_canonical_example_1():
    # spec: minMaxSums([1, 2, 3], 2) -> 24
    # Subsequence sums: [1] -> 2, [2] -> 4, [3] -> 6, [1, 2] -> 3, [1, 3] -> 4, [2, 3] -> 5
    # Total = 2 + 4 + 6 + 3 + 4 + 5 = 24
    assert minMaxSums([1, 2, 3], 2) == 24

def test_canonical_example_2():
    # spec: minMaxSums([5, 0, 6], 1) -> 22
    # Single element subsequences: [5] -> 10, [0] -> 0, [6] -> 12
    # Total = 10 + 0 + 12 = 22
    assert minMaxSums([5, 0, 6], 1) == 22

def test_canonical_example_3():
    # spec: minMaxSums([1, 1, 1], 2) -> 12
    # All subsequences with at most 2 elements yield min and max as 1
    # Each appears 3 times: [1] -> 2, [1, 1] -> 2
    # Total = 3 * 2 + 3 * 2 = 12
    assert minMaxSums([1, 1, 1], 2) == 12
