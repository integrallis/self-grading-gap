## Implementation

```python
from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    nums.sort()

    # Calculate contributions of each number as min and max
    for i in range(n):
        # Calculate the number of ways this element can be the minimum
        min_count = min(k, i + 1) * (1 << (n - i - 1))
        # Calculate the number of ways this element can be the maximum
        max_count = min(k, n - i) * (1 << i)

        # Update the total sum
        total_sum += (nums[i] * (min_count + max_count)) % MOD
        total_sum %= MOD

    return total_sum
```
