from solution import maxDifference

def test_canonical_example_1():
    # spec: maxDifference("aaaaabbc") -> 3
    # 'a' has frequency 5 (odd), 'b' has frequency 2 (even)
    assert maxDifference("aaaaabbc") == 3

def test_canonical_example_2():
    # spec: maxDifference("abcabcab") -> 1
    # 'a' has frequency 3 (odd), 'c' has frequency 2 (even)
    assert maxDifference("abcabcab") == 1

def test_multiple_even_frequencies():
    # maxDifference("aaabbbbcc") -> 1
    # 'b' has frequency 4 (even), 'c' has frequency 2 (even), 'a' has frequency 3 (odd)
    # Maximum difference is 3 - 2 = 1
    assert maxDifference("aaabbbbcc") == 1

def test_negative_difference():
    # maxDifference("abbbcccc") -> -1
    # 'b' has frequency 3 (odd), 'c' has frequency 4 (even)
    # Maximum difference is 3 - 4 = -1
    assert maxDifference("abbbcccc") == -1

def test_minimum_length():
    # maxDifference("aab") -> -1
    # 'a' has frequency 2 (even), 'b' has frequency 1 (odd)
    # Maximum difference is 1 - 2 = -1
    assert maxDifference("aab") == -1
