from typing import List
from solution import subarraySum

def test_canonical_example_1():
    # spec: nums = [2, 3, 1]
    # subarray sums:
    # i=0: nums[0] = [2] -> sum = 2
    # i=1: nums[0 ... 1] = [2, 3] -> sum = 5
    # i=2: nums[1 ... 2] = [3, 1] -> sum = 4
    # Total sum = 2 + 5 + 4 = 11
    assert subarraySum([2, 3, 1]) == 11

def test_canonical_example_2():
    # spec: nums = [3, 1, 1, 2]
    # subarray sums:
    # i=0: nums[0] = [3] -> sum = 3
    # i=1: nums[0 ... 1] = [3, 1] -> sum = 4
    # i=2: nums[1 ... 2] = [1, 1] -> sum = 2
    # i=3: nums[1 ... 3] = [1, 1, 2] -> sum = 4
    # Total sum = 3 + 4 + 2 + 4 = 13
    assert subarraySum([3, 1, 1, 2]) == 13

def test_minimum_length():
    # spec: nums = [1]
    # subarray sums:
    # i=0: nums[0] = [1] -> sum = 1
    # Total sum = 1

def test_maximum_length():
    # spec: nums = [1] * 100
    # subarray sums:
    # For each i from 0 to 99, start = max(0, i-1) = 0 for all, hence every sum is 1.
    # Total sum = 100 * 1 = 100
    assert subarraySum([1] * 100) == 100

def test_maximum_element_value():
    # spec: nums = [1, 1000]
    # subarray sums:
    # i=0: nums[0] = [1] -> sum = 1
    # i=1: nums[0 ... 1] = [1, 1000] -> sum = 1001
    # Total sum = 1 + 1001 = 1002
