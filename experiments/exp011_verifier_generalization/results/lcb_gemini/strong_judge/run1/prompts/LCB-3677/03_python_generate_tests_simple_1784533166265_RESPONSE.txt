from typing import List
from solution import maximumAmount

def test_canonical_example_1():
    # Input: [[0, 1, -1], [1, -2, 3], [2, -3, 4]]
    # Optimal path:
    # Start at (0, 0) with 0 coins -> (0, 1) gain 1 -> (1, 1) neutralize -2 -> (1, 2) gain 3 -> (2, 2) gain 4
    # Total coins = 0 + 1 + 0 + 3 + 4 = 8
    assert maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) == 8

def test_canonical_example_2():
    # Input: [[10, 10, 10], [10, 10, 10]]
    # Optimal path:
    # Start at (0, 0) with 10 coins -> (0, 1) gain 10 -> (0, 2) gain 10 -> (1, 2) gain 10
    # Total coins = 10 + 10 + 10 + 10 = 40
    assert maximumAmount([[10, 10, 10], [10, 10, 10]]) == 40

def test_1x1_grid_expects_value_of_the_single_cell():
    # Input: [[10]]
    # Only one cell with value 10
    assert maximumAmount([[10]]) == 10

def test_1x1_grid_with_robber_expects_zero():
    # Input: [[-10]]
    # Only one cell with a robber; must neutralize to get 0 coins
    assert maximumAmount([[-10]]) == 0

def test_single_row_grid_with_choice_of_robbers():
    # Input: [[0, -10, -1, -20]]
    # Optimal path: Start at 0, neutralize -10, then -1, total = -1
    assert maximumAmount([[0, -10, -1, -20]]) == -1

def test_single_column_grid():
    # Input: [[0], [-10], [5], [-20]]
    # Total coins: 0 - 10 + 5 - 20 = -25, but must neutralize one robber (-10) to get 5
    assert maximumAmount([[0], [-10], [5], [-20]]) == 5

def test_two_neutralizations_are_optimal():
    # Input: [[0, 1, 1], [-10, -20, 1]]
    # Optimal path: Start at 0, gain 1, neutralize -10, gain 1, total = 3
    assert maximumAmount([[0, 1, 1], [-10, -20, 1]]) == 3

def test_negative_result():
    # Input: [[-1, -1], [-1, -1]]
    # Total coins: -1 - 1 - 1 - 1 = -4, must neutralize two to get -1
    assert maximumAmount([[-1, -1], [-1, -1]]) == -1

def test_boundary_values():
    # Input: [[0, 1000], [-1000, 0]]
    # Total coins: 0 + 1000 - 1000 + 0 = 0, neutralizations not needed
    assert maximumAmount([[0, 1000], [-1000, 0]]) == 1000
