You are given a 2D array coins, where coins[i] = [l_i, r_i, c_i] denotes that every bag from l_i to r_i contains c_i coins.

Output: 6
Explanation:
Selecting bags at positions [1, 2] gives the maximum number of coins: 3 + 3 = 6.

1 <= coins.length <= 10^5
1 <= k <= 10^9
coins[i] == [l_i, r_i, c_i]
1 <= l_i <= r_i <= 10^9
1 <= c_i <= 1000
The given segments are non-overlapping.

```python
from typing import List
```


```python
def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Use a list to keep track of coin counts at each position
    max_pos = 0
    for l, r, c in coins:
        max_pos = max(max_pos, r)
    coin_count = [0] * (max_pos + 1)

    # Fill the coin_count array with the number of coins for each position
    for l, r, c in coins:
        for i in range(l, r + 1):
            coin_count[i] += c

    # Initialize variables to track the maximum coins collected
    max_coins = 0
    current_coins = 0

    # Sliding window over the coin_count array to find max for k consecutive bags
    for i in range(max_pos + 1):
        current_coins += coin_count[i]
        if i >= k:
            current_coins -= coin_count[i - k]
        if i >= k - 1:
            max_coins = max(max_coins, current_coins)

    return max_coins

```
