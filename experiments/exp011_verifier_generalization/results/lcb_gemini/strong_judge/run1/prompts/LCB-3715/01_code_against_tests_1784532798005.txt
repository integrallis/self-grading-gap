```python
from typing import List

def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Initialize a dictionary to hold the number of coins at each position
    coin_map = {}
    
    # Fill the coin_map with the number of coins for each bag
    for l, r, c in coins:
        for i in range(l, r + 1):
            coin_map[i] = coin_map.get(i, 0) + c
    
    # Create a sorted list of positions/bags
    positions = sorted(coin_map.keys())
    
    # Initialize variables to track the maximum coins collected
    max_coins = 0
    current_coins = 0
    left = 0
    
    # Iterate through the positions to calculate maximum coins for k consecutive bags
    for right in range(len(positions)):
        # Add coins from the current bag position
        current_coins += coin_map[positions[right]]
        
        # If we have more than k bags, slide the window
        if right - left + 1 > k:
            current_coins -= coin_map[positions[left]]
            left += 1
        
        # Update max_coins if we have exactly k bags
        if right - left + 1 == k:
            max_coins = max(max_coins, current_coins)
    
    return max_coins
```
