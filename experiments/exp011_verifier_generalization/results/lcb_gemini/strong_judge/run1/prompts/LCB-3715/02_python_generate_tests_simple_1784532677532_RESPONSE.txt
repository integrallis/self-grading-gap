from typing import List
from solution import maximumCoins

def test_canonical_example_1():
    # spec: coins = [[8,10,1],[1,3,2],[5,6,4]], k = 4
    # Selecting bags at positions [3, 4, 5, 6] gives 2 + 0 + 4 + 4 = 10
    assert maximumCoins([[8,10,1],[1,3,2],[5,6,4]], 4) == 10

def test_canonical_example_2():
    # spec: coins = [[1,10,3]], k = 2
    # Selecting bags at positions [1, 2] gives 3 + 3 = 6
    assert maximumCoins([[1,10,3]], 2) == 6

def test_boundary_case_k_equals_1():
    # spec: coins = [[8,10,1],[1,3,2],[5,6,4]], k = 1
    # The maximum coins in any single bag is 4 (from bags at positions 5 and 6)
    assert maximumCoins([[8,10,1],[1,3,2],[5,6,4]], 1) == 4

def test_segment_of_length_1():
    # spec: coins = [[2,2,10],[5,7,3]], k = 2
    # Selecting bags at positions [2, 3] gives 10 + 0 = 10
    assert maximumCoins([[2,2,10],[5,7,3]], 2) == 10
