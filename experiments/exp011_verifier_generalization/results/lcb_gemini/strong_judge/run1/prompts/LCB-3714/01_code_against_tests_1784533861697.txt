```python
from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Calculate contributions from subsequences of size 1 to k
    for size in range(1, k + 1):
        # For each element, calculate how many subsequences it contributes to as min and max
        for i in range(n):
            # Count how many ways to choose (size-1) elements from the remaining (n-1) elements
            if size > 1:
                contrib = comb(n - 1, size - 1)
            else:
                contrib = 1

            # Each element contributes its value multiplied by the number of subsequences it is part of
            total_sum += (nums[i] * contrib) % MOD
            total_sum %= MOD

    # Each element contributes once as min and once as max in its subsequences
    total_sum = (2 * total_sum) % MOD
    return total_sum


def comb(n: int, k: int) -> int:
    if k > n:
        return 0
    if k == 0 or k == n:
        return 1
    k = min(k, n - k)  # Take advantage of symmetry
    c = 1
    for i in range(k):
        c = c * (n - i) // (i + 1)
    return c

```
