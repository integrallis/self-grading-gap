from typing import List
from solution import minMaxSums

def test_canonical_example_1():
    # spec: input [1, 2, 3], k = 2
    # All subsequences with at most 2 elements:
    # [1] -> min=1, max=1 -> sum=2
    # [2] -> min=2, max=2 -> sum=4
    # [3] -> min=3, max=3 -> sum=6
    # [1, 2] -> min=1, max=2 -> sum=3
    # [1, 3] -> min=1, max=3 -> sum=4
    # [2, 3] -> min=2, max=3 -> sum=5
    # Total = 2 + 4 + 6 + 3 + 4 + 5 = 24
    assert minMaxSums([1, 2, 3], 2) == 24

def test_canonical_example_2():
    # spec: input [5, 0, 6], k = 1
    # All subsequences with exactly 1 element:
    # [5] -> min=5, max=5 -> sum=10
    # [0] -> min=0, max=0 -> sum=0
    # [6] -> min=6, max=6 -> sum=12
    # Total = 10 + 0 + 12 = 22
    assert minMaxSums([5, 0, 6], 1) == 22

def test_canonical_example_3():
    # spec: input [1, 1, 1], k = 2
    # All subsequences with at most 2 elements:
    # [1] -> min=1, max=1 -> sum=2 (3 times)
    # [1, 1] -> min=1, max=1 -> sum=2 (3 times)
    # Total = 3*(2) + 3*(2) = 12
    assert minMaxSums([1, 1, 1], 2) == 12

def test_large_sum_example():
    # Testing large sums with modulo
    # spec: input [10**9, 10**9, 10**9], k = 2
    # All subsequences with at most 2 elements:
    # [10**9] -> min=10**9, max=10**9 -> sum=2 * 10**9 (3 times)
    # [10**9, 10**9] -> min=10**9, max=10**9 -> sum=2 * 10**9 (3 times)
    # Total = 3*(2*10**9) + 3*(2*10**9) = 12 * 10**9
    # Total % (10**9 + 7) = (12 * 10**9) % (10**9 + 7) = 12 * 10**9 % (10**9 + 7)
    assert minMaxSums([10**9, 10**9, 10**9], 2) == (12 * 10**9) % (10**9 + 7)

def test_single_element():
    # Testing the boundary condition where len(nums) = 1
    # spec: input [100], k = 1
    # Only one subsequence: [100] -> min=100, max=100 -> sum=200
    assert minMaxSums([100], 1) == 200

def test_k_equals_length():
    # Testing the boundary condition where k = len(nums)
    # spec: input [10, 20], k = 2
    # Subsequences: [10] -> 20, [20] -> 40, [10, 20] -> 30
    # Total = 20 + 40 + 30 = 90
    assert minMaxSums([10, 20], 2) == 90
