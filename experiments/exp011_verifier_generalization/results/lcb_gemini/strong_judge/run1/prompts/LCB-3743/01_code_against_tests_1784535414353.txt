```python
from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    total_duration = sum(end - start for start, end in zip(startTime, endTime))
    free_time = eventTime - total_duration

    if k >= n:
        return max(0, free_time)

    # Create gaps between meetings, including the start and end of the event.
    gaps = []
    previous_end = 0

    for i in range(n):
        current_start = startTime[i]
        if previous_end < current_start:
            gaps.append(current_start - previous_end)
        previous_end = endTime[i]

    if previous_end < eventTime:
        gaps.append(eventTime - previous_end)

    # Sort gaps in descending order
    gaps.sort(reverse=True)

    # We can remove at most k gaps to maximize free time
    return sum(gaps[:k]) + free_time

```
