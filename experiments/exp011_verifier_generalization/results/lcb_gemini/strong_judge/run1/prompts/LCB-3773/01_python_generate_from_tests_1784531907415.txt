from typing import List

def minimumPairRemoval(nums: List[int]) -> int:
    operations = 0
    while True:
        # Find the adjacent pair with the minimum sum
        min_sum = float('inf')
        min_index = -1

        for i in range(len(nums) - 1):
            current_sum = nums[i] + nums[i + 1]
            if current_sum < min_sum:
                min_sum = current_sum
                min_index = i

        # Check if the array is non-decreasing
        if (len(nums) < 2) or (all(nums[i] <= nums[i + 1] for i in range(len(nums) - 1)) and nums[min_index] != nums[min_index + 1]):
            break

        # Replace the pair with their sum
        nums[min_index] = min_sum
        del nums[min_index + 1]  # Remove the next element
        operations += 1

    return operations

```
