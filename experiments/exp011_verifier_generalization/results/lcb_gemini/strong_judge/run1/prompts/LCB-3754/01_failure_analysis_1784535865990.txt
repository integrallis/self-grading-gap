def maxDistance(s: str, k: int) -> int:
    # Initialize coordinates
    x, y = 0, 0
    max_distance = 0
    # Iterate through each character in the string
    for move in s:
        # Move in the specified direction
        if move == 'N':
            y += 1
        elif move == 'S':
            y -= 1
        elif move == 'E':
            x += 1
        elif move == 'W':
            x -= 1
        # Calculate the current Manhattan distance
        current_distance = abs(x) + abs(y)
        # Update the maximum distance
        max_distance = max(max_distance, current_distance)
    # Add the maximum possible changes to the distance
    return max_distance + k
