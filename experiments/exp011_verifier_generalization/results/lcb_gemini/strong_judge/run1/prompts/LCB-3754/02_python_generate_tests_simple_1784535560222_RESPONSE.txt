```python
from solution import maxDistance

def test_canonical_example_1():
    # spec: maxDistance("NWSE", 1) -> 3
    assert maxDistance("NWSE", 1) == 3

def test_canonical_example_2():
    # spec: maxDistance("NSWWEW", 3) -> 6
    assert maxDistance("NSWWEW", 3) == 6

def test_no_changes_allowed():
    # no changes allowed, max distance is achieved without any modifications
    # Path: "NESW" results in positions (0,1) -> (1,1) -> (1,0) -> (0,0)
    # Max distance from origin is 2 (position (1,1) or (1,0))
    assert maxDistance("NESW", 0) == 2

def test_maximal_changes():
    # all characters can be changed, maximum distance is the length of the string
    # Path: "SW" can be changed to "NN" for a final position of (0,2)
    assert maxDistance("SW", 2) == 2

def test_minimal_string_length():
    # minimal string length of 1, maximum distance is 1
    assert maxDistance("W", 0) == 1

def test_homogeneous_string():
    # string with all identical characters, max distance is just the length
    assert maxDistance("NNNN", 0) == 4

def test_canceling_string():
    # alternating 'N' and 'S', canceling movements with maximum distance equal to 1
    assert maxDistance("NSNS", 0) == 1
```
