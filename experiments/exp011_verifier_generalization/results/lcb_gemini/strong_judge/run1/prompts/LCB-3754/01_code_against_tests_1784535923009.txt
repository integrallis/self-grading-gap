## Implementation

```python
def maxDistance(s: str, k: int) -> int:
    # Initialize coordinates
    x, y = 0, 0
    max_achievable_distance = 0
    # Iterate through each character in the string
    for i in range(1, len(s) + 1):
        move = s[i - 1]
        # Move in the specified direction
        if move == 'N':
            y += 1
        elif move == 'S':
            y -= 1
        elif move == 'E':
            x += 1
        elif move == 'W':
            x -= 1
        # Calculate the current Manhattan distance
        current_distance = abs(x) + abs(y)
        # Calculate the maximum distance that can be achieved with k changes
        max_distance_with_changes = min(i, current_distance + 2 * k)
        # Update the overall maximum achievable distance
        max_achievable_distance = max(max_achievable_distance, max_distance_with_changes)
    return max_achievable_distance

```
