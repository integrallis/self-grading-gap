```python
def maxDistance(s: str, k: int) -> int:
    # Initialize coordinates
    x, y = 0, 0
    max_distance = 0
    # Count moves in each direction
    north = s.count('N')
    south = s.count('S')
    east = s.count('E')
    west = s.count('W')
    # Calculate net vertical and horizontal movements
    vertical = north - south
    horizontal = east - west
    # Calculate the manhattan distance before any changes
    max_distance = abs(vertical) + abs(horizontal)
    # Calculate how many moves we can change to maximize distance
    # The effective number of available changes is limited by 'k'
    total_moves = north + south + east + west
    # The maximum distance we could achieve if we could change all moves
    possible_max_distance = total_moves + min(k, total_moves)
    # The maximum distance is the maximum of the calculated distances
    return max(max_distance, possible_max_distance)
```
