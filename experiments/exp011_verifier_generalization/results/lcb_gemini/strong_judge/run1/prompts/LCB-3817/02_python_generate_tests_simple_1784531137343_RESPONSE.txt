from solution import maxContainers

def test_canonical_example_1():
    # spec: maxContainers(2, 3, 15) -> 4
    # The deck has 4 cells (2x2), each container weighs 3. Total weight = 3 * 4 = 12 <= 15
    assert maxContainers(2, 3, 15) == 4

def test_canonical_example_2():
    # spec: maxContainers(3, 5, 20) -> 4
    # The deck has 9 cells (3x3), each container weighs 5. Total weight = 5 * 4 = 20 <= 20
    assert maxContainers(3, 5, 20) == 4

def test_lower_boundary_conditions():
    # spec: maxContainers(1, 1, 1) -> 1
    # The deck has 1 cell (1x1), and one container weighs 1. Total weight = 1 * 1 = 1 <= 1
    assert maxContainers(1, 1, 1) == 1

def test_upper_boundary_conditions():
    # spec: maxContainers(1000, 1000, 10**9) -> 1000
    # The deck has 1000000 cells (1000x1000), each container weighs 1000. Total weight = 1000 * 1000 = 1000000 <= 10**9
    assert maxContainers(1000, 1000, 10**9) == 1000000

def test_insufficient_capacity():
    # spec: maxContainers(10, 100, 99) -> 0
    # The deck has 100 cells (10x10), each container weighs 100. Total weight = 100 * 1 = 100 > 99
    assert maxContainers(10, 100, 99) == 0
