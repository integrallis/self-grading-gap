from typing import List
from solution import maxSum

def test_canonical_example_1():
    # spec: maxSum([1, 2, 3, 4, 5]) -> 15 (sum of all unique elements)
    assert maxSum([1, 2, 3, 4, 5]) == 15

def test_canonical_example_2():
    # spec: maxSum([1, 1, 0, 1, 1]) -> 1 (only unique element is 1)
    assert maxSum([1, 1, 0, 1, 1]) == 1

def test_canonical_example_3():
    # spec: maxSum([1, 2, -1, -2, 1, 0, -1]) -> 3 (selecting unique elements 1 and 2)
    assert maxSum([1, 2, -1, -2, 1, 0, -1]) == 3

def test_no_positive_unique_numbers():
    # spec: maxSum([-10, -20, -10]) -> -10 (the maximum unique number)
    assert maxSum([-10, -20, -10]) == -10

def test_single_element():
    # spec: maxSum([50]) -> 50 (only one element)
    assert maxSum([50]) == 50
