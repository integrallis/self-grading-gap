```python
from typing import List

def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers
    online_status = [True] * numberOfUsers
    event_queue = []

    for event in events:
        event_queue.append((int(event[1]), event))

    event_queue.sort(key=lambda x: x[0])  # Sort events by timestamp

    for timestamp, event in event_queue:
        if event[0] == "OFFLINE":
            user_id = int(event[2])
            online_status[user_id] = False

            # Set user back online after 60 time units
            for i in range(len(event_queue)):
                if event_queue[i][0] > timestamp:
                    event_queue[i] = (event_queue[i][0] + 60, event_queue[i][1])
                    break
        elif event[0] == "MESSAGE":
            mentions_string = event[2].split()
            mentioned_users = []
            for mention in mentions_string:
                if mention.startswith("id"):
                    user_id = int(mention[2:])
                    mentioned_users.append(user_id)  # Count direct id mentions
                elif mention == "ALL":
                    mentioned_users.extend(range(numberOfUsers))  # Mention all users
                elif mention == "HERE":
                    mentioned_users.extend(i for i in range(numberOfUsers) if online_status[i])  # Only online users

            # Count occurrences of each mention
            for user_id in mentioned_users:
                mentions[user_id] += mentions_string.count(f"id{user_id}")

    return mentions

```
