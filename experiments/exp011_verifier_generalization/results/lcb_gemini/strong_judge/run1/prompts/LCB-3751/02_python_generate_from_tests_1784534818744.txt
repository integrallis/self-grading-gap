from typing import List


def maxFrequency(nums: List[int], k: int) -> int:
    count = 0
    freq = [0] * 51  # Since nums[i] and k are between 1 and 50
    for num in nums:
        freq[num] += 1
    initial_k_count = freq[k]  # Count of k in the original array
    max_gain = 0

    for num in range(1, 51):
        if num != k:
            gain = freq[num]  # We can convert these to k
            # We can add (k - num) to any of these occurrences
            max_gain = max(max_gain, gain)

    return initial_k_count + max_gain

```
