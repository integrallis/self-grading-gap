from typing import List
from solution import minCosts

def test_canonical_example_1():
    # spec: input [5, 3, 4, 1, 3, 2]
    # Minimum costs to reach positions:
    # i = 0 -> cost[0] = 5
    # i = 1 -> cost[1] = 3
    # i = 2 -> cost[1] = 3 (then free swap with 2)
    # i = 3 -> cost[3] = 1
    # i = 4 -> cost[3] = 1 (then free swap with 4)
    # i = 5 -> cost[3] = 1 (then free swap with 5)
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]

def test_canonical_example_2():
    # spec: input [1, 2, 4, 6, 7]
    # Minimum costs to reach positions:
    # i = 0 -> cost[0] = 1
    # i = 1 -> cost[0] = 1 (then free swap with 1)
    # i = 2 -> cost[0] = 1 (then free swap with 2)
    # i = 3 -> cost[0] = 1 (then free swap with 3)
    # i = 4 -> cost[0] = 1 (then free swap with 4)
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]
