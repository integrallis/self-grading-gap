from solution import calculateScore

def test_canonical_example_1():
    # spec: calculateScore("aczzx") -> 5
    # Steps:
    # i = 0. No j, skip.
    # i = 1. No j, skip.
    # i = 2. j = 0 (mirror: 'a' -> 'z'), score += 2 - 0 = 2
    # i = 3. No j, skip.
    # i = 4. j = 1 (mirror: 'b' -> 'y'), score += 4 - 1 = 3
    # Total score = 2 + 3 = 5
    assert calculateScore("aczzx") == 5
