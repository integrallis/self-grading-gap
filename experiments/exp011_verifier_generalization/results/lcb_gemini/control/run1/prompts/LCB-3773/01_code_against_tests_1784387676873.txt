```python
from typing import List


def minimumPairRemoval(nums: List[int]) -> int:
    operations = 0
    while True:
        # Find the adjacent pair with the minimum sum
        min_sum = float('inf')
        min_index = -1
        for i in range(len(nums) - 1):
            current_sum = nums[i] + nums[i + 1]
            if current_sum < min_sum:
                min_sum = current_sum
                min_index = i

        # If no pair needs to be merged, break
        if min_index == -1 or all(nums[i] <= nums[i + 1] for i in range(len(nums) - 1)):
            break

        # Merge the pair
        nums[min_index] += nums[min_index + 1]
        del nums[min_index + 1]
        operations += 1

    return operations

```
