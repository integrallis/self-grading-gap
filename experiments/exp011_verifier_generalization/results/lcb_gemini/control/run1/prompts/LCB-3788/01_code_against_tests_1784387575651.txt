```python
from typing import List

def maxSum(nums: List[int]) -> int:
    unique_elements = set()
    max_sum = 0
    current_sum = 0

    for num in nums:
        if num not in unique_elements:
            unique_elements.add(num)
            current_sum += num
        else:
            # When a duplicate is found, we need to reset and calculate max
            max_sum = max(max_sum, current_sum)
            # Reset current_sum and unique_elements
            while num in unique_elements:
                unique_elements.remove(nums[current_sum])
                current_sum -= nums[current_sum]
            unique_elements.add(num)
            current_sum += num

    # Final check for the last unique segment
    max_sum = max(max_sum, current_sum)

    return max_sum
```
