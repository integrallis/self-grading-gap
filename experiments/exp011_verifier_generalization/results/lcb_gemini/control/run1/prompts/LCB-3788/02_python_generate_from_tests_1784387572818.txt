from typing import List

def maxSum(nums: List[int]) -> int:
    unique_elements = set()
    max_sum = 0
    current_sum = 0

    for num in nums:
        if num not in unique_elements:
            unique_elements.add(num)
            current_sum += num
        else:
            # Calculate max sum with the current unique elements
            max_sum = max(max_sum, current_sum)
            # Reset the current sum and unique elements
            current_sum = num if num not in unique_elements else 0
            unique_elements = {num} if current_sum != 0 else set()
            current_sum += num

    # Final check for the last unique segment
    max_sum = max(max_sum, current_sum)

    return max_sum
```
