from solution import maxContainers

def test_canonical_example_1():
    # spec: n = 2, w = 3, maxWeight = 15
    # total cells = 2 * 2 = 4
    # max weight with all containers = 4 * 3 = 12, which is <= 15
    assert maxContainers(2, 3, 15) == 4

def test_canonical_example_2():
    # spec: n = 3, w = 5, maxWeight = 20
    # total cells = 3 * 3 = 9
    # max weight with all containers = 9 * 5 = 45, which is > 20
    # max containers that can be loaded without exceeding maxWeight = 20 // 5 = 4
    assert maxContainers(3, 5, 20) == 4
