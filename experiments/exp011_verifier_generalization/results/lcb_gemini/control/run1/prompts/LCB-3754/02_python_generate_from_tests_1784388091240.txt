def maxDistance(s: str, k: int) -> int:
    # Direction vectors for N, S, E, W
    direction_vectors = {'N': (0, 1), 'S': (0, -1), 'E': (1, 0), 'W': (-1, 0)}
    x, y = 0, 0
    max_distance = 0

    # Count the number of movements in each direction
    count_n = s.count('N')
    count_s = s.count('S')
    count_e = s.count('E')
    count_w = s.count('W')

    # Calculate the initial position after all moves
    x += count_e - count_w
    y += count_n - count_s
    max_distance = abs(x) + abs(y)

    # Number of moves that can be changed
    total_moves = len(s)
    can_change = min(k, total_moves)

    # Calculate the maximum distance possible by changing moves
    # The maximum distance can be achieved in any direction up to the limit of k
    # If we change moves to maximize distance, we can move in the best direction for the available changes
    max_distance += can_change

    return max_distance
