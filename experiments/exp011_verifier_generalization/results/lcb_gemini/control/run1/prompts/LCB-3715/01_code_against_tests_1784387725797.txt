```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Dictionary to keep track of coins in each bag position
    coin_map = {}
    
    # Fill the coin_map with the coins in respective positions
    for l, r, c in coins:
        for pos in range(l, r + 1):
            coin_map[pos] = coin_map.get(pos, 0) + c
    
    # Prepare a list of coins in order of positions
    max_position = max(coin_map.keys(), default=0)
    bags = [coin_map.get(i, 0) for i in range(1, max_position + 1)]
    
    # If k is greater than the number of bags, return the sum of all available bags
    if k > len(bags):
        return sum(bags)
    
    # Calculate the maximum coins in k consecutive bags using sliding window
    current_sum = sum(bags[:k])
    max_coins = current_sum
    
    for i in range(k, len(bags)):
        current_sum += bags[i] - bags[i - k]
        max_coins = max(max_coins, current_sum)
    
    return max_coins
```
