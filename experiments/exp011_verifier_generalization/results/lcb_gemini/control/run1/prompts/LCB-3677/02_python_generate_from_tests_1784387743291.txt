from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[-float('inf')] * (n + 1) for _ in range(m + 1)]
    dp[0][1] = 0  # Start at (0, 0) with 0 coins

    for i in range(m):
        for j in range(n):
            current = coins[i][j]
            for used_neutralizations in range(3):
                if j + 1 <= n:
                    if current < 0 and used_neutralizations < 2:
                        dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j] + 0)  # Neutralize robber
                    else:
                        dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j] + current)
                if i + 1 <= m:
                    if current < 0 and used_neutralizations < 2:

                    else:
                        dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j] + current)

    return dp[m][n]  # Return maximum coins at (m-1, n-1)
