```python
from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[-float('inf')] * (n + 1) for _ in range(m + 1)]
    dp[0][0] = 0  # Starting point has 0 coins

    for i in range(m):
        for j in range(n):
            current = coins[i][j]
            for used_neutralizations in range(3):
                if i + 1 < m:  # Move down
                    if current < 0 and used_neutralizations < 2:
                        dp[i + 1][j] = max(dp[i + 1][j], dp[i][j])  # Neutralize robber
                    else:
                        dp[i + 1][j] = max(dp[i + 1][j], dp[i][j] + current)
                if j + 1 < n:  # Move right
                    if current < 0 and used_neutralizations < 2:
                        dp[i][j + 1] = max(dp[i][j + 1], dp[i][j])  # Neutralize robber
                    else:
                        dp[i][j + 1] = max(dp[i][j + 1], dp[i][j] + current)

    return dp[m - 1][n - 1]  # Return maximum coins at (m-1, n-1)
```
