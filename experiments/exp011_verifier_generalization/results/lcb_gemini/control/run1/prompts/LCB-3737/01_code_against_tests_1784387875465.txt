```python
from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    dp = [[0]*3 for _ in range(n)]
    for j in range(3):
        dp[0][j] = cost[0][j]
    for i in range(1, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(dp[i-1][(j+1) % 3], dp[i-1][(j+2) % 3])
    min_cost = float('inf')
    for j in range(3):
        min_cost = min(min_cost, dp[n-1][j])
    return min_cost

```
