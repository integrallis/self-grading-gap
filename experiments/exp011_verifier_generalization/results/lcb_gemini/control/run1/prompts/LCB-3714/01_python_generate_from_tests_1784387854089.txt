from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    sorted_nums = sorted(nums)

    # Calculate the contribution of each number in nums
    for i in range(n):
        # The number nums[i] will be the minimum in subsequences of size 1 to k
        min_contributions = (1 << min(k, n - i)) - 1  # 2^min(k, n-i) - 1
        # The number nums[i] will be the maximum in subsequences of size 1 to k
        max_contributions = (1 << min(k, i + 1)) - 1  # 2^min(k, i+1) - 1

        # Ensure we only count valid contributions without duplicates
        total_sum += (sorted_nums[i] * (min_contributions + max_contributions)) % MOD
        total_sum %= MOD

    return total_sum

```
