## Implementation

```python
from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    sorted_nums = sorted(nums)

    # Calculate contributions for each number as both min and max
    for i in range(n):
        min_contrib = (1 << min(k, n - i)) - 1  # Number of subsequences where sorted_nums[i] is the minimum
        max_contrib = (1 << min(k, i + 1)) - 1  # Number of subsequences where sorted_nums[i] is the maximum

        total_sum += (sorted_nums[i] * (min_contrib + max_contrib)) % MOD
        total_sum %= MOD

    return total_sum

```
