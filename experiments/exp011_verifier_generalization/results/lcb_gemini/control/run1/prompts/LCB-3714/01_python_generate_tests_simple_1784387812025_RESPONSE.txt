from typing import List
from solution import minMaxSums

def test_canonical_example_1():
    # spec: minMaxSums([1, 2, 3], 2) -> 24
    # Subsequences with at most 2 elements:
    # [1] -> min: 1, max: 1, sum: 2
    # [2] -> min: 2, max: 2, sum: 4
    # [3] -> min: 3, max: 3, sum: 6
    # [1, 2] -> min: 1, max: 2, sum: 3
    # [1, 3] -> min: 1, max: 3, sum: 4
    # [2, 3] -> min: 2, max: 3, sum: 5
    # Total = 2 + 4 + 6 + 3 + 4 + 5 = 24
    assert minMaxSums([1, 2, 3], 2) == 24

def test_canonical_example_2():
    # spec: minMaxSums([5, 0, 6], 1) -> 22
    # Subsequences with exactly 1 element:
    # [5] -> min: 5, max: 5, sum: 10 (included twice)
    # [0] -> min: 0, max: 0, sum: 0 (included twice)
    # [6] -> min: 6, max: 6, sum: 12 (included twice)
    # Total = 10 + 0 + 12 = 22
    assert minMaxSums([5, 0, 6], 1) == 22

def test_canonical_example_3():
    # spec: minMaxSums([1, 1, 1], 2) -> 12
    # Subsequences with at most 2 elements:
    # [1] -> min: 1, max: 1, sum: 2 (each appears 3 times)
    # [1, 1] -> min: 1, max: 1, sum: 2 (each appears 3 times)
    # Total = 3 * 2 + 3 * 2 = 12
    assert minMaxSums([1, 1, 1], 2) == 12
