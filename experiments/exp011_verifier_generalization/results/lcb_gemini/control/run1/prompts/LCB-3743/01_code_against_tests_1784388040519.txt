```python
from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_times = []
    last_end = 0

    # Collect free time intervals before the meetings
    for i in range(n):
        if startTime[i] > last_end:
            free_times.append((last_end, startTime[i]))
        last_end = endTime[i]

    # Collect free time after the last meeting
    if last_end < eventTime:
        free_times.append((last_end, eventTime))

    # Calculate gaps between meetings
    gaps = []
    for i in range(len(free_times) - 1):
        gaps.append(free_times[i + 1][0] - free_times[i][1])

    # Sort gaps to prioritize the largest ones
    gaps.sort(reverse=True)

    # Maximizing free time by moving k meetings
    max_free_time = 0
    for i in range(min(k, len(gaps))):
        max_free_time += gaps[i]

    return max_free_time

```
