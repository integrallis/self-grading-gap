from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    meetings = list(zip(startTime, endTime))
    # Add boundaries at the start and the end of the event
    meetings = [(0, 0)] + meetings + [(eventTime, eventTime)]
    free_times = []

    # Calculate gaps between meetings
    for i in range(1, len(meetings)):
        free_time = meetings[i][0] - meetings[i - 1][1]
        if free_time > 0:
            free_times.append(free_time)

    # Maximizing free time by moving k meetings
    max_free_time = 0
    if len(free_times) == 0:
        return 0
    if k >= len(free_times):  
        return sum(free_times)  

    # Sort the available free slots in descending order
    free_times.sort(reverse=True)

    # Calculate max free time after moving k meetings
    max_free_time = sum(free_times[:k])

    return max_free_time

```
