from typing import List
from solution import maxLength

def test_canonical_example_1():
    # For input [1, 2, 1, 2, 1, 1, 1]:
    # Longest product equivalent subarray is [1, 2, 1, 1, 1]
    # prod([1, 2, 1, 1, 1]) = 2
    # gcd([1, 2, 1, 1, 1]) = 1
    # lcm([1, 2, 1, 1, 1]) = 2
    # Since prod(arr) == lcm(arr) * gcd(arr), the output should be 5
    assert maxLength([1, 2, 1, 2, 1, 1, 1]) == 5
