from typing import List
from solution import maxSum

def test_canonical_example_1():
    # Example: input [1, 2, 3, 4, 5] -> output 15
    # The sum of the entire array is 1 + 2 + 3 + 4 + 5 = 15
    assert maxSum([1, 2, 3, 4, 5]) == 15

def test_canonical_example_2():
    # Example: input [1, 1, 0, 1, 1] -> output 1
    # The maximum unique sum can be achieved by keeping just one '1' 
    assert maxSum([1, 1, 0, 1, 1]) == 1

def test_canonical_example_3():
    # Example: input [1, 2, -1, -2, 1, 0, -1] -> output 3
    # The maximum unique sum can be achieved by selecting the subarray [1, 2]
    # which gives 1 + 2 = 3
    assert maxSum([1, 2, -1, -2, 1, 0, -1]) == 3
