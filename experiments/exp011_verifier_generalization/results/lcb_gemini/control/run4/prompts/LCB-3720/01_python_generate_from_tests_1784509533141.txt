from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    edges.sort(key=lambda x: x[2])  # Sort edges by weight
    left, right = 1, 10**6  # Weight range

    def can_form_graph(max_weight):
        from collections import defaultdict, deque
        graph = defaultdict(list)

        # Construct the graph without limiting the outgoing edges initially
        for u, v, w in edges:
            if w <= max_weight:
                graph[u].append(v)

        visited = [False] * n
        out_degree = [0] * n  # Initialize out_degree after graph construction

        queue = deque([0])  # Start BFS from node 0
        visited[0] = True

        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    out_degree[node] += 1  # Count outgoing edges from the node
                    if out_degree[node] < threshold:
                        queue.append(neighbor)

        return all(visited)

    while left < right:
        mid = (left + right) // 2
        if can_form_graph(mid):
            right = mid  # Try for a smaller max weight
        else:
            left = mid + 1  # Increase max weight

    if can_form_graph(left):
        return left
    return -1

```
