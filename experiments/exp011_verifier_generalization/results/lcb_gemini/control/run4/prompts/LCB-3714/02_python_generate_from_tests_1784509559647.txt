from typing import List


def minMaxSums(nums: List[int], k: int) -> int:
    MOD = 10**9 + 7
    n = len(nums)
    nums.sort()
    total_sum = 0

    # Contribution of each number as max
    for i in range(n):
        # Each number can be max in subsequences of size 1 to k
        total_sum += nums[i] * (pow(2, min(k, n - 1), MOD) - 1)  # All combinations excluding empty
        total_sum %= MOD

    # Contribution of each number as min
    for i in range(n):
        # Each number can be min in subsequences of size 1 to k
        total_sum += nums[i] * (pow(2, min(k, n - 1), MOD) - 1)
        total_sum %= MOD

    return total_sum % MOD
```
