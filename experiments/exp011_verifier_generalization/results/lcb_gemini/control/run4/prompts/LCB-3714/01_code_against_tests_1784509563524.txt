```python
from typing import List


def minMaxSums(nums: List[int], k: int) -> int:
    MOD = 10**9 + 7
    n = len(nums)
    total_sum = 0
    nums.sort()

    # Prefix powers of 2 for combinations count
    pow2 = [1] * (k + 1)
    for i in range(1, k + 1):
        pow2[i] = (pow2[i - 1] * 2) % MOD

    # Calculate contribution as max
    for i in range(n):
        contribution_as_max = nums[i] * pow2[min(k, i + 1)] % MOD
        total_sum = (total_sum + contribution_as_max) % MOD

    # Calculate contribution as min
    for i in range(n):
        contribution_as_min = nums[i] * pow2[min(k, n - i)] % MOD
        total_sum = (total_sum + contribution_as_min) % MOD

    return total_sum % MOD

```
