def maxDistance(s: str, k: int) -> int:
    # Initialize coordinates
    x, y = 0, 0
    # Count movements in each direction
    for char in s:
        if char == 'N':
            y += 1
        elif char == 'S':
            y -= 1
        elif char == 'E':
            x += 1
        elif char == 'W':
            x -= 1
    # Calculate maximum distance considering changes allowed by k
    max_distance = abs(x) + abs(y) + k
    return max_distance

```
