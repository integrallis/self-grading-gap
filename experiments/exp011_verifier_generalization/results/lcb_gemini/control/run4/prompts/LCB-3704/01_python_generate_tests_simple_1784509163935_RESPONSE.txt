from solution import countPartitions

def test_canonical_example_1():
    # spec: [10], [10, 3, 7, 6] -> sum difference = 10 - 26 = -16 (even)
    # spec: [10, 10], [3, 7, 6] -> sum difference = 20 - 16 = 4 (even)

    # spec: No partitions result in an even sum difference.
