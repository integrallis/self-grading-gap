```python
from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[float('-inf')] * (n + 1) for _ in range(m + 1)]
    dp[0][0] = 0  # Starting point with 0 coins

    for i in range(m):
        for j in range(n):
            if coins[i][j] >= 0:
                dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j + 1] + coins[i][j], dp[i + 1][j] + coins[i][j])
            else:
                # If there's a robber, we can choose to neutralize it or not
                for k in range(3):  # k = 0, 1, 2 neutralizations used
                    if k == 0:
                        dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j + 1] + coins[i][j], dp[i + 1][j] + coins[i][j])
                    elif k == 1:
                        dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j + 1] + coins[i][j] + coins[i][j], dp[i + 1][j] + coins[i][j] + coins[i][j])
                    elif k == 2:
                        dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j + 1] + coins[i][j] + coins[i][j] + coins[i][j], dp[i + 1][j] + coins[i][j] + coins[i][j] + coins[i][j])
    return dp[m][n]

```
