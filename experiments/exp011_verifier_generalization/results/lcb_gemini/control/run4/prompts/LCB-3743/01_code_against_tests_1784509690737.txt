from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_intervals = []

    # Calculate initial free intervals
    if startTime[0] > 0:
        free_intervals.append((0, startTime[0]))
    for i in range(n - 1):
        if endTime[i] < startTime[i + 1]:
            free_intervals.append((endTime[i], startTime[i + 1]))
    if endTime[-1] < eventTime:
        free_intervals.append((endTime[-1], eventTime))

    # Calculate the lengths of free time intervals
    free_lengths = [end - start for start, end in free_intervals]

    # Identify how many meetings can potentially be rescheduled
    rescheduled_intervals = []
    for i in range(n):
        reschedule_start = max(0, startTime[i] - (endTime[i] - startTime[i]))
        if (i > 0 and reschedule_start < startTime[i]):
            rescheduled_intervals.append((reschedule_start, startTime[i]))
        if (i + 1 < n and endTime[i] + (endTime[i] - startTime[i]) <= startTime[i + 1]):
            rescheduled_intervals.append((endTime[i], endTime[i] + (endTime[i] - startTime[i])))

    # Sort rescheduled intervals by length
    rescheduled_intervals.sort(key=lambda x: x[1] - x[0], reverse=True)

    # Maximize free time
    max_free_time = 0
    for i in range(min(k, len(rescheduled_intervals))):
        max_free_time = max(max_free_time, rescheduled_intervals[i][1] - rescheduled_intervals[i][0])

    total_free_time = sum(free_lengths)
    return max(max_free_time, total_free_time)
