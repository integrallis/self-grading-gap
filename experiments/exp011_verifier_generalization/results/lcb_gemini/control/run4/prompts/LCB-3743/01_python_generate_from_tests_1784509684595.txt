from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_intervals = []

    # Calculate initial free intervals
    if startTime[0] > 0:
        free_intervals.append((0, startTime[0]))
    for i in range(n - 1):
        if endTime[i] < startTime[i + 1]:
            free_intervals.append((endTime[i], startTime[i + 1]))
    if endTime[-1] < eventTime:
        free_intervals.append((endTime[-1], eventTime))

    # Calculate the lengths of free time intervals
    free_lengths = [end - start for start, end in free_intervals]

    # Calculate the total time occupied by meetings
    total_occupied_time = 0
    for i in range(n):
        total_occupied_time += endTime[i] - startTime[i]

    # Calculate the maximum free time possible
    max_free_time = free_intervals[0][0]  # Time before the first meeting
    for i in range(len(free_lengths)):
        max_free_time = max(max_free_time, free_lengths[i])

    # We can reschedule k meetings to create more free time
    if k > 0:
        total_time = eventTime - total_occupied_time
        if total_time > 0:
            max_free_time = max(max_free_time, total_time)

    return max_free_time

```
