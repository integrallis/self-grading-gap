def test_canonical_example_1():
    # Input: nums = [2,3,1]
    # Subarray sums:
    # i = 0: sum(nums[0:0+1]) = 2
    # i = 1: sum(nums[0:1+1]) = 5
    # i = 2: sum(nums[1:2+1]) = 4
    # Total = 2 + 5 + 4 = 11
    assert subarraySum([2, 3, 1]) == 11

def test_canonical_example_2():
    # Input: nums = [3,1,1,2]
    # Subarray sums:
    # i = 0: sum(nums[0:0+1]) = 3
    # i = 1: sum(nums[0:1+1]) = 4
    # i = 2: sum(nums[1:2+1]) = 2
    # i = 3: sum(nums[1:3+1]) = 4
    # Total = 3 + 4 + 2 + 4 = 13
