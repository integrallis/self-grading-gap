from typing import List
from solution import minCosts

def test_canonical_example_1():
    # For input [5, 3, 4, 1, 3, 2]:
    # Cost to reach position 0 = 5 (swap with person 0)
    # Cost to reach position 1 = 3 (swap with person 1)
    # Cost to reach position 2 = 3 (swap with person 1 then free swap with person 2)
    # Cost to reach position 3 = 1 (swap with person 3)
    # Cost to reach position 4 = 1 (swap with person 3 then free swap with person 4)
    # Cost to reach position 5 = 1 (swap with person 3 then free swap with person 5)
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]

def test_canonical_example_2():
    # For input [1, 2, 4, 6, 7]:
    # Cost to reach position 0 = 1 (swap with person 0)
    # Cost to reach position 1 = 1 (swap with person 0 then free swap with person 1)
    # Cost to reach position 2 = 1 (swap with person 0 then free swap with person 1 and 2)
    # Cost to reach position 3 = 1 (swap with person 0 then free swap with person 1, 2, and 3)
    # Cost to reach position 4 = 1 (swap with person 0 then free swap with person 1, 2, 3, and 4)
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]
