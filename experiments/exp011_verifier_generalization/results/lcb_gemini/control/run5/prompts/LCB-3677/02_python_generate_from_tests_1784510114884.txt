from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[-float('inf')] * (n + 1) for _ in range(m + 1)]
    dp[0][1] = 0

    for i in range(m):
        for j in range(n):
            current_coins = coins[i][j]
            for k in range(3):  # k = 0, 1, 2 for neutralizations
                if current_coins < 0 and k == 0:
                    continue
                if i > 0:
                    dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j + 1] + (current_coins if current_coins >= 0 else 0))
                    dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j + 1] - abs(current_coins) + (0 if current_coins < 0 and k == 1 else 0))
                if j > 0:
                    dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i + 1][j] + (current_coins if current_coins >= 0 else 0))
                    dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i + 1][j] - abs(current_coins) + (0 if current_coins < 0 and k == 1 else 0))

    return dp[m][n]
```
