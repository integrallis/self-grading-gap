from typing import List


def totalNumbers(digits: List[int]) -> int:
    from itertools import permutations
    even_count = 0
    distinct_numbers = set()
    for digit in digits:
        if digit % 2 == 0:
            # Generate permutations where `digit` is at the last position (even number)
            for perm in permutations(digits, 3):
                if perm[2] == digit and perm[0] != 0:
                    distinct_numbers.add(perm)
    return len(distinct_numbers)
