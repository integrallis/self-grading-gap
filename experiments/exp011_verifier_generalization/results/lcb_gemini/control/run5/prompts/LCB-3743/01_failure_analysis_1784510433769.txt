from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    intervals = list(zip(startTime, endTime))
    free_times = []
    prev_end = 0

    # Calculate initial free times between meetings
    for i in range(n):
        if intervals[i][0] > prev_end:
            free_times.append((prev_end, intervals[i][0]))
        prev_end = intervals[i][1]

    # Add free time after the last meeting
    if prev_end < eventTime:
        free_times.append((prev_end, eventTime))

    # Maximum free time is the largest gap
    max_free_time = max((end - start for start, end in free_times), default=0)

    # Calculate gaps between meetings
    gaps = []
    for i in range(1, n):
        gap = intervals[i][0] - intervals[i-1][1]
        if gap > 0:
            gaps.append(gap)

    # We can reschedule at most k meetings to merge the largest gaps into free intervals
    gaps.sort(reverse=True)
    added_gap = sum(gaps[:k])

    return max_free_time + added_gap
