from typing import List

def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    left, right = 1, max(W for _, _, W in edges)  # Binary search bounds

    def canReachWithMaxWeight(maxWeight):
        new_graph = [[] for _ in range(n)]  # Create a new graph
        out_degree = [0] * n
        for A, B, W in edges:
            if W <= maxWeight:
                new_graph[B].append(A)  # Reverse the direction for reachability
                out_degree[B] += 1
                if out_degree[B] > threshold:
                    return False

        # Check if we can reach node 0 from all other nodes
        reachable = [False] * n
        reachable[0] = True
        queue = [0]
        while queue:
            node = queue.pop(0)
            for neighbor in new_graph[node]:
                if not reachable[neighbor]:
                    reachable[neighbor] = True
                    queue.append(neighbor)

        return all(reachable)

    answer = -1
    while left <= right:
        mid = (left + right) // 2
        if canReachWithMaxWeight(mid):
            answer = mid
            right = mid - 1
        else:
            left = mid + 1

    return answer
