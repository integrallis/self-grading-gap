    # First operation: (3+9)%10=2, (9+0)%10=9, (0+2)%10=2 -> "292"

    # First operation: (3+4)%10=7, (4+7)%10=1, (7+8)%10=5, (8+9)%10=7 -> "7157"
    # Second operation: (7+1)%10=8, (1+5)%10=6 -> "862"

    # Since "48" has different digits, output is False
