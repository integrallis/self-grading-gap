```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    max_coins = 0
    coin_map = {}

    # Building a coins map from the ranges
    for l, r, c in coins:
        for pos in range(l, r + 1):
            if pos in coin_map:
                coin_map[pos] += c
            else:
                coin_map[pos] = c

    # Sorting the keys to process range sums
    sorted_positions = sorted(coin_map.keys())
    current_coins = 0
    left_index = 0

    # Sliding window to find the maximum coins in k consecutive bags
    for right_index in range(len(sorted_positions)):
        current_pos = sorted_positions[right_index]
        current_coins += coin_map[current_pos]

        # Ensure the window is of size k
        while current_pos - sorted_positions[left_index] + 1 > k:
            current_coins -= coin_map[sorted_positions[left_index]]
            left_index += 1

        # Update max_coins if we have a valid window
        if current_pos - sorted_positions[left_index] + 1 == k:
            max_coins = max(max_coins, current_coins)

    return max_coins

```
