```python
from typing import List
import math


def maxLength(nums: List[int]) -> int:
    n = len(nums)
    max_length = 0
    for i in range(n):
        prod = 1
        gcd_value = nums[i]
        lcm_value = nums[i]
        for j in range(i, n):
            prod *= nums[j]
            gcd_value = math.gcd(gcd_value, nums[j])
            lcm_value = (lcm_value * nums[j]) // math.gcd(lcm_value, nums[j])
            if prod == lcm_value * gcd_value:
                max_length = max(max_length, j - i + 1)
    return max_length

```
