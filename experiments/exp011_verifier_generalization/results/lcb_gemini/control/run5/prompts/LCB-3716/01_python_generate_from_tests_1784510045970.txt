from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n == 0:
        return 0
    
    # dp[i] will store the length of the longest subsequence ending at index i
    dp = [1] * n
    
    # To store the maximum length of valid subsequence
    max_length = 1
    
    # Iterate through each number in nums
    for i in range(n):
        for j in range(i):
            # Calculate the absolute difference
            diff1 = abs(nums[i] - nums[j])
            # Check if we can form a valid subsequence with non-increasing differences
            if abs(nums[j] - nums[j-1]) >= diff1 and dp[j] + 1 > dp[i]:
                dp[i] = dp[j] + 1
        max_length = max(max_length, dp[i])
    
    return max_length

```
