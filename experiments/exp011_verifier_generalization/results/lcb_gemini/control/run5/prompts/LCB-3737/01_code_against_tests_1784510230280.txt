```python
from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    dp = [[0] * 3 for _ in range(n)]

    # Initialize the first house costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the dp table for the first half
    for i in range(1, n // 2 + 1):
        dp[i][0] = cost[i][0] + min(dp[i - 1][1], dp[i - 1][2])  # Color 1
        dp[i][1] = cost[i][1] + min(dp[i - 1][0], dp[i - 1][2])  # Color 2
        dp[i][2] = cost[i][2] + min(dp[i - 1][0], dp[i - 1][1])  # Color 3

    first_half_cost = min(dp[n // 2][0], dp[n // 2][1], dp[n // 2][2])

    # Fill the dp table for the second half
    for i in range(n // 2, n - 1):
        j = n - 1 - i  # Corresponding index in the first half
        dp[i + 1][0] = cost[i + 1][0] + min(dp[i][1], dp[i][2])
        dp[i + 1][1] = cost[i + 1][1] + min(dp[i][0], dp[i][2])
        dp[i + 1][2] = cost[i + 1][2] + min(dp[i][0], dp[i][1])

    second_half_cost = min(dp[n - 1][0], dp[n - 1][1], dp[n - 1][2])

    return first_half_cost + second_half_cost

```
