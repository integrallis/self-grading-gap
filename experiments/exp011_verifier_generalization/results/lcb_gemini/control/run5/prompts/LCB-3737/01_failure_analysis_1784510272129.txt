def minCost(n: int, cost: List[List[int]]) -> int:
    dp = [[0] * 3 for _ in range(n)]

    # Initialize the first house costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the dp table for all houses
    for i in range(1, n):
        for color in range(3):
            dp[i][color] = cost[i][color] + min(dp[i - 1][(color + 1) % 3], dp[i - 1][(color + 2) % 3])

    # Calculate the minimum cost while considering equidistant houses
    min_cost = float('inf')

        min_cost = min(min_cost, dp[n - 1][color] + dp[n - 2][(color + 1) % 3], dp[n - 1][color] + dp[n - 2][(color + 2) % 3])

    return min_cost
