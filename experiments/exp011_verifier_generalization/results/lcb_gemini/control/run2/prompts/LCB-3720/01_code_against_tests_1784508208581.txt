```python
from typing import List
import sys
from collections import defaultdict, deque


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Step 1: Determine the min and max weights
    min_weight = min(edge[2] for edge in edges)
    max_weight = max(edge[2] for edge in edges)

    def can_reach_with_max_weight(max_weight: int) -> bool:
        graph = defaultdict(list)
        out_degree = defaultdict(int)

        # Build graph with edges <= max_weight
        for u, v, w in edges:
            if w <= max_weight:
                graph[u].append(v)
                out_degree[u] += 1
                if out_degree[u] > threshold:
                    return False

        # BFS to check if all nodes can reach 0
        reached = [False] * n
        queue = deque([0])
        reached[0] = True

        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if not reached[neighbor]:
                    reached[neighbor] = True
                    queue.append(neighbor)

        return all(reached)

    # Step 2: Binary search over edge weights
    left, right = min_weight, max_weight
    answer = -1

    while left <= right:
        mid = (left + right) // 2
        if can_reach_with_max_weight(mid):
            answer = mid
            right = mid - 1
        else:
            left = mid + 1

    return answer
```
