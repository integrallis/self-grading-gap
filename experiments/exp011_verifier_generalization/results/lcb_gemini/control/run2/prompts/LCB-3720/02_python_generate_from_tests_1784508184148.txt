from typing import List
import sys
from collections import defaultdict, deque


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Step 1: Sort edges by weight
    edges.sort(key=lambda x: x[2])

    def can_reach_with_max_weight(max_weight: int) -> bool:
        graph = defaultdict(list)
        outgoing_count = [0] * n

        # Build graph with edges <= max_weight
        for u, v, w in edges:
            if w <= max_weight and outgoing_count[u] < threshold:
                graph[u].append(v)
                outgoing_count[u] += 1

        # BFS to check if all nodes can reach 0
        reached = [False] * n
        queue = deque([0])
        reached[0] = True

        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if not reached[neighbor]:
                    reached[neighbor] = True
                    queue.append(neighbor)

        return all(reached)

    # Step 2: Binary search over edge weights
    left, right = 1, 10**6
    answer = -1

    while left <= right:
        mid = (left + right) // 2
        if can_reach_with_max_weight(mid):
            answer = mid
            right = mid - 1
        else:
            left = mid + 1

    return answer

```
