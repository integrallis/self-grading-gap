```python
from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize the dp array with costs
    dp = [[0] * 3 for _ in range(n)]
    dp[0] = cost[0]

    # Fill dp table based on the constraints
    for i in range(1, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(
                dp[i - 1][(j + 1) % 3],  # Different color from previous
                dp[i - 1][(j + 2) % 3]   # Different color from previous
            )

    # Equidistant condition checks (0 with n-1, 1 with n-2, etc.)
    # We need to ensure symmetry is maintained between houses.
    for i in range(n // 2):
        for j in range(3):
            if (j == 0):  # If we're considering house n-1 painted with color 1 (j == 0)
                dp[n - 1 - i][j] = float('inf')  # Cannot pick this color for equidistant house
            else:
                dp[n - 1 - i][j] = cost[n - 1 - i][j] + min(
                    dp[i][(j + 1) % 3],
                    dp[i][(j + 2) % 3]
                )

    # Return minimum cost for the last house considering all colors
    return min(dp[n - 1])

```
