from typing import List
from solution import countPartitions

def test_canonical_example_1():
    # spec: countPartitions([10, 10, 3, 7, 6]) -> 4
    # Explanation:
    # [10], [10, 3, 7, 6] -> difference = 10 - 26 = -16 (even)
    # [10, 10], [3, 7, 6] -> difference = 20 - 16 = 4 (even)
    # [10, 10, 3], [7, 6] -> difference = 23 - 13 = 10 (even)
    # [10, 10, 3, 7], [6] -> difference = 30 - 6 = 24 (even)
    assert countPartitions([10, 10, 3, 7, 6]) == 4

def test_canonical_example_2():
    # spec: countPartitions([1, 2, 2]) -> 0
    # Explanation:
    # No partition results in an even sum difference.
    assert countPartitions([1, 2, 2]) == 0

def test_canonical_example_3():
    # spec: countPartitions([2, 4, 6, 8]) -> 3
    # Explanation:
    # All partitions result in even sum differences.
    # Partitions:
    # [2], [4, 6, 8] -> difference = 2 - 18 = -16 (even)
    # [2, 4], [6, 8] -> difference = 6 - 14 = -8 (even)
    # [2, 4, 6], [8] -> difference = 12 - 8 = 4 (even)
    assert countPartitions([2, 4, 6, 8]) == 3
