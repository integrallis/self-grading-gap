from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n < 2:
        return n
    dp = [1] * n
    max_length = 1
    for i in range(n):
        for j in range(i):
            diff1 = abs(nums[i] - nums[j])
            if j > 0:
                diff2 = abs(nums[j] - nums[j-1])
                if diff1 <= diff2:
                    dp[i] = max(dp[i], dp[j] + 1)
            max_length = max(max_length, dp[i])
    return max_length
```
