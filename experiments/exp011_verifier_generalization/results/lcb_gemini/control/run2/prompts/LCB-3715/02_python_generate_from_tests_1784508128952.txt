You are also given an integer k.
Return the maximum amount of coins you can obtain by collecting k consecutive bags.

Selecting bags at positions [3, 4, 5, 6] gives the maximum number of coins: 2 + 0 + 4 + 4 = 10.

Example 2:

Output: 6
Explanation:
Selecting bags at positions [1, 2] gives the maximum number of coins: 3 + 3 = 6.

 
Constraints:

1 <= k <= 10^9
coins[i] == [l_i, r_i, c_i]
1 <= l_i <= r_i <= 10^9
1 <= c_i <= 1000

```python
from typing import List
```


```python
def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a list to hold the count of coins for each position
    max_position = 0
    for l, r, c in coins:
        max_position = max(max_position, r)

    # Initialize a list for counting coins in bags
    coin_count = [0] * (max_position + 2)

    # Fill the coin_count array based on the coins input
    for l, r, c in coins:
        for pos in range(l, r + 1):
            coin_count[pos] += c

    # Calculate the maximum coins we can obtain from k consecutive bags
    max_coins = 0
    current_coins = sum(coin_count[1:k + 1])  # Sum of the first k bags
    max_coins = max(max_coins, current_coins)

    # Slide the window across the positions
    for i in range(2, max_position + 1):
        current_coins = current_coins - coin_count[i - 1] + coin_count[i + k - 1]
        max_coins = max(max_coins, current_coins)

    return max_coins
```
