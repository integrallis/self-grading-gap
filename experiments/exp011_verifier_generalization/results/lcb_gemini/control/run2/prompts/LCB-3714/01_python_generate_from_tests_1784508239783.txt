from typing import List
from math import comb

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    for i in range(1, k + 1):
        for num in nums:
            count = comb(n - 1, i - 1)
            total_sum += (num * count * 2) % MOD
            total_sum %= MOD
    return total_sum

```
