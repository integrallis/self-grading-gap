from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    for i in range(1, k + 1):
        # Calculate the contribution of each element as min and max
        # Using combinations, we need to calculate how many times each element is the min or max
        for num in nums:
            # Number of ways to choose (i-1) elements from the rest (n-1)
            count_min = comb(n - 1, i - 1)
            count_max = comb(n - 1, i - 1)
            total_sum += (num * (count_min + count_max)) % MOD
            total_sum %= MOD
    return total_sum

from math import comb
