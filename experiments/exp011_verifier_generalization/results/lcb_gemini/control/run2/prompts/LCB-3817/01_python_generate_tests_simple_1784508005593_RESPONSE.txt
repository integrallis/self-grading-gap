from solution import maxContainers

def test_canonical_example_1():
    # spec: n = 2, w = 3, maxWeight = 15
    # total cells = 2 * 2 = 4, total weight for 4 containers = 4 * 3 = 12
    # maxWeight = 15, so we can load all 4 containers
    assert maxContainers(2, 3, 15) == 4

def test_canonical_example_2():
    # spec: n = 3, w = 5, maxWeight = 20
    # total cells = 3 * 3 = 9, total weight for 9 containers = 9 * 5 = 45
    # maxWeight = 20, we can only load 4 containers (4 * 5 = 20)
    assert maxContainers(3, 5, 20) == 4
