from typing import List
from solution import minCosts

def test_canonical_example_1():
    # spec: input [5, 3, 4, 1, 3, 2] 
    # - i = 0: cost = 5
    # - i = 1: cost = 3
    # - i = 2: can swap with 1 for 3, then swap with 2 for free
    # - i = 3: cost = 1
    # - i = 4: swap with 3 for 1, then swap with 4 for free
    # - i = 5: swap with 3 for 1, then swap with 5 for free
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]

def test_canonical_example_2():
    # spec: input [1, 2, 4, 6, 7]
    # - i = 0: cost = 1
    # - i = 1: swap with 0 for 1
    # - i = 2: swap with 0 for 1
    # - i = 3: swap with 0 for 1
    # - i = 4: swap with 0 for 1
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]
