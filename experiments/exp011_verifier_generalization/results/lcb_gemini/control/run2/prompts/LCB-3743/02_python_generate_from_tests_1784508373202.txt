from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_times = []

    # Calculate free time intervals before the first meeting and after the last meeting
    if startTime[0] > 0:
        free_times.append(startTime[0])
    if endTime[-1] < eventTime:
        free_times.append(eventTime - endTime[-1])

    # Calculate free times between meetings
    for i in range(1, n):
        free_time = startTime[i] - endTime[i - 1]
        if free_time > 0:
            free_times.append(free_time)

    # Sort free times in descending order
    free_times.sort(reverse=True)

    # Sum the largest k free times
    max_free_time = sum(free_times[:k])
    return max_free_time
```
