from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[float('-inf')] * (n + 1) for _ in range(m + 1)]
    dp[0][1] = 0  # Starting point (0, 0) with 0 coins

    for i in range(m):
        for j in range(n):
            current = coins[i][j]
            if current < 0:
                # If there's a robber, we need to consider using a neutralization
                dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j + 1] - abs(current), dp[i][j + 1])
                dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i + 1][j] - abs(current), dp[i + 1][j])
            else:
                # If it's coins, we just add the coins
                dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j + 1] + current, dp[i][j + 1])
                dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i + 1][j] + current, dp[i + 1][j])
            # Handle the case of using one or two neutralizations
            if i > 0 and j > 0:
                dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i - 1][j] + current)
                dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j - 1] + current)

    return dp[m][n]
