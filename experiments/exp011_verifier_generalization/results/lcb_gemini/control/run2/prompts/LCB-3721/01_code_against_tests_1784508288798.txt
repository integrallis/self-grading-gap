```python
from typing import List


def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers
    online_status = [True] * numberOfUsers

    for event in events:
        event_type, timestamp, *args = event
        timestamp = int(timestamp)
        if event_type == "OFFLINE":
            user_id = int(args[0])
            online_status[user_id] = False
        elif event_type == "MESSAGE":
            mentions_string = args[0]
            if mentions_string == "ALL":
                for i in range(numberOfUsers):
                    mentions[i] += 1
            elif mentions_string == "HERE":
                for i in range(numberOfUsers):
                    if online_status[i]:
                        mentions[i] += 1
            else:
                user_ids = mentions_string.split()
                for user_id in user_ids:
                    user_id = int(user_id[2:])  # Extracting the id from id<number>
                    mentions[user_id] += 1

        # Process user coming back online after 60 time units
        if event_type == "OFFLINE":
            for j in range(len(events)):  
                if events[j][0] == "OFFLINE" and online_status[int(events[j][2])]:
                    continue
                if events[j][0] == "MESSAGE":
                    msg_timestamp = int(events[j][1])
                    if msg_timestamp >= timestamp + 60:
                        online_status[user_id] = True

    return mentions

```
