from solution import countMentions

def test_canonical_example_1():
    # spec: countMentions(2, [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","71","HERE"]]) 
    # mentions user 1 and 0 at timestamp 10, 
    # user 0 goes offline at timestamp 11, 
    # user 0 is not counted for "HERE" at timestamp 71, 
    # hence mentions = [2, 2]
    assert countMentions(2, [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","71","HERE"]]) == [2, 2]

def test_canonical_example_2():
    # spec: countMentions(2, [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","12","ALL"]]) 
    # mentions user 1 and 0 at timestamp 10, 
    # user 0 goes offline at timestamp 11, 
    # "ALL" at timestamp 12 mentions both user 0 and 1, 
    # hence mentions = [2, 2]
    assert countMentions(2, [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","12","ALL"]]) == [2, 2]

def test_canonical_example_3():
    # spec: countMentions(2, [["OFFLINE","10","0"],["MESSAGE","12","HERE"]]) 
    # user 0 goes offline at timestamp 10, 
    # "HERE" at timestamp 12 mentions only user 1, 
    # hence mentions = [0, 1]
    assert countMentions(2, [["OFFLINE","10","0"],["MESSAGE","12","HERE"]]) == [0, 1]
