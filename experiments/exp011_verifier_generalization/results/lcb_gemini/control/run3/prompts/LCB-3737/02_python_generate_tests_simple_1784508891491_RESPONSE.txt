from typing import List
from solution import minCost

def test_canonical_example_1():
    # Example: input '4\n[[3, 5, 7], [6, 2, 9], [4, 8, 1], [7, 3, 5]]' 
    # The optimal painting sequence is [1, 2, 3, 2] with corresponding costs [3, 2, 1, 3]. 
    # Minimum cost = 3 + 2 + 1 + 3 = 9
    assert minCost(4, [[3, 5, 7], [6, 2, 9], [4, 8, 1], [7, 3, 5]]) == 9

def test_canonical_example_2():
    # Example: input '6\n[[2, 4, 6], [5, 3, 8], [7, 1, 9], [4, 6, 2], [3, 5, 7], [8, 2, 4]]'
    # The optimal painting sequence is [1, 3, 2, 3, 1, 2] with corresponding costs [2, 8, 1, 2, 3, 2].
    # Minimum cost = 2 + 8 + 1 + 2 + 3 + 2 = 18
    assert minCost(6, [[2, 4, 6], [5, 3, 8], [7, 1, 9], [4, 6, 2], [3, 5, 7], [8, 2, 4]]) == 18

def test_minimum_values():
    # Test minimum values: n = 2, costs = [[0, 0, 0], [0, 0, 0]]
    # The only valid painting sequence is [1, 2] or [2, 1], both costs = 0.
    assert minCost(2, [[0, 0, 0], [0, 0, 0]]) == 0

def test_maximum_values():
    # Test maximum values: n = 100000, costs with maximum values
    # All costs are 100000, since n is large, any sequence will be equally costly.
    # Total cost will be 100000 * 100000 / 2 = 5000000000 (with max alternation)
    costs = [[100000, 100000, 100000]] * 100000
    assert minCost(100000, costs) == 5000000000

def test_variety_costs():
    # Test with a variety of costs: n = 4, costs = [[1, 100, 1], [100, 1, 100], [1, 100, 1], [100, 1, 100]]
    # The optimal painting sequence is [1, 2, 1, 2] with costs = [1, 1, 1, 1].
    # Minimum cost = 1 + 1 + 1 + 1 = 4
    assert minCost(4, [[1, 100, 1], [100, 1, 100], [1, 100, 1], [100, 1, 100]]) == 4
