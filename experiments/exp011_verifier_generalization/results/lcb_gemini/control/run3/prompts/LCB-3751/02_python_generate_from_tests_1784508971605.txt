from typing import List

def maxFrequency(nums: List[int], k: int) -> int:
    # Count frequency of each number
    count = [0] * 51
    for num in nums:
        count[num] += 1

    max_freq = 0

    # Iterate over possible values to transform to k
    for x in range(1, 51):
        if x <= k:
            # Calculate how many values can be transformed to k
            total = 0
            for i in range(x, k + 1):
                total += count[i]
            # Update max frequency
            max_freq = max(max_freq, total)

    return max_freq
```
