from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a map to hold the number of coins in each bag position
    coin_map = {}

    # Fill the coin_map with the coins at each position based on the input segments
    for l, r, c in coins:
        for pos in range(l, r + 1):
            coin_map[pos] = coin_map.get(pos, 0) + c

    # Create a list of positions sorted and a corresponding list of coins
    positions = sorted(coin_map.keys())
    max_coins = 0

    # We will use a sliding window to calculate the maximum coins for k consecutive bags
    start = 0
    current_sum = 0

    for end in range(len(positions)):
        current_sum += coin_map[positions[end]]

        # When we have more than k positions, we slide the window
        if end - start + 1 > k:
            current_sum -= coin_map[positions[start]]
            start += 1

        # If we have exactly k positions, we check for maximum
        if end - start + 1 == k:
            max_coins = max(max_coins, current_sum)

    return max_coins

```
