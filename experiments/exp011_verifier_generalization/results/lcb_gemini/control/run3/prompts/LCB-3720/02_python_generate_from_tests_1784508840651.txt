from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    from collections import defaultdict, deque
    import sys

    def bfs(max_weight):
        graph = defaultdict(list)
        out_degree = defaultdict(int)
        for u, v, w in edges:
            if w <= max_weight:
                graph[v].append(u)
                out_degree[u] += 1
                if out_degree[u] > threshold:
                    return False
        queue = deque([0])
        reachable = set([0])
        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if neighbor not in reachable:
                    reachable.add(neighbor)
                    queue.append(neighbor)
        return len(reachable) == n

    left, right = 1, 10**6
    answer = -1
    while left <= right:
        mid = (left + right) // 2
        if bfs(mid):
            answer = mid
            right = mid - 1
        else:
            left = mid + 1
    return answer
```
