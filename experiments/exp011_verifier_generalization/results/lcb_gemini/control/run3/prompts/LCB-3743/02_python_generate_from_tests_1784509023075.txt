from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_times = []

    # Calculate free time before the first meeting
    if startTime[0] > 0:
        free_times.append(startTime[0])

    # Calculate free times between meetings
    for i in range(1, n):
        free_time = startTime[i] - endTime[i - 1]
        if free_time > 0:
            free_times.append(free_time)

    # Calculate free time after the last meeting
    if endTime[-1] < eventTime:
        free_times.append(eventTime - endTime[-1])

    # If we can reschedule meetings, we can combine the largest free times
    free_times.sort(reverse=True)
    return sum(free_times[:k]) if k <= len(free_times) else sum(free_times)
```
