from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_times = []

    # Calculate free time before the first meeting
    if startTime[0] > 0:
        free_times.append(startTime[0])

    # Calculate free times between meetings
    for i in range(1, n):
        free_time = startTime[i] - endTime[i - 1]
        if free_time > 0:
            free_times.append(free_time)

    # Calculate free time after the last meeting
    if endTime[-1] < eventTime:
        free_times.append(eventTime - endTime[-1])

    # If we can reschedule meetings, we can create new gaps
    # We need to consider the total free time after k meetings are rescheduled
    if k >= n:
        return eventTime - (endTime[-1] - startTime[0])

    # Store potential gaps we can create by rescheduling meetings
    potential_free_times = []
    for i in range(min(k, n)):  
        if i < n - 1:
            gap = startTime[i + 1] - endTime[i]
            if gap > 0:
                potential_free_times.append(gap)

    # Sort potential free times in descending order to maximize free time
    potential_free_times.sort(reverse=True)
    max_free_time = sum(potential_free_times[:k])

    return max_free_time + sum(free_times)

```
