from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n < 2:
        return n
    dp = [0] * n  # dp[i] will store the length of the longest subsequence ending with nums[i]
    max_length = 1

    for i in range(n):
        max_diff = 0  # variable to track the maximum length of subsequences for current index
        for j in range(i):
            diff = abs(nums[i] - nums[j])
            if dp[j] > 0 and (max_diff == 0 or diff <= max_diff):
                dp[i] = max(dp[i], dp[j] + 1)
                max_diff = diff
        if dp[i] == 0:
            dp[i] = 1  # for the case where no previous element contributes to the subsequence
        max_length = max(max_length, dp[i])

    return max_length
```
