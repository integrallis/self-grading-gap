```python
from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n < 2:
        return n
    dp = [[1] * (n + 1) for _ in range(n + 1)]  # dp[i][j] is the length of the longest subsequence ending at i with j as the last absolute difference
    max_length = 1

    for i in range(n):
        for j in range(i):
            diff = abs(nums[i] - nums[j])
            length = 2  # Starting length (includes nums[i] and nums[j])
            for k in range(j):
                if abs(nums[j] - nums[k]) >= diff:
                    length = max(length, dp[j][abs(nums[j] - nums[k])]+1)
            dp[i][diff] = length
            max_length = max(max_length, length)

    return max_length
```
