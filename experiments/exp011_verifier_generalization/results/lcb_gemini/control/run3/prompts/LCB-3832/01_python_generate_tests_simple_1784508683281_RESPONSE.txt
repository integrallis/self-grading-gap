from typing import List
from solution import minCosts

def test_canonical_example_1():
    # input: [5, 3, 4, 1, 3, 2]
    # cost to reach position 0: swap with person 0 (cost 5)
    # cost to reach position 1: swap with person 1 (cost 3)
    # cost to reach position 2: swap with person 1 (cost 3), then swap with person 2 for free
    # cost to reach position 3: swap with person 3 (cost 1)
    # cost to reach position 4: swap with person 3 (cost 1), then swap with person 4 for free
    # cost to reach position 5: swap with person 3 (cost 1), then swap with person 5 for free
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]

def test_canonical_example_2():

    # cost to reach position 1: swap with person 0 (cost 1)
    # cost to reach position 2: swap with person 0 (cost 1)
    # cost to reach position 3: swap with person 0 (cost 1)
