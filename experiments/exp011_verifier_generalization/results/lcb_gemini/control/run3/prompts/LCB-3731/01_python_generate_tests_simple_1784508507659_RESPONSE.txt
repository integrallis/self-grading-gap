from solution import subarraySum

def test_canonical_example_1():
    # spec: input: [2, 3, 1]
    # subarrays:
    # nums[0] = [2] -> sum = 2
    # nums[0 ... 1] = [2, 3] -> sum = 5
    # nums[1 ... 2] = [3, 1] -> sum = 4
    # total = 2 + 5 + 4 = 11
    assert subarraySum([2, 3, 1]) == 11

def test_canonical_example_2():
    # spec: input: [3, 1, 1, 2]
    # subarrays:
    # nums[0] = [3] -> sum = 3
    # nums[0 ... 1] = [3, 1] -> sum = 4
    # nums[1 ... 2] = [1, 1] -> sum = 2
    # nums[1 ... 3] = [1, 1, 2] -> sum = 4
    # total = 3 + 4 + 2 + 4 = 13
    assert subarraySum([3, 1, 1, 2]) == 13
