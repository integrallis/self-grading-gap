```python
from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    for i in range(1, k + 1):
        # Calculate contribution of each number as min and max in subsequences of length i
        for num in nums:
            # Each number can be min in C(n-1, i-1) ways (choose i-1 from the rest)
            min_contrib = num * comb(n - 1, i - 1) % MOD
            # Each number can be max in C(n-1, i-1) ways
            max_contrib = num * comb(n - 1, i - 1) % MOD
            total_sum = (total_sum + min_contrib + max_contrib) % MOD
    return total_sum

from math import comb
```
