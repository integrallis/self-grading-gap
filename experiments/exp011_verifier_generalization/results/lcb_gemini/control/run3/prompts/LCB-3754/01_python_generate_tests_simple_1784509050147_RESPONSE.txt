def test_canonical_example_1():
    # input: "NWSE", k = 1
    # Explanation: Change s[2] from 'S' to 'N'. The string becomes "NWNE".
    # Maximum Manhattan distance is 3.

def test_canonical_example_2():
    # input: "NSWWEW", k = 3
    # Explanation: Change s[1] from 'S' to 'N', and s[4] from 'E' to 'W'.
    # The string becomes "NNWWWW". Maximum Manhattan distance is 6.
    assert maxDistance("NSWWEW", 3) == 6
