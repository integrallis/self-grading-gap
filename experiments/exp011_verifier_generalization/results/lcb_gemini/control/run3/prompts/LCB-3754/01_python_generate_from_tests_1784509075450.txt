def maxDistance(s: str, k: int) -> int:
    # Count movements in each direction
    n_count = s.count('N')
    s_count = s.count('S')
    e_count = s.count('E')
    w_count = s.count('W')

    # Calculate the net movement in y and x directions

    net_x = e_count - w_count

    # Calculate the maximum distance from the origin
    max_distance = abs(net_x) + abs(net_y) + k

    # The maximum distance is limited by the total number of moves in any direction
    return max_distance
```
