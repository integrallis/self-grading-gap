## Implementation

```python
def calculateScore(s: str) -> int:
    score = 0
    marked = set()
    n = len(s)
    for i in range(n):
        if i in marked:
            continue
        mirror_char = chr(219 - ord(s[i]))  # mirror of s[i]
        for j in range(i - 1, -1, -1):
            if j in marked:
                continue
            if s[j] == mirror_char:
                marked.add(i)
                marked.add(j)
                score += i - j
                break
    return score

```
