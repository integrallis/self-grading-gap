from solution import calculateScore

def test_canonical_example_1():
    # For input "aczzx":
    # i = 0: no j found, skip
    # i = 1: no j found, skip
    # i = 2: j = 0 (a <-> z), score += 2 - 0 = 2
    # i = 3: no j found, skip
    # i = 4: j = 1 (c <-> x), score += 4 - 1 = 3
    # Total score = 2 + 3 = 5
    assert calculateScore("aczzx") == 5

def test_canonical_example_2():
    # For input "abcdef":
    # i = 0: no j found, skip
    # i = 1: no j found, skip
    # i = 2: no j found, skip
    # i = 3: no j found, skip
    # i = 4: no j found, skip
    # Total score = 0
    assert calculateScore("abcdef") == 0
