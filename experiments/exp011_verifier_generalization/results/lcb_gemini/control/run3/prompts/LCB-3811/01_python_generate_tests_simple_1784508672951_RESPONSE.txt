from solution import reverseDegree

def test_canonical_example_1():
    # Input: "abc"
    # 'a' = 26, position 1 -> 26 * 1 = 26
    # 'b' = 25, position 2 -> 25 * 2 = 50
    # 'c' = 24, position 3 -> 24 * 3 = 72
    # Total = 26 + 50 + 72 = 148
    assert reverseDegree("abc") == 148

def test_canonical_example_2():
    # Input: "zaza"
    # 'z' = 1, position 1 -> 1 * 1 = 1
    # 'a' = 26, position 2 -> 26 * 2 = 52
    # 'z' = 1, position 3 -> 1 * 3 = 3
    # 'a' = 26, position 4 -> 26 * 4 = 104
    # Total = 1 + 52 + 3 + 104 = 160
    assert reverseDegree("zaza") == 160
