from solution import hasSameDigits

def test_canonical_example_1():
    # spec: hasSameDigits("3902") -> True
    # Explanation:
    # Initial: "3902"
    # Operation 1: "292" (3 + 9 = 12 -> 2, 9 + 0 = 9, 0 + 2 = 2)
    # Operation 2: "11" (2 + 9 = 11 -> 1, 9 + 2 = 11 -> 1)
    # Final digits: '1' and '1' are the same
    assert hasSameDigits("3902") is True

def test_canonical_example_2():
    # spec: hasSameDigits("34789") -> False
    # Explanation:
    # Initial: "34789"
    # Operation 1: "7157" (3 + 4 = 7, 4 + 7 = 11 -> 1, 7 + 8 = 15 -> 5, 8 + 9 = 17 -> 7)
    # Operation 2: "862" (7 + 1 = 8, 1 + 5 = 6, 5 + 7 = 2)
    # Operation 3: "48" (8 + 6 = 14 -> 4, 6 + 2 = 8)
    # Final digits: '4' and '8' are not the same
