from solution import maxContainers

def test_canonical_example_1():
    # spec: maxContainers(2, 3, 15) -> 4
    # The deck has 4 cells (2x2), each container weighs 3. 
    # Maximum weight if all containers loaded = 4 * 3 = 12, which is <= 15.
    assert maxContainers(2, 3, 15) == 4

def test_canonical_example_2():
    # spec: maxContainers(3, 5, 20) -> 4
    # The deck has 9 cells (3x3), each container weighs 5. 
    # Maximum weight if all containers loaded = 9 * 5 = 45, but maxWeight = 20.
    # Maximum number of containers = 20 // 5 = 4.
    assert maxContainers(3, 5, 20) == 4

def test_non_divisible_weight_limit():
    # spec: maxContainers(3, 6, 20) -> 3
    # The deck has 9 cells (3x3), each container weighs 6. 
    # Maximum weight if all containers loaded = 9 * 6 = 54, but maxWeight = 20.
    # Maximum number of containers = 20 // 6 = 3.
    assert maxContainers(3, 6, 20) == 3

def test_insufficient_capacity():
    # spec: maxContainers(1, 2, 1) -> 0
    # The deck has 1 cell, but the container weighs 2, which exceeds maxWeight of 1.
    assert maxContainers(1, 2, 1) == 0

def test_minimum_boundaries():
    # spec: maxContainers(1, 1, 1) -> 1
    # The deck has 1 cell, and the container weighs 1, which is equal to maxWeight.
    assert maxContainers(1, 1, 1) == 1

def test_maximum_boundaries():
    # spec: maxContainers(1000, 1000, 10**9) -> 1_000_000
    # The deck has 1,000,000 cells (1000x1000), each container weighs 1000.
    # Maximum weight if all containers loaded = 1,000,000 * 1000 = 1,000,000,000, which is <= 10**9.
    assert maxContainers(1000, 1000, 10**9) == 1_000_000
