from typing import List
from solution import minCosts

def test_canonical_example_1():
    # Input: cost = [5, 3, 4, 1, 3, 2]
    # Explanation:
    # i = 0: cost[0] = 5
    # i = 1: cost[1] = 3
    # i = 2: swap with 1 for 3, then 2 for free -> 3
    # i = 3: cost[3] = 1
    # i = 4: swap with 3 for 1, then 4 for free -> 1
    # i = 5: swap with 3 for 1, then 5 for free -> 1
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]

def test_canonical_example_2():
    # Input: cost = [1, 2, 4, 6, 7]
    # Explanation:
    # i = 0: cost[0] = 1
    # i = 1: swap with 0 for 1 -> 1
    # i = 2: swap with 0 for 1 -> 1
    # i = 3: swap with 0 for 1 -> 1
    # i = 4: swap with 0 for 1 -> 1
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]

def test_single_element():
    # Input: cost = [100]
    # Explanation:
    # i = 0: cost[0] = 100
    assert minCosts([100]) == [100]

def test_maximum_length():
    # Input: cost = [100] * 100
    # Explanation:
    # For each position, the cost to swap with the first person remains 100
    assert minCosts([100] * 100) == [100] * 100
