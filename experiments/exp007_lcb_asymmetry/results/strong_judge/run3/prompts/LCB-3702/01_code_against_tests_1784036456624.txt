```python
from typing import List
import math


def maxLength(nums: List[int]) -> int:
    n = len(nums)
    max_len = 0
    for start in range(n):
        prod = 1
        g = nums[start]
        l = nums[start]
        for end in range(start, n):
            prod *= nums[end]
            g = math.gcd(g, nums[end])
            l = (l * nums[end]) // math.gcd(l, nums[end])
            if prod == g * l:
                max_len = max(max_len, end - start + 1)
    return max_len

```
