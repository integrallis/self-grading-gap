```python
from typing import List

def minimumPairRemoval(nums: List[int]) -> int:
    operations = 0
    while True:
        # Check if the array is non-decreasing
        if all(nums[i] <= nums[i + 1] for i in range(len(nums) - 1)):
            break
        # Find the pair with the minimum sum
        min_sum = float('inf')
        min_pair_index = -1
        for i in range(len(nums) - 1):
            pair_sum = nums[i] + nums[i + 1]
            if pair_sum < min_sum:
                min_sum = pair_sum
                min_pair_index = i
        # Merge the leftmost minimum pair
        nums[min_pair_index] += nums[min_pair_index + 1]
        del nums[min_pair_index + 1]
        operations += 1
    return operations

```
