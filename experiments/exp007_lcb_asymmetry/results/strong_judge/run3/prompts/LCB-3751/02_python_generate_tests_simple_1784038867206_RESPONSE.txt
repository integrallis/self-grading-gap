from typing import List
from solution import maxFrequency

def test_canonical_example_1():
    # spec: maxFrequency([1, 2, 3, 4, 5, 6], 1) -> 2
    # Explanation: After adding -5 to nums[2..5], 1 has a frequency of 2 in [1, 2, -2, -1, 0, 1].
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2

def test_canonical_example_2():
    # spec: maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) -> 4
    # Explanation: After adding 8 to nums[1..9], 10 has a frequency of 4 in [10, 10, 11, 12, 13, 13, 12, 11, 10, 10].
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4

def test_minimum_length():
    # spec: maxFrequency([1], 1) -> 1
    # Explanation: The single element is equal to k, so frequency is 1.
    assert maxFrequency([1], 1) == 1

def test_boundary_value():
    # spec: maxFrequency([50], 50) -> 1
    # Explanation: The single element is equal to k, so frequency is 1.
    assert maxFrequency([50], 50) == 1

def test_maximum_length():
    # spec: maxFrequency([1] * 100000, 1) -> 100000
    # Explanation: All elements are equal to k, so frequency is 100000.
    assert maxFrequency([1] * 100000, 1) == 100000

def test_already_equal_to_k():
    # spec: maxFrequency([5, 5, 5], 5) -> 3
    # Explanation: All elements are equal to k, so frequency is 3.
    assert maxFrequency([5, 5, 5], 5) == 3

def test_no_initial_k():
    # spec: maxFrequency([2, 2, 3, 2], 1) -> 3
    # Explanation: Need to convert 2s to 1 by adding -1 to nums[0..2] or similar.
    assert maxFrequency([2, 2, 3, 2], 1) == 3

def test_contiguity_tradeoff():
    # spec: maxFrequency([2, 1, 2], 1) -> 2
    # Explanation: Convert both 2s to 1 by adding -1 to nums[0..2], losing the existing 1.
    assert maxFrequency([2, 1, 2], 1) == 2
