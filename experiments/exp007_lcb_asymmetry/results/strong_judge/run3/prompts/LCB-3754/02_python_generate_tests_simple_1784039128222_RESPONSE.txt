from solution import maxDistance

def test_canonical_example_1():
    # input: "NWSE", k = 1
    # Explanation: Change s[2] from 'S' to 'N'. The string s becomes "NWNE".
    # Maximum Manhattan distance = 3
    assert maxDistance("NWSE", 1) == 3

def test_canonical_example_2():
    # input: "NSWWEW", k = 3
    # Explanation: Change s[1] from 'S' to 'N', and s[4] from 'E' to 'W'.
    # The string s becomes "NNWWWW".
    # Maximum Manhattan distance = 6
    assert maxDistance("NSWWEW", 3) == 6

def test_minimum_length_zero_edit():
    # input: "N", k = 0
    # No changes allowed, maximum Manhattan distance = 1
    assert maxDistance("N", 0) == 1

def test_full_edit_budget():
    # input: "NS", k = 2
    # Change both characters to maximize distance, maximum Manhattan distance = 2
    assert maxDistance("NS", 2) == 2

def test_maximum_length():
    # input: "NS" * 50000, k = 100000
    # All characters can be changed to reach maximum distance, maximum Manhattan distance = 100000
    assert maxDistance("NS" * 50000, 100000) == 100000
