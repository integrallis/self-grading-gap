You are given a 2D array coins, where coins[i] = [l_i, r_i, c_i] denotes that every bag from l_i to r_i contains c_i coins.

Output: 6
Explanation:
Selecting bags at positions [1, 2] gives the maximum number of coins: 3 + 3 = 6.

1 <= k <= 10^9
coins[i] == [l_i, r_i, c_i]
1 <= l_i <= r_i <= 10^9
1 <= c_i <= 1000
The given segments are non-overlapping.

```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    max_coins = 0
    n = len(coins)
    events = []

    # Create events for the start and end of each interval
    for l, r, c in coins:
        events.append((l, c))  # start of interval
        events.append((r + 1, -c))  # end of interval (r + 1 to make it exclusive)

    # Sort events by coordinate
    events.sort()
    current_coins = 0
    last_position = 0

    # Use a two-pointer approach to calculate maximum coins in a window of size k
    for position, change in events:
        # Calculate the length of the current segment
        if last_position < position:
            segment_length = position - last_position
            # If current segment is within the window size k
            if segment_length + current_coins > k:
                # Calculate how many full bags we can take
                excess = (segment_length + current_coins) - k
                current_coins -= excess
            # Update the maximum coins possible
            max_coins = max(max_coins, current_coins)
        # Update the current number of coins
        current_coins += change
        last_position = position

    # Final check in case the last segment contributes to the maximum
    max_coins = max(max_coins, current_coins)

    return max_coins
```
