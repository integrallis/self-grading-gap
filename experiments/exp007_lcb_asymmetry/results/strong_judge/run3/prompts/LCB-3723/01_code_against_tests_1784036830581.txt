## Implementation

```python
from typing import List

def sumOfGoodNumbers(nums: List[int], k: int) -> int:
    total = 0
    n = len(nums)
    for i in range(n):
        current = nums[i]
        left_index = i - k
        right_index = i + k
        if (left_index < 0 or current > nums[left_index]) and (right_index >= n or current > nums[right_index]):
            total += current
    return total

```
