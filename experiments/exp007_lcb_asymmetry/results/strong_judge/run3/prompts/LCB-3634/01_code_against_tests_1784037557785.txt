## Implementation

```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirror_map = {chr(i): chr(219 - i) for i in range(ord('a'), ord('z') + 1)}
    for i in range(len(s)):
        if marked[i]:
            continue
        mirror = mirror_map[s[i]]
        for j in range(i - 1, -1, -1):
            if s[j] == mirror and not marked[j]:
                score += i - j
                marked[i] = True
                marked[j] = True
                break
    return score
```
