from solution import calculateScore

def test_canonical_example_1():
    # Input: "aczzx"
    # Iteration 2: j=0 (mirror of 'c' is 'x'), score += 2 - 0 = 2
    # Iteration 4: j=1 (mirror of 'z' is 'a'), score += 4 - 1 = 3
    # Total score = 2 + 3 = 5
    assert calculateScore("aczzx") == 5

def test_canonical_example_2():
    # Input: "abcdef"
    # No pairs of indices (i, j) where s[j] is the mirror of s[i]
    # Total score = 0
    assert calculateScore("abcdef") == 0
