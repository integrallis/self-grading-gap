from solution import maxDifference

def test_canonical_example_1():
    # Example: input "aaaaabbc" -> character 'a' has an odd frequency of 5, character 'b' has an even frequency of 2.
    # Maximum difference = 5 - 2 = 3
    assert maxDifference("aaaaabbc") == 3

def test_canonical_example_2():
    # Example: input "abcabcab" -> character 'a' has an odd frequency of 3, character 'c' has an even frequency of 2.
    # Maximum difference = 3 - 2 = 1
    assert maxDifference("abcabcab") == 1

def test_minimum_length_input():
    # Input "aab" has character 'a' with even frequency 2 and character 'b' with odd frequency 1.
    # Maximum difference = 1 - 2 = -1
    assert maxDifference("aab") == -1

def test_maximum_length_input():
    # Input "a" * 97 + "b" * 2 + "c" has character 'a' with odd frequency 97, character 'b' with even frequency 2.
    # Maximum difference = 97 - 2 = 95
    assert maxDifference("a" * 97 + "b" * 2 + "c") == 95
