```python
from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    if n == 0:
        return eventTime
    
    # Calculate gaps between meetings
    gaps = [startTime[0]] + [startTime[i] - endTime[i - 1] for i in range(1, n)] + [eventTime - endTime[-1]]
    
    # Calculate the maximum sum of k + 1 consecutive gaps
    max_free_time = 0
    window = sum(gaps[:k + 1])
    max_free_time = window
    
    for i in range(k + 1, len(gaps)):
        window += gaps[i] - gaps[i - (k + 1)]
        max_free_time = max(max_free_time, window)
    
    return max_free_time
```
