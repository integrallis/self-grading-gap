from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_intervals = []
    last_end = 0

    # Calculate initial free intervals
    for i in range(n):
        if startTime[i] > last_end:
            free_intervals.append((last_end, startTime[i]))
        last_end = endTime[i]

    if last_end < eventTime:
        free_intervals.append((last_end, eventTime))

    # Calculate maximum free time without rescheduling
    max_free_time = 0
    for start, end in free_intervals:
        max_free_time = max(max_free_time, end - start)

    # If k meetings can be rescheduled, check for the potential increase in free time
    if k > 0:
        # We can try to push meetings to the left or right
        possible_gaps = []
        for i in range(n):
            # Moving the i-th meeting to the left
            if i > 0:
                new_start = startTime[i - 1] + (endTime[i] - startTime[i])
                if new_start < startTime[i]:
                    possible_gaps.append((new_start, startTime[i]))
            # Moving the i-th meeting to the right
            if i < n - 1:
                new_end = endTime[i + 1] - (endTime[i] - startTime[i])
                if new_end > endTime[i]:
                    possible_gaps.append((endTime[i], new_end))

        # Calculate possible maximum free time by considering rescheduling
        for start, end in possible_gaps:
            max_free_time = max(max_free_time, end - start)

    return max_free_time
