```python
from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    nums.sort()

    for i in range(n):
        # Each number nums[i] contributes as minimum in several subsequences
        min_contribution = nums[i] * pow(2, min(k, i), MOD) % MOD
        # Each number nums[i] contributes as maximum in several subsequences
        max_contribution = nums[i] * pow(2, min(k, n - 1 - i), MOD) % MOD
        total_sum = (total_sum + min_contribution + max_contribution) % MOD

    return total_sum % MOD
```
