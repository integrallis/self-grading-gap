from typing import List
from solution import minMaxSums

def test_canonical_example_1():
    # spec: nums = [1, 2, 3], k = 2
    # Subsequences: [1], [2], [3], [1, 2], [1, 3], [2, 3]
    # Sum = 2 + 4 + 6 + 3 + 4 + 5 = 24
    assert minMaxSums([1, 2, 3], 2) == 24

def test_canonical_example_2():
    # spec: nums = [5, 0, 6], k = 1
    # Subsequences: [5], [0], [6]
    # Sum = 5 + 5 + 0 + 0 + 6 + 6 = 22
    assert minMaxSums([5, 0, 6], 1) == 22

def test_canonical_example_3():
    # spec: nums = [1, 1, 1], k = 2
    # Subsequences: [1], [1], [1], [1, 1], [1, 1], [1, 1]
    # Each subsequence has min = 1 and max = 1
    # Total sum = 3 * 2 + 3 * 2 = 12
    assert minMaxSums([1, 1, 1], 2) == 12

def test_single_element():
    # Single element case: nums = [7], k = 1
    # Subsequences: [7]
    # Sum = 7 + 7 = 14
    assert minMaxSums([7], 1) == 14

def test_maximum_value_and_modulo():
    # Maximum value case: nums = [1_000_000_000], k = 1
    # Subsequences: [1_000_000_000]
    # Sum = 1_000_000_000 + 1_000_000_000 = 2_000_000_000
    # Result should be 2_000_000_000 % (10**9 + 7) = 999_999_993
    assert minMaxSums([1_000_000_000], 1) == 999_999_993

def test_k_equals_full_array_length():
    # Test where k equals the full array length: nums = [1, 2, 3], k = 3
    # Subsequences: [1], [2], [3], [1, 2], [1, 3], [2, 3], [1, 2, 3]
    # Sum = 2 + 4 + 6 + 3 + 4 + 5 + 6 = 30
    assert minMaxSums([1, 2, 3], 3) == 28

def test_k_70_boundary():
    # Test case where k = 70: nums = [1] * 70, k = 70
    # Each subsequence has min = 1 and max = 1
    # Total sum = 70 * (1 + 1) = 140
    assert minMaxSums([1] * 70, 70) == 540_032_504

def test_maximum_length_constraint():
    # Test case with maximum length: nums = [0] * 100_000, k = 70
    # All subsequences will have min = 0 and max = 0
    # Total sum = 0
    assert minMaxSums([0] * 100_000, 70) == 0
