from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    pow2 = pow(2, n, MOD) - 1  # Total non-empty subsequences

    for i in range(n):
        # For each number, calculate its contribution to total_sum
        # Contribution = element * (number of subsequences where it is the min or max)
        contribution = nums[i] * (pow(2, min(k, n - 1), MOD) * (pow2 - pow(2, n - k - 1, MOD)) % MOD) % MOD
        total_sum = (total_sum + contribution) % MOD

    return (total_sum * 2) % MOD

```
