from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Calculate the contribution of each element as min and max
    for i in range(n):
        # Calculate contribution for nums[i] being the minimum
        min_contribution = nums[i] * pow(2, min(k, n-1), MOD) % MOD
        # Calculate contribution for nums[i] being the maximum
        max_contribution = nums[i] * pow(2, min(k, n-1), MOD) % MOD
        total_sum = (total_sum + min_contribution + max_contribution) % MOD

    return total_sum % MOD
```
