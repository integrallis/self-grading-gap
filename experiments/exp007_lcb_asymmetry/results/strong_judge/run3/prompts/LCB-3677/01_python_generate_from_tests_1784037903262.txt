from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[-float('inf')] * (n + 1) for _ in range(m + 1)]
    dp[0][0] = coins[0][0]  # Starting point

    for i in range(m):
        for j in range(n):
            if i == 0 and j == 0:
                continue
            gain = coins[i][j]
            if gain < 0:
                gain = 0  # Neutralized robber

            # Normal path values
            from_top = dp[i - 1][j] if i > 0 else -float('inf')
            from_left = dp[i][j - 1] if j > 0 else -float('inf')
            dp[i][j] = max(dp[i][j], from_top + gain, from_left + gain)

            # Handling robber neutralizations
            if coins[i][j] < 0:
                # Neutralize this robber
                if i > 0:
                    dp[i][j] = max(dp[i][j], dp[i - 1][j] + (0 if gain == 0 else gain))
                if j > 0:
                    dp[i][j] = max(dp[i][j], dp[i][j - 1] + (0 if gain == 0 else gain))

            # Check possible neutralization of previous robbers
            if i > 0 and j > 0:
                if coins[i - 1][j] < 0:
                    dp[i][j] = max(dp[i][j], dp[i - 1][j] + gain - coins[i - 1][j])
                if coins[i][j - 1] < 0:
                    dp[i][j] = max(dp[i][j], dp[i][j - 1] + gain - coins[i][j - 1])

    return max(dp[m - 1][n - 1], 0)  # Ensure we do not return negative profits

```
