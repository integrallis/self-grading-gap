from typing import List
from solution import maximumAmount

def test_canonical_example_1():
    # Example: input '[[0, 1, -1], [1, -2, 3], [2, -3, 4]]' -> output '8'
    assert maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) == 8

def test_canonical_example_2():
    # Example: input '[[10, 10, 10], [10, 10, 10]]' -> output '40'
    assert maximumAmount([[10, 10, 10], [10, 10, 10]]) == 40

def test_two_neutralizations():
    # Input requires using both neutralizations: 
    # Start at (0, 0) with 5 coins, neutralize -7, neutralize -11, gain 4
    assert maximumAmount([[5, -7, -11, 4]]) == 9  # 5 + 0 + 0 + 4 = 9

def test_negative_after_neutralizations():
    # Input that remains negative after using both neutralizations:
    # Start at (0, 0) with -1, neutralize -2, neutralize -3
    assert maximumAmount([[-1, -2, -3, -4]]) == -3  # -1 + 0 + 0 = -1

def test_neutralize_starting_robber():
    # Neutralizing a robber at the starting cell
    assert maximumAmount([[-5, 1]]) == 1  # Neutralize -5, gain 1 => 1
    # Neutralizing a robber at the ending cell
    assert maximumAmount([[1, -5]]) == 1  # Gain 1, neutralize -5 => 1

def test_single_cell():
    # Minimum grid size 1x1
    assert maximumAmount([[-1000]]) == 0  # Only cell is a robber, neutralize it => 0

def test_single_column():
    # m x 1 boundary test
    assert maximumAmount([[-1], [-2], [-3]]) == -1  # Neutralize -1 => -1

def test_maximum_dimension():
    # Maximum dimension test with all cells having the maximum value
    assert maximumAmount([[1000] * 500 for _ in range(500)]) == 999000000  # 500*500 - 1 = 999000
