```python
from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[-float('inf')] * (n + 1) for _ in range(m + 1)]
    dp[0][1] = 0  # Starting point, coins[0][0] considered separately

    for i in range(m):
        for j in range(n):
            if i == 0 and j == 0:
                continue
            gain = coins[i][j]
            if gain < 0:
                gain = 0  # Neutralized robber

            # Normal path values
            dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j + 1] + gain)
            dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i + 1][j] + gain)

            # Handling robber neutralizations
            if coins[i][j] < 0:
                # Neutralize this robber
                dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j + 1] + gain + (0 if j == 0 else dp[i + 1][j - 1]))
                dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i + 1][j] + gain + (0 if i == 0 else dp[i - 1][j]))

    return max(dp[m][n], 0)  # Ensure we do not return negative profits
```
