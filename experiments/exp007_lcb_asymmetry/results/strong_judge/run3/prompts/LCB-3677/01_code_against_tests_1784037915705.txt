from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    # Create a 3D DP table to keep track of maximum coins with 0, 1, or 2 neutralizations
    dp = [[[-float('inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0] if coins[0][0] >= 0 else 0  # Starting point

    for i in range(m):
        for j in range(n):
            for k in range(3):  # Iterate over neutralizations used
                if i == 0 and j == 0 and k == 0:
                    continue
                if i > 0:
                    base_coins = dp[i-1][j][k]  # Coming from the top
                    if coins[i][j] < 0 and k > 0:
                        dp[i][j][k] = max(dp[i][j][k], base_coins - abs(coins[i][j]))  # Neutralize robber
                    elif coins[i][j] >= 0:
                        dp[i][j][k] = max(dp[i][j][k], base_coins + coins[i][j])  # Gain coins
                    else:
                        dp[i][j][k] = max(dp[i][j][k], base_coins)  # Robber without neutralization
                    if coins[i][j] < 0 and k < 2:
                        dp[i][j][k + 1] = max(dp[i][j][k + 1], base_coins)  # Use neutralization
                if j > 0:
                    base_coins = dp[i][j-1][k]  # Coming from the left
                    if coins[i][j] < 0 and k > 0:
                        dp[i][j][k] = max(dp[i][j][k], base_coins - abs(coins[i][j]))  # Neutralize robber
                    elif coins[i][j] >= 0:
                        dp[i][j][k] = max(dp[i][j][k], base_coins + coins[i][j])  # Gain coins
                    else:
                        dp[i][j][k] = max(dp[i][j][k], base_coins)  # Robber without neutralization
                    if coins[i][j] < 0 and k < 2:
                        dp[i][j][k + 1] = max(dp[i][j][k + 1], base_coins)  # Use neutralization

    return max(max(dp[m-1][n-1]), 0)  # Ensure we do not return negative profits
