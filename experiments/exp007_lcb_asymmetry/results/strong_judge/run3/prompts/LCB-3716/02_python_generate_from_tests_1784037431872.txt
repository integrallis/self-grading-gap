from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n == 0:
        return 0

    # dp[i] will hold the length of the longest subsequence ending at index i
    dp = [1] * n
    max_length = 1

    # Iterate over each pair (i, j) such that j < i
    for i in range(n):
        for j in range(i):
            if abs(nums[i] - nums[j]) >= (max_length - 1):
                dp[i] = max(dp[i], dp[j] + 1)
                max_length = max(max_length, dp[i])

    return max_length
```
