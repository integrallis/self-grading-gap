from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    dp = [[0] * 3 for _ in range(n)]
    dp[0][0] = cost[0][0]
    dp[0][1] = cost[0][1]
    dp[0][2] = cost[0][2]

    for i in range(1, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(
                dp[i-1][(j+1) % 3],  # Previous house different color
                dp[i-1][(j+2) % 3]   # Previous house different color
            )

    total_min_cost = min(dp[n-1][0], dp[n-1][1], dp[n-1][2])

    # Check equidistant house constraint
    for j in range(3):
        dp[n//2][j] = float('inf')  # Disallow painting the n/2 house with color j

    return min(total_min_cost, min(dp[n-1][0], dp[n-1][1], dp[n-1][2]))

```
