```python
from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize the first house costs
    r, g, b = cost[0]

    # Iterate through each house from 1 to n-1
    for i in range(1, n):
        r_old, g_old, b_old = r, g, b  # Store previous costs
        r = cost[i][0] + min(g_old, b_old)  # Current red cost
        g = cost[i][1] + min(r_old, b_old)  # Current green cost
        b = cost[i][2] + min(r_old, g_old)  # Current blue cost

    # Return the minimum cost
    return min(r, g, b)
```
