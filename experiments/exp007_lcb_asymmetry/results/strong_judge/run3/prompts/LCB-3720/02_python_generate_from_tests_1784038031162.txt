from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    from collections import defaultdict, deque
    import sys

    # Create a graph based on the edges
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))

    # Function to check if we can reach node 0 with the max weight limit
    def canReach(maxWeight):
        out_edges = defaultdict(int)
        # Create a new graph with edges that are within the max weight limit
        new_graph = defaultdict(list)
        for u in graph:
            for v, w in graph[u]:
                if w <= maxWeight:
                    new_graph[u].append((v, w))
                    out_edges[u] += 1
                    # Ensure we do not exceed the threshold
                    if out_edges[u] > threshold:
                        return False

        # Perform BFS or DFS to check reachability to node 0
        visited = [False] * n
        queue = deque([0])
        visited[0] = True
        while queue:
            curr = queue.popleft()
            for neighbor, _ in new_graph[curr]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    queue.append(neighbor)

        # Check if all nodes are reachable
        return all(visited[i] or out_edges[i] == 0 for i in range(n))

    # Binary search on the edge weights
    low, high = 1, 10**6
    result = -1
    while low <= high:
        mid = (low + high) // 2
        if canReach(mid):
            result = mid
            high = mid - 1
        else:
            low = mid + 1

    return result
```
