from solution import minMaxWeight

def test_canonical_example_1():
    # input: n = 5, edges = [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]], threshold = 2
    # After removing edge 2 -> 0, the max weight is 1
    assert minMaxWeight(5, [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]], 2) == 1

def test_canonical_example_2():
    # input: n = 5, edges = [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]], threshold = 1
    # Impossible to reach node 0 from node 2, so return -1
    assert minMaxWeight(5, [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]], 1) == -1

def test_canonical_example_3():
    # input: n = 5, edges = [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]], threshold = 1
    # Remove edges 1 -> 3 and 1 -> 4, max weight among remaining is 2
    assert minMaxWeight(5, [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]], 1) == 2

def test_canonical_example_4():
    # input: n = 5, edges = [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]], threshold = 1
    # Impossible to reach node 0 from node 3 as there are no valid edges, so return -1
    assert minMaxWeight(5, [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]], 1) == -1

def test_no_removal():
    # input: n = 2, edges = [[1, 0, 1]], threshold = 1
    # No edges need to be removed; max weight is 1
    assert minMaxWeight(2, [[1, 0, 1]], 1) == 1

def test_maximum_weight():
    # input: n = 2, edges = [[1, 0, 1_000_000]], threshold = 1
    # No edges need to be removed; max weight is 1_000_000
    assert minMaxWeight(2, [[1, 0, 1_000_000]], 1) == 1_000_000

def test_parallel_edges():
    # input: n = 3, edges = [[1, 0, 1], [1, 0, 5], [2, 1, 2]], threshold = 1
    # Retain edge 2 -> 1 and one of the edges 1 -> 0 (either works); max weight is 2
    assert minMaxWeight(3, [[1, 0, 1], [1, 0, 5], [2, 1, 2]], 1) == 2

def test_maximum_size_boundary():
    # input: n = 100000, threshold = 99999, edges = [[i, 0, 1] for i in range(1, 100000)] + [[1, 0, 2]]
    # We can retain edges from every node to 0 with weight 1, and one edge 1 -> 0 with weight 2; max weight is 1
    n = 100000
    edges = [[i, 0, 1] for i in range(1, n)] + [[1, 0, 2]]
    assert minMaxWeight(n, edges, n - 1) == 1
