from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    from collections import defaultdict, deque

    # Create a graph based on the edges
to_graph = defaultdict(list)
    for u, v, w in edges:
        to_graph[v].append((u, w))  # reverse graph to check reachability to node 0

    # Function to check if all nodes can reach node 0 with the max weight limit
    def canReach(maxWeight):
        new_graph = defaultdict(list)
        out_edges = [0] * n
        # Create a new graph with edges that are within the max weight limit
        for u, v, w in edges:
            if w <= maxWeight:
                new_graph[v].append((u, w))  # reversed direction 
                out_edges[u] += 1
                # Ensure we do not exceed the threshold
                if out_edges[u] > threshold:
                    return False

        # Perform BFS to check reachability to node 0
        visited = [False] * n
        queue = deque([0])
        visited[0] = True
        while queue:
            curr = queue.popleft()
            for neighbor, _ in new_graph[curr]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    queue.append(neighbor)

        # Check if all nodes can reach node 0
        for node in range(n):
            if not visited[node] and out_edges[node] > 0:
                return False
        return True

    # Binary search on the edge weights
    low, high = 1, 10**6
    result = -1
    while low <= high:
        mid = (low + high) // 2
        if canReach(mid):
            result = mid
            high = mid - 1
        else:
            low = mid + 1

    return result
