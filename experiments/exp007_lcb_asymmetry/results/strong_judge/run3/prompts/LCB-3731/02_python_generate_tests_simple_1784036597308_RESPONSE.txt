def test_maximum_length():
    # spec: input [1] * 100 (100 elements of value 1)
    # subarray sums: [1], [1, 1], ..., [1, 1, ..., 1] (up to 100 elements)
    # total sum calculations: 1 + 2 + 3 + ... + 100 = 5050
    # Each index contributes: 
    # index 0 -> 1, index 1 -> 2, index 2 -> 3, ..., index 99 -> 100
    # total = 1 + 2 + ... + 100 = (100 * (100 + 1)) / 2 = 5050
    assert subarraySum([1] * 100) == 5050
