from solution import calculateScore

def test_canonical_example_1():
    # spec: calculateScore("aczzx") -> 5
    # Explanation:
    # i = 0. No j found, skip.
    # i = 1. No j found, skip.
    # i = 2. j = 0 (mirror of 'z' is 'a'), score += 2 - 0 = 2.
    # i = 3. No j found, skip.
    # i = 4. j = 1 (mirror of 'x' is 'c'), score += 4 - 1 = 3.
    # Total score = 2 + 3 = 5
    assert calculateScore("aczzx") == 5

def test_canonical_example_2():
    # spec: calculateScore("abcdef") -> 0
    # Explanation:
    # For each index i, there is no index j that satisfies the conditions.
    # Total score = 0
    assert calculateScore("abcdef") == 0

def test_closest_unmarked_candidate():
    # spec: calculateScore("aaz") -> 1
    # Explanation:
    # i = 0. No j found, skip.
    # i = 1. j = 0 (mirror of 'a' is 'z'), score += 1 - 0 = 1.
    # i = 2. No j found, skip.
    # Total score = 1
    assert calculateScore("aaz") == 1

def test_marking_indices():
    # spec: calculateScore("aza") -> 1
    # Explanation:
    # i = 0. No j found, skip.
    # i = 1. j = 0 (mirror of 'a' is 'z'), score += 1 - 0 = 1.
    # i = 2. No j found (index 1 is marked), skip.
    # Total score = 1
    assert calculateScore("aza") == 1

def test_no_future_mirror():
    # spec: calculateScore("za") -> 0
    # Explanation:
    # i = 0. No j found, skip.
    # i = 1. No j found (mirror of 'a' is 'z' but j must be < i), skip.
    # Total score = 0
    assert calculateScore("za") == 0

def test_minimum_length():
    # spec: calculateScore("a") -> 0
    # Explanation:
    # There is only one character, so no j can be found.
    # Total score = 0
    assert calculateScore("a") == 0

def test_maximum_length():
    # spec: calculateScore("a" * 50000 + "z" * 50000) -> 2500000000
    # Explanation:
    # Each 'a' at indices 0 to 49999 will match with 'z' at indices 50000 to 99999.
    # Total score = (50000 - 0) + (50001 - 1) + ... + (99999 - 49999) = 50000 * 50000 = 2500000000
    assert calculateScore("a" * 50000 + "z" * 50000) == 2500000000
