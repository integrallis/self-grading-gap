```python
def calculateScore(s: str) -> int:
    score = 0
    stacks = [[] for _ in range(26)]  # 26 stacks for each letter
    for i in range(len(s)):
        char_index = ord(s[i]) - ord('a')
        mirror_index = 25 - char_index  # Calculate mirror index
        if char_index < 13:
            stacks[char_index].append(i)  # Only store lower-half letters
        elif stacks[mirror_index]:
            j = stacks[mirror_index].pop()  # Get the closest unmarked index for the mirror
            score += i - j  # Update score
    return score

```
