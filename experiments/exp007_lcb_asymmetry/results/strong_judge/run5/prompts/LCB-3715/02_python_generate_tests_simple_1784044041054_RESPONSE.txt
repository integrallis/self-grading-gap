from typing import List
from solution import maximumCoins

def test_canonical_example_1():
    # coins = [[8,10,1],[1,3,2],[5,6,4]], k = 4
    # Selecting bags at positions [3, 4, 5, 6] gives:
    # 2 (from bag 3) + 0 (from bag 4) + 4 (from bag 5) + 4 (from bag 6) = 10
    assert maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) == 10

def test_canonical_example_2():
    # coins = [[1,10,3]], k = 2
    # Selecting bags at positions [1, 2] gives:
    # 3 (from bag 1) + 3 (from bag 2) = 6
    assert maximumCoins([[1, 10, 3]], 2) == 6

def test_k_equals_1():
    # coins = [[5, 5, 7], [7, 7, 9]], k = 1
    # Selecting bag at position 7 gives:
    # 9 (from bag 7) = 9
    assert maximumCoins([[5, 5, 7], [7, 7, 9]], 1) == 9

def test_adjacent_segments():
    # coins = [[1, 2, 2], [3, 4, 5]], k = 3
    # Selecting bags at positions [1, 2, 3] gives:
    # 2 (from bag 1) + 2 (from bag 2) + 5 (from bag 3) = 12
    assert maximumCoins([[1, 2, 2], [3, 4, 5]], 3) == 12

def test_k_exceeds_total_covered_span():
    # coins = [[10, 11, 4]], k = 5
    # Selecting bags at positions [10, 11, 12, 13, 14] gives:
    # 4 (from bags 10 and 11) + 0 + 0 + 0 = 8
    assert maximumCoins([[10, 11, 4]], 5) == 8

def test_boundary_values():
    # coins = [[1_000_000_000, 1_000_000_000, 1000]], k = 1_000_000_000
    # Selecting bags at position 1_000_000_000 gives:
    # 1000 = 1000
    assert maximumCoins([[1_000_000_000, 1_000_000_000, 1000]], 1_000_000_000) == 1000

def test_large_input():
    # Generating 100,000 valid non-overlapping segments.
    coins = [[i, i + 1, 1] for i in range(1, 200_001, 2)]
    # k = 100,000, should give us 100,000 coins, since each segment gives 1 coin.
    assert maximumCoins(coins, 100_000) == 100_000
