```python
from typing import List

def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a map to hold the number of coins at each position
    coin_map = {}
    for l, r, c in coins:
        for pos in range(l, r + 1):
            coin_map[pos] = coin_map.get(pos, 0) + c

    # Create a list of positions and their corresponding coin values
    positions = sorted(coin_map.keys())
    coin_values = [coin_map[pos] for pos in positions]

    # Now we need to find the maximum coins we can get from k consecutive bags
    max_coins = 0
    n = len(positions)

    # Using a sliding window technique to calculate the number of coins in k consecutive bags
    left = 0
    current_sum = 0

    for right in range(n):
        current_sum += coin_values[right]
        while positions[right] - positions[left] + 1 > k:
            current_sum -= coin_values[left]
            left += 1
        max_coins = max(max_coins, current_sum)

    return max_coins
```
