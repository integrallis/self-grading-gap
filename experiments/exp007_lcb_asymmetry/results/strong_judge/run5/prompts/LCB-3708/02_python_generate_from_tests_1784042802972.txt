from typing import List

def zigzagTraversal(grid: List[List[int]]) -> List[int]:
    result = []
    for i in range(len(grid)):
        if i % 2 == 0:
            # Even row index: traverse left to right
            result.extend(grid[i][j] for j in range(len(grid[i])) if j % 2 == 0)
        else:
            # Odd row index: traverse right to left
            result.extend(grid[i][j] for j in range(len(grid[i]) - 1, -1, -1) if (len(grid[i]) - 1 - j) % 2 == 0)
    return result
