## Implementation

```python
from typing import List

MOD = 10**9 + 7

# Precompute factorials and inverse factorials
MAX_N = 100000
MAX_K = 70
factorials = [1] * (MAX_N + 1)
inv_factorials = [1] * (MAX_N + 1)

for i in range(2, MAX_N + 1):
    factorials[i] = factorials[i - 1] * i % MOD

inv_factorials[MAX_N] = pow(factorials[MAX_N], MOD - 2, MOD)
for i in range(MAX_N - 1, 0, -1):
    inv_factorials[i] = inv_factorials[i + 1] * (i + 1) % MOD

def binomial(n: int, k: int) -> int:
    if k < 0 or k > n:
        return 0
    return factorials[n] * inv_factorials[k] % MOD * inv_factorials[n - k] % MOD

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    nums.sort()
    total = 0

    for i in range(n):
        num = nums[i]
        min_count = sum(binomial(n - i - 1, j) for j in range(k)) % MOD
        max_count = sum(binomial(i, j) for j in range(k)) % MOD

        total += num * (min_count + max_count) % MOD
        total %= MOD

    return total % MOD
```
