from typing import List
from solution import minMaxSums

def test_canonical_example_1():
    # spec: nums = [1, 2, 3], k = 2
    # Subsequences: [1], [2], [3], [1, 2], [1, 3], [2, 3]
    # Min/Max sums: 2 + 4 + 6 + 3 + 4 + 5 = 24
    assert minMaxSums([1, 2, 3], 2) == 24

def test_canonical_example_2():
    # spec: nums = [5, 0, 6], k = 1
    # Subsequences: [5], [0], [6]
    # Min/Max sums: 5 + 5 + 0 + 0 + 6 + 6 = 22
    assert minMaxSums([5, 0, 6], 1) == 22

def test_canonical_example_3():
    # spec: nums = [1, 1, 1], k = 2
    # Subsequences: [1], [1], [1], [1, 1], [1, 1], [1, 1]
    # Min/Max sums: 12 (each subsequence contributes 2)
    assert minMaxSums([1, 1, 1], 2) == 12

def test_singleton_maximum_value():
    # spec: nums = [1_000_000_000], k = 1
    # Subsequence: [1_000_000_000]
    # Min/Max sums: 1_000_000_000 + 1_000_000_000 = 2_000_000_000
    # Output modulo 10^9 + 7: 2_000_000_000 % 1_000_000_007 = 999_999_993
    assert minMaxSums([1_000_000_000], 1) == 999_999_993

def test_maximum_k():
    # spec: nums = [1] * 70, k = 70
    # Every non-empty subsequence contributes 2
    # Total contributions: 2 * (2^70 - 1) modulo 10^9 + 7
    total = 2 * (pow(2, 70) - 1) % 1_000_000_007
    assert minMaxSums([1] * 70, 70) == total

def test_maximum_array_length():
    # spec: nums = [0] * 100_000, k = 70
    # All values are zero, so the total is 0
    assert minMaxSums([0] * 100_000, 70) == 0
