```python
from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total = 0
    # For each element, calculate its contribution
    for i, num in enumerate(nums):
        # Calculate the number of subsequences including this element
        # It can be the min or max in subsequences of size 1 to k
        # The count of combinations of picking other elements
        min_count = (pow(2, min(n - 1, k - 1), MOD) - 1) % MOD
        max_count = (pow(2, min(n - 1, k - 1), MOD) - 1) % MOD
        total += (num * (min_count + max_count)) % MOD
        total %= MOD
    return total
```
