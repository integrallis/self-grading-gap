from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total = 0

    # Calculate the contribution of each number as min and max
    for i in range(n):
        num = nums[i]
        min_count = pow(2, min(k - 1, n - 1), MOD)
        max_count = pow(2, min(k - 1, n - 1), MOD)

        # Each number contributes its value times the number of subsequences it can be min or max
        total += num * (min_count + max_count) % MOD
        total %= MOD

    return total % MOD
