from typing import List
import heapq


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Prepare a graph from the edges
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[v].append((u, w))  # Reverse the edges for reachability check

    def canReachNodeZero(max_weight):
        # Use BFS or DFS to check if all nodes can reach node 0
        visited = [False] * n
        queue = []
        for u in range(n):
            for v, w in graph[u]:
                if w <= max_weight:
                    queue.append(u)
                    visited[u] = True
                    break
        while queue:
            node = queue.pop()
            for v, w in graph[node]:
                if not visited[v] and w <= max_weight:
                    visited[v] = True
                    queue.append(v)
        return all(visited)

    # Binary search over the possible maximum weights
    left, right = 1, 10**6
    result = -1
    while left <= right:
        mid = (left + right) // 2
        if canReachNodeZero(mid):
            result = mid
            right = mid - 1
        else:
            left = mid + 1

    return result

```
