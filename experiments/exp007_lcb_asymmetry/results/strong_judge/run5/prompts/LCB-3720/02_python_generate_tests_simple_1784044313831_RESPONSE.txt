from typing import List
from solution import minMaxWeight

def test_canonical_example_1():
    # spec: minMaxWeight(5, [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]], 2) -> 1
    assert minMaxWeight(5, [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]], 2) == 1

def test_canonical_example_2():
    # spec: minMaxWeight(5, [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]], 1) -> -1
    assert minMaxWeight(5, [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]], 1) == -1

def test_canonical_example_3():
    # spec: minMaxWeight(5, [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]], 1) -> 2
    assert minMaxWeight(5, [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]], 1) == 2

def test_canonical_example_4():
    # spec: minMaxWeight(5, [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]], 1) -> -1
    assert minMaxWeight(5, [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]], 1) == -1

def test_no_removal_needed():
    # No edges need to be removed: 1->0 and 2->0, max weight is 4
    assert minMaxWeight(3, [[1, 0, 4], [2, 0, 2]], 2) == 4

def test_parallel_edges():
    # Multiple edges from 1 to 0, keep the one with weight 2 as max weight
    assert minMaxWeight(3, [[1, 0, 5], [1, 0, 1], [2, 0, 2]], 1) == 2

def test_minimum_size_and_numeric_boundary():
    # Minimum size graph with max weight 1_000_000
    assert minMaxWeight(2, [[1, 0, 1_000_000]], 1) == 1_000_000

def test_constraint_scale_chain():
    # Large chain of nodes with weight 1
    n = 100000
    edges = [[i, i - 1, 1] for i in range(1, n)]
    assert minMaxWeight(n, edges, n - 1) == 1
