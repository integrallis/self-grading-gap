from typing import List
from solution import maxSum

def test_canonical_example_1():
    # spec: maxSum([1, 2, 3, 4, 5]) -> 15 (select entire array)
    assert maxSum([1, 2, 3, 4, 5]) == 15

def test_canonical_example_2():
    # spec: maxSum([1, 1, 0, 1, 1]) -> 1 (delete all but one 1 to maximize unique sum)
    assert maxSum([1, 1, 0, 1, 1]) == 1

def test_canonical_example_3():
    # spec: maxSum([1, 2, -1, -2, 1, 0, -1]) -> 3 (delete -1 and -2, select [1, 2])
    assert maxSum([1, 2, -1, -2, 1, 0, -1]) == 3

def test_single_negative_value_must_be_selected():
    # spec: maxSum([-100]) -> -100 (only one value, must be selected)
    assert maxSum([-100]) == -100

def test_upper_value_boundary():
    # spec: maxSum([100]) -> 100 (only one value, must be selected)
    assert maxSum([100]) == 100

def test_maximum_length_boundary():
    # spec: maxSum([1] * 100) -> 1 (all are the same, select one)
    assert maxSum([1] * 100) == 1

def test_all_nonpositive_values_choose_best_nonempty_unique_selection():
    # spec: maxSum([-3, -1, -2, -1, 0]) -> 0 (select 0 as the best nonnegative unique)
    assert maxSum([-3, -1, -2, -1, 0]) == 0
