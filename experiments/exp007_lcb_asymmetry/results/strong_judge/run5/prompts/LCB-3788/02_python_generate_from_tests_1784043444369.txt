from typing import List

def maxSum(nums: List[int]) -> int:
    unique_nums = set()  # To track unique elements
    total_sum = 0
    for num in nums:
        if num not in unique_nums:  # If the number is unique
            unique_nums.add(num)  # Add it to the set
            total_sum += num  # Increase the total sum
    return total_sum
```
