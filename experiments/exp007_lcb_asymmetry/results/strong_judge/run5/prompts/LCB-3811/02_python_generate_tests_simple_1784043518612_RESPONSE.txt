from solution import reverseDegree

def test_canonical_example_1():
    # Input: "abc"
    # Reverse degree calculation:
    # 'a': 26 * 1 = 26
    # 'b': 25 * 2 = 50
    # 'c': 24 * 3 = 72
    # Total: 26 + 50 + 72 = 148
    assert reverseDegree("abc") == 148

def test_canonical_example_2():
    # Input: "zaza"
    # Reverse degree calculation:
    # 'z': 1 * 1 = 1
    # 'a': 26 * 2 = 52
    # 'z': 1 * 3 = 3
    # 'a': 26 * 4 = 104
    # Total: 1 + 52 + 3 + 104 = 160
    assert reverseDegree("zaza") == 160

def test_minimum_length():
    # Input: "z"
    # Reverse degree calculation:
    # 'z': 1 * 1 = 1
    # Total: 1 = 1
    assert reverseDegree("z") == 1

def test_maximum_length():
    # Input: "a" * 1000
    # Reverse degree calculation:
    # 'a': 26 * i for i from 1 to 1000
    # Total: 26 * (1 + 2 + ... + 1000) = 26 * (1000 * 1001 / 2)
    # Total: 26 * 500500 = 13013000
    assert reverseDegree("a" * 1000) == 13_013_000
