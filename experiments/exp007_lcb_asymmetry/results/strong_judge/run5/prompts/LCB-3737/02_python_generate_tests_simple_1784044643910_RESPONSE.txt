from typing import List
from solution import minCost

def test_canonical_example_1():
    # Example: input '4\n[[3, 5, 7], [6, 2, 9], [4, 8, 1], [7, 3, 5]]' 
    # The optimal painting sequence is [1, 2, 3, 2] with costs [3, 2, 1, 3].
    # Minimum cost is 3 + 2 + 1 + 3 = 9
    assert minCost(4, [[3, 5, 7], [6, 2, 9], [4, 8, 1], [7, 3, 5]]) == 9

def test_canonical_example_2():
    # Example: input '6\n[[2, 4, 6], [5, 3, 8], [7, 1, 9], [4, 6, 2], [3, 5, 7], [8, 2, 4]]' 
    # The optimal painting sequence is [1, 3, 2, 3, 1, 2] with costs [2, 8, 1, 2, 3, 2].
    # Minimum cost is 2 + 8 + 1 + 2 + 3 + 2 = 18
    assert minCost(6, [[2, 4, 6], [5, 3, 8], [7, 1, 9], [4, 6, 2], [3, 5, 7], [8, 2, 4]]) == 18

def test_minimum_size():
    # Test minimum size case: input '2\n[[0, 0, 0], [0, 0, 0]]' 
    # The optimal painting sequence is [1, 2] or [2, 1] with costs both 0.
    assert minCost(2, [[0, 0, 0], [0, 0, 0]]) == 0

def test_upper_cost():
    # Test upper cost case: input '2\n[[100000, 100000, 100000], [100000, 100000, 100000]]' 
    # The optimal painting sequence is [1, 2] or [2, 1] with costs both 200000.
    assert minCost(2, [[100000, 100000, 100000], [100000, 100000, 100000]]) == 200000

def test_maximum_size():
    # Test maximum size case: input '100000' with uniform costs.
    # All costs are 100000, so the minimum cost is 100000 * (n // 2) = 100000 * 50000 = 10_000_000_000.
    assert minCost(100000, [[100000, 100000, 100000] for _ in range(100000)]) == 10_000_000_000
