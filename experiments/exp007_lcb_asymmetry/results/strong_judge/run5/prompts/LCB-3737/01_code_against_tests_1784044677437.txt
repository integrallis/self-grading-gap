```python
from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    dp = [[float('inf')] * 3 for _ in range(n)]

    # Initialize the first house costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the dp table for the first half
    for i in range(1, n//2 + 1):
        for j in range(3):
            for k in range(3):
                if j != k:
                    dp[i][j] = min(dp[i][j], dp[i - 1][k] + cost[i][j])

    # Calculate the minimum cost for the second half with equidistant condition
    for i in range(n//2):
        for j in range(3):
            for k in range(3):
                if j != k and j != (2 - k):
                    dp[n//2 + i][j] = min(dp[n//2 + i][j], dp[n//2 - 1 - i][k] + cost[n//2 + i][j])

    # The answer is the minimum cost of painting the last house
    return min(dp[n - 1])
```
