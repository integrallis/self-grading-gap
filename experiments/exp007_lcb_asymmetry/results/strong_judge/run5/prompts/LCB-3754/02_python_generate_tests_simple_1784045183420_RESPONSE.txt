from solution import maxDistance

def test_canonical_example_1():
    # example: input "NWSE", k = 1
    # Changing 'S' to 'N' gives max distance = 3
    assert maxDistance("NWSE", 1) == 3

def test_canonical_example_2():
    # example: input "NSWWEW", k = 3
    # Changing 'S' to 'N' and 'E' to 'W' gives max distance = 6
    assert maxDistance("NSWWEW", 3) == 6

def test_no_change_budget():
    # input "NSWE", k = 0
    # No changes allowed, maximum distance = 1
    assert maxDistance("NSWE", 0) == 1

def test_full_change_budget():
    # input "SSW", k = 3
    # All 'S' can be changed to 'N' for max distance = 3
    assert maxDistance("SSW", 3) == 3

def test_minimum_length():
    # input "N", k = 0
    # No changes allowed, maximum distance = 1
    assert maxDistance("N", 0) == 1

def test_maximum_length():
    # input of length 100000 with all 'S', k = 100000
    # All 'S' can be changed to 'N' for max distance = 100000
    assert maxDistance("S" * 100000, 100000) == 100000
