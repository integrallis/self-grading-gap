```python
def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_dist = 0
    for move in s:
        if move == 'N':
            y += 1
        elif move == 'S':
            y -= 1
        elif move == 'E':
            x += 1
        elif move == 'W':
            x -= 1
        max_dist = max(max_dist, abs(x) + abs(y))
    # Calculate how far we can go by changing up to k moves
    total_moves = len(s)
    max_possible_distance = max_dist + k
    return min(max_possible_distance, total_moves + k)
```
