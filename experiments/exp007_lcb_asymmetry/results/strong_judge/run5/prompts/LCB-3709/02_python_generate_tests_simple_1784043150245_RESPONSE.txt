def test_canonical_example_1():
    # spec: hasSpecialSubstring("aaabaaa", 3) -> True
    assert hasSpecialSubstring("aaabaaa", 3) is True

def test_canonical_example_2():
    # spec: hasSpecialSubstring("abc", 2) -> False

def test_valid_substring_at_start():
    # hasSpecialSubstring("aaab", 3) -> True
    # The substring "aaa" has length 3, is uniform, has preceding 'b' which is different.
    assert hasSpecialSubstring("aaab", 3) is True

def test_valid_substring_strictly_inside():
    # hasSpecialSubstring("baaac", 3) -> True

    # hasSpecialSubstring("aaaa", 3) -> False

def test_minimum_boundary():
    # hasSpecialSubstring("a", 1) -> True
    # The single character 'a' is a valid substring of length 1.

def test_maximum_length_boundary():
    # hasSpecialSubstring("a" * 100, 100) -> True
    # The entire string of 'a's is a valid substring of length 100.
