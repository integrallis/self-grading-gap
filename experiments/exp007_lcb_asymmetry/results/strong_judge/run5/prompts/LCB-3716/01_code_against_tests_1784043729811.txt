```python
from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = [[1, float('inf')]] * n  # dp[i] holds [length, last_difference] of longest subsequence ending at index i
    max_length = 1

    for i in range(n):
        for j in range(i):
            abs_diff = abs(nums[i] - nums[j])
            # Check if we can extend the subsequence from j to i
            if abs_diff >= dp[j][1]:
                dp[i][0] = max(dp[i][0], dp[j][0] + 1)
                dp[i][1] = abs_diff
                max_length = max(max_length, dp[i][0])

    return max_length
```
