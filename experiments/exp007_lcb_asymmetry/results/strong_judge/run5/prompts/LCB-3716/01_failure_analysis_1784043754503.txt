def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = [1] * n  # dp[i] holds the length of the longest subsequence ending at index i
    max_length = 1
    diffs = [0] * n  # diffs[i] holds the last difference in the longest subsequence ending at index i

    for i in range(n):
        for j in range(i):
            abs_diff = abs(nums[i] - nums[j])
            # Check if we can extend the subsequence from j to i
            if abs_diff <= diffs[j] or j == 0:
                if j == 0 or abs_diff != 0:
                    dp[i] = max(dp[i], dp[j] + 1)
                    diffs[i] = abs_diff
            max_length = max(max_length, dp[i])
