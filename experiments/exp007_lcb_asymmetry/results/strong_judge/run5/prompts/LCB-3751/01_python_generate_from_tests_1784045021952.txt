from typing import List


def maxFrequency(nums: List[int], k: int) -> int:
    nums.sort()  # Sort the numbers to consider a sliding window
    left = 0  # Left pointer for the window
    total = 0  # Total of the current window
    max_freq = 0  # Maximum frequency of k found

    for right in range(len(nums)):
        total += nums[right]  # Add the current number to the total
        # While the current window cannot be converted to k with k additions
        while nums[right] * (right - left + 1) - total > k:
            total -= nums[left]  # Shrink the window from the left
            left += 1  # Move left pointer to the right
        max_freq = max(max_freq, right - left + 1)  # Update the maximum frequency

    return max_freq
```
