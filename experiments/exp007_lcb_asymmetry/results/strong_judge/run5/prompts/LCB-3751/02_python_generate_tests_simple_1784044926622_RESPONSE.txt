from typing import List
from solution import maxFrequency

def test_canonical_example_1():
    # spec: After adding -5 to nums[2..5], 1 has a frequency of 2 in [1, 2, -2, -1, 0, 1].
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2

def test_canonical_example_2():
    # spec: After adding 8 to nums[1..9], 10 has a frequency of 4 in [10, 10, 11, 12, 13, 13, 12, 11, 10, 10].
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4

def test_minimum_length():
    # spec: Only one element, cannot change anything, so frequency of k = 1 is 0.
    assert maxFrequency([2], 1) == 0

def test_no_op_case():
    # spec: All elements are k, selecting x = 0 keeps all occurrences of k.
    assert maxFrequency([1, 1, 1], 1) == 3

def test_left_boundary_case():
    # spec: Adding -1 to nums[0..2] makes all elements 1, yielding 3 occurrences of 1.
    assert maxFrequency([2, 2, 1], 1) == 3

def test_numeric_upper_bound():
    # spec: Adding 0 to change last 50 to 50 increases frequency of 50 to 3.
    assert maxFrequency([1, 1, 50], 50) == 3

def test_maximum_length():
    # spec: All elements are 1, so frequency of 1 is 100000.
    assert maxFrequency([1] * 100_000, 1) == 100_000
