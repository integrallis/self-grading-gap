from typing import List

def maxFrequency(nums: List[int], k: int) -> int:
    count = 0
    for i in range(len(nums)):
        for j in range(i, len(nums)):
            x = k - nums[i]  # Calculate what we need to add
            new_nums = nums[i:j+1]
            transformed = [num + x for num in new_nums]
            count_k = transformed.count(k)
            count = max(count, count_k)
    return count
```
