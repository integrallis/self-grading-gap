from typing import List
from solution import countPartitions

def test_canonical_example_1():
    # spec: countPartitions([10, 10, 3, 7, 6]) -> 4
    # Partitions are: 
    # [10], [10, 3, 7, 6] -> sum difference = 10 - 26 = -16 (even)
    # [10, 10], [3, 7, 6] -> sum difference = 20 - 16 = 4 (even)
    # [10, 10, 3], [7, 6] -> sum difference = 23 - 13 = 10 (even)
    # [10, 10, 3, 7], [6] -> sum difference = 30 - 6 = 24 (even)
    assert countPartitions([10, 10, 3, 7, 6]) == 4

def test_canonical_example_2():
    # spec: countPartitions([1, 2, 2]) -> 0
    # No partitions result in an even sum difference.
    assert countPartitions([1, 2, 2]) == 0

def test_canonical_example_3():
    # spec: countPartitions([2, 4, 6, 8]) -> 3
    # Partitions are:
    # [2], [4, 6, 8] -> sum difference = 2 - 18 = -16 (even)
    # [2, 4], [6, 8] -> sum difference = 6 - 14 = -8 (even)
    # [2, 4, 6], [8] -> sum difference = 12 - 8 = 4 (even)
    assert countPartitions([2, 4, 6, 8]) == 3

def test_minimum_length():
    # spec: countPartitions([1, 1]) -> 1
    # Only one partition: [1], [1] -> sum difference = 1 - 1 = 0 (even)
    assert countPartitions([1, 1]) == 1

def test_maximum_length():
    # spec: countPartitions([1] * 100) -> 99
    # The total sum is 100 (even), so all 99 partitions are valid.
    assert countPartitions([1] * 100) == 99

def test_maximum_element_value():
