from typing import List
from solution import maxFrequency

def test_canonical_example_1():
    # spec: maxFrequency([1, 2, 3, 4, 5, 6], 1) -> 2
    # Explanation: After adding -5 to nums[2..5], 1 has a frequency of 2 in [1, 2, -2, -1, 0, 1].
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2

def test_canonical_example_2():
    # spec: maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) -> 4
    # Explanation: After adding 8 to nums[1..9], 10 has a frequency of 4 in [10, 10, 11, 12, 13, 13, 12, 11, 10, 10].
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4

def test_minimum_boundary():
    # spec: maxFrequency([1], 1) -> 1
    # Explanation: The only element is 1, so its frequency is 1.
    assert maxFrequency([1], 1) == 1

def test_maximum_value_boundary():
    # spec: maxFrequency([50], 50) -> 1
    # Explanation: The only element is 50, so its frequency is 1.
    assert maxFrequency([50], 50) == 1

def test_all_k_preservation():
    # spec: maxFrequency([7, 7, 7], 7) -> 3
    # Explanation: All elements are 7, so their frequency is 3.
    assert maxFrequency([7, 7, 7], 7) == 3

def test_no_initial_k():
    # spec: maxFrequency([2, 2, 3, 3], 1) -> 2
    # Explanation: Adding -1 to nums[0..3] makes two elements equal to 1.
    assert maxFrequency([2, 2, 3, 3], 1) == 2

def test_uniform_addition():
    # spec: maxFrequency([2, 3, 2, 3], 1) -> 2
    # Explanation: You cannot make both 2 and 3 into 1 by a single addition.
    assert maxFrequency([2, 3, 2, 3], 1) == 2

def test_maximum_length():
    # spec: maxFrequency([50] * 100000, 50) -> 100000
    # Explanation: All elements are 50, so their frequency is 100000.
    assert maxFrequency([50] * 100000, 50) == 100000
