from solution import calculateScore

def test_canonical_example_1():
    # spec: calculateScore("aczzx") -> 5
    # Explanation:
    # i = 0: no j, skip.
    # i = 1: no j, skip.
    # i = 2: j = 0 (mirror 'a' is 'z'), score += 2 - 0 = 2.
    # i = 3: no j, skip.
    # i = 4: j = 1 (mirror 'c' is 'x'), score += 4 - 1 = 3.
    # Total score = 2 + 3 = 5
    assert calculateScore("aczzx") == 5

def test_canonical_example_2():
    # spec: calculateScore("abcdef") -> 0
    # Explanation:
    # For each index i, there is no index j that satisfies the conditions.
    assert calculateScore("abcdef") == 0

def test_closest_prior_mirror():
    # spec: calculateScore("aaz") -> 1
    # Explanation:
    # i = 0: no j, skip.
    # i = 1: j = 0 (mirror 'a' is 'z'), score += 1 - 0 = 1.
    # i = 2: no j, skip.
    # Total score = 1
    assert calculateScore("aaz") == 1

def test_current_index_marked():
    # spec: calculateScore("aza") -> 1
    # Explanation:
    # i = 0: no j, skip.
    # i = 1: j = 0 (mirror 'a' is 'z'), score += 1 - 0 = 1.
    # i = 2: no j, skip (index 1 is marked).
    # Total score = 1
    assert calculateScore("aza") == 1

def test_minimum_length():
    # spec: calculateScore("a") -> 0
    # Explanation:
    # Only one character, no j exists.
    assert calculateScore("a") == 0

def test_maximum_length():
    # spec: calculateScore("a" * 50_000 + "z" * 50_000) -> 2_500_000_000
    # Explanation:
    # Each 'a' at indices 0 to 49,999 will match with 'z' at indices 50,000 to 99,999.
    # Total score = sum of (i - j) for all pairs, which is 49,999 + 49,998 + ... + 0 = 2_500_000_000.
    s = "a" * 50_000 + "z" * 50_000
    assert calculateScore(s) == 2_500_000_000
