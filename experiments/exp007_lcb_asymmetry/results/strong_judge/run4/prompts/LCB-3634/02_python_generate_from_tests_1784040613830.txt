```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirror = {chr(i): chr(122 - (i - 97)) for i in range(97, 123)}  # a-z to z-a

    for i in range(len(s)):
        if marked[i]:  # skip if current index is already marked
            continue
        for j in range(i - 1, -1, -1):  # find closest unmarked index j
            if not marked[j] and mirror[s[i]] == s[j]:
                score += i - j
                marked[i] = marked[j] = True  # mark both indices
                break
    return score

```
