from typing import List
from solution import maxLength

def test_canonical_example_1():
    # spec: longest product equivalent subarray is [1, 2, 1, 1, 1]
    # prod([1, 2, 1, 1, 1]) = 2, gcd([1, 2, 1, 1, 1]) = 1, lcm([1, 2, 1, 1, 1]) = 2
    assert maxLength([1, 2, 1, 2, 1, 1, 1]) == 5

def test_canonical_example_2():
    # spec: longest product equivalent subarray is [3, 4, 5]
    # prod([3, 4, 5]) = 60, gcd([3, 4, 5]) = 1, lcm([3, 4, 5]) = 60
    assert maxLength([2, 3, 4, 5, 6]) == 3

def test_canonical_example_3():
    # spec: longest product equivalent subarray is [3, 1, 4, 5, 1]
    # prod([3, 1, 4, 5, 1]) = 60, gcd([3, 1, 4, 5, 1]) = 1, lcm([3, 1, 4, 5, 1]) = 60
    assert maxLength([1, 2, 3, 1, 4, 5, 1]) == 5

def test_minimum_length_boundary():
    # spec: longest product equivalent subarray is [10, 10]
    # prod([10, 10]) = 100, gcd([10, 10]) = 10, lcm([10, 10]) = 10
    assert maxLength([10, 10]) == 2

def test_maximum_input_length_boundary():
    # spec: longest product equivalent subarray is [1, 1, ..., 1] (100 times)
    # prod([1] * 100) = 1, gcd([1] * 100) = 1, lcm([1] * 100) = 1
    assert maxLength([1] * 100) == 100
