from typing import List
from solution import zigzagTraversal

def test_canonical_example_1():
    # input: [[1, 2], [3, 4]]
    # zigzag traversal with skips: visit (0,0) -> (1,1) -> output: [1, 4]
    assert zigzagTraversal([[1, 2], [3, 4]]) == [1, 4]

def test_canonical_example_2():
    # input: [[2, 1], [2, 1], [2, 1]]
    # zigzag traversal with skips: visit (0,0) -> (1,1) -> (2,0) -> output: [2, 1, 2]
    assert zigzagTraversal([[2, 1], [2, 1], [2, 1]]) == [2, 1, 2]

def test_canonical_example_3():
    # input: [[1, 2, 3], [4, 5, 6], [7, 8, 9]]
    # zigzag traversal with skips: visit (0,0) -> (0,2) -> (1,1) -> (2,0) -> (2,2) -> output: [1, 3, 5, 7, 9]
    assert zigzagTraversal([[1, 2, 3], [4, 5, 6], [7, 8, 9]]) == [1, 3, 5, 7, 9]

def test_maximum_dimensions():
    # input: 50x50 grid with distinct values from 1 to 2500
    grid = [[(i * 50 + j + 1) for j in range(50)] for i in range(50)]
    # zigzag traversal with skips will visit cells at (0,0), (0,2), ..., (0,48), 
    # (1,49), (1,47), ..., (1,1), (2,0), (2,2), ..., (2,48), ..., (49,49)
    # The expected output will be the values at those positions.
    expected_output = [1 + 2 * i for i in range(50)] + [50 * (i + 1) - 1 for i in range(49)]
    assert zigzagTraversal(grid) == expected_output

def test_max_value_in_grid():
    # input: grid containing maximum allowed value 2500 in visited and skipped positions
    grid = [[2500, 1], [1, 2500]]
    # zigzag traversal with skips: visit (0,0) -> (1,1) -> output: [2500, 2500]
    assert zigzagTraversal(grid) == [2500, 2500]
