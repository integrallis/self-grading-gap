from typing import List
import heapq


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    graph = [[] for _ in range(n)]
    for a, b, w in edges:
        graph[a].append((b, w))  # Build the graph as directed from A to B

    def can_reach_node_0(max_weight: int) -> bool:
        out_degree = [0] * n
        for u in range(n):
            count = 0
            for v, w in graph[u]:
                if w <= max_weight:
                    count += 1
            out_degree[u] = count
            if out_degree[u] > threshold:
                return False

        queue = []
        visited = [False] * n
        visited[0] = True  # Node 0 must be reachable

        for u in range(1, n):
            if out_degree[u] > 0:
                queue.append(u)

        while queue:
            node = queue.pop()
            for neighbor, w in graph[node]:
                if w <= max_weight and not visited[neighbor]:
                    visited[neighbor] = True
                    queue.append(neighbor)

        return all(visited[i] for i in range(n))

    left, right = 1, 10**6  # The weights range from 1 to 10^6
    answer = -1
    while left <= right:
        mid = (left + right) // 2
        if can_reach_node_0(mid):
            answer = mid
            right = mid - 1
        else:
            left = mid + 1

    return answer

```
