```python
from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_intervals = []
    free_intervals.append((0, startTime[0]))  # Free time before the first meeting

    for i in range(1, n):
        free_intervals.append((endTime[i - 1], startTime[i]))  # Gaps between meetings

    free_intervals.append((endTime[-1], eventTime))  # Free time after the last meeting

    # Calculate free times lengths
    free_lengths = [end - start for start, end in free_intervals]

    # Sort free lengths to allow for rescheduling k meetings, which can merge k gaps
    free_lengths.sort(reverse=True)

    # If k is greater than or equal to the number of gaps, all gaps can be merged
    max_free_time = sum(free_lengths) - len(free_lengths) + 1 if k >= len(free_lengths) else sum(free_lengths[:k + 1])

    return max_free_time
```
