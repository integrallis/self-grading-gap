```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a list to accumulate the number of coins at each position
    max_pos = 0
    for l, r, c in coins:
        max_pos = max(max_pos, r)

    # Create an array to hold the coins in each bag position
    bags = [0] * (max_pos + 2)  # +2 to handle edge cases

    # Fill the bags array based on the coins input
    for l, r, c in coins:
        for i in range(l, r + 1):
            bags[i] += c

    # Now we need to find the maximum coins we can collect from k consecutive bags
    max_coins = 0
    current_coins = sum(bags[1:k + 1])  # Initial sum for the first k bags
    max_coins = current_coins

    for i in range(1, max_pos - k + 2):
        current_coins = current_coins - bags[i] + bags[i + k]
        max_coins = max(max_coins, current_coins)

    return max_coins

```
