from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n == 0:
        return 0

    dp = [1] * n
    max_length = 1

    for i in range(n):
        for j in range(i):
            if abs(nums[i] - nums[j]) >= abs(nums[j] - nums[j - 1]) if j > 0 else 0:
                dp[i] = max(dp[i], dp[j] + 1)
                max_length = max(max_length, dp[i])

    return max_length
```
