def test_canonical_example_1():
    # spec: input [2, 3, 1] -> output 11
    # Calculations:
    # i=0: subarray [2] -> sum = 2
    # i=1: subarray [2, 3] -> sum = 5
    # i=2: subarray [3, 1] -> sum = 4
    # Total = 2 + 5 + 4 = 11
    assert subarraySum([2, 3, 1]) == 11

def test_canonical_example_2():

    # i=1: subarray [3, 1] -> sum = 4
    # i=2: subarray [1, 1] -> sum = 2
    # i=3: subarray [1, 1, 2] -> sum = 4

def test_maximum_length_and_value():
    # spec: input [1000] * 100 -> output 5050000
    # Calculations:
    # For each i from 0 to 99, the subarray will be [1000] * (i+1)
    # The sum for each i will be (i+1) * 1000
