from solution import reverseDegree

def test_canonical_example_1():
    # spec: reverseDegree("abc") 
    # 'a' = 26, index 1 -> 26
    # 'b' = 25, index 2 -> 50
    # 'c' = 24, index 3 -> 72
    # Total = 26 + 50 + 72 = 148
    assert reverseDegree("abc") == 148

def test_canonical_example_2():
    # spec: reverseDegree("zaza") 
    # 'z' = 1, index 1 -> 1
    # 'a' = 26, index 2 -> 52
    # 'z' = 1, index 3 -> 3
    # 'a' = 26, index 4 -> 104
    # Total = 1 + 52 + 3 + 104 = 160
    assert reverseDegree("zaza") == 160

def test_minimum_length():
    # spec: reverseDegree("a") 
    # 'a' = 26, index 1 -> 26
    # Total = 26
    assert reverseDegree("a") == 26

def test_maximum_length():
    # spec: reverseDegree("a" * 1000) 
    # 'a' = 26, index i -> 26 * i for i from 1 to 1000
    # Total = 26 * (1 + 2 + ... + 1000) = 26 * (1000 * 1001 / 2) = 13013000
    assert reverseDegree("a" * 1000) == 13013000
