```python
def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    # Count the number of each direction
    count_N = count_S = count_E = count_W = 0
    for direction in s:
        if direction == 'N':
            y += 1
            count_N += 1
        elif direction == 'S':
            y -= 1
            count_S += 1
        elif direction == 'E':
            x += 1
            count_E += 1
        elif direction == 'W':
            x -= 1
            count_W += 1
        current_distance = abs(x) + abs(y)
        max_distance = max(max_distance, current_distance)
    # Calculate the maximum possible distance
    total_moves = count_N + count_S + count_E + count_W
    # The furthest point we can reach by changing up to k moves
    max_possible_distance = max_distance + min(k, total_moves) * 2
    return max_possible_distance
```
