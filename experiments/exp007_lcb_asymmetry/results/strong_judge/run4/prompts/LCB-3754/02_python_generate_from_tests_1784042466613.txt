def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    for direction in s:
        if direction == 'N':
            y += 1
        elif direction == 'S':
            y -= 1
        elif direction == 'E':
            x += 1
        elif direction == 'W':
            x -= 1
        current_distance = abs(x) + abs(y)
        max_distance = max(max_distance, current_distance)

    # Add the maximum possible distance we can achieve by changing k moves
    return max_distance + k
```
