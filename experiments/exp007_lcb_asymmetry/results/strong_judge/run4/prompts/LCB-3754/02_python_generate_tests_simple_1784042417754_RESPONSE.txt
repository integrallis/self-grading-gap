from solution import maxDistance

def test_canonical_example_1():
    # Example: input "NWSE", k = 1
    # Maximum Manhattan distance can be achieved by changing 'S' to 'N'.
    # New path: "NWNE", positions will be (0, 1) -> (-1, 1) -> (-1, 2) -> (0, 2)
    # Maximum distance = 3
    assert maxDistance("NWSE", 1) == 3

def test_canonical_example_2():
    # Example: input "NSWWEW", k = 3
    # Change 'S' to 'N' and 'E' to 'W', making the path "NNWWWW"
    # Positions will be (0, 1) -> (0, 2) -> (-1, 2) -> (-2, 2) -> (-3, 2) -> (-4, 2)
    # Maximum distance = 6
    assert maxDistance("NSWWEW", 3) == 6

def test_min_length():
    # Minimum length input: input "N", k = 0
    # Maximum Manhattan distance = 1
    assert maxDistance("N", 0) == 1

def test_k_zero():
    # Test with k = 0: input "NSWE", k = 0
    # No changes allowed, maximum distance = 1
    assert maxDistance("NSWE", 0) == 1

def test_k_equals_length():
    # Test with k = len(s): input "NSWE", k = 4
    # All positions can be changed to reach maximum distance = 4
    assert maxDistance("NSWE", 4) == 4

def test_large_input():
    # Test with maximum length input: 100000 'N's, k = 0
    # Only moving north, maximum distance = 100000
    assert maxDistance("N" * 100000, 0) == 100000
