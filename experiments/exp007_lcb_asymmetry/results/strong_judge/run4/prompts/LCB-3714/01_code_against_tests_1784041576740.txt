```python
from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    total_combinations = 0

    # Precompute powers of 2 up to k
    power_of_2 = [1] * (k + 1)
    for i in range(1, k + 1):
        power_of_2[i] = (power_of_2[i - 1] * 2) % MOD

    # Calculate contributions of each number as min and max
    for i in range(n):
        num = nums[i]
        # Each number can be in subsequences of size 1 to k
        # Count how many times this number will be min or max
        total_contributions = 0
        for j in range(1, k + 1):
            total_contributions = (total_contributions + power_of_2[j - 1]) % MOD
        total_sum = (total_sum + (num * total_contributions * 2) % MOD) % MOD

    return total_sum % MOD

```
