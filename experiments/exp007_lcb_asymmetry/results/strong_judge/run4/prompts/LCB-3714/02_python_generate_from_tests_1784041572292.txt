from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    
    # Calculate contributions from the minimums and maximums
    for i in range(1, k + 1):
        # Number of subsequences choosing i elements
        num_subsequences = (1 << n) - 1
        # Each element contributes to num_subsequences / n times
        for num in nums:
            total_sum += (num * 2 * num_subsequences) % MOD
    
    return total_sum % MOD

```
