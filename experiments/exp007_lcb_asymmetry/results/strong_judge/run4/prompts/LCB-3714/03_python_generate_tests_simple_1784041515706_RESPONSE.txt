from typing import List
from solution import minMaxSums

def test_canonical_example_1():
    # spec: sum of min and max of subsequences of [1, 2, 3] with at most 2 elements
    # subsequences: [1], [2], [3], [1, 2], [1, 3], [2, 3]
    # min/max sums: 2 + 4 + 6 + 3 + 4 + 5 = 24
    assert minMaxSums([1, 2, 3], 2) == 24

def test_canonical_example_2():
    # spec: sum of min and max of subsequences of [5, 0, 6] with exactly 1 element
    # subsequences: [5], [0], [6]
    # min/max sums: 5 + 5 + 0 + 0 + 6 + 6 = 22
    assert minMaxSums([5, 0, 6], 1) == 22

def test_k_equals_array_length():
    # spec: sum of min and max of subsequences of [1_000_000_000, 1_000_000_000, 1_000_000_000] with k = 3
    # subsequences: [1_000_000_000], [1_000_000_000], [1_000_000_000], [1_000_000_000, 1_000_000_000], [1_000_000_000, 1_000_000_000], [1_000_000_000, 1_000_000_000]
    # min/max sums: 3 * (1_000_000_000 + 1_000_000_000) = 6_000_000_000
    # 6_000_000_000 % (10**9 + 7) = 999_999_909
    assert minMaxSums([1_000_000_000, 1_000_000_000, 1_000_000_000], 3) == 999_999_909

    # spec: sum of min and max of subsequences of [1] * 70 with k = 70
    # all subsequences will contribute: 70 * (1 + 1) = 140

    assert minMaxSums([1] * 70, 70) == 540032504
