```python
from typing import List

def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers
    online_status = [True] * numberOfUsers
    event_map = []

    # Process events to handle OFFLINE and MESSAGE
    for event in events:
        event_type, timestamp, *data = event
        timestamp = int(timestamp)
        if event_type == "OFFLINE":
            user_id = int(data[0][2:])  # Extract id<number>
            # Set user offline
            online_status[user_id] = False
            # Schedule return
            event_map.append((timestamp + 60, user_id, True))
        elif event_type == "MESSAGE":
            mentions_string = data[0]
            # Handle mentions
            # Process returning users if their time has come
            while event_map and event_map[0][0] <= timestamp:
                _, user_id, status = event_map.pop(0)
                online_status[user_id] = status
            # Split mentions
            if mentions_string == "ALL":
                for i in range(numberOfUsers):
                    mentions[i] += 1
            elif mentions_string == "HERE":
                for i in range(numberOfUsers):
                    if online_status[i]:
                        mentions[i] += 1
            else:
                mentioned_ids = mentions_string.split()
                for mention in mentioned_ids:
                    user_id = int(mention[2:])  # Extract id<number>
                    mentions[user_id] += 1

    # Process any returning users at the end of events
    for return_time, user_id, status in event_map:
        if status:
            online_status[user_id] = status

    return mentions
```
