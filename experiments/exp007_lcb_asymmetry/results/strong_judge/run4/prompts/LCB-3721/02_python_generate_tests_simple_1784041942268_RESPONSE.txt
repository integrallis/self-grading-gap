from typing import List
from solution import countMentions

def test_canonical_example_1():
    # spec: Input: numberOfUsers = 2, events = [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","71","HERE"]]
    # At timestamp 10, id1 and id0 are mentioned. mentions = [1,1]
    # At timestamp 11, id0 goes offline.
    # At timestamp 71, id0 comes back online and "HERE" is mentioned. mentions = [2,2]
    assert countMentions(2, [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "71", "HERE"]]) == [2, 2]

def test_canonical_example_2():
    # spec: Input: numberOfUsers = 2, events = [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","12","ALL"]]
    # At timestamp 10, id1 and id0 are mentioned. mentions = [1,1]
    # At timestamp 11, id0 goes offline.
    # At timestamp 12, "ALL" is mentioned. This includes offline users, so both id0 and id1 are mentioned. mentions = [2,2]
    assert countMentions(2, [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "12", "ALL"]]) == [2, 2]

def test_canonical_example_3():
    # spec: Input: numberOfUsers = 2, events = [["OFFLINE","10","0"],["MESSAGE","12","HERE"]]
    # At timestamp 10, id0 goes offline.
    # At timestamp 12, "HERE" is mentioned. Because id0 is still offline, they will not be mentioned. mentions = [0,1]
    assert countMentions(2, [["OFFLINE", "10", "0"], ["MESSAGE", "12", "HERE"]]) == [0, 1]

def test_duplicate_id_mentions():
    # spec: Input: numberOfUsers = 3, events = [["MESSAGE", "1", "id2 id2 id0"]]
    # id2 is mentioned twice and id0 once. mentions = [1, 0, 2]
    assert countMentions(3, [["MESSAGE", "1", "id2 id2 id0"]]) == [1, 0, 2]

def test_offline_before_message():
    # spec: Input: numberOfUsers = 2, events = [["MESSAGE", "10", "HERE"], ["OFFLINE", "10", "0"]]
    # At timestamp 10, id0 goes offline, but "HERE" mentions only id1. mentions = [0, 1]
    assert countMentions(2, [["MESSAGE", "10", "HERE"], ["OFFLINE", "10", "0"]]) == [0, 1]

def test_explicit_id_mentions_while_offline():
    # spec: Input: numberOfUsers = 2, events = [["OFFLINE", "1", "0"], ["MESSAGE", "2", "id0"]]
    # id0 is offline when mentioned, but still counts. mentions = [1, 0]
    assert countMentions(2, [["OFFLINE", "1", "0"], ["MESSAGE", "2", "id0"]]) == [1, 0]

def test_minimum_boundary():
    # spec: Input: numberOfUsers = 1, events = [["MESSAGE", "1", "id0"]]
    # Only one user is mentioned. mentions = [1]
    assert countMentions(1, [["MESSAGE", "1", "id0"]]) == [1]

def test_maximum_boundary():
    # spec: Input: numberOfUsers = 100, events = [["MESSAGE", "1", "id99"] * 100 + ["ALL"] * 99]
    # id99 is mentioned 100 times and ALL mentions all users including offline. mentions = [99] * 99 + [199]
    events = [["MESSAGE", "1", "id99"] for _ in range(100)] + [["ALL", str(i + 2), "ALL"] for i in range(99)]
    assert countMentions(100, events) == [99] * 99 + [199]
