from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize a DP array to store the minimum costs
    dp = [[0] * 3 for _ in range(n)]
    # Set the base costs for the first house
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the DP array for all houses
    for i in range(1, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(
                dp[i - 1][(j + 1) % 3],
                dp[i - 1][(j + 2) % 3]
            )

    # Calculate the minimum cost considering equidistant houses
    total_min_cost = float('inf')
    for j in range(3):
        total_min_cost = min(total_min_cost, dp[n - 1][j])

    return total_min_cost
