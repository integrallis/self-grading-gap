from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize a DP array to store the minimum costs
    dp = [[0] * 3 for _ in range(n)]
    # Set the base costs for the first house
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the DP array for all houses
    for i in range(1, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(
                dp[i - 1][(j + 1) % 3],
                dp[i - 1][(j + 2) % 3]
            )

    # Handle equidistant house constraint
    min_cost = float('inf')
    for color in range(3):
        for opposite_color in range(3):
            if opposite_color != color:
                min_cost = min(min_cost, dp[n - 1][color] + dp[n - 2][opposite_color])

    return min_cost

```
