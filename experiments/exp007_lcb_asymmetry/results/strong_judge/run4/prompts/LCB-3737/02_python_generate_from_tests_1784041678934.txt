from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize a DP array to store the minimum costs
    dp = [[0] * 3 for _ in range(n)]
    # Set the base costs for the first house
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the DP array for the first half of houses
    for i in range(1, n // 2 + 1):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(
                dp[i - 1][(j + 1) % 3],
                dp[i - 1][(j + 2) % 3]
            )

    # Calculate the minimum cost for the second half of houses
    total_min_cost = float('inf')
    for j in range(3):
        dp[n // 2][j] += dp[n // 2 - 1][(j + 1) % 3] + dp[n // 2 - 1][(j + 2) % 3]
        total_min_cost = min(total_min_cost, dp[n // 2][j])

    return total_min_cost
```
