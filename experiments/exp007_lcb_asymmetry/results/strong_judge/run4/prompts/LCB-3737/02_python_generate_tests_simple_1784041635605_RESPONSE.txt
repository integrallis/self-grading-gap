from typing import List
from solution import minCost

def test_canonical_example_1():
    # spec: input '4\n[[3, 5, 7], [6, 2, 9], [4, 8, 1], [7, 3, 5]]' -> output '9'
    assert minCost(4, [[3, 5, 7], [6, 2, 9], [4, 8, 1], [7, 3, 5]]) == 9

def test_canonical_example_2():
    # spec: input '6\n[[2, 4, 6], [5, 3, 8], [7, 1, 9], [4, 6, 2], [3, 5, 7], [8, 2, 4]]' -> output '18'
    assert minCost(6, [[2, 4, 6], [5, 3, 8], [7, 1, 9], [4, 6, 2], [3, 5, 7], [8, 2, 4]]) == 18

def test_minimum_size():
    # Minimum n = 2 with costs [0, 5, 7] and [0, 3, 4]: output should be 3 (choose color 2 for house 1)
    assert minCost(2, [[0, 5, 7], [0, 3, 4]]) == 3

def test_maximum_cell_cost():
    # Maximum cell cost: for n = 2, all costs are 100000; output should be 200000 (color choice does not matter)
    assert minCost(2, [[100000, 100000, 100000], [100000, 100000, 100000]]) == 200000

def test_large_n():
    # Large n = 100000 with all-zero costs should return 0
    assert minCost(100000, [[0, 0, 0]] * 100000) == 0
