from typing import List
from solution import maximumAmount

def test_canonical_example_1():
    # spec: maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) -> 8
    # Explanation:
    # Start at (0, 0) with 0 coins (total coins = 0).
    # Move to (0, 1), gaining 1 coin (total coins = 0 + 1 = 1).
    # Move to (1, 1), where there's a robber stealing 2 coins. 
    # The robot uses one neutralization here, avoiding the robbery (total coins = 1).
    # Move to (1, 2), gaining 3 coins (total coins = 1 + 3 = 4).
    # Move to (2, 2), gaining 4 coins (total coins = 4 + 4 = 8).
    assert maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) == 8

def test_canonical_example_2():
    # spec: maximumAmount([[10, 10, 10], [10, 10, 10]]) -> 40
    # Explanation:
    # Start at (0, 0) with 10 coins (total coins = 10).
    # Move to (0, 1), gaining 10 coins (total coins = 10 + 10 = 20).
    # Move to (0, 2), gaining another 10 coins (total coins = 20 + 10 = 30).
    # Move to (1, 2), gaining the final 10 coins (total coins = 30 + 10 = 40).
    assert maximumAmount([[10, 10, 10], [10, 10, 10]]) == 40

def test_two_neutralizations_negative_optimum():
    # Using two neutralizations to avoid robbers, the path can yield:
    # Start at (0, 0) with -5 coins (total coins = -5).
    # Move to (0, 1), gaining -1 coins (total coins = -5 - 1 = -6).
    # Move to (0, 2), gaining -7 coins (total coins = -6 - 7 = -13).
    # Neutralizing both robbers gives 0 coins.
    assert maximumAmount([[-5, -1, -7]]) == -1

def test_one_by_one_negative_cell():
    # Single cell with a negative value, can neutralize it
    # Start at (0, 0) with -1000 coins (total coins = -1000).
    # Neutralizing it gives 0 coins.
    assert maximumAmount([[-1000]]) == 0

def test_single_column():
    # A single column with positive and negative cells.
    # Start at (0, 0) with 1 coin (total coins = 1).
    # Move to (1, 0), gaining -5 coins (total coins = 1 - 5 = -4).
    # Move to (2, 0), gaining 2 coins (total coins = -4 + 2 = -2).
    # Neutralizing -5 gives total coins = 1 + 2 = 3.
    assert maximumAmount([[1], [-5], [2]]) == 3

def test_boundary_values():
    # Test a row with maximum values and a negative robber.
    # Start at (0, 0) with 1000 coins (total coins = 1000).
    # Move to (0, 1), gaining -1000 (total coins = 1000 - 1000 = 0).
    # Move to (0, 2), gaining -1000 (total coins = 0 - 1000 = -1000).
    # Move to (0, 3), gaining -1000 (total coins = -1000 - 1000 = -2000).
    # Neutralizing the first -1000 gives total coins = 0.
    assert maximumAmount([[1000, -1000, -1000, -1000]]) == 0

def test_maximum_dimension():
    # Test the maximum grid size filled with zeros
    # The robot moves through 500 * 500 cells, all with 0 coins.
    assert maximumAmount([[0] * 500 for _ in range(500)]) == 0
