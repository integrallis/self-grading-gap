from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[[float('-inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0]
    if coins[0][0] < 0:
        dp[0][0][1] = 0  # Neutralizing the starting cell if it's negative

    for i in range(m):
        for j in range(n):
            for k in range(3):  # k is the number of neutralizations used
                if i == 0 and j == 0:
                    continue
                gain = coins[i][j]
                if gain >= 0:
                    from_top = dp[i-1][j][k] + gain if i > 0 else float('-inf')
                    from_left = dp[i][j-1][k] + gain if j > 0 else float('-inf')
                    dp[i][j][k] = max(dp[i][j][k], from_top, from_left)
                else:
                    from_top = dp[i-1][j][k] if i > 0 else float('-inf')
                    from_left = dp[i][j-1][k] if j > 0 else float('-inf')
                    dp[i][j][k] = max(dp[i][j][k], from_top, from_left)
                    if k > 0:
                        dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k-1] if i > 0 else float('-inf'), dp[i][j-1][k-1] if j > 0 else float('-inf'))
    return max(dp[m-1][n-1])
