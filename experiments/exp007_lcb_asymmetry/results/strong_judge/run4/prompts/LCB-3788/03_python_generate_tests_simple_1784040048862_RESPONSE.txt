def test_canonical_example_2():

    assert maxSum([1, 1, 0, 1, 1]) == 1

def test_canonical_example_3():
    # spec: maxSum([1, 2, -1, -2, 1, 0, -1]) -> 3 (select subarray [2, 1])

    # spec: maxSum([-100]) -> -100 (only one element, which is negative)

    # spec: maxSum([100]) -> 100 (only one element, which is positive)

    # spec: maxSum(list(range(1, 101))) -> 5050 (sum of all unique elements from 1 to 100)

    # spec: maxSum([-3, -1, -2]) -> -1 (select the maximum negative value)
