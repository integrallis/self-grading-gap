from typing import List

def maxSum(nums: List[int]) -> int:
    unique_nums = set()
    max_sum = float('-inf')
    current_sum = 0
    for num in nums:
        if num not in unique_nums:
            unique_nums.add(num)
            current_sum += num
        else:
            max_sum = max(max_sum, current_sum)
            # Reset current_sum and unique_nums until the duplicate is found
            while num in unique_nums:
                unique_nums.remove(nums[0])
                nums = nums[1:]
            unique_nums.add(num)
            current_sum = sum(unique_nums)
    max_sum = max(max_sum, current_sum)
    return max(max_sum, max(nums, default=0))
```
