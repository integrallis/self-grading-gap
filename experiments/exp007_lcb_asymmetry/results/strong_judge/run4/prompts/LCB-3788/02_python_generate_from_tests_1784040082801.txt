from typing import List

def maxSum(nums: List[int]) -> int:
    unique_nums = set()
    max_sum = float('-inf')
    current_sum = 0

    for num in nums:
        if num not in unique_nums:
            unique_nums.add(num)
            current_sum += num
        else:
            max_sum = max(max_sum, current_sum)
            # Reset the current sum and unique set
            current_sum = num
            unique_nums = {num}

    max_sum = max(max_sum, current_sum)
    return max(max_sum, 0)
```
