def test_canonical_example_1():
    # spec: For nums = [1,2,3] and k = 2,

def test_k_equals_length():
    # spec: For nums = [1, 2, 3, 4] and k = 4,
    # subsequences: [1], [2], [3], [4], [1, 2], [1, 3], [1, 4], [2, 3], [2, 4], [3, 4],
    # [1, 2, 3], [1, 2, 4], [1, 3, 4], [2, 3, 4], [1, 2, 3, 4]
    # Total sum of min and max elements = 75
    assert minMaxSums([1, 2, 3, 4], 4) == 75

def test_modulo_and_max_value():
    # spec: For nums = [1_000_000_000] and k = 1,
    # subsequences: [1_000_000_000]
    # Total sum of min and max elements = 1_000_000_000 + 1_000_000_000 = 2_000_000_000
    # which is 2_000_000_000 % (10^9 + 7) = 999_999_993
    assert minMaxSums([1_000_000_000], 1) == 999_999_993

def test_upper_k_boundary():
    # spec: For nums = [1_000_000_000] * 70 and k = 70,
    # Each subsequence contributes 1_000_000_000 + 1_000_000_000 = 2_000_000_000
    # Total subsequences = 2^70 - 1 (all non-empty subsets)
    # Total sum of min and max elements = (2^70 - 1) * 2_000_000_000
    # Result needs to be calculated modulo (10^9 + 7)
    total_subsequences = (1 << 70) - 1
    expected_sum = total_subsequences * 2_000_000_000
    assert minMaxSums([1_000_000_000] * 70, 70) == expected_sum % (10**9 + 7)

def test_large_length():
    # spec: Testing with the maximum length constraint
    # For nums = [1] * 10^5 and k = 70,
    # All subsequences contribute 1 + 1 = 2
    # Total number of subsequences with at most 70 elements = sum(C(100000, r) for r in range(1, 71))
    # Total sum of min and max elements = 2 * sum(C(100000, r) for r in range(1, 71)) % (10^9 + 7)
    from math import comb
    total_sum = sum(comb(100000, r) for r in range(1, 71)) * 2
    assert minMaxSums([1] * 100000, 70) == total_sum % (10**9 + 7)
