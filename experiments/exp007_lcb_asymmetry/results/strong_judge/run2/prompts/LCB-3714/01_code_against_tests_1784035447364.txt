## Implementation

```python
from typing import List
from math import comb

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    # Calculate contribution of each number as min and max
    for i in range(n):
        # Each number appears as min in C(n-1, j) subsequences of size j+1
        # and as max in C(n-1, j) subsequences of size j+1
        for j in range(k):
            total_sum += nums[i] * (comb(n - 1, j) * 2)
            total_sum %= MOD
    return total_sum
```
