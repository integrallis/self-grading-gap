from solution import maximumAmount

def test_canonical_example_1():
    # spec: maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) -> 8
    assert maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) == 8

def test_canonical_example_2():
    # spec: maximumAmount([[10, 10, 10], [10, 10, 10]]) -> 40
    assert maximumAmount([[10, 10, 10], [10, 10, 10]]) == 40

def test_two_neutralizations():
    # spec: maximumAmount([[-5, -6, -7]]) -> -5
    # Neutralizing -6 and -7 leaves -5.
    assert maximumAmount([[-5, -6, -7]]) == -5

def test_negative_optimal_result():
    # spec: maximumAmount([[-1, -2, -3, -4]]) -> -1
    # Robot can neutralize -1 and end up with -1.
    assert maximumAmount([[-1, -2, -3, -4]]) == -1

def test_minimum_size_grid():
    # spec: maximumAmount([[-1000]]) -> 0
    # Neutralizing the only robber leaves 0.
    assert maximumAmount([[-1000]]) == 0

def test_one_column_grid():
    # spec: maximumAmount([[5], [-10], [4]]) -> 9
    # Collect 5, neutralize -10, collect 4.
    assert maximumAmount([[5], [-10], [4]]) == 9

def test_boundary_values():
    # spec: maximumAmount([[-1000, 1000]]) -> 1000
    # Neutralizing -1000 leaves 1000.
    assert maximumAmount([[-1000, 1000]]) == 1000

def test_large_grid():
    # Test a 500 x 500 grid with all zeros.
    # spec: maximumAmount([[0]*500]*500) -> 0
    assert maximumAmount([[0]*500]*500) == 0
