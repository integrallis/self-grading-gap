from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[-float('inf')] * (n + 1) for _ in range(m + 1)]
    dp[0][1] = 0

    for i in range(m):
        for j in range(n):
            current_coins = coins[i][j]
            # If moving from above
            if i > 0:
                dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j + 1] + current_coins)
            # If moving from the left
            if j > 0:
                dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i + 1][j] + current_coins)

            # If current_coins is negative, consider neutralizing it
            if current_coins < 0:
                if i > 0:
                    dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j + 1] - current_coins)
                if j > 0:
                    dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i + 1][j] - current_coins)

    max_coins = -float('inf')
    for i in range(m + 1):
        for j in range(n + 1):
            if dp[i][j] != -float('inf'):
                max_coins = max(max_coins, dp[i][j])

    return max_coins

```
