from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[-float('inf')] * (n + 1) for _ in range(m + 1)]
    dp[0][0] = 0

    for i in range(m):
        for j in range(n):
            current_coins = coins[i][j]
            # If moving from above
            if i > 0:
                dp[i][j] = max(dp[i][j], dp[i - 1][j] + current_coins)
            # If moving from the left
            if j > 0:
                dp[i][j] = max(dp[i][j], dp[i][j - 1] + current_coins)

            # If current_coins is negative, consider neutralizing it
            if current_coins < 0:
                if i > 0:
                    dp[i][j] = max(dp[i][j], dp[i - 1][j] - current_coins)
                if j > 0:
                    dp[i][j] = max(dp[i][j], dp[i][j - 1] - current_coins)

    return max(dp[m - 1][n - 1], 0)

```
