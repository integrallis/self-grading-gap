```python
from typing import List
from collections import defaultdict, deque


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Create an adjacency list to store the graph
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[v].append((u, w))  # Store reversed directed edges to check reachability to node 0

    def canReachZero(maxWeight):
        # BFS to check if all nodes can reach node 0 with maxWeight constraints
        reachable = [False] * n
        queue = deque([0])  # Start from node 0
        reachable[0] = True
        while queue:
            current = queue.popleft()
            count = 0  # Track outgoing edges
            for neighbor, weight in graph[current]:
                if weight <= maxWeight:
                    if not reachable[neighbor]:
                        reachable[neighbor] = True
                        queue.append(neighbor)
                count += 1
                if count > threshold:
                    break
        return all(reachable)

    # Binary search over the possible edge weights
    low, high = 1, 10**6
    answer = -1
    while low <= high:
        mid = (low + high) // 2
        if canReachZero(mid):
            answer = mid
            high = mid - 1  # try for a smaller maximum weight
        else:
            low = mid + 1  # increase the weight

    return answer

```
