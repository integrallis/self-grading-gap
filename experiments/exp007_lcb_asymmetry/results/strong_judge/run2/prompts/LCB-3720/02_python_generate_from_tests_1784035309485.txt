from typing import List
import heapq


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Create an adjacency list to store the graph
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[v].append((u, w))  # Reverse edges for incoming paths to 0

    def canReachZero(maxWeight):
        # BFS to check if all nodes can reach node 0 with maxWeight constraints
        reachable = [0] * n
        queue = []
        for v in range(n):
            if len(graph[v]) <= threshold:
                queue.append(v)  # only consider nodes with <= threshold edges
        for v in queue:
            reachable[v] = 1
        while queue:
            current = queue.pop(0)
            for u, weight in graph[current]:
                if weight <= maxWeight and not reachable[u]:
                    reachable[u] = 1
                    queue.append(u)
        return all(reachable)

    # Binary search over the possible edge weights
    low, high = 1, 10**6
    answer = -1
    while low <= high:
        mid = (low + high) // 2
        if canReachZero(mid):
            answer = mid
            high = mid - 1  # try for a smaller maximum weight
        else:
            low = mid + 1  # increase the weight

    return answer

```
