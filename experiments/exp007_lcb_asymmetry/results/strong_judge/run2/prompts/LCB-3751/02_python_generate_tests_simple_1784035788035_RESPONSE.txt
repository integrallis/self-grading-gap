from typing import List
from solution import maxFrequency

def test_canonical_example_1():
    # spec: maxFrequency([1, 2, 3, 4, 5, 6], 1) -> 2
    # Explanation: After adding -5 to nums[2..5], 1 has a frequency of 2.
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2

def test_canonical_example_2():
    # spec: maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) -> 4
    # Explanation: After adding 8 to nums[1..9], 10 has a frequency of 4.
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4

def test_contiguity_sensitive():
    # spec: maxFrequency([2, 1, 2], 1) -> 2
    # Explanation: After adding -1 to nums[0..2], 1 has a frequency of 2 (changed both 2s).
    assert maxFrequency([2, 1, 2], 1) == 2

def test_shared_addition():
    # spec: maxFrequency([2, 3], 1) -> 1
    # Explanation: Cannot change both 2 and 3 to 1 with a single x.
    assert maxFrequency([2, 3], 1) == 1

def test_zero_addition():
    # spec: maxFrequency([7, 7, 7], 7) -> 3
    # Explanation: Adding 0 retains the frequency of 3 for 7.
    assert maxFrequency([7, 7, 7], 7) == 3

def test_minimum_length_boundary():
    # spec: maxFrequency([50], 50) -> 1
    # Explanation: With only one element, the frequency of 50 is 1.
    assert maxFrequency([50], 50) == 1

def test_maximum_length_constraint():
    # spec: maxFrequency([1] * 100000, 1) -> 100000
    # Explanation: All elements are 1, so the frequency of 1 is 100000.
    assert maxFrequency([1] * 100000, 1) == 100000
