```python
from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # Dynamic programming table to store the minimum costs
    dp = [[0] * 3 for _ in range(n)]

    # Initialize the first house costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the DP table
    for i in range(1, n):
        for j in range(3):
            # Cost to paint house i with color j
            dp[i][j] = cost[i][j] + min(dp[i-1][(j+1)%3], dp[i-1][(j+2)%3])

    min_cost = float('inf')
    # Calculate the minimum cost considering the constraints
    for j in range(3):
        first_half_cost = dp[n//2 - 1][j]
        second_half_cost = min(dp[n - 1][(j + 1) % 3], dp[n - 1][(j + 2) % 3])
        total_cost = first_half_cost + second_half_cost
        min_cost = min(min_cost, total_cost)

    return min_cost

```
