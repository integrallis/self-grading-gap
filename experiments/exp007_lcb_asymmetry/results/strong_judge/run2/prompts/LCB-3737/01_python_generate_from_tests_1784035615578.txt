from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # Dynamic programming table to store the minimum costs
    dp = [[0] * 3 for _ in range(n)]

    # Initialize the first house costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the DP table
    for i in range(1, n):
        for j in range(3):
            # Cost to paint house i with color j
            dp[i][j] = cost[i][j] + min(dp[i-1][(j+1)%3], dp[i-1][(j+2)%3])

    # Calculate the minimum cost considering symmetry
    min_cost = float('inf')
    for j in range(3):
        # First half painted with color j
        first_half_cost = dp[n//2-1][j]
        # Second half must not be the same color as the first
        second_half_cost = min(dp[n-1][(j+1)%3], dp[n-1][(j+2)%3])
        min_cost = min(min_cost, first_half_cost + second_half_cost)

    return min_cost

```
