```python
def calculateScore(s: str) -> int:
    score = 0
    stacks = {chr(i + 97): [] for i in range(26)}  # Stacks for unmatched indices
    mirror = {chr(i + 97): chr(122 - i) for i in range(26)}  # Mirror mapping

    for i, ch in enumerate(s):
        mirror_char = mirror[ch]  # Get the mirror character
        if stacks[mirror_char]:  # Check if there's an unmatched index for the mirror char
            j = stacks[mirror_char].pop()  # Pop the closest unmatched index
            score += i - j  # Update score
        else:
            stacks[ch].append(i)  # No match found, push current index to the stack

    return score

```
