def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirror = {chr(i + 97): chr(122 - i) for i in range(26)}

    for i in range(len(s)):
        if marked[i]:
            continue
        for j in range(i - 1, -1, -1):
            if not marked[j] and mirror[s[i]] == s[j]:
                marked[i] = marked[j] = True
                score += i - j
                break

    return score

```
