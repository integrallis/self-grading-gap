from solution import calculateScore

def test_canonical_example_1():
    # spec: calculateScore("aczzx") -> 5
    # Explanation:
    # i = 0. No j found, skip.
    # i = 1. No j found, skip.
    # i = 2. j = 0 (mirror of 'a' is 'z'), score += 2 - 0 = 2
    # i = 3. No j found, skip.
    # i = 4. j = 1 (mirror of 'b' is 'y'), score += 4 - 1 = 3
    # Total score = 2 + 3 = 5
    assert calculateScore("aczzx") == 5

def test_canonical_example_2():
    # spec: calculateScore("abcdef") -> 0
    # Explanation:
    # For each index i (0 to 5), no j found that satisfies the conditions.
    # Total score = 0
    assert calculateScore("abcdef") == 0

def test_closest_match_and_no_reuse():
    # spec: calculateScore("aazz") -> 4
    # Explanation:
    # i = 0. No j found, skip.
    # i = 1. No j found, skip.
    # i = 2. j = 1 (mirror of 'a' is 'z'), score += 2 - 1 = 1
    # i = 3. j = 0 (mirror of 'a' is 'z'), score += 3 - 0 = 3
    # Total score = 1 + 3 = 4
    assert calculateScore("aazz") == 4

def test_non_example_mirror_pair():
    # spec: calculateScore("ev") -> 1
    # Explanation:
    # i = 0. No j found, skip.
    # i = 1. j = 0 (mirror of 'e' is 'v'), score += 1 - 0 = 1
    # Total score = 1
    assert calculateScore("ev") == 1

def test_minimum_length():
    # spec: calculateScore("a") -> 0
    # Explanation:
    # Only one character, no j found, score = 0.
    assert calculateScore("a") == 0

def test_maximum_length():
    # spec: calculateScore("a" * 100000) -> 0
    # Explanation:
    # All characters are 'a', no j found for any i, score = 0.
    assert calculateScore("a" * 100000) == 0
