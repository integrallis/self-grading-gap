from typing import List
from solution import countPartitions

def test_canonical_example_1():
    # spec: input [10, 10, 3, 7, 6] 
    # Partitions are:
    # [10], [10, 3, 7, 6] -> 10 - 26 = -16 (even)
    # [10, 10], [3, 7, 6] -> 20 - 16 = 4 (even)
    # [10, 10, 3], [7, 6] -> 23 - 13 = 10 (even)
    # [10, 10, 3, 7], [6] -> 30 - 6 = 24 (even)
    # Total = 4
    assert countPartitions([10, 10, 3, 7, 6]) == 4

def test_canonical_example_2():
    # spec: input [1, 2, 2]
    # No partitions result in an even sum difference.
    assert countPartitions([1, 2, 2]) == 0

def test_canonical_example_3():
    # spec: input [2, 4, 6, 8]
    # Partitions are:
    # [2], [4, 6, 8] -> 2 - 18 = -16 (even)
    # [2, 4], [6, 8] -> 6 - 14 = -8 (even)
    # [2, 4, 6], [8] -> 12 - 8 = 4 (even)
    # Total = 3
    assert countPartitions([2, 4, 6, 8]) == 3

def test_min_length():
    # spec: input [1, 1]
    # Only one partition: [1], [1] -> 1 - 1 = 0 (even)
    assert countPartitions([1, 1]) == 1

def test_max_length():
    # spec: input [100] * 100
    # Total sum is 10000 (even), all partitions are valid
    # There are 100 - 1 = 99 valid partitions
    assert countPartitions([100] * 100) == 99
