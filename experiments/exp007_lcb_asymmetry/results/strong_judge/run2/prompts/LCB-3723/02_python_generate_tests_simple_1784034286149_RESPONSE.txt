from typing import List
from solution import sumOfGoodNumbers

def test_canonical_example_1():
    # spec: sumOfGoodNumbers([1, 3, 2, 1, 5, 4], 2) -> 12
    # Good numbers are 3 (nums[1]), 5 (nums[4]), and 4 (nums[5])

def test_canonical_example_2():
    # spec: sumOfGoodNumbers([2, 1], 1) -> 2
    # Good number is 2 (nums[0]) because it is greater than nums[1]
    assert sumOfGoodNumbers([2, 1], 1) == 2

def test_interior_good_element():
    # Good number is 5 (nums[1]) because it is greater than nums[0] (1) and nums[2] (2)
    assert sumOfGoodNumbers([1, 5, 2], 1) == 5

def test_strict_comparison_equality():
    # No good numbers since 5 is not greater than 5 (nums[2])

    # Good number is 1000 (nums[50]) because it is greater than nums[0:50] (all 1s)
    assert sumOfGoodNumbers([1] * 50 + [1000] + [1] * 49, 50) == 1000
