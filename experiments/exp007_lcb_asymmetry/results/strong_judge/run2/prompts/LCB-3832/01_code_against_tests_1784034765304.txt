## Implementation

```python
from typing import List

def minCosts(cost: List[int]) -> List[int]:
    n = len(cost)
    answer = [0] * n
    answer[0] = cost[0]  # Cost to swap with the first person
    
    for i in range(1, n):
        answer[i] = min(answer[i - 1], cost[i])  # Minimum cost to reach position i
    
    return answer

```
