from typing import List
from solution import minCosts

def test_canonical_example_1():
    # spec: input [5, 3, 4, 1, 3, 2] -> output [5, 3, 3, 1, 1, 1]
    # Explanation:
    # i = 0: cost[0] = 5
    # i = 1: cost[1] = 3
    # i = 2: swap with person 1 (cost 3) -> free swap with person 2
    # i = 3: cost[3] = 1
    # i = 4: swap with person 3 (cost 1) -> free swap with person 4
    # i = 5: swap with person 3 (cost 1) -> free swap with person 5
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]

def test_canonical_example_2():
    # spec: input [1, 2, 4, 6, 7] -> output [1, 1, 1, 1, 1]
    # Explanation:
    # i = 0: cost[0] = 1
    # i = 1: swap with person 0 (cost 1) -> free for position 1
    # i = 2: swap with person 0 (cost 1) -> free for position 2
    # i = 3: swap with person 0 (cost 1) -> free for position 3
    # i = 4: swap with person 0 (cost 1) -> free for position 4
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]

def test_minimum_length():
    # spec: input [100] -> output [100]
    # Explanation:
    # i = 0: cost[0] = 100
    assert minCosts([100]) == [100]

def test_maximum_length():
    # spec: input [100] * 100 -> output [100] * 100
    # Explanation:
    # Each cost is 100, and for each position i, the cost remains 100.
    assert minCosts([100] * 100) == [100] * 100
