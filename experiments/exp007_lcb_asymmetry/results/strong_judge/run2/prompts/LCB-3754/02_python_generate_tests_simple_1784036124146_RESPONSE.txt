from solution import maxDistance

def test_canonical_example_1():
    # spec: maxDistance("NWSE", 1) -> 3
    # Change s[2] from 'S' to 'N'. The string s becomes "NWNE".
    # Maximum Manhattan distance from the origin is 3.
    assert maxDistance("NWSE", 1) == 3

def test_canonical_example_2():
    # spec: maxDistance("NSWWEW", 3) -> 6
    # Change s[1] from 'S' to 'N', and s[4] from 'E' to 'W'. The string s becomes "NNWWWW".
    # Maximum Manhattan distance from the origin is 6.
    assert maxDistance("NSWWEW", 3) == 6

def test_zero_edit():
    # spec: maxDistance("NS", 0) -> 1
    # No changes allowed, maximum distance is 1 (moving from (0, 0) to (0, 1) or (-1, 0)).
    assert maxDistance("NS", 0) == 1

def test_minimum_length():
    # spec: maxDistance("N", 0) -> 1
    # No changes allowed, maximum distance is 1 (moving from (0, 0) to (0, 1)).
    assert maxDistance("N", 0) == 1

def test_full_edit_budget():
    # spec: maxDistance("NS", 2) -> 2
    # We can change both moves to either direction, reaching maximum distance of 2.
    assert maxDistance("NS", 2) == 2

def test_maximum_length():
    # spec: maxDistance("N" * 100_000, 0) -> 100_000
    # No changes allowed, maximum distance is 100,000 (moving north continuously).
    assert maxDistance("N" * 100_000, 0) == 100_000

def test_directional_symmetry_south():
    # spec: maxDistance("SN", 1) -> 2
    # Change either 'S' or 'N' to maximize distance to (0, 2) or (0, -2).
    assert maxDistance("SN", 1) == 2

def test_directional_symmetry_east():
    # spec: maxDistance("EW", 1) -> 2
    # Change either 'E' or 'W' to maximize distance to (2, 0) or (-2, 0).
    assert maxDistance("EW", 1) == 2
