def maxDistance(s: str, k: int) -> int:
    # Initialize the starting position and maximum distance
    x, y = 0, 0
    max_distance = 0
    n = len(s)

    # Iterate over each move to track position and calculate maximum distance
    for i in range(n):
        if s[i] == 'N':
            y += 1
        elif s[i] == 'S':
            y -= 1
        elif s[i] == 'E':
            x += 1
        elif s[i] == 'W':
            x -= 1

        # Calculate the current Manhattan distance
        current_distance = abs(x) + abs(y)
        max_distance = max(max_distance, current_distance)

        # Consider all four orientations and calculate potential maximum distances
        for sx, sy in [(1, 1), (1, -1), (-1, 1), (-1, -1)]:
            score = (x * sx + y * sy)  # score aligns with current orientation
            opposing_moves = 0
            # Count how many moves are opposing the current orientation
            for j in range(i + 1):
