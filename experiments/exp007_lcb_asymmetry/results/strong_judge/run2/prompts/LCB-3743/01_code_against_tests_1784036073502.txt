```python
from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    if n == 0:
        return eventTime

    # Calculate gaps
    gaps = [startTime[0]] + [startTime[i] - endTime[i - 1] for i in range(1, n)] + [eventTime - endTime[-1]]

    # If k is 0, return the maximum single gap
    if k == 0:
        return max(gaps)

    # Use sliding window to find the maximum free time
    max_free_time = 0
    current_sum = sum(gaps[:k + 1])
    max_free_time = current_sum

    for i in range(1, len(gaps) - k):
        current_sum = current_sum - gaps[i - 1] + gaps[i + k]
        max_free_time = max(max_free_time, current_sum)

    return max_free_time
```
