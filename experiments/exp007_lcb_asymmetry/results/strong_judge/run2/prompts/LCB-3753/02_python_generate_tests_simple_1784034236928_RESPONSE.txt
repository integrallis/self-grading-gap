from solution import maxDifference

def test_canonical_example_1():
    # spec: maxDifference("aaaaabbc") -> 3
    # 'a' has an odd frequency of 5, 'b' has an even frequency of 2. 
    # Maximum difference is 5 - 2 = 3
    assert maxDifference("aaaaabbc") == 3

def test_canonical_example_2():
    # spec: maxDifference("abcabcab") -> 1
    # 'a' has an odd frequency of 3, 'c' has an even frequency of 2. 
    # Maximum difference is 3 - 2 = 1
    assert maxDifference("abcabcab") == 1

def test_minimum_length():
    # spec: maxDifference("aab") -> -1
    # 'a' has an even frequency of 2, 'b' has an odd frequency of 1. 
    # Maximum difference is 1 - 2 = -1
    assert maxDifference("aab") == -1

def test_maximum_length():
    # spec: maxDifference("a" * 97 + "bb" + "c") -> 95
    # 'a' has an odd frequency of 97, 'b' has an even frequency of 2, 'c' has an odd frequency of 1.
    # Maximum difference is 97 - 2 = 95
    assert maxDifference("a" * 97 + "bb" + "c") == 95

def test_multiple_even_frequencies():
    # spec: maxDifference("aaaaabbbbcc") -> 3
    # 'a' has an odd frequency of 5, 'b' has an even frequency of 4, 'c' has an even frequency of 2.
    # Maximum difference is 5 - 2 = 3 (selecting the smallest even frequency)
    assert maxDifference("aaaaabbbbcc") == 3
