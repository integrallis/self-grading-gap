from typing import List
from solution import countMentions

def test_canonical_example_1():
    # spec: numberOfUsers = 2, events = [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","71","HERE"]]
    # At timestamp 10, id1 and id0 are mentioned. mentions = [1,1]
    # At timestamp 11, id0 goes offline.
    # At timestamp 71, id0 comes back online and "HERE" is mentioned. mentions = [2,2]
    assert countMentions(2, [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","71","HERE"]]) == [2, 2]

def test_canonical_example_2():
    # spec: numberOfUsers = 2, events = [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","12","ALL"]]
    # At timestamp 10, id1 and id0 are mentioned. mentions = [1,1]
    # At timestamp 11, id0 goes offline.
    # At timestamp 12, "ALL" is mentioned. Both id0 and id1 are mentioned. mentions = [2,2]
    assert countMentions(2, [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","12","ALL"]]) == [2, 2]

def test_canonical_example_3():
    # spec: numberOfUsers = 2, events = [["OFFLINE","10","0"],["MESSAGE","12","HERE"]]
    # At timestamp 10, id0 goes offline.
    # At timestamp 12, "HERE" is mentioned. id0 is offline, so id1 is mentioned. mentions = [0,1]
    assert countMentions(2, [["OFFLINE","10","0"],["MESSAGE","12","HERE"]]) == [0, 1]

def test_duplicate_mentions():
    # spec: numberOfUsers = 2, events = [["MESSAGE", "5", "id0 id0 id1"]]
    # id0 is mentioned twice and id1 once. mentions = [2, 1]
    assert countMentions(2, [["MESSAGE", "5", "id0 id0 id1"]]) == [2, 1]

def test_explicit_id_to_offline_user():
    # spec: numberOfUsers = 2, events = [["OFFLINE", "5", "0"], ["MESSAGE", "6", "id0 id1"]]
    # id0 is offline, id1 is online. mentions = [1, 1]
    assert countMentions(2, [["OFFLINE", "5", "0"], ["MESSAGE", "6", "id0 id1"]]) == [1, 1]

def test_same_timestamp_offline_priority():
    # spec: numberOfUsers = 2, events = [["MESSAGE", "10", "HERE"], ["MESSAGE", "20", "HERE"], ["OFFLINE", "20", "0"]]
    # At timestamp 10, both are online. At timestamp 20, id0 goes offline. mentions = [1, 2]
    assert countMentions(2, [["MESSAGE", "10", "HERE"], ["MESSAGE", "20", "HERE"], ["OFFLINE", "20", "0"]]) == [1, 2]

def test_out_of_order_timestamp():
    # spec: numberOfUsers = 2, events = [["MESSAGE", "100", "HERE"], ["OFFLINE", "10", "0"], ["MESSAGE", "20", "HERE"]]
    # At timestamp 10, id0 goes offline. At timestamp 20, id1 is mentioned. At timestamp 100, id1 is still online. mentions = [1, 2]
    assert countMentions(2, [["MESSAGE", "100", "HERE"], ["OFFLINE", "10", "0"], ["MESSAGE", "20", "HERE"]]) == [1, 2]

def test_boundary_one_user():
    # spec: numberOfUsers = 1, events = [["MESSAGE", "10", "id0"]]
    # Only one user, id0 is mentioned. mentions = [1]
    assert countMentions(1, [["MESSAGE", "10", "id0"]]) == [1]

def test_boundary_maximum_conditions():
    # spec: numberOfUsers = 100, events = [["MESSAGE", "1", " ".join(f"id{i}" for i in range(100))] for _ in range(100)]
    # All users mentioned in each message. mentions should be [100]*100 = [100]*100
    result = countMentions(100, [["MESSAGE", "1", " ".join(f"id{i}" for i in range(100))] for _ in range(100)])
    assert result == [100] * 100
