```python
from typing import List

def maxFrequency(nums: List[int], k: int) -> int:
    base = nums.count(k)
    max_gain = 0
    current_gain = 0

    for num in nums:
        if num == k:
            current_gain -= 1
        elif num < k:
            current_gain += 1
        else:
            current_gain = 0
        max_gain = max(max_gain, current_gain)

    return base + max(0, max_gain)
```
