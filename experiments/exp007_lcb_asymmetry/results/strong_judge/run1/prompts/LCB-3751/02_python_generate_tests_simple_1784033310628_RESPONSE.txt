from typing import List
from solution import maxFrequency

def test_canonical_example_1():
    # spec: maxFrequency([1, 2, 3, 4, 5, 6], 1) -> 2
    # After adding -5 to nums[2..5], 1 has a frequency of 2 in [1, 2, -2, -1, 0, 1].
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2

def test_canonical_example_2():
    # spec: maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) -> 4
    # After adding 8 to nums[1..9], 10 has a frequency of 4 in [10, 10, 11, 12, 13, 13, 12, 11, 10, 10].
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4

def test_minimum_size():
    # spec: maxFrequency([1], 1) -> 1
    # The only element is 1, so its frequency is 1.
    assert maxFrequency([1], 1) == 1

def test_maximum_constraint():
    # spec: maxFrequency([50] * 100000, 50) -> 100000
    # All elements are 50, so the frequency of 50 is 100000.
    assert maxFrequency([50] * 100000, 50) == 100000

def test_one_element_subarray():
    # spec: maxFrequency([2, 1, 3], 1) -> 2
    # Adding 0 to the subarray [1] makes 1 appear twice.
    assert maxFrequency([2, 1, 3], 1) == 2

def test_entire_array_selection():
    # spec: maxFrequency([2, 2, 2], 1) -> 3
    # Adding -1 to the entire array makes all 2's into 1's.
    assert maxFrequency([2, 2, 2], 1) == 3

def test_zero_optimal_selection():
    # spec: maxFrequency([1, 1, 1], 1) -> 3
    # Adding 0 to the entire array retains all 1's, giving a frequency of 3.
    assert maxFrequency([1, 1, 1], 1) == 3
