from typing import List

def maxFrequency(nums: List[int], k: int) -> int:
    nums.sort()
    n = len(nums)
    left = 0
    total_add = 0
    max_freq = 0

    for right in range(n):
        total_add += nums[right]
        while nums[right] * (right - left + 1) > total_add + k:
            total_add -= nums[left]
            left += 1
        max_freq = max(max_freq, right - left + 1)

    return max_freq
```
