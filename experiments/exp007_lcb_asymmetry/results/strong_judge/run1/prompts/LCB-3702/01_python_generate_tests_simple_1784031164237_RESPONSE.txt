from typing import List
from math import gcd
from functools import reduce
from solution import maxLength

def lcm(a, b):
    return abs(a * b) // gcd(a, b)

def prod(arr):
    result = 1
    for num in arr:
        result *= num
    return result

def test_canonical_example_1():
    # spec: longest subarray [1, 2, 1, 1, 1] => prod = 2, gcd = 1, lcm = 2
    assert maxLength([1, 2, 1, 2, 1, 1, 1]) == 5

def test_canonical_example_2():
    # spec: longest subarray [3, 4, 5] => prod = 60, gcd = 1, lcm = 60
    assert maxLength([2, 3, 4, 5, 6]) == 3

def test_canonical_example_3():
