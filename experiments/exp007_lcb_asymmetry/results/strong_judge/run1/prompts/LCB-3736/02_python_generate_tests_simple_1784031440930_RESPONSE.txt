from solution import findValidPair

def test_canonical_example_1():
    # s = "2523533": digit '2' appears 2 times and digit '3' appears 3 times.
    # The valid pair is "23".
    assert findValidPair("2523533") == "23"

def test_canonical_example_2():
    # s = "221": digit '2' appears 2 times and digit '1' appears 1 time.
    # The valid pair is "21".
    assert findValidPair("221") == "21"

def test_canonical_example_3():
    # s = "22": there are no valid adjacent pairs.
    assert findValidPair("22") == ""

def test_first_match_example():
    # s = "122333": both "12" and "23" are valid pairs.
    # The first valid pair is "12".
    assert findValidPair("122333") == "12"

def test_maximum_length():
    # s = "122" + "9" * 97: digit '1' appears 1 time, digit '2' appears 2 times.
    # The valid pair is "12".
    assert findValidPair("122" + "9" * 97) == "12"
