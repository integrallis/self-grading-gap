from solution import calculateScore

def test_canonical_example_1():
    # spec: calculateScore("aczzx") -> 5
    # Explanation breakdown:
    # i = 2, j = 0 -> score += (2 - 0) = 2
    # i = 4, j = 1 -> score += (4 - 1) = 3
    # Total score = 2 + 3 = 5
    assert calculateScore("aczzx") == 5

def test_canonical_example_2():
    # spec: calculateScore("abcdef") -> 0
    # Explanation:
    # For every index i, there is no index j that satisfies the conditions.
    assert calculateScore("abcdef") == 0

def test_closest_match():
    # spec: calculateScore("aaz") -> 1
    # Explanation breakdown:
    # i = 2, j = 1 -> score += (2 - 1) = 1
    # Total score = 1
    assert calculateScore("aaz") == 1

def test_marked_index_exclusion():
    # spec: calculateScore("azaz") -> 2
    # Explanation breakdown:
    # i = 1, j = 0 -> score += (1 - 0) = 1
    # i = 3 is skipped because j = 0 and j = 1 are marked.
    # Total score = 1
    assert calculateScore("azaz") == 2

def test_minimum_length():
    # spec: calculateScore("a") -> 0
    # Explanation:
    # There are no pairs possible.
    assert calculateScore("a") == 0

def test_another_mirror_pair():
    # spec: calculateScore("by") -> 1
    # Explanation breakdown:
    # i = 1, j = 0 -> score += (1 - 0) = 1
    # Total score = 1
    assert calculateScore("by") == 1

def test_maximum_length():
    # spec: calculateScore("a" * 50000 + "z" * 50000) -> 2500000000
    # Explanation breakdown:
    # Each 'a' at index 0, 1, ..., 49999 pairs with each 'z' at index 50000, ..., 99999
    # Score for each pair = 1 + 2 + ... + 50000 = 50000 * (50000 + 1) // 2 = 1250025000
    # Total score = 1250025000 * 2 = 2500000000
    assert calculateScore("a" * 50000 + "z" * 50000) == 2500000000
