```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # Mapping for a-z to z-a

    for i in range(len(s)):
        if marked[i]:
            continue
        for j in range(i - 1, -1, -1):
            if not marked[j] and s[j] == mirror[s[i]]:
                marked[i] = marked[j] = True
                score += (i - j)
                break
    return score
```
