## Implementation

```python
def calculateScore(s: str) -> int:
    score = 0
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # Mapping for a-z to z-a
    unmatched = {ch: [] for ch in mirror.keys()}

    for i, ch in enumerate(s):
        target = mirror[ch]
        if unmatched[target]:
            j = unmatched[target].pop()
            score += (i - j)
        else:
            unmatched[ch].append(i)

    return score
```
