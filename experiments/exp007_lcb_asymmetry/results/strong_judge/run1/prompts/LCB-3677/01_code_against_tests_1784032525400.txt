```python
from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[[float('-inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0]
    if coins[0][0] < 0:
        dp[0][0][1] = 0

    for i in range(m):
        for j in range(n):
            for k in range(3):
                if i == 0 and j == 0:
                    continue
                if i > 0:
                    current_coins = dp[i-1][j][k] + coins[i][j]
                    if coins[i][j] < 0 and k > 0:
                        current_coins = max(current_coins, dp[i-1][j][k-1])
                    dp[i][j][k] = max(dp[i][j][k], current_coins)
                if j > 0:
                    current_coins = dp[i][j-1][k] + coins[i][j]
                    if coins[i][j] < 0 and k > 0:
                        current_coins = max(current_coins, dp[i][j-1][k-1])
                    dp[i][j][k] = max(dp[i][j][k], current_coins)

    return max(dp[m-1][n-1])
```
