from typing import List
from solution import zigzagTraversal

def test_canonical_example_1():
    # Input: [[1,2],[3,4]] 
    # Zigzag traversal: 1 (0,0) -> 4 (1,1) -> skipped cells -> Output: [1, 4]
    assert zigzagTraversal([[1, 2], [3, 4]]) == [1, 4]

def test_canonical_example_2():
    # Input: [[2,1],[2,1],[2,1]] 
    # Zigzag traversal: 2 (0,0) -> 1 (0,1) -> (1,0) -> (1,1) -> (2,0) -> Output: [2, 1, 2]
    assert zigzagTraversal([[2, 1], [2, 1], [2, 1]]) == [2, 1, 2]

def test_canonical_example_3():
    # Input: [[1,2,3],[4,5,6],[7,8,9]] 
    # Zigzag traversal: 1 (0,0) -> 3 (0,2) -> 5 (1,1) -> 7 (2,0) -> 9 (2,2) -> Output: [1, 3, 5, 7, 9]
    assert zigzagTraversal([[1, 2, 3], [4, 5, 6], [7, 8, 9]]) == [1, 3, 5, 7, 9]

def test_maximum_dimensions():
    # Create a 50 x 50 grid filled with values from 1 to 2500
    grid = [[(i * 50 + j + 1) for j in range(50)] for i in range(50)]
    # Zigzag traversal will skip alternate cells, starting from (0,0)
    # So we will take every second element in zigzag order
    expected_output = []
    for i in range(50):
        if i % 2 == 0:  # even row, go left to right
            expected_output.extend(grid[i][j] for j in range(0, 50, 2))
        else:  # odd row, go right to left
            expected_output.extend(grid[i][j] for j in range(49, -1, -1)[::2])
    assert zigzagTraversal(grid) == expected_output

def test_max_value():
    # Input: A grid of size 2x2 with maximum allowed value
    grid = [[2500, 2500], [2500, 2500]]
    # Zigzag traversal: 2500 (0,0) -> 2500 (1,1) -> skipped cells -> Output: [2500, 2500]
    assert zigzagTraversal(grid) == [2500, 2500]
