from solution import reverseDegree

def test_canonical_example_1():
    # spec: reverseDegree("abc") -> 148
    # Calculation: 
    # 'a' = 26 * 1 = 26
    # 'b' = 25 * 2 = 50
    # 'c' = 24 * 3 = 72
    # Total = 26 + 50 + 72 = 148
    assert reverseDegree("abc") == 148

def test_canonical_example_2():
    # spec: reverseDegree("zaza") -> 160
    # Calculation:
    # 'z' = 1 * 1 = 1
    # 'a' = 26 * 2 = 52
    # 'z' = 1 * 3 = 3
    # 'a' = 26 * 4 = 104
    # Total = 1 + 52 + 3 + 104 = 160
    assert reverseDegree("zaza") == 160

def test_minimum_length():
    # spec: reverseDegree("a") -> 26
    # Calculation:
    # 'a' = 26 * 1 = 26
    assert reverseDegree("a") == 26

def test_maximum_length():
    # spec: reverseDegree("z" * 1000) -> 500500
    # Calculation:
    # Each 'z' contributes 1 * its position (1 to 1000).
    # Total = 1 + 2 + ... + 1000 = 500500 (using the sum formula n(n + 1)/2)
    assert reverseDegree("z" * 1000) == 500500
