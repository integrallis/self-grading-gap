def test_canonical_example_1():
    # spec: reverseDegree("abc") -> 148
    # Calculation: 
    # 'a' = 26 * 1 = 26
    # 'b' = 25 * 2 = 50
    # 'c' = 24 * 3 = 72
    # Total = 26 + 50 + 72 = 148
    assert reverseDegree("abc") == 148

def test_canonical_example_2():
    # spec: reverseDegree("zaza") -> 160
    # Calculation:
    # 'z' = 1 * 1 = 1
    # 'a' = 26 * 2 = 52
    # 'z' = 1 * 3 = 3
    # 'a' = 26 * 4 = 104
    # Total = 1 + 52 + 3 + 104 = 160
