from typing import List
from solution import maxSum

def test_canonical_example_1():
    # spec: maxSum([1, 2, 3, 4, 5]) -> 15 (select the entire array)
    assert maxSum([1, 2, 3, 4, 5]) == 15

def test_canonical_example_2():
    # spec: maxSum([1, 1, 0, 1, 1]) -> 1 (keep one of the 1's)
    assert maxSum([1, 1, 0, 1, 1]) == 1

def test_canonical_example_3():
    # spec: maxSum([1, 2, -1, -2, 1, 0, -1]) -> 3 (select subarray [1, 2])
    assert maxSum([1, 2, -1, -2, 1, 0, -1]) == 3

def test_singleton_negative():
    # spec: maxSum([-7]) -> -7 (only one element available)
    assert maxSum([-7]) == -7

def test_max_length():
    # spec: maxSum with length 100 (use a mix of unique and duplicates)
    assert maxSum(list(range(1, 101))) == sum(range(1, 101))  # 1 + 2 + ... + 100 = 5050

def test_min_value():
    # spec: maxSum([-100, -99]) -> -99 (select the maximum negative)
    assert maxSum([-100, -99]) == -99

def test_max_value():
    # spec: maxSum([100, 100, -1]) -> 100 (keep one of the 100's)
    assert maxSum([100, 100, -1]) == 100

def test_deletion_sensitive():
    # spec: maxSum([1, 2, 1, 2, 3, 3]) -> 6 (select distinct 1, 2, 3)
    assert maxSum([1, 2, 1, 2, 3, 3]) == 6
