```python
from typing import List

def maxSum(nums: List[int]) -> int:
    unique_nums = set()
    max_sum = nums[0]  # Initialize to the first element
    current_sum = 0
    left = 0
    for right in range(len(nums)):
        while left <= right and (nums[right] in unique_nums or current_sum < 0):
            unique_nums.remove(nums[left])
            current_sum -= nums[left]
            left += 1
        unique_nums.add(nums[right])
        current_sum += nums[right]
        max_sum = max(max_sum, current_sum)
    return max_sum
```
