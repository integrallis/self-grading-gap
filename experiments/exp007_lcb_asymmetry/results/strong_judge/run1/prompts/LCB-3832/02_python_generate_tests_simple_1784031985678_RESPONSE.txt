from typing import List
from solution import minCosts

def test_canonical_example_1():
    # spec: input [5, 3, 4, 1, 3, 2] 
    # costs to reach positions are: 
    # 0: swap with 0 for cost 5
    # 1: swap with 1 for cost 3
    # 2: swap with 1 (cost 3), then 2 for free
    # 3: swap with 3 for cost 1
    # 4: swap with 3 (cost 1), then 4 for free
    # 5: swap with 3 (cost 1), then 5 for free
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]

def test_canonical_example_2():
    # spec: input [1, 2, 4, 6, 7]
    # costs to reach positions are:
    # 0: swap with 0 for cost 1
    # 1: swap with 0 (cost 1), then 1 for free
    # 2: swap with 0 (cost 1), then 2 for free
    # 3: swap with 0 (cost 1), then 3 for free
    # 4: swap with 0 (cost 1), then 4 for free
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]

def test_singleton_boundary():
    # spec: input [1]
    # only one person, cost to swap with them is 1
    assert minCosts([1]) == [1]

def test_maximum_length_and_cost_boundary():
    # spec: input [100] * 100
    # every person in line has a cost of 100
    # costs to reach each position are the same as their cost
    assert minCosts([100] * 100) == [100] * 100
