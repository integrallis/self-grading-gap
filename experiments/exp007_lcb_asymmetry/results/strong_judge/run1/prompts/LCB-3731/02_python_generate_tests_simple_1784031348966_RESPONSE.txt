def test_canonical_example_1():
    # spec: input [2, 3, 1]
    # Subarray sums:
    # 0: nums[0] = [2] -> 2
    # 1: nums[0...1] = [2, 3] -> 5
    # 2: nums[1...2] = [3, 1] -> 4
    # Total sum = 2 + 5 + 4 = 11
    assert subarraySum([2, 3, 1]) == 11

def test_canonical_example_2():
    # spec: input [3, 1, 1, 2]
    # Subarray sums:
    # 0: nums[0] = [3] -> 3
    # 1: nums[0...1] = [3, 1] -> 4
    # 2: nums[1...2] = [1, 1] -> 2
    # 3: nums[1...3] = [1, 1, 2] -> 4
    # Total sum = 3 + 4 + 2 + 4 = 13
    assert subarraySum([3, 1, 1, 2]) == 13

def test_singleton_input():
    # spec: input [1]

def test_maximum_length_input():
    # spec: input [1] * 100
    # Subarray sums:
    # For each index i from 0 to 99, start = max(0, i - 1) = max(0, i - 1)
    # 0: nums[0] = [1] -> 1
    # 1: nums[0...1] = [1, 1] -> 2
    # 2: nums[1...2] = [1, 1] -> 2
    # ...
    # 99: nums[98...99] = [1, 1] -> 2
    # Total sum = 1 + 2 + 2 + ... + 2 (for 99 times) = 1 + 2 * 99 = 199
    assert subarraySum([1] * 100) == 199

def test_maximum_element_value():
    # spec: input [1000, 1]

    # 0: nums[0] = [1000] -> 1000
    # 1: nums[0...1] = [1000, 1] -> 1001
    # Total sum = 1000 + 1001 = 2001
    assert subarraySum([1000, 1]) == 2001
