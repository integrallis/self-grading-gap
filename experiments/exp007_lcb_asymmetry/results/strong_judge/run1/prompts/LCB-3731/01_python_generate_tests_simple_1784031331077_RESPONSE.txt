from solution import subarraySum

def test_canonical_example_1():
    # spec: input [2, 3, 1]
    # Subarray sums:
    # 0: nums[0] = [2] -> 2
    # 1: nums[0...1] = [2, 3] -> 5
    # 2: nums[1...2] = [3, 1] -> 4
    # Total sum = 2 + 5 + 4 = 11
    assert subarraySum([2, 3, 1]) == 11

def test_canonical_example_2():
    # spec: input [3, 1, 1, 2]
    # Subarray sums:
    # 0: nums[0] = [3] -> 3
    # 1: nums[0...1] = [3, 1] -> 4
    # 2: nums[1...2] = [1, 1] -> 2
    # 3: nums[1...3] = [1, 1, 2] -> 4
    # Total sum = 3 + 4 + 2 + 4 = 13
    assert subarraySum([3, 1, 1, 2]) == 13
