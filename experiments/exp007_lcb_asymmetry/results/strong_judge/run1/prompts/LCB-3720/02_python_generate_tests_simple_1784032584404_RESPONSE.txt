from typing import List
from solution import minMaxWeight

def test_canonical_example_1():
    # Input: n = 5, edges = [[1,0,1],[2,0,2],[3,0,1],[4,3,1],[2,1,1]], threshold = 2
    # Output: 1
    # Explanation: Remove the edge 2 -> 0. The maximum weight among the remaining edges is 1.
    assert minMaxWeight(5, [[1,0,1],[2,0,2],[3,0,1],[4,3,1],[2,1,1]], 2) == 1

def test_canonical_example_2():
    # Input: n = 5, edges = [[0,1,1],[0,2,2],[0,3,1],[0,4,1],[1,2,1],[1,4,1]], threshold = 1
    # Output: -1
    # Explanation: It is impossible to reach node 0 from node 2.
    assert minMaxWeight(5, [[0,1,1],[0,2,2],[0,3,1],[0,4,1],[1,2,1],[1,4,1]], 1) == -1

def test_canonical_example_3():
    # Input: n = 5, edges = [[1,2,1],[1,3,3],[1,4,5],[2,3,2],[3,4,2],[4,0,1]], threshold = 1
    # Output: 2
    # Explanation: Remove the edges 1 -> 3 and 1 -> 4. The maximum weight among the remaining edges is 2.
    assert minMaxWeight(5, [[1,2,1],[1,3,3],[1,4,5],[2,3,2],[3,4,2],[4,0,1]], 1) == 2

def test_canonical_example_4():
    # Input: n = 5, edges = [[1,2,1],[1,3,3],[1,4,5],[2,3,2],[4,0,1]], threshold = 1
    # Output: -1
    # Explanation: It is impossible to reach node 0 from node 3.
    assert minMaxWeight(5, [[1,2,1],[1,3,3],[1,4,5],[2,3,2],[4,0,1]], 1) == -1

def test_no_removal_needed_at_smallest_graph_and_maximum_threshold():
    # Input: n = 2, edges = [[1, 0, 7]], threshold = 1
    # Output: 7
    # Explanation: No edges need to be removed and maximum weight is 7.
    assert minMaxWeight(2, [[1, 0, 7]], 1) == 7

def test_parallel_edges_choose_lower_weight():
    # Input: n = 2, edges = [[1, 0, 9], [1, 0, 3]], threshold = 1
    # Output: 3
    # Explanation: The lower weight edge 1->0 with weight 3 is retained.
    assert minMaxWeight(2, [[1, 0, 9], [1, 0, 3]], 1) == 3

def test_maximum_weight_is_valid_answer():
    # Input: n = 2, edges = [[1, 0, 1_000_000]], threshold = 1
    # Output: 1_000_000
    # Explanation: The maximum weight edge is retained.
    assert minMaxWeight(2, [[1, 0, 1_000_000]], 1) == 1_000_000

def test_large_valid_input():
    # Input: n = 100_000, edges = [[i, i - 1, 1] for i in range(1, 100_000)] + [[0, 99_999, 1]], threshold = 1
    # Output: 1
    # Explanation: All edges allow node 0 to be reachable with maximum weight 1.
    assert minMaxWeight(100_000, [[i, i - 1, 1] for i in range(1, 100_000)] + [[0, 99_999, 1]], 1) == 1
