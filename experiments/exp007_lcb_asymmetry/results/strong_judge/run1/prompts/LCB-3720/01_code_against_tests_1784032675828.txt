from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Create adjacency list for the graph
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[v].append((u, w))  # Reverse the graph for reachability to 0

    # Binary search for the minimum maximum weight
    left, right = 1, 10**6
    def can_reach_with_max_weight(max_weight):
        out_degree = [0] * n
        queue = []
        for v in range(n):
            for u, w in graph[v]:
                if w <= max_weight:
                    out_degree[v] += 1
                    if out_degree[v] <= threshold:
                        queue.append(v)
        visited = [False] * n
        while queue:
            current = queue.pop()
            if visited[current]:
                continue
            visited[current] = True
            for u, w in graph[current]:
                if w <= max_weight:
                    out_degree[u] -= 1
                    if out_degree[u] < threshold and not visited[u]:
                        queue.append(u)
        return visited[0]  # Check if 0 is reachable

    while left < right:
        mid = (left + right) // 2
        if can_reach_with_max_weight(mid):
            right = mid
        else:
            left = mid + 1

    return left if can_reach_with_max_weight(left) else -1
