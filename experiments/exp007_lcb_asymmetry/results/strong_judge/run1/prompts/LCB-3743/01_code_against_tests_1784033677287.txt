```python
from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    if n == 0:
        return eventTime

    # Create a list to hold the gaps between meetings
    gaps = [startTime[0]] + [startTime[i] - endTime[i - 1] for i in range(1, n)] + [eventTime - endTime[-1]]

    max_free_time = 0
    current_sum = 0

    # Sliding window to find the maximum sum of k + 1 contiguous gaps
    for i in range(min(k + 1, len(gaps))):
        current_sum += gaps[i]

    max_free_time = current_sum

    for i in range(min(k + 1, len(gaps)), len(gaps)):
        current_sum += gaps[i] - gaps[i - (k + 1)]
        max_free_time = max(max_free_time, current_sum)

    return max_free_time
```
