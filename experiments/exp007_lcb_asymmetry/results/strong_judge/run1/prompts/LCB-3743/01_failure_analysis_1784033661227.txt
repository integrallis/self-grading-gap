```python
from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    # Create a list to hold the gaps between meetings
    gaps = []
    # The end of the last meeting starts from 0
    last_end = 0

    # Calculate gaps before the first meeting and between meetings
    for i in range(len(startTime)):
        if startTime[i] > last_end:
            gaps.append(startTime[i] - last_end)
        last_end = endTime[i]

    # Add gap after the last meeting
    if last_end < eventTime:
        gaps.append(eventTime - last_end)

    # If there are no gaps, return 0
    if not gaps:
        return 0

    # Sort gaps in descending order to maximize the free time
    gaps.sort(reverse=True)

    # Sum the largest k gaps
    max_free_time = sum(gaps[:k])

    # The maximum free time is the total event time minus occupied time plus the gaps we can use
    return max_free_time + sum(gaps[k:])
```
