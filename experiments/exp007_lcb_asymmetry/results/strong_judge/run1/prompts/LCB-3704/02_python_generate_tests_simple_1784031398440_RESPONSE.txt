from typing import List
from solution import countPartitions

def test_canonical_example_1():
    # For input [10, 10, 3, 7, 6], the partitions are:
    # [10], [10, 3, 7, 6] -> difference = 10 - 26 = -16 (even)
    # [10, 10], [3, 7, 6] -> difference = 20 - 16 = 4 (even)
    # [10, 10, 3], [7, 6] -> difference = 23 - 13 = 10 (even)
    # [10, 10, 3, 7], [6] -> difference = 30 - 6 = 24 (even)
    # Total even differences = 4
    assert countPartitions([10, 10, 3, 7, 6]) == 4

def test_canonical_example_2():
    # For input [1, 2, 2], there are no valid partitions:
    # [1], [2, 2] -> difference = 1 - 4 = -3 (odd)
    # [1, 2], [2] -> difference = 3 - 2 = 1 (odd)
    # Total even differences = 0
    assert countPartitions([1, 2, 2]) == 0

def test_canonical_example_3():
    # For input [2, 4, 6, 8], the partitions are:
    # [2], [4, 6, 8] -> difference = 2 - 18 = -16 (even)
    # [2, 4], [6, 8] -> difference = 6 - 14 = -8 (even)
    # [2, 4, 6], [8] -> difference = 12 - 8 = 4 (even)
    # Total even differences = 3
    assert countPartitions([2, 4, 6, 8]) == 3

def test_minimum_length():
    # For input [1, 1], the only partition is:
    # [1], [1] -> difference = 1 - 1 = 0 (even)
    # Total even differences = 1
    assert countPartitions([1, 1]) == 1

def test_maximum_length():
    # For input [100] * 100, the partitions are all even:
    # All differences will be even since both sums will always be even.
    # Total even differences = 99 (there are 99 possible partitions)
    assert countPartitions([100] * 100) == 99
