from typing import List
from solution import maximumCoins

def test_canonical_example_1():
    # spec: coins = [[8,10,1],[1,3,2],[5,6,4]], k = 4
    # Selecting bags at positions [3, 4, 5, 6] gives 2 + 0 + 4 + 4 = 10
    assert maximumCoins([[8,10,1],[1,3,2],[5,6,4]], 4) == 10

def test_canonical_example_2():
    # spec: coins = [[1,10,3]], k = 2
    # Selecting bags at positions [1, 2] gives 3 + 3 = 6
    assert maximumCoins([[1,10,3]], 2) == 6

def test_single_bag_k_1():
    # Test for k == 1
    # spec: coins = [[2, 4, 7]], k = 1
    # Selecting bag at position [2] gives 7
    assert maximumCoins([[2, 4, 7]], 1) == 7

def test_large_k_with_zero_coin_bags():
    # Test for k == 10**9 with a narrow segment
    # spec: coins = [[5, 5, 3]], k = 10**9
    # Selecting bags from 1 to 10**9 will only count bag at position 5 which has 3 coins
    assert maximumCoins([[5, 5, 3]], 10**9) == 3

def test_single_coordinate_segment_upper_bounds():
    # Test single-coordinate segment at the coordinate and coin-value upper bounds
    # spec: coins = [[10**9, 10**9, 1000]], k = 1
    # Selecting bag at position [10**9] gives 1000
    assert maximumCoins([[10**9, 10**9, 1000]], 1) == 1000

def test_large_input():
    # Test with many non-overlapping segments
    # spec: coins = [[1, 1, 1], [2, 2, 2], ..., [100000, 100000, 1000]]
    # This will create 100000 bags with 1 coin each, so the total for k = 100000 is 100000
    coins = [[i, i, 1000] for i in range(1, 100001)]
    assert maximumCoins(coins, 100000) == 100000

def test_large_k_with_large_coins():
    # Test with large k and large coin values
    # spec: coins = [[1, 100, 1000]], k = 100
    # Selecting bags from position 1 to 100 will give us 1000*100 = 100000
    assert maximumCoins([[1, 100, 1000]], 100) == 100000
