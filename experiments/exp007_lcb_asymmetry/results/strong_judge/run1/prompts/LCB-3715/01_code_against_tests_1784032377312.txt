```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a mapping of coin counts across the number line
    from collections import defaultdict
    coin_map = defaultdict(int)
    
    # Fill the coin_map with the coins in the specified ranges
    for l, r, c in coins:
        for i in range(l, r + 1):
            coin_map[i] += c
    
    # Now we need to find the maximum coins we can collect from k consecutive bags
    max_coins = 0
    current_coins = 0
    
    # Traverse the coin_map and compute the sum of coins for the first k bags
    sorted_positions = sorted(coin_map.keys())
    left_index = 0
    right_index = 0
    bag_count = 0
    
    while right_index < len(sorted_positions):
        # Add coins from the current right position
        current_coins += coin_map[sorted_positions[right_index]]
        bag_count += 1
        right_index += 1
        
        # If we have more than k bags, slide the window
        while bag_count > k:
            current_coins -= coin_map[sorted_positions[left_index]]
            bag_count -= 1
            left_index += 1
        
        # Update max_coins if we have exactly k bags
        if bag_count == k:
            max_coins = max(max_coins, current_coins)

    return max_coins

```
