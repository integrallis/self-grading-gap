from typing import List


def minMaxSums(nums: List[int], k: int) -> int:
    MOD = 10**9 + 7
    n = len(nums)
    total_sum = 0

    # Sort the array to easily calculate contributions
    nums.sort()

    # Calculate contributions from subsequences of all sizes from 1 to k
    for size in range(1, k + 1):
        # Each number contributes to the total as min and max
        for i in range(n):
            # Each number is the max in combinations of size-1 from the remaining
            max_contrib = nums[i] * (1 << (n - 1 - i)) % MOD
            # Each number is the min in combinations of size-1 from the preceding numbers
            min_contrib = nums[i] * (1 << i) % MOD
            # Add to total sum
            total_sum = (total_sum + max_contrib + min_contrib) % MOD

    return total_sum

```
