```python
from typing import List


def minMaxSums(nums: List[int], k: int) -> int:
    MOD = 10**9 + 7
    n = len(nums)
    total_sum = 0

    # Calculate contributions from subsequences of all sizes from 1 to k
    for size in range(1, k + 1):
        # Calculate the number of times each number is the min and max
        for num in nums:
            # Each number is the max in combinations of size-1 from the remaining
            max_contrib = num * (1 << (n - size)) % MOD
            # Each number is the min in combinations of size-1 from the remaining
            min_contrib = num * (1 << (n - size)) % MOD
            total_sum = (total_sum + max_contrib + min_contrib) % MOD

    return total_sum

```
