from solution import maxDistance

def test_canonical_example_1():
    # Input: "NWSE", k = 1
    # Explanation: Change s[2] from 'S' to 'N'. The string s becomes "NWNE".
    # Maximum Manhattan distance achieved is 3.
    assert maxDistance("NWSE", 1) == 3

def test_canonical_example_2():
    # Input: "NSWWEW", k = 3
    # Explanation: Change s[1] from 'S' to 'N', and s[4] from 'E' to 'W'.
    # The string s becomes "NNWWWW". Maximum Manhattan distance is 6.
    assert maxDistance("NSWWEW", 3) == 6

def test_minimum_length_zero_edit():
    # Input: "N", k = 0
    # Explanation: No changes allowed, maximum Manhattan distance is 1.
    assert maxDistance("N", 0) == 1

def test_zero_edit_cancellation():
    # Input: "NS", k = 0
    # Explanation: No changes allowed, maximum Manhattan distance is 1.
    assert maxDistance("NS", 0) == 1

def test_full_edit_budget_boundary():
    # Input: "NS", k = 2
    # Explanation: Change both 'N' and 'S' to 'E' or 'W', maximum distance is 2.
    assert maxDistance("NS", 2) == 2

def test_maximum_length_constraint():
    # Input: "N" * 100000, k = 0
    # Explanation: No changes allowed, maximum Manhattan distance is 100000.
    assert maxDistance("N" * 100000, 0) == 100000
