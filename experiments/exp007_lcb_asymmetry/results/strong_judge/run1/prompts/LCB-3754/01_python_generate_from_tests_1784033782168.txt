```python
def maxDistance(s: str, k: int) -> int:
    direction_map = {'N': (0, 1), 'S': (0, -1), 'E': (1, 0), 'W': (-1, 0)}
    x, y = 0, 0
    max_distance = 0

    for move in s:
        dx, dy = direction_map[move]
        x += dx
        y += dy
        current_distance = abs(x) + abs(y)
        max_distance = max(max_distance, current_distance)

    # Calculate the maximum distance we can achieve by changing up to k moves
    changes_possible = min(k, len(s))
    max_possible_distance = max_distance + changes_possible

    return max(max_distance, max_possible_distance)

```
