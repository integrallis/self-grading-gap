## Implementation

```python
def maxDistance(s: str, k: int) -> int:
    direction_map = {'N': (0, 1), 'S': (0, -1), 'E': (1, 0), 'W': (-1, 0)}
    x, y = 0, 0
    distances = []

    for move in s:
        dx, dy = direction_map[move]
        x += dx
        y += dy
        distances.append(abs(x) + abs(y))

    max_distance = max(distances) if distances else 0
    max_distance_with_changes = min(len(s), k) + max_distance

    return max(max_distance, max_distance_with_changes)
```
