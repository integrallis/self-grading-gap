from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    dp = [[0] * 3 for _ in range(n)]
    # Fill in the first pair of houses
    for j in range(3):
        dp[0][j] = cost[0][j]
        dp[1][j] = float('inf')  # We will calculate these later

    for i in range(2, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(
                dp[i-1][(j+1) % 3],
                dp[i-1][(j+2) % 3]
            )

    # Handle equidistant constraints
    min_cost = float('inf')
    for j in range(3):
        for k in range(3):
            if j != k:
                # In between paired selections: j for the front end and k for the back end
                current_cost = dp[n-1][j] + cost[n // 2 - 1][k]
                # Consider minimum valid pairing costs
                min_cost = min(min_cost, current_cost)

    return min_cost
```
