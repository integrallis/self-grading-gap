from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = [1] * n
    max_length = 1

    for j in range(n):
        current_diff = float('inf')
        for i in range(j):
            diff = abs(nums[j] - nums[i])
            if diff <= current_diff:
                dp[j] = max(dp[j], dp[i] + 1)
                current_diff = diff
        max_length = max(max_length, dp[j])

    return max_length
```
