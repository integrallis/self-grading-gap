from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = [1] * n
    diffs = [[0] * n for _ in range(n)]

    # Calculate the absolute differences
    for i in range(n):
        for j in range(i + 1, n):
            diffs[i][j] = abs(nums[j] - nums[i])

    # Find the longest subsequence with non-increasing differences
    for j in range(n):
        for i in range(j):
            if diffs[i][j] >= diffs[i][i + 1]:
                dp[j] = max(dp[j], dp[i] + 1)

    return max(dp)
```
