from solution import maxDifference

def test_canonical_example_1():
    # For input "aaaaabbc", frequencies are: a: 5 (odd), b: 2 (even)
    # Maximum difference = 5 - 2 = 3
    assert maxDifference("aaaaabbc") == 3

def test_canonical_example_2():
    # For input "abcabcab", frequencies are: a: 3 (odd), b: 2 (even), c: 2 (even)
    # Maximum difference = 3 - 2 = 1
    assert maxDifference("abcabcab") == 1

def test_multiple_unequal_frequencies():
    # For input "aaaaabbbccdddd", frequencies are: a: 5 (odd), b: 3 (odd), c: 2 (even), d: 4 (even)
    # Maximum difference = 5 (odd) - 2 (even) = 3
    assert maxDifference("aaaaabbbccdddd") == 3

def test_negative_result():
    # For input "abbbbcc", frequencies are: a: 1 (odd), b: 4 (even), c: 2 (even)
    # Maximum difference = 1 (odd) - 2 (even) = -1
    assert maxDifference("abbbbcc") == -1

def test_minimum_length():
    # For input "aab", frequencies are: a: 2 (even), b: 1 (odd)
    # Maximum difference = 1 (odd) - 2 (even) = -1
    assert maxDifference("aab") == -1

def test_maximum_length():
    # For input "a" * 98 + "bc", frequencies are: a: 98 (even), b: 1 (odd), c: 1 (odd)
    # Maximum difference = 1 (odd) - 2 (even) = -97
    assert maxDifference("a" * 98 + "bc") == -97
