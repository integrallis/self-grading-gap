from typing import List
from solution import countMentions

def test_canonical_example_1():
    # Input: numberOfUsers = 2, events = [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","71","HERE"]]
    # At timestamp 10, id1 and id0 are mentioned. mentions = [1,1]
    # At timestamp 11, id0 goes offline.
    # At timestamp 71, id0 comes back online and "HERE" is mentioned. mentions = [2,2]
    assert countMentions(2, [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","71","HERE"]]) == [2, 2]

def test_canonical_example_2():
    # Input: numberOfUsers = 2, events = [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","12","ALL"]]
    # At timestamp 10, id1 and id0 are mentioned. mentions = [1,1]
    # At timestamp 11, id0 goes offline.
    # At timestamp 12, "ALL" is mentioned. Both id0 and id1 are mentioned. mentions = [2,2]
    assert countMentions(2, [["MESSAGE","10","id1 id0"],["OFFLINE","11","0"],["MESSAGE","12","ALL"]]) == [2, 2]

def test_canonical_example_3():
    # Input: numberOfUsers = 2, events = [["OFFLINE","10","0"],["MESSAGE","12","HERE"]]
    # At timestamp 10, id0 goes offline.
    # At timestamp 12, "HERE" is mentioned. id0 is offline, so mentions = [0,1]
    assert countMentions(2, [["OFFLINE","10","0"],["MESSAGE","12","HERE"]]) == [0, 1]

    # Input: numberOfUsers = 2, events = [["MESSAGE", "10", "HERE"], ["OFFLINE", "10", "0"]]
    # The OFFLINE status change at timestamp 10 is processed before the HERE message at timestamp 10, so only id1 is online. mentions = [0, 1]
    assert countMentions(2, [["MESSAGE", "10", "HERE"], ["OFFLINE", "10", "0"]]) == [0, 1]

    assert countMentions(2, [["MESSAGE", "12", "HERE"], ["OFFLINE", "10", "0"]]) == [0, 1]

    # At timestamp 61, id0 and id1 are offline, id2 is online, so mentions = [1, 0, 1]

    assert countMentions(3, [["OFFLINE","1","0"],["OFFLINE","2","1"],["MESSAGE","61","HERE"],["MESSAGE","62","HERE"]]) == [2, 1, 2]
