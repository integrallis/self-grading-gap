from solution import maximumAmount


def test_canonical_example_1():
    # Path 0 -> 1 -> neutralize -2 -> 3 -> 4 totals 0 + 1 + 0 + 3 + 4 = 8.
    assert maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) == 8


def test_canonical_example_2():
    # A path contains four cells worth 10 coins each: 10 + 10 + 10 + 10 = 40.
    assert maximumAmount([[10, 10, 10], [10, 10, 10]]) == 40


def test_single_robber_cell_can_be_neutralized():
    # The only path visits -7, and neutralizing that robber yields 0 coins.
    assert maximumAmount([[-7]]) == 0


def test_two_neutralizations_leave_remaining_robber_loss():
    # The path total is -5 - 1 - 10 - 2 = -18; neutralizing -10 and -5 gives -3.
    assert maximumAmount([[-5, -1, -10, -2]]) == -3


def test_optimal_route_and_neutralization_are_chosen_together():
    # Right, right, down gives 0 - 100 + 10 + 5 = -85; neutralizing -100 yields 15.
    assert maximumAmount([[0, -100, 10], [5, -1, 5]]) == 15


def test_maximum_grid_dimensions_all_zeroes():
    # Every path visits only zero-valued cells, so the total is 0.
    coins = [[0] * 500 for _ in range(500)]
    assert maximumAmount(coins) == 0


def test_coin_value_bounds_with_negative_bound_neutralized():
    # Neutralizing -1000 leaves 1000 + 0 + 1000 = 2000.
    assert maximumAmount([[1000, -1000, 1000]]) == 2000


def test_single_column_down_only_path():
    # The only path is 1 -> -5 -> 2; neutralizing -5 gives 1 + 0 + 2 = 3.
    assert maximumAmount([[1], [-5], [2]]) == 3
