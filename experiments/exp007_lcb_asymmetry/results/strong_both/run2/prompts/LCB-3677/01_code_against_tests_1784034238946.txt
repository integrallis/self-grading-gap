```python
from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    # DP array to store maximum coins up to (i, j)
    dp = [[-float('inf')] * (n + 1) for _ in range(m + 1)]
    dp[0][0] = 0

    for i in range(m):
        for j in range(n):
            # Directly adding coins from the current cell without neutralization
            current_value = coins[i][j]
            # If current cell is negative, this means a robber is present
            if current_value < 0:
                dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j] - current_value)  # neutralize if possible
            else:
                dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j] + current_value)
            # Propagate from the left and top cells
            if i > 0:
                dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j + 1])  # coming from above
            if j > 0:
                dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i + 1][j])  # coming from left

    # Returning the maximum profit at the bottom-right corner
    return dp[m][n]
```
