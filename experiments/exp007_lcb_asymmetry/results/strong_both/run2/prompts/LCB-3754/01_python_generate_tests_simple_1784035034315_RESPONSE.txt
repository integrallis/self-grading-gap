from solution import maxDistance


def test_canonical_example_1():
    # Change "NWSE" to "NWNE"; its prefix distances are 1, 2, 3, 2, so max is 3.
    assert maxDistance("NWSE", 1) == 3


def test_canonical_example_2():
    # Change "NSWWEW" to "NNWWWW"; after all 6 moves, distance is 6.
    assert maxDistance("NSWWEW", 3) == 6


def test_no_changes_uses_the_best_prefix_not_just_final_position():
    # "NNSS" has prefix distances 1, 2, 1, 0; the maximum is 2.
    assert maxDistance("NNSS", 0) == 2


def test_one_change_can_turn_an_opposite_move_into_a_favorable_move():
    # Change "NSEW" to "NNEW"; prefix distances become 1, 2, 3, 2, so max is 3.
    assert maxDistance("NSEW", 1) == 3


def test_enough_changes_can_make_every_move_in_one_direction():
    # Change "SWNE" to "NNNN"; after 4 moves, Manhattan distance is 4.
    assert maxDistance("SWNE", 4) == 4


def test_manhattan_distance_can_combine_north_and_east_progress():
    # With no changes, "NE" reaches (1, 1), whose Manhattan distance is |1| + |1| = 2.
    assert maxDistance("NE", 0) == 2
