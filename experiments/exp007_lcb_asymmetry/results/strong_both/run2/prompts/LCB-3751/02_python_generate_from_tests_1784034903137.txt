from typing import List


def maxFrequency(nums: List[int], k: int) -> int:
    count = [0] * 51  # since nums[i] and k are between 1 and 50
    for num in nums:
        count[num] += 1

    max_freq = count[k]  # initial frequency of k
    for x in range(1, 51):  # check every possible number that can be converted
        if x == k:
            continue
        needed_addition = k - x
        if 1 <= needed_addition <= 50:
            max_freq = max(max_freq, count[x] + count[k])

    return max_freq
```
