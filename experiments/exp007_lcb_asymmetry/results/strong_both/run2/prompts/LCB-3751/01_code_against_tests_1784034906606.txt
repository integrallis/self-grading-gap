```python
from typing import List


def maxFrequency(nums: List[int], k: int) -> int:
    count = [0] * 51  # since nums[i] and k are between 1 and 50
    for num in nums:
        count[num] += 1

    max_freq = count[k]  # initial frequency of k
    total_sum = 0
    left = 0
    n = len(nums)
    for right in range(n):
        total_sum += nums[right]
        while total_sum + (right - left + 1) * k > (right - left + 1) * k:
            total_sum -= nums[left]
            left += 1
        max_freq = max(max_freq, right - left + 1)

    return max_freq
```
