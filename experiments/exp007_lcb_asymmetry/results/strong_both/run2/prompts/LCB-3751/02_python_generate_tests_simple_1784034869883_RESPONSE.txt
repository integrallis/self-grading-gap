from solution import maxFrequency


def test_canonical_example_1():
    # Adding -5 to nums[2..5] changes the final 6 to 1, so 1 appears twice.
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2


def test_canonical_example_2():
    # Adding 8 to nums[1..9] converts the three 2s to 10; with the initial 10, total is 4.
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4


def test_single_element_can_be_shifted_to_k():
    # Adding -49 to the only element changes 50 to 1, giving frequency 1.
    assert maxFrequency([50], 1) == 1


def test_zero_addition_preserves_an_array_already_equal_to_k():
    # Selecting any subarray and adding 0 keeps all three 5s, so the frequency remains 3.
    assert maxFrequency([5, 5, 5], 5) == 3


def test_one_shared_addition_cannot_convert_distinct_values_to_k():
    # A single x can map only one of 1, 2, and 3 to 4, so the best frequency is 1.
    assert maxFrequency([1, 2, 3], 4) == 1


def test_best_subarray_can_avoid_losing_too_many_existing_k_values():
    # Converting either adjacent pair of 2s to 1 adds 2 to the five existing 1s, for 7.
    # Covering both pairs would also shift the three middle 1s away from k.
    assert maxFrequency([1, 2, 2, 1, 1, 1, 2, 2, 1], 1) == 7


def test_maximum_length_and_upper_k_boundary():
    # Adding 0 preserves all 100000 occurrences of 50, so the frequency is 100000.
    assert maxFrequency([50] * 100000, 50) == 100000
