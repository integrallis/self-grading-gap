from solution import reverseDegree


def test_canonical_example_1():
    # 'a': 26 * 1, 'b': 25 * 2, 'c': 24 * 3; 26 + 50 + 72 = 148
    assert reverseDegree("abc") == 148


def test_canonical_example_2():
    # 'z': 1 * 1, 'a': 26 * 2, 'z': 1 * 3, 'a': 26 * 4;
    # 1 + 52 + 3 + 104 = 160
    assert reverseDegree("zaza") == 160


def test_single_character_minimum_length():
    # 'z' has reversed-alphabet position 1 and string position 1: 1 * 1 = 1
    assert reverseDegree("z") == 1


def test_maximum_length_all_a_characters():
    # Each 'a' contributes 26 * position; 26 * (1 + ... + 1000)
    # = 26 * (1000 * 1001 / 2) = 13,013,000
    assert reverseDegree("a" * 1000) == 13013000
