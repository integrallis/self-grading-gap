```python
from typing import List
import math

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    # Calculate contributions from subsequences of size 1 to k
    for size in range(1, k + 1):
        for i in range(n):
            # Each element contributes to the sum as both min and max
            contribution = (nums[i] * 2) % MOD
            # Count ways to choose (size-1) from (n-1) elements
            counts = math.comb(n - 1, size - 1) % MOD
            total_sum = (total_sum + contribution * counts) % MOD
    return total_sum

```
