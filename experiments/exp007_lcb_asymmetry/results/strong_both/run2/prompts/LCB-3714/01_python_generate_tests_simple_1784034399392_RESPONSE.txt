from solution import minMaxSums


def test_canonical_example_1():
    # [1], [2], [3], [1,2], [1,3], [2,3] contribute
    # 2 + 4 + 6 + 3 + 4 + 5 = 24.
    assert minMaxSums([1, 2, 3], 2) == 24


def test_canonical_example_2():
    # With k=1, each element contributes itself as both minimum and maximum:
    # 2*5 + 2*0 + 2*6 = 22.
    assert minMaxSums([5, 0, 6], 1) == 22


def test_canonical_example_3():
    # Three singleton subsequences contribute 2 each and three length-2
    # subsequences contribute 2 each: 3*2 + 3*2 = 12.
    assert minMaxSums([1, 1, 1], 2) == 12


def test_single_element_array():
    # The only non-empty subsequence is [7], contributing 7 + 7 = 14.
    assert minMaxSums([7], 1) == 14


def test_includes_subsequences_up_to_array_length():
    # For [1,2,3], all non-empty subsequences contribute:
    # singletons 2+4+6, pairs 3+4+5, and [1,2,3] contributes 1+3.
    # Total = 12 + 12 + 4 = 28.
    assert minMaxSums([1, 2, 3], 3) == 28


def test_returns_result_modulo_1000000007():
    # Three non-empty subsequences of [10^9, 10^9] contribute
    # 2*10^9 each, for 6,000,000,000 total.
    # 6,000,000,000 mod 1,000,000,007 = 999,999,965.
    assert minMaxSums([1_000_000_000, 1_000_000_000], 2) == 999_999_965
