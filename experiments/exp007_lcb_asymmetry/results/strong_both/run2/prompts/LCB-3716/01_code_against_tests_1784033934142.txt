```python
from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    # dp[i] will store the length of the longest subsequence ending at index i
    dp = [1] * n
    max_length = 1
    # This will store the previous absolute difference for comparison
    prev_diff = float('inf')

    for i in range(n):
        for j in range(i):
            diff = abs(nums[i] - nums[j])
            if diff <= prev_diff:
                dp[i] = max(dp[i], dp[j] + 1)
        prev_diff = diff
        max_length = max(max_length, dp[i])

    return max_length
```
