from solution import longestSubsequence


def test_canonical_example_1():
    # [16, 6, 3] has adjacent absolute differences [10, 3], and 10 >= 3.
    assert longestSubsequence([16, 6, 3]) == 3


def test_canonical_example_2():
    # The subsequence [6, 4, 2, 1] has differences [2, 2, 1], which are non-increasing.
    assert longestSubsequence([6, 5, 3, 4, 2, 1]) == 4


def test_canonical_example_3():
    # The subsequence [10, 20, 10, 19, 10] has differences [10, 10, 9, 9].
    assert longestSubsequence([10, 20, 10, 19, 10, 20]) == 5


def test_equal_values_can_form_a_full_subsequence():
    # All adjacent differences are 0, so [7, 7, 7, 7] has differences [0, 0, 0].
    assert longestSubsequence([7, 7, 7, 7]) == 4


def test_can_skip_elements_to_form_the_longest_subsequence():
    # Skipping 11 gives [1, 10, 2, 3] with differences [9, 8, 1].
    assert longestSubsequence([1, 10, 11, 2, 3]) == 4


def test_increasing_differences_do_not_extend_a_subsequence():
    # The only length-3 subsequence is [1, 2, 4], with increasing differences [1, 2];
    # any pair is valid, so the longest length is 2.
    assert longestSubsequence([1, 2, 4]) == 2


def test_two_elements_form_a_valid_subsequence():
    # A two-element subsequence has one absolute difference, which is always non-increasing.
    assert longestSubsequence([300, 1]) == 2
