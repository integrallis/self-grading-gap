from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n == 0:
        return 0

    # dp[i] will store the length of the longest subsequence ending at index i
    dp = [1] * n
    max_length = 1

    for i in range(n):
        for j in range(i):
            # Calculate the absolute difference
            diff1 = abs(nums[i] - nums[j])
            # The last difference in the subsequence formed by j
            if dp[j] > 0 and (i == 0 or diff1 >= abs(nums[j] - (nums[dp[j] - 1] if dp[j] > 1 else nums[j]))):
                dp[i] = max(dp[i], dp[j] + 1)
                max_length = max(max_length, dp[i])

    return max_length

```
