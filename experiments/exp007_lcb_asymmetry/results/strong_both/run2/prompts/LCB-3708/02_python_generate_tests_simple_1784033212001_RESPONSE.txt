from solution import zigzagTraversal


def test_canonical_example_1():
    # Zigzag order is [1, 2, 4, 3]; keeping alternate cells gives [1, 4].
    assert zigzagTraversal([[1, 2], [3, 4]]) == [1, 4]


def test_canonical_example_2():
    # Zigzag order is [2, 1, 1, 2, 2, 1]; keeping alternate cells gives [2, 1, 2].
    assert zigzagTraversal([[2, 1], [2, 1], [2, 1]]) == [2, 1, 2]


def test_canonical_example_3():
    # Zigzag order is [1, 2, 3, 6, 5, 4, 7, 8, 9]; keeping alternate cells gives [1, 3, 5, 7, 9].
    assert zigzagTraversal([[1, 2, 3], [4, 5, 6], [7, 8, 9]]) == [1, 3, 5, 7, 9]


def test_maximum_row_count_with_maximum_visited_value():
    grid = [[2 * row + 1, 2 * row + 2] for row in range(50)]
    grid[48][0] = 2500

    # Each 2-cell row contributes its first zigzag cell: column 0 on even rows and column 1 on odd rows.
    expected = [
        grid[row][0] if row % 2 == 0 else grid[row][1]
        for row in range(50)
    ]

    assert zigzagTraversal(grid) == expected


def test_maximum_column_count_across_two_rows():
    grid = [list(range(1, 51)), list(range(51, 101))]

    # Zigzag order is [1..50, 100..51]; keeping alternate positions gives odd values from
    # the first row followed by [100, 98, ..., 52] from the reversed second row.
    expected = list(range(1, 51, 2)) + list(range(100, 51, -2))

    assert zigzagTraversal(grid) == expected
