```python
from typing import List

def maxSum(nums: List[int]) -> int:
    unique_values = set()  # To store unique elements
    max_sum = 0  # To keep track of the maximum sum, start with 0
    for num in nums:
        if num not in unique_values:
            unique_values.add(num)  # Add to set if unique
            if num > 0:
                max_sum += num  # Sum positive unique values
            elif max_sum == 0:  # Keep track of the largest negative if all are negative
                max_sum = num if max_sum < num else max_sum
    return max_sum
```
