from typing import List

def maxSum(nums: List[int]) -> int:
    unique_values = set()  # To store unique elements
    max_sum = float('-inf')  # To keep track of the maximum sum
    for num in nums:
        if num not in unique_values:
            unique_values.add(num)  # Add to set if unique
            # Update max_sum only if it is not negative
            if num > 0:
                max_sum += num
            elif max_sum == float('-inf'):
                max_sum = num
    return max_sum if max_sum != float('-inf') else 0
```
