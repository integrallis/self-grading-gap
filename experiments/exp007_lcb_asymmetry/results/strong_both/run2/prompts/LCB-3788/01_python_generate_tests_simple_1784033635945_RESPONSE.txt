from solution import maxSum


def test_canonical_example_1():
    # All values are unique, so the whole array sums to 1 + 2 + 3 + 4 + 5 = 15.
    assert maxSum([1, 2, 3, 4, 5]) == 15


def test_canonical_example_2():
    # After deletions, selecting one unique value [1] gives the maximum sum of 1.
    assert maxSum([1, 1, 0, 1, 1]) == 1


def test_canonical_example_3():
    # Delete negative values and duplicates to select unique values [2, 1], summing to 3.
    assert maxSum([1, 2, -1, -2, 1, 0, -1]) == 3


def test_selects_each_distinct_positive_value_once():
    # The best unique selection contains 3 and 2, so the sum is 3 + 2 = 5.
    assert maxSum([3, -1, 3, 2, -2, 2]) == 5


def test_returns_zero_when_zero_is_better_than_negative_values():
    # A nonempty unique selection [0] has sum 0, which is greater than -1.
    assert maxSum([0, 0, -1]) == 0


def test_returns_largest_value_when_all_values_are_negative():
    # Only one value can be selected to maximize the sum; max(-4, -2, -7) = -2.
    assert maxSum([-4, -2, -7]) == -2
