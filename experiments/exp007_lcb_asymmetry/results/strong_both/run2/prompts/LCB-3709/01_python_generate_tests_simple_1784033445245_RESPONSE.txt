from solution import hasSpecialSubstring


def test_canonical_example_1():
    # "aaa" at the end has length 3 and is preceded by different character "b".
    assert hasSpecialSubstring("aaabaaa", 3) is True


def test_canonical_example_2():
    # No length-2 substring of "abc" consists of one repeated character.
    assert hasSpecialSubstring("abc", 2) is False


def test_exact_length_run_between_different_characters_is_special():
    # "bbb" has exactly length 3 and is bounded by different characters "a" and "c".
    assert hasSpecialSubstring("abbbc", 3) is True


def test_longer_run_does_not_contain_special_substring_of_shorter_length():
    # Each "aaa" length-3 candidate is adjacent to another "a", so none is special.
    assert hasSpecialSubstring("aaaa", 3) is False


def test_entire_string_can_be_special_substring():
    # "aaaa" has length 4 and has no neighboring characters that could violate the condition.
    assert hasSpecialSubstring("aaaa", 4) is True


def test_single_character_substring_requires_different_neighbors():
    # The middle "b" has length 1 and both adjacent characters are different "a".
    assert hasSpecialSubstring("aba", 1) is True


def test_single_character_inside_run_is_not_special():
    # Every length-1 "a" candidate has an adjacent "a", so none is special.
    assert hasSpecialSubstring("aa", 1) is False


def test_can_find_special_run_after_nonmatching_runs():
    # "cc" has exactly length 2 and is bounded by different characters "b" and "d".
    assert hasSpecialSubstring("aabccdde", 2) is True


def test_runs_of_wrong_lengths_do_not_match():
    # Runs have lengths 2 ("aa"), 1 ("b"), and 3 ("ccc"), so no run has exactly length 4.
    assert hasSpecialSubstring("aabccc", 4) is False
