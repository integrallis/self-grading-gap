from solution import sumOfGoodNumbers


def test_canonical_example_1():
    # Good elements are 3 (index 1), 5 (index 4), and 4 (index 5): 3 + 5 + 4 = 12.
    assert sumOfGoodNumbers([1, 3, 2, 1, 5, 4], 2) == 12


def test_canonical_example_2():
    # Only 2 (index 0) is greater than its existing k-distance neighbor: 2.
    assert sumOfGoodNumbers([2, 1], 1) == 2


def test_equal_k_distance_neighbors_are_not_good():
    # Neither 3 is strictly greater than the other; boundary elements also fail: sum = 0.
    assert sumOfGoodNumbers([1, 3, 3, 1], 1) == 0
