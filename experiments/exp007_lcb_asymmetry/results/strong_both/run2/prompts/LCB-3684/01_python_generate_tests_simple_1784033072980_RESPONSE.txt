from solution import hasMatch


def test_canonical_example_1():
    # "ee*e" can replace "*" with "tcod" to match substring "eetcode".
    assert hasMatch("leetcode", "ee*e") is True


def test_canonical_example_2():
    # No replacement for "*" makes "c*v" match a substring of "car".
    assert hasMatch("car", "c*v") is False


def test_canonical_example_3():
    # Replacing "*" with "" makes "u*" match the substring "u".
    assert hasMatch("luck", "u*") is True


def test_wildcard_can_match_zero_characters_between_fixed_parts():
    # Replacing "*" with "" makes "a*b" equal "ab", which is a substring of "zabz".
    assert hasMatch("zabz", "a*b") is True


def test_fixed_parts_must_appear_in_order_within_one_substring():
    # "a" occurs after "b" in "ba", so no substring can match "a*b".
    assert hasMatch("ba", "a*b") is False


def test_wildcard_can_match_characters_between_fixed_parts():
    # Replacing "*" with "xy" makes "ab*cd" equal substring "abxycd".
    assert hasMatch("zzabxycdzz", "ab*cd") is True


def test_pattern_with_only_wildcard_matches_a_nonempty_substring():
    # Replacing "*" with "a" makes the pattern match the non-empty substring "a".
    assert hasMatch("abc", "*") is True


def test_pattern_ending_with_wildcard_requires_its_prefix():
    # No substring of "bbb" starts with "a", so "a*" cannot match.
    assert hasMatch("bbb", "a*") is False


def test_pattern_starting_with_wildcard_can_match_suffix_at_end():
    # Replacing "*" with "ab" makes "*c" equal the substring "abc".
    assert hasMatch("zabc", "*c") is True
