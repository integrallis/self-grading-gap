## Implementation

```python
def hasMatch(s: str, p: str) -> bool:
    # Split the pattern into two parts using '*'
    fixed_start, fixed_end = p.split('*')
    # Check if the fixed start can be found in the string
    start_index = s.find(fixed_start)
    while start_index != -1:
        # Check if the end part can be found after the start_index
        if s.find(fixed_end, start_index + len(fixed_start)) != -1:
            return True
        # Look for the next occurrence of fixed_start
        start_index = s.find(fixed_start, start_index + 1)
    return False

```
