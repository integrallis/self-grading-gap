## Implementation

```python
def findValidPair(s: str) -> str:
    from collections import Counter

    # Count occurrences of each digit in the string
    counts = Counter(s)

    # Iterate over the string looking for valid pairs
    for i in range(len(s) - 1):
        first = s[i]
        second = s[i + 1]
        # Check the pair is valid
        if first != second:
            if counts[first] == int(first) and counts[second] == int(second):
                return first + second
    return ""
```
