from solution import findValidPair


def test_canonical_example_1():
    # "23" is the first valid adjacent pair: '2' appears 2 times and '3' appears 3 times.
    assert findValidPair("2523533") == "23"


def test_canonical_example_2():
    # "21" is valid because '2' appears 2 times and '1' appears 1 time.
    assert findValidPair("221") == "21"


def test_canonical_example_3():
    # The only adjacent pair is "22", whose digits are equal, so no valid pair exists.
    assert findValidPair("22") == ""


def test_returns_leftmost_when_multiple_valid_pairs_exist():
    # In "122333", both "12" and later "23" are valid; the first valid pair is "12".
    assert findValidPair("122333") == "12"


def test_returns_empty_when_adjacent_digits_do_not_have_required_global_counts():
    # In "123", '1' appears once, but '2' and '3' also each appear once, not 2 and 3 times.
    assert findValidPair("123") == ""
