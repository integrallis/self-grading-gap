## Implementation

```python
from typing import List
import math


def maxLength(nums: List[int]) -> int:
    n = len(nums)
    max_length = 0

    for start in range(n):
        product = 1
        g = nums[start]
        l = nums[start]
        for end in range(start, n):
            product *= nums[end]
            g = math.gcd(g, nums[end])
            l = (l * nums[end]) // math.gcd(l, nums[end])

            if product == g * l:
                max_length = max(max_length, end - start + 1)

    return max_length

```
