from solution import maxLength


def test_canonical_example_1():
    # [1, 2, 1, 1, 1] has product 2, gcd 1, and lcm 2, so it is product equivalent.
    # Every length-6 subarray has product 4 but gcd * lcm = 2.
    assert maxLength([1, 2, 1, 2, 1, 1, 1]) == 5


def test_canonical_example_2():
    # [3, 4, 5] has product 60, gcd 1, and lcm 60, so it is product equivalent.
    # Neither length-4 subarray is product equivalent.
    assert maxLength([2, 3, 4, 5, 6]) == 3


def test_canonical_example_3():
    # [3, 1, 4, 5, 1] has product 60, gcd 1, and lcm 60, so it is product equivalent.
    # Both length-6 subarrays have product 120 but gcd * lcm = 60.
    assert maxLength([1, 2, 3, 1, 4, 5, 1]) == 5


def test_minimum_length_with_maximum_element():
    # [10, 10] has product 100, gcd 10, and lcm 10, so gcd * lcm is 100.
    assert maxLength([10, 10]) == 2


def test_maximum_length_all_ones():
    # For 100 ones, product, gcd, and lcm are all 1, so product = gcd * lcm.
    assert maxLength([1] * 100) == 100
