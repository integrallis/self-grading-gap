```python
from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    intervals = []

    # Collect intervals of meetings
    for i in range(n):
        intervals.append((startTime[i], endTime[i]))

    # Calculate free time gaps
    free_gaps = []
    if intervals:
        # Free time before the first meeting
        if intervals[0][0] > 0:
            free_gaps.append(intervals[0][0])

        # Free time between meetings
        for i in range(1, n):
            free_time = intervals[i][0] - intervals[i-1][1]
            if free_time > 0:
                free_gaps.append(free_time)

        # Free time after the last meeting
        if intervals[-1][1] < eventTime:
            free_gaps.append(eventTime - intervals[-1][1])

    # Sort the free gaps descending
    free_gaps.sort(reverse=True)

    # We can remove the duration of at most `k` meetings to maximize gaps
    total_free_time = sum(free_gaps)
    if k > 0 and n > 0:
        # We can only merge `k` gaps into the largest free gaps
        for i in range(min(k, n)):
            if i < len(free_gaps):
                total_free_time += intervals[i][1] - intervals[i][0]

    return total_free_time
```
