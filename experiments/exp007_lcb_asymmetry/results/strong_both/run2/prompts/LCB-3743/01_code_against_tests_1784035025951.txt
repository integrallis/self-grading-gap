```python
from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    total_free_time = 0
    merged_start = [0]
    merged_end = [0]

    # Collect all the end times and start times based on the constraints
    for i in range(n):
        merged_start.append(startTime[i])
        merged_end.append(endTime[i])
    merged_start.append(eventTime)
    merged_end.append(eventTime)

    # Calculate initial free time before the first meeting or after the last meeting
    for i in range(1, n + 1):
        free_time = merged_start[i] - merged_end[i - 1]
        if free_time > 0:
            total_free_time += free_time

    # Prepare the list of intervals with their durations
    durations = [endTime[i] - startTime[i] for i in range(n)]

    # Create a pair list to represent possible free times based on moving meetings
    potential_gaps = []
    for i in range(1, n):
        gap = merged_start[i] - merged_end[i - 1]
        if gap > 0:
            potential_gaps.append((gap, durations[i-1]))  # store (gap size, duration)

    # Sort potential gaps based on the gap length descending
    potential_gaps.sort(reverse=True, key=lambda x: x[0])

    # Calculate the maximum free time by possibly merging k gaps
    for i in range(min(k, len(potential_gaps))):
        total_free_time += potential_gaps[i][1]  # add the duration of the meetings we can shift

    return total_free_time
```
