from solution import largestInteger


def test_canonical_example_1():
    # 3 and 7 each occur in one size-3 subarray; the larger value is 7.
    assert largestInteger([3, 9, 2, 1, 7], 3) == 7


def test_canonical_example_2():
    # Only 3 occurs in exactly one size-4 subarray.
    assert largestInteger([3, 9, 7, 2, 1, 7], 4) == 3


def test_canonical_example_3():
    # 0 occurs in both size-1 subarrays, so no value occurs in exactly one.
    assert largestInteger([0, 0], 1) == -1


def test_repeated_value_in_one_subarray_counts_as_one_subarray():
    # There is only one size-2 subarray, [5, 5], so 5 appears in exactly one subarray.
    assert largestInteger([5, 5], 2) == 5


def test_whole_array_as_window_returns_largest_distinct_value():
    # With one size-3 subarray, each present value appears in exactly one; largest is 9.
    assert largestInteger([2, 9, 4], 3) == 9


def test_single_element_array():
    # The only value appears in the only size-1 subarray.
    assert largestInteger([0], 1) == 0
