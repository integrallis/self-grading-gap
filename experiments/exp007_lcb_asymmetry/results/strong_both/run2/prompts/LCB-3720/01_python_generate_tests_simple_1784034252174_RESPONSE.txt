from solution import minMaxWeight


def test_canonical_example_1():
    # Nodes 1, 2, 3, and 4 can reach 0 using only weight-1 edges:
    # 1->0, 2->1, 3->0, and 4->3, so the minimum maximum is 1.
    assert minMaxWeight(
        5,
        [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]],
        2,
    ) == 1


def test_canonical_example_2():
    # Node 2 has no directed path to node 0, so satisfying reachability is impossible.
    assert minMaxWeight(
        5,
        [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]],
        1,
    ) == -1


def test_canonical_example_3():
    # Keeping 1->2 (1), 2->3 (2), 3->4 (2), and 4->0 (1)
    # lets every node reach 0 with maximum weight 2.
    assert minMaxWeight(
        5,
        [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]],
        1,
    ) == 2


def test_canonical_example_4():
    # Node 3 has no directed path to node 0, so satisfying reachability is impossible.
    assert minMaxWeight(
        5,
        [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]],
        1,
    ) == -1


def test_chooses_lower_weight_indirect_path_over_heavier_direct_edge():
    # Use 1->2 (3) and 2->0 (4) rather than 1->0 (9);
    # all nodes reach 0 and the resulting maximum is 4.
    assert minMaxWeight(
        3,
        [[1, 0, 9], [1, 2, 3], [2, 0, 4]],
        1,
    ) == 4


def test_removes_extra_outgoing_edge_to_meet_threshold():
    # With threshold 1, keep 1->2, 2->0, and 3->0, all of weight 1.
    # Removing 1->0 leaves every node able to reach 0, so the minimum maximum is 1.
    assert minMaxWeight(
        4,
        [[1, 0, 5], [1, 2, 1], [2, 0, 1], [3, 0, 1]],
        1,
    ) == 1


def test_multiple_edges_between_same_nodes_can_use_the_lighter_one():
    # Keep 1->0 with weight 3 and 2->0 with weight 4;
    # node 2 requires weight 4, so the minimum possible maximum is 4.
    assert minMaxWeight(
        3,
        [[1, 0, 8], [1, 0, 3], [2, 0, 4]],
        1,
    ) == 4
