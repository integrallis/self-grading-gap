from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    from collections import defaultdict, deque

    # Build the reverse graph
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[v].append((u, w))  # Reverse the direction

    # Function to check if we can reach node 0 with a certain max weight
    def canReachMaxWeight(maxWeight: int) -> bool:
        outgoing_count = defaultdict(int)
        queue = deque([0])
        visited = set([0])

        while queue:
            node = queue.popleft()
            for neighbor, weight in graph[node]:
                if weight <= maxWeight and outgoing_count[node] < threshold:
                    outgoing_count[node] += 1
                    if neighbor not in visited:
                        visited.add(neighbor)
                        queue.append(neighbor)
        return len(visited) == n

    # Binary search for the minimum possible maximum edge weight
    left, right = 1, 10**6
    result = -1

    while left <= right:
        mid = (left + right) // 2
        if canReachMaxWeight(mid):
            result = mid
            right = mid - 1
        else:
            left = mid + 1

    return result

```
