from solution import transformArray


def test_canonical_example_1():
    # 4 and 2 become 0; 3 and 1 become 1; sorting gives [0, 0, 1, 1].
    assert transformArray([4, 3, 2, 1]) == [0, 0, 1, 1]


def test_canonical_example_2():
    # 4 and 2 become 0; 1, 5, and 1 become 1; sorting gives [0, 0, 1, 1, 1].
    assert transformArray([1, 5, 1, 4, 2]) == [0, 0, 1, 1, 1]


def test_single_element():
    # 1 is odd, so it becomes 1; a one-element array is already sorted.
    assert transformArray([1]) == [1]


def test_maximum_length_and_value():
    # Each of the 100 values is 1000, an even number, so all become 0 and remain sorted.
    assert transformArray([1000] * 100) == [0] * 100
