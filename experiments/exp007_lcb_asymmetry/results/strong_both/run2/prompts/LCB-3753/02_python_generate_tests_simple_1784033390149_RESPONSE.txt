from solution import maxDifference


def test_canonical_example_1():
    # 'a' occurs 5 times (odd) and 'b' occurs 2 times (even), so 5 - 2 = 3.
    assert maxDifference("aaaaabbc") == 3


def test_canonical_example_2():
    # 'a' occurs 3 times (odd) and 'c' occurs 2 times (even), so 3 - 2 = 1.
    assert maxDifference("abcabcab") == 1


def test_minimum_length_can_return_negative_difference():
    # 'a' occurs 1 time (odd) and 'b' occurs 2 times (even), so 1 - 2 = -1.
    assert maxDifference("abb") == -1


def test_chooses_smallest_frequency_among_multiple_even_characters():
    # 'a' occurs 5 times (odd); 'b' and 'c' occur 2 and 4 times (even).
    # The maximum difference is 5 - min(2, 4) = 5 - 2 = 3.
    assert maxDifference("aaaaabbcccc") == 3


def test_maximum_length_input():
    # Length is 97 + 2 + 1 = 100. 'a' occurs 97 times (odd) and 'b' occurs
    # 2 times (even), so 97 - 2 = 95.
    assert maxDifference("a" * 97 + "bb" + "c") == 95
