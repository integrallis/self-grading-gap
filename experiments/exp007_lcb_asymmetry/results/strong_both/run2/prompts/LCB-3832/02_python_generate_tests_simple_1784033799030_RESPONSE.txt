from solution import minCosts


def test_canonical_example_1():
    # Prefix minima: [5, min(5, 3), min(3, 4), min(3, 1), min(1, 3), min(1, 2)].
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]


def test_canonical_example_2():
    # Paying 1 for person 0 makes every later position reachable for free.
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]


def test_single_person_with_maximum_cost():
    # With one person, reaching position 0 requires paying that person's cost: 100.
    assert minCosts([100]) == [100]


def test_maximum_length_all_maximum_costs():
    # Every prefix contains only cost 100, so each of the 100 answers is 100.
    assert minCosts([100] * 100) == [100] * 100


def test_non_example_prefix_minimum_updates_and_ties():
    # Prefix minima are [4, min(4,2), min(2,5), min(2,2), min(2,1)].
    assert minCosts([4, 2, 5, 2, 1]) == [4, 2, 2, 2, 1]
