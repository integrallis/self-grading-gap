from solution import countMentions


def test_canonical_example_1():
    events = [
        ["MESSAGE", "10", "id1 id0"],
        ["OFFLINE", "11", "0"],
        ["MESSAGE", "71", "HERE"],
    ]

    # At t=10 both users receive one explicit mention; at t=71 user 0 is online
    # again, so HERE adds one mention to each user: [1, 1] + [1, 1] = [2, 2].
    assert countMentions(2, events) == [2, 2]


def test_canonical_example_2():
    events = [
        ["MESSAGE", "10", "id1 id0"],
        ["OFFLINE", "11", "0"],
        ["MESSAGE", "12", "ALL"],
    ]

    # The explicit message gives [1, 1], and ALL mentions both users even though
    # user 0 is offline: [1, 1] + [1, 1] = [2, 2].
    assert countMentions(2, events) == [2, 2]


def test_canonical_example_3():
    events = [
        ["OFFLINE", "10", "0"],
        ["MESSAGE", "12", "HERE"],
    ]

    # User 0 remains offline at t=12, so HERE mentions only user 1: [0, 1].
    assert countMentions(2, events) == [0, 1]


def test_explicit_ids_count_duplicates_and_can_mention_offline_users():
    events = [
        ["OFFLINE", "5", "0"],
        ["MESSAGE", "6", "id0 id0 id1"],
    ]

    # Explicit id0 appears twice and is counted twice despite being offline;
    # id1 appears once, producing [2, 1].
    assert countMentions(2, events) == [2, 1]


def test_offline_status_is_processed_before_message_at_same_timestamp():
    events = [
        ["MESSAGE", "10", "HERE"],
        ["OFFLINE", "10", "0"],
    ]

    # The offline change at t=10 is processed before the t=10 message, so HERE
    # mentions only still-online user 1: [0, 1].
    assert countMentions(2, events) == [0, 1]
