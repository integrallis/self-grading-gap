from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # Create a 2D DP table
    dp = [[0] * 3 for _ in range(n)]
    # Initialize the first house's costs
    dp[0] = cost[0]

    for i in range(1, n):
        for j in range(3):
            # Calculate the cost for painting house i with color j
            dp[i][j] = cost[i][j] + min(dp[i - 1][(j + 1) % 3], dp[i - 1][(j + 2) % 3])

    # Calculate the minimum cost for the last house considering the mirroring condition
    total_cost = float('inf')
    for j in range(3):
        # Ensure the last house's color is not the same as the first house and its mirror
        total_cost = min(total_cost, dp[n - 1][j] + (cost[0][(j + 1) % 3] + cost[n - 1][(j + 2) % 3]))

    return total_cost
```
