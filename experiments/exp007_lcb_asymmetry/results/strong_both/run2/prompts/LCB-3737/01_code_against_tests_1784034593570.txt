```python
from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # Create a 2D DP table
    dp = [[0] * 3 for _ in range(n)]
    # Initialize the first house's costs
    dp[0] = cost[0]

    for i in range(1, n):
        for j in range(3):
            # Calculate the cost for painting house i with color j
            dp[i][j] = cost[i][j] + min(dp[i - 1][(j + 1) % 3], dp[i - 1][(j + 2) % 3])

    # To handle the equidistant constraint for the first half and second half of the houses
    res = float('inf')
    for j in range(3):
        # Apply constraints between first and last house and calculate respective costs
        cost_with_constraints = dp[n - 1][j]
        for k in range(3):
            if k != j and k != (j + 1) % 3:
                cost_with_constraints += cost[0][k]
                res = min(res, cost_with_constraints)

    return res
```
