from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # DP table for the current and previous states
    dp = [[float('inf')] * 3 for _ in range(3)]

    # Initialize DP for first house with no constraints
    for a in range(3):
        for b in range(3):
            if a != b:
                dp[0][a] = cost[0][a] + cost[n - 1][b]  # First and last houses must be different

    # DP for every pair of houses
    for i in range(1, n // 2):  # only need to iterate through half the houses
        new_dp = [[float('inf')] * 3 for _ in range(3)]
        for a in range(3):
            for b in range(3):
                if a != b:
                    current_cost = cost[i][a] + cost[n - 1 - i][b]
                    for prev_a in range(3):
                        for prev_b in range(3):
                            if a != prev_a and b != prev_b:
                                new_dp[a][b] = min(new_dp[a][b], dp[prev_a][prev_b] + current_cost)
        dp = new_dp

    # Find the minimum cost from the last DP stage
    res = float('inf')
    for a in range(3):
        for b in range(3):
            if a != b:
                res = min(res, dp[a][b])

    return res
```
