from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # Create a 2D DP table
    dp = [[0] * 3 for _ in range(n)]
    # Initialize the first house's costs
    dp[0] = cost[0]

    for i in range(1, n):
        for j in range(3):
            # Calculate the cost for painting house i with color j
            # We add the cost of the current color and the minimum of the previous house's costs
            dp[i][j] = cost[i][j] + min(dp[i - 1][(j + 1) % 3], dp[i - 1][(j + 2) % 3])

    # Calculate the minimum cost for the last house considering the mirroring condition
    total_cost = float('inf')
    for j in range(3):
        # We need to ensure that the last house's color is not the same as the first and its mirror
        total_cost = min(total_cost, dp[n - 1][j] + (cost[0][j] if j != 0 else min(dp[0][1], dp[0][2])) + (cost[n // 2][j] if j != (0 if n // 2 == 0 else 1) else min(dp[n // 2][1], dp[n // 2][2])))

    return total_cost
```
