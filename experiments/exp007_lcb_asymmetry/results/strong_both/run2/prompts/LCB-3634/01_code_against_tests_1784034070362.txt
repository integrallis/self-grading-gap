## Implementation

```python
def calculateScore(s: str) -> int:
    score = 0
    stacks = {}

    for i in range(len(s)):
        mirror_char = chr(219 - ord(s[i]))  # 219 = 'a' + 'z'
        if mirror_char in stacks and stacks[mirror_char]:
            j = stacks[mirror_char].pop()
            score += i - j
        else:
            if s[i] not in stacks:
                stacks[s[i]] = []
            stacks[s[i]].append(i)
    return score

```
