from solution import calculateScore


def test_canonical_example_1():
    # i=2 pairs with i=0 for 2 points; i=4 pairs with i=1 for 3 points: 2 + 3 = 5.
    assert calculateScore("aczzx") == 5


def test_canonical_example_2():
    # No character has an earlier unmarked mirror, so the score is 0.
    assert calculateScore("abcdef") == 0


def test_uses_closest_unmarked_mirror_index():
    # At i=2, 'z' pairs with the closest earlier 'a' at i=1: 1;
    # at i=3, 'z' pairs with the remaining 'a' at i=0: 3; total = 1 + 3 = 4.
    assert calculateScore("aazz") == 4


def test_marked_indices_cannot_be_reused():
    # i=1 pairs indices 0 and 1 for 1 point; i=2 cannot reuse marked index 0, so total = 1.
    assert calculateScore("aza") == 1
