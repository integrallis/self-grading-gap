```python
def calculateScore(s: str) -> int:
    score = 0
    marked = set()

    for i in range(len(s)):
        if i in marked:
            continue
        mirror_char = chr(219 - ord(s[i]))  # 219 = 'a' + 'z'
        for j in range(i - 1, -1, -1):
            if j in marked:
                continue
            if s[j] == mirror_char:
                score += i - j
                marked.add(i)
                marked.add(j)
                break
    return score

```
