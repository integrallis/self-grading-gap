from solution import minimumPairRemoval


def test_canonical_example_1():
    # (3, 1) -> 4 gives [5, 2, 4]; then (2, 4) -> 6 gives [5, 6], so 2 operations.
    assert minimumPairRemoval([5, 2, 3, 1]) == 2


def test_canonical_example_2():
    # [1, 2, 2] is already non-decreasing, so no operations are needed.
    assert minimumPairRemoval([1, 2, 2]) == 0


def test_single_element_array_needs_no_operations():
    # A one-element array has no previous element to violate non-decreasing order, so 0 operations.
    assert minimumPairRemoval([7]) == 0


def test_leftmost_pair_is_used_when_minimum_sums_tie():
    # Both adjacent sums are 2: (3, -1) and (-1, 3).
    # The leftmost pair becomes 2, producing [2, 3], which is non-decreasing after 1 operation.
    assert minimumPairRemoval([3, -1, 3]) == 1


def test_merges_continue_until_array_becomes_non_decreasing():
    # (2, 1) -> 3 gives [4, 3, 1]; then (3, 1) -> 4 gives [4, 4], so 2 operations.
    assert minimumPairRemoval([4, 2, 1, 1]) == 2
