from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a dictionary to store the total coins at each position
    coin_map = {}

    # Fill the coin_map with coins in the specified ranges
    for l, r, c in coins:
        for i in range(l, r + 1):
            if i not in coin_map:
                coin_map[i] = 0
            coin_map[i] += c

    # Generate a sorted list of unique positions
    positions = sorted(coin_map.keys())
    max_coins = 0
    current_coins = 0
    left = 0

    # Use a sliding window approach to find the maximum coins in k consecutive bags
    for right in range(len(positions)):
        current_coins += coin_map[positions[right]]

        # Adjust the left pointer if the window exceeds k
        while positions[right] - positions[left] + 1 > k:
            current_coins -= coin_map[positions[left]]
            left += 1

        # Update max_coins if the current window is valid
        if positions[right] - positions[left] + 1 <= k:
            max_coins = max(max_coins, current_coins)

    return max_coins
