from solution import maximumCoins


def test_canonical_example_1():
    # Bags [3, 4, 5, 6] contain 2 + 0 + 4 + 4 = 10 coins.
    assert maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) == 10


def test_canonical_example_2():
    # Any two bags within [1, 10] contain 3 + 3 = 6 coins.
    assert maximumCoins([[1, 10, 3]], 2) == 6


def test_k_one_with_point_segments_and_maximum_values():
    # A one-bag window can select position 1,000,000,000 for 1,000 coins.
    assert maximumCoins([[1, 1, 1], [1000000000, 1000000000, 1000]], 1) == 1000


def test_maximum_k_window_can_extend_into_zero_coin_bags():
    # A length-1,000,000,000 window can include the sole bag containing 1,000 coins.
    assert maximumCoins([[1, 1, 1000]], 1000000000) == 1000


def test_adjacent_segments_can_be_collected_in_one_window():
    # Bags [2, 3, 4] contain 5 + 7 + 7 = 19 coins.
    assert maximumCoins([[1, 2, 5], [3, 4, 7]], 3) == 19


def test_maximum_number_of_point_segments():
    coins = [[2 * index + 1, 2 * index + 1, 1] for index in range(100000)]

    # Any 100,000 consecutive positions cover at most 50,000 odd-coordinate point bags.
    assert maximumCoins(coins, 100000) == 50000
