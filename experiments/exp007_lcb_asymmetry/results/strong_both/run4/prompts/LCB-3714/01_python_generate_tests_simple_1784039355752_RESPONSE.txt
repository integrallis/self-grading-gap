from solution import minMaxSums


def test_canonical_example_1():
    # [1], [2], [3], [1,2], [1,3], [2,3] contribute 2+4+6+3+4+5 = 24.
    assert minMaxSums([1, 2, 3], 2) == 24


def test_canonical_example_2():
    # With k=1, each element contributes itself as both minimum and maximum:
    # 2*5 + 2*0 + 2*6 = 22.
    assert minMaxSums([5, 0, 6], 1) == 22


def test_canonical_example_3():
    # Three singleton and three length-two subsequences each contribute 1+1,
    # so 6 * 2 = 12.
    assert minMaxSums([1, 1, 1], 2) == 12


def test_includes_subsequences_up_to_the_full_array_length():
    # Contributions are:
    # [0]=0, [2]=4, [4]=8, [0,2]=2, [0,4]=4, [2,4]=6, [0,2,4]=4;
    # total = 28.
    assert minMaxSums([0, 2, 4], 3) == 28


def test_single_element_array():
    # The only subsequence is [7], whose minimum plus maximum is 7+7 = 14.
    assert minMaxSums([7], 1) == 14


def test_returns_result_modulo_1000000007():
    # All 2^70-1 nonempty subsequences have min+max = 2,000,000,000.
    # (2,000,000,000 * (2^70 - 1)) mod 1,000,000,007 = 219,772,500.
    assert minMaxSums([1_000_000_000] * 70, 70) == 219_772_500
