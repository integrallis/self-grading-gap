from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    # We will calculate the contribution of each number
    for i in range(n):
        # Each number contributes to the minimums and maximums of the subsequences
        # of size 1 to k. The count of such subsequences can be calculated
        # using combinations (n-1 choose j-1) where j is the size of the subsequence
        # and i is the index of the current number.
        for j in range(1, k + 1):
            # Count of subsequences of size j that can be formed with nums[i] as min or max
            count = 1 << (n - 1)  # This is 2^(n-1), as we can choose to include/exclude other elements
            # We can choose (j-1) elements from (n-1)
            if j - 1 <= n - 1:
                count = (count * comb(n - 1, j - 1)) % MOD
            # Contribution of nums[i] as min
            total_sum = (total_sum + nums[i] * count) % MOD
            # Contribution of nums[i] as max
            total_sum = (total_sum + nums[i] * count) % MOD

    return total_sum % MOD


def comb(n: int, k: int) -> int:
    if k > n:
        return 0
    if k == 0 or k == n:
        return 1
    k = min(k, n - k)  # Take advantage of symmetry
    c = 1
    for i in range(k):
        c = c * (n - i) // (i + 1)
    return c
```
