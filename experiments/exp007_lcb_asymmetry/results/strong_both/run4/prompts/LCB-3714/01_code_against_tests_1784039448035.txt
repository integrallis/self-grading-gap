## Implementation

```python
from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Sort the input to get proper min and max handling
    nums.sort()

    # Calculate contribution based on sorted array
    for i in range(n):
        # Use combinations to find all possible contributions of the current element
        if i < k:
            # Each element contributes as max in all combinations it can form
            total_sum += nums[i] * pow(2, min(i, k-1), MOD) % MOD
        if n - i - 1 < k:
            # Each element contributes as min in all combinations it can form
            total_sum += nums[i] * pow(2, min(n - i - 1, k-1), MOD) % MOD

    return total_sum % MOD
```
