from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Calculate the contribution of each element as both min and max
    for i in range(n):
        # Each element contributes to all subsequences of size 1 to k
        count_as_min = 0
        count_as_max = 0

        # Calculate combinations for all lengths from 1 to k
        for j in range(1, k + 1):
            if j <= n:
                # Count the number of subsequences of size j including nums[i]
                count = comb(n - 1, j - 1)
                count_as_min = (count_as_min + count) % MOD
                count_as_max = (count_as_max + count) % MOD

        # Each nums[i] contributes its value multiplied by the counts
        total_sum = (total_sum + nums[i] * (count_as_min + count_as_max)) % MOD

    return total_sum % MOD


def comb(n: int, k: int) -> int:
    if k > n:
        return 0
    if k == 0 or k == n:
        return 1
    k = min(k, n - k)  # Take advantage of symmetry
    c = 1
    for i in range(k):
        c = c * (n - i) // (i + 1)
    return c

```
