from solution import minCosts


def test_canonical_example_1():
    # Prefix minima of [5, 3, 4, 1, 3, 2] are [5, 3, 3, 1, 1, 1].
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]


def test_canonical_example_2():
    # Paying person 0 costs 1, so every position has minimum cost 1.
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]


def test_single_person_with_maximum_cost():
    # With one person, reaching position 0 requires paying that person's cost: 100.
    assert minCosts([100]) == [100]


def test_maximum_length_with_equal_maximum_costs():
    # Every prefix minimum of 100 repeated 100 times remains 100.
    assert minCosts([100] * 100) == [100] * 100
