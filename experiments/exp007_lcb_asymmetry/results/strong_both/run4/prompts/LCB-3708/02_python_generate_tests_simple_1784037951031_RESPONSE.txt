from solution import zigzagTraversal


def test_canonical_example_1():
    # Zigzag order is [1, 2, 4, 3]; skipping alternate cells keeps [1, 4].
    assert zigzagTraversal([[1, 2], [3, 4]]) == [1, 4]


def test_canonical_example_2():
    # Zigzag order is [2, 1, 1, 2, 2, 1]; skipping alternate cells keeps [2, 1, 2].
    assert zigzagTraversal([[2, 1], [2, 1], [2, 1]]) == [2, 1, 2]


def test_canonical_example_3():
    # Zigzag order is [1, 2, 3, 6, 5, 4, 7, 8, 9]; skipping alternate cells keeps [1, 3, 5, 7, 9].
    assert zigzagTraversal([[1, 2, 3], [4, 5, 6], [7, 8, 9]]) == [1, 3, 5, 7, 9]


def test_maximum_dimensions_and_cell_value():
    grid = [[r * 50 + c + 1 for c in range(50)] for r in range(50)]
    # Each row is traversed in its zigzag direction, keeping every other global traversal cell.
    expected = [
        r * 50 + c + 1
        for r in range(50)
        for c in (range(0, 50, 2) if r % 2 == 0 else range(49, -1, -2))
    ]
    assert zigzagTraversal(grid) == expected
