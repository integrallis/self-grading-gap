from solution import maximumCoins


def test_canonical_example_1():
    # Bags [3, 4, 5, 6] contain 2 + 0 + 4 + 4 coins.
    assert maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) == 10


def test_canonical_example_2():
    # Any two bags within [1, 10] contain 3 + 3 coins.
    assert maximumCoins([[1, 10, 3]], 2) == 6


def test_window_can_include_partial_segments_at_both_ends():
    # Bags [3, 4, 5, 6] contain 2 + 5 + 5 + 5 coins.
    assert maximumCoins([[1, 3, 2], [4, 6, 5]], 4) == 17


def test_window_can_span_a_gap_between_segments():
    # Bags [2, 3, 4, 5, 6] contain 4 + 4 + 0 + 7 + 7 coins.
    assert maximumCoins([[1, 3, 4], [5, 6, 7]], 5) == 22


def test_window_shorter_than_a_high_value_segment():
    # Bags [3, 4, 5] within the segment contain 9 + 9 + 9 coins.
    assert maximumCoins([[1, 10, 9]], 3) == 27


def test_window_longer_than_all_coin_segments():
    # A length-10 window covering both segments contains 2 + 2 + 2 + 5 + 5 coins.
    assert maximumCoins([[2, 4, 2], [8, 9, 5]], 10) == 16


def test_unsorted_non_overlapping_segments():
    # Bags [9, 10] contain 8 + 8 coins, exceeding every other length-2 choice.
    assert maximumCoins([[10, 12, 8], [1, 2, 3], [7, 8, 1]], 2) == 16


def test_minimum_window_with_single_bag_segment():
    # The only selected bag contains 7 coins.
    assert maximumCoins([[1, 1, 7]], 1) == 7


def test_upper_bound_coordinate_window_and_coin_value():
    # The length-1,000,000,000 window can include the one populated bag with 1000 coins.
    assert maximumCoins([[1_000_000_000, 1_000_000_000, 1000]], 1_000_000_000) == 1000


def test_maximum_number_of_segments():
    coins = [[2 * i + 1, 2 * i + 1, 1] for i in range(100_000)]
    # The length-199,999 window from bag 1 through bag 199,999 contains all 100,000 coin bags.
    assert maximumCoins(coins, 199_999) == 100_000
