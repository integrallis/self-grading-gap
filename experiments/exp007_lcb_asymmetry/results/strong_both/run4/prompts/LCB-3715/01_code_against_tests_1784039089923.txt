```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a dictionary to store the number of coins at each position
    coin_map = {}
    for l, r, c in coins:
        for pos in range(l, r + 1):
            coin_map[pos] = coin_map.get(pos, 0) + c

    # Create a sorted list of unique positions
    sorted_positions = sorted(coin_map.keys())
    max_coins = 0
    window_coins = 0
    left_index = 0  # This will maintain the left edge of the sliding window

    # Iterate through the sorted positions to maintain a sliding window
    for right_index in range(len(sorted_positions)):
        pos = sorted_positions[right_index]
        window_coins += coin_map[pos]  # Add coins at the right edge

        # Move the left index to maintain the window size of k
        while pos - sorted_positions[left_index] >= k:
            window_coins -= coin_map[sorted_positions[left_index]]  # Subtract coins at the left edge
            left_index += 1  # Move left edge right

        max_coins = max(max_coins, window_coins)

    return max_coins

```
