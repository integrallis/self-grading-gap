from solution import largestInteger


def test_canonical_example_1():
    # 3 appears only in [3, 9, 2] and 7 only in [2, 1, 7]; largest is 7.
    assert largestInteger([3, 9, 2, 1, 7], 3) == 7


def test_canonical_example_2():
    # Only 3 appears in exactly one size-4 subarray: [3, 9, 7, 2].
    assert largestInteger([3, 9, 7, 2, 1, 7], 4) == 3


def test_canonical_example_3():
    # 0 appears in both size-1 subarrays, so no value appears in exactly one.
    assert largestInteger([0, 0], 1) == -1


def test_full_array_is_the_only_subarray():
    # The only size-4 subarray contains 2, 8, 3, and 8; all present values appear once,
    # so the largest is 8.
    assert largestInteger([2, 8, 3, 8], 4) == 8


def test_repeated_value_within_one_subarray_counts_once():
    # There is one size-3 subarray, [5, 5, 1]; 5 appears in that one subarray,
    # and is larger than 1.
    assert largestInteger([5, 5, 1], 3) == 5


def test_size_one_subarrays_require_value_to_occur_once_in_nums():
    # Size-1 subarrays are [2], [5], [5], [3]; only 2 and 3 occur in exactly one,
    # and the largest is 3.
    assert largestInteger([2, 5, 5, 3], 1) == 3


def test_singleton_array_boundary():
    # The sole size-1 subarray is [50], so 50 appears in exactly one subarray.
    assert largestInteger([50], 1) == 50


def test_maximum_length_array_boundary():
    # The sole size-50 subarray contains 50, so 50 appears in exactly one subarray.
    assert largestInteger([50] * 50, 50) == 50
