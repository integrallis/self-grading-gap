from solution import maxFrequency


def test_canonical_example_1():
    # Add -5 to nums[2..5]: [1, 2, -2, -1, 0, 1], so k=1 appears 2 times.
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2


def test_canonical_example_2():
    # Add 8 to nums[1..9]: [10, 10, 11, 12, 13, 13, 12, 11, 10, 10],
    # so k=10 appears 4 times.
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4


def test_can_use_zero_addition_to_keep_all_existing_k_values():
    # Select any one-element subarray and add 0, leaving all three 7s unchanged.
    assert maxFrequency([7, 7, 7], 7) == 3


def test_converting_values_across_k_can_trade_one_existing_k_for_two_new_ones():
    # Add -1 to the whole array: [2, 1, 2] becomes [1, 0, 1],
    # producing two occurrences of k=1.
    assert maxFrequency([2, 1, 2], 1) == 2


def test_one_addition_cannot_convert_different_source_values_to_k():
    # Adding -1 converts the two 2s to 1, while adding -2 converts the two 3s
    # to 1; no single x converts both values, so the maximum is 2.
    assert maxFrequency([2, 3, 2, 3], 1) == 2


def test_best_subarray_can_avoid_losing_existing_k_values():
    # Add -1 to either adjacent pair of 2s, creating two new 1s while keeping
    # the three existing 1s outside that subarray, for a total of 5.
    assert maxFrequency([2, 2, 1, 1, 1, 2, 2], 1) == 5


def test_maximum_length_array_with_one_convertible_value():
    nums = [2] * 100_000
    # Add -1 to the whole array, converting all 100000 values to k=1.
    assert maxFrequency(nums, 1) == 100_000
