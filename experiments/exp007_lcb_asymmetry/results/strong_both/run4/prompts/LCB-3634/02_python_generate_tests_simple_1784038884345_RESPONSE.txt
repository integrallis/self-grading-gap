from solution import calculateScore


def test_canonical_example_1():
    # i=2 pairs with i=0 for 2 points; i=4 pairs with i=1 for 3 points; total = 5.
    assert calculateScore("aczzx") == 5


def test_canonical_example_2():
    # No character has an earlier unmarked mirror character, so total = 0.
    assert calculateScore("abcdef") == 0


def test_uses_closest_available_mirror_index():
    # At i=2, 'z' pairs with closest earlier 'a' at i=1: 2-1=1.
    # At i=3, 'z' pairs with remaining 'a' at i=0: 3-0=3; total = 4.
    assert calculateScore("aazz") == 4


def test_marked_indices_are_not_reused():
    # i=1 pairs with i=0 for 1 point, and i=3 pairs with i=2 for 1 point; total = 2.
    assert calculateScore("azaz") == 2


def test_minimum_length_string_has_zero_score():
    # The only index has no earlier index, so no pair can be formed; total = 0.
    assert calculateScore("a") == 0


def test_matches_mirror_in_reverse_alphabet_direction():
    # At i=1, 'a' pairs with earlier mirror 'z' at i=0: 1-0=1.
    assert calculateScore("za") == 1


def test_current_matched_index_is_marked():
    # i=1 pairs with i=0 for 1 point; i=2 cannot reuse marked i=1; total = 1.
    assert calculateScore("aza") == 1


def test_maximum_length_valid_input():
    # Each z at i=50000+k pairs with closest remaining a at i=49999-k.
    # Sum of distances is 1 + 3 + ... + 99999 = 50000^2 = 2500000000.
    assert calculateScore("a" * 50000 + "z" * 50000) == 2500000000
