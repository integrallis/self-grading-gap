```python
from typing import List

def maxSum(nums: List[int]) -> int:
    unique_nums = set()
    max_sum = 0
    for num in nums:
        if num > 0:
            unique_nums.add(num)
    max_sum = sum(unique_nums)
    return max(max_sum, max(nums)) if nums else 0
```
