from solution import maxAdjacentDistance


def test_canonical_example_1():
    # max(|1 - 2|, |2 - 4|, |4 - 1|) = max(1, 2, 3) = 3
    assert maxAdjacentDistance([1, 2, 4]) == 3


def test_canonical_example_2():
    # max(|-5 - (-10)|, |-10 - (-5)|, |-5 - (-5)|) = max(5, 5, 0) = 5
    assert maxAdjacentDistance([-5, -10, -5]) == 5


def test_two_elements_at_value_extrema():
    # The two circular adjacencies both have difference |-100 - 100| = 200.
    assert maxAdjacentDistance([-100, 100]) == 200


def test_maximum_length_array():
    # All adjacent 0s differ by 0; each adjacency involving 100 differs by |100 - 0| = 100.
    assert maxAdjacentDistance([0] * 99 + [100]) == 100
