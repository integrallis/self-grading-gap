```python
from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[[float('-inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0]

    for i in range(m):
        for j in range(n):
            for k in range(3):
                if i + 1 < m:
                    new_value = dp[i][j][k] + coins[i + 1][j]
                    if coins[i + 1][j] < 0:
                        if k > 0:
                            dp[i + 1][j][k - 1] = max(dp[i + 1][j][k - 1], new_value)
                        else:
                            dp[i + 1][j][k] = max(dp[i + 1][j][k], new_value)
                    else:
                        dp[i + 1][j][k] = max(dp[i + 1][j][k], new_value)

                if j + 1 < n:
                    new_value = dp[i][j][k] + coins[i][j + 1]
                    if coins[i][j + 1] < 0:
                        if k > 0:
                            dp[i][j + 1][k - 1] = max(dp[i][j + 1][k - 1], new_value)
                        else:
                            dp[i][j + 1][k] = max(dp[i][j + 1][k], new_value)
                    else:
                        dp[i][j + 1][k] = max(dp[i][j + 1][k], new_value)

    max_profit = float('-inf')
    for k in range(3):
        max_profit = max(max_profit, dp[m - 1][n - 1][k])

    return max_profit

```
