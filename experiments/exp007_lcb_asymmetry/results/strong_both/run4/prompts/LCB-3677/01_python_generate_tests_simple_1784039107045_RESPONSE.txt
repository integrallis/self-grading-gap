from solution import maximumAmount


def test_canonical_example_1():
    # Optimal route collects 0 + 1 + 0 (neutralized -2) + 3 + 4 = 8.
    assert maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) == 8


def test_canonical_example_2():
    # Any route visits four cells worth 10 coins each: 10 + 10 + 10 + 10 = 40.
    assert maximumAmount([[10, 10, 10], [10, 10, 10]]) == 40


def test_can_neutralize_a_robber_at_the_starting_cell():
    # The only path contains -7, which can be neutralized, so the total is 0.
    assert maximumAmount([[-7]]) == 0


def test_can_neutralize_two_robbers_on_a_path():
    # Neutralize -7 and -5; collect 4 + 2 - 3 = 3.
    assert maximumAmount([[-5, 4, -7, 2, -3]]) == 3


def test_profit_can_remain_negative_after_two_neutralizations():
    # Neutralize -4 and -3; the remaining robber losses are -1 + -2 = -3.
    assert maximumAmount([[-1, -2, -3, -4]]) == -3


def test_selects_the_route_with_the_best_profit():
    # Going right, right, down neutralizes -10 and collects 1 + 100 + 1 = 102.
    assert maximumAmount([[1, -10, 100], [50, -100, 1]]) == 102
