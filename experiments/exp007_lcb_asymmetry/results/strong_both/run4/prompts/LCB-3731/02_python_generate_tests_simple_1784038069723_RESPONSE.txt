from solution import subarraySum


def test_canonical_example_1():
    # i=0: sum([2])=2; i=1: sum([2, 3])=5; i=2: sum([3, 1])=4; total=2+5+4=11
    assert subarraySum([2, 3, 1]) == 11


def test_canonical_example_2():
    # i=0: sum([3])=3; i=1: sum([3, 1])=4; i=2: sum([1, 1])=2; i=3: sum([1, 1, 2])=4; total=3+4+2+4=13
    assert subarraySum([3, 1, 1, 2]) == 13


def test_single_element_minimum_length():
    # i=0: start=max(0, 0-1)=0, so sum([1])=1
    assert subarraySum([1]) == 1


def test_maximum_length_all_ones():
    # i=0 contributes 1; each of i=1 through i=99 contributes [1, 1] with sum 2; total=1+99*2=199
    assert subarraySum([1] * 100) == 199


def test_maximum_element_value():
    # i=0: start=max(0, 0-1000)=0, so sum([1000])=1000
    assert subarraySum([1000]) == 1000
