from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n == 0:
        return 0

    dp = [[0] * (n + 1) for _ in range(n)]
    for i in range(n):
        dp[i][0] = 1

    max_length = 1
    for j in range(1, n + 1):
        for i in range(j):
            diff1 = abs(nums[j - 1] - nums[i]) if i < j - 1 else 0
            for k in range(i):
                diff2 = abs(nums[i] - nums[k])
                if diff1 >= diff2:
                    dp[j - 1][i + 1] = max(dp[j - 1][i + 1], dp[i][k + 1] + 1)
                    max_length = max(max_length, dp[j - 1][i + 1])

    return max_length

```
