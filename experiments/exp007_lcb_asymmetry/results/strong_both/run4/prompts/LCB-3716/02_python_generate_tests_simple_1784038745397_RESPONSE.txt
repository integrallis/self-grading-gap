from solution import longestSubsequence


def test_canonical_example_1():
    # [16, 6, 3] has adjacent absolute differences [10, 3], and 10 >= 3.
    assert longestSubsequence([16, 6, 3]) == 3


def test_canonical_example_2():
    # [6, 4, 2, 1] is a subsequence with differences [2, 2, 1], where 2 >= 2 >= 1.
    assert longestSubsequence([6, 5, 3, 4, 2, 1]) == 4


def test_canonical_example_3():
    # [10, 20, 10, 19, 10] has differences [10, 10, 9, 9], where 10 >= 10 >= 9 >= 9.
    assert longestSubsequence([10, 20, 10, 19, 10, 20]) == 5


def test_subsequence_can_delete_elements_to_be_longer_than_any_contiguous_valid_segment():
    # Selecting [1, 11, 6, 5] gives differences [10, 5, 1], where 10 >= 5 >= 1.
    assert longestSubsequence([1, 11, 10, 6, 5]) == 4


def test_minimum_length_with_lowest_and_highest_legal_values():
    # The two elements [1, 300] form a length-2 subsequence; one difference needs no comparison.
    assert longestSubsequence([1, 300]) == 2


def test_maximum_length_with_zero_differences():
    # All 10,000 selected values are 300, so all 9,999 adjacent differences are 0 and 0 >= 0.
    assert longestSubsequence([300] * 10_000) == 10_000
