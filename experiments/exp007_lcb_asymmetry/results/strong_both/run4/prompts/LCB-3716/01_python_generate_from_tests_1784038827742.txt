from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n == 0:
        return 0

    dp = [1] * n  # Each element can at least form a subsequence of length 1
    max_length = 1

    for i in range(n):
        for j in range(i):
            diff1 = abs(nums[i] - nums[j])
            # Check if this forms a valid subsequence
            if dp[j] > 0:
                # Update dp[i] if we can extend the subsequence
                for k in range(j):
                    diff2 = abs(nums[j] - nums[k])
                    if diff1 >= diff2:
                        dp[i] = max(dp[i], dp[j] + 1)
                        max_length = max(max_length, dp[i])

    return max_length
```
