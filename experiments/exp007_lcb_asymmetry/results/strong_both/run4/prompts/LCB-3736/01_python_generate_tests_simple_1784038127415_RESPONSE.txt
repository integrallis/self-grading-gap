from solution import findValidPair


def test_canonical_example_1():
    # '2' appears twice and '3' appears three times; the first valid adjacent pair is "23".
    assert findValidPair("2523533") == "23"


def test_canonical_example_2():
    # '2' appears twice and '1' appears once; "22" is equal digits, so the first valid pair is "21".
    assert findValidPair("221") == "21"


def test_canonical_example_3():
    # Although '2' appears twice, the only adjacent pair is "22", whose digits are equal.
    assert findValidPair("22") == ""


def test_returns_leftmost_when_multiple_valid_pairs_exist():
    # Counts are '1': 1, '2': 2, '3': 3, and '4': 4; both "12" and later pairs qualify, so return "12".
    assert findValidPair("1223334444") == "12"


def test_rejects_adjacent_digits_when_one_digit_count_does_not_match_value():
    # '2' appears once rather than twice, so "23" is invalid even though '3' appears three times.
    assert findValidPair("2333") == ""


def test_finds_valid_pair_after_equal_adjacent_digits():
    # '9' appears nine times and '1' appears once; after the equal "99" pairs, "91" is valid.
    assert findValidPair("9999999991") == "91"
