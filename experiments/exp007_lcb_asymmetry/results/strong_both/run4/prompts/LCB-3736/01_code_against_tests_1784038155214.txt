## Implementation

```python
def findValidPair(s: str) -> str:
    from collections import Counter

    # Count the occurrences of each digit in the string
    counts = Counter(s)

    # Iterate through the string to find the first valid pair
    for i in range(len(s) - 1):
        first, second = s[i], s[i + 1]
        if first != second:
            # Check if the count of each digit matches its value
            if counts[first] == int(first) and counts[second] == int(second):
                return first + second

    return ""
```
