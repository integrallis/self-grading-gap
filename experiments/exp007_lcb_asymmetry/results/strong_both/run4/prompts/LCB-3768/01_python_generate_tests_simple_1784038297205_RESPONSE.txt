from solution import hasSameDigits


def test_canonical_example_1():
    # "3902" -> "292" -> "11", so the final two digits are equal.
    assert hasSameDigits("3902") is True


def test_canonical_example_2():
    # "34789" -> "7157" -> "862" -> "48", and 4 != 8.
    assert hasSameDigits("34789") is False


def test_three_digits_performs_one_operation():
    # "123" -> "35", and 3 != 5.
    assert hasSameDigits("123") is False


def test_sums_are_reduced_modulo_ten():
    # "199" -> "08" because (1 + 9) % 10 = 0 and (9 + 9) % 10 = 8; 0 != 8.
    assert hasSameDigits("199") is False
