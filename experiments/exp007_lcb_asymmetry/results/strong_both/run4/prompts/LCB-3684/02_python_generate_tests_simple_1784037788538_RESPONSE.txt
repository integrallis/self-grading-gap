from solution import hasMatch


def test_canonical_example_1():
    # "ee*e" can replace "*" with "tcod" to form "eetcode", a substring of "leetcode".
    assert hasMatch("leetcode", "ee*e") is True


def test_canonical_example_2():
    # No replacement for "*" makes "c*v" match a substring of "car".
    assert hasMatch("car", "c*v") is False


def test_canonical_example_3():
    # Replacing "*" with "" forms "u", which is a substring of "luck".
    assert hasMatch("luck", "u*") is True


def test_wildcard_can_match_zero_characters():
    # Replacing "*" with "" forms "ab", which is a substring of "ab".
    assert hasMatch("ab", "a*b") is True


def test_wildcard_can_match_characters_before_a_suffix():
    # Replacing "*" with "xx" forms "xxab", which is a substring of "xxab".
    assert hasMatch("xxab", "*ab") is True


def test_match_can_start_and_end_inside_the_string():
    # Replacing "*" with "x" forms "abxcd", a substring of "zabxcdz".
    assert hasMatch("zabxcdz", "ab*cd") is True


def test_fixed_parts_must_appear_in_pattern_order():
    # "a" occurs after "b" in "ba", so no replacement makes "a*b" a substring.
    assert hasMatch("ba", "a*b") is False


def test_pattern_with_only_wildcard_can_form_nonempty_substring():
    # Replacing "*" with "a" forms the non-empty substring "a" of "ab".
    assert hasMatch("ab", "*") is True


def test_minimum_length_string_with_only_wildcard():
    # Replacing "*" with "a" forms the non-empty substring "a" of "a".
    assert hasMatch("a", "*") is True


def test_maximum_length_string_and_pattern():
    # Replacing "*" with "a" forms 50 "a" characters, the whole substring of s.
    assert hasMatch("a" * 50, "a" * 49 + "*") is True
