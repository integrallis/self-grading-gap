from solution import minimumPairRemoval


def test_canonical_example_1():
    # (3, 1) sums to 4 -> [5, 2, 4]; then (2, 4) sums to 6 -> [5, 6].
    assert minimumPairRemoval([5, 2, 3, 1]) == 2


def test_canonical_example_2():
    # [1, 2, 2] is already non-decreasing, so no replacements are needed.
    assert minimumPairRemoval([1, 2, 2]) == 0


def test_single_element_array_needs_no_operations():
    # A one-element array has no adjacent decreasing pair, so it is non-decreasing.
    assert minimumPairRemoval([7]) == 0


def test_minimum_sum_pair_is_chosen_even_when_it_is_not_decreasing_pair():
    # (1, 2) has smaller sum than (3, 1): [3, 1, 2] -> [3, 3].
    assert minimumPairRemoval([3, 1, 2]) == 1


def test_leftmost_pair_is_chosen_when_minimum_sums_tie():
    # Both pairs sum to 4; choose leftmost (3, 1): [3, 1, 3] -> [4, 3] -> [7].
    assert minimumPairRemoval([3, 1, 3]) == 2


def test_negative_values_follow_minimum_pair_sum_rule():
    # (-1, 2) sums to 1 and (2, -3) sums to -1: [-1, 2, -3] -> [-1, -1].
    assert minimumPairRemoval([-1, 2, -3]) == 1

    assert minimumPairRemoval([1000] * 50) == 0


def test_exact_numeric_endpoints_in_non_decreasing_order_need_no_operations():
    # -1000 <= 1000, so this endpoint array is already non-decreasing.
    assert minimumPairRemoval([-1000, 1000]) == 0


def test_exact_numeric_endpoints_in_decreasing_order_need_one_operation():
    # 1000 > -1000; merging the only pair produces [0], which is non-decreasing.
    assert minimumPairRemoval([1000, -1000]) == 1
