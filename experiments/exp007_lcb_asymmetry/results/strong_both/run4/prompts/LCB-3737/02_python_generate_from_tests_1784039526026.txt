from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    dp = [[0] * 3 for _ in range(n)]
    dp[0] = cost[0]

    for i in range(1, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(dp[i - 1][(j + 1) % 3], dp[i - 1][(j + 2) % 3])

    min_cost = float('inf')
    for j in range(3):
        if j == 0:
            min_cost = min(min_cost, dp[n - 1][j] + min(dp[n // 2 - 1][(1 + j) % 3], dp[n // 2 - 1][(2 + j) % 3]))
        elif j == 1:
            min_cost = min(min_cost, dp[n - 1][j] + min(dp[n // 2 - 1][(0 + j) % 3], dp[n // 2 - 1][(2 + j) % 3]))
        else:
            min_cost = min(min_cost, dp[n - 1][j] + min(dp[n // 2 - 1][(0 + j) % 3], dp[n // 2 - 1][(1 + j) % 3]))

    return min_cost

```
