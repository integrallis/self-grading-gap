def maxDistance(s: str, k: int) -> int:
    # Initialize position at the origin
    x, y = 0, 0
    max_distance = 0
    # Count the number of each direction
    move_counts = {'N': 0, 'S': 0, 'E': 0, 'W': 0}
    for char in s:
        # Update position based on the current movement
        if char == 'N':
            y += 1
            move_counts['N'] += 1
        elif char == 'S':
            y -= 1
            move_counts['S'] += 1
        elif char == 'E':
            x += 1
            move_counts['E'] += 1
        elif char == 'W':
            x -= 1
            move_counts['W'] += 1
        # Calculate the current Manhattan distance
        current_distance = abs(x) + abs(y)
        # Calculate the number of moves that can be changed to maximize distance
        total_moves = move_counts['N'] + move_counts['S'] + move_counts['E'] + move_counts['W']
