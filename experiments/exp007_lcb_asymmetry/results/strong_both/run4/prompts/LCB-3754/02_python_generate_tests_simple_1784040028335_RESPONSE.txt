from solution import maxDistance


def test_canonical_example_1():
    # Change "NWSE" to "NWNE"; prefix positions reach (-1, 2), distance 3.
    assert maxDistance("NWSE", 1) == 3


def test_canonical_example_2():
    # Change "NSWWEW" to "NNWWWW"; all 6 moves increase distance from the origin.
    assert maxDistance("NSWWEW", 3) == 6


def test_no_changes_uses_the_best_prefix_not_necessarily_the_final_position():
    # "NNSS" reaches distance 2 after "NN", then returns to distance 0.
    assert maxDistance("NNSS", 0) == 2


def test_one_change_can_make_a_two_move_prefix_fully_outward():
    # Change "NS" to "NN"; after two moves the distance is 2.
    assert maxDistance("NS", 1) == 2


def test_changes_are_limited_by_k():
    # Change S -> N and W -> E, making "SENW" into "NENE" and reaching distance 4.
    assert maxDistance("SENW", 2) == 4


def test_existing_moves_can_already_reach_the_full_length_distance():
    # Every move in "NNNW" increases distance from the origin, so the maximum is 4.
    assert maxDistance("NNNW", 0) == 4


def test_minimum_length_string():
    # One southward move ends at (0, -1), whose Manhattan distance is 1.
    assert maxDistance("S", 0) == 1


def test_edit_budget_equal_to_string_length():
    # Change all moves in "NSWE" to N; after four moves the distance is 4.
    assert maxDistance("NSWE", 4) == 4


def test_maximum_length_string():
    # All 100000 northward moves increase distance, reaching 100000.
    assert maxDistance("N" * 100000, 0) == 100000
