from solution import hasSpecialSubstring


def test_canonical_example_1():
    # The trailing "aaa" has length 3 and is preceded by different character "b".
    assert hasSpecialSubstring("aaabaaa", 3) is True


def test_canonical_example_2():
    # No length-2 substring of "abc" consists of one repeated character.
    assert hasSpecialSubstring("abc", 2) is False


def test_entire_string_is_a_special_substring():
    # "aaaa" is a uniform substring of length 4 with no neighboring characters.
    assert hasSpecialSubstring("aaaa", 4) is True


def test_run_longer_than_k_is_not_special():
    # Every length-2 "aa" lies within the longer run "aaa", so it has an equal neighbor.
    assert hasSpecialSubstring("aaa", 2) is False


def test_exact_internal_run_with_different_neighbors_is_special():
    # The middle "bb" has length 2 and is bounded by different characters "a" and "c".
    assert hasSpecialSubstring("abbc", 2) is True


def test_single_character_substring_requires_different_neighbors():
    # The "b" has length 1 and is bounded by different characters "a" and "c".
    assert hasSpecialSubstring("abc", 1) is True


def test_single_character_inside_repeated_run_is_not_special():
    # Each length-1 "a" has an immediately adjacent equal "a".
    assert hasSpecialSubstring("aa", 1) is False


def test_special_run_at_start_needs_only_different_right_neighbor():
    # The initial "bb" has length 2, no left neighbor, and right neighbor "a".
    assert hasSpecialSubstring("bba", 2) is True


def test_no_exact_length_run_exists():
    # The runs have lengths 2 ("aa") and 3 ("bbb"), so none has exactly length 1.
    assert hasSpecialSubstring("aabbb", 1) is False
