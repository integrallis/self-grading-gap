```python
from typing import List


def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers
    offline_until = [0] * numberOfUsers
    online_users = set(range(numberOfUsers))  # Initially all users are online

    # Sort events by timestamp
    events.sort(key=lambda x: int(x[1]))

    for event in events:
        event_type, timestamp, arg = event
        timestamp = int(timestamp)
        # Process offline events first
        if event_type == "OFFLINE":
            user_id = int(arg)
            offline_until[user_id] = timestamp + 60  # User goes offline for 60 time units
            online_users.discard(user_id)  # Remove user from online set

        # Process message events
        elif event_type == "MESSAGE":
            # Update online users based on offline times
            for user_id in range(numberOfUsers):
                if timestamp >= offline_until[user_id]:
                    online_users.add(user_id)

            # Count mentions
            if arg == "ALL":
                # All users are mentioned
                for user_id in range(numberOfUsers):
                    mentions[user_id] += 1
            elif arg == "HERE":
                # Only online users are mentioned
                for user_id in online_users:
                    mentions[user_id] += 1
            else:
                # Mentioned users are specified
                mentioned_ids = arg.split()
                for mentioned_id in mentioned_ids:
                    user_id = int(mentioned_id[2:])  # Extract user number from 'id<number>'
                    mentions[user_id] += 1

    return mentions
```
