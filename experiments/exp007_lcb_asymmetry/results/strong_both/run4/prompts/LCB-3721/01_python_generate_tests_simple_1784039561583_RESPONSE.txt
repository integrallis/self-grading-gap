from solution import countMentions


def test_canonical_example_1():
    events = [
        ["MESSAGE", "10", "id1 id0"],
        ["OFFLINE", "11", "0"],
        ["MESSAGE", "71", "HERE"],
    ]

    # At 10 each user is directly mentioned once; at 71 both are online for HERE.
    assert countMentions(2, events) == [2, 2]


def test_canonical_example_2():
    events = [
        ["MESSAGE", "10", "id1 id0"],
        ["OFFLINE", "11", "0"],
        ["MESSAGE", "12", "ALL"],
    ]

    # At 10 each user is directly mentioned once; ALL at 12 includes offline id0.
    assert countMentions(2, events) == [2, 2]


def test_canonical_example_3():
    events = [
        ["OFFLINE", "10", "0"],
        ["MESSAGE", "12", "HERE"],
    ]

    # id0 is offline at 12, so HERE mentions only id1.
    assert countMentions(2, events) == [0, 1]


def test_direct_id_mentions_count_duplicates_and_can_mention_offline_users():
    events = [
        ["OFFLINE", "5", "1"],
        ["MESSAGE", "6", "id1 id1 id0"],
    ]

    # id1 is offline but is directly mentioned twice; id0 is directly mentioned once.
    assert countMentions(2, events) == [1, 2]


def test_here_mentions_all_users_when_everyone_is_online():
    events = [
        ["MESSAGE", "10", "HERE"],
    ]

    # All three users start online, so HERE contributes one mention to each.
    assert countMentions(3, events) == [1, 1, 1]


def test_offline_status_is_processed_before_message_at_the_same_timestamp():
    events = [
        ["MESSAGE", "10", "HERE"],
        ["OFFLINE", "10", "0"],
    ]

    # The offline change at 10 is processed before the message at 10, so HERE mentions only id1.
    assert countMentions(2, events) == [0, 1]


def test_user_returns_online_exactly_sixty_time_units_after_going_offline():
    events = [
        ["OFFLINE", "20", "1"],
        ["MESSAGE", "80", "HERE"],
    ]

    # id1 returns at 20 + 60 = 80, so HERE at 80 mentions both users.
    assert countMentions(2, events) == [1, 1]
