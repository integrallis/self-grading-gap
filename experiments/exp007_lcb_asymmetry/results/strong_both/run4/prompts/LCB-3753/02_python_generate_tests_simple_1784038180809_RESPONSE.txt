from solution import maxDifference


def test_canonical_example_1():
    # 'a' occurs 5 times (odd) and 'b' occurs 2 times (even): 5 - 2 = 3.
    assert maxDifference("aaaaabbc") == 3


def test_canonical_example_2():
    # 'a' occurs 3 times (odd) and 'c' occurs 2 times (even): 3 - 2 = 1.
    assert maxDifference("abcabcab") == 1


def test_selects_smallest_even_frequency_among_multiple_even_characters():
    # Frequencies are a=5 (odd), b=4 (even), c=2 (even): 5 - 2 = 3.
    assert maxDifference("aaaaabbbbcc") == 3


def test_valid_minimum_length():
    # Frequencies are a=2 (even) and b=1 (odd): 1 - 2 = -1.
    assert maxDifference("aab") == -1


def test_valid_maximum_length():
    # Frequencies are a=95 (odd), b=4 (even), c=1 (odd): 95 - 4 = 91.
    assert maxDifference("a" * 95 + "b" * 4 + "c") == 91
