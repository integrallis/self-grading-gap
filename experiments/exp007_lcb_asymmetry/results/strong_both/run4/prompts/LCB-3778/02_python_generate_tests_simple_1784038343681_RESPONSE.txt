from solution import transformArray


def test_canonical_example_1():
    # 4 and 2 are even -> 0; 3 and 1 are odd -> 1; sorting gives [0, 0, 1, 1].
    assert transformArray([4, 3, 2, 1]) == [0, 0, 1, 1]


def test_canonical_example_2():
    # 4 and 2 are even -> 0; 1, 5, and 1 are odd -> 1; sorting gives [0, 0, 1, 1, 1].
    assert transformArray([1, 5, 1, 4, 2]) == [0, 0, 1, 1, 1]


def test_single_element_minimum_length():
    # 1 is odd -> 1; a one-element transformed array is already sorted.
    assert transformArray([1]) == [1]


def test_maximum_length_with_maximum_value():
    nums = [1000] + [1] * 50 + [2] * 49
    # 1000 and the 49 values of 2 are even -> 50 zeros; 50 values of 1 are odd -> 50 ones.
    # Sorting gives all zeros before all ones.
    assert transformArray(nums) == [0] * 50 + [1] * 50
