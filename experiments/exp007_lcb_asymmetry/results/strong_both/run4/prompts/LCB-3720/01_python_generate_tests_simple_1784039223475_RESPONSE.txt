from solution import minMaxWeight


def test_canonical_example_1():
    # Keeping 1->0, 3->0, 4->3, and 2->1 gives every node a path to 0;
    # all kept edges have weight 1, so the minimum possible maximum is 1.
    assert minMaxWeight(
        5,
        [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]],
        2,
    ) == 1


def test_canonical_example_2():
    # Node 2 has no directed path to node 0, so satisfying reachability is impossible.
    assert minMaxWeight(
        5,
        [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]],
        1,
    ) == -1


def test_canonical_example_3():
    # Keeping 1->2 (1), 2->3 (2), 3->4 (2), and 4->0 (1)
    # reaches 0 from every node with maximum kept weight 2.
    assert minMaxWeight(
        5,
        [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]],
        1,
    ) == 2


def test_canonical_example_4():
    # Node 3 has no path to node 0 because the only route onward would require 3->4,
    # which is absent; therefore the required reachability is impossible.
    assert minMaxWeight(
        5,
        [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]],
        1,
    ) == -1


def test_single_required_edge_sets_the_maximum_weight():
    # The only path from node 1 to node 0 is 1->0 of weight 7,
    # so the minimum possible maximum weight is 7.
    assert minMaxWeight(2, [[1, 0, 7]], 1) == 7


def test_cheaper_indirect_paths_are_preferred_over_heavier_direct_edge():
    # Keeping 1->2, 2->0, and 3->1 gives paths to 0 using only weight-3 edges;
    # node 1 cannot reach 0 with a maximum below 3, so the answer is 3.
    assert minMaxWeight(
        4,
        [[1, 0, 9], [1, 2, 3], [2, 0, 3], [3, 1, 3]],
        1,
    ) == 3


def test_multiple_edges_between_the_same_nodes_can_use_the_lighter_weight():
    # Keeping the weight-2 edge 1->0 and 2->1 reaches node 0 from both nodes,
    # making the maximum kept weight 2 instead of the parallel edge's weight 8.
    assert minMaxWeight(3, [[1, 0, 8], [1, 0, 2], [2, 1, 2]], 1) == 2
