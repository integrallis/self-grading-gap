from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    from collections import defaultdict, deque
    import heapq

    # Group edges by weight
    edges_by_weight = defaultdict(list)
    for u, v, w in edges:
        edges_by_weight[w].append((u, v))

    # Function to check if we can reach node 0 from all nodes with given max weight
    def canReachZero(max_weight: int) -> bool:
        graph = defaultdict(list)
        out_degree = defaultdict(int)

        # Build the graph with edges not exceeding max_weight
        for weight in sorted(edges_by_weight.keys()):
            if weight > max_weight:
                break
            for u, v in edges_by_weight[weight]:
                graph[u].append(v)
                out_degree[u] += 1
                if out_degree[u] > threshold:
                    return False

        # Perform BFS from node 0 to check reachability
        reachable = set()
        queue = deque([0])
        while queue:
            node = queue.popleft()
            if node in reachable:
                continue
            reachable.add(node)
            for neighbor in graph[node]:
                if neighbor not in reachable:
                    queue.append(neighbor)

        return len(reachable) == n

    # Binary search over the weights
    low, high = 1, 10**6
    result = -1
    while low <= high:
        mid = (low + high) // 2
        if canReachZero(mid):
            result = mid
            high = mid - 1
        else:
            low = mid + 1

    return result
