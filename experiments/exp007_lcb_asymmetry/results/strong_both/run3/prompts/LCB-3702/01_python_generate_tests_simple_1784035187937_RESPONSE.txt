from solution import maxLength


def test_canonical_example_1():
    # [1, 2, 1, 1, 1] has product 2, gcd 1, and lcm 2, so its length is 5.
    assert maxLength([1, 2, 1, 2, 1, 1, 1]) == 5


def test_canonical_example_2():
    # [3, 4, 5] has product 60, gcd 1, and lcm 60, so its length is 3.
    assert maxLength([2, 3, 4, 5, 6]) == 3


def test_canonical_example_3():
    # [3, 1, 4, 5, 1] has product 60, gcd 1, and lcm 60, so its length is 5.
    assert maxLength([1, 2, 3, 1, 4, 5, 1]) == 5


def test_all_ones_entire_array_is_product_equivalent():
    # Product, gcd, and lcm of [1, 1, 1, 1] are all 1, so the full length is 4.
    assert maxLength([1, 1, 1, 1]) == 4


def test_returns_two_when_no_length_three_subarray_is_product_equivalent():
    # [2, 4, 8] has product 64, while gcd * lcm is 2 * 8 = 16;
    # each length-2 subarray is product equivalent, so the maximum length is 2.
    assert maxLength([2, 4, 8]) == 2
