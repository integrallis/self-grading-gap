from solution import largestInteger


def test_canonical_example_1():
    # 3 occurs only in [3, 9, 2] and 7 only in [2, 1, 7]; the larger is 7.
    assert largestInteger([3, 9, 2, 1, 7], 3) == 7


def test_canonical_example_2():
    # Only 3 appears in exactly one size-4 subarray: [3, 9, 7, 2].
    assert largestInteger([3, 9, 7, 2, 1, 7], 4) == 3


def test_canonical_example_3():
    # Each 0 appears in a different size-1 subarray, so 0 occurs in two subarrays.
    assert largestInteger([0, 0], 1) == -1


def test_full_array_is_the_only_subarray():
    # There is one size-3 subarray, [4, 9, 4], containing both 4 and 9; largest is 9.
    assert largestInteger([4, 9, 4], 3) == 9


def test_repeated_value_in_one_subarray_counts_once_for_that_subarray():
    # The only size-2 subarray is [5, 5], so 5 appears in exactly one subarray.
    assert largestInteger([5, 5], 2) == 5


def test_size_one_selects_largest_value_with_one_occurrence():
    # Size-1 subarrays are [1], [4], [1], [3]; values occurring once are 4 and 3.
    assert largestInteger([1, 4, 1, 3], 1) == 4


def test_returns_minus_one_when_every_value_is_in_multiple_subarrays():
    # Both 1 and 2 occur in each of the two size-2 subarrays, [1, 2] and [1, 2].
    assert largestInteger([1, 2, 1, 2], 2) == -1
