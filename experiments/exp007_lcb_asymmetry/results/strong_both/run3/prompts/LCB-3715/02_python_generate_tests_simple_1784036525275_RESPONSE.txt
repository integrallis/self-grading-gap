from solution import maximumCoins


def test_canonical_example_1():
    # Bags [3, 4, 5, 6] contain 2 + 0 + 4 + 4 = 10 coins.
    assert maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) == 10


def test_canonical_example_2():
    # Any two bags within [1, 10] contain 3 + 3 = 6 coins.
    assert maximumCoins([[1, 10, 3]], 2) == 6


def test_k_one_selects_the_highest_coin_bag():
    # A one-bag window can select bag 3, which contains 7 coins.
    assert maximumCoins([[1, 1, 2], [3, 3, 7]], 1) == 7


def test_singleton_segment_can_be_collected_in_a_larger_window():
    # Any three consecutive bags containing position 5 collect its 9 coins and two zero bags.
    assert maximumCoins([[5, 5, 9]], 3) == 9


def test_optimal_window_ends_partway_through_a_segment():
    # Bags [1, 4] contain 9 + 0 + 2 + 2 = 13 coins.
    assert maximumCoins([[1, 1, 9], [3, 12, 2]], 4) == 13


def test_large_k_includes_zero_coin_bags_outside_segments():
    # A five-bag window containing bags 5 and 6 collects 4 + 4 = 8 coins.
    assert maximumCoins([[5, 6, 4]], 5) == 8


def test_maximum_k_coordinate_and_coin_value_boundaries():
    # The length-10**9 window [1, 10**9] collects 1000 + 1 = 1001 coins.
    assert maximumCoins([[1, 1, 1000], [10**9, 10**9, 1]], 10**9) == 1001


def test_maximum_number_of_segments():
    coins = [[2 * i + 1, 2 * i + 1, (i % 1000) + 1] for i in range(100000)]

    # With k = 1, the best singleton bag has the largest assigned value, 1000.
    assert maximumCoins(coins, 1) == 1000
