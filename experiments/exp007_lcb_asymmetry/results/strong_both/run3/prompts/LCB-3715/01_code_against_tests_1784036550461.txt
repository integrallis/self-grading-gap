```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a dictionary to hold the total coins at each position
    coin_map = {}
    for l, r, c in coins:
        for pos in range(l, r + 1):
            coin_map[pos] = coin_map.get(pos, 0) + c

    # To find the maximum coins in k consecutive bags
    max_coins = 0
    current_coins = 0
    sorted_positions = sorted(coin_map.keys())
    left = 0

    for right in range(len(sorted_positions)):
        current_coins += coin_map[sorted_positions[right]]
        # If the window size exceeds k, slide the window
        while sorted_positions[right] - sorted_positions[left] + 1 > k:
            current_coins -= coin_map[sorted_positions[left]]
            left += 1
        # Update the maximum coins
        max_coins = max(max_coins, current_coins)

    return max_coins
```
