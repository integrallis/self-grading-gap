from solution import subarraySum


def test_canonical_example_1():
    # [2] + [2, 3] + [3, 1] = 2 + 5 + 4 = 11
    assert subarraySum([2, 3, 1]) == 11


def test_canonical_example_2():
    # [3] + [3, 1] + [1, 1] + [1, 1, 2] = 3 + 4 + 2 + 4 = 13
    assert subarraySum([3, 1, 1, 2]) == 13


def test_single_element_array():
    # [1] = 1
    assert subarraySum([1]) == 1


def test_values_larger_than_index_include_entire_prefix():
    # [1] + [1, 2] + [1, 2, 3] = 1 + 3 + 6 = 10
    assert subarraySum([1, 2, 3]) == 10


def test_subarrays_use_value_based_start_index():
    # [1] + [1, 2] + [2, 3] + [3, 1] = 1 + 3 + 5 + 4 = 13
    assert subarraySum([1, 2, 3, 1]) == 13


def test_maximum_length_array():
    # Each index uses its own [1], except index 0 also has [1]:
    # 100 subarrays each sum to 1, plus indices 1 through 99 include one prior 1.
    # Total = 1 + 99 * 2 = 199.
    assert subarraySum([1] * 100) == 199


def test_maximum_element_value():
    # [1000] = 1000
    assert subarraySum([1000]) == 1000
