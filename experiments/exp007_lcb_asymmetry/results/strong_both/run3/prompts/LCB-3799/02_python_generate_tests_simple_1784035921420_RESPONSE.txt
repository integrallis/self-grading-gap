from solution import totalNumbers


def test_canonical_example_1():
    # Even final digits 2 or 4 each allow 3 nonzero first choices and 2 middle choices:
    # 2 * 3 * 2 = 12.
    assert totalNumbers([1, 2, 3, 4]) == 12


def test_canonical_example_2():
    # The distinct valid three-digit even numbers are 202 and 220.
    assert totalNumbers([0, 2, 2]) == 2


def test_canonical_example_3():
    # All three available digits form the single distinct number 666.
    assert totalNumbers([6, 6, 6]) == 1


def test_canonical_example_4():
    # All available digits are odd, so no formed number can be even.
    assert totalNumbers([1, 3, 5]) == 0


def test_maximum_length_and_digit_upper_bound():
    # With digits 0 through 9, choose an even final digit, a nonzero distinct first
    # digit, and then a distinct middle digit: 4*8*8 + 1*9*8 = 328.
    assert totalNumbers(list(range(10))) == 328
