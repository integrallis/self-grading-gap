def test_canonical_example_1():
    # Good elements are 3 (greater than 1), 5 (greater than 2), and 4 (greater than 1): 3 + 5 + 4 = 12.

def test_requires_strictly_greater_not_equal():
    # Each element equals its k-distance counterpart, so no element is strictly greater: sum = 0.
    assert sumOfGoodNumbers([5, 1, 5, 1], 2) == 0
