from solution import hasSpecialSubstring


def test_canonical_example_1():
    # "aaa" at the end has length 3 and is preceded by different character "b".
    assert hasSpecialSubstring("aaabaaa", 3) is True


def test_canonical_example_2():
    # No length-2 substring of "abc" consists of one repeated character.
    assert hasSpecialSubstring("abc", 2) is False


def test_returns_false_when_matching_characters_extend_past_substring_length():
    # Each "aaa" in "aaaa" has an adjacent "a", so no length-3 substring is bounded properly.
    assert hasSpecialSubstring("aaaa", 3) is False


def test_returns_true_for_run_with_exact_requested_length_between_different_characters():
    # "bbbb" has length 4 and is bounded by "a" and "c", both different from "b".
    assert hasSpecialSubstring("abbbbc", 4) is True


def test_returns_true_when_entire_string_is_the_special_substring():
    # "zzzz" has length 4, consists only of "z", and has no neighboring characters.
    assert hasSpecialSubstring("zzzz", 4) is True


def test_returns_true_for_single_character_substring_with_different_neighbors():
    # The "b" in "aba" has length 1 and is bounded by "a" characters different from "b".
    assert hasSpecialSubstring("aba", 1) is True
