```python
from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize the DP table
    dp = [[float('inf')] * 3 for _ in range(3)]

    # Setup initial costs for the color pairs at the outer houses
    for left_color in range(3):
        for right_color in range(3):
            if left_color != right_color:
                dp[left_color][right_color] = cost[0][left_color] + cost[n - 1][right_color]

    # Iterate over pairs of houses
    for i in range(1, n // 2):
        new_dp = [[float('inf')] * 3 for _ in range(3)]
        for left_color in range(3):
            for right_color in range(3):
                if dp[left_color][right_color] == float('inf'):
                    continue  # Skip if there was no valid configuration
                # Check new color pairs
                for new_left in range(3):
                    for new_right in range(3):
                        if new_left != new_right and new_left != left_color and new_right != right_color:
                            new_dp[new_left][new_right] = min(new_dp[new_left][new_right], dp[left_color][right_color] + cost[i][new_left] + cost[n - 1 - i][new_right])
        dp = new_dp  # Move to the next stage of DP

    # Find the minimum cost from the last DP table
    return min(min(dp[left_color][right_color] for right_color in range(3) for left_color in range(3) if left_color != right_color), default=float('inf'))
```
