from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # dp[i][j] will hold the minimum cost to paint house i with color j
    dp = [[0] * 3 for _ in range(n)]

    # Initialize the first house costs
    dp[0][0] = cost[0][0]
    dp[0][1] = cost[0][1]
    dp[0][2] = cost[0][2]

    for i in range(1, n):
        for j in range(3):
            # The cost to paint house i with color j
            dp[i][j] = cost[i][j] + min(
                dp[i-1][(j+1) % 3],  # Previous house with a different color
                dp[i-1][(j+2) % 3]   # Previous house with a different color
            )

    # Calculate the minimum cost considering the equidistant house constraint
    min_cost = float('inf')
    for j in range(3):
        # Compare the costs from the last house with color j
        min_cost = min(min_cost, dp[n-1][j])

    # Now we have to ensure the equidistant condition is satisfied
    for j in range(3):
        # Take the last house with color j
        cost_with_j = dp[n-1][j]
        if n > 2:
            # Add the cost of the first house with a different color than j
            cost_with_j += min(dp[0][(j+1) % 3], dp[0][(j+2) % 3])
        min_cost = min(min_cost, cost_with_j)

    return min_cost
