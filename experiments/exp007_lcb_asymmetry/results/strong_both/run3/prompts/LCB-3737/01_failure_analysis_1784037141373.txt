from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # Possible color pairs for symmetric houses
    min_cost = float('inf')

    # Iterate over the first half of the houses
    for left_color in range(3):  
        for right_color in range(3):
            if left_color == right_color:
                continue  # Skip same color pairs
            total_cost = cost[0][left_color] + cost[n - 1][right_color]

            # Calculate cost for the remaining houses
            for i in range(1, n // 2):
                next_left_color = (left_color + 1) % 3
                next_right_color = (right_color + 1) % 3
                temp_cost = float('inf')
                # Check valid transitions avoiding adjacent same colors
                for new_left_color in range(3):
                    for new_right_color in range(3):
                        if new_left_color != left_color and new_right_color != right_color:
                            temp_cost = min(temp_cost, cost[i][new_left_color] + cost[n - 1 - i][new_right_color])
                total_cost += temp_cost
                left_color = new_left_color
                right_color = new_right_color

            min_cost = min(min_cost, total_cost)

    return min_cost
