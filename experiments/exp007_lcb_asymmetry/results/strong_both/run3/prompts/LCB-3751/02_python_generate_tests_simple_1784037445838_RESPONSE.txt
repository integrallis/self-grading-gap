from solution import maxFrequency


def test_canonical_example_1():
    # Adding -5 to nums[2..5] changes the final 6 to 1, so 1 appears twice.
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2


def test_canonical_example_2():
    # Adding 8 to nums[1..9] changes three 2s into 10; with the original 10, frequency is 4.
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4


def test_keeps_all_existing_target_values_when_no_change_is_best():
    # Choosing x = 0 preserves every element, so all three existing 4s remain.
    assert maxFrequency([4, 4, 4], 4) == 3


def test_converts_matching_values_while_preserving_target_outside_subarray():
    # Adding 2 to nums[1..3] converts both 2s to 4; the 4 at index 0 remains, totaling 3.
    assert maxFrequency([4, 2, 4, 2, 3], 4) == 3


def test_single_element_array():
    # Adding 0 to the only element keeps it equal to k, so its frequency is 1.
    assert maxFrequency([7], 7) == 1


def test_maximum_array_length():
    # Choosing x = 0 preserves all 100000 occurrences of 50.
    assert maxFrequency([50] * 100000, 50) == 100000


def test_maximum_permitted_value_for_nums_and_target():
    # Adding 49 to the final two elements converts both 1s to 50, totaling three 50s.
    assert maxFrequency([50, 1, 1], 50) == 3
