```python
from typing import List


def maxFrequency(nums: List[int], k: int) -> int:
    n = len(nums)
    count = 0
    total_add = 0
    left = 0

    # Sort the array to use a sliding window approach
    nums.sort()

    for right in range(n):
        # Calculate the total number of elements to add to convert
        # nums[left..right] to k
        total_add += nums[right]

        while (right - left + 1) * k > total_add + (right - left + 1) * k:
            total_add -= nums[left]
            left += 1

        count = max(count, right - left + 1)

    return count

```
