```python
from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[[-float('inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0]  # Starting point with 0 robbers neutralized

    for i in range(m):
        for j in range(n):
            for k in range(3):  # k represents the number of robbers neutralized
                if i == 0 and j == 0:
                    continue  # Skip the starting cell

                if i > 0:
                    dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k] + coins[i][j])
                if j > 0:
                    dp[i][j][k] = max(dp[i][j][k], dp[i][j-1][k] + coins[i][j])

                # If the current cell is a robber
                if coins[i][j] < 0:
                    if k > 0:
                        dp[i][j][k] = max(dp[i][j][k], dp[i][j][k-1] + coins[i][j])

    # Get the maximum coins collected at the bottom-right corner with at most 2 neutralizations
    return max(dp[m-1][n-1][0], dp[m-1][n-1][1], dp[m-1][n-1][2])
```
