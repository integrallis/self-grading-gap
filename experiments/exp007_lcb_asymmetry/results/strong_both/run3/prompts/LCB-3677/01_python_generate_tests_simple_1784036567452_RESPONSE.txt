from solution import maximumAmount


def test_canonical_example_1():
    # 0 + 1 + neutralized(-2) + 3 + 4 = 8.
    assert maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) == 8


def test_canonical_example_2():
    # A route through four cells each worth 10 gains 10 + 10 + 10 + 10 = 40.
    assert maximumAmount([[10, 10, 10], [10, 10, 10]]) == 40


def test_single_negative_start_and_destination_can_be_neutralized():
    # The only cell is a robber, so neutralizing it changes -7 to 0.
    assert maximumAmount([[-7]]) == 0


def test_at_most_two_robbers_can_be_neutralized_and_total_can_remain_negative():
    # On the only route, neutralize -5 and -4; the remaining total is -3 + -2 = -5.
    assert maximumAmount([[-5, -4, -3, -2]]) == -5


def test_chooses_the_best_route_instead_of_neutralizing_more_robbers_on_another_route():
    # Right, right, down with -100 neutralized gives 0 + 0 + 100 + 10 = 110.
    assert maximumAmount([[0, -100, 100], [10, -1, 10]]) == 110
