from typing import List
import math

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Calculate contribution of each number as min and max in subsequences
    for i in range(n):
        # Each number nums[i] can be the minimum or maximum of all subsequences that include it
        # It can participate in C(n-1, j-1) subsequences of size j (1 <= j <= k)
        for j in range(1, k + 1):
            if j <= n:
                contribution = nums[i] * math.comb(n - 1, j - 1)
                total_sum += contribution * 2  # Each contributes as both min and max
                total_sum %= MOD

    return total_sum

```
