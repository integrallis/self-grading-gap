import math

from solution import minMaxSums


def test_canonical_example_1():
    # [1], [2], [3], [1,2], [1,3], [2,3] contribute
    # 2 + 4 + 6 + 3 + 4 + 5 = 24.
    assert minMaxSums([1, 2, 3], 2) == 24


def test_canonical_example_2():
    # With k=1, each singleton contributes twice its value:
    # 2*5 + 2*0 + 2*6 = 22.
    assert minMaxSums([5, 0, 6], 1) == 22


def test_canonical_example_3():
    # Three singleton and three two-element subsequences each contribute 1 + 1:
    # 6 * 2 = 12.
    assert minMaxSums([1, 1, 1], 2) == 12


def test_includes_subsequences_up_to_the_full_array_length():
    # For [2,1,3] and k=3:
    # singletons: 4 + 2 + 6 = 12;
    # pairs: 3 + 5 + 4 = 12;
    # full array: 1 + 3 = 4; total = 28.
    assert minMaxSums([2, 1, 3], 3) == 28


def test_returns_result_modulo_1000000007():
    # With three values of 1,000,000,000 and k=2, there are
    # 3 singletons and 3 pairs, each contributing 2,000,000,000:
    # 6 * 2,000,000,000 = 12,000,000,000,
    # and 12,000,000,000 mod 1,000,000,007 = 999,999,923.
    assert minMaxSums([1_000_000_000, 1_000_000_000, 1_000_000_000], 2) == 999_999_923


def test_singleton_array_boundary():
    # The only subsequence is [7], whose minimum plus maximum is 7 + 7 = 14.
    assert minMaxSums([7], 1) == 14


def test_maximum_length_and_k():
    # Every subsequence of r identical ones contributes 1 + 1 = 2.
    # There are C(100000, r) subsequences for each r from 1 through 70.
    nums = [1] * 100_000
    expected = (2 * sum(math.comb(100_000, r) for r in range(1, 71))) % 1_000_000_007
    assert minMaxSums(nums, 70) == expected
