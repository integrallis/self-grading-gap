```python
from typing import List
import math

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    min_sum = 0
    max_sum = 0

    for num in nums:
        # Each element can serve as a minimum (in subsequences of size 1 to k)
        min_sum += num * (math.comb(n - 1, k - 1) if k <= n else 0)
        # Each element can serve as a maximum
        max_sum += num * (math.comb(n - 1, k - 1) if k <= n else 0)

    total_sum = (min_sum + max_sum) % MOD
    return total_sum
```
