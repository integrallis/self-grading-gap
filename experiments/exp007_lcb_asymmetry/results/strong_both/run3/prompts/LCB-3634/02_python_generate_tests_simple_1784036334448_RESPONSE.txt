from solution import calculateScore


def test_canonical_example_1():
    # spec: pairs (0, 2) for 2 points and (1, 4) for 3 points; total = 5
    assert calculateScore("aczzx") == 5


def test_canonical_example_2():
    # spec: no character has an earlier unmarked mirror, so total = 0
    assert calculateScore("abcdef") == 0


def test_marked_indices_cannot_be_reused():
    # i=1 pairs with i=0 for 1; i=3 pairs with i=2 for 1; total = 2
    assert calculateScore("azaz") == 2


def test_matches_use_available_unmarked_mirrors():
    # i=2 pairs with closest z at i=1 for 1; i=3 pairs with remaining z at i=0 for 3; total = 4
    assert calculateScore("zzaa") == 4


def test_minimum_length_string_has_no_earlier_mirror():
    # The only index has no earlier index, so total = 0
    assert calculateScore("a") == 0


def test_another_reversed_alphabet_mirror_pair():
    # b at i=1 mirrors y at i=0, contributing 1 - 0 = 1
    assert calculateScore("yb") == 1


def test_maximum_length_string_accumulates_all_pair_distances():
    # Each z at index 50000+k pairs with a at index 49999-k:
    # sum((50000+k) - (49999-k) for k in range(50000)) = 2,500,000,000
    s = "a" * 50000 + "z" * 50000
    assert calculateScore(s) == 2500000000
