from solution import countMentions


def test_canonical_example_1():
    events = [
        ["MESSAGE", "10", "id1 id0"],
        ["OFFLINE", "11", "0"],
        ["MESSAGE", "71", "HERE"],
    ]

    # At 10 both users are explicitly mentioned; at 71 user 0 is online again, so HERE adds one each.
    assert countMentions(2, events) == [2, 2]


def test_canonical_example_2():
    events = [
        ["MESSAGE", "10", "id1 id0"],
        ["OFFLINE", "11", "0"],
        ["MESSAGE", "12", "ALL"],
    ]

    # The explicit message gives one each, and ALL at 12 includes offline user 0, giving two each.
    assert countMentions(2, events) == [2, 2]


def test_canonical_example_3():
    events = [
        ["OFFLINE", "10", "0"],
        ["MESSAGE", "12", "HERE"],
    ]

    # User 0 remains offline at 12, so HERE mentions only user 1.
    assert countMentions(2, events) == [0, 1]


def test_explicit_ids_count_duplicates_and_can_mention_offline_users():
    events = [
        ["OFFLINE", "5", "1"],
        ["MESSAGE", "6", "id1 id1 id2"],
    ]

    # Offline user 1 is explicitly mentioned twice and user 2 once; user 0 is not mentioned.
    assert countMentions(3, events) == [0, 2, 1]


def test_offline_status_change_precedes_message_at_same_timestamp():
    events = [
        ["MESSAGE", "10", "HERE"],
        ["OFFLINE", "10", "0"],
    ]

    # The offline change at 10 is processed before HERE at 10, so only user 1 is mentioned.
    assert countMentions(2, events) == [0, 1]


def test_here_uses_each_users_individual_return_time():
    events = [
        ["OFFLINE", "1", "0"],
        ["OFFLINE", "2", "1"],
        ["MESSAGE", "61", "HERE"],
    ]

    # User 0 returns at 61, user 1 remains offline until 62, and user 2 was always online.
    assert countMentions(3, events) == [1, 0, 1]


def test_events_are_processed_by_timestamp_for_online_status():
    events = [
        ["MESSAGE", "70", "HERE"],
        ["OFFLINE", "10", "0"],
        ["MESSAGE", "9", "HERE"],
    ]

    # At 9 HERE mentions both; user 0 returns at 70, so HERE at 70 also mentions both.
    assert countMentions(2, events) == [2, 2]


def test_minimum_number_of_users_with_one_event():
    events = [["MESSAGE", "1", "HERE"]]

    # The only initially online user is mentioned once by HERE.
    assert countMentions(1, events) == [1]


def test_maximum_number_of_users_can_mention_id99():
    events = [["MESSAGE", "1", "id99"]]

    # Only user 99 is explicitly mentioned once.
    assert countMentions(100, events) == [0] * 99 + [1]


def test_exactly_one_explicit_id_token_counts_once():
    events = [["MESSAGE", "1", "id0"]]

    # The one explicit id0 token produces one mention for user 0.
    assert countMentions(1, events) == [1]


def test_one_hundred_duplicate_explicit_id_tokens_count_individually():
    events = [["MESSAGE", "1", " ".join(["id0"] * 100)]]

    # Each of the 100 id0 tokens is a separate mention of user 0.
    assert countMentions(1, events) == [100]


def test_one_hundred_events_accumulate_mentions():
    events = [["MESSAGE", str(timestamp), "id0"] for timestamp in range(1, 101)]

    # Each of the 100 message events explicitly mentions user 0 once.
    assert countMentions(1, events) == [100]


def test_timestamp_bounds_apply_status_changes_chronologically():
    events = [
        ["MESSAGE", "100000", "HERE"],
        ["OFFLINE", "1", "0"],
        ["MESSAGE", "1", "HERE"],
    ]

    # At time 1 user 0 goes offline before HERE, so only user 1 is mentioned;
    # by time 100000 user 0 has returned, so HERE mentions both users.
    assert countMentions(2, events) == [1, 2]
