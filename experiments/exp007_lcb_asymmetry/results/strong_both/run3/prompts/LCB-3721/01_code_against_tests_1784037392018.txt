```python
from typing import List


def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers
    online = [True] * numberOfUsers
    return_time = [0] * numberOfUsers

    # Sort events by timestamp
    events.sort(key=lambda x: int(x[1]))

    for event in events:
        event_type, timestamp_str, data = event
        timestamp = int(timestamp_str)

        # Update online status for users going offline at this timestamp
        for user_id in range(numberOfUsers):
            if not online[user_id] and return_time[user_id] <= timestamp:
                online[user_id] = True

        # Process offline events first
        if event_type == 'OFFLINE':
            user_id = int(data)
            online[user_id] = False
            return_time[user_id] = timestamp + 60

        # Process message events
        elif event_type == 'MESSAGE':
            mentions_list = data.split()
            for mention in mentions_list:
                if mention.startswith('id'):
                    user_id = int(mention[2:])
                    mentions[user_id] += 1
                elif mention == 'ALL':
                    for user_id in range(numberOfUsers):
                        mentions[user_id] += 1
                elif mention == 'HERE':
                    for user_id in range(numberOfUsers):
                        if online[user_id]:
                            mentions[user_id] += 1

    return mentions
```
