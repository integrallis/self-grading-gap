from solution import findValidPair


def test_canonical_example_1():
    # '2' occurs twice and '3' occurs three times; the first valid adjacent pair is "23".
    assert findValidPair("2523533") == "23"


def test_canonical_example_2():
    # '2' occurs twice and '1' occurs once; the first valid adjacent pair is "21".
    assert findValidPair("221") == "21"


def test_canonical_example_3():
    # The only adjacent pair is "22", whose digits are equal, so no valid pair exists.
    assert findValidPair("22") == ""


def test_returns_leftmost_when_multiple_valid_pairs_exist():
    # In "122333", counts are '1': 1, '2': 2, '3': 3; both "12" and later "23" are valid,
    # so left-to-right traversal returns "12".
    assert findValidPair("122333") == "12"


def test_returns_empty_when_adjacent_digits_do_not_have_required_counts():
    # In "1234", only '1' occurs once; no adjacent pair has both digits occurring their values.
    assert findValidPair("1234") == ""
