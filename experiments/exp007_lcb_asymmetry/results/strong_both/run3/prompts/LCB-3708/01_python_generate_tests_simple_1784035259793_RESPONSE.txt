from solution import zigzagTraversal


def test_canonical_example_1():
    grid = [[1, 2], [3, 4]]

    # Zigzag order is 1, 2, 4, 3; visiting alternate cells gives 1, 4.
    assert zigzagTraversal(grid) == [1, 4]


def test_canonical_example_2():
    grid = [[2, 1], [2, 1], [2, 1]]

    # Zigzag order is 2, 1, 1, 2, 2, 1; visiting alternate cells gives 2, 1, 2.
    assert zigzagTraversal(grid) == [2, 1, 2]


def test_canonical_example_3():
    grid = [[1, 2, 3], [4, 5, 6], [7, 8, 9]]

    # Zigzag order is 1, 2, 3, 6, 5, 4, 7, 8, 9; visiting alternate cells gives 1, 3, 5, 7, 9.
    assert zigzagTraversal(grid) == [1, 3, 5, 7, 9]
