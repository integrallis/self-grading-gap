from solution import zigzagTraversal


def test_canonical_example_1():
    grid = [[1, 2], [3, 4]]

    # Zigzag order is 1, 2, 4, 3; visiting alternate cells gives 1, 4.
    assert zigzagTraversal(grid) == [1, 4]


def test_canonical_example_2():
    grid = [[2, 1], [2, 1], [2, 1]]

    # Zigzag order is 2, 1, 1, 2, 2, 1; visiting alternate cells gives 2, 1, 2.
    assert zigzagTraversal(grid) == [2, 1, 2]


def test_canonical_example_3():
    grid = [[1, 2, 3], [4, 5, 6], [7, 8, 9]]

    # Zigzag order is 1, 2, 3, 6, 5, 4, 7, 8, 9; visiting alternate cells gives 1, 3, 5, 7, 9.
    assert zigzagTraversal(grid) == [1, 3, 5, 7, 9]


def test_maximum_row_count_with_distinct_values():
    grid = [[2 * row + 1, 2 * row + 2] for row in range(50)]

    # Each two-cell row starts at an even traversal position: select its left value on
    # left-to-right rows and its right value on right-to-left rows.
    expected = [
        1, 4, 5, 8, 9, 12, 13, 16, 17, 20,
        21, 24, 25, 28, 29, 32, 33, 36, 37, 40,
        41, 44, 45, 48, 49, 52, 53, 56, 57, 60,
        61, 64, 65, 68, 69, 72, 73, 76, 77, 80,
        81, 84, 85, 88, 89, 92, 93, 96, 97, 100,
    ]

    assert zigzagTraversal(grid) == expected


def test_maximum_column_count_with_distinct_values():
    grid = [list(range(1, 51)), list(range(51, 101))]

    # Select odd-positioned values from the first row, then every other value while
    # traversing the second row from 100 down to 51.
    expected = list(range(1, 51, 2)) + list(range(100, 50, -2))

    assert zigzagTraversal(grid) == expected


def test_maximum_cell_value_is_returned_when_visited():
    grid = [[2500, 1], [2, 3]]

    # Zigzag order is 2500, 1, 3, 2; visiting alternate cells gives 2500, 3.
    assert zigzagTraversal(grid) == [2500, 3]
