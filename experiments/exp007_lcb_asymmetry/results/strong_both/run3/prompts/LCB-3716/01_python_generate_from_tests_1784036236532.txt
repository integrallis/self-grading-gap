from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = {}  # Function to hold lengths keyed by (index, last_diff)

    max_length = 1

    # Initialize dp for all pairs (i, j) with valid lengths 2 subsequences
    for i in range(n):
        for j in range(i):
            last_diff = abs(nums[i] - nums[j])
            dp[(i, last_diff)] = max(dp.get((i, last_diff), 0), 2)  # Length of 2 subsequence from (j, i)
            for k in range(j):
                prev_diff = abs(nums[j] - nums[k])
                if prev_diff >= last_diff:  # Ensure the non-increasing condition
                    dp[(i, last_diff)] = max(dp[(i, last_diff)], dp.get((j, prev_diff), 1) + 1)
                    max_length = max(max_length, dp[(i, last_diff)])

    return max_length
```
