from solution import longestSubsequence


def test_canonical_example_1():
    # [16, 6, 3] has adjacent absolute differences [10, 3], and 10 >= 3.
    assert longestSubsequence([16, 6, 3]) == 3


def test_canonical_example_2():
    # [6, 4, 2, 1] is a subsequence with differences [2, 2, 1],
    # and 2 >= 2 >= 1; no valid subsequence has length 5 or 6.
    assert longestSubsequence([6, 5, 3, 4, 2, 1]) == 4


def test_canonical_example_3():
    # [10, 20, 10, 19, 10] has differences [10, 10, 9, 9],
    # and 10 >= 10 >= 9 >= 9; no valid subsequence has length 6.
    assert longestSubsequence([10, 20, 10, 19, 10, 20]) == 5


def test_two_elements_form_a_valid_subsequence():
    # Any two-element subsequence has only one difference, so it is valid.
    assert longestSubsequence([1, 300]) == 2


def test_all_equal_values_keep_zero_differences():
    # The whole array has differences [0, 0, 0], and 0 >= 0 >= 0.
    assert longestSubsequence([7, 7, 7, 7]) == 4


def test_subsequence_can_skip_elements_to_make_differences_non_increasing():
    # [1, 4, 7] is a subsequence with differences [3, 3].
    # The full array has differences [1, 2, 3], which are increasing.
    assert longestSubsequence([1, 2, 4, 7]) == 3
