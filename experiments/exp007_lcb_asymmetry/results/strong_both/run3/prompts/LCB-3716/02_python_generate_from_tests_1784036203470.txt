from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = [1] * n  # Each element is a subsequence of at least length 1.

    for i in range(n):
        for j in range(i):
            diff1 = abs(nums[i] - nums[j])
            if j == 0 or abs(nums[j] - nums[j-1]) >= diff1:
                dp[i] = max(dp[i], dp[j] + 1)

    return max(dp)
```
