```python
from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = [1] * n  # Each element is a subsequence of at least length 1.
    max_len = 1

    for i in range(n):
        for j in range(i):
            diff1 = abs(nums[i] - nums[j])
            # Check for valid subsequence condition
            if dp[j] > 1:
                prev_diff = abs(nums[j] - nums[j - 1]) if j > 0 else float('inf')
                if diff1 >= prev_diff:
                    dp[i] = max(dp[i], dp[j] + 1)
                    max_len = max(max_len, dp[i])

    return max_len

```
