from solution import maxDistance


def test_canonical_example_1():
    # Change "NWSE" to "NWNE"; after "NWN" the position is (-1, 2), distance 3.
    assert maxDistance("NWSE", 1) == 3


def test_canonical_example_2():
    # Change "NSWWEW" to "NNWWWW"; the final position is (-4, 2), distance 6.
    assert maxDistance("NSWWEW", 3) == 6


def test_no_changes_uses_the_best_prefix_not_necessarily_final_position():
    # Positions for "NS" have distances 1 then 0, so the maximum at any time is 1.
    assert maxDistance("NS", 0) == 1


def test_one_change_can_make_all_moves_in_a_prefix_point_away_from_origin():
    # Change the first "S" in "SNNN" to "N"; after four moves the distance is 4.
    assert maxDistance("SNNN", 1) == 4


def test_at_most_k_changes_allows_leaving_already_optimal_moves_unchanged():
    # "NN" already reaches position (0, 2), whose Manhattan distance is 2.
    assert maxDistance("NN", 1) == 2


def test_changing_every_character_can_reach_distance_equal_to_string_length():
    # Change all four moves in "NSEW" to "N"; after four moves the distance is 4.
    assert maxDistance("NSEW", 4) == 4


def test_minimum_length_string():
    # A single north move reaches position (0, 1), whose distance is 1.
    assert maxDistance("N", 0) == 1


def test_maximum_length_string():
    # 100000 north moves reach position (0, 100000), whose distance is 100000.
    assert maxDistance("N" * 100000, 0) == 100000
