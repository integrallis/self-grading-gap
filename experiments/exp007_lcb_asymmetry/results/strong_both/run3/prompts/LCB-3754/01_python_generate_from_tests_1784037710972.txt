def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    for char in s:
        if char == 'N':
            y += 1
        elif char == 'S':
            y -= 1
        elif char == 'E':
            x += 1
        elif char == 'W':
            x -= 1
        max_distance = max(max_distance, abs(x) + abs(y))
    # Calculate how many opposing moves can be changed
    opposing_moves = 0
    if 'N' in s and 'S' in s:
        opposing_moves += min(s.count('N'), s.count('S'))
    if 'E' in s and 'W' in s:
        opposing_moves += min(s.count('E'), s.count('W'))
    # The maximum distance achievable is the max distance + changes we can make
    max_possible_distance = max_distance + min(k, opposing_moves * 2)
    return max_possible_distance
```
