```python
def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    for char in s:
        if char == 'N':
            y += 1
        elif char == 'S':
            y -= 1
        elif char == 'E':
            x += 1
        elif char == 'W':
            x -= 1
        max_distance = max(max_distance, abs(x) + abs(y))
    # Total moves made
    total_moves = len(s)
    # The best distance we can achieve with k changes
    max_possible_distance = max_distance + k
    # The maximum distance is limited to total moves as we can't exceed that
    return min(max_possible_distance, total_moves)
```
