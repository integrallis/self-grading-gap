from solution import minCosts


def test_canonical_example_1():
    # Pay 5, then 3; person 1 enables positions 2 for free; person 3 costs 1 and enables 4 and 5.
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]


def test_canonical_example_2():
    # Paying person 0 costs 1, and every later position is then reachable for free.
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]


def test_single_person_line():
    # The only reachable position requires swapping with person 0, costing 100.
    assert minCosts([100]) == [100]


def test_each_new_lower_cost_reduces_future_minimum():
    # Costs to positions are 4, then min(4, 2)=2, then min(2, 3)=2, then min(2, 1)=1.
    assert minCosts([4, 2, 3, 1]) == [4, 2, 2, 1]
