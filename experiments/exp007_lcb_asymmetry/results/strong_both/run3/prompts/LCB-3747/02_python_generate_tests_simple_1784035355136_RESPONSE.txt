from solution import maxAdjacentDistance


def test_canonical_example_1():
    # max(|1 - 2|, |2 - 4|, |4 - 1|) = max(1, 2, 3) = 3
    assert maxAdjacentDistance([1, 2, 4]) == 3


def test_canonical_example_2():
    # max(|-5 - (-10)|, |-10 - (-5)|, |-5 - (-5)|) = max(5, 5, 0) = 5
    assert maxAdjacentDistance([-5, -10, -5]) == 5


def test_minimum_length_with_value_boundaries():
    # The two adjacent circular comparisons are both |-100 - 100| = 200.
    assert maxAdjacentDistance([-100, 100]) == 200


def test_maximum_length_with_value_boundaries():
    # |-100 - 100| = 200; every other adjacent difference is at most |100 - 0| = 100.
    assert maxAdjacentDistance([-100, 100] + [0] * 98) == 200
