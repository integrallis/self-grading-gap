from solution import maxContainers


def test_canonical_example_1():
    # Deck holds 2 * 2 = 4 containers; weight allows 15 // 3 = 5, so result is 4.
    assert maxContainers(2, 3, 15) == 4


def test_canonical_example_2():
    # Deck holds 3 * 3 = 9 containers; weight allows 20 // 5 = 4, so result is 4.
    assert maxContainers(3, 5, 20) == 4


def test_lower_constraint_boundaries():
    # A 1 * 1 deck has 1 cell, and capacity allows 1 // 1 = 1 container.
    assert maxContainers(1, 1, 1) == 1


def test_insufficient_weight_capacity_loads_no_containers():
    # Capacity allows 1 // 2 = 0 containers, despite the deck having 1 cell.
    assert maxContainers(1, 2, 1) == 0


def test_upper_constraint_boundaries():
    # Deck holds 1000 * 1000 = 1_000_000 containers; weight allows 10**9 // 1000 = 1_000_000.
    assert maxContainers(1000, 1000, 10**9) == 1_000_000
