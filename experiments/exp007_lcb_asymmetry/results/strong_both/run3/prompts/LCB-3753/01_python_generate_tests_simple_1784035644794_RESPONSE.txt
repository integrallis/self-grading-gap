from solution import maxDifference


def test_canonical_example_1():
    # 'a' occurs 5 times (odd) and 'b' occurs 2 times (even), so 5 - 2 = 3.
    assert maxDifference("aaaaabbc") == 3


def test_canonical_example_2():
    # 'a' occurs 3 times (odd) and 'c' occurs 2 times (even), so 3 - 2 = 1.
    assert maxDifference("abcabcab") == 1


def test_uses_largest_odd_frequency():
    # Odd frequencies are 3 ('a') and 5 ('c'); even frequency is 4 ('b'), so 5 - 4 = 1.
    assert maxDifference("aaabbbbccccc") == 1


def test_uses_smallest_even_frequency():
    # Odd frequency is 5 ('a'); even frequencies are 4 ('b') and 2 ('c'), so 5 - 2 = 3.
    assert maxDifference("aaaaabbbbcc") == 3


def test_can_return_negative_difference():
    # Odd frequency is 1 ('b'); even frequencies are 4 ('a') and 2 ('c'), so 1 - 2 = -1.
    assert maxDifference("aaaabcc") == -1
