from solution import minMaxWeight


def test_canonical_example_1():
    # Keeping 1->0, 3->0, 4->3, and 2->1 makes every node reach 0 with maximum weight 1.
    assert minMaxWeight(
        5,
        [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]],
        2,
    ) == 1


def test_canonical_example_2():
    # Node 2 has no directed path to node 0, so no permitted edge removals can satisfy reachability.
    assert minMaxWeight(
        5,
        [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]],
        1,
    ) == -1


def test_canonical_example_3():
    # Keep 1->2 (1), 2->3 (2), 3->4 (2), and 4->0 (1), so the maximum is 2.
    assert minMaxWeight(
        5,
        [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]],
        1,
    ) == 2


def test_canonical_example_4():
    # Node 3 has no path to node 0 because the only route through node 4 is absent.
    assert minMaxWeight(
        5,
        [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]],
        1,
    ) == -1


def test_chooses_lower_bottleneck_indirect_path():
    # Node 1 can reach 0 directly with weight 10 or through 2 with weights 2 and 3;
    # together with 3->0 (4), the minimum possible maximum is 4.
    assert minMaxWeight(
        4,
        [[1, 0, 10], [1, 2, 2], [2, 0, 3], [3, 0, 4]],
        1,
    ) == 4


def test_multiple_edges_between_same_nodes_can_use_lighter_edge():
    # Keep 1->0 (2) instead of the parallel weight-9 edge, plus 2->1 (3);
    # all nodes reach 0 and the maximum retained weight is 3.
    assert minMaxWeight(
        3,
        [[1, 0, 9], [1, 0, 2], [2, 1, 3]],
        1,
    ) == 3
