from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Binary search to find the minimum possible maximum weight
    left, right = 1, 10**6

    def can_form_graph(max_weight: int) -> bool:
        from collections import defaultdict, deque

        # Create a graph with edges filtered by max_weight
        graph = defaultdict(list)
        out_degree = [0] * n

        for A, B, W in edges:
            if W <= max_weight:
                graph[B].append(A)
                out_degree[A] += 1

        # BFS to check reachability to node 0
        queue = deque([0])
        reachable = set([0])

        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if neighbor not in reachable:
                    reachable.add(neighbor)
                    queue.append(neighbor)

        # Check if all nodes are reachable
        if len(reachable) != n:
            return False

        # Check if outgoing edges of each node respect the threshold
        for A in range(n):
            if out_degree[A] > threshold:
                return False

        return True

    # Perform binary search to find the minimum possible maximum weight
    while left < right:
        mid = (left + right) // 2
        if can_form_graph(mid):
            right = mid
        else:
            left = mid + 1

    return left if can_form_graph(left) else -1
```
