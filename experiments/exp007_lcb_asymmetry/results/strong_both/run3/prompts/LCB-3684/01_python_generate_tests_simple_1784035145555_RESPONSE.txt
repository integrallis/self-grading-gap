from solution import hasMatch


def test_canonical_example_1():
    # "ee*e" becomes "eetcode" when * is replaced by "tcod", a substring of "leetcode".
    assert hasMatch("leetcode", "ee*e") is True


def test_canonical_example_2():
    # No replacement for * makes "c*v" match a substring of "car".
    assert hasMatch("car", "c*v") is False


def test_canonical_example_3():
    # "u*" becomes "u" when * is empty, and "u" is a substring of "luck".
    assert hasMatch("luck", "u*") is True


def test_star_can_match_zero_characters():
    # "ab*c" becomes "abc" when * is empty, and "abc" is a substring of "abc".
    assert hasMatch("abc", "ab*c") is True


def test_match_can_be_a_proper_substring():
    # "ab*cd" becomes "abxxcd" when * is "xx", a substring of "zzabxxcdyy".
    assert hasMatch("zzabxxcdyy", "ab*cd") is True


def test_star_at_beginning_can_match_prefix_before_suffix():
    # "*bc" becomes "abc" when * is "a", and "abc" is a substring of "abc".
    assert hasMatch("abc", "*bc") is True


def test_required_parts_must_appear_in_order():
    # "c*a" cannot match a substring of "abc" because no "a" occurs after "c".
    assert hasMatch("abc", "c*a") is False


def test_star_only_can_match_a_nonempty_substring():
    # "*" becomes "a" when * is "a", and "a" is a non-empty substring of "a".
    assert hasMatch("a", "*") is True
