from solution import minimumPairRemoval


def test_canonical_example_1():
    # (3, 1) has minimum sum 4 -> [5, 2, 4];
    # then (2, 4) has minimum sum 6 -> [5, 6], so 2 operations.
    assert minimumPairRemoval([5, 2, 3, 1]) == 2


def test_canonical_example_2():
    # [1, 2, 2] is already non-decreasing, so no operations are needed.
    assert minimumPairRemoval([1, 2, 2]) == 0


def test_single_element_is_non_decreasing():
    # A one-element array has no previous element to violate ordering, so 0 operations.
    assert minimumPairRemoval([7]) == 0


def test_leftmost_pair_is_chosen_when_minimum_sums_tie():
    # Both adjacent pairs sum to 3; choose leftmost (2, 1) -> [3, 2].
    # Then merge (3, 2) -> [5], requiring 2 operations.
    assert minimumPairRemoval([2, 1, 2]) == 2


def test_negative_values_participate_in_minimum_sum_selection():
    # (-1, 2) sums to 1 and (2, -3) sums to -1, so merge (2, -3) -> [-1, -1].
    # The result is non-decreasing after 1 operation.
    assert minimumPairRemoval([-1, 2, -3]) == 1


def test_maximum_length_already_non_decreasing():
    # Fifty zeros are non-decreasing because every element equals its previous element,
    # so 0 operations are needed.
    assert minimumPairRemoval([0] * 50) == 0


def test_numeric_endpoints_can_require_a_merge():
    # 1000 > -1000, so the array is not non-decreasing; its only pair merges to [0],
    # which is non-decreasing, requiring 1 operation.
    assert minimumPairRemoval([1000, -1000]) == 1
