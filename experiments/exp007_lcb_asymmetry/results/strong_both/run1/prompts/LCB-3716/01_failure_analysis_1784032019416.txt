from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n < 2:
        return n

    # Initialize a list to store the maximum length of subsequence ending at each index with their last differences
    states = [{} for _ in range(n)]

    # Iterate through the nums array to fill the states array
    for i in range(n):
        for j in range(i):
            # Calculate the absolute difference
            d = abs(nums[i] - nums[j])
            best = 2  # At least nums[j] and nums[i] can form a subsequence
            # Extend the states based on the valid subsequences ending at j
            for previous_d, length in states[j].items():
                if previous_d >= d:
                    best = max(best, length + 1)
            states[i][d] = max(states[i].get(d, 0), best)

    return max((max(state.values()) for state in states), default=1)
