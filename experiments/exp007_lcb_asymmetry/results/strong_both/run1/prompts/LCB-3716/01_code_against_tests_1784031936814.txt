```python
from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n < 2:
        return n

    # Initialize a list to store the maximum length of subsequence ending at each index
    dp = [1] * n

    # Iterate through the nums array to fill the dp array
    for i in range(n):
        for j in range(i):
            # Calculate the absolute difference
            diff1 = abs(nums[i] - nums[j])
            # Compare with previous elements to ensure non-increasing differences
            is_non_increasing = True
            for k in range(j):
                diff2 = abs(nums[j] - nums[k])
                if diff1 > diff2:
                    is_non_increasing = False
                    break
            if is_non_increasing:
                dp[i] = max(dp[i], dp[j] + 1)

    return max(dp)
```
