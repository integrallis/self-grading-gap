from solution import longestSubsequence


def test_canonical_example_1():
    # [16, 6, 3] has adjacent differences [10, 3], and 10 >= 3.
    assert longestSubsequence([16, 6, 3]) == 3


def test_canonical_example_2():
    # The subsequence [6, 4, 2, 1] has differences [2, 2, 1], where 2 >= 2 >= 1.
    assert longestSubsequence([6, 5, 3, 4, 2, 1]) == 4


def test_canonical_example_3():
    # The subsequence [10, 20, 10, 19, 10] has differences [10, 10, 9, 9].
    assert longestSubsequence([10, 20, 10, 19, 10, 20]) == 5


def test_equal_adjacent_differences_are_non_increasing():
    # [1, 300, 1, 300] has differences [299, 299, 299], which are non-increasing.
    assert longestSubsequence([1, 300, 1, 300]) == 4


def test_zero_differences_are_allowed():
    # [5, 5, 5, 5] has differences [0, 0, 0], which are non-increasing.
    assert longestSubsequence([5, 5, 5, 5]) == 4


def test_subsequence_can_skip_elements_to_preserve_difference_order():
    # [1, 4, 7] is a subsequence with differences [3, 3]; all four elements give [1, 2, 3].
    assert longestSubsequence([1, 2, 4, 7]) == 3


def test_two_element_array_has_length_two_subsequence():
    # Any two-element sequence has one adjacent difference, so [1, 300] is valid.
    assert longestSubsequence([1, 300]) == 2
