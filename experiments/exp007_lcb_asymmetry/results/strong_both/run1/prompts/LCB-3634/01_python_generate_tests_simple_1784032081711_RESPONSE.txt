from solution import calculateScore


def test_canonical_example_1():
    # spec: "aczzx" pairs indices (0, 2) for 2 and (1, 4) for 3; total = 2 + 3 = 5
    assert calculateScore("aczzx") == 5


def test_canonical_example_2():
    # spec: "abcdef" has no earlier unmarked mirror for any character; total = 0
    assert calculateScore("abcdef") == 0


def test_single_character_has_no_possible_prior_mirror():
    # index 0 has no earlier index; total = 0
    assert calculateScore("a") == 0


def test_uses_closest_unmarked_matching_mirror():
    # "aazz": index 2 pairs with closest 'a' at 1 for 1, then index 3 pairs with 0 for 3; total = 1 + 3 = 4
    assert calculateScore("aazz") == 4


def test_marked_indices_are_not_available_for_later_pairs():
    # "azza": indices 0 and 1 pair for 1; index 2 cannot use marked 0, and index 3 pairs with 2 for 1; total = 2
    assert calculateScore("azza") == 2


def test_middle_alphabet_mirror_letters_pair():
    # 'm' and 'n' are mirrors; indices 0 and 1 pair for 1
    assert calculateScore("mn") == 1
