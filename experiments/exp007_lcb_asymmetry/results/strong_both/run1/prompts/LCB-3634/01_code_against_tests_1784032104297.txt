## Implementation

```python
def calculateScore(s: str) -> int:
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # Create a mirror mapping from a-z to z-a
    marked = set()  # To keep track of marked indices
    score = 0

    for i in range(len(s)):
        if i in marked:
            continue  # Skip marked indices
        # Find the closest unmarked mirror character
        for j in range(i - 1, -1, -1):
            if j in marked:
                continue  # Skip marked indices
            if mirror[s[i]] == s[j]:
                marked.add(i)
                marked.add(j)
                score += i - j
                break  # We found our closest mirror, break the loop

    return score
```
