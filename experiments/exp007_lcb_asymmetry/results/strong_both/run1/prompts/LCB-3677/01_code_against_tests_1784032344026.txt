```python
from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[[-float('inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0] if coins[0][0] >= 0 else 0  # starting point

    for i in range(m):
        for j in range(n):
            for k in range(3):  # number of neutralizations used
                if i == 0 and j == 0:  # skip the starting cell
                    continue
                # collect coins from the top
                if i > 0:
                    current = dp[i-1][j][k]
                    if coins[i][j] < 0:
                        if k > 0:
                            dp[i][j][k] = max(dp[i][j][k], dp[i-1][j][k-1])  # neutralize from top
                        dp[i][j][k] = max(dp[i][j][k], current + coins[i][j])
                    else:
                        dp[i][j][k] = max(dp[i][j][k], current + coins[i][j])
                # collect coins from the left
                if j > 0:
                    current = dp[i][j-1][k]
                    if coins[i][j] < 0:
                        if k > 0:
                            dp[i][j][k] = max(dp[i][j][k], dp[i][j-1][k-1])  # neutralize from left
                        dp[i][j][k] = max(dp[i][j][k], current + coins[i][j])
                    else:
                        dp[i][j][k] = max(dp[i][j][k], current + coins[i][j])

    return max(dp[m-1][n-1])

```
