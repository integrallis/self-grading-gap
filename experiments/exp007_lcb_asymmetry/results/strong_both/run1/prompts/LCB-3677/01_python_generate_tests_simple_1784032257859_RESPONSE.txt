from solution import maximumAmount


def test_canonical_example_1():
    # Path 0 -> 1 -> neutralize -2 -> 3 -> 4 gives 0 + 1 + 0 + 3 + 4 = 8.
    assert maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) == 8


def test_canonical_example_2():
    # Any route visits four cells worth 10 coins each: 10 + 10 + 10 + 10 = 40.
    assert maximumAmount([[10, 10, 10], [10, 10, 10]]) == 40


def test_neutralizes_the_two_largest_robbers_when_the_path_is_fixed():
    # The only path is -1 -> -2 -> -3; neutralize -2 and -3, leaving -1.
    assert maximumAmount([[-1, -2, -3]]) == -1


def test_can_neutralize_a_robber_at_the_start_cell():
    # The single robber cell can be neutralized, so its contribution is 0.
    assert maximumAmount([[-1000]]) == 0


def test_can_neutralize_robbers_at_both_endpoints():
    # Neutralize start -5 and end -7; choose the path through 3, for 0 + 3 + 0 = 3.
    assert maximumAmount([[-5, 2], [3, -7]]) == 3


def test_chooses_route_with_best_result_after_neutralizations():
    # Right-right-down neutralizes -100: 0 + 0 + 5 + 5 = 10,
    # which is better than down-right-right: 0 + 0 + 0 + 5 = 5.
    assert maximumAmount([[0, -100, 5], [-1, -1, 5]]) == 10
