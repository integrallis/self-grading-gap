from solution import findValidPair


def test_canonical_example_1():
    # '2' occurs 2 times and '3' occurs 3 times; the first valid adjacent pair is "23".
    assert findValidPair("2523533") == "23"


def test_canonical_example_2():
    # '2' occurs 2 times and '1' occurs 1 time; "21" is the first valid adjacent pair.
    assert findValidPair("221") == "21"


def test_canonical_example_3():
    # The only adjacent pair is "22", whose digits are equal, so no valid pair exists.
    assert findValidPair("22") == ""


def test_returns_leftmost_when_multiple_valid_pairs_exist():
    # In "122333", '1' occurs once, '2' occurs twice, and '3' occurs three times;
    # both "12" and later "23" are valid, so the first is "12".
    assert findValidPair("122333") == "12"


def test_rejects_adjacent_distinct_digits_with_incorrect_frequencies():
    # In "1233", counts are '1': 1, '2': 1, and '3': 2;
    # every adjacent distinct pair includes a digit whose count does not equal its value.
    assert findValidPair("1233") == ""


def test_finds_valid_pair_with_digit_nine_at_maximum_length():
    # The string has 100 characters: '1' occurs once, '9' occurs nine times,
    # and '2' occurs 90 times, so the leftmost adjacent pair "19" is valid.
    assert findValidPair("19" + "9" * 8 + "2" * 90) == "19"
