def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    for char in s:
        if char == 'N':
            y += 1
        elif char == 'S':
            y -= 1
        elif char == 'E':
            x += 1
        elif char == 'W':
            x -= 1
        current_distance = abs(x) + abs(y)
        max_distance = max(max_distance, current_distance)

    # Calculate the maximum distance achievable with k changes
    total_moves = len(s)
    horizontal_moves = abs(x)
    vertical_moves = abs(y)
    total_distance = horizontal_moves + vertical_moves + k

    return total_distance

```
