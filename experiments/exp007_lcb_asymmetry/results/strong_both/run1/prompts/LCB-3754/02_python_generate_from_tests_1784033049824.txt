def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    for char in s:
        if char == 'N':
            y += 1
        elif char == 'S':
            y -= 1
        elif char == 'E':
            x += 1
        elif char == 'W':
            x -= 1
        current_distance = abs(x) + abs(y)
        max_distance = max(max_distance, current_distance)

    # Maximum possible distance by changing k moves
    total_moves = len(s)
    possible_distance = max_distance + k

    return possible_distance
```
