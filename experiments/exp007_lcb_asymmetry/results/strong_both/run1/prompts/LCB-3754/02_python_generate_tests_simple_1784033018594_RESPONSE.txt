from solution import maxDistance


def test_canonical_example_1():
    # Change "NWSE" to "NWNE"; after three moves position is (-1, 2), distance 3.
    assert maxDistance("NWSE", 1) == 3


def test_canonical_example_2():
    # Change "NSWWEW" to "NNWWWW"; after all six moves distance is 6.
    assert maxDistance("NSWWEW", 3) == 6


def test_no_changes_uses_greatest_distance_reached_before_the_end():
    # "NNSS" reaches distances 1, 2, 1, 0, so its maximum is 2.
    assert maxDistance("NNSS", 0) == 2


def test_one_change_can_turn_an_opposite_pair_into_two_moves_away():
    # Change "NS" to "NN"; after two moves the distance is 2.
    assert maxDistance("NS", 1) == 2


def test_changes_apply_to_horizontal_directions_too():
    # Change "EW" to "EE"; after two moves the distance is 2.
    assert maxDistance("EW", 1) == 2


def test_full_budget_can_make_every_move_point_away_from_origin():
    # Change "NSWE" to "NNNN"; after four moves the distance is 4.
    assert maxDistance("NSWE", 4) == 4


def test_at_most_k_changes_allows_keeping_already_optimal_moves():
    # "NNN" already reaches positions with distances 1, 2, 3, so the maximum is 3.
    assert maxDistance("NNN", 1) == 3


def test_minimum_length_string():
    # The single south move reaches (0, -1), whose Manhattan distance is 1.
    assert maxDistance("S", 0) == 1


def test_maximum_length_string():
    # 100000 north moves reach (0, 100000), whose Manhattan distance is 100000.
    assert maxDistance("N" * 100000, 0) == 100000
