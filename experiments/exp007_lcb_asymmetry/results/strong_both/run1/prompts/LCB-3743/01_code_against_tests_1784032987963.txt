```python
from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    gaps = []
    prev_end = 0

    # Calculate the free time gaps between meetings
    for i in range(n):
        if startTime[i] > prev_end:
            gaps.append(startTime[i] - prev_end)
        prev_end = endTime[i]

    # Also consider the gap after the last meeting
    if prev_end < eventTime:
        gaps.append(eventTime - prev_end)

    # If there are no gaps or k is 0, return the maximum gap
    if not gaps or k == 0:
        return max(gaps) if gaps else 0

    # Calculate the maximum free time by merging gaps
    gaps.sort(reverse=True)
    max_free_time = 0

    # We can combine k gaps at most, so we take the largest k gaps
    for i in range(min(k, len(gaps) - 1)):
        max_free_time += gaps[i]

    # The maximum free time is the sum of the largest k gaps
    if k < len(gaps):
        max_free_time += gaps[k]

    return max_free_time
```
