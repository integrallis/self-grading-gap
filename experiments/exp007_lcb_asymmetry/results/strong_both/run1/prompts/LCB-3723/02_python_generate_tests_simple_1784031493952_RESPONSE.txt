from solution import sumOfGoodNumbers


def test_canonical_example_1():
    # nums[1] = 3, nums[4] = 5, and nums[5] = 4 are good: 3 + 5 + 4 = 12.
    assert sumOfGoodNumbers([1, 3, 2, 1, 5, 4], 2) == 12


def test_canonical_example_2():
    # Only nums[0] = 2 is good: 2.
    assert sumOfGoodNumbers([2, 1], 1) == 2


def test_element_without_comparison_indices_is_good():
    # With k = 1, only nums[2] = 3 is good: 3.
    assert sumOfGoodNumbers([1, 2, 3], 1) == 3


def test_equal_comparison_values_are_not_good_at_upper_boundaries():
    # Every value has an existing comparison value equal to 1000, so none is strictly greater: 0.
    assert sumOfGoodNumbers([1000] * 100, 50) == 0
