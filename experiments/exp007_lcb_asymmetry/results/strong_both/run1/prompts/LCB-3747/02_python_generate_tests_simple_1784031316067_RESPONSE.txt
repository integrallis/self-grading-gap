from solution import maxAdjacentDistance


def test_canonical_example_1():
    # |1 - 2| = 1, |2 - 4| = 2, and circular |4 - 1| = 3.
    assert maxAdjacentDistance([1, 2, 4]) == 3


def test_canonical_example_2():
    # |-5 - (-10)| = 5, |-10 - (-5)| = 5, and circular |-5 - (-5)| = 0.
    assert maxAdjacentDistance([-5, -10, -5]) == 5


def test_two_elements_at_numeric_limits():
    # The two adjacent comparisons are both |-100 - 100| = 200.
    assert maxAdjacentDistance([-100, 100]) == 200


def test_maximum_length_array():
    # All 0-to-0 differences are 0; the two pairs involving 100 have difference 100.
    assert maxAdjacentDistance([0] * 99 + [100]) == 100
