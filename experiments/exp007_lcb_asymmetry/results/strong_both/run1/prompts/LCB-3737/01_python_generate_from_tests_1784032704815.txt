from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # Dynamic programming table to store the minimum cost
    dp = [[0] * 3 for _ in range(n)]
    dp[0] = cost[0]

    for i in range(1, n):
        for j in range(3):
            # Calculate the minimum cost for painting house i with color j
            dp[i][j] = cost[i][j] + min(dp[i - 1][(j + 1) % 3], dp[i - 1][(j + 2) % 3])

    # Now, handle the mirrored conditions
    min_cost = float('inf')

    for first_color in range(3):
        for last_color in range(3):
            if first_color != last_color:
                # Calculate total cost if first house is painted first_color and last house last_color
                total_cost = dp[n - 1][first_color] + cost[n - 1][last_color]
                min_cost = min(min_cost, total_cost)

    return min_cost

```
