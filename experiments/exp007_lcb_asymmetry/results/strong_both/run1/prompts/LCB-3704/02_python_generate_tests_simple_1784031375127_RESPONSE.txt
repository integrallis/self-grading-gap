from solution import countPartitions


def test_canonical_example_1():
    # Prefix/suffix differences are -16, 4, 10, and 24; all 4 are even.
    assert countPartitions([10, 10, 3, 7, 6]) == 4


def test_canonical_example_2():
    # Differences are 1 - 4 = -3 and 3 - 2 = 1; neither is even.
    assert countPartitions([1, 2, 2]) == 0


def test_canonical_example_3():
    # Differences are 2 - 18 = -16, 6 - 14 = -8, and 12 - 8 = 4; all 3 are even.
    assert countPartitions([2, 4, 6, 8]) == 3


def test_minimum_length_two_equal_odd_values():
    # The only split is [1] and [1], with difference 1 - 1 = 0, which is even.
    assert countPartitions([1, 1]) == 1


def test_maximum_length_with_maximum_element_values():
    # Every split has an even sum on both sides, so all 100 - 1 = 99 partitions qualify.
    assert countPartitions([100] * 100) == 99
