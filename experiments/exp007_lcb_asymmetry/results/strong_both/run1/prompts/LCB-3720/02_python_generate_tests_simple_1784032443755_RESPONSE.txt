from solution import minMaxWeight


def test_canonical_example_1():
    # Keeping 1->0, 3->0, 4->3, and 2->1 lets every node reach 0; max weight is 1.
    assert minMaxWeight(
        5,
        [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]],
        2,
    ) == 1


def test_canonical_example_2():
    # Node 2 has no directed path to node 0, so satisfying reachability is impossible.
    assert minMaxWeight(
        5,
        [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]],
        1,
    ) == -1


def test_canonical_example_3():
    # With one outgoing edge per node, 1->2->3->4->0 reaches 0 from all nodes; max weight is 2.
    assert minMaxWeight(
        5,
        [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]],
        1,
    ) == 2


def test_canonical_example_4():
    # Node 3 has no path to node 0, so satisfying reachability is impossible.
    assert minMaxWeight(
        5,
        [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]],
        1,
    ) == -1


def test_parallel_edges_with_distinct_weights_uses_lighter_edge():
    # Retaining the parallel edge 1->0 with weight 3 reaches node 0; 3 is the smaller weight.
    assert minMaxWeight(2, [[1, 0, 7], [1, 0, 3]], 1) == 3


def test_minimum_nodes_single_edge_maximum_weight_and_maximum_threshold():
    # The only edge 1->0 must remain, so the resulting maximum weight is 1_000_000.
    assert minMaxWeight(2, [[1, 0, 1_000_000]], 1) == 1_000_000


def test_maximum_nodes_and_maximum_edge_list_size():
    n = 100_000
    edges = [[i, i - 1, 1] for i in range(1, n)] + [[0, n - 1, 1]]

    # Every node follows the weight-1 chain to node 0, so the minimum possible maximum is 1.
    assert minMaxWeight(n, edges, 1) == 1
