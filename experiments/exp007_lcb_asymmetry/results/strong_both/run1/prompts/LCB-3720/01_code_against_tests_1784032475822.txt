from typing import List
import heapq


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Create adjacency list for the graph
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[v].append((u, w))  # Reverse the edges for backward reachability

    def canReach(target_weight: int) -> bool:
        # Check if we can reach node 0 from all nodes with the given maximum weight
        reachable = [0] * n
        queue = []

        # Start from node 0
        reachable[0] = 1
        for v in range(n):
            for u, w in graph[v]:
                if w <= target_weight:
                    queue.append((u, v))

        while queue:
            u, v = queue.pop()
            if reachable[u] < threshold:
                reachable[u] += 1
                for next_u, next_w in graph[u]:
                    if next_w <= target_weight:
                        queue.append((next_u, u))

        return all(reachable[i] > 0 for i in range(1, n))

    # Binary search on the edge weights
    left, right = 1, 10**6
    while left < right:
        mid = (left + right) // 2
        if canReach(mid):
            right = mid
        else:
            left = mid + 1

    return left if canReach(left) else -1
