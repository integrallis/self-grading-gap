from solution import hasSpecialSubstring


def test_canonical_example_1():
    # "aaa" at the end has length 3, is uniform, and is preceded by different "b".
    assert hasSpecialSubstring("aaabaaa", 3) is True


def test_canonical_example_2():
    # No length-2 substring of "abc" consists of one repeated character.
    assert hasSpecialSubstring("abc", 2) is False


def test_accepts_exact_length_uniform_run_between_different_characters():
    # "bbb" has length 3 and is bounded by "a" and "c", both different from "b".
    assert hasSpecialSubstring("abbbc", 3) is True


def test_rejects_uniform_run_longer_than_k():
    # Every length-2 "aa" substring belongs to the length-3 run "aaa", so one boundary is "a".
    assert hasSpecialSubstring("aaa", 2) is False


def test_accepts_entire_string_when_it_is_exact_uniform_length():
    # The whole string "zzzz" is uniform with exactly length 4 and has no neighboring characters.
    assert hasSpecialSubstring("zzzz", 4) is True


def test_accepts_isolated_single_character_for_k_one():
    # The substring "b" has length 1 and is bounded by different characters "a" and "c".
    assert hasSpecialSubstring("abc", 1) is True


def test_rejects_k_one_inside_a_longer_run():
    # Each "a" is adjacent to another "a", so no length-1 "a" has different neighbors where present.
    assert hasSpecialSubstring("aa", 1) is False
