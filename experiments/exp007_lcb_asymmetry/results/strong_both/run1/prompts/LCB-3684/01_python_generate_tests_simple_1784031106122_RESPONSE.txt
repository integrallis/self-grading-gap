from solution import hasMatch


def test_canonical_example_1():
    # "ee" followed later by "e" occurs in "leetcode": "eetcode" matches "ee*e".
    assert hasMatch("leetcode", "ee*e") is True


def test_canonical_example_2():
    # No substring of "car" starts with "c" and later ends with "v".
    assert hasMatch("car", "c*v") is False


def test_canonical_example_3():
    # Replacing "*" with "" makes the non-empty substring "u" match "u*".
    assert hasMatch("luck", "u*") is True


def test_wildcard_can_match_zero_characters_between_fixed_parts():
    # Replacing "*" with "" makes the substring "ab" match "a*b".
    assert hasMatch("zzabyy", "a*b") is True


def test_wildcard_can_match_characters_between_fixed_parts():
    # Replacing "*" with "x" makes the substring "abxcd" match "ab*cd".
    assert hasMatch("zzabxcdyy", "ab*cd") is True


def test_wildcard_at_beginning_allows_a_suffix_substring():
    # Replacing "*" with "" makes the substring "cd" match "*cd".
    assert hasMatch("abcd", "*cd") is True


def test_fixed_parts_must_appear_in_pattern_order():
    # "a" appears after "b" in "ba", so no substring can match "a*b".
    assert hasMatch("ba", "a*b") is False


def test_pattern_of_only_wildcard_matches_a_non_empty_substring():
    # Replacing "*" with "a" makes the non-empty substring "a" match "*".
    assert hasMatch("a", "*") is True
