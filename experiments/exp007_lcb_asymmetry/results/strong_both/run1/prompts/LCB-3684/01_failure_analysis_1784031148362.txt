def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    # Fixed prefix before the '*'
    prefix = p[:star_index]
    # Fixed suffix after the '*'
    suffix = p[star_index + 1:]
    prefix_len = len(prefix)
    suffix_len = len(suffix)
    # Iterate through s to find a match
    for i in range(len(s) - prefix_len - suffix_len + 1):
        if s[i:i + prefix_len] == prefix:
            if s[i + prefix_len:i + prefix_len + suffix_len] == suffix:
                return True
    return False
