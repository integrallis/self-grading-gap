from solution import largestInteger


def test_canonical_example_1():
    # 3 and 7 each occur in one size-3 subarray; the larger value is 7.
    assert largestInteger([3, 9, 2, 1, 7], 3) == 7


def test_canonical_example_2():
    # Only 3 occurs in exactly one size-4 subarray.
    assert largestInteger([3, 9, 7, 2, 1, 7], 4) == 3


def test_canonical_example_3():
    # 0 occurs in both size-1 subarrays, so no value occurs in exactly one.
    assert largestInteger([0, 0], 1) == -1


def test_k_one_returns_largest_value_appearing_once():
    # Size-1 subarrays contain individual elements; only 4 and 3 occur once, so return 4.
    assert largestInteger([2, 4, 2, 5, 5, 3], 1) == 4


def test_full_array_is_the_only_subarray():
    # There is one size-4 subarray, so every present value appears in exactly one; max is 4.
    assert largestInteger([4, 1, 4, 3], 4) == 4


def test_repeated_occurrences_in_one_subarray_count_once_for_that_subarray():
    # The only size-2 subarray is [5, 5], so 5 appears in exactly one subarray.
    assert largestInteger([5, 5], 2) == 5


def test_returns_negative_one_when_every_value_is_in_multiple_windows():
    # Both 1 and 2 appear in each of the two size-2 subarrays.
    assert largestInteger([1, 2, 1], 2) == -1


def test_single_element_array():
    # [0] is the only size-1 subarray, so 0 appears in exactly one subarray.
    assert largestInteger([0], 1) == 0


def test_maximum_element_value():
    # [50] is the only size-1 subarray, so 50 appears in exactly one subarray.
    assert largestInteger([50], 1) == 50


def test_maximum_length_array_with_full_length_window():
    # The 50-element array is the sole size-50 subarray, so 50 appears in exactly one subarray.
    assert largestInteger([50] * 50, 50) == 50
