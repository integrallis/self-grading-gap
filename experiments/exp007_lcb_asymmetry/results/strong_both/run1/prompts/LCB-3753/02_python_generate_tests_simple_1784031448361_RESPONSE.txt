from solution import maxDifference


def test_canonical_example_1():
    # 'a' occurs 5 times (odd) and 'b' occurs 2 times (even): 5 - 2 = 3.
    assert maxDifference("aaaaabbc") == 3


def test_canonical_example_2():
    # 'a' occurs 3 times (odd) and 'c' occurs 2 times (even): 3 - 2 = 1.
    assert maxDifference("abcabcab") == 1


def test_chooses_smallest_of_multiple_even_frequencies():
    # 'a' occurs 5 times (odd); 'b' occurs 4 times and 'c' occurs 2 times (even):
    # the maximum is 5 - 2 = 3.
    assert maxDifference("aaaaabbbbcc") == 3


def test_minimum_length_with_negative_difference():
    # 'a' occurs 1 time (odd) and 'b' occurs 2 times (even): 1 - 2 = -1.
    assert maxDifference("abb") == -1


def test_maximum_length():
    # 'a' occurs 97 times and 'c' occurs 1 time (odd); 'b' occurs 2 times (even):
    # the maximum is 97 - 2 = 95.
    assert maxDifference("a" * 97 + "bb" + "c") == 95
