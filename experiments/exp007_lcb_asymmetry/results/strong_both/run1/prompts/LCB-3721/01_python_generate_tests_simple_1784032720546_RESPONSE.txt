from solution import countMentions


def test_canonical_example_1():
    # At 10 both users are directly mentioned; at 71 user 0 has returned, so HERE adds one to both.
    assert countMentions(
        2,
        [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "71", "HERE"]],
    ) == [2, 2]


def test_canonical_example_2():
    # At 10 both users are directly mentioned; ALL at 12 mentions both, including offline user 0.
    assert countMentions(
        2,
        [["MESSAGE", "10", "id1 id0"], ["OFFLINE", "11", "0"], ["MESSAGE", "12", "ALL"]],
    ) == [2, 2]


def test_canonical_example_3():
    # User 0 is offline from 10 through before 70, so HERE at 12 mentions only user 1.
    assert countMentions(
        2,
        [["OFFLINE", "10", "0"], ["MESSAGE", "12", "HERE"]],
    ) == [0, 1]


def test_direct_id_mentions_count_duplicates_and_can_mention_offline_users():
    # User 1 is offline, but direct id1 appears twice; id0 appears once, giving [1, 2, 0].
    assert countMentions(
        3,
        [["OFFLINE", "5", "1"], ["MESSAGE", "6", "id1 id0 id1"]],
    ) == [1, 2, 0]


def test_here_mentions_only_users_online_at_that_timestamp():
    # User 0 is offline at 20, while user 2 is offline at 30; HERE at 35 mentions only user 1.
    assert countMentions(
        3,
        [["OFFLINE", "10", "0"], ["OFFLINE", "20", "2"], ["MESSAGE", "35", "HERE"]],
    ) == [0, 1, 0]


def test_here_includes_user_when_their_60_unit_offline_period_ends():
    # User 0 returns at 70, so HERE at exactly 70 mentions both users once.
    assert countMentions(
        2,
        [["OFFLINE", "10", "0"], ["MESSAGE", "70", "HERE"]],
    ) == [1, 1]


def test_offline_status_is_processed_before_message_at_same_timestamp():
    # The OFFLINE event at 10 takes effect before HERE at 10, so only user 1 is mentioned.
    assert countMentions(
        2,
        [["MESSAGE", "10", "HERE"], ["OFFLINE", "10", "0"]],
    ) == [0, 1]


def test_events_are_processed_by_timestamp_for_online_status():
    # Although listed first, HERE occurs at 20 after user 0 went offline at 10, so it mentions only user 1.
    assert countMentions(
        2,
        [["MESSAGE", "20", "HERE"], ["OFFLINE", "10", "0"]],
    ) == [0, 1]


def test_all_mentions_every_user_regardless_of_online_status():
    # Users 0 and 2 are offline at 30, but ALL still mentions each of the three users once.
    assert countMentions(
        3,
        [["OFFLINE", "5", "0"], ["OFFLINE", "10", "2"], ["MESSAGE", "30", "ALL"]],
    ) == [1, 1, 1]


def test_only_offline_events_produce_no_mentions():
    # No MESSAGE event occurs, so every user's mention total remains zero.
    assert countMentions(2, [["OFFLINE", "10", "1"]]) == [0, 0]
