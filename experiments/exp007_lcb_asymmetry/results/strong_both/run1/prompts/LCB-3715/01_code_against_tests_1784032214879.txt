from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a map to hold the number of coins at each bag position
    coin_map = {}
    for l, r, c in coins:
        for i in range(l, r + 1):
            coin_map[i] = coin_map.get(i, 0) + c

    # Create a sorted list of unique bag positions
    positions = sorted(coin_map.keys())
    n = len(positions)
    max_coins = 0
    current_coins = 0
    left = 0

    # Use a sliding window to calculate the maximum coins in k consecutive bags
    for right in range(n):
        current_coins += coin_map[positions[right]]  # Add the coins from the right position

        # Adjust the left pointer to maintain a window of size at most k
        while (positions[right] - positions[left] + 1) > k:
            current_coins -= coin_map[positions[left]]  # Remove the coins from the left position
            left += 1

        # Update max_coins if the current window size is at least k
        if (positions[right] - positions[left] + 1) >= k:
            max_coins = max(max_coins, current_coins)

    return max_coins
