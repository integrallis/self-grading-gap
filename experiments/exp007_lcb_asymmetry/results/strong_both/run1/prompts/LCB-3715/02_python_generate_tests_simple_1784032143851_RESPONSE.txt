from solution import maximumCoins


def test_canonical_example_1():
    # Bags [3, 4, 5, 6] contain 2 + 0 + 4 + 4 = 10 coins.
    assert maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) == 10


def test_canonical_example_2():
    # Any two bags inside [1, 10] contain 3 + 3 = 6 coins.
    assert maximumCoins([[1, 10, 3]], 2) == 6


def test_window_can_include_empty_bags_between_segments():
    # Bags [9, 10, 11] contain 0 + 6 + 6 = 12 coins.
    assert maximumCoins([[1, 2, 5], [10, 11, 6]], 3) == 12


def test_window_can_take_part_of_adjacent_segments():
    # Bags [1, 2, 3, 4, 5, 6] contain 5 + 5 + 5 + 5 + 2 + 2 = 24 coins.
    assert maximumCoins([[1, 4, 5], [5, 10, 2]], 6) == 24


def test_k_can_exceed_the_length_of_a_coin_segment():
    # A five-bag window containing bags [5, 6] collects 7 + 7 = 14 coins.
    assert maximumCoins([[5, 6, 7]], 5) == 14


def test_large_coordinate_and_total_coin_count():
    # All 1,000,000,000 bags contain 1,000 coins:
    # 1,000,000,000 * 1,000 = 1,000,000,000,000.
    assert maximumCoins([[1, 1_000_000_000, 1000]], 1_000_000_000) == 1_000_000_000_000


def test_minimum_k_selects_the_highest_single_bag_value():
    # With one bag selected, bag 5 contributes 9 coins, exceeding bags 1 through 3 with 2 each.
    assert maximumCoins([[1, 3, 2], [5, 5, 9]], 1) == 9


def test_single_bag_segment():
    # The only selectable coin-bearing bag is bag 4, which contains 7 coins.
    assert maximumCoins([[4, 4, 7]], 1) == 7


def test_maximum_number_of_non_overlapping_segments():
    coins = [[2 * index + 1, 2 * index + 1, index % 1000 + 1] for index in range(100_000)]

    # A one-bag window can select a segment with value 1,000, and no segment contains more.
    assert maximumCoins(coins, 1) == 1000
