from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    # Calculate contribution of each number as max and min
    for i in range(n):
        # Each number can be the min in (choose k-1 from the other n-1)
        min_contrib = pow(2, n - 1, MOD)  # 2^(n-1)
        # Each number can be the max in (choose k-1 from the other n-1)
        max_contrib = pow(2, n - 1, MOD)  # 2^(n-1)
        # We consider only combinations that are valid for k
        if k < n:
            min_contrib = (min_contrib * (k - 1)) % MOD
            max_contrib = (max_contrib * (k - 1)) % MOD
        total_sum += (nums[i] * (min_contrib + max_contrib)) % MOD
        total_sum %= MOD
    return total_sum

```
