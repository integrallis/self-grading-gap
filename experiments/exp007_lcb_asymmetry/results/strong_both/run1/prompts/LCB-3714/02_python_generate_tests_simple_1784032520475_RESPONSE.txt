from solution import minMaxSums


def test_canonical_example_1():
    # Singletons contribute 2+4+6=12; pairs contribute 3+4+5=12; total 24.
    assert minMaxSums([1, 2, 3], 2) == 24


def test_canonical_example_2():
    # With k=1, each element contributes itself as both min and max: 2*(5+0+6)=22.
    assert minMaxSums([5, 0, 6], 1) == 22


def test_canonical_example_3():
    # Three singleton subsequences and three length-two subsequences each contribute 1+1=2: 6*2=12.
    assert minMaxSums([1, 1, 1], 2) == 12


def test_includes_subsequences_up_to_the_full_array_length():
    # For [2,5,9]: singletons 4+10+18=32, pairs 7+11+14=32, triple 2+9=11; total 75.
    assert minMaxSums([2, 5, 9], 3) == 75


def test_returns_result_modulo_1_000_000_007():
    # With k=1, two 1_000_000_000 values contribute 4_000_000_000;
    # 4_000_000_000 mod 1_000_000_007 = 999_999_979.
    assert minMaxSums([1_000_000_000, 1_000_000_000], 1) == 999_999_979


def test_minimum_array_length():
    # The only subsequence is [7], whose minimum plus maximum is 7+7=14.
    assert minMaxSums([7], 1) == 14


def test_maximum_k_boundary_with_seventy_elements():
    # All 2^70-1 nonempty subsequences have min+max = 1+1 = 2.
    # 2*(2^70-1) mod 1_000_000_007 = 540_032_504.
    assert minMaxSums([1] * 70, 70) == 540_032_504


def test_maximum_array_length():
    # With k=1, each of 100_000 singleton subsequences contributes 1+1=2:
    # 100_000*2 = 200_000.
    assert minMaxSums([1] * 100_000, 1) == 200_000
