```python
from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    # Calculate contribution of each number as min and max
    for num in nums:
        # Each number can be the min in (choose 0 to k-1 from the other n-1)
        min_contrib = (pow(2, n - 1, MOD) * (min(k, n) - 1)) % MOD
        # Each number can be the max in (choose 0 to k-1 from the other n-1)
        max_contrib = (pow(2, n - 1, MOD) * (min(k, n) - 1)) % MOD
        total_sum += (num * (min_contrib + max_contrib)) % MOD
        total_sum %= MOD
    total_sum = (total_sum + (k * sum(nums)) % MOD) % MOD
    return total_sum

```
