## Implementation

```python
from typing import List

def zigzagTraversal(grid: List[List[int]]) -> List[int]:
    result = []
    for i in range(len(grid)):
        if i % 2 == 0:
            # even index row: left to right
            result.extend(grid[i][j] for j in range(0, len(grid[i]), 2))
        else:
            # odd index row: right to left
            result.extend(grid[i][j] for j in range(len(grid[i]) - 1, -1, -1) if j % 2 == 1)
    return result
```
