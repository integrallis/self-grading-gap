from solution import zigzagTraversal


def test_canonical_example_1():
    # Zigzag order is [1, 2, 4, 3]; taking alternate cells gives [1, 4].
    assert zigzagTraversal([[1, 2], [3, 4]]) == [1, 4]


def test_canonical_example_2():
    # Zigzag order is [2, 1, 1, 2, 2, 1]; taking alternate cells gives [2, 1, 2].
    assert zigzagTraversal([[2, 1], [2, 1], [2, 1]]) == [2, 1, 2]


def test_canonical_example_3():
    # Zigzag order is [1, 2, 3, 6, 5, 4, 7, 8, 9]; taking alternate cells gives [1, 3, 5, 7, 9].
    assert zigzagTraversal([[1, 2, 3], [4, 5, 6], [7, 8, 9]]) == [1, 3, 5, 7, 9]


def test_maximum_row_count():
    grid = [[row * 2 + 1, row * 2 + 2] for row in range(50)]

    # Each two-cell row begins at an included traversal position:
    # even rows contribute their left value, and odd rows contribute their right value.
    expected = [
        row * 2 + 1 if row % 2 == 0 else row * 2 + 2
        for row in range(50)
    ]

    assert zigzagTraversal(grid) == expected


def test_maximum_column_count():
    grid = [
        list(range(1, 51)),
        list(range(51, 101)),
    ]

    # Zigzag order is 1 through 50, then 100 down through 51.
    # Taking alternate cells gives odd values through 49, then even values from 100 down through 52.
    expected = list(range(1, 51, 2)) + list(range(100, 50, -2))

    assert zigzagTraversal(grid) == expected


def test_maximum_cell_value_is_retained_when_visited():
    # Zigzag order is [1, 2, 2500, 3]; taking alternate cells gives [1, 2500].
    assert zigzagTraversal([[1, 2], [3, 2500]]) == [1, 2500]
