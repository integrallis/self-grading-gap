from solution import maxFrequency


def test_canonical_example_1():
    # Add -5 to [3, 4, 5, 6], producing one additional 1; total frequency is 2.
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2


def test_canonical_example_2():
    # Add 8 to [2, 3, 4, 5, 5, 4, 3, 2, 2], producing three additional 10s;
    # together with the original 10, the frequency is 4.
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4


def test_single_element_can_be_changed_to_k():
    # Add -2 to the only element 7, making it 5, so the frequency is 1.
    assert maxFrequency([7], 5) == 1


def test_zero_addition_preserves_all_existing_k_values():
    # Add 0 to any nonempty subarray; all three existing 4s remain 4.
    assert maxFrequency([4, 4, 4], 4) == 3


def test_k_values_between_matching_values_limit_the_best_subarray():
    # Starting with one 1, either adjacent 2 can become 1 without changing that 1,
    # for a total of 2; converting both 2s requires including and losing the middle 1.
    assert maxFrequency([2, 1, 2], 1) == 2


def test_single_addition_can_only_convert_one_original_value():
    # Adding one x can turn either both 2s or both 3s into 1, but not both values,
    # so the maximum frequency is 2.
    assert maxFrequency([2, 3, 2, 3], 1) == 2


def test_subarray_can_include_k_when_the_gain_is_larger_than_the_loss():
    # Add -1 to the whole array: four 2s become 1 while the original 1 changes,
    # yielding a frequency of 4.
    assert maxFrequency([2, 2, 1, 2, 2], 1) == 4


def test_maximum_length_and_upper_value_boundaries():
    # Add 0 to any subarray; every one of the 100000 values already equals 50.
    assert maxFrequency([50] * 100000, 50) == 100000
