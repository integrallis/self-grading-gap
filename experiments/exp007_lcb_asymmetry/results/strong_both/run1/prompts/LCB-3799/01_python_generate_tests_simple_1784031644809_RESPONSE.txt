from solution import totalNumbers


def test_canonical_example_1():
    # 4 choices for the hundreds digit, then valid distinct even arrangements total 12.
    assert totalNumbers([1, 2, 3, 4]) == 12


def test_canonical_example_2():
    # The valid numbers are 202 and 220; leading-zero arrangements are not allowed.
    assert totalNumbers([0, 2, 2]) == 2


def test_canonical_example_3():
    # All three available digits are 6, so the only number is 666.
    assert totalNumbers([6, 6, 6]) == 1


def test_canonical_example_4():
    # All available digits are odd, so no formed three-digit number can be even.
    assert totalNumbers([1, 3, 5]) == 0
