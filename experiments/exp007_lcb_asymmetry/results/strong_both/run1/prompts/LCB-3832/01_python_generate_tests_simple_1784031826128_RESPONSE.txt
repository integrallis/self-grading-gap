from solution import minCosts


def test_canonical_example_1():
    # Paying person 3 costs 1 and then every later position is reachable for free;
    # earlier positions use the cheapest cost seen through that position.
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]


def test_canonical_example_2():
    # Paying person 0 costs 1, after which every position is reachable for free.
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]
