from solution import minCosts


def test_canonical_example_1():
    # Running minimum: [5, min(5, 3), min(5, 3, 4), min(5, 3, 4, 1), ...].
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]


def test_canonical_example_2():
    # Person 0 costs 1, the minimum through every position.
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]


def test_singleton_maximum_cost():
    # With one person, reaching position 0 requires paying that person's cost: 100.
    assert minCosts([100]) == [100]


def test_maximum_length_all_maximum_costs():
    # Every prefix contains only cost 100, so each of the 100 answers is 100.
    assert minCosts([100] * 100) == [100] * 100
