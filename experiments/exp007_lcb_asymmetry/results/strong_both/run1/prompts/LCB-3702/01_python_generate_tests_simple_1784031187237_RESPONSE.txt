from solution import maxLength


def test_canonical_example_1():
    # [1, 2, 1, 1, 1] has prod = 2, lcm = 2, gcd = 1, so length 5 is valid.
    assert maxLength([1, 2, 1, 2, 1, 1, 1]) == 5


def test_canonical_example_2():
    # [3, 4, 5] has prod = 60, lcm = 60, gcd = 1, so length 3 is valid.
    assert maxLength([2, 3, 4, 5, 6]) == 3


def test_canonical_example_3():
    # [3, 1, 4, 5, 1] has prod = 60, lcm = 60, gcd = 1, so length 5 is valid.
    assert maxLength([1, 2, 3, 1, 4, 5, 1]) == 5


def test_all_ones_entire_array_is_product_equivalent():
    # prod([1, 1, 1, 1]) = 1 and lcm * gcd = 1 * 1 = 1, so length 4 is valid.
    assert maxLength([1, 1, 1, 1]) == 4


def test_pairwise_coprime_values_can_form_a_long_product_equivalent_subarray():
    # prod([2, 3, 5, 7]) = 210 and lcm * gcd = 210 * 1 = 210, so length 4 is valid.
    assert maxLength([2, 3, 5, 7]) == 4


def test_returns_pair_length_when_repeated_values_make_the_triple_invalid():
    # Any pair [2, 2] has prod = 4 and lcm * gcd = 2 * 2 = 4;
    # [2, 2, 2] has prod = 8 but lcm * gcd = 2 * 2 = 4, so the maximum is 2.
    assert maxLength([2, 2, 2]) == 2
