from solution import minCosts


def test_canonical_example_1():
    # Pay 5, 3, 3, 1; after paying 3 or 1, later positions are reachable for free.
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]


def test_canonical_example_2():
    # Pay 1 to swap with person 0, then every later position is reachable for free.
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]


def test_single_person_in_front():
    # The only reachable position requires swapping with person 0 for cost 7.
    assert minCosts([7]) == [7]


def test_later_cheaper_swap_reduces_future_minimum_cost():
    # Costs to positions are min(4), min(4, 2), min(4, 2, 3), min(4, 2, 3, 1).
    assert minCosts([4, 2, 3, 1]) == [4, 2, 2, 1]
