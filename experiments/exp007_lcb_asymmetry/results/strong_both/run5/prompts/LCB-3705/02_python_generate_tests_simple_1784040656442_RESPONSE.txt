from solution import largestInteger


def test_canonical_example_1():
    # 3 and 7 each appear in exactly one size-3 subarray; the largest is 7.
    assert largestInteger([3, 9, 2, 1, 7], 3) == 7


def test_canonical_example_2():
    # Only 3 appears in exactly one size-4 subarray.
    assert largestInteger([3, 9, 7, 2, 1, 7], 4) == 3


def test_canonical_example_3():
    # 0 appears in both size-1 subarrays, so no value appears in exactly one.
    assert largestInteger([0, 0], 1) == -1


def test_single_element_array():
    # The sole value appears in the only size-1 subarray.
    assert largestInteger([50], 1) == 50


def test_maximum_length_single_window_with_repeated_values():
    # There is one size-50 subarray; both 0 and 50 appear in it, and 50 is larger.
    assert largestInteger([50, 50] + [0] * 48, 50) == 50
