```python
from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    # Calculate the gaps between meetings
    gaps = []
    # Initial gap from the start of the event to the first meeting
    if startTime:
        gaps.append(startTime[0] - 0)
    # Gaps between consecutive meetings
    for i in range(1, len(startTime)):
        gaps.append(startTime[i] - endTime[i - 1])
    # Final gap from the last meeting to the end of the event
    if endTime:
        gaps.append(eventTime - endTime[-1])

    # We can merge k gaps into one larger gap
    # Sort the gaps in descending order to maximize the free time
    gaps.sort(reverse=True)
    # Only the first k gaps can be merged, hence we take k + 1 gaps
    max_free_time = sum(gaps[:k + 1])
    return max_free_time
```
