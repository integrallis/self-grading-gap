from solution import findValidPair


def test_canonical_example_1():
    # "2" appears 2 times and "3" appears 3 times; "23" is the first valid adjacent pair.
    assert findValidPair("2523533") == "23"


def test_canonical_example_2():
    # "2" appears 2 times and "1" appears 1 time; adjacent pair "21" is valid.
    assert findValidPair("221") == "21"


def test_canonical_example_3():
    # Although "2" appears 2 times, the only adjacent pair is "22", whose digits are equal.
    assert findValidPair("22") == ""


def test_returns_leftmost_when_multiple_valid_pairs_exist():
    # Counts are: "1" appears once, "2" appears twice, and "3" appears three times;
    # both "12" and later "23" are valid, so the first pair is "12".
    assert findValidPair("123323") == "12"


def test_returns_empty_when_adjacent_digits_do_not_both_have_matching_counts():
    # "1" appears once, but "3" appears twice rather than three times, so "13" is invalid.
    assert findValidPair("133") == ""
