from typing import List


def totalNumbers(digits: List[int]) -> int:
    from collections import Counter
    count = Counter(digits)
    total = 0
    for last_digit in count:
        if last_digit % 2 == 0:
            # Avoid leading zeros
            for first_digit in count:
                if first_digit == 0:
                    continue
                if count[first_digit] > (1 if last_digit == first_digit else 0):
                    # Choose remaining digit for the middle position
                    for middle_digit in count:
                        if middle_digit == first_digit or middle_digit == last_digit:
                            continue
                        total += 1
            # Handle case where last_digit is used as first_digit
            if count[last_digit] > 1:
                for middle_digit in count:
                    if middle_digit == last_digit:
                        continue
                    total += 1
    return total
```
