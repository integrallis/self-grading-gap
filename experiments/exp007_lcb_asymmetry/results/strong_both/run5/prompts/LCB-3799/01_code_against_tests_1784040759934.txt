```python
from typing import List


def totalNumbers(digits: List[int]) -> int:
    from collections import Counter
    count = Counter(digits)
    total = 0
    for last_digit in count:
        if last_digit % 2 == 0:
            for first_digit in count:
                if first_digit == 0:
                    continue
                for middle_digit in count:
                    if (Counter([first_digit, middle_digit, last_digit])[first_digit] <= count[first_digit] and 
                        Counter([first_digit, middle_digit, last_digit])[middle_digit] <= count[middle_digit] and 
                        Counter([first_digit, middle_digit, last_digit])[last_digit] <= count[last_digit]):
                        total += 1
    return total

```
