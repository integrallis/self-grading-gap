from solution import countPartitions


def test_canonical_example_1():
    # Prefix sums are 10, 20, 23, and 30; differences from the remaining sums are all even.
    assert countPartitions([10, 10, 3, 7, 6]) == 4


def test_canonical_example_2():
    # Total sum is 5; the two differences are 2*1-5=-3 and 2*3-5=1, both odd.
    assert countPartitions([1, 2, 2]) == 0


def test_canonical_example_3():
    # The three split differences are 4-20=-16, 12-20=-8, and 24-20=4, all even.
    assert countPartitions([2, 4, 6, 8]) == 3


def test_minimum_length_with_maximum_values():
    # The sole split is [100] and [100], with difference 100-100=0, which is even.
    assert countPartitions([100, 100]) == 1


def test_maximum_length_all_maximum_values():
    # There are 99 valid splits; both sides always have even sums, so every difference is even.
    assert countPartitions([100] * 100) == 99
