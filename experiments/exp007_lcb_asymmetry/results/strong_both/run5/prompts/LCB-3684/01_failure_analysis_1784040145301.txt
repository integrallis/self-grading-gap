def hasMatch(s: str, p: str) -> bool:
    # Find the position of '*'
    star_index = p.index('*')
    # Get the fixed parts before and after '*'
    prefix = p[:star_index]
    suffix = p[star_index + 1:]
    # Check for all possible starting points for the prefix in s
    for i in range(len(s) - len(prefix) - len(suffix) + 1):
        # Check if the current position matches the prefix
        if s[i:i + len(prefix)] == prefix:
            # Check for the suffix after the matched prefix
            if s[i + len(prefix):].startswith(suffix):
                return True
    return False
