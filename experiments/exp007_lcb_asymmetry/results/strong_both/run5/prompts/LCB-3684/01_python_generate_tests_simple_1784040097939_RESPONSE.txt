from solution import hasMatch


def test_canonical_example_1():
    # "ee*e" can replace "*" with "tcod" to match substring "eetcode".
    assert hasMatch("leetcode", "ee*e") is True


def test_canonical_example_2():
    # No replacement for "*" makes "c*v" match a substring of "car".
    assert hasMatch("car", "c*v") is False


def test_canonical_example_3():
    # Replacing "*" with "" makes the non-empty substring "u" match.
    assert hasMatch("luck", "u*") is True


def test_wildcard_can_match_zero_characters_between_fixed_parts():
    # Replacing "*" with "" makes "ab*cd" become substring "abcd".
    assert hasMatch("xxabcdyy", "ab*cd") is True


def test_wildcard_can_match_characters_between_fixed_parts():
    # Replacing "*" with "123" is not valid input characters; use "xyz": "abxyzcd" matches.
    assert hasMatch("zzabxyzcdzz", "ab*cd") is True


def test_pattern_with_wildcard_at_start_matches_suffix():
    # Replacing "*" with "ab" makes "*cd" match substring "abcd".
    assert hasMatch("zabcdz", "*cd") is True


def test_fixed_parts_must_appear_in_order():
    # "cd" occurs before "ab", so no replacement makes "ab*cd" a substring.
    assert hasMatch("cdxxab", "ab*cd") is False


def test_wildcard_only_pattern_can_match_a_nonempty_substring():
    # Replacing "*" with "a" makes the pattern match non-empty substring "a".
    assert hasMatch("a", "*") is True
