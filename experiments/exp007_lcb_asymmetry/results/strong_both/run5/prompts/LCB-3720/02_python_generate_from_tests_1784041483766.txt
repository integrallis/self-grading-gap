from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    from collections import defaultdict, deque
    import sys

    # Sort edges by weight
    edges.sort(key=lambda x: x[2])
    left, right = 1, 10**6  # The range of weights
    result = -1

    while left <= right:
        mid = (left + right) // 2

        # Build graph with edges <= mid
        graph = defaultdict(list)
        indegree = [0] * n

        for u, v, w in edges:
            if w <= mid:
                graph[u].append(v)
                indegree[v] += 1

        # Check if we can reach node 0 from all nodes while respecting the threshold
        reachable = [False] * n
        queue = deque([0])
        reachable[0] = True
        count = 1  # Count reachable nodes

        while queue:
            node = queue.popleft()
            outgoing_count = 0

            for neighbor in graph[node]:
                if reachable[neighbor]:
                    continue
                outgoing_count += 1
                if outgoing_count > threshold:
                    break
                reachable[neighbor] = True
                count += 1
                queue.append(neighbor)

            if outgoing_count > threshold:
                break

        if count == n:
            result = mid  # All nodes can reach node 0
            right = mid - 1  # Try to minimize maximum weight
        else:
            left = mid + 1  # Increase the weight limit

    return result
```
