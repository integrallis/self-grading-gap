from solution import minMaxWeight


def test_canonical_example_1():
    # Nodes 1, 2, 3, and 4 can reach 0 using only weight-1 edges, so the minimum maximum is 1.
    assert minMaxWeight(
        5,
        [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]],
        2,
    ) == 1


def test_canonical_example_2():
    # Node 2 has no directed path to node 0, so satisfying reachability is impossible.
    assert minMaxWeight(
        5,
        [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]],
        1,
    ) == -1


def test_canonical_example_3():
    # Keeping 1->2 (1), 2->3 (2), 3->4 (2), and 4->0 (1) reaches 0 from every node;
    # weight 2 is required because node 2 must use 2->3.
    assert minMaxWeight(
        5,
        [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]],
        1,
    ) == 2


def test_canonical_example_4():
    # Node 3 has no path to node 0, so satisfying reachability is impossible.
    assert minMaxWeight(
        5,
        [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]],
        1,
    ) == -1
