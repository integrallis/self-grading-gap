from solution import minMaxWeight


def test_canonical_example_1():
    # Retaining 1->0, 2->1, 3->0, and 4->3 gives every node a path to 0 with maximum weight 1.
    assert minMaxWeight(
        5,
        [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]],
        2,
    ) == 1


def test_canonical_example_2():
    # Node 2 has no outgoing edge, so it cannot reach node 0 and the answer is -1.
    assert minMaxWeight(
        5,
        [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]],
        1,
    ) == -1


def test_canonical_example_3():
    # Retaining 1->2 (1), 2->3 (2), 3->4 (2), and 4->0 (1) works; node 2 requires weight 2.
    assert minMaxWeight(
        5,
        [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]],
        1,
    ) == 2


def test_canonical_example_4():
    # Node 3 has no outgoing edge, so it cannot reach node 0 and the answer is -1.
    assert minMaxWeight(
        5,
        [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]],
        1,
    ) == -1


def test_minimum_size_single_required_edge_needs_no_removal():
    # The only edge 1->0 must remain, so the maximum retained weight is 7.
    assert minMaxWeight(2, [[1, 0, 7]], 1) == 7


def test_parallel_edges_choose_lower_weight():
    # Keep 2->1 (1) and the lower parallel edge 1->0 (2), giving maximum weight 2.
    assert minMaxWeight(3, [[1, 0, 9], [1, 0, 2], [2, 1, 1]], 1) == 2


def test_maximum_allowed_edge_weight():
    # The only required edge has the maximum permitted weight, so the answer is 1_000_000.
    assert minMaxWeight(2, [[1, 0, 1_000_000]], 1) == 1_000_000


def test_large_chain_at_input_size_limit():
    # Each node i reaches 0 through i->i-1; all retained edges have weight 1.
    n = 100_000
    edges = [[node, node - 1, 1] for node in range(1, n)]
    assert minMaxWeight(n, edges, 1) == 1
