from solution import transformArray


def test_canonical_example_1():
    # 4 and 2 are even -> 0; 3 and 1 are odd -> 1; sorted: [0, 0, 1, 1].
    assert transformArray([4, 3, 2, 1]) == [0, 0, 1, 1]


def test_canonical_example_2():
    # 4 and 2 are even -> 0; 1, 5, and 1 are odd -> 1; sorted: [0, 0, 1, 1, 1].
    assert transformArray([1, 5, 1, 4, 2]) == [0, 0, 1, 1, 1]


def test_minimum_length_and_minimum_value():
    # 1 is odd -> 1; sorting a one-element array leaves it unchanged.
    assert transformArray([1]) == [1]


def test_maximum_length_and_maximum_value():
    # Each of the 100 values is even -> 0; the resulting 100 zeros are already sorted.
    assert transformArray([1000] * 100) == [0] * 100
