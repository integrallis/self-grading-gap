```python
from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n == 0:
        return 0
    # dp[i] will store the length of the longest subsequence ending at index i
    dp = [1] * n
    max_length = 1
    
    for i in range(n):
        for j in range(i):
            difference1 = abs(nums[i] - nums[j])
            valid = True
            # Check all previous elements to maintain the non-increasing condition
            for k in range(j):
                difference2 = abs(nums[j] - nums[k])
                if difference1 > difference2:
                    valid = False
                    break
            if valid:
                dp[i] = max(dp[i], dp[j] + 1)
                max_length = max(max_length, dp[i])
    return max_length
```
