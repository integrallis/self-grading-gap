from solution import longestSubsequence


def test_canonical_example_1():
    # [16, 6, 3] has adjacent absolute differences [10, 3], and 10 >= 3.
    assert longestSubsequence([16, 6, 3]) == 3


def test_canonical_example_2():
    # [6, 4, 2, 1] is a subsequence with differences [2, 2, 1], where 2 >= 2 >= 1.
    assert longestSubsequence([6, 5, 3, 4, 2, 1]) == 4


def test_canonical_example_3():
    # [10, 20, 10, 19, 10] has differences [10, 10, 9, 9], where 10 >= 10 >= 9 >= 9.
    assert longestSubsequence([10, 20, 10, 19, 10, 20]) == 5


def test_two_elements_always_form_a_valid_subsequence():
    # A two-element subsequence has only one difference, so [300, 1] has valid length 2.
    assert longestSubsequence([300, 1]) == 2


def test_equal_values_allow_zero_differences():
    # [7, 7, 7, 7] has differences [0, 0, 0], and 0 >= 0 >= 0.
    assert longestSubsequence([7, 7, 7, 7]) == 4


def test_can_skip_elements_when_full_array_differences_increase():
    # The full array has differences [1, 8, 1], which is not non-increasing;
    # [1, 10, 11] has differences [9, 1], so the longest valid length is 3.
    assert longestSubsequence([1, 2, 10, 11]) == 3
