from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n == 0:
        return 0

    # dp will map last_difference to the best length of subsequence ending at current index
    dp = [{} for _ in range(n)]
    max_length = 1

    for i in range(n):
        for j in range(i):
            difference = abs(nums[i] - nums[j])
            # Always record a length-2 subsequence for the pair (nums[j], nums[i])
            dp[i][difference] = max(dp[i].get(difference, 1), 2)
            # Extend the states from dp[j]
            for prev_diff, length in dp[j].items():
                if prev_diff >= difference:
                    dp[i][difference] = max(dp[i][difference], length + 1)
                    max_length = max(max_length, dp[i][difference])

    return max_length
```
