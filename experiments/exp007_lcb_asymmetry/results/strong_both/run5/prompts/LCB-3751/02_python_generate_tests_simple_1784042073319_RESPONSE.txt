from solution import maxFrequency


def test_canonical_example_1():
    # Add -5 to nums[2..5], producing [1, 2, -2, -1, 0, 1], so k=1 appears 2 times.
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2


def test_canonical_example_2():
    # Add 8 to nums[1..9], changing the three 2s to 10 while the initial 10 remains,
    # so k=10 appears 4 times.
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4


def test_singleton_upper_bound_no_op():
    # With one element already equal to k, choose x=0; k appears once.
    assert maxFrequency([50], 50) == 1


def test_full_array_at_maximum_length():
    # Add -1 to the entire array, converting all 100,000 copies of 2 into k=1.
    assert maxFrequency([2] * 100_000, 1) == 100_000


def test_interior_subarray_preserves_k_at_both_ends():
    # Add -1 to the interior [2, 2], producing [1, 1, 1, 1], so k appears 4 times.
    assert maxFrequency([1, 2, 2, 1], 1) == 4


def test_distinct_source_values_cannot_both_convert_with_one_addition():
    # A single x can change either 2 or 3 into 1, but not both, so the maximum is 1.
    assert maxFrequency([2, 3], 1) == 1
