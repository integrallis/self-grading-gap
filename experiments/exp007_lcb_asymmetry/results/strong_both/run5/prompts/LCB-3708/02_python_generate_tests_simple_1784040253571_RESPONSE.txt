from solution import zigzagTraversal


def test_canonical_example_1():
    # Zigzag order is [1, 2, 4, 3]; keeping traversal positions 0 and 2 gives [1, 4].
    assert zigzagTraversal([[1, 2], [3, 4]]) == [1, 4]


def test_canonical_example_2():
    # Zigzag order is [2, 1, 1, 2, 2, 1]; keeping positions 0, 2, and 4 gives [2, 1, 2].
    assert zigzagTraversal([[2, 1], [2, 1], [2, 1]]) == [2, 1, 2]


def test_canonical_example_3():
    # Zigzag order is [1, 2, 3, 6, 5, 4, 7, 8, 9]; keeping even positions gives [1, 3, 5, 7, 9].
    assert zigzagTraversal([[1, 2, 3], [4, 5, 6], [7, 8, 9]]) == [1, 3, 5, 7, 9]


def test_maximum_dimensions_and_cell_value():
    # A 50-by-50 grid has 2500 traversal cells; keeping every other cell yields 1250 values, all 2500.
    grid = [[2500] * 50 for _ in range(50)]
    assert zigzagTraversal(grid) == [2500] * 1250
