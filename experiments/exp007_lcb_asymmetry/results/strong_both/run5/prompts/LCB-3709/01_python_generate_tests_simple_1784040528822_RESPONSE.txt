from solution import hasSpecialSubstring


def test_canonical_example_1():
    # "aaa" at the end has length 3, one distinct character, and is preceded by "b".
    assert hasSpecialSubstring("aaabaaa", 3) is True


def test_canonical_example_2():
    # No length-2 substring of "abc" consists of only one distinct character.
    assert hasSpecialSubstring("abc", 2) is False


def test_accepts_run_exactly_k_surrounded_by_different_characters():
    # "bbb" has length 3 and is bounded by "a" and "c", both different from "b".
    assert hasSpecialSubstring("abbbc", 3) is True


def test_rejects_run_longer_than_k():
    # The only uniform run is "aaaa" of length 4, so no isolated uniform substring has exactly length 3.
    assert hasSpecialSubstring("aaaa", 3) is False


def test_accepts_entire_string_when_its_length_is_k():
    # The whole string "zzzz" has exactly length 4 and has no neighboring characters.
    assert hasSpecialSubstring("zzzz", 4) is True


def test_accepts_single_character_run_for_k_one():
    # The substring "b" has length 1 and is bounded by different characters "a" and "a".
    assert hasSpecialSubstring("aba", 1) is True


def test_rejects_k_one_inside_a_longer_run():
    # Each "a" belongs to the run "aa", so neither length-1 "a" substring has different adjacent "a" characters.
    assert hasSpecialSubstring("aa", 1) is False
