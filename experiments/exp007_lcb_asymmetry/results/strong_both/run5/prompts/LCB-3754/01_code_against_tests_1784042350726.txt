## Implementation

```python
def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0
    for i, char in enumerate(s, 1):
        if char == 'N':
            y += 1
        elif char == 'S':
            y -= 1
        elif char == 'E':
            x += 1
        elif char == 'W':
            x -= 1
        # Calculate the current Manhattan distance
        current_distance = abs(x) + abs(y)
        # Update the maximum distance considering allowed changes
        max_distance = max(max_distance, min(i, current_distance + 2 * k))
    return max_distance

```
