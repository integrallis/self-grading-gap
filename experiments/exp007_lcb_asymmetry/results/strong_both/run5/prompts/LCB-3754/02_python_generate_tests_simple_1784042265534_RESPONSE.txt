from solution import maxDistance


def test_canonical_example_1():
    # Change "NWSE" to "NWNE"; distances are 1, 2, 3, 2, so the maximum is 3.
    assert maxDistance("NWSE", 1) == 3


def test_canonical_example_2():
    # Change "NSWWEW" to "NNWWWW"; after all 6 moves the distance is 6.
    assert maxDistance("NSWWEW", 3) == 6


def test_single_move_without_changes():
    # "S" reaches (0, -1), whose Manhattan distance from the origin is 1.
    assert maxDistance("S", 0) == 1


def test_no_changes_uses_greatest_distance_over_all_prefixes():
    # "NESW" has prefix distances 1, 2, 1, 0, so the maximum is 2.
    assert maxDistance("NESW", 0) == 2


def test_change_can_make_all_moves_point_away_from_origin():
    # Change one S to N, giving either "NNS" or "NSN"; the greatest prefix distance is 2.
    assert maxDistance("NSS", 1) == 2


def test_at_most_k_changes_need_not_be_used():
    # "NNN" already has prefix distances 1, 2, 3, so its maximum remains 3.
    assert maxDistance("NNN", 2) == 3


def test_available_changes_cannot_exceed_prefix_length():
    # "SWWW" already reaches (-3, -1) after four moves, at distance 4.
    assert maxDistance("SWWW", 4) == 4


def test_best_distance_can_occur_before_final_move():
    # Change the first S in "SNEW" to N: "NNEW" reaches distance 3 after three moves.
    assert maxDistance("SNEW", 1) == 3


def test_change_from_north_to_south_can_be_optimal():
    # Change the final N to S, producing "SSS" and distance 3 after three moves.
    assert maxDistance("SSN", 1) == 3


def test_change_from_west_to_east_can_be_optimal():
    # Change the final W to E, producing "EEE" and distance 3 after three moves.
    assert maxDistance("EEW", 1) == 3


def test_maximum_length_with_no_changes():
    # 100000 north moves end at (0, 100000), whose distance is 100000.
    assert maxDistance("N" * 100000, 0) == 100000
