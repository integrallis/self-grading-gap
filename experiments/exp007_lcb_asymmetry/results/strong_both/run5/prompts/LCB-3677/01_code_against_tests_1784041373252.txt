from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[-float('inf')] * (n + 3) for _ in range(m + 3)]
    dp[0][0] = coins[0][0]

    for i in range(m):
        for j in range(n):
            if i > 0:
                dp[i][j] = max(dp[i][j], dp[i - 1][j] + coins[i][j])
            if j > 0:
                dp[i][j] = max(dp[i][j], dp[i][j - 1] + coins[i][j])

    max_coins = -float('inf')
    for i in range(m):
        for j in range(n):
            for a in range(max(0, i - 2), min(m, i + 1)):
                for b in range(max(0, j - 2), min(n, j + 1)):
                    if coins[i][j] < 0:
                        loss = -coins[i][j]
                        if (a, b) != (i, j):  # If not the same cell
                            offset = 0
                            if coins[a][b] < 0:
                                offset = -coins[a][b]
                            dp[i][j] = max(dp[i][j], dp[a][b] + (coins[i][j] + offset))
                    else:
                        dp[i][j] = max(dp[i][j], dp[a][b] + coins[i][j])

            max_coins = max(max_coins, dp[i][j])
    return max(0, max_coins)
