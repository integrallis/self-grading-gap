from solution import maximumAmount


def test_canonical_example_1():
    # Path 0 -> 1 -> neutralize -2 -> 3 -> 4 gives 0 + 1 + 0 + 3 + 4 = 8.
    assert maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) == 8


def test_canonical_example_2():
    # Any top-right/down path visits four cells worth 10 coins each: 10 + 10 + 10 + 10 = 40.
    assert maximumAmount([[10, 10, 10], [10, 10, 10]]) == 40


def test_single_negative_cell_can_be_neutralized():
    # The only cell is a robber worth -9; neutralizing it produces 0 coins.
    assert maximumAmount([[-9]]) == 0


def test_two_largest_robber_losses_are_neutralized_on_forced_path():
    # The forced path totals -5 - 2 - 7 - 1 = -15; neutralizing -7 and -5 gives -15 + 7 + 5 = -3.
    assert maximumAmount([[-5, -2, -7, -1]]) == -3


def test_route_and_neutralizations_are_chosen_together():
    # Top route: 0 -> -100 -> 100 -> 50, neutralizing -100, gives 150.
    # Bottom route gives at most 0 + 50 + 0 + 50 = 100 after neutralizing -1.
    assert maximumAmount([[0, -100, 100], [50, -1, 50]]) == 150
