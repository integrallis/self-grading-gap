from solution import sumOfGoodNumbers


def test_canonical_example_1():
    # nums[1] = 3, nums[4] = 5, and nums[5] = 4 are good; 3 + 5 + 4 = 12.
    assert sumOfGoodNumbers([1, 3, 2, 1, 5, 4], 2) == 12


def test_canonical_example_2():
    # Only nums[0] = 2 is greater than its existing k-distance neighbor; sum = 2.
    assert sumOfGoodNumbers([2, 1], 1) == 2


def test_requires_strictly_greater_than_both_existing_neighbors():
    # nums[1] = 2 equals nums[2] = 2, so neither is good; only nums[3] = 3 is good.
    assert sumOfGoodNumbers([1, 2, 2, 3], 1) == 3


def test_includes_elements_with_only_one_existing_k_distance_neighbor():
    # nums[0] = 5 > 1 and nums[3] = 4 > 1; 5 + 4 = 9.
    assert sumOfGoodNumbers([5, 1, 1, 4], 2) == 9
