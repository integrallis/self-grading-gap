from typing import List


def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers
    online_status = [True] * numberOfUsers  # Track online status
    offline_until = [0] * numberOfUsers  # Track when users will be back online

    for event in sorted(events, key=lambda x: int(x[1])):  # Sort events by timestamp
        event_type, timestamp, detail = event
        timestamp = int(timestamp)

        if event_type == "OFFLINE":
            user_id = int(detail)
            online_status[user_id] = False
            offline_until[user_id] = timestamp + 60
        elif event_type == "MESSAGE":
            # Update online statuses before processing message
            for user_id in range(numberOfUsers):
                if timestamp >= offline_until[user_id]:
                    online_status[user_id] = True

            if detail == "HERE":
                # Mention all online users
                for user_id in range(numberOfUsers):
                    if online_status[user_id]:
                        mentions[user_id] += 1
            elif detail == "ALL":
                # Mention all users
                for user_id in range(numberOfUsers):
                    mentions[user_id] += 1
            else:
                # Explicit mentions
                user_ids = detail.split()
                for user_id in user_ids:
                    if user_id.startswith("id"):
                        uid = int(user_id[2:])
                        mentions[uid] += 1

    return mentions
