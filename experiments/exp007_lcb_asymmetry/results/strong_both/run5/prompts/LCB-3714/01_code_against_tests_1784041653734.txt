```python
from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total = 0
    # Calculate contributions for subsequences of size 1 to k
    for size in range(1, k + 1):
        # Each element contributes as min and max in comb(n-1, size-1) subsequences
        for i in range(n):
            contribution = (nums[i] * 2) % MOD  # each element contributes both as min and max
            # Count how many subsequences of this size include nums[i] as min/max
            count = comb(n - 1, size - 1) % MOD
            total = (total + (contribution * count) % MOD) % MOD
    return total


def comb(n, k):
    if k > n:
        return 0
    if k == 0 or k == n:
        return 1
    k = min(k, n - k)  # Take advantage of symmetry
    c = 1
    for i in range(k):
        c = c * (n - i) // (i + 1)
    return c

```
