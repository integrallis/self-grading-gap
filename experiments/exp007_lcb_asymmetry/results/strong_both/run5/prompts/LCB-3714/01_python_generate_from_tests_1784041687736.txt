from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total = 0

    # Sort the numbers to easily find the min and max for subsequences
    nums.sort()

    # Calculate contributions for subsequences of size 1 to k
    for size in range(1, k + 1):
        # Each number contributes to many subsequences as min and max
        # Each number nums[i] will be the minimum in subsequences that include it
        # and are formed by taking any size-1 subset from the remaining numbers.
        for i in range(n):
            # Count how many subsequences of this size include nums[i] as min
            min_count = comb(n - 1, size - 1) % MOD
            # Count how many subsequences of this size include nums[i] as max
            max_count = comb(n - 1, size - 1) % MOD
            total = (total + (nums[i] * (min_count + max_count)) % MOD) % MOD

    return total


def comb(n, k):
    if k > n:
        return 0
    if k == 0 or k == n:
        return 1
    k = min(k, n - k)  # Take advantage of symmetry
    c = 1
    for i in range(k):
        c = c * (n - i) // (i + 1)
    return c

```
