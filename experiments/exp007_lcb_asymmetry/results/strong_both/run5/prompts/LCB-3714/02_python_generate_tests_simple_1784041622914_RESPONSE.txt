from solution import minMaxSums


def test_canonical_example_1():
    # [1], [2], [3], [1,2], [1,3], [2,3] contribute
    # 2 + 4 + 6 + 3 + 4 + 5 = 24.
    assert minMaxSums([1, 2, 3], 2) == 24


def test_canonical_example_2():
    # With k = 1, each element contributes itself as both minimum and maximum:
    # 2*5 + 2*0 + 2*6 = 22.
    assert minMaxSums([5, 0, 6], 1) == 22


def test_canonical_example_3():
    # Three singleton subsequences contribute 3 * 2, and three pairs contribute
    # 3 * 2, so the total is 6 + 6 = 12.
    assert minMaxSums([1, 1, 1], 2) == 12


def test_includes_subsequences_up_to_the_full_array_length():
    # Singles contribute 2 + 4 + 8 = 14; pairs contribute 3 + 5 + 6 = 14;
    # and [1,2,4] contributes 1 + 4 = 5. Total: 14 + 14 + 5 = 33.
    assert minMaxSums([1, 2, 4], 3) == 33


def test_returns_result_modulo_1000000007():
    # For k = 1, 100000 copies of 10^9 each contribute twice:
    # 2 * 100000 * 10^9 = 200000000000000,
    # and 200000000000000 mod 1000000007 = 998600007.
    assert minMaxSums([1_000_000_000] * 100_000, 1) == 998_600_007


def test_single_element_array():
    # The only subsequence is [7], whose minimum plus maximum is 7 + 7 = 14.
    assert minMaxSums([7], 1) == 14


def test_maximum_allowed_k():
    # Every nonempty subsequence of 70 ones has minimum plus maximum 1 + 1 = 2.
    # Thus the total is 2 * (2**70 - 1) mod 1000000007 = 540032504.
    assert minMaxSums([1] * 70, 70) == 540_032_504
