from solution import maxLength


def test_canonical_example_1():
    # [1, 2, 1, 1, 1] has product 2, gcd 1, and lcm 2, so 2 == 1 * 2.
    assert maxLength([1, 2, 1, 2, 1, 1, 1]) == 5


def test_canonical_example_2():
    # [3, 4, 5] has product 60, gcd 1, and lcm 60, so 60 == 1 * 60.
    assert maxLength([2, 3, 4, 5, 6]) == 3


def test_canonical_example_3():
    # [3, 1, 4, 5, 1] has product 60, gcd 1, and lcm 60, so 60 == 1 * 60.
    assert maxLength([1, 2, 3, 1, 4, 5, 1]) == 5


def test_all_ones_at_maximum_input_length():
    # All 100 ones have product 1, gcd 1, and lcm 1, so 1 == 1 * 1.
    assert maxLength([1] * 100) == 100


def test_repeated_non_one_values_limit_the_longest_subarray():
    # [2, 2] has product 4, gcd 2, and lcm 2, so 4 == 2 * 2;
    # [2, 2, 2] has product 8 while gcd * lcm is 2 * 2 = 4.
    assert maxLength([2, 2, 2]) == 2


def test_minimum_length_with_maximum_value():
    # [2, 10] has product 20, gcd 2, and lcm 10, so 20 == 2 * 10.
    assert maxLength([2, 10]) == 2
