from solution import minimumPairRemoval


def test_canonical_example_1():
    # (3, 1) has the minimum sum: [5, 2, 4]; then (2, 4): [5, 6].
    assert minimumPairRemoval([5, 2, 3, 1]) == 2


def test_canonical_example_2():
    # 1 <= 2 <= 2, so the input is already non-decreasing.
    assert minimumPairRemoval([1, 2, 2]) == 0


def test_single_element_is_non_decreasing():
    # A one-element array has no previous element to violate non-decreasing order.
    assert minimumPairRemoval([42]) == 0


def test_equal_minimum_pair_sums_choose_leftmost_pair():
    # Both adjacent sums are 4; choose leftmost (3, 1) -> [4, 3],
    # then merge (4, 3) -> [7], requiring two operations.
    assert minimumPairRemoval([3, 1, 3]) == 2


def test_negative_values_use_the_minimum_adjacent_sum():
    # 0 + (-2) = -2 is smaller than -2 + 3 = 1;
    # merge (0, -2) -> [-2, 3], which is non-decreasing.
    assert minimumPairRemoval([0, -2, 3]) == 1


def test_maximum_length_already_non_decreasing():
    # range(50) is non-decreasing, so no merges are needed.
    assert minimumPairRemoval(list(range(50))) == 0


def test_numeric_boundary_values_can_be_merged():
    # 1000 > -1000, so merge the only pair to obtain [0] after one operation.
    assert minimumPairRemoval([1000, -1000]) == 1
