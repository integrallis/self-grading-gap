```python
from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize a dp array to store the minimum costs
    dp = [[0] * 3 for _ in range(n)]
    # Base case: the cost of painting the first house
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the dp array for the first half of the houses
    for i in range(1, n // 2 + 1):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(dp[i - 1][(j + 1) % 3], dp[i - 1][(j + 2) % 3])

    # Calculate the minimum cost for the second half of the houses
    for i in range(n // 2, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(
                dp[i - 1][(j + 1) % 3], dp[i - 1][(j + 2) % 3]
            )

    # Ensure that the current house does not match its mirrored house
    for j in range(3):
        dp[n - 1][j] = cost[n - 1][j] + min(
            dp[n // 2 - 1][(j + 1) % 3], dp[n // 2 - 1][(j + 2) % 3]
        )

    # The result is the minimum cost of painting all houses
    return min(dp[n - 1])

```
