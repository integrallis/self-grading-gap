from solution import minCost


def test_canonical_example_1():
    cost = [[3, 5, 7], [6, 2, 9], [4, 8, 1], [7, 3, 5]]

    # Colors [1, 2, 3, 2] satisfy both rules and cost 3 + 2 + 1 + 3 = 9.
    assert minCost(4, cost) == 9


def test_canonical_example_2():
    cost = [[2, 4, 6], [5, 3, 8], [7, 1, 9], [4, 6, 2], [3, 5, 7], [8, 2, 4]]

    # Colors [1, 3, 2, 3, 1, 2] satisfy both rules and cost 2 + 8 + 1 + 2 + 3 + 2 = 18.
    assert minCost(6, cost) == 18


def test_two_houses_must_use_different_colors():
    cost = [[0, 5, 9], [0, 2, 3]]

    # The two houses are adjacent and equidistant, so they differ; colors [1, 2] cost 0 + 2 = 2.
    assert minCost(2, cost) == 2


def test_equally_distant_end_houses_cannot_share_the_cheapest_color():
    cost = [[0, 100, 100], [100, 0, 100], [100, 100, 0], [0, 100, 100]]

    # The individually cheapest colors [1, 2, 3, 1] are invalid because houses 0 and 3 match.
    # Colors [3, 2, 3, 1] satisfy both rules and cost 100 + 0 + 0 + 0 = 100.
    assert minCost(4, cost) == 100


def test_maximum_number_of_houses_with_zero_costs():
    n = 100000
    cost = [[0, 0, 0] for _ in range(n)]

    # Any valid alternating coloring has total cost 0 + ... + 0 = 0.
    assert minCost(n, cost) == 0


def test_maximum_individual_cost_value():
    cost = [[100000, 100000, 100000], [100000, 100000, 100000]]

    # The two houses must differ, and any two colors cost 100000 + 100000 = 200000.
    assert minCost(2, cost) == 200000
