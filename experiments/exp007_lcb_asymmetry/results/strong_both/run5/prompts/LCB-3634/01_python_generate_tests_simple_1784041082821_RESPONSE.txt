from solution import calculateScore


def test_canonical_example_1():
    # i=2 pairs with j=0 for 2 points; i=4 pairs with j=1 for 3 points; total = 5.
    assert calculateScore("aczzx") == 5


def test_canonical_example_2():
    # No character has an earlier unmarked mirrored character, so the total is 0.
    assert calculateScore("abcdef") == 0


def test_uses_closest_unmarked_mirrored_index():
    # i=2 ('z') pairs with closest 'a' at j=1 for 1; i=3 pairs with j=0 for 3; total = 4.
    assert calculateScore("aazz") == 4


def test_marked_indices_cannot_be_reused():
    # i=1 pairs with j=0 for 1; i=2 has no unmarked 'a'; i=3 pairs with unmarked j=2 for 1; total = 2.
    assert calculateScore("azza") == 2
