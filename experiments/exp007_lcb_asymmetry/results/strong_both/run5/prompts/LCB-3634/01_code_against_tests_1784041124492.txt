## Implementation

```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # a-z mirrors

    for i in range(len(s)):
        if marked[i]:  # Skip if already marked
            continue
        for j in range(i - 1, -1, -1):  # Look for closest unmarked j
            if marked[j]:
                continue
            if mirror[s[i]] == s[j]:  # Check if they are mirrors
                marked[i] = marked[j] = True  # Mark both
                score += (i - j)
                break  # Found closest j
    return score
```
