from solution import maximumCoins


def test_canonical_example_1():
    # Bags [3, 4, 5, 6] contain 2 + 0 + 4 + 4 = 10 coins.
    assert maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) == 10


def test_canonical_example_2():
    # Any two bags within [1, 10] contain 3 + 3 = 6 coins.
    assert maximumCoins([[1, 10, 3]], 2) == 6


def test_window_can_include_gaps_between_segments():
    # Bags [1, 5] contain 5 + 5 + 0 + 0 + 4 = 14 coins.
    assert maximumCoins([[1, 2, 5], [5, 6, 4]], 5) == 14


def test_window_can_take_partial_segment_before_high_value_bag():
    # Bags [7, 11] contain 1 + 1 + 1 + 1 + 100 = 104 coins.
    assert maximumCoins([[1, 10, 1], [11, 11, 100]], 5) == 104


def test_window_long_enough_to_collect_every_segment():
    # All covered bags contribute 3 * 2 + 2 * 5 = 16 coins.
    assert maximumCoins([[10, 12, 2], [20, 21, 5]], 15) == 16


def test_window_matching_a_segment_length_collects_its_full_value():
    # Bags [4, 6] contain 7 + 7 + 7 = 21 coins.
    assert maximumCoins([[4, 6, 7]], 3) == 21
