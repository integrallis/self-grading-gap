```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a dictionary to hold the total coins at each bag position
    coin_dict = {}
    
    # Fill the dictionary with the coins in each range
    for l, r, c in coins:
        for pos in range(l, r + 1):
            coin_dict[pos] = coin_dict.get(pos, 0) + c
    
    # Prepare a sorted list of positions
    positions = sorted(coin_dict.keys())
    
    # Use a sliding window approach to find the maximum coins obtainable
    max_coins = 0
    current_coins = 0
    left = 0
    
    for right in range(len(positions)):
        # Add the coins in the rightmost position of the window
        current_coins += coin_dict[positions[right]]
        
        # Move the left side of the window as necessary
        while positions[right] - positions[left] + 1 > k:
            current_coins -= coin_dict[positions[left]]
            left += 1
        
        # Update the maximum coins found
        max_coins = max(max_coins, current_coins)
    
    return max_coins
```
