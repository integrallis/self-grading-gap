```python
def maxDistance(s: str, k: int) -> int:
    # Initialize variables for position and maximum distance
    x, y = 0, 0
    max_dist = 0
    # Count the number of moves in each direction
    move_counts = {'N': 0, 'S': 0, 'E': 0, 'W': 0}

    # Iterate through the movements
    for move in s:
        if move == 'N':
            y += 1
        elif move == 'S':
            y -= 1
        elif move == 'E':
            x += 1
        elif move == 'W':
            x -= 1
        move_counts[move] += 1

        # Calculate current Manhattan distance
        current_distance = abs(x) + abs(y)
        # Update maximum distance considering k changes
        max_dist = max(max_dist, current_distance + min(k, (move_counts['N'] + move_counts['S'] + move_counts['E'] + move_counts['W'])))

    # The final maximum distance achievable
    return max_dist
```
