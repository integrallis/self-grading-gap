```python
from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[v].append((u, w))  # reversing the direction for reachability to node 0

    def canReachWithMaxWeight(maxWeight):
        out_degree_count = [0] * n
        queue = []
        visited = [False] * n

        visited[0] = True
        for starting_node in range(n):
            if visited[starting_node]:
                for v, w in graph[starting_node]:
                    if w <= maxWeight:
                        out_degree_count[starting_node] += 1
                        if out_degree_count[starting_node] <= threshold:
                            if not visited[v]:
                                visited[v] = True
                                queue.append(v)

        while queue:
            current = queue.pop(0)
            for next_node, w in graph[current]:
                if w <= maxWeight and not visited[next_node]:
                    visited[next_node] = True
                    queue.append(next_node)

        return all(visited)

    left, right = 1, 10**6
    result = -1
    while left <= right:
        mid = (left + right) // 2
        if canReachWithMaxWeight(mid):
            result = mid
            right = mid - 1
        else:
            left = mid + 1

    return result
```
