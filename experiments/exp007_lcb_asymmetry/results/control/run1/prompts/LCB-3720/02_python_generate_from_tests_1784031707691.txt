from typing import List
import heapq


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[v].append((u, w))  # reversing the direction for reachability to node 0

    def canReachWithMaxWeight(maxWeight):
        out_degree_count = [0] * n
        queue = []
        visited = [False] * n

        # Start from node 0
        visited[0] = True
        for u in range(n):
            for v, w in graph[u]:
                if w <= maxWeight:
                    out_degree_count[u] += 1
                    if out_degree_count[u] <= threshold and not visited[v]:
                        visited[v] = True
                        queue.append(v)

        # BFS to check if all nodes can reach 0
        while queue:
            current = queue.pop(0)
            for next_node, w in graph[current]:
                if w <= maxWeight and not visited[next_node]:
                    visited[next_node] = True
                    queue.append(next_node)

        return all(visited)

    # Binary search for the minimum max weight
    left, right = 1, 10**6
    result = -1
    while left <= right:
        mid = (left + right) // 2
        if canReachWithMaxWeight(mid):
            result = mid
            right = mid - 1
        else:
            left = mid + 1

    return result
```
