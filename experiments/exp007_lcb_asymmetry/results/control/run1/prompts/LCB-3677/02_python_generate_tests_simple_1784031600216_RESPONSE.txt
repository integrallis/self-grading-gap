from typing import List
from solution import maximumAmount

def test_canonical_example_1():
    # spec: maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) -> 8
    # Explanation: 
    # Start at (0, 0): 0 coins
    # Move to (0, 1): gain 1 coin -> total = 1
    # Move to (1, 1): robber steals 2 coins, neutralize -> total = 1
    # Move to (1, 2): gain 3 coins -> total = 4
    # Move to (2, 2): gain 4 coins -> total = 8
    assert maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) == 8

def test_canonical_example_2():
    # spec: maximumAmount([[10, 10, 10], [10, 10, 10]]) -> 40
    # Explanation:
    # Start at (0, 0): 10 coins
    # Move to (0, 1): gain 10 coins -> total = 20
    # Move to (0, 2): gain 10 coins -> total = 30
    # Move to (1, 2): gain 10 coins -> total = 40
    assert maximumAmount([[10, 10, 10], [10, 10, 10]]) == 40

def test_empty_grid():
    # spec: maximumAmount([[]]) -> 0
    # Explanation: An empty grid means no coins to collect.
    assert maximumAmount([[]]) == 0

def test_grid_with_all_robbers():
    # spec: maximumAmount([[-1, -1], [-1, -1]]) -> 2
    # Explanation: Start with 0 coins, neutralize 2 robbers and end with 0 coins.
    assert maximumAmount([[-1, -1], [-1, -1]]) == 2

    # spec: maximumAmount([[1, 1], [1, 1]]) -> 4

    # spec: maximumAmount([[-1]]) -> 0

    # spec: maximumAmount([[1] * 500] * 500) -> 250000
