from typing import List
from solution import minCosts

def test_canonical_example_1():
    # spec: cost = [5, 3, 4, 1, 3, 2] 
    # answer[0] = 5, answer[1] = 3, answer[2] = 3, answer[3] = 1, answer[4] = 1, answer[5] = 1
    # Therefore, output should be [5, 3, 3, 1, 1, 1]
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]

def test_canonical_example_2():
    # spec: cost = [1, 2, 4, 6, 7] 
    # answer[0] = 1, answer[1] = 1, answer[2] = 1, answer[3] = 1, answer[4] = 1
    # Therefore, output should be [1, 1, 1, 1, 1]
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]
