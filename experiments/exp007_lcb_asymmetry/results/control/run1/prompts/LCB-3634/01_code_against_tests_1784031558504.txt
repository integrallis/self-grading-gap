```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirrors = {chr(i): chr(219 - i) for i in range(97, 123)}  # Create a mirror mapping for a-z

    for i in range(len(s)):
        if marked[i]:  # Skip if already marked
            continue
        mirror_char = mirrors[s[i]]  # Find the mirror character
        # Find the closest unmarked index j
        for j in range(i - 1, -1, -1):
            if not marked[j] and s[j] == mirror_char:
                marked[i] = True
                marked[j] = True
                score += i - j  # Update score
                break  # Stop searching after finding the first unmarked mirror
    return score
```
