from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    dp = [[0] * 3 for _ in range(n)]
    dp[0] = cost[0]
    dp[1][0] = cost[1][0] + dp[0][1]
    dp[1][1] = cost[1][1] + min(dp[0][0], dp[0][2])
    dp[1][2] = cost[1][2] + dp[0][1]
    for i in range(2, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(
                dp[i-1][(j+1)%3],
                dp[i-1][(j+2)%3]
            )
    total_cost = [dp[n-1][i] + dp[n-2][(i+1)%3] for i in range(3)]
    return min(total_cost)
```
