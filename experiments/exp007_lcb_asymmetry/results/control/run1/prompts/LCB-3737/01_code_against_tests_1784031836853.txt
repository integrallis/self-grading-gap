```python
from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    dp = [[0] * 3 for _ in range(n)]
    dp[0] = cost[0]
    for j in range(3):
        if j == 0:
            dp[1][j] = cost[1][j] + dp[0][1]
        else:
            dp[1][j] = cost[1][j] + min(dp[0][(j-1)%3], dp[0][(j+1)%3])
    for i in range(2, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(
                dp[i-1][(j-1)%3],
                dp[i-1][(j+1)%3]
            )
    total_cost = [dp[n-1][i] + dp[n-2][(i+1)%3] for i in range(3)]
    return min(total_cost)
```
