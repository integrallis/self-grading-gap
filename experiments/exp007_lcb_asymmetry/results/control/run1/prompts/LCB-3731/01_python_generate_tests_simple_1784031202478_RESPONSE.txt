from typing import List
from solution import subarraySum

def test_canonical_example_1():
    # Given input: [2, 3, 1]
    # Subarrays:
    # i=0: nums[0] = [2] -> sum = 2
    # i=1: nums[0...1] = [2, 3] -> sum = 2 + 3 = 5
    # i=2: nums[1...2] = [3, 1] -> sum = 3 + 1 = 4
    # Total sum = 2 + 5 + 4 = 11
    assert subarraySum([2, 3, 1]) == 11

def test_canonical_example_2():
    # Given input: [3, 1, 1, 2]
    # Subarrays:
    # i=0: nums[0] = [3] -> sum = 3
    # i=1: nums[0...1] = [3, 1] -> sum = 3 + 1 = 4
    # i=2: nums[1...2] = [1, 1] -> sum = 1 + 1 = 2
    # i=3: nums[1...3] = [1, 1, 2] -> sum = 1 + 1 + 2 = 4
    # Total sum = 3 + 4 + 2 + 4 = 13
    assert subarraySum([3, 1, 1, 2]) == 13
