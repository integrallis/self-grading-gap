from solution import maxContainers

def test_canonical_example_1():
    # spec: maxContainers(2, 3, 15) 
    # Total cells: 2*2 = 4, Total weight if all loaded: 4 * 3 = 12
    # 12 <= 15, so maximum containers = 4

def test_canonical_example_2():
    # spec: maxContainers(3, 5, 20) 
    # Total cells: 3*3 = 9, Total weight if all loaded: 9 * 5 = 45
    # 45 > 20, maximum containers = 20 // 5 = 4
    assert maxContainers(3, 5, 20) == 4
