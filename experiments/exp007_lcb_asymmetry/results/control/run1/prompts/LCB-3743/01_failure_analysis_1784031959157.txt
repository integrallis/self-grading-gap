from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_times = []

    # Keep track of the free time segments
    prev_end = 0
    for i in range(n):
        if startTime[i] > prev_end:
            free_times.append((prev_end, startTime[i]))
        prev_end = endTime[i]

    # Add a final segment if there's time after the last meeting
    if prev_end < eventTime:
        free_times.append((prev_end, eventTime))

    # Calculate free time segments without rescheduling
    max_free_time = 0
    for start, end in free_times:
        max_free_time = max(max_free_time, end - start)

    # Attempt to maximize free time by rescheduling
    # Using a list to keep track of possible moves
    possible_moves = []
    for i in range(n):
        move_start = max(0, startTime[i] - (endTime[i] - startTime[i]))
        move_end = min(eventTime, endTime[i] + (endTime[i] - startTime[i]))
        if move_start < startTime[i]:
            possible_moves.append((move_start, startTime[i]))
        if move_end > endTime[i]:
            possible_moves.append((endTime[i], move_end))

    # Sort possible moves by start time
    possible_moves.sort()

    max_possible_free_time = max_free_time
    for i in range(min(k, len(possible_moves))):
        start_move, end_move = possible_moves[i]
        # Calculate the new free time after the ith move
        new_free_time = end_move - start_move
        max_possible_free_time = max(max_possible_free_time, new_free_time)

    return max_possible_free_time
