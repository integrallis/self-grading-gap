MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    nums.sort()

    # Calculate contributions of each number as min and max
    for i in range(n):
        # Contributions as min
        if i < k:
            total_sum += nums[i] * pow(2, i, MOD) * pow(2, n - 1 - i, MOD) % MOD
        # Contributions as max
        if n - 1 - i < k:
            total_sum += nums[i] * pow(2, n - 1 - i, MOD) * pow(2, i, MOD) % MOD

    return total_sum % MOD
