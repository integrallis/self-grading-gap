from typing import List

def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a list to hold the number of coins at each position
    max_position = 0
    for l, r, c in coins:
        max_position = max(max_position, r)

    # Create a coin array initialized to zero
    coin_array = [0] * (max_position + 2)

    # Fill the coin array based on the coins data
    for l, r, c in coins:
        for i in range(l, r + 1):
            coin_array[i] += c

    # Calculate the maximum coins that can be collected from k consecutive bags
    max_coins = 0
    current_coins = 0

    # Calculate the initial sum of the first k bags, if applicable
    for i in range(1, min(k + 1, max_position + 1)):
        current_coins += coin_array[i]

    max_coins = current_coins

    # Use a sliding window to calculate the sum of k consecutive bags
    for i in range(k + 1, max_position + 1):
        current_coins += coin_array[i] - coin_array[i - k]
        max_coins = max(max_coins, current_coins)

    return max_coins
