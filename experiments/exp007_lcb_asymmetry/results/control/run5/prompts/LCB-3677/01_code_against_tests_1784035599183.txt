```python
from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m = len(coins)
    n = len(coins[0]) if m > 0 else 0
    dp = [[float('-inf')] * (n + 1) for _ in range(m + 1)]
    dp[0][0] = 0  # Starting point with 0 coins

    for i in range(m):
        for j in range(n):
            for used in range(3):  # 0 to 2 robbers neutralized
                current = coins[i][j]
                if dp[i][j] != float('-inf'):
                    new_coins = dp[i][j] + (current if current >= 0 else 0)
                    if current < 0 and used < 2:
                        # Neutralizing the robbery
                        dp[i + 1][j] = max(dp[i + 1][j], new_coins)
                        dp[i][j + 1] = max(dp[i][j + 1], new_coins)
                    else:
                        # Updating without neutralizing the robbery
                        reduced_coins = (0 if current >= 0 else current)
                        dp[i + 1][j] = max(dp[i + 1][j], new_coins + reduced_coins)
                        dp[i][j + 1] = max(dp[i][j + 1], new_coins + reduced_coins)

    return max(dp[m][n], 0)  # Return max coins at bottom-right corner or 0 if negative

```
