from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    free_intervals = []
    last_end = 0

    # Calculate the free time intervals between meetings
    for i in range(len(startTime)):
        if last_end < startTime[i]:
            free_intervals.append((last_end, startTime[i]))
        last_end = endTime[i]

    if last_end < eventTime:
        free_intervals.append((last_end, eventTime))

    # Sort the free intervals by their length
    free_intervals.sort(key=lambda x: x[1] - x[0], reverse=True)

    # Try to maximize free time by rescheduling meetings
    rescheduled = 0
    total_free_time = 0

    for start, end in free_intervals:
        if rescheduled < k:
            total_free_time += (end - start)
            rescheduled += 1
        else:
            break

    return total_free_time
```
