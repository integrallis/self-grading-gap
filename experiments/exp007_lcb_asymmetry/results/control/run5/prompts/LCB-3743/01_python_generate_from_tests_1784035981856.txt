from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    # Create a list to store free time slots
    free_intervals = []
    last_end = 0

    # Calculate the free time intervals between meetings
    for i in range(len(startTime)):
        if last_end < startTime[i]:
            free_intervals.append((last_end, startTime[i]))
        last_end = endTime[i]

    if last_end < eventTime:
        free_intervals.append((last_end, eventTime))

    # Calculate the maximum free time possible by moving meetings
    max_free_time = 0
    free_time_slots = []

    # Collect free time slots
    for start, end in free_intervals:
        free_time_slots.append(end - start)

    # If we have more free time slots than k, we will take the largest k free time slots
    free_time_slots.sort(reverse=True)
    max_free_time = sum(free_time_slots[:k]) if k <= len(free_time_slots) else sum(free_time_slots)

    # Finally, return the maximum free time
    return max_free_time
