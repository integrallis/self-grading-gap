from typing import List

def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    def canReachNode0(maxWeight):
        graph = [[] for _ in range(n)]
        for u, v, w in edges:
            if w <= maxWeight:
                graph[u].append(v)  # Corrected direction

        reachable = [False] * n
        reachable[0] = True
        queue = [0]
        while queue:
            node = queue.pop(0)  # Changed to BFS
            for neighbor in graph[node]:
                if not reachable[neighbor]:
                    reachable[neighbor] = True
                    queue.append(neighbor)

        return all(reachable)

    def countOutgoingEdges(maxWeight):
        graph = [[] for _ in range(n)]
        for u, v, w in edges:
            if w <= maxWeight:
                graph[u].append((v, w))

        outgoingCount = [0] * n
        for u in range(n):
            for v, _ in graph[u]:
                outgoingCount[u] += 1
                if outgoingCount[u] > threshold:
                    return False
        return True

    left, right = 1, 10**6
    result = -1
    while left <= right:
        mid = (left + right) // 2
        if canReachNode0(mid) and countOutgoingEdges(mid):
            result = mid
            right = mid - 1
        else:
            left = mid + 1

    return result
```
