from typing import List

MOD = 10**9 + 7

from math import comb

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    nums.sort()
    total_sum = 0

    for i in range(n):
        min_contrib = nums[i] * 
        (comb(i, 0) * (1 << min(k - 1, n - i - 1))) % MOD
        total_sum = (total_sum + min_contrib) % MOD

        max_contrib = nums[i] * 
        (comb(n - i - 1, 0) * (1 << min(k - 1, i))) % MOD
        total_sum = (total_sum + max_contrib) % MOD

    return total_sum % MOD
```
