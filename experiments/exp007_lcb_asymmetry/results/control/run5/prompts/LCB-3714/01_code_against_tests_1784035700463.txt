## Implementation

```python
from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    nums.sort()

    total_sum = 0

    # Count contributions of min and max
    for i in range(n):
        # Count how many times nums[i] is a min in subsequences of size 1 to k
        # For each position i, it can be the minimum of subsequences of size j
        # where j ranges from 1 to k, and nums[i] is the minimum in these subsequences.
        min_contrib = nums[i] * (1 << min(k, i)) % MOD
        total_sum = (total_sum + min_contrib) % MOD

        # Count how many times nums[i] is a max in subsequences of size 1 to k
        # For each position i, it can be the maximum of subsequences of size j
        # where j ranges from 1 to k, and nums[i] is the maximum in these subsequences.
        max_contrib = nums[i] * (1 << min(k, n - 1 - i)) % MOD
        total_sum = (total_sum + max_contrib) % MOD

    return total_sum % MOD
```
