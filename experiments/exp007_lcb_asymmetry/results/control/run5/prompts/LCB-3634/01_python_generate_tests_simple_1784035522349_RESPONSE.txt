from solution import calculateScore

def test_canonical_example_1():
    # spec: calculateScore("aczzx") -> 5
    # Explanation:
    # i = 0. No j found. Skip.
    # i = 1. No j found. Skip.
    # i = 2. Closest j = 0 (mirror 'a' = 'z'). Score += 2 - 0 = 2.
    # i = 3. No j found. Skip.
    # i = 4. Closest j = 1 (mirror 'c' = 'x'). Score += 4 - 1 = 3.
    # Total score = 2 + 3 = 5
    assert calculateScore("aczzx") == 5

def test_canonical_example_2():
    # spec: calculateScore("abcdef") -> 0
    # Explanation:
    # For each index i, there is no j that satisfies the conditions.
    assert calculateScore("abcdef") == 0
