def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers
    online_status = [True] * numberOfUsers
    event_list = []

    # Process events and track online status
    for event in events:
        event_type, timestamp, *details = event
        timestamp = int(timestamp)
        if event_type == "OFFLINE":
            user_id = int(details[0])
            online_status[user_id] = False
            event_list.append((timestamp + 60, user_id, 'ONLINE'))  # Schedule coming back online
        elif event_type == "MESSAGE":
            event_list.append((timestamp, 'MESSAGE', details[0]))

    # Sort events by timestamp
    event_list.sort(key=lambda x: x[0])

    # Process sorted events
    for event in event_list:
        if event[1] == 'MESSAGE':
            mentions_string = event[2]
            mentioned_users = set()
            if mentions_string == "ALL":
                mentioned_users = set(range(numberOfUsers))
            elif mentions_string == "HERE":
                mentioned_users = {i for i in range(numberOfUsers) if online_status[i]}
            else:
                for mention in mentions_string.split():
                    if mention.startswith("id"):
                        user_id = int(mention[2:])
                        if online_status[user_id]:
                            mentioned_users.add(user_id)
            for user_id in mentioned_users:
                mentions[user_id] += 1
        else:
            # Handle online event
            _, user_id, status = event
            if status == 'ONLINE':
                online_status[user_id] = True

    return mentions
