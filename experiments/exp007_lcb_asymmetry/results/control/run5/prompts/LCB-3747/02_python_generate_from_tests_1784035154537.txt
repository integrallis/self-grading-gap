from typing import List

def maxAdjacentDistance(nums: List[int]) -> int:
    n = len(nums)
    max_distance = 0
    for i in range(n):
        # Calculate absolute difference between adjacent elements
        distance = abs(nums[i] - nums[(i + 1) % n])
        max_distance = max(max_distance, distance)
    return max_distance
```
