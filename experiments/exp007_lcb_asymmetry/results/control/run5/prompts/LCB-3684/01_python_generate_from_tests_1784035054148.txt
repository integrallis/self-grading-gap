def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]

    for i in range(len(s) - len(prefix) - len(suffix) + 1):
        if s.startswith(prefix, i) and s[i + len(prefix):].startswith(suffix): 
            return True
    return False

```
