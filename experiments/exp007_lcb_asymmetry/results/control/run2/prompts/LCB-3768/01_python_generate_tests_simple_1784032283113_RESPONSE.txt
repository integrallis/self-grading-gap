from solution import hasSameDigits

def test_canonical_example_1():
    # spec: hasSameDigits("3902") -> True
    # Explanation: 
    # Initial s = "3902"
    # First operation: (3+9)%10=2, (9+0)%10=9, (0+2)%10=2 -> s = "292"
    # Second operation: (2+9)%10=1, (9+2)%10=1 -> s = "11"
    # Final digits are the same '1' == '1'
    assert hasSameDigits("3902") is True

def test_canonical_example_2():
    # spec: hasSameDigits("34789") -> False
    # Explanation: 
    # Initial s = "34789"
    # First operation: (3+4)%10=7, (4+7)%10=1, (7+8)%10=5, (8+9)%10=7 -> s = "7157"
    # Second operation: (7+1)%10=8, (1+5)%10=6, (5+7)%10=2 -> s = "862"
    # Third operation: (8+6)%10=4, (6+2)%10=8 -> s = "48"
    # Final digits are different '4' != '8'
    assert hasSameDigits("34789") is False
