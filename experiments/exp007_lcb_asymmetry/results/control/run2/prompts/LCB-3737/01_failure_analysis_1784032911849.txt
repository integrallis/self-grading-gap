from typing import List

def minCost(n: int, cost: List[List[int]]) -> int:
    # dp[i][c] will be the minimum cost to paint up to house i with color c
    dp = [[0] * 3 for _ in range(n)]
    # Initialize the first house with its corresponding costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill in the dp table for each house
    for i in range(1, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(dp[i - 1][(j + 1) % 3], dp[i - 1][(j + 2) % 3])

    # Handle the equidistant condition
    min_cost = float('inf')

    for color1 in range(3):
        for color2 in range(3):
            if color1 != color2:
                # Cost for first half with color1 and second half with color2
                total_cost = 0
                for i in range(n // 2):
                    total_cost += dp[i][(color1 + i) % 3]
                for i in range(n // 2, n):
                    total_cost += dp[i][(color2 + (i - n // 2)) % 3]
                min_cost = min(min_cost, total_cost)

    return min_cost
