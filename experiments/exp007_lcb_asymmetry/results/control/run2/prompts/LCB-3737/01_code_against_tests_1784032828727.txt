```python
from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # dp[i][j] will be the minimum cost to paint up to house i with color j
    dp = [[0] * 3 for _ in range(n)]
    # Initialize the first house with its corresponding costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill in the dp table for each house
    for i in range(1, n):
        for j in range(3):
            # Choose the minimum cost of painting the previous house with a different color
            dp[i][j] = cost[i][j] + min(dp[i - 1][(j + 1) % 3], dp[i - 1][(j + 2) % 3])

    # Handle the equidistant condition
    # Consider two cases, painting the first half with one set of colors and the second half with another
    min_cost = float('inf')

    for color1 in range(3):
        for color2 in range(3):
            if color1 != color2:
                # Calculate total cost for this combination
                total_cost = dp[n//2 - 1][color1] + dp[n - 1][color2]
                min_cost = min(min_cost, total_cost)

    return min_cost
```
