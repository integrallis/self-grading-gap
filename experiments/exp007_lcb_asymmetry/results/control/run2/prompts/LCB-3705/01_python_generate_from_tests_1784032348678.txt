from typing import List

def largestInteger(nums: List[int], k: int) -> int:
    from collections import defaultdict

    count = defaultdict(int)
    n = len(nums)

    # Count occurrences of each integer in each subarray of size k
    for i in range(n - k + 1):
        for j in range(i, i + k):
            count[nums[j]] += 1

    # Find the largest integer that appears in exactly one subarray
    largest_almost_missing = -1
    for num, occurrences in count.items():
        if occurrences == 1:
            largest_almost_missing = max(largest_almost_missing, num)

    return largest_almost_missing
```
