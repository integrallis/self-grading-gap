from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a list to hold the number of coins in each bag
    coin_count = {}

    # Fill the coin_count dictionary based on the coins array
    for l, r, c in coins:
        for i in range(l, r + 1):
            coin_count[i] = coin_count.get(i, 0) + c

    # Convert the dictionary to a sorted list of bags (positions)
    positions = sorted(coin_count.keys())
    total_bags = len(positions)

    # If k is greater than the number of available bags, return 0
    if k > total_bags:
        return 0

    # Initialize the maximum coins we can get from a sliding window
    max_coins = 0
    current_coins = 0
    left = 0

    # Slide through the positions to find the maximum coins in k consecutive bags
    for right in range(total_bags):
        current_coins += coin_count[positions[right]]

        # If we have more than k bags, slide the window
        if right - left + 1 > k:
            current_coins -= coin_count[positions[left]]
            left += 1

        # Update max_coins if we have exactly k bags
        if right - left + 1 == k:
            max_coins = max(max_coins, current_coins)

    return max_coins
