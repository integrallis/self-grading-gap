```python
def calculateScore(s: str) -> int:
    score = 0
    marked = [False] * len(s)
    mirror = {chr(i + 97): chr(122 - i) for i in range(26)}  # Create a mirror mapping for 'a' to 'z'
    for i in range(len(s)):
        if marked[i]:  # Skip if already marked
            continue
        for j in range(i - 1, -1, -1):  # Look for the closest unmarked index j < i
            if not marked[j] and s[j] == mirror[s[i]]:  # Found a mirror
                marked[i] = marked[j] = True
                score += i - j
                break  # Stop once we find the first valid j
    return score

```
