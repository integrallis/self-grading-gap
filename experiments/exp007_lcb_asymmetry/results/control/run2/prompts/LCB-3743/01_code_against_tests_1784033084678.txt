```python
from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    prev_end = 0
    free_intervals = []

    # Collect the free intervals before the meetings start
    if startTime[0] > 0:
        free_intervals.append((0, startTime[0]))

    for i in range(n):
        if startTime[i] > prev_end:
            free_intervals.append((prev_end, startTime[i]))
        prev_end = endTime[i]

    # Collect free time after the last meeting
    if prev_end < eventTime:
        free_intervals.append((prev_end, eventTime))

    # Calculate gaps between meetings
    gaps = []
    for i in range(1, n):
        gap = startTime[i] - endTime[i - 1]
        if gap > 0:
            gaps.append(gap)

    # Sort gaps in descending order to maximize free time by rescheduling
    gaps.sort(reverse=True)
    max_free_time = 0

    # Calculate maximum free time without any rescheduling
    if free_intervals:
        max_free_time = max(free_end - free_start for free_start, free_end in free_intervals)

    # Calculate extra free time by moving up to k meetings
    extra_free_time = sum(gaps[:k])

    return max(max_free_time, extra_free_time)
```
