from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_intervals = []
    prev_end = 0

    # Collect the free intervals before the meetings start
    for i in range(n):
        if startTime[i] > prev_end:
            free_intervals.append((prev_end, startTime[i]))
        prev_end = endTime[i]

    # Collect free time after the last meeting
    if prev_end < eventTime:
        free_intervals.append((prev_end, eventTime))

    # We need to find the maximum free time possible by rescheduling meetings
    max_free_time = 0
    for i in range(len(free_intervals)):
        free_start, free_end = free_intervals[i]
        max_free_time = max(max_free_time, free_end - free_start)

    # Attempt to reschedule up to k meetings to maximize free time
    rescheduled_free_times = []
    for i in range(n):
        if i > 0:
            free_time_after = startTime[i] - endTime[i - 1]
            rescheduled_free_times.append(free_time_after)

    rescheduled_free_times.sort(reverse=True)

    # Sum up the largest k free times that can be created by rescheduling
    extra_free_time = sum(rescheduled_free_times[:k]) if k <= len(rescheduled_free_times) else sum(rescheduled_free_times)

    return max(max_free_time, extra_free_time)
