from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Sort edges by weight
    edges.sort(key=lambda x: x[2])
    left, right = 0, 10**6
    result = -1

    def canReachNode0(maxWeight):
        from collections import defaultdict, deque
        graph = defaultdict(list)
        for u, v, w in edges:
            if w <= maxWeight:
                graph[u].append(v)  # Change direction to build graph from u to v

        visited = [False] * n
        queue = deque([0])
        visited[0] = True

        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    queue.append(neighbor)

        return all(visited[i] for i in range(n))

    while left <= right:
        mid = (left + right) // 2
        if canReachNode0(mid):
            result = mid
            right = mid - 1
        else:
            left = mid + 1

    if result == -1:
        return -1

    # Check if the resulting graph can have at most threshold edges per node
    outgoing_count = [0] * n
    for u, v, w in edges:
        if w <= result:
            outgoing_count[u] += 1
            if outgoing_count[u] > threshold:
                return -1

    return result

```
