def maxDistance(s: str, k: int) -> int:
    # Initialize the starting position at the origin
    x, y = 0, 0
    # Track the maximum distance from the origin
    max_distance = 0
    # Count the number of moves in each direction
    count_n = count_s = count_e = count_w = 0

    for char in s:
        if char == 'N':
            y += 1
            count_n += 1
        elif char == 'S':
            y -= 1
            count_s += 1
        elif char == 'E':
            x += 1
            count_e += 1
        elif char == 'W':
            x -= 1
            count_w += 1
        # Calculate the current distance
        current_distance = abs(x) + abs(y)
        # Update the maximum distance
        max_distance = max(max_distance, current_distance)

    # Total moves made
    total_moves = len(s)
    # Calculate the maximum possible distance after changing k characters
    max_possible_distance = max_distance + k
    return max_possible_distance
```
