from solution import maxDistance

def test_canonical_example_1():
    # Example: input "NWSE", k = 1
    # Change s[2] from 'S' to 'N', the string becomes "NWNE".
    # The maximum Manhattan distance from the origin is 3.
    assert maxDistance("NWSE", 1) == 3

def test_canonical_example_2():
    # Example: input "NSWWEW", k = 3
    # Change s[1] from 'S' to 'N', and s[4] from 'E' to 'W'.
    # The string becomes "NNWWWW".
    # The maximum Manhattan distance from the origin is 6.
    assert maxDistance("NSWWEW", 3) == 6
