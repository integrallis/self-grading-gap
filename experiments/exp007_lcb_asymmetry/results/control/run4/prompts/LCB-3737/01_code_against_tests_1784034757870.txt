```python
from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # dp[i][j] will hold the minimum cost to paint up to house i with color j
    dp = [[0] * 3 for _ in range(n)]

    # Initialize the first house's costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the dp table
    for i in range(1, n):
        for j in range(3):
            # Get the minimum cost for the previous house with a different color
            dp[i][j] = cost[i][j] + min(dp[i - 1][(j + 1) % 3], dp[i - 1][(j + 2) % 3])

    # Calculate the minimum cost considering the equidistant condition
    # We need to check pairs (0, n-1), (1, n-2), ..., (n/2 - 1, n/2)
    min_cost = float('inf')

    for j in range(3):
        for k in range(3):
            if j != k:  # Check the color condition for equidistant houses
                total_cost = dp[n - 1][j] + dp[n - 2][k]
                min_cost = min(min_cost, total_cost)

    return min_cost
```
