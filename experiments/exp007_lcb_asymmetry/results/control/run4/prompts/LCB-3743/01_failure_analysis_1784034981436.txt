```python
from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    unavailable_times = []

    # Gather all unavailable time intervals from meetings
    for i in range(n):
        unavailable_times.append((startTime[i], endTime[i]))

    # Add event edges as unavailable times
    unavailable_times.append((0, 0))  # Before event starts
    unavailable_times.append((eventTime, eventTime))  # After event ends

    # Sort the unavailable times
    unavailable_times.sort() 

    free_time_intervals = []

    # Calculate free time intervals based on unavailable times
    for i in range(1, len(unavailable_times)):
        start = unavailable_times[i-1][1]
        end = unavailable_times[i][0]
        if start < end:
            free_time_intervals.append(end - start)

    # If there are no free time intervals, return 0
    if not free_time_intervals:
        return 0

    # Calculate maximum free time after moving k meetings
    max_free = 0
    current_time = 0

    for i in range(len(free_time_intervals)):
        if i < k:
            current_time += free_time_intervals[i]
        else:
            current_time += free_time_intervals[i] - (free_time_intervals[i - k] if i >= k else 0)
        max_free = max(max_free, current_time)

    return max_free
```
