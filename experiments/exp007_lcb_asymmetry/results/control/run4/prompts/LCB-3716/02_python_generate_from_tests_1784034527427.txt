from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n < 2:
        return n

    # dp[i] will store the length of the longest subsequence ending with nums[i]
    dp = [1] * n

    # Iterate through the array to fill dp array
    for i in range(n):
        for j in range(i):
            # Calculate the absolute difference
            diff1 = abs(nums[i] - nums[j])
            # Check if we can extend the sequence from nums[j] to nums[i]
            if dp[j] > 0:
                diff2 = abs(nums[j] - nums[j - 1]) if j > 0 else float('inf')
                # We can consider nums[j] if it maintains the non-increasing condition
                if diff1 >= diff2:
                    dp[i] = max(dp[i], dp[j] + 1)

    return max(dp)
```
