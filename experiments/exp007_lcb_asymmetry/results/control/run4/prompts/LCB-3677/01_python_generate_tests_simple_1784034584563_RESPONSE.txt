from typing import List
from solution import maximumAmount

def test_canonical_example_1():
    # spec: input [[0, 1, -1], [1, -2, 3], [2, -3, 4]]
    # Explanation:
    # Start at (0, 0) with 0 coins.
    # Move to (0, 1) -> total coins = 0 + 1 = 1.
    # Move to (1, 1) -> total coins = 1 - 2 = -1. Use neutralization here -> total = 1.
    # Move to (1, 2) -> total coins = 1 + 3 = 4.
    # Move to (2, 2) -> total coins = 4 + 4 = 8.
    assert maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) == 8

def test_canonical_example_2():
    # spec: input [[10, 10, 10], [10, 10, 10]]
    # Explanation:
    # Start at (0, 0) with 10 coins.
    # Move to (0, 1) -> total coins = 10 + 10 = 20.
    # Move to (0, 2) -> total coins = 20 + 10 = 30.
    # Move to (1, 2) -> total coins = 30 + 10 = 40.
    assert maximumAmount([[10, 10, 10], [10, 10, 10]]) == 40
