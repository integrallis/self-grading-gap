```python
from typing import List

def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a list to represent the amount of coins at each position
    max_position = 10**9
    coin_bags = [0] * (max_position + 1)

    # Fill the coin bags based on the input segments
    for l, r, c in coins:
        for i in range(l, r + 1):
            coin_bags[i] += c

    # Use a sliding window to find the maximum coins in k consecutive bags
    current_sum = sum(coin_bags[1:k+1])  # Initial sum for the first k bags
    max_coins = current_sum

    for i in range(1, max_position - k + 1):
        current_sum = current_sum - coin_bags[i] + coin_bags[i + k]
        max_coins = max(max_coins, current_sum)

    return max_coins
```
