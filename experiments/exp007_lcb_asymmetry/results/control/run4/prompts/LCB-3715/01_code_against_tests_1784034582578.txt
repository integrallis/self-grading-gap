```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a dictionary to represent the amount of coins at each position
    coin_map = {}

    # Fill the coin map based on the input segments
    for l, r, c in coins:
        for i in range(l, r + 1):
            if i in coin_map:
                coin_map[i] += c
            else:
                coin_map[i] = c

    # Extract the positions and corresponding coins
    positions = sorted(coin_map.keys())
    max_coins = 0

    # Sliding window approach to find maximum coins in k consecutive bags
    current_sum = 0
    left_index = 0

    for right_index in range(len(positions)):
        current_sum += coin_map[positions[right_index]]

        # Ensure we have k bags in the window
        while positions[right_index] - positions[left_index] + 1 > k:
            current_sum -= coin_map[positions[left_index]]
            left_index += 1

        # Update maximum coins found
        if positions[right_index] - positions[left_index] + 1 == k:
            max_coins = max(max_coins, current_sum)

    return max_coins
```
