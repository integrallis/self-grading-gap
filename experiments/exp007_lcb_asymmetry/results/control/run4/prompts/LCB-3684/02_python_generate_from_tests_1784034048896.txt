'''The solution package.'''

__version__ = "0.1.0"

def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]
    for i in range(len(s) - len(prefix) - len(suffix) + 1):
        if s[i:i + len(prefix)] == prefix:
            if s[i + len(prefix):].startswith(suffix):
                return True
    return False
```
