from typing import List
import math

def lcm(a: int, b: int) -> int:
    return abs(a * b) // math.gcd(a, b)

def maxLength(nums: List[int]) -> int:
    n = len(nums)
    max_length = 0

    for start in range(n):
        current_gcd = nums[start]
        current_lcm = nums[start]
        product = nums[start]
        length = 1

        for end in range(start + 1, n):
            current_gcd = math.gcd(current_gcd, nums[end])
            current_lcm = lcm(current_lcm, nums[end])
            product *= nums[end]
            length += 1

            # Check if product is equivalent
            if product == current_gcd * current_lcm:
                max_length = max(max_length, length)

gcd, lcm, product, length = current_gcd, current_lcm, product, length

    return max_length
