from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    for i in range(1, k + 1):
        # Each number contributes to the subsequences in which it is max and min
        for num in nums:
            num_contribution = num * comb(n - 1, i - 1) % MOD
            total_sum += num_contribution * (pow(2, n - i, MOD)) % MOD
            total_sum %= MOD
    total_sum = (total_sum * 2) % MOD  # Each number contributes as both min and max
    return total_sum


def comb(n: int, k: int) -> int:
    if k > n:
        return 0
    if k == 0 or k == n:
        return 1
    k = min(k, n - k)  # Take advantage of symmetry
    c = 1
    for i in range(k):
        c = c * (n - i) % MOD
        c = c * pow(i + 1, MOD - 2, MOD) % MOD  # Modular inverse
    return c

```
