from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    # Iterate through each element in nums
    for i in range(n):
        min_contribution = nums[i] * (pow(2, min(k, n - 1), MOD) - 1) % MOD * pow(2, n - 1 - min(k, n - 1), MOD) % MOD
        max_contribution = nums[i] * (pow(2, min(k, n - 1), MOD) - 1) % MOD * pow(2, n - 1 - min(k, n - 1), MOD) % MOD
        total_sum = (total_sum + min_contribution + max_contribution) % MOD
    return total_sum

```
