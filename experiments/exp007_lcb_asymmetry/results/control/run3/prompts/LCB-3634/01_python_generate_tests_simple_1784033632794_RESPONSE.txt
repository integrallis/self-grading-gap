from solution import calculateScore

def test_canonical_example_1():
    # spec: calculateScore("aczzx") should return 5
    # Explanation:
    # i = 0. No j satisfies condition, skip.
    # i = 1. No j satisfies condition, skip.
    # i = 2. j = 0 (mirror of 'a' is 'z'), score += 2 - 0 = 2.
    # i = 3. No j satisfies condition, skip.
    # i = 4. j = 1 (mirror of 'b' is 'y'), score += 4 - 1 = 3.
    # Total score = 2 + 3 = 5
    assert calculateScore("aczzx") == 5

def test_canonical_example_2():
    # spec: calculateScore("abcdef") should return 0
    # Explanation:
    # For each index i, there is no index j that satisfies the conditions.
    assert calculateScore("abcdef") == 0
