from typing import List

def zigzagTraversal(grid: List[List[int]]) -> List[int]:
    result = []
    for i in range(len(grid)):
        if i % 2 == 0:
            # Traverse right
            for j in range(len(grid[i])):
                if j % 2 == 0:
                    result.append(grid[i][j])
        else:
            # Traverse left
            for j in range(len(grid[i])):
                if (len(grid[i]) - j - 1) % 2 == 0:
                    result.append(grid[i][len(grid[i]) - j - 1])
    return result
```
