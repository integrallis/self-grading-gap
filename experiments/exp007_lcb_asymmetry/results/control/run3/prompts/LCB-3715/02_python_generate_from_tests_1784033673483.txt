from typing import List

def maximumCoins(coins: List[List[int]], k: int) -> int:
    max_coins = 0
    current_coins = 0
    bag_values = {}

    # Fill the bag_values dictionary with the coins in each bag position
    for l, r, c in coins:
        for position in range(l, r + 1):
            bag_values[position] = bag_values.get(position, 0) + c

    # Create a sorted list of all unique bag positions
    sorted_positions = sorted(bag_values.keys())
    n = len(sorted_positions)

    # Use a sliding window approach to find the maximum coins in k consecutive bags
    left = 0
    for right in range(n):
        current_coins += bag_values[sorted_positions[right]]

        # If we have more than k bags, slide the window to the right
        if right - left + 1 > k:
            current_coins -= bag_values[sorted_positions[left]]
            left += 1

        # Update max_coins if we have exactly k bags
        if right - left + 1 == k:
            max_coins = max(max_coins, current_coins)

    return max_coins
```
