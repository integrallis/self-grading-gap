def test_canonical_example_1():
    # spec: hasSpecialSubstring("aaabaaa", 3) -> True 
    # Explanation: substring "aaa" at index 4-6 has length 3, 
    # consists of one distinct character 'a', and is surrounded by 'b' (before) 
    # and no character (after).
    assert hasSpecialSubstring("aaabaaa", 3) is True

def test_canonical_example_2():
    # spec: hasSpecialSubstring("abc", 2) -> False 
    # Explanation: no substring of length 2 exists that consists of one distinct character.
