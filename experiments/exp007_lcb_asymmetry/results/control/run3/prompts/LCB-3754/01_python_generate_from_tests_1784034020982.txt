def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0

    for char in s:
        if char == 'N':
            y += 1
        elif char == 'S':
            y -= 1
        elif char == 'E':
            x += 1
        elif char == 'W':
            x -= 1

        # Current Manhattan distance from the origin
        current_distance = abs(x) + abs(y)
        max_distance = max(max_distance, current_distance)

    # After all movements, apply at most k changes
    # Maximum distance can be increased by 2 * k
    return max_distance + 2 * k
```
