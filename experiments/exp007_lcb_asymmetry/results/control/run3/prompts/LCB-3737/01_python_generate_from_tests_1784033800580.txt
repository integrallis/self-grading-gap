from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    dp = [[0] * 3 for _ in range(n)]
    dp[0] = cost[0]

    for i in range(1, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(
                dp[i - 1][(j + 1) % 3],  # previous house, different color
                dp[i - 1][(j + 2) % 3]   # previous house, different color
            )

    for i in range(n // 2):
        for j in range(3):
            dp[i + n // 2][j] = float('inf')
            if j != (i + n // 2) % 3:
                dp[i + n // 2][j] = min(dp[i + n // 2][j], 
                                          cost[i + n // 2][j] + 
                                          min(dp[i][(j + 1) % 3], 
                                              dp[i][(j + 2) % 3]))

    return min(dp[n - 1])

```
