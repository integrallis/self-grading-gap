from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)

    # Maximum free time considering the boundaries
    max_free_time = max(0, startTime[0])

    # This will store potential additional free time intervals from gaps
    potential_gaps = []

    # Iterate through meetings to calculate the gaps
    for i in range(1, n):
        gap = startTime[i] - endTime[i - 1]
        if gap > 0:
            potential_gaps.append(gap)

    # Determine free time after the last meeting
    max_free_time = max(max_free_time, eventTime - endTime[-1])  # After the last meeting

    # We sort the potential gaps in descending order
    potential_gaps.sort(reverse=True)

    # If we can reschedule up to k meetings, we can maximize gaps
    shifts_gained = sum(potential_gaps[:k]) if k <= len(potential_gaps) else sum(potential_gaps)

    # The final free time is the maximum free time considered with the potential shifts
    return max_free_time + shifts_gained
