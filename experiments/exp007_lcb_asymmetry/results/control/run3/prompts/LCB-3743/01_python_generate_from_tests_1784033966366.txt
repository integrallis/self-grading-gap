from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_intervals = []

    # Collect free time intervals before, between, and after meetings
    if startTime[0] > 0:
        free_intervals.append((0, startTime[0]))
    for i in range(1, n):
        if startTime[i] > endTime[i - 1]:
            free_intervals.append((endTime[i - 1], startTime[i]))
    if endTime[-1] < eventTime:
        free_intervals.append((endTime[-1], eventTime))

    # Calculate total free time without any rescheduling
    total_free_time = 0
    for start, end in free_intervals:
        total_free_time += end - start

    # Calculate gaps between meetings
    gaps = []
    for i in range(n - 1):
        gap = startTime[i + 1] - endTime[i]
        if gap > 0:
            gaps.append(gap)

    # Calculate maximum free time by rescheduling up to k meetings
    gaps.sort(reverse=True)  # Sort decreasing to maximize free time
    max_gain = sum(gaps[:k]) if k <= len(gaps) else sum(gaps)
    return total_free_time + max_gain
```
