## Implementation

```python
from typing import List

def maxSum(nums: List[int]) -> int:
    unique_nums = set(nums)  # Use a set to store unique elements
    # Filter out negative values and sum only non-negative unique values
    positive_sum = sum(x for x in unique_nums if x > 0)
    # If the positive sum is 0, return the maximum value from the original list
    if positive_sum > 0:
        return positive_sum
    else:
        return max(unique_nums) if unique_nums else 0
```
