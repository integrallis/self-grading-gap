from solution import maxSum


def test_canonical_example_1():
    # All values are unique, so 1 + 2 + 3 + 4 + 5 = 15.
    assert maxSum([1, 2, 3, 4, 5]) == 15


def test_canonical_example_2():
    # Keeping one unique 1 gives the greatest sum: 1.
    assert maxSum([1, 1, 0, 1, 1]) == 1


def test_canonical_example_3():
    # Delete negatives and one repeated 1; the unique subarray [2, 1] sums to 3.
    assert maxSum([1, 2, -1, -2, 1, 0, -1]) == 3


def test_counts_each_positive_value_at_most_once():
    # The best unique selection contains 5 and 4 once each: 5 + 4 = 9.
    assert maxSum([5, -100, 5, 4, 4]) == 9


def test_zero_is_best_when_all_other_values_are_negative():
    # A nonempty unique subarray containing 0 has sum 0, greater than either negative value.
    assert maxSum([-3, 0, -1, -3]) == 0


def test_returns_largest_value_when_all_values_are_negative():
    # The required nonempty selection with greatest sum is [-2], so the sum is -2.
    assert maxSum([-7, -2, -5, -2]) == -2
