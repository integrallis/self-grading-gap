```python
from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    gaps = []

    # Calculate initial gaps between meetings
    previous_end = 0
    for i in range(n):
        if startTime[i] > previous_end:
            gaps.append(startTime[i] - previous_end)
        previous_end = endTime[i]

    # Add gap after last meeting if there's time left in the event
    if previous_end < eventTime:
        gaps.append(eventTime - previous_end)

    # If k is greater than or equal to the number of gaps, we can combine all gaps
    if k >= len(gaps):
        return sum(gaps)

    # Sort gaps in reverse order (largest to smallest)
    gaps.sort(reverse=True)
    # Sum the largest (k + 1) gaps, which we can make contiguous
    max_free_time = sum(gaps[:k + 1])

    return max_free_time
```
