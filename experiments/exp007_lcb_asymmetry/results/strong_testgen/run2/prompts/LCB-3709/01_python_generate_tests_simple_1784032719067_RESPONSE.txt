from solution import hasSpecialSubstring


def test_canonical_example_1():
    # "aaa" at the end has length 3 and is preceded by different character "b".
    assert hasSpecialSubstring("aaabaaa", 3) is True


def test_canonical_example_2():
    # No length-2 substring of "abc" contains only one distinct character.
    assert hasSpecialSubstring("abc", 2) is False


def test_entire_string_is_a_special_substring():
    # The whole string "aaaa" has length 4, one distinct character, and no neighbors.
    assert hasSpecialSubstring("aaaa", 4) is True


def test_run_longer_than_k_is_not_a_special_substring():
    # "aaaa" contains length-3 uniform substrings, but each has an adjacent "a".
    assert hasSpecialSubstring("aaaa", 3) is False


def test_isolated_internal_run_of_exact_length_is_special():
    # "bbb" has length 3 and is bounded by different characters "a" and "c".
    assert hasSpecialSubstring("abbbca", 3) is True


def test_single_character_k_requires_an_isolated_character():
    # "b" is a length-1 uniform substring bounded by different characters "a" and "c".
    assert hasSpecialSubstring("abc", 1) is True


def test_single_character_inside_a_run_is_not_special():
    # Each length-1 "a" has an adjacent "a", so none is special.
    assert hasSpecialSubstring("aa", 1) is False


def test_no_uniform_substring_of_requested_length():
    # Every length-2 substring of "ababa" has two distinct characters.
    assert hasSpecialSubstring("ababa", 2) is False
