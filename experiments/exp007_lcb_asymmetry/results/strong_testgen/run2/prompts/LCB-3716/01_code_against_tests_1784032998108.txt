```python
from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n == 0:
        return 0

    # dp[i] will hold the length of the longest subsequence ending at index i
    dp = [1] * n
    max_length = 1

    # Iterate through each pair of elements to find valid subsequences
    for i in range(n):
        for j in range(i):
            diff1 = abs(nums[i] - nums[j])
            # Only consider the next element if the difference is valid
            if dp[j] + 1 > dp[i] and (j == 0 or abs(nums[j] - nums[j-1]) >= diff1):
                dp[i] = dp[j] + 1
            max_length = max(max_length, dp[i])

    return max_length

```
