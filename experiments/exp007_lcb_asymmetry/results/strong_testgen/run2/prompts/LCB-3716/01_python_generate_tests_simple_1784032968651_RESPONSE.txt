from solution import longestSubsequence


def test_canonical_example_1():
    # [16, 6, 3] has adjacent absolute differences [10, 3], and 10 >= 3.
    assert longestSubsequence([16, 6, 3]) == 3


def test_canonical_example_2():
    # [6, 4, 2, 1] is a subsequence with differences [2, 2, 1],
    # and 2 >= 2 >= 1; no valid subsequence has length 5 or 6.
    assert longestSubsequence([6, 5, 3, 4, 2, 1]) == 4


def test_canonical_example_3():
    # [10, 20, 10, 19, 10] has differences [10, 10, 9, 9],
    # and 10 >= 10 >= 9 >= 9; no valid subsequence has length 6.
    assert longestSubsequence([10, 20, 10, 19, 10, 20]) == 5


def test_equal_adjacent_values_allow_zero_differences():
    # [7, 7, 7, 7] has differences [0, 0, 0], and 0 >= 0 >= 0.
    assert longestSubsequence([7, 7, 7, 7]) == 4


def test_subsequence_can_skip_elements_to_preserve_non_increasing_differences():
    # [1, 10, 11] has differences [9, 1], and 9 >= 1.
    assert longestSubsequence([1, 10, 20, 11]) == 3


def test_increasing_differences_prevent_using_all_elements():
    # Any length-3 subsequence of [1, 2, 4] has increasing differences:
    # [1, 2, 4] gives [1, 2], so the best valid length is 2.
    assert longestSubsequence([1, 2, 4]) == 2


def test_two_element_input_has_length_two_subsequence():
    # The entire array [1, 300] is valid because a one-difference sequence is non-increasing.
    assert longestSubsequence([1, 300]) == 2
