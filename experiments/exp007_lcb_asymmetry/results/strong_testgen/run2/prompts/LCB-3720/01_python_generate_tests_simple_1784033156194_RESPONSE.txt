from solution import minMaxWeight


def test_canonical_example_1():
    # Nodes 1, 2, 3, and 4 can reach 0 using only weight-1 edges:
    # 1 -> 0, 2 -> 1, 3 -> 0, and 4 -> 3.
    assert minMaxWeight(
        5,
        [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]],
        2,
    ) == 1


def test_canonical_example_2():
    # Node 2 has no directed path to node 0, so satisfying reachability is impossible.
    assert minMaxWeight(
        5,
        [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]],
        1,
    ) == -1


def test_canonical_example_3():
    # Keeping 1 -> 2 -> 3 -> 4 -> 0 reaches 0 from every node;
    # its largest edge weight is 2.
    assert minMaxWeight(
        5,
        [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]],
        1,
    ) == 2


def test_canonical_example_4():
    # Node 3 has no path to node 0, so no edge removal can satisfy reachability.
    assert minMaxWeight(
        5,
        [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]],
        1,
    ) == -1


def test_uses_heavier_edge_when_it_is_required_for_reachability():
    # Node 1 can reach 0 only through weight 7; node 2 reaches 0 via 2 -> 1.
    # Therefore the minimum possible maximum weight is 7.
    assert minMaxWeight(3, [[1, 0, 7], [2, 1, 3], [2, 0, 9]], 1) == 7


def test_selects_lightest_of_multiple_edges_between_same_nodes():
    # Keep 1 -> 0 with weight 2 and 2 -> 0 with weight 1, so the maximum is 2.
    assert minMaxWeight(3, [[1, 0, 8], [1, 0, 2], [2, 0, 1]], 1) == 2
