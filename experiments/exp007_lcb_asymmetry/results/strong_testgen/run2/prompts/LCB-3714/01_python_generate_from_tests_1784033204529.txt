from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    total_min = 0
    total_max = 0
    pow2 = [1] * (k + 1)
    for i in range(1, k + 1):
        pow2[i] = (pow2[i - 1] * 2) % MOD

    for num in nums:
        total_max = (total_max + num * (pow2[k] - 1)) % MOD
        total_min = (total_min + num * (pow2[k] - 1)) % MOD

    total_sum = (total_max + total_min) % MOD
    return total_sum

```
