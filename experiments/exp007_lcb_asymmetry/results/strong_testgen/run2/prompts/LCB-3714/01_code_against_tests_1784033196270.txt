## Implementation

```python
from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0
    for i in range(1, k + 1):
        # Calculate the contribution of each element as max and min
        total_sum += (sum(nums) * (1 << (i - 1))) % MOD
        total_sum %= MOD
        for num in nums:
            total_sum += (num * (1 << (n - i))) % MOD
            total_sum %= MOD
    return total_sum % MOD

```
