from solution import minMaxSums


def test_canonical_example_1():
    # Singles contribute 2 + 4 + 6 = 12; pairs contribute 3 + 4 + 5 = 12; total = 24.
    assert minMaxSums([1, 2, 3], 2) == 24


def test_canonical_example_2():
    # With k = 1, each singleton contributes twice its value: 2*5 + 2*0 + 2*6 = 22.
    assert minMaxSums([5, 0, 6], 1) == 22


def test_canonical_example_3():
    # Three singletons contribute 3*(1 + 1) = 6; three pairs contribute 3*(1 + 1) = 6; total = 12.
    assert minMaxSums([1, 1, 1], 2) == 12
