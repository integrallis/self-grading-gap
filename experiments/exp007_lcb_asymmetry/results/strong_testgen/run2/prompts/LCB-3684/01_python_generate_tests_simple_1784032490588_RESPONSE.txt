from solution import hasMatch


def test_canonical_example_1():
    # "ee*e" matches substring "eetcode" when '*' is replaced by "tcod".
    assert hasMatch("leetcode", "ee*e") is True


def test_canonical_example_2():
    # No substring of "car" starts with "c" and later ends with "v".
    assert hasMatch("car", "c*v") is False


def test_canonical_example_3():
    # "u*" matches substring "u" when '*' is replaced by an empty sequence.
    assert hasMatch("luck", "u*") is True


def test_star_can_match_zero_characters_between_literal_parts():
    # "a*c" matches contiguous substring "ac" when '*' is replaced by an empty sequence.
    assert hasMatch("xxacxx", "a*c") is True


def test_star_can_match_characters_before_a_suffix():
    # "*ab" matches substring "cab" when '*' is replaced by "c".
    assert hasMatch("zzcabzz", "*ab") is True


def test_literal_parts_must_appear_in_pattern_order():
    # "ab*cd" cannot match because "ab" occurs after "cd" in "cdab".
    assert hasMatch("cdab", "ab*cd") is False


def test_pattern_containing_only_star_matches_a_nonempty_substring():
    # "*" matches substring "a" when '*' is replaced by "a".
    assert hasMatch("abc", "*") is True
