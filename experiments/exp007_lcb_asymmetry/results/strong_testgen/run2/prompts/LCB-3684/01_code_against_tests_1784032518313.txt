## Implementation

```python
def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]
    # Check for all possible starting positions where prefix can match
    for i in range(len(s) - len(prefix) + 1):
        if s[i:i + len(prefix)] == prefix:
            # Check if there is enough length left for suffix
            if i + len(prefix) + len(suffix) <= len(s):
                if s[i + len(prefix):i + len(prefix) + len(suffix)] == suffix:
                    return True
    # Now check if the suffix can match at the end of the string
    for i in range(len(s) - len(suffix), -1, -1):
        if s[i:i + len(suffix)] == suffix:
            # Check if there is enough length for prefix
            if i - len(prefix) >= 0:
                if s[i - len(prefix):i] == prefix:
                    return True
    return False
```
