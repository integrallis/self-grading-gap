from solution import maxFrequency


def test_canonical_example_1():
    # Adding -5 to nums[2..5] converts the final 6 to 1, so 1 appears twice.
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2


def test_canonical_example_2():
    # Adding 8 to nums[1..9] converts the three 2s to 10; together with the
    # original 10, the frequency of 10 is 4.
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4


def test_single_element_can_be_converted_to_k():
    # Add -2 to the only element, changing 7 to 5, so 5 appears once.
    assert maxFrequency([7], 5) == 1


def test_existing_k_values_can_be_left_unchanged():
    # Choose x = 0 for the single-element subarray; all three existing 4s remain.
    assert maxFrequency([4, 4, 4], 4) == 3


def test_one_added_value_cannot_convert_different_values_together():
    # A single x can convert either both 2s or both 3s to 1, but not both values,
    # so the maximum frequency is 2.
    assert maxFrequency([2, 3, 2, 3], 1) == 2


def test_k_values_inside_a_conversion_subarray_are_lost():
    # Converting all three 2s to 1 over the full array also changes two existing
    # 1s, leaving 3 occurrences of 1.
    assert maxFrequency([2, 1, 2, 1, 2], 1) == 3
