from solution import minCost


def test_canonical_example_1():
    cost = [[3, 5, 7], [6, 2, 9], [4, 8, 1], [7, 3, 5]]

    # Optimal colors [1, 2, 3, 2] cost 3 + 2 + 1 + 3 = 9.
    assert minCost(4, cost) == 9


def test_canonical_example_2():
    cost = [[2, 4, 6], [5, 3, 8], [7, 1, 9], [4, 6, 2], [3, 5, 7], [8, 2, 4]]

    # Optimal colors [1, 3, 2, 3, 1, 2] cost 2 + 8 + 1 + 2 + 3 + 2 = 18.
    assert minCost(6, cost) == 18


def test_two_houses_must_use_different_colors():
    cost = [[0, 9, 9], [0, 5, 8]]

    # The two houses are adjacent and equidistant, so their colors differ;
    # colors 1 and 2 cost 0 + 5 = 5, the cheapest such choice.
    assert minCost(2, cost) == 5


def test_mirrored_pairs_can_force_costlier_choices():
    cost = [[0, 100, 100], [100, 0, 100], [100, 0, 100], [0, 100, 100]]

    # Each preferred mirrored pair uses the same color, so each pair requires
    # one 100-cost change; colors [1, 2, 1, 2] cost 0 + 0 + 100 + 100 = 200.
    assert minCost(4, cost) == 200


def test_zero_cost_valid_painting():
    cost = [[0, 4, 4], [4, 0, 4], [4, 4, 0], [4, 0, 4]]

    # Colors [1, 2, 3, 2] satisfy both rules and cost 0 + 0 + 0 + 0 = 0.
    assert minCost(4, cost) == 0
