from solution import maxLength


def test_canonical_example_1():
    # [1, 2, 1, 1, 1] has product 2, gcd 1, and lcm 2, so 2 == 1 * 2.
    assert maxLength([1, 2, 1, 2, 1, 1, 1]) == 5


def test_canonical_example_2():
    # [3, 4, 5] has product 60, gcd 1, and lcm 60, so 60 == 1 * 60.
    assert maxLength([2, 3, 4, 5, 6]) == 3


def test_canonical_example_3():
    # [3, 1, 4, 5, 1] has product 60, gcd 1, and lcm 60, so 60 == 1 * 60.
    assert maxLength([1, 2, 3, 1, 4, 5, 1]) == 5


def test_entire_all_ones_array_is_product_equivalent():
    # [1, 1, 1, 1] has product 1, gcd 1, and lcm 1, so 1 == 1 * 1.
    assert maxLength([1, 1, 1, 1]) == 4


def test_repeated_non_one_values_can_limit_the_longest_subarray():
    # Any two adjacent 2s have product 4, gcd 2, and lcm 2, so 4 == 2 * 2;
    # all three have product 8 while gcd * lcm is 2 * 2 = 4.
    assert maxLength([2, 2, 2]) == 2
