from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a map to store the number of coins at each position
    coin_map = {}

    # Fill in the coin_map based on the provided segments
    for l, r, c in coins:
        for position in range(l, r + 1):
            coin_map[position] = coin_map.get(position, 0) + c

    # Sort the positions to create a list of bags
    positions = sorted(coin_map.keys())
    n = len(positions)

    # Prepare a prefix sum array to calculate the total coins efficiently
    prefix_sum = [0] * (n + 1)
    for i in range(n):
        prefix_sum[i + 1] = prefix_sum[i] + coin_map[positions[i]]

    max_coins = 0

    # Slide a window of size k across the positions
    for i in range(n):
        # Find the right-most position that is within k from the current position
        right_bound = positions[i] + k - 1
        # Use binary search to find the upper index
        j = i
        while j < n and positions[j] <= right_bound:
            j += 1

        # Calculate the total coins in the window [i, j-1]
        total_coins = prefix_sum[j] - prefix_sum[i]
        max_coins = max(max_coins, total_coins)

    return max_coins

```
