from solution import maximumCoins


def test_canonical_example_1():
    # Bags [3, 4, 5, 6] contain 2 + 0 + 4 + 4 = 10 coins.
    assert maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) == 10


def test_canonical_example_2():
    # Any two bags within [1, 10] contain 3 + 3 = 6 coins.
    assert maximumCoins([[1, 10, 3]], 2) == 6


def test_window_can_include_a_gap_between_segments():
    # Bags [2, 3, 4, 5] contain 5 + 0 + 0 + 5 = 10 coins.
    assert maximumCoins([[1, 2, 5], [5, 6, 5]], 4) == 10


def test_window_can_take_part_of_a_segment_at_its_boundary():
    # Bags [2, 3, 4] contain 4 + 4 + 7 = 15 coins.
    assert maximumCoins([[1, 3, 4], [4, 5, 7]], 3) == 15


def test_unsorted_non_overlapping_segments():
    # Bags [10, 11, 12] contain 2 + 2 + 2 = 6 coins.
    assert maximumCoins([[10, 12, 2], [1, 2, 1]], 3) == 6


def test_k_larger_than_all_coin_containing_span():
    # A window containing both segments collects 2 + 2 + 3 = 7 coins.
    assert maximumCoins([[1, 2, 1], [5, 5, 3]], 10) == 7


def test_large_coordinates_and_coin_total():
    # Bags [1_000_000_000 - 1, 1_000_000_000] contain 1000 + 1000 = 2000 coins.
    assert maximumCoins([[999_999_999, 1_000_000_000, 1000]], 2) == 2000
