from solution import minimumPairRemoval


def test_canonical_example_1():
    # Minimum pair sums: (3, 1) -> 4 gives [5, 2, 4];
    # then (2, 4) -> 6 gives [5, 6], so 2 operations.
    assert minimumPairRemoval([5, 2, 3, 1]) == 2


def test_canonical_example_2():
    # [1, 2, 2] is already non-decreasing, so no operations are needed.
    assert minimumPairRemoval([1, 2, 2]) == 0


def test_single_element_is_already_non_decreasing():
    # A one-element array has no previous element to violate ordering, so 0 operations.
    assert minimumPairRemoval([7]) == 0


def test_merges_the_adjacent_pair_with_the_smallest_sum():
    # Pair sums are 5 for (3, 2) and 3 for (2, 1).
    # Merge (2, 1) -> 3, producing [3, 3], so 1 operation.
    assert minimumPairRemoval([3, 2, 1]) == 1


def test_chooses_leftmost_pair_when_minimum_sums_tie():
    # Pair sums are 3 for (5, -2), 3 for (-2, 5), and 10 for (5, 5).
    # The leftmost minimum pair merges to 3, producing [3, 5, 5], so 1 operation.
    assert minimumPairRemoval([5, -2, 5, 5]) == 1
