```python
from typing import List


def minimumPairRemoval(nums: List[int]) -> int:
    operations = 0
    while True:
        # Find the minimum adjacent pair
        min_sum = float('inf')
        min_index = -1
        for i in range(len(nums) - 1):
            pair_sum = nums[i] + nums[i + 1]
            if pair_sum < min_sum:
                min_sum = pair_sum
                min_index = i
        # If no pair was found, the array is non-decreasing
        if min_index == -1:
            break
        # Merge the found pair
        nums[min_index] += nums[min_index + 1]
        del nums[min_index + 1]
        operations += 1
    return operations
```
