from solution import countMentions


def test_canonical_example_1():
    events = [
        ["MESSAGE", "10", "id1 id0"],
        ["OFFLINE", "11", "0"],
        ["MESSAGE", "71", "HERE"],
    ]

    # At 10 both users are mentioned; at 71 user 0 has returned, so HERE mentions both.
    assert countMentions(2, events) == [2, 2]


def test_canonical_example_2():
    events = [
        ["MESSAGE", "10", "id1 id0"],
        ["OFFLINE", "11", "0"],
        ["MESSAGE", "12", "ALL"],
    ]

    # The first message mentions both users, and ALL at 12 mentions both including offline user 0.
    assert countMentions(2, events) == [2, 2]


def test_canonical_example_3():
    events = [
        ["OFFLINE", "10", "0"],
        ["MESSAGE", "12", "HERE"],
    ]

    # User 0 remains offline at 12, so HERE mentions only online user 1.
    assert countMentions(2, events) == [0, 1]


def test_explicit_id_mentions_count_duplicates_and_include_offline_users():
    events = [
        ["OFFLINE", "1", "1"],
        ["MESSAGE", "2", "id1 id1 id2"],
    ]

    # id1 is offline but is explicitly mentioned twice; id2 is mentioned once; id0 is not mentioned.
    assert countMentions(3, events) == [0, 2, 1]


def test_here_mentions_only_users_online_at_that_timestamp():
    events = [
        ["OFFLINE", "5", "0"],
        ["OFFLINE", "6", "2"],
        ["MESSAGE", "10", "HERE"],
        ["MESSAGE", "65", "HERE"],
    ]

    # At 10 only id1 is online; at 65 id0 has returned while id2 is still offline.
    assert countMentions(3, events) == [1, 2, 0]


def test_status_change_at_same_timestamp_is_processed_before_message():
    events = [
        ["MESSAGE", "10", "HERE"],
        ["OFFLINE", "10", "0"],
    ]

    # The offline change at 10 is processed before HERE at 10, so only user 1 is mentioned.
    assert countMentions(2, events) == [0, 1]
