```python
from typing import List

def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers
    online_status = [True] * numberOfUsers
    event_queue = []

    for event in events:
        event_queue.append((int(event[1]), event[0], event[2]))

    event_queue.sort()  # Sort events by timestamp

    for timestamp, event_type, data in event_queue:
        # Process any online events triggered by the OFFLINE event
        for i in range(len(event_queue)):
            if event_queue[i][0] <= timestamp:
                if event_queue[i][1] == 'OFFLINE':
                    user_id = int(event_queue[i][2])
                    online_status[user_id] = False
                elif event_queue[i][1] == 'ONLINE':
                    user_id = event_queue[i][2]
                    online_status[user_id] = True

        if event_type == 'MESSAGE':
            if data == 'HERE':
                mentioned_users = [i for i in range(numberOfUsers) if online_status[i]]
            elif data == 'ALL':
                mentioned_users = range(numberOfUsers)
            else:
                mentioned_users = []
                ids = data.split()
                for user_id in ids:
                    if user_id.startswith('id'):
                        mentioned_users.append(int(user_id[2:]))

            for user_id in mentioned_users:
                mentions[user_id] += 1

    return mentions

```
