from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[-float('inf')] * (n + 1) for _ in range(m + 1)]
    dp[0][1] = 0

    for i in range(m):
        for j in range(n):
            current_coins = coins[i][j]
            for k in range(3):  # k represents the number of neutralizations used
                if current_coins < 0 and k == 0:
                    continue  # Cannot neutralize if no neutralizations left
                # Calculate the gain if we are at (i, j)
                if j + 1 < n:
                    # Move right
                    dp[i][j + 1] = max(dp[i][j + 1], dp[i][j] + (current_coins if current_coins >= 0 else 0))
                if i + 1 < m:
                    # Move down
                    dp[i + 1][j] = max(dp[i + 1][j], dp[i][j] + (current_coins if current_coins >= 0 else 0))
                # Neutralize the robber
                if current_coins < 0:
                    dp[i][j] = max(dp[i][j], dp[i][j] - current_coins)

    return max(dp[m-1][n-1], -float('inf'))
```
