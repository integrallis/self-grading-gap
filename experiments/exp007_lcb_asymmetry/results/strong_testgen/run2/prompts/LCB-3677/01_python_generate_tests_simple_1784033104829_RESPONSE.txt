from solution import maximumAmount


def test_canonical_example_1():
    # Path 0 -> 1 -> neutralize -2 -> 3 -> 4 collects 0 + 1 + 0 + 3 + 4 = 8.
    assert maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) == 8


def test_canonical_example_2():
    # A path contains four cells worth 10 coins each: 10 + 10 + 10 + 10 = 40.
    assert maximumAmount([[10, 10, 10], [10, 10, 10]]) == 40


def test_can_neutralize_robbers_at_start_and_destination():
    # Right then down: neutralize -5 and -7, collecting 0 + 4 + 0 = 4.
    assert maximumAmount([[-5, 4], [3, -7]]) == 4


def test_at_most_two_neutralizations_can_leave_a_negative_profit():
    # The only path totals -1 - 5 - 2 - 4 = -12; neutralizing -5 and -4 gives -3.
    assert maximumAmount([[-1, -5, -2, -4]]) == -3


def test_path_choice_accounts_for_two_neutralizations():
    # Across the top then down: 0 + 0 + 0 + 100 + 10 = 110 after neutralizing both -100 cells.
    assert maximumAmount([[0, -100, -100, 100], [10, 10, 10, 10]]) == 110
