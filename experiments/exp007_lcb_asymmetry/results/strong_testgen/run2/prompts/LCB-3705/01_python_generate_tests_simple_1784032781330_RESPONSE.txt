from solution import largestInteger


def test_canonical_example_1():
    # Size-3 windows contain 3 only in the first and 7 only in the last; max(3, 7) = 7.
    assert largestInteger([3, 9, 2, 1, 7], 3) == 7


def test_canonical_example_2():
    # Of the values, only 3 occurs in exactly one size-4 window.
    assert largestInteger([3, 9, 7, 2, 1, 7], 4) == 3


def test_canonical_example_3():
    # Each 0 occurs in a size-1 window, so 0 appears in two such subarrays and none qualifies.
    assert largestInteger([0, 0], 1) == -1


def test_k_equals_array_length_counts_each_value_in_the_single_subarray_once():
    # There is one size-4 subarray containing every value; largest distinct value is 50.
    assert largestInteger([1, 50, 50, 2], 4) == 50


def test_k_one_returns_largest_value_that_occurs_once():
    # Size-1 subarrays contain 2 once, 3 twice, and 5 once; max(2, 5) = 5.
    assert largestInteger([3, 2, 5, 3], 1) == 5


def test_repeated_value_in_one_window_is_almost_missing():
    # The single size-2 subarray contains 8, so 8 appears in exactly one subarray.
    assert largestInteger([8, 8], 2) == 8
