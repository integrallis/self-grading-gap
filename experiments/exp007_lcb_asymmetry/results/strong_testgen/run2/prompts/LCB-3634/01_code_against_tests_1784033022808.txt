## Implementation

```python
def calculateScore(s: str) -> int:
    score = 0
    marked = set()
    n = len(s)
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # create mirror mapping

    for i in range(n):
        if i in marked:
            continue
        for j in range(i - 1, -1, -1):
            if j in marked:
                continue
            if s[j] == mirror[s[i]]:
                score += i - j
                marked.add(i)
                marked.add(j)
                break
    return score
```
