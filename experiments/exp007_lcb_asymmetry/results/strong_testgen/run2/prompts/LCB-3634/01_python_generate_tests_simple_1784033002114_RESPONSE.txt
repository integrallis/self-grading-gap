from solution import calculateScore


def test_canonical_example_1():
    # spec: pairs (0, 2) and (1, 4), so (2 - 0) + (4 - 1) = 5.
    assert calculateScore("aczzx") == 5


def test_canonical_example_2():
    # spec: no character has an earlier unmarked mirror, so the score is 0.
    assert calculateScore("abcdef") == 0


def test_uses_closest_unmarked_matching_index():
    # "aazz": z at 2 pairs with closest a at 1, then z at 3 pairs with a at 0:
    # (2 - 1) + (3 - 0) = 4.
    assert calculateScore("aazz") == 4


def test_marked_index_cannot_be_reused():
    # "azz": z at 1 pairs with a at 0 for 1; z at 2 has no unmarked a, so total is 1.
    assert calculateScore("azz") == 1


def test_middle_alphabet_mirror_pairing():
    # "mnnm": pairs are (0, 1) and (2, 3), so (1 - 0) + (3 - 2) = 2.
    assert calculateScore("mnnm") == 2
