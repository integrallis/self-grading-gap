from solution import maxDistance


def test_canonical_example_1():
    # Change "NWSE" to "NWNE": distances are 1, 2, 3, 2, so the maximum is 3.
    assert maxDistance("NWSE", 1) == 3


def test_canonical_example_2():
    # Change "NSWWEW" to "NNWWWW": all six moves increase distance, so maximum is 6.
    assert maxDistance("NSWWEW", 3) == 6


def test_no_changes_uses_maximum_over_all_prefixes():
    # "NS" reaches distance 1 after N and returns to distance 0 after S; maximum is 1.
    assert maxDistance("NS", 0) == 1


def test_no_changes_handles_all_four_directions():
    # "NESW" has distances 1, 2, 1, 0 after each move, so the maximum is 2.
    assert maxDistance("NESW", 0) == 2


def test_one_change_can_turn_an_opposite_move_into_same_direction():
    # Change the first S in "SS" to N, producing "NS" only gives 0 at end;
    # instead change one S to N and perform "NN", reaching distance 2.
    assert maxDistance("SS", 1) == 2


def test_changes_can_make_every_move_increase_distance():
    # With four changes, "SENW" can become "NNNN"; after four moves distance is 4.
    assert maxDistance("SENW", 4) == 4
