from solution import sumOfGoodNumbers


def test_canonical_example_1():
    # nums[1] = 3, nums[4] = 5, and nums[5] = 4 are good: 3 + 5 + 4 = 12.
    assert sumOfGoodNumbers([1, 3, 2, 1, 5, 4], 2) == 12


def test_canonical_example_2():
    # Only nums[0] = 2 is good: 2.
    assert sumOfGoodNumbers([2, 1], 1) == 2


def test_requires_strictly_greater_than_both_existing_neighbors():
    # nums[1] = 4 and nums[3] = 3 exceed both k-distance neighbors: 4 + 3 = 7.
    assert sumOfGoodNumbers([1, 4, 2, 3, 1], 1) == 7


def test_equal_k_distance_neighbors_are_not_good():
    # Each element equals its existing k-distance neighbor, so none is strictly greater: 0.
    assert sumOfGoodNumbers([5, 1, 5, 1], 2) == 0
