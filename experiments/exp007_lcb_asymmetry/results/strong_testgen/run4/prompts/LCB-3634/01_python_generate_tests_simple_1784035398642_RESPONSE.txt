```python
from solution import calculateScore


def test_canonical_example_1():
    # spec: "aczzx" -> pairs (0, 2) worth 2 and (1, 4) worth 3; total = 5
    assert calculateScore("aczzx") == 5


def test_canonical_example_2():
    # spec: "abcdef" has no earlier unmarked mirror for any character; total = 0
    assert calculateScore("abcdef") == 0


def test_uses_closest_unmarked_mirror_index():
    # "aazz": index 2 pairs with closest 'a' at 1 (2 - 1 = 1);
    # index 3 then pairs with remaining 'a' at 0 (3 - 0 = 3); total = 4
    assert calculateScore("aazz") == 4


def test_marked_indices_cannot_be_reused():
    # "azaz": indices 0 and 1 pair for 1; indices 2 and 3 pair for 1;
    # marked index 0 cannot pair again; total = 2
    assert calculateScore("azaz") == 2


def test_matches_interior_alphabet_mirrors():
    # "by": 'b' and 'y' are mirrors, so index 1 pairs with index 0 for 1 - 0 = 1
    assert calculateScore("by") == 1
```
