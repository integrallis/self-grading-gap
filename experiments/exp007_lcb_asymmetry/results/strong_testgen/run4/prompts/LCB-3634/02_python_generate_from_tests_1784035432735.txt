```python
def calculateScore(s: str) -> int:
    score = 0
    marked = set()
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # Create mirror mapping

    for i in range(len(s)):
        if i in marked:
            continue
        for j in range(i - 1, -1, -1):  # Look for the closest unmarked mirror
            if j in marked:
                continue
            if mirror[s[i]] == s[j]:
                marked.add(i)
                marked.add(j)
                score += i - j
                break
    return score
```
