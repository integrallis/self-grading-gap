from solution import countMentions


def test_canonical_example_1():
    events = [
        ["MESSAGE", "10", "id1 id0"],
        ["OFFLINE", "11", "0"],
        ["MESSAGE", "71", "HERE"],
    ]

    # At time 10 each user is explicitly mentioned once; at time 71 both are online for HERE.
    assert countMentions(2, events) == [2, 2]


def test_canonical_example_2():
    events = [
        ["MESSAGE", "10", "id1 id0"],
        ["OFFLINE", "11", "0"],
        ["MESSAGE", "12", "ALL"],
    ]

    # Each user is explicitly mentioned at time 10, and ALL adds one mention for both users.
    assert countMentions(2, events) == [2, 2]


def test_canonical_example_3():
    events = [
        ["OFFLINE", "10", "0"],
        ["MESSAGE", "12", "HERE"],
    ]

    # User 0 remains offline at time 12, so HERE mentions only user 1.
    assert countMentions(2, events) == [0, 1]


def test_explicit_ids_count_duplicates_even_when_user_is_offline():
    events = [
        ["OFFLINE", "5", "1"],
        ["MESSAGE", "6", "id1 id1 id0"],
    ]

    # Explicit id1 appears twice despite being offline, and id0 appears once.
    assert countMentions(2, events) == [1, 2]


def test_here_includes_user_who_returns_online_exactly_after_sixty_time_units():
    events = [
        ["OFFLINE", "10", "0"],
        ["MESSAGE", "70", "HERE"],
    ]

    # User 0 returns at time 70 before the message is handled, so HERE mentions both users.
    assert countMentions(2, events) == [1, 1]


def test_offline_status_change_at_same_timestamp_precedes_here_message():
    events = [
        ["MESSAGE", "10", "HERE"],
        ["OFFLINE", "10", "0"],
    ]

    # The offline change at time 10 is processed before HERE, so only user 1 is mentioned.
    assert countMentions(2, events) == [0, 1]


def test_events_are_processed_by_timestamp_for_online_status():
    events = [
        ["MESSAGE", "20", "HERE"],
        ["OFFLINE", "10", "0"],
    ]

    # User 0 went offline at time 10 and is still offline at time 20, so HERE mentions only user 1.
    assert countMentions(2, events) == [0, 1]
