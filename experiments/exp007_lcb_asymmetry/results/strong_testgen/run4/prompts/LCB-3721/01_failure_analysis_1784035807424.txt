from typing import List

def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers
    online_status = [True] * numberOfUsers
    user_offline_time = [0] * numberOfUsers

    for event in events:
        event_type, timestamp, *args = event
        timestamp = int(timestamp)

        if event_type == 'OFFLINE':
            user_id = int(args[0])
            online_status[user_id] = False
            user_offline_time[user_id] = timestamp + 60
        elif event_type == 'MESSAGE':
            mentions_string = args[0]
            if mentions_string == 'ALL':
                for i in range(numberOfUsers):
                    mentions[i] += 1
            elif mentions_string == 'HERE':
                for i in range(numberOfUsers):
                    if online_status[i]:
                        mentions[i] += 1
            else:
                mentioned_ids = mentions_string.split()
                for user in mentioned_ids:
                    user_id = int(user[2:])  # Extract the user id after 'id'
                    mentions[user_id] += 1

        # Process online status changes
        for user_id in range(numberOfUsers):
            if not online_status[user_id] and timestamp >= user_offline_time[user_id]:
                online_status[user_id] = True

    return mentions
