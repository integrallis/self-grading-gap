```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a dictionary to hold the number of coins at each bag position
    coin_map = {}
    for l, r, c in coins:
        for pos in range(l, r + 1):
            coin_map[pos] = coin_map.get(pos, 0) + c

    # Create a list of positions and their corresponding coin counts
    positions = sorted(coin_map.keys())
    max_coins = 0

    for i in range(len(positions)):
        # Calculate the range of bags we are collecting
        start = positions[i]
        end = start + k - 1

        # Calculate the total coins in this range
        total_coins = 0
        for j in range(i, len(positions)):
            pos = positions[j]
            if pos > end:
                break
            total_coins += coin_map[pos]

        max_coins = max(max_coins, total_coins)

    return max_coins
```
