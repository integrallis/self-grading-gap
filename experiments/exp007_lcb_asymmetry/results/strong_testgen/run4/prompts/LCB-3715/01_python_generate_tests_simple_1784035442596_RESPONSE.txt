from solution import maximumCoins


def test_canonical_example_1():
    # [3, 4, 5, 6] contains 2 + 0 + 4 + 4 = 10 coins.
    assert maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) == 10


def test_canonical_example_2():
    # Any two bags within [1, 10] contain 3 + 3 = 6 coins.
    assert maximumCoins([[1, 10, 3]], 2) == 6


def test_window_can_include_empty_bags_between_segments():
    # Bags [1, 2, 3, 4] contain 10 + 10 + 0 + 0 = 20 coins.
    assert maximumCoins([[1, 2, 10], [5, 7, 4]], 4) == 20


def test_window_can_collect_only_part_of_a_segment():
    # Bags [7, 8, 9] contain 5 + 5 + 5 = 15 coins.
    assert maximumCoins([[1, 5, 2], [7, 10, 5]], 3) == 15


def test_k_can_exceed_the_length_of_all_coin_containing_bags():
    # A five-bag window containing [10, 11, 12] has 5 + 5 + 5 = 15 coins.
    assert maximumCoins([[10, 12, 5]], 5) == 15


def test_segments_need_not_be_given_in_coordinate_order():
    # Bags [3, 4, 5, 6] contain 2 + 2 + 4 + 4 = 12 coins.
    assert maximumCoins([[5, 6, 4], [1, 4, 2]], 4) == 12


def test_coordinates_at_the_upper_constraint_bound():
    # Either one-bag selection contains 1000 coins.
    assert maximumCoins([[999_999_999, 1_000_000_000, 1000]], 1) == 1000
