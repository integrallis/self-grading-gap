from solution import subarraySum


def test_canonical_example_1():
    # i=0: 2, i=1: 2+3=5, i=2: 3+1=4; total = 2+5+4 = 11.
    assert subarraySum([2, 3, 1]) == 11


def test_canonical_example_2():
    # i=0: 3, i=1: 3+1=4, i=2: 1+1=2, i=3: 1+1+2=4; total = 13.
    assert subarraySum([3, 1, 1, 2]) == 13


def test_single_element_array():
    # i=0: the subarray is [1000], so the total is 1000.
    assert subarraySum([1000]) == 1000


def test_subarrays_with_starts_clamped_to_zero():
    # i=0: 1, i=1: 1+2=3, i=2: 1+2+3=6, i=3: 1+2+3+4=10; total = 20.
    assert subarraySum([1, 2, 3, 4]) == 20


def test_subarrays_with_varying_window_starts():
    # i=0: 1, i=1: 1+4=5, i=2: 4+1=5, i=3: 4+1+2=7, i=4: 2+1=3; total = 21.
    assert subarraySum([1, 4, 1, 2, 1]) == 21
