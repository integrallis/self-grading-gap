from solution import hasSpecialSubstring


def test_canonical_example_1():
    # "aaa" at the end has length 3; its preceding character is "b".
    assert hasSpecialSubstring("aaabaaa", 3) is True


def test_canonical_example_2():
    # No length-2 substring of "abc" consists of one repeated character.
    assert hasSpecialSubstring("abc", 2) is False


def test_entire_string_is_a_special_substring():
    # The whole string "aaaa" has exactly length 4 and no neighboring characters.
    assert hasSpecialSubstring("aaaa", 4) is True


def test_run_longer_than_k_is_not_a_special_substring():
    # Every length-3 "aaa" is part of the length-4 run "aaaa", so it has an equal neighbor.
    assert hasSpecialSubstring("aaaa", 3) is False


def test_middle_run_with_different_neighbors_is_special():
    # "bbb" has length 3 and is bordered by different characters "a" and "c".
    assert hasSpecialSubstring("abbbca", 3) is True


def test_single_character_substring_requires_different_neighbors():
    # The "b" has length 1 and is bordered by different characters "a" and "c".
    assert hasSpecialSubstring("abc", 1) is True


def test_single_character_inside_repeated_run_is_not_special():
    # Each length-1 "a" has an immediately adjacent equal "a".
    assert hasSpecialSubstring("aa", 1) is False


def test_can_find_special_run_after_an_earlier_nonmatching_run():
    # The first run "aa" has the wrong length; "bbb" has exactly length 3 and is at the end.
    assert hasSpecialSubstring("aabbb", 3) is True
