from solution import maxDifference


def test_canonical_example_1():
    # 'a' appears 5 times (odd) and 'b' appears 2 times (even): 5 - 2 = 3.
    assert maxDifference("aaaaabbc") == 3


def test_canonical_example_2():
    # 'a' appears 3 times (odd) and 'c' appears 2 times (even): 3 - 2 = 1.
    assert maxDifference("abcabcab") == 1


def test_returns_negative_difference_when_even_frequency_is_larger():
    # 'b' and 'c' each appear 1 time (odd), while 'a' appears 4 times (even): 1 - 4 = -3.
    assert maxDifference("aaaabc") == -3


def test_uses_largest_odd_frequency_and_smallest_even_frequency():
    # Odd frequencies are 5 ('a') and 3 ('b'); even frequencies are 2 ('c') and 4 ('d'): 5 - 2 = 3.
    assert maxDifference("aaaaabbbccdddd") == 3
