from solution import maxLength


def test_canonical_example_1():
    # [1, 2, 1, 1, 1] has product 2, gcd 1, and lcm 2; every length-6
    # subarray has product 4 but gcd * lcm is 2, so the maximum is 5.
    assert maxLength([1, 2, 1, 2, 1, 1, 1]) == 5


def test_canonical_example_2():
    # [3, 4, 5] has product 60, gcd 1, and lcm 60; each length-4
    # subarray has product different from gcd * lcm, so the maximum is 3.
    assert maxLength([2, 3, 4, 5, 6]) == 3


def test_canonical_example_3():
    # [3, 1, 4, 5, 1] has product 60, gcd 1, and lcm 60; each length-6
    # subarray has product 120 but gcd * lcm is 60, so the maximum is 5.
    assert maxLength([1, 2, 3, 1, 4, 5, 1]) == 5


def test_all_ones_entire_array_is_product_equivalent():
    # product([1, 1, 1, 1, 1, 1]) = 1, gcd = 1, and lcm = 1, so all 6
    # elements form a product equivalent subarray.
    assert maxLength([1, 1, 1, 1, 1, 1]) == 6


def test_repeated_non_one_value_limits_length_to_two():
    # Any pair [2, 2] has product 4, gcd 2, and lcm 2; all three elements
    # have product 8 while gcd * lcm is 4, so the maximum is 2.
    assert maxLength([2, 2, 2]) == 2


def test_two_element_subarray_with_common_factor():
    # product([6, 10]) = 60, gcd = 2, and lcm = 30, so 60 = 2 * 30.
    assert maxLength([6, 10]) == 2
