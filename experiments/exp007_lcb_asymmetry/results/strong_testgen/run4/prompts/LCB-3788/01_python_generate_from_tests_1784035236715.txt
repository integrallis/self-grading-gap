from typing import List

def maxSum(nums: List[int]) -> int:
    unique_nums = set()
    max_sum = float('-inf')
    current_sum = 0

    for num in nums:
        if num not in unique_nums:
            unique_nums.add(num)
            current_sum += num
        else:
            # Update max_sum if current_sum is greater
            max_sum = max(max_sum, current_sum)
            # Reset the current sum and unique set since we cannot use the duplicated number
            unique_nums = set()
            current_sum = 0
            # Recalculate current_sum with unique numbers until we find a duplicate
            for n in nums:
                if n not in unique_nums:
                    unique_nums.add(n)
                    current_sum += n
                if n == num:
                    break

    # Update max_sum with the last seen current_sum
    return max(max_sum, current_sum) if unique_nums else 0
