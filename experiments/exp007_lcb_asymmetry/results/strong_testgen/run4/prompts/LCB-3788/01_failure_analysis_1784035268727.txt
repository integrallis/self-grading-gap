```python
from typing import List

def maxSum(nums: List[int]) -> int:
    unique_nums = set()
    total_sum = 0
    max_single_value = float('-inf')

    for num in nums:
        if num not in unique_nums:
            unique_nums.add(num)
            total_sum += num
        max_single_value = max(max_single_value, num)

    # If total_sum is zero or positive, return it; otherwise, return the largest single value
    return total_sum if total_sum > 0 else max_single_value
```
