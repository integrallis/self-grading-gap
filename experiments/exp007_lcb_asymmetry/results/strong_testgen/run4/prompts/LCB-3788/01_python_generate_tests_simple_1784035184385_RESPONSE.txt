from solution import maxSum


def test_canonical_example_1():
    # All values are unique, so selecting [1, 2, 3, 4, 5] gives 1 + 2 + 3 + 4 + 5 = 15.
    assert maxSum([1, 2, 3, 4, 5]) == 15


def test_canonical_example_2():
    # Delete duplicates and 0; selecting one [1] gives the maximum sum of 1.
    assert maxSum([1, 1, 0, 1, 1]) == 1


def test_canonical_example_3():
    # Delete -1 and -2; selecting unique values [1, 2] gives 1 + 2 = 3.
    assert maxSum([1, 2, -1, -2, 1, 0, -1]) == 3


def test_single_negative_element_must_be_selected():
    # The array cannot be made empty, so the only selectable subarray [-7] sums to -7.
    assert maxSum([-7]) == -7


def test_all_negative_values_selects_largest_single_value():
    # Any additional negative value lowers the sum; selecting [-2] gives the maximum sum -2.
    assert maxSum([-4, -2, -2, -3]) == -2


def test_zero_is_preferred_over_negative_values():
    # Selecting [0] gives 0, which is greater than selecting either negative value.
    assert maxSum([-5, 0, -1]) == 0


def test_duplicate_positive_values_are_counted_once():
    # Delete duplicate occurrences; selecting unique values [5, 4] gives 5 + 4 = 9.
    assert maxSum([5, 5, 4, 4]) == 9


def test_deletions_allow_positive_values_separated_by_duplicates_and_negatives():
    # Delete repeated 2 and -3; selecting unique values [2, 6] gives 2 + 6 = 8.
    assert maxSum([2, -3, 2, 6, -3]) == 8
