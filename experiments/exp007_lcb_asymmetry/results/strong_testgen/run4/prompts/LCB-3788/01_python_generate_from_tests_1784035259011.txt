from typing import List

def maxSum(nums: List[int]) -> int:
    unique_nums = set()
    max_sum = float('-inf')
    current_sum = 0
    
    for num in nums:
        if num not in unique_nums:
            unique_nums.add(num)
            current_sum += num
    
    max_sum = max(max_sum, current_sum)
    
    # Handle the case where all values are negative or zero
    return max(max_sum, max(nums), 0)
```
