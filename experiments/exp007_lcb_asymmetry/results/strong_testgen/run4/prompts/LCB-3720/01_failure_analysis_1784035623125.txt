from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    from collections import defaultdict, deque

    edges.sort(key=lambda x: x[2])  # Sort edges by weight
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))

    def bfs(max_weight):
        visited = [False] * n
        queue = deque([0])
        visited[0] = True
        while queue:
            node = queue.popleft()
            for neighbor, weight in graph[node]:
                if not visited[neighbor] and weight <= max_weight:
                    visited[neighbor] = True
                    queue.append(neighbor)
        return all(visited)

    low, high = 1, max(weight for _, _, weight in edges)
    result = -1

    while low <= high:
        mid = (low + high) // 2
        # Create a subgraph with max weight <= mid
        subgraph = defaultdict(list)
        for u, v, w in edges:
            if w <= mid:
                subgraph[u].append(v)

        # Check if we can reach node 0 from all nodes with the current threshold
        if bfs(mid):
            result = mid
            high = mid - 1
        else:
            low = mid + 1

    return result
