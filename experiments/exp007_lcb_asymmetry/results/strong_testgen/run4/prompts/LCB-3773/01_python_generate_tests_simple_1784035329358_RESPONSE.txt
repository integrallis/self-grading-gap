from solution import minimumPairRemoval


def test_canonical_example_1():
    # (3, 1) has minimum sum 4 -> [5, 2, 4];
    # then (2, 4) has minimum sum 6 -> [5, 6], which is non-decreasing.
    assert minimumPairRemoval([5, 2, 3, 1]) == 2


def test_canonical_example_2():
    # 1 <= 2 <= 2, so the array is already non-decreasing.
    assert minimumPairRemoval([1, 2, 2]) == 0


def test_single_element_is_already_non_decreasing():
    # A one-element array has no previous element that could violate the ordering.
    assert minimumPairRemoval([7]) == 0


def test_merges_minimum_sum_pair_even_when_it_is_not_the_first_decreasing_pair():
    # (1, 2) has minimum sum 3 -> [4, 3];
    # then (4, 3) -> [7], which is non-decreasing.
    assert minimumPairRemoval([4, 1, 2]) == 2


def test_uses_leftmost_pair_when_minimum_sums_tie():
    # All initial adjacent sums are -2, so merge the leftmost (3, -5) -> [-2, 3, -5, 3].
    # The minimum sum is then -2 for both (3, -5) and (-5, 3), so merge the leftmost
    # (3, -5) -> [-2, -2, 3], which is non-decreasing.
    assert minimumPairRemoval([3, -5, 3, -5, 3]) == 2


def test_negative_values_can_require_merging_to_one_element():
    # (-2, -3) has minimum sum -5 -> [-1, -5];
    # then (-1, -5) -> [-6], which is non-decreasing.
    assert minimumPairRemoval([-1, -2, -3]) == 2
