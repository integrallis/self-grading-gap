from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total = 0
    for i in range(1, k + 1):
        # Calculate the contribution of each element as min and max
        contribution = 0
        for j in range(n):
            # Each element appears in C(n-1, i-1) subsequences where it's max
            # and C(n-1, i-1) where it's min
            contribution += nums[j] * (pow(2, n - 1, MOD) % MOD) % MOD
        total += (contribution * i) % MOD
    return total % MOD
```
