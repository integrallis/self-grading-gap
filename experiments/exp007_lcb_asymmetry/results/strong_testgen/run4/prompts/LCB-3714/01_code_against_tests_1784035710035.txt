## Implementation

```python
from typing import List

MOD = 10**9 + 7


def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total = 0
    # Precompute powers of 2 up to n
    power_of_2 = [1] * (n + 1)
    for i in range(1, n + 1):
        power_of_2[i] = (power_of_2[i - 1] * 2) % MOD
    
    for i in range(1, k + 1):
        # For each element, calculate its total contribution
        for j in range(n):
            # Contribution of nums[j] as both min and max in subsequences of size i
            contribution = (nums[j] * power_of_2[n - 1] * 2) % MOD
            total = (total + contribution) % MOD
    
    return total % MOD

```
