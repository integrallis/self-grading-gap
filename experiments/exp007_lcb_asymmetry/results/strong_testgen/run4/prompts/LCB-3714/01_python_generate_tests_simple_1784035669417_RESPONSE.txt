from solution import minMaxSums


def test_canonical_example_1():
    # Singles contribute 2 + 4 + 6 = 12; pairs contribute 3 + 4 + 5 = 12.
    assert minMaxSums([1, 2, 3], 2) == 24


def test_canonical_example_2():
    # With k = 1, each element is both min and max: 2*5 + 2*0 + 2*6 = 22.
    assert minMaxSums([5, 0, 6], 1) == 22


def test_canonical_example_3():
    # Three singleton subsequences and three size-two subsequences each contribute 2.
    assert minMaxSums([1, 1, 1], 2) == 12


def test_single_element_subsequence():
    # The only subsequence is [7], whose minimum plus maximum is 7 + 7 = 14.
    assert minMaxSums([7], 1) == 14


def test_k_includes_all_nonempty_subsequences():
    # Singles: 4 + 10 + 2 = 16; pairs: 7 + 5 + 6 = 18; triple: 1 + 5 = 6.
    assert minMaxSums([2, 5, 1], 3) == 40


def test_zero_values_contribute_zero_when_minimum_and_maximum():
    # Singles contribute 0 + 0 + 6 = 6; pairs contribute 0 + 3 + 3 = 6.
    assert minMaxSums([0, 0, 3], 2) == 12


def test_result_is_returned_modulo_1000000007():
    # All 2^31 - 1 nonempty subsequences contribute 10^9 + 10^9 = 2*10^9.
    # (2*10^9) * (2^31 - 1) mod 1_000_000_007 = 935_229_159.
    assert minMaxSums([1_000_000_000] * 31, 31) == 935_229_159
