```python
from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[-float('inf')] * (n + 1) for _ in range(m + 1)]
    dp[0][1] = 0  # Start at (0, 0) with 0 coins

    for i in range(m):
        for j in range(n):
            # Calculate the maximum coins if moving right
            if j + 1 <= n:
                dp[i][j + 1] = max(dp[i][j + 1], dp[i][j] + coins[i][j])  # Move right
            # Calculate the maximum coins if moving down
            if i + 1 <= m:
                dp[i + 1][j] = max(dp[i + 1][j], dp[i][j] + coins[i][j])  # Move down
            # Handle robbers (negative coins)
            if coins[i][j] < 0:
                # Neutralize robber by considering both cases
                if j + 1 <= n:
                    dp[i][j + 1] = max(dp[i][j + 1], dp[i][j] - coins[i][j])  # Use neutralization
                if i + 1 <= m:
                    dp[i + 1][j] = max(dp[i + 1][j], dp[i][j] - coins[i][j])  # Use neutralization

    # Return the maximum coins collected at (m-1, n-1) or 0 if negative
    return max(dp[m - 1][n], 0)
```
