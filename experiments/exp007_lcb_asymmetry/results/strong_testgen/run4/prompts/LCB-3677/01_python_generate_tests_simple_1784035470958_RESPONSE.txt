from solution import maximumAmount


def test_canonical_example_1():
    # Path 0 -> 1 -> neutralize -2 -> 3 -> 4 gives 0 + 1 + 0 + 3 + 4 = 8.
    assert maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) == 8


def test_canonical_example_2():
    # A path visits four cells worth 10 each: 10 + 10 + 10 + 10 = 40.
    assert maximumAmount([[10, 10, 10], [10, 10, 10]]) == 40


def test_single_negative_cell_can_be_neutralized():
    # The only robber is on the path and can be neutralized, so the total is 0.
    assert maximumAmount([[-1000]]) == 0


def test_single_positive_cell_is_collected():
    # The only path contains the single cell, so the total is 7.
    assert maximumAmount([[7]]) == 7


def test_only_two_robbers_can_be_neutralized():
    # Neutralize -5 and -4; the remaining -3 is stolen, for a total of -3.
    assert maximumAmount([[-5, -4, -3]]) == -3


def test_robot_can_choose_a_route_with_better_profit():
    # Going down then right twice collects 5 + 5 + 5 + 5 = 20, avoiding -100.
    assert maximumAmount([[5, -100, 5], [5, 5, 5]]) == 20
