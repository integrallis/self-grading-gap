from solution import largestInteger


def test_canonical_example_1():
    # 3 and 7 each appear in exactly one size-3 subarray; the largest is 7.
    assert largestInteger([3, 9, 2, 1, 7], 3) == 7


def test_canonical_example_2():
    # Only 3 appears in exactly one size-4 subarray.
    assert largestInteger([3, 9, 7, 2, 1, 7], 4) == 3


def test_canonical_example_3():
    # 0 appears in both size-1 subarrays, so no integer appears in exactly one.
    assert largestInteger([0, 0], 1) == -1


def test_full_array_is_the_only_subarray():
    # There is one size-3 subarray, so both 4 and 2 appear in exactly one; largest is 4.
    assert largestInteger([4, 4, 2], 3) == 4


def test_k_one_counts_each_array_position_as_a_subarray():
    # For size 1, only 1 and 2 occur in exactly one subarray; the largest is 2.
    assert largestInteger([1, 3, 3, 2], 1) == 2


def test_no_value_appears_in_exactly_one_window():
    # With size-2 windows [8, 1] and [1, 8], both 8 and 1 appear in two subarrays.
    assert largestInteger([8, 1, 8], 2) == -1
