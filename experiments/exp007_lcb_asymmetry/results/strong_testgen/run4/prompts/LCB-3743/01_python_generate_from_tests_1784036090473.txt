from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    gaps = []

    # Calculate the initial gaps before any meetings
    prev_end = 0
    for i in range(n):
        if startTime[i] > prev_end:
            gaps.append(startTime[i] - prev_end)
        prev_end = endTime[i]

    # Add final gap if any
    if prev_end < eventTime:
        gaps.append(eventTime - prev_end)

    # Sort gaps in descending order to maximize the free time
    gaps.sort(reverse=True)

    # Calculate the maximum free time possible by merging gaps
    max_free_time = 0
    for i in range(min(k, len(gaps))):
        max_free_time += gaps[i]

    return max_free_time
```
