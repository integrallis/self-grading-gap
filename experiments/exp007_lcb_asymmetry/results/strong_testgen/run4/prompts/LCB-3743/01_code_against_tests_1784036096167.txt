```python
from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    gaps = []

    # Calculate the initial gaps before any meetings
    prev_end = 0
    for i in range(n):
        if startTime[i] > prev_end:
            gaps.append(startTime[i] - prev_end)
        prev_end = endTime[i]

    # Add final gap if any
    if prev_end < eventTime:
        gaps.append(eventTime - prev_end)

    # Sort gaps in descending order to maximize the free time
    gaps.sort(reverse=True)

    # Calculate the maximum free time possible by merging gaps
    max_free_time = 0
    merged_gaps = min(len(gaps), k)
    total_merge = 0

    # Calculate the total length of gaps that would be merged
    for i in range(merged_gaps):
        total_merge += gaps[i]

    # The maximum free time is the sum of merged gaps plus any remaining free time
    if merged_gaps < len(gaps):
        return total_merge + sum(gaps[merged_gaps:])

    return total_merge

```
