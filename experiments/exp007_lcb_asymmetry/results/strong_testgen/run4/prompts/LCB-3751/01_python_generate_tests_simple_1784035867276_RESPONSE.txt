from solution import maxFrequency


def test_canonical_example_1():
    # Adding -5 to nums[2..5] produces [1, 2, -2, -1, 0, 1], so 1 appears twice.
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2


def test_canonical_example_2():
    # Adding 8 to nums[1..9] produces four occurrences of 10.
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4


def test_converts_a_subarray_while_preserving_k_values_outside_it():
    # Adding -1 to the middle [2, 2] gives [1, 1, 1, 1], so 1 appears four times.
    assert maxFrequency([1, 2, 2, 1], 1) == 4


def test_selecting_k_inside_a_nonzero_shift_can_reduce_its_frequency():
    # Converting either 2 to 1 gives two 1s; shifting all three converts two 2s but loses the middle 1.
    assert maxFrequency([2, 1, 2], 1) == 2


def test_one_shift_cannot_convert_different_source_values_to_k():
    # A single x can convert either both 2s or both 3s to 1, but not both values, so the maximum is two.
    assert maxFrequency([2, 3, 2, 3], 1) == 2


def test_already_all_k_values_can_remain_k():
    # Choosing x = 0 leaves all three values equal to 7, so the frequency is three.
    assert maxFrequency([7, 7, 7], 7) == 3
