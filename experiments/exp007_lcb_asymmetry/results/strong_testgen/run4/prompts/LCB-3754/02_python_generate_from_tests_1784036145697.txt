def maxDistance(s: str, k: int) -> int:
    # Initialize coordinates and max distance
    x, y, max_dist = 0, 0, 0

    # Process each move in the string
    for move in s:
        # Update coordinates based on the move
        if move == 'N':
            y += 1
        elif move == 'S':
            y -= 1
        elif move == 'E':
            x += 1
        elif move == 'W':
            x -= 1

        # Calculate current Manhattan distance from the origin
        current_distance = abs(x) + abs(y)
        max_dist = max(max_dist, current_distance)

        # If we have changes left, consider the impact of optimal changes
        if k > 0:
            # Increase the maximum distance possible by changing moves
            max_dist = max(max_dist, current_distance + k)

    return max_dist
```
