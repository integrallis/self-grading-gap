from solution import maxDistance


def test_canonical_example_1():
    # Change "NWSE" to "NWNE": distances are 1, 2, 3, 2, so maximum is 3.
    assert maxDistance("NWSE", 1) == 3


def test_canonical_example_2():
    # Change "NSWWEW" to "NNWWWW": after all 6 moves, distance is 6.
    assert maxDistance("NSWWEW", 3) == 6


def test_single_move_without_changes():
    # "N" moves from (0, 0) to (0, 1), whose Manhattan distance is 1.
    assert maxDistance("N", 0) == 1


def test_no_changes_uses_largest_distance_reached_during_path():
    # "NS": distances after each move are 1 and 0, so the maximum is 1.
    assert maxDistance("NS", 0) == 1


def test_change_can_make_opposite_moves_continue_away_from_origin():
    # Change "NS" to "NN": distances after each move are 1 and 2, so maximum is 2.
    assert maxDistance("NS", 1) == 2


def test_at_most_k_changes_allows_no_change_when_already_optimal():
    # "NN" already reaches (0, 2), whose Manhattan distance is 2.
    assert maxDistance("NN", 1) == 2


def test_changing_every_move_can_reach_the_number_of_moves():
    # Change "NSWE" to "NNNN": after 4 north moves, distance is 4.
    assert maxDistance("NSWE", 4) == 4
