from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n < 2:
        return n

    # Create an array to store the longest subsequence lengths
    dp = [1] * n
    max_length = 1

    # Iterate through the numbers
    for i in range(n):
        for j in range(i):
            # Calculate the absolute difference
            if abs(nums[i] - nums[j]) >= (dp[i] - dp[j]):
                dp[i] = max(dp[i], dp[j] + 1)
                max_length = max(max_length, dp[i])

    return max_length

```
