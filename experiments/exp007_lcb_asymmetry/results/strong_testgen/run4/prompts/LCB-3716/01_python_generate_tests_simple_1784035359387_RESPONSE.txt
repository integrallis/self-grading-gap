from solution import longestSubsequence


def test_canonical_example_1():
    # [16, 6, 3] has adjacent differences [10, 3], and 10 >= 3.
    assert longestSubsequence([16, 6, 3]) == 3


def test_canonical_example_2():
    # [6, 4, 2, 1] is a subsequence with adjacent differences [2, 2, 1].
    assert longestSubsequence([6, 5, 3, 4, 2, 1]) == 4


def test_canonical_example_3():
    # [10, 20, 10, 19, 10] has adjacent differences [10, 10, 9, 9].
    assert longestSubsequence([10, 20, 10, 19, 10, 20]) == 5


def test_equal_values_allow_zero_differences():
    # The full subsequence [42, 42, 42, 42] has differences [0, 0, 0].
    assert longestSubsequence([42, 42, 42, 42]) == 4


def test_two_elements_form_a_valid_subsequence():
    # The full array has one adjacent difference, so its length is 2.
    assert longestSubsequence([1, 300]) == 2


def test_can_skip_elements_to_keep_differences_non_increasing():
    # [1, 3, 2] has differences [2, 1]; all four elements give [2, 1, 2], which increases.
    assert longestSubsequence([1, 3, 2, 4]) == 3
