```python
from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    if n < 2:
        return n

    dp = [1] * n
    diffs = [0] * (n * (n - 1) // 2)
    index = 0

    for i in range(n):
        for j in range(i):
            diff = abs(nums[i] - nums[j])
            diffs[index] = diff
            index += 1
            if dp[j] + 1 > dp[i]:
                dp[i] = dp[j] + 1

    max_length = 1
    prev_length = float('inf')

    for i in range(n):
        if dp[i] > 1:
            length = dp[i]
            current_diff = diffs[i] if i > 0 else float('inf')
            if length >= prev_length:  # Ensure non-increasing.
                max_length = max(max_length, length)
            prev_length = current_diff

    return max_length
```
