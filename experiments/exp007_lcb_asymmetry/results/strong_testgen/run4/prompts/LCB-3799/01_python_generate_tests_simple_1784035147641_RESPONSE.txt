from solution import totalNumbers


def test_canonical_example_1():
    # 12 distinct valid even numbers: 124, 132, 134, 142, 214, 234,
    # 312, 314, 324, 342, 412, and 432.
    assert totalNumbers([1, 2, 3, 4]) == 12


def test_canonical_example_2():
    # The two valid even numbers are 202 and 220.
    assert totalNumbers([0, 2, 2]) == 2


def test_canonical_example_3():
    # All three available digits form the single valid number 666.
    assert totalNumbers([6, 6, 6]) == 1


def test_canonical_example_4():
    # No available digit can be the final digit of an even number.
    assert totalNumbers([1, 3, 5]) == 0


def test_does_not_allow_leading_zero():
    # From 0, 1, and 2, the valid even numbers are 102, 120, and 210;
    # 012 is not a three-digit number.
    assert totalNumbers([0, 1, 2]) == 3


def test_respects_available_duplicate_copies():
    # The valid even numbers are 112, 122, and 212; 222 needs three 2s.
    assert totalNumbers([1, 1, 2, 2]) == 3
