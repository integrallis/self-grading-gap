from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    dp = [[float('inf')] * 3 for _ in range(n)]

    # Initialize the first house cost
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the dp table for all houses
    for i in range(1, n):
        for j in range(3):
            for k in range(3):
                if j != k:
                    dp[i][j] = min(dp[i][j], dp[i - 1][k] + cost[i][j])

    # Handle equidistant constraints
    for i in range(n // 2):
        for j in range(3):
            dp[n - 1 - i][j] = float('inf')
            for k in range(3):
                if j != k:
                    dp[n - 1 - i][j] = min(dp[n - 1 - i][j], dp[n // 2 + i - 1][k] + cost[n - 1 - i][j])

    return min(dp[n - 1])

```
