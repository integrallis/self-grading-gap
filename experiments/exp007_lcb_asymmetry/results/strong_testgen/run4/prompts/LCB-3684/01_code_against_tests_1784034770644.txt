## Implementation

```python
def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]

    # Check for all possible starting positions for the prefix
    for i in range(len(s) - len(prefix) - len(suffix) + 1):
        # Check if the prefix matches
        if s[i:i + len(prefix)] == prefix:
            # Check if the suffix matches
            if s[i + len(prefix):].startswith(suffix):
                return True
    return False

```
