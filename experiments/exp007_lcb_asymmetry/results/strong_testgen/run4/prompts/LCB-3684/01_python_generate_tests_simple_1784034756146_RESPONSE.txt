from solution import hasMatch


def test_canonical_example_1():
    # "ee*" can use "tcod", producing "eetcode", a substring of "leetcode".
    assert hasMatch("leetcode", "ee*e") is True


def test_canonical_example_2():
    # No replacement for "*" makes a substring of "car" start with "c" and end with "v".
    assert hasMatch("car", "c*v") is False


def test_canonical_example_3():
    # Replacing "*" with "" makes "u", which is a substring of "luck".
    assert hasMatch("luck", "u*") is True


def test_star_matches_zero_characters_between_prefix_and_suffix():
    # Replacing "*" with "" makes "abc", which is a substring of "zabcx".
    assert hasMatch("zabcx", "ab*c") is True


def test_star_matches_characters_between_prefix_and_suffix():
    # Replacing "*" with "xx" makes "abxxc", which is a substring of "zabxxcy".
    assert hasMatch("zabxxcy", "ab*c") is True


def test_pattern_can_match_a_substring_not_the_entire_string():
    # Replacing "*" with "" makes "bc", which is a substring of "abcd".
    assert hasMatch("abcd", "b*c") is True


def test_star_at_beginning_can_match_zero_characters():
    # Replacing "*" with "" makes "bc", which is a substring of "abcd".
    assert hasMatch("abcd", "*bc") is True


def test_star_at_end_can_match_zero_characters():
    # Replacing "*" with "" makes "bc", which is a substring of "abcd".
    assert hasMatch("abcd", "bc*") is True


def test_required_parts_must_appear_in_order():
    # No replacement for "*" can make "ba" occur as a substring of "abcd".
