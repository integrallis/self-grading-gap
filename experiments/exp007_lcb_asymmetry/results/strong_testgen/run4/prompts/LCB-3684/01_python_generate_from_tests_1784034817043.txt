def hasMatch(s: str, p: str) -> bool:
    star_index = p.index('*')
    prefix = p[:star_index]
    suffix = p[star_index + 1:]
    prefix_len = len(prefix)
    suffix_len = len(suffix)
    s_len = len(s)

    # Loop through and find the start of the prefix in s
    for i in range(s_len - prefix_len - suffix_len + 1):
        if s[i:i + prefix_len] == prefix:
            # Check if the suffix matches after the prefix
            if s[i + prefix_len:i + prefix_len + suffix_len] == suffix or suffix_len == 0:
                return True
    return False

```
