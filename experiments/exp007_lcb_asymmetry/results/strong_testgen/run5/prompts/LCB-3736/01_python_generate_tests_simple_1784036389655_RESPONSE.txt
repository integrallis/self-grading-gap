from solution import findValidPair


def test_canonical_example_1():
    # '2' occurs 2 times and '3' occurs 3 times; the first valid adjacent pair is "23".
    assert findValidPair("2523533") == "23"


def test_canonical_example_2():
    # '2' occurs 2 times and '1' occurs 1 time; "22" is equal digits, so "21" is first valid.
    assert findValidPair("221") == "21"


def test_canonical_example_3():
    # The only adjacent pair is "22", whose digits are equal, so no valid pair exists.
    assert findValidPair("22") == ""


def test_returns_leftmost_when_multiple_valid_pairs_exist():
    # '1' occurs once, '2' occurs twice, and '3' occurs three times; both "12" and "23" qualify,
    # so the leftmost valid pair is "12".
    assert findValidPair("122333") == "12"


def test_returns_empty_when_adjacent_digits_differ_but_counts_do_not_match_values():
    # In "12", '1' occurs once but '2' occurs once rather than twice, so "12" is not valid.
    assert findValidPair("12") == ""


def test_supports_digit_nine_when_its_count_matches_its_value():
    # '1' occurs once and '9' occurs nine times, so the adjacent pair "19" is valid.
    assert findValidPair("1999999999") == "19"
