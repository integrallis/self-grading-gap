def hasMatch(s: str, p: str) -> bool:
    prefix, _, suffix = p.partition('*')
    len_prefix = len(prefix)
    len_suffix = len(suffix)
    for i in range(len(s) - len_prefix - len_suffix + 1):
        if s[i:i + len_prefix] == prefix and s[i + len_prefix:i + len_prefix + len_suffix] == suffix:
            return True
    return False
```
