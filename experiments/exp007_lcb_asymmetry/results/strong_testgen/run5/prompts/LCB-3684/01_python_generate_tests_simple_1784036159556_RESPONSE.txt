from solution import hasMatch


def test_canonical_example_1():
    # "ee*" can use "tcod", producing "eetcode", a substring of "leetcode".
    assert hasMatch("leetcode", "ee*e") is True


def test_canonical_example_2():
    # No replacement makes a substring of "car" start with "c" and end with "v".
    assert hasMatch("car", "c*v") is False


def test_canonical_example_3():
    # "*" can be empty, so "u" is a substring of "luck".
    assert hasMatch("luck", "u*") is True


def test_wildcard_can_match_zero_characters_between_fixed_parts():
    # Replacing "*" with "" makes "ab", which is a substring of "zabz".
    assert hasMatch("zabz", "a*b") is True


def test_wildcard_can_match_characters_between_fixed_parts():
    # Replacing "*" with "bc" makes "abcd", which is a substring of "xabcdx".
    assert hasMatch("xabcdx", "a*d") is True


def test_prefix_and_suffix_must_appear_in_order_within_one_substring():
    # "ab" occurs only after "c", so no substring can match "ab*c".
    assert hasMatch("cab", "ab*c") is False


def test_pattern_with_wildcard_at_start_matches_suffix():
    # Replacing "*" with "ab" makes "abc", which is a substring of "zabc".
    assert hasMatch("zabc", "*c") is True


def test_pattern_with_wildcard_at_end_matches_prefix():
    # Replacing "*" with "" makes "ab", which is a substring of "zab".
    assert hasMatch("zab", "ab*") is True
