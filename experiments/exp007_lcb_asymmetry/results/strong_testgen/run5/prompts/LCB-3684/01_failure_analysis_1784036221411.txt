def hasMatch(s: str, p: str) -> bool:
    prefix, _, suffix = p.partition('*')
    len_prefix = len(prefix)
    for i in range(len(s) - len_prefix + 1):
        if s[i:i + len_prefix] == prefix and s[i + len_prefix:].startswith(suffix):
            return True
    return False
