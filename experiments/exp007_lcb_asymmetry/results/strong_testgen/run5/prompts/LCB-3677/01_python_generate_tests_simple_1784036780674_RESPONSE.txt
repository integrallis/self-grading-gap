from solution import maximumAmount


def test_canonical_example_1():
    # Path 0 -> 1 -> neutralized -2 -> 3 -> 4 collects 0 + 1 + 0 + 3 + 4 = 8.
    assert maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) == 8


def test_canonical_example_2():
    # Any top-left to bottom-right path visits four 10-value cells: 10 + 10 + 10 + 10 = 40.
    assert maximumAmount([[10, 10, 10], [10, 10, 10]]) == 40


def test_single_negative_cell_can_be_neutralized():
    # The only cell is a robber worth -7; neutralizing it makes the total 0.
    assert maximumAmount([[-7]]) == 0


def test_single_nonnegative_cell_is_collected():
    # The robot starts and finishes on the only cell, collecting 9 coins.
    assert maximumAmount([[9]]) == 9


def test_at_most_two_robbers_can_be_neutralized_on_a_fixed_path():
    # Neutralize -4 and -3 on the only path, leaving -1 + -2 = -3.
    assert maximumAmount([[-1, -2, -3, -4]]) == -3


def test_robot_can_choose_a_path_that_benefits_from_neutralization():
    # Going right, right, down yields 0 + neutralized(-100) + 100 + 1 = 101.
    assert maximumAmount([[0, -100, 100], [1, 1, 1]]) == 101
