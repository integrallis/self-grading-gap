from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[float('-inf')] * (n + 1) for _ in range(m + 1)]
    dp[0][1] = 0  # Initialize starting point

    for i in range(m):
        for j in range(n):
            if dp[i][j + 1] != float('-inf') or dp[i + 1][j] != float('-inf'):
                if coins[i][j] >= 0:
                    dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j + 1] + coins[i][j], dp[i + 1][j] + coins[i][j])
                else:
                    rob = -coins[i][j]
                    if rob <= 2:
                        dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j + 1] - rob, dp[i + 1][j] - rob)
                    dp[i + 1][j + 1] = max(dp[i + 1][j + 1], dp[i][j + 1] + coins[i][j], dp[i + 1][j] + coins[i][j])

    return dp[m][n] if dp[m][n] != float('-inf') else 0
