from typing import List

def maximumAmount(coins: List[List[int]]) -> int:
    m, n = len(coins), len(coins[0])
    dp = [[[float('-inf')] * 3 for _ in range(n + 1)] for _ in range(m + 1)]
    dp[0][1][0] = 0  # Initialize starting point

    for i in range(m):
        for j in range(n):
            for k in range(3):  # k represents number of neutralizations used
                if dp[i][j + 1][k] != float('-inf') or dp[i + 1][j][k] != float('-inf'):
                    if coins[i][j] >= 0:
                        dp[i + 1][j + 1][k] = max(dp[i + 1][j + 1][k], dp[i][j + 1][k] + coins[i][j], dp[i + 1][j][k] + coins[i][j])
                    else:
                        rob = -coins[i][j]
                        # Situation where we encounter a robber
                        if k < 2:
                            # We use a neutralization
                            dp[i + 1][j + 1][k + 1] = max(dp[i + 1][j + 1][k + 1], dp[i][j + 1][k], dp[i + 1][j][k])
                        # Not using a neutralization
                        dp[i + 1][j + 1][k] = max(dp[i + 1][j + 1][k], dp[i][j + 1][k] - rob, dp[i + 1][j][k] - rob)
                        # Collecting value as per the current cell
                        dp[i + 1][j + 1][k] = max(dp[i + 1][j + 1][k], dp[i][j + 1][k] + coins[i][j], dp[i + 1][j][k] + coins[i][j])

    result = max(dp[m][n])
    return result if result != float('-inf') else 0
