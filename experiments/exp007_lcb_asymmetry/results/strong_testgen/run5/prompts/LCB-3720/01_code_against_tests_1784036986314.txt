```python
from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    from collections import defaultdict, deque

    edges.sort(key=lambda x: x[2])  # Sort edges by weight

    def canReach(max_weight):
        graph = defaultdict(list)

        # Construct the graph with edges <= max_weight
        for u, v, w in edges:
            if w <= max_weight:
                graph[u].append(v)

        # Check if all nodes can reach node 0
        reachable = [False] * n
        queue = deque([0])
        reachable[0] = True

        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if not reachable[neighbor]:
                    reachable[neighbor] = True
                    queue.append(neighbor)

        return all(reachable)

    # Binary search over the edge weights
    left, right = 1, 10**6
    result = -1

    while left <= right:
        mid = (left + right) // 2
        if canReach(mid):
            result = mid
            right = mid - 1  # Try for a smaller maximum weight
        else:
            left = mid + 1  # Increase the maximum weight

    return result

```
