from solution import minMaxWeight


def test_canonical_example_1():
    # Paths 1->0, 2->1->0, 3->0, and 4->3->0 use only weight 1.
    assert minMaxWeight(
        5,
        [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]],
        2,
    ) == 1


def test_canonical_example_2():
    # Node 2 has no directed path to node 0, so satisfying reachability is impossible.
    assert minMaxWeight(
        5,
        [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]],
        1,
    ) == -1


def test_canonical_example_3():
    # Keeping 1->2 (1), 2->3 (2), 3->4 (2), and 4->0 (1) gives maximum weight 2.
    assert minMaxWeight(
        5,
        [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]],
        1,
    ) == 2


def test_canonical_example_4():
    # Node 1 cannot reach node 0 because the only route toward 0 requires missing edge 3->4.
    assert minMaxWeight(
        5,
        [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]],
        1,
    ) == -1


def test_chooses_lower_bottleneck_indirect_path():
    # Node 1 can reach 0 directly with weight 10 or via 1->2->0 with maximum weight 3.
    assert minMaxWeight(3, [[1, 0, 10], [1, 2, 2], [2, 0, 3]], 1) == 3


def test_parallel_edges_can_use_the_lighter_weight():
    # Of the two edges from 1 to 0, retaining the weight-3 edge makes the maximum 3.
    assert minMaxWeight(2, [[1, 0, 7], [1, 0, 3]], 1) == 3
