def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0

    # Calculate the number of moves in each direction
    north_moves = s.count('N')
    south_moves = s.count('S')
    east_moves = s.count('E')
    west_moves = s.count('W')

    # Calculate the effect of changing moves
    total_vertical_moves = north_moves - south_moves
    total_horizontal_moves = east_moves - west_moves

    # Calculate the initial manhattan distance
    initial_distance = abs(total_vertical_moves) + abs(total_horizontal_moves)

    # Use the budget k to maximize distance
    max_distance = initial_distance + k

    # Check maximum distance reachable at every position in the string
    for char in s:
        # Update position based on current character
        if char == 'N':
            y += 1
        elif char == 'S':
            y -= 1
        elif char == 'E':
            x += 1
        elif char == 'W':
            x -= 1

        # Calculate current Manhattan distance
        current_distance = abs(x) + abs(y)
        # Update max distance considering k changes
        max_distance = max(max_distance, current_distance + k)

    return max_distance
```
