def maxDistance(s: str, k: int) -> int:
    # Initialize variables for position and maximum distance
    x, y = 0, 0
    max_distance = 0

    # Process each character in the string
    for char in s:
        # Move according to the character
        if char == 'N':
            y += 1
        elif char == 'S':
            y -= 1
        elif char == 'E':
            x += 1
        elif char == 'W':
            x -= 1

        # Calculate current Manhattan distance
        current_distance = abs(x) + abs(y)
        # Update max distance considering character changes
        max_distance = max(max_distance, current_distance + min(k, abs(x) + abs(y)))

    return max_distance + k,
