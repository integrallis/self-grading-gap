def maxDistance(s: str, k: int) -> int:
    x, y = 0, 0
    max_distance = 0

    # Check maximum distance reachable at every position in the string
    for char in s:
        # Update position based on current character
        if char == 'N':
            y += 1
        elif char == 'S':
            y -= 1
        elif char == 'E':
            x += 1
        elif char == 'W':
            x -= 1

        # Calculate current Manhattan distance
        current_distance = abs(x) + abs(y)

        # Update the maximum distance considering k changes
        max_distance = max(max_distance, current_distance + k)

        # We can change the current move to maximize the distance
        if k > 0:
            # If we have budget left, we consider changing the last move to the optimal one
