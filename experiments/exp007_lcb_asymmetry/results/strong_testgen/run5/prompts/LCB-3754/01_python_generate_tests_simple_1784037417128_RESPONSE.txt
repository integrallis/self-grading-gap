from solution import maxDistance


def test_canonical_example_1():
    # Change "NWSE" to "NWNE"; after three moves position is (-1, 2), distance 3.
    assert maxDistance("NWSE", 1) == 3


def test_canonical_example_2():
    # Change "NSWWEW" to "NNWWWW"; after all six moves distance is 6.
    assert maxDistance("NSWWEW", 3) == 6


def test_no_changes_uses_the_farthest_prefix():
    # Positions for "NSEW" have distances 1, 0, 1, 0, so the maximum is 1.
    assert maxDistance("NSEW", 0) == 1


def test_one_change_can_turn_an_opposite_move_into_a_matching_move():
    # Change "NS" to "NN"; after two moves position is (0, 2), distance 2.
    assert maxDistance("NS", 1) == 2


def test_maximum_can_occur_before_the_final_move():
    # Change the third move in "NNSS" to N: "NNNS" reaches distance 3 after move three.
    assert maxDistance("NNSS", 1) == 3


def test_budget_can_change_every_move():
    # Change all moves in "NSEW" to N; after four moves position is (0, 4), distance 4.
    assert maxDistance("NSEW", 4) == 4
