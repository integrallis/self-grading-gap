from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    dp = [1] * n  # dp[i] will hold the length of the longest subsequence ending at index i
    max_len = 1

    for i in range(n):
        for j in range(i):
            if (i == 0) or (abs(nums[i] - nums[j]) >= abs(nums[j] - nums[j - 1])):
                dp[i] = max(dp[i], dp[j] + 1)
        max_len = max(max_len, dp[i])

    return max_len

```
