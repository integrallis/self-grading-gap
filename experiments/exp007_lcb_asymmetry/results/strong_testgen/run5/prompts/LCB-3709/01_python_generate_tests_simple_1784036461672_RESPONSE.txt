from solution import hasSpecialSubstring


def test_canonical_example_1():
    # "aaa" at the end has length 3 and is preceded by different character "b".
    assert hasSpecialSubstring("aaabaaa", 3) is True


def test_canonical_example_2():
    # No length-2 substring of "abc" consists of one repeated character.
    assert hasSpecialSubstring("abc", 2) is False


def test_entire_string_is_a_special_substring():
    # "aaaa" is a uniform substring of length 4 with no neighboring characters.
    assert hasSpecialSubstring("aaaa", 4) is True


def test_run_longer_than_k_is_not_special():
    # Any length-3 "aaa" within "aaaa" has an adjacent "a", so no valid substring exists.
    assert hasSpecialSubstring("aaaa", 3) is False


def test_exact_run_between_different_characters_is_special():
    # "bbb" has length 3 and is bordered by different characters "a" and "c".
    assert hasSpecialSubstring("abbbc", 3) is True


def test_single_character_substring_requires_different_neighbors():
    # The middle "b" has length 1 and both adjacent characters are different ("a" and "c").
    assert hasSpecialSubstring("abc", 1) is True


def test_no_uniform_substring_of_requested_length():
    # Every length-2 substring ("ab", "bc", "cd") contains two distinct characters.
    assert hasSpecialSubstring("abcd", 2) is False
