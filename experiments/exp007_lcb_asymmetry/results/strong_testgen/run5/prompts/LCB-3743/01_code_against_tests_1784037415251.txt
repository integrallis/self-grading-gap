```python
from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    # Calculate gaps between meetings
    gaps = []
    previous_end = 0
    for i in range(len(startTime)):
        if startTime[i] > previous_end:
            gaps.append(startTime[i] - previous_end)
        previous_end = endTime[i]
    if previous_end < eventTime:
        gaps.append(eventTime - previous_end)

    # If we can move k meetings, we need to maximize the gaps we can merge
    # Sort gaps in descending order
    gaps.sort(reverse=True)

    # We can only merge up to k gaps
    max_mergeable_free_time = sum(gaps[i] for i in range(k)) if k <= len(gaps) else sum(gaps)

    # Maximum free time is the sum of all free time after merging
    max_free_time_possible = sum(gaps) + max_mergeable_free_time - (k if k <= len(gaps) else len(gaps))

    return max_free_time_possible
```
