from typing import List

def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    # Calculate gaps between meetings
    gaps = []
    previous_end = 0
    for i in range(len(startTime)):
        if startTime[i] > previous_end:
            gaps.append(startTime[i] - previous_end)
        previous_end = endTime[i]
    if previous_end < eventTime:
        gaps.append(eventTime - previous_end)

    # Calculate total free time as total available time minus the duration of meetings
    total_meeting_duration = sum(end - start for start, end in zip(startTime, endTime))
    total_free_time = eventTime - total_meeting_duration

    # If we can move k meetings, we need to maximize the gaps we can merge
    # Sort gaps in descending order
    gaps.sort(reverse=True)

    # We need to merge gaps considering k movements
    max_free_time_possible = total_free_time
    for i in range(min(k, len(gaps))):
        if i < len(gaps) - 1:
            # Merging gaps i and i+1
            max_free_time_possible += gaps[i] + gaps[i + 1]
    
    return max_free_time_possible
```
