from solution import maxFrequency


def test_canonical_example_1():
    # Adding -5 to nums[2..5] produces [1, 2, -2, -1, 0, 1], so 1 occurs 2 times.
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2


def test_canonical_example_2():
    # Adding 8 to nums[1..9] produces four occurrences of 10.
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4


def test_single_element_can_be_changed_to_k():
    # Add 3 to the only element: [2] becomes [5], so 5 occurs once.
    assert maxFrequency([2], 5) == 1


def test_zero_addition_can_preserve_all_existing_k_values():
    # Add 0 to any subarray; all three existing 4 values remain unchanged.
    assert maxFrequency([4, 4, 4], 4) == 3


def test_one_addition_cannot_convert_different_source_values_to_k():
    # A single x can change either 1 or 2 into 3, but not both, so the maximum is 1.
    assert maxFrequency([1, 2], 3) == 1


def test_optimal_subarray_can_avoid_losing_existing_k_values():
    # There are three 1s; changing either adjacent pair of 2s to 1 adds two, for 5 total.
    assert maxFrequency([2, 2, 1, 1, 1, 2, 2], 1) == 5


def test_neutral_values_inside_subarray_do_not_prevent_converting_matching_values():
    # Add -1 to the whole array: both 2s become 1 while 3 becomes 2, for 2 total 1s.
    assert maxFrequency([2, 3, 2], 1) == 2
