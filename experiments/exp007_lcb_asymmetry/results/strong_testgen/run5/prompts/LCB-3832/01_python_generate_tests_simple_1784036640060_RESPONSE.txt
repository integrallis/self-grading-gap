from solution import minCosts


def test_canonical_example_1():
    # Pay 5, 3, 3, 1; after paying a minimum cost, later positions are reachable free.
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]


def test_canonical_example_2():
    # Pay 1 to swap with person 0, then every later position is reachable for free.
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]


def test_single_person_line():
    # The only position requires swapping with person 0 for cost 7.
    assert minCosts([7]) == [7]


def test_cost_drops_and_ties():
    # Running minimum swap costs are min(4)=4, min(4,2)=2, min(4,2,2)=2,
    # min(4,2,2,3)=2, and min(4,2,2,3,1)=1.
    assert minCosts([4, 2, 2, 3, 1]) == [4, 2, 2, 2, 1]
