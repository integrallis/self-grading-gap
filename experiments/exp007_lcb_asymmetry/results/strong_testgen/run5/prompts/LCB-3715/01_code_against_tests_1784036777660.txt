```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Prepare a list of coin contributions across a number line
    coin_map = {}
    for l, r, c in coins:
        coin_map[l] = coin_map.get(l, 0) + c
        coin_map[r + 1] = coin_map.get(r + 1, 0) - c

    # Create a sorted list of positions
    positions = sorted(coin_map.keys())
    max_coins = 0
    current_coins = 0
    idx = 0
    position_count = 0

    # Calculate the contributions using a sliding window
    for pos in range(positions[0], positions[-1] + 1):
        if pos in coin_map:
            current_coins += coin_map[pos]
        if position_count < k:
            position_count += 1
        else:
            # Maintain window size of k
            left_position = pos - k
            if left_position in coin_map:
                current_coins -= coin_map[left_position]

        # Update max coins
        max_coins = max(max_coins, current_coins)

    return max_coins

```
