from solution import zigzagTraversal


def test_canonical_example_1():
    # Zigzag order is [1, 2, 4, 3]; taking every alternate cell gives [1, 4].
    assert zigzagTraversal([[1, 2], [3, 4]]) == [1, 4]


def test_canonical_example_2():
    # Zigzag order is [2, 1, 1, 2, 2, 1]; taking every alternate cell gives [2, 1, 2].
    assert zigzagTraversal([[2, 1], [2, 1], [2, 1]]) == [2, 1, 2]


def test_canonical_example_3():
    # Zigzag order is [1, 2, 3, 6, 5, 4, 7, 8, 9]; taking every alternate cell gives [1, 3, 5, 7, 9].
    assert zigzagTraversal([[1, 2, 3], [4, 5, 6], [7, 8, 9]]) == [1, 3, 5, 7, 9]


def test_zigzag_skips_continue_across_an_even_width_row_boundary():
    # Zigzag order is [10, 11, 12, 13, 17, 16, 15, 14]; taking every alternate cell gives [10, 12, 17, 15].
    assert zigzagTraversal([[10, 11, 12, 13], [14, 15, 16, 17]]) == [10, 12, 17, 15]
