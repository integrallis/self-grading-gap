HERE: mentions all online users.




Offline Event: ["OFFLINE", "timestamp_i", "id_i"]

This event indicates that the user id_i had become offline at timestamp_i for 60 time units. The user will automatically be online again at time timestamp_i + 60.



Return an array mentions where mentions[i] represents the number of mentions the user with id i has across all MESSAGE events.
All users are initially online, and if a user goes offline or comes back online, their status change is processed before handling any message event that occurs at the same timestamp.

```python
from typing import List


def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers
    online_status = [True] * numberOfUsers
    user_offline_until = [0] * numberOfUsers

    for event in sorted(events, key=lambda x: int(x[1])):
        event_type, timestamp, *args = event
        timestamp = int(timestamp)

        if event_type == "OFFLINE":
            user_id = int(args[0])
            online_status[user_id] = False
            user_offline_until[user_id] = timestamp + 60
        elif event_type == "MESSAGE":
            mentions_string = args[0]
            mentioned_users = set()

            if mentions_string == "ALL":
                mentioned_users = set(range(numberOfUsers))
            elif mentions_string == "HERE":
                mentioned_users = {i for i in range(numberOfUsers) if online_status[i]}
            else:
                for mention in mentions_string.split():
                    if mention.startswith("id"):
                        user_id = int(mention[2:])
                        if user_id not in mentioned_users:
                            mentioned_users.add(user_id)

            for user_id in mentioned_users:
                if online_status[user_id] or timestamp < user_offline_until[user_id]:
                    mentions[user_id] += 1

            # Process user online status changes
            for i in range(numberOfUsers):
                if not online_status[i] and timestamp >= user_offline_until[i]:
                    online_status[i] = True

    return mentions

```
