from solution import countMentions


def test_canonical_example_1():
    events = [
        ["MESSAGE", "10", "id1 id0"],
        ["OFFLINE", "11", "0"],
        ["MESSAGE", "71", "HERE"],
    ]

    # At time 10 both users are mentioned; at time 71 user 0 is online again,
    # so HERE adds one mention to each user: [1, 1] + [1, 1] = [2, 2].
    assert countMentions(2, events) == [2, 2]


def test_canonical_example_2():
    events = [
        ["MESSAGE", "10", "id1 id0"],
        ["OFFLINE", "11", "0"],
        ["MESSAGE", "12", "ALL"],
    ]

    # The first message mentions both users, and ALL mentions both users even
    # though user 0 is offline: [1, 1] + [1, 1] = [2, 2].
    assert countMentions(2, events) == [2, 2]


def test_canonical_example_3():
    events = [
        ["OFFLINE", "10", "0"],
        ["MESSAGE", "12", "HERE"],
    ]

    # User 0 remains offline at time 12, so HERE mentions only user 1.
    assert countMentions(2, events) == [0, 1]


def test_explicit_id_mentions_count_duplicates_separately():
    events = [["MESSAGE", "5", "id0 id0 id1"]]

    # id0 appears twice and id1 appears once in the message.
    assert countMentions(2, events) == [2, 1]


def test_explicit_id_mention_can_target_an_offline_user():
    events = [
        ["OFFLINE", "10", "0"],
        ["MESSAGE", "11", "id0"],
    ]

    # Explicit id0 mentions count even while user 0 is offline.
    assert countMentions(2, events) == [1, 0]


def test_offline_status_is_processed_before_message_at_same_timestamp():
    events = [
        ["MESSAGE", "10", "HERE"],
        ["OFFLINE", "10", "0"],
    ]

    # The offline status change at time 10 is processed before HERE, so only
    # user 1 is mentioned.
    assert countMentions(2, events) == [0, 1]


def test_user_returns_online_exactly_sixty_time_units_after_offline():
    events = [
        ["OFFLINE", "10", "0"],
        ["MESSAGE", "70", "HERE"],
    ]

    # User 0 returns at time 10 + 60 = 70, so HERE mentions both users.
    assert countMentions(2, events) == [1, 1]
