from solution import maxSum


def test_canonical_example_1():
    # All five values are unique, so their sum is 1 + 2 + 3 + 4 + 5 = 15.
    assert maxSum([1, 2, 3, 4, 5]) == 15


def test_canonical_example_2():
    # Delete duplicates and 0; selecting one unique 1 gives sum 1.
    assert maxSum([1, 1, 0, 1, 1]) == 1


def test_canonical_example_3():
    # Delete -1 and -2, then select unique values [2, 1] for 2 + 1 = 3.
    assert maxSum([1, 2, -1, -2, 1, 0, -1]) == 3


def test_keeps_each_distinct_positive_value_once():
    # Delete one 2; the unique subarray [2, 3] has sum 2 + 3 = 5.
    assert maxSum([2, 2, 3]) == 5


def test_deletions_allow_skipping_negative_values_between_positives():
    # Delete -10 and one 4; the unique subarray [4, 6] has sum 4 + 6 = 10.
    assert maxSum([4, -10, 6, 4]) == 10


def test_all_negative_values_selects_the_largest_single_value():
    # A nonempty selection is required; [-2] is the maximum-sum unique subarray.
    assert maxSum([-5, -2, -2, -3]) == -2


def test_all_zero_values_have_maximum_sum_zero():
    # Select one 0 so all elements are unique; its sum is 0.
    assert maxSum([0, 0, 0]) == 0
