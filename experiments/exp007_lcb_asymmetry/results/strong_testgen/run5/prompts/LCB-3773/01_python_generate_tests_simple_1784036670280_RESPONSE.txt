from solution import minimumPairRemoval


def test_canonical_example_1():
    # (3, 1) sums to 4 -> [5, 2, 4]; then (2, 4) sums to 6 -> [5, 6].
    assert minimumPairRemoval([5, 2, 3, 1]) == 2


def test_canonical_example_2():
    # 1 <= 2 <= 2, so the array is already non-decreasing.
    assert minimumPairRemoval([1, 2, 2]) == 0


def test_single_element_is_non_decreasing():
    # A one-element array has no previous element to violate non-decreasing order.
    assert minimumPairRemoval([7]) == 0


def test_leftmost_pair_is_chosen_when_minimum_sums_tie():
    # All initial adjacent sums are 1, so merge the leftmost (2, -1) -> [1, 2, -1].
    # Then merge the minimum-sum pair (2, -1) -> [1, 1], which is non-decreasing.
    assert minimumPairRemoval([2, -1, 2, -1]) == 2
