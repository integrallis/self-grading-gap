from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Sort the numbers to easily calculate min and max contributions
    nums.sort()

    # Calculate contributions for each element considering its min and max role
    for i in range(n):
        # Number of ways to choose the remaining elements
        min_contrib = (pow(2, min(k, n - 1), MOD) - 1) % MOD
        max_contrib = (pow(2, min(k, n - 1), MOD) - 1) % MOD

        # Add to total sum considering this number as min and max
        total_sum += (nums[i] * (min_contrib + max_contrib)) % MOD
        total_sum %= MOD

    return total_sum

```
