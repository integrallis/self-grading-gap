## Implementation

```python
from typing import List

MOD = 10**9 + 7

def minMaxSums(nums: List[int], k: int) -> int:
    n = len(nums)
    total_sum = 0

    # Calculate contribution of each element as min and max
    for i in range(n):
        # Count how many subsequences of size up to k include nums[i] as min
        min_contrib = (1 << (min(k, n - 1))) - 1
        # Count how many subsequences of size up to k include nums[i] as max
        max_contrib = (1 << (min(k, n - 1))) - 1
        total_sum += (nums[i] * (min_contrib + max_contrib)) % MOD
        total_sum %= MOD

    return total_sum
```
