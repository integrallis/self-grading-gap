from solution import minMaxSums


def test_canonical_example_1():
    # Singles contribute 2 + 4 + 6 = 12; pairs contribute 3 + 4 + 5 = 12.
    assert minMaxSums([1, 2, 3], 2) == 24


def test_canonical_example_2():
    # With k = 1, each element contributes itself as both min and max:
    # 2*5 + 2*0 + 2*6 = 22.
    assert minMaxSums([5, 0, 6], 1) == 22


def test_canonical_example_3():
    # Three singleton subsequences contribute 3 * 2, and three pairs contribute 3 * 2.
    assert minMaxSums([1, 1, 1], 2) == 12


def test_single_element_subsequence():
    # The only subsequence is [7], whose minimum plus maximum is 7 + 7 = 14.
    assert minMaxSums([7], 1) == 14


def test_includes_subsequences_up_to_array_length():
    # Singles: 4 + 8 + 14 = 26; pairs: 6 + 9 + 11 = 26;
    # triple: 2 + 7 = 9; total = 26 + 26 + 9 = 61.
    assert minMaxSums([2, 4, 7], 3) == 61


def test_duplicate_values_are_counted_by_subsequence_occurrence():
    # Singles: 4 + 4 + 10 = 18; pairs: [2,2] gives 4 and each [2,5]
    # gives 7, so 4 + 7 + 7 = 18; total = 36.
    assert minMaxSums([2, 2, 5], 2) == 36


def test_returns_result_modulo_1000000007():
    # For k = 2, singles contribute 4,000,000,000 and the pair contributes
    # 2,000,000,000, for 6,000,000,000 total. Modulo 1,000,000,007 this is 999,999,965.
    assert minMaxSums([1_000_000_000, 1_000_000_000], 2) == 999_999_965
