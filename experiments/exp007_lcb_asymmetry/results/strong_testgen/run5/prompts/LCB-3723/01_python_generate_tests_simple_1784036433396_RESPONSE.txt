from solution import sumOfGoodNumbers


def test_canonical_example_1():
    # Good values are 3, 5, and 4, so the sum is 3 + 5 + 4 = 12.
    assert sumOfGoodNumbers([1, 3, 2, 1, 5, 4], 2) == 12


def test_canonical_example_2():
    # Only 2 is greater than its existing k-distance neighbor 1, so the sum is 2.
    assert sumOfGoodNumbers([2, 1], 1) == 2


def test_equal_k_distance_values_are_not_good():
    # Each value equals its existing k-distance neighbor, so none is strictly greater; sum = 0.
    assert sumOfGoodNumbers([7, 7, 7, 7], 2) == 0


def test_interior_value_must_exceed_both_existing_k_distance_neighbors():
    # Good values are 10 (greater than 1 and 4) and 3 (greater than 2), so sum = 10 + 3 = 13.
    assert sumOfGoodNumbers([1, 2, 10, 3, 4], 2) == 13
