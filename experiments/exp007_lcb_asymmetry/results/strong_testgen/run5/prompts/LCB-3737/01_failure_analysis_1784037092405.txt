from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize the dp array to hold costs for painting houses
    dp = [[0] * 3 for _ in range(n)]
    # Initialize the first house's costs
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the dp array for houses 1 to n-1
    for i in range(1, n):
        for j in range(3):
            # Calculate minimum cost excluding the same color for adjacent houses
            dp[i][j] = cost[i][j] + min(dp[i-1][(j+1) % 3], dp[i-1][(j+2) % 3])

    # Handle symmetry constraint for mirrored houses
    total_cost = float('inf')
    for j in range(3):
        # Calculate the total cost when the last house has color j
        total_cost = min(total_cost, dp[n-1][j] + min(dp[n//2-1][(j+1) % 3], dp[n//2-1][(j+2) % 3]))

    return total_cost
