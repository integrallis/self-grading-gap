from solution import largestInteger


def test_canonical_example_1():
    # 3 and 7 each appear in one size-3 subarray; the larger value is 7.
    assert largestInteger([3, 9, 2, 1, 7], 3) == 7


def test_canonical_example_2():
    # Only 3 appears in exactly one size-4 subarray.
    assert largestInteger([3, 9, 7, 2, 1, 7], 4) == 3


def test_canonical_example_3():
    # Each 0 appears in a different size-1 subarray, so 0 appears in two subarrays.
    assert largestInteger([0, 0], 1) == -1


def test_k_one_returns_largest_value_appearing_once():
    # Size-1 subarrays are [4], [2], [4], [3]; values appearing once are 2 and 3.
    assert largestInteger([4, 2, 4, 3], 1) == 3


def test_duplicate_within_one_subarray_counts_that_subarray_once():
    # The only size-3 subarray is [5, 5, 1]; both 5 and 1 appear in one subarray.
    assert largestInteger([5, 5, 1], 3) == 5


def test_repeated_value_across_overlapping_subarrays_is_not_almost_missing():
    # Size-3 subarrays are [8, 8, 1] and [8, 1, 2]; only 2 appears in exactly one.
    assert largestInteger([8, 8, 1, 2], 3) == 2
