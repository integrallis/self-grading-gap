from solution import findValidPair


def test_canonical_example_1():
    # '2' appears twice and '3' appears three times; the first valid adjacent pair is "23".
    assert findValidPair("2523533") == "23"


def test_canonical_example_2():
    # '2' appears twice and '1' appears once; the valid adjacent pair is "21".
    assert findValidPair("221") == "21"


def test_canonical_example_3():
    # The only adjacent pair is "22", whose digits are equal, so no valid pair exists.
    assert findValidPair("22") == ""


def test_returns_first_when_multiple_valid_pairs_exist():
    # '2' appears twice and '1' appears once; both "21" and "12" are valid, so return first "21".
    assert findValidPair("212") == "21"


def test_returns_pair_at_start_when_its_digit_counts_match_values():
    # '1' appears once and '2' appears twice, so the first adjacent pair "12" is valid.
    assert findValidPair("122") == ""
