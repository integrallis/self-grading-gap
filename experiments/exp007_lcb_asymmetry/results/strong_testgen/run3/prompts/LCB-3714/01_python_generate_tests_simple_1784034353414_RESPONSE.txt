from solution import minMaxSums


def test_canonical_example_1():
    # Singletons contribute 2 + 4 + 6, and pairs contribute 3 + 4 + 5: total 24.
    assert minMaxSums([1, 2, 3], 2) == 24


def test_canonical_example_2():
    # With k = 1, each element contributes itself as both minimum and maximum:
    # 2 * (5 + 0 + 6) = 22.
    assert minMaxSums([5, 0, 6], 1) == 22


def test_canonical_example_3():
    # There are three singleton and three two-element subsequences; each contributes 1 + 1:
    # 6 * 2 = 12.
    assert minMaxSums([1, 1, 1], 2) == 12


def test_includes_subsequences_of_every_size_up_to_k():
    # Singletons: 4 + 8 + 14 = 26; pairs: 6 + 9 + 11 = 26;
    # triple: 2 + 7 = 9; total 26 + 26 + 9 = 61.
    assert minMaxSums([2, 4, 7], 3) == 61


def test_handles_unsorted_input():
    # Singletons contribute 6 + 2 + 4 = 12; pairs contribute
    # (3 + 1) + (3 + 2) + (1 + 2) = 4 + 5 + 3 = 12; total 24.
    assert minMaxSums([3, 1, 2], 2) == 24


def test_returns_result_modulo_1000000007():
    # With k = 1, the two singletons contribute 2_000_000_000 each:
    # 4_000_000_000 mod 1_000_000_007 = 999_999_979.
    assert minMaxSums([1_000_000_000, 1_000_000_000], 1) == 999_999_979


def test_k_at_its_maximum_allowed_value():
    # All 2^70 - 1 nonempty subsequences are included, each contributing 2.
    # 2 * (2^70 - 1) mod 1_000_000_007 = 540_032_504.
    assert minMaxSums([1] * 70, 70) == 540_032_504
