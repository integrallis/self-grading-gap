```python
from typing import List


def minMaxSums(nums: List[int], k: int) -> int:
    MOD = 10**9 + 7
    n = len(nums)
    nums.sort()

    total_sum = 0
    # Contribution from single elements
    total_sum += 2 * sum(nums) % MOD

    # Calculate contributions from combinations of elements up to size k
    for i in range(1, k + 1):
        min_contribution = sum(nums[:i]) % MOD
        max_contribution = sum(nums[-i:]) % MOD
        total_sum += (min_contribution + max_contribution) * (1 << (n - i)) % MOD
        total_sum %= MOD

    return total_sum

```
