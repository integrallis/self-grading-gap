from solution import minimumPairRemoval


def test_canonical_example_1():
    # (3, 1) has minimum sum 4 -> [5, 2, 4];
    # then (2, 4) has minimum sum 6 -> [5, 6], so 2 operations.
    assert minimumPairRemoval([5, 2, 3, 1]) == 2


def test_canonical_example_2():
    # [1, 2, 2] is already non-decreasing, so no operations are needed.
    assert minimumPairRemoval([1, 2, 2]) == 0


def test_single_element_is_non_decreasing():
    # A one-element array has no previous element to violate ordering, so 0 operations.
    assert minimumPairRemoval([7]) == 0


def test_merges_minimum_sum_pair_until_sorted():
    # (1, 3) has minimum sum 4 -> [4, 4], which is non-decreasing after 1 operation.
    assert minimumPairRemoval([4, 1, 3]) == 1


def test_tied_minimum_sum_uses_leftmost_pair():
    # Both adjacent sums are 3; merge the leftmost (2, 1) -> [3, 2].
    # Then merge (3, 2) -> [5], so 2 operations.
    assert minimumPairRemoval([2, 1, 2]) == 2


def test_negative_values_follow_minimum_pair_sum_rule():
    # (-2, -3) has minimum sum -5 -> [-1, -5];
    # then merge (-1, -5) -> [-6], so 2 operations.
    assert minimumPairRemoval([-1, -2, -3]) == 2
