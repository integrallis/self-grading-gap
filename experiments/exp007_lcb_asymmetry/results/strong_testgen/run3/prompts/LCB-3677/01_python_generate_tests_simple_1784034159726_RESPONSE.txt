from solution import maximumAmount


def test_canonical_example_1():
    # Best route: 0 + 1 + neutralized(-2) + 3 + 4 = 8.
    assert maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) == 8


def test_canonical_example_2():
    # A top-row-then-down route collects 10 + 10 + 10 + 10 = 40.
    assert maximumAmount([[10, 10, 10], [10, 10, 10]]) == 40


def test_can_neutralize_robbers_at_start_and_end_of_path():
    # The only path is -5, -1, -10, -2; neutralize -5 and -10:
    # -1 + -2 = -3.
    assert maximumAmount([[-5, -1, -10, -2]]) == -3


def test_returns_negative_profit_when_more_than_two_robbers_are_unavoidable():
    # The only path totals -5 + -4 + -3 + -2 = -14; neutralizing
    # the two largest thefts (-5 and -4) improves it by 9, giving -5.
    assert maximumAmount([[-5], [-4], [-3], [-2]]) == -5


def test_single_robber_cell_can_be_neutralized():
    # The start and destination are the same robber cell; neutralizing -7 gives 0.
    assert maximumAmount([[-7]]) == 0
