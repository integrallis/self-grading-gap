If coins[i][j] >= 0, the robot gains that many coins.
If coins[i][j] < 0, the robot encounters a robber, and the robber steals the absolute value of coins[i][j] coins.

The robot has a special ability to neutralize robbers in at most 2 cells on its path, preventing them from stealing coins in those cells.
Note: The robot's total coins can be negative.

Move to (1, 2), gaining 3 coins (total coins = 1 + 3 = 4).

```python
from typing import List


def maximumAmount(coins: List[List[int]]) -> int:
    m = len(coins)
    n = len(coins[0]) if m > 0 else 0
    dp = [[[float('-inf')] * 3 for _ in range(n)] for _ in range(m)]
    dp[0][0][0] = coins[0][0]  # Start at the top-left corner

    # Fill the dp array
    for i in range(m):
        for j in range(n):
            for k in range(3):  # k is the number of neutralizations used
                if i == 0 and j == 0:
                    continue
                if i > 0:
                    # Move from above
                    current_coins = dp[i-1][j][k]
                    if coins[i][j] < 0 and k > 0:
                        current_coins = max(current_coins, dp[i-1][j][k-1] + coins[i][j])
                    elif coins[i][j] >= 0:
                        current_coins = max(current_coins, dp[i-1][j][k] + coins[i][j])
                    dp[i][j][k] = max(dp[i][j][k], current_coins)
                if j > 0:
                    # Move from left
                    current_coins = dp[i][j-1][k]
                    if coins[i][j] < 0 and k > 0:
                        current_coins = max(current_coins, dp[i][j-1][k-1] + coins[i][j])
                    elif coins[i][j] >= 0:
                        current_coins = max(current_coins, dp[i][j-1][k] + coins[i][j])
                    dp[i][j][k] = max(dp[i][j][k], current_coins)

    # Get the maximum value from the bottom-right corner with up to 2 neutralizations
    return max(dp[m-1][n-1][0], dp[m-1][n-1][1], dp[m-1][n-1][2])
```
