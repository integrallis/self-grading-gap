from solution import maxDistance


def test_canonical_example_1():
    # Change "NWSE" to "NWNE": after three moves position is (-1, 2), distance 1 + 2 = 3.
    assert maxDistance("NWSE", 1) == 3


def test_canonical_example_2():
    # Change to "NNWWWW": after all six moves position is (-4, 2), distance 4 + 2 = 6.
    assert maxDistance("NSWWEW", 3) == 6


def test_no_changes_uses_best_intermediate_position():
    # "NNSS" reaches (0, 2) after two moves, distance 2, although it ends at the origin.
    assert maxDistance("NNSS", 0) == 2


def test_single_move_has_distance_one_without_changes():
    # One south move reaches (0, -1), whose Manhattan distance is 1.
    assert maxDistance("S", 0) == 1


def test_changes_can_make_every_move_point_away_from_origin():
    # Change the first "S" in "SNNN" to "N", producing "NNNN" and distance 4.
    assert maxDistance("SNNN", 1) == 4


def test_at_most_k_changes_can_leave_an_already_optimal_path_unchanged():
    # "NNN" already reaches (0, 3), distance 3; no path of three moves can exceed distance 3.
    assert maxDistance("NNN", 2) == 3


def test_change_can_target_a_prefix_before_later_moves():
    # Change the first "S" to "W": "WWNE" reaches (-2, 1) after three moves, distance 3.
    assert maxDistance("SWNE", 1) == 3
