from typing import List

def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers  # Initialize mention counts
    online_status = [True] * numberOfUsers  # Track online status
    event_queue = []  # Queue to process events

    for event in events:
        event_type, timestamp, *details = event
        timestamp = int(timestamp)
        if event_type == 'OFFLINE':
            user_id = int(details[0])
            online_status[user_id] = False  # User goes offline
            # Schedule user to come back online after 60 time units
            event_queue.append((timestamp + 60, user_id, 'ONLINE'))
        elif event_type == 'MESSAGE':
            # Process any queued events before this message
            while event_queue and event_queue[0][0] <= timestamp:
                _, user_id, status = event_queue.pop(0)
                online_status[user_id] = (status == 'ONLINE')
            mentions_string = details[0]
            # Count mentions
            if mentions_string == 'ALL':
                for i in range(numberOfUsers):
                    mentions[i] += 1  # Mention all users
            elif mentions_string == 'HERE':
                for i in range(numberOfUsers):
                    if online_status[i]:
                        mentions[i] += 1  # Mention only online users
            else:
                for mention in mentions_string.split():
                    if mention.startswith('id'):
                        user_id = int(mention[2:])
                        mentions[user_id] += 1  # Count specific user mentions
    return mentions
