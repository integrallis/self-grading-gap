from solution import countMentions


def test_canonical_example_1():
    events = [
        ["MESSAGE", "10", "id1 id0"],
        ["OFFLINE", "11", "0"],
        ["MESSAGE", "71", "HERE"],
    ]

    # At t=10 both users are mentioned once; at t=71 user 0 is online again,
    # so HERE adds one mention to each user: [1, 1] + [1, 1] = [2, 2].
    assert countMentions(2, events) == [2, 2]


def test_canonical_example_2():
    events = [
        ["MESSAGE", "10", "id1 id0"],
        ["OFFLINE", "11", "0"],
        ["MESSAGE", "12", "ALL"],
    ]

    # The explicit message gives [1, 1], and ALL mentions both users even
    # though user 0 is offline: [1, 1] + [1, 1] = [2, 2].
    assert countMentions(2, events) == [2, 2]


def test_canonical_example_3():
    events = [
        ["OFFLINE", "10", "0"],
        ["MESSAGE", "12", "HERE"],
    ]

    # User 0 remains offline at t=12, so HERE mentions only user 1: [0, 1].
    assert countMentions(2, events) == [0, 1]


def test_explicit_ids_count_offline_users_and_duplicates_separately():
    events = [
        ["OFFLINE", "5", "1"],
        ["MESSAGE", "6", "id1 id1 id0 id1"],
    ]

    # Explicit ids can mention offline users; id0 appears once and id1 appears
    # three times, so the counts are [1, 3].
    assert countMentions(2, events) == [1, 3]


def test_here_mentions_all_users_initially_online():
    events = [
        ["MESSAGE", "1", "HERE"],
    ]

    # All three users start online, so HERE gives each user one mention: [1, 1, 1].
    assert countMentions(3, events) == [1, 1, 1]


def test_offline_status_at_same_timestamp_is_processed_before_here_message():
    events = [
        ["MESSAGE", "10", "HERE"],
        ["OFFLINE", "10", "1"],
    ]

    # The offline change is processed before the same-timestamp message, so HERE
    # mentions users 0 and 2 but not user 1: [1, 0, 1].
    assert countMentions(3, events) == [1, 0, 1]
