from solution import longestSubsequence


def test_canonical_example_1():
    # [16, 6, 3] has differences [10, 3], and 10 >= 3, so all 3 elements qualify.
    assert longestSubsequence([16, 6, 3]) == 3


def test_canonical_example_2():
    # [6, 4, 2, 1] has differences [2, 2, 1], so the longest valid subsequence has length 4.
    assert longestSubsequence([6, 5, 3, 4, 2, 1]) == 4


def test_canonical_example_3():
    # [10, 20, 10, 19, 10] has differences [10, 10, 9, 9], so the longest length is 5.
    assert longestSubsequence([10, 20, 10, 19, 10, 20]) == 5


def test_all_equal_values_allow_zero_differences():
    # [7, 7, 7, 7] has differences [0, 0, 0], which are non-increasing, so length is 4.
    assert longestSubsequence([7, 7, 7, 7]) == 4


def test_absolute_differences_allow_direction_changes():
    # [1, 5, 2, 4] has absolute differences [4, 3, 2], so all 4 elements qualify.
    assert longestSubsequence([1, 5, 2, 4]) == 4


def test_subsequence_can_skip_elements_to_make_differences_non_increasing():
    # [1, 4, 7] is a subsequence with differences [3, 3], so length 3 is achievable;
    # no length-4 subsequence works because [1, 2, 4, 7] has differences [1, 2, 3].
    assert longestSubsequence([1, 2, 4, 7]) == 3


def test_two_elements_always_form_a_valid_subsequence():
    # A two-element subsequence has only one difference, so [1, 300] has valid length 2.
    assert longestSubsequence([1, 300]) == 2
