```python
from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    # Initialize a DP array to store the minimum cost for each house
    dp = [[0] * 3 for _ in range(n)]

    # Fill the DP table for the first house
    for j in range(3):
        dp[0][j] = cost[0][j]

    # Fill the DP table for the remaining houses with respect to colors
    for i in range(1, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(dp[i-1][(j+1) % 3], dp[i-1][(j+2) % 3])

    # To handle equidistant houses, calculate the minimum cost
    result = float('inf')
    for color1 in range(3):
        for color2 in range(3):
            if color1 != color2:
                # Ensure that the color at position n/2 is different from the end colors
                cost_with_equidistant = (dp[n-1][color1] + dp[n//2-1][color2])
                result = min(result, cost_with_equidistant)

    return result
```
