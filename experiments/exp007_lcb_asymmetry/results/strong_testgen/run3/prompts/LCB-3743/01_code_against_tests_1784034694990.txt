```python
from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_times = []
    last_end = 0

    # Account for free time before the first meeting
    if startTime and startTime[0] > 0:
        free_times.append(startTime[0] - last_end)

    # Gather free time gaps between meetings
    for i in range(n):
        if startTime[i] > last_end:
            free_times.append(startTime[i] - last_end)
        last_end = endTime[i]

    # Add free time after the last meeting up to event time
    if last_end < eventTime:
        free_times.append(eventTime - last_end)

    # If k is 0, return the largest free time
    if k == 0:
        return max(free_times) if free_times else 0

    # Try to merge free time by rescheduling meetings
    max_free_time = 0
    for i in range(len(free_times) - k):
        # Calculate the total free time that can be obtained by merging k gaps
        current_free_time = sum(free_times[i:i + k + 1])
        max_free_time = max(max_free_time, current_free_time)

    return max_free_time
```
