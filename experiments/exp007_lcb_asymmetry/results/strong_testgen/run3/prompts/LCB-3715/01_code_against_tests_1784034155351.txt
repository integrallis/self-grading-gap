```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a dictionary to store the number of coins in each bag position
    coin_map = {}

    # Fill the coin_map with coins for each bag position
    for l, r, c in coins:
        for pos in range(l, r + 1):
            coin_map[pos] = coin_map.get(pos, 0) + c

    # Find the maximum coins in any window of size k
    max_coins = 0
    current_coins = 0

    # Get sorted positions from the coin_map
    sorted_positions = sorted(coin_map.keys())
    num_positions = len(sorted_positions)

    # Initialize the sliding window sum
    start_index = 0
    for end_index in range(num_positions):
        current_coins += coin_map[sorted_positions[end_index]]

        # If the window exceeds size k, slide the window
        while sorted_positions[end_index] - sorted_positions[start_index] + 1 > k:
            current_coins -= coin_map[sorted_positions[start_index]]
            start_index += 1

        # Update the max_coins with the current window's sum
        if sorted_positions[end_index] - sorted_positions[start_index] + 1 <= k:
            max_coins = max(max_coins, current_coins)

    return max_coins
```
