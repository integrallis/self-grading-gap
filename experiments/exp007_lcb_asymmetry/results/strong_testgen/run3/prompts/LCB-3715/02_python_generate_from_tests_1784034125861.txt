You are given a 2D array coins, where coins[i] = [l_i, r_i, c_i] denotes that every bag from l_i to r_i contains c_i coins.

Return the maximum amount of coins you can obtain by collecting k consecutive bags.

```python
from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a dictionary to store the number of coins in each bag position
    coin_map = {}

    # Fill the coin_map with coins for each bag position
    for l, r, c in coins:
        for pos in range(l, r + 1):
            coin_map[pos] = coin_map.get(pos, 0) + c

    # Find the maximum coins in any window of size k
    max_coins = 0
    current_coins = 0

    # Use a sliding window to sum the coins over the first k positions
    for i in range(1, k + 1):
        current_coins += coin_map.get(i, 0)

    max_coins = current_coins

    # Slide the window over the remaining positions
    for i in range(k + 1, max(coin_map.keys()) + 1):
        current_coins += coin_map.get(i, 0) - coin_map.get(i - k, 0)
        max_coins = max(max_coins, current_coins)

    return max_coins
```
