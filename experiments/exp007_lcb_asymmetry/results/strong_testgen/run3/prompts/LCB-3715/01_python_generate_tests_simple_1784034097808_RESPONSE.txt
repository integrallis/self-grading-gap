from solution import maximumCoins


def test_canonical_example_1():
    # Bags [3, 4, 5, 6] contain 2 + 0 + 4 + 4 = 10 coins.
    assert maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) == 10


def test_canonical_example_2():
    # Any two bags within [1, 10] contain 3 + 3 = 6 coins.
    assert maximumCoins([[1, 10, 3]], 2) == 6


def test_window_includes_gap_between_segments():
    # Bags [1, 2, 3, 4, 5] contain 4 + 4 + 0 + 3 + 3 = 14 coins.
    assert maximumCoins([[1, 2, 4], [4, 5, 3]], 5) == 14


def test_optimal_window_can_start_inside_a_segment():
    # Bags [5, 6, 7, 8] contain 1 + 10 + 10 + 10 = 31 coins.
    assert maximumCoins([[1, 5, 1], [6, 8, 10]], 4) == 31


def test_k_can_exceed_occupied_segment_length():
    # A length-10 window can include both occupied bags, containing 7 + 7 = 14 coins.
    assert maximumCoins([[5, 6, 7]], 10) == 14
