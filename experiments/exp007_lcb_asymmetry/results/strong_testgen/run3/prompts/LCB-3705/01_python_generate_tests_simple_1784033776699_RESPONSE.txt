from solution import largestInteger


def test_canonical_example_1():
    # 3 and 7 each appear in exactly one size-3 subarray; the larger is 7.
    assert largestInteger([3, 9, 2, 1, 7], 3) == 7


def test_canonical_example_2():
    # Only 3 appears in exactly one size-4 subarray.
    assert largestInteger([3, 9, 7, 2, 1, 7], 4) == 3


def test_canonical_example_3():
    # 0 appears in both size-1 subarrays, so no value appears in exactly one.
    assert largestInteger([0, 0], 1) == -1


def test_single_subarray_returns_largest_distinct_value():
    # With k equal to the array length, there is one subarray containing 2, 4, and 9;
    # all appear in exactly one subarray, so return 9.
    assert largestInteger([4, 9, 2, 9], 4) == 9


def test_repeated_value_within_one_subarray_counts_once():
    # There is one size-2 subarray, [5, 5]; value 5 appears in that one subarray.
    assert largestInteger([5, 5], 2) == 5


def test_size_one_subarrays_require_value_to_occur_once():
    # For k=1, 1 appears in two subarrays, while 2 and 4 each appear once;
    # the largest qualifying value is 4.
    assert largestInteger([1, 4, 1, 2], 1) == 4


def test_zero_can_be_the_largest_almost_missing_integer():
    # The only size-1 subarray is [0], so 0 appears in exactly one subarray.
    assert largestInteger([0], 1) == 0
