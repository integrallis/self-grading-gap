from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Create an adjacency list with reverse edges to check reachability to node 0
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[v].append((u, w))  # Reverse the direction for reachability check

    # Function to check if all nodes can reach node 0 with max edge weight <= max_weight
    def can_reach_with_max_weight(max_weight: int) -> bool:
        out_degree = [0] * n   # Track out-degrees for each node
        reachable_from_zero = [False] * n   # Track reachability to node 0
        queue = []

        # Start by adding all nodes that can connect to 0 with a weight <= max_weight
        for node in range(n):
            for neighbor, weight in graph[node]:
                if weight <= max_weight:
                    out_degree[node] += 1
                    if out_degree[node] <= threshold:
                        queue.append(node)

        visited = [False] * n   # Visited nodes set
        while queue:
            node = queue.pop(0)
            if not visited[node]:
                visited[node] = True
                for neighbor, weight in graph[node]:
                    if weight <= max_weight and out_degree[node] <= threshold:
                        reachable_from_zero[neighbor] = True
                        for next_neighbor, next_weight in graph[neighbor]:  # Explore edges backwards
                            if next_weight <= max_weight and out_degree[neighbor] < threshold:
                                out_degree[neighbor] += 1
                                queue.append(next_neighbor)

        return all(reachable_from_zero[i] for i in range(n))  # Check if all nodes can reach node 0

    # Binary search on maximum edge weight
    left, right = 1, 10**6
    answer = -1

    while left <= right:
        mid = (left + right) // 2
        if can_reach_with_max_weight(mid):
            answer = mid
            right = mid - 1
        else:
            left = mid + 1

    return answer
