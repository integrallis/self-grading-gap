from solution import minMaxWeight


def test_canonical_example_1():
    edges = [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]]

    # Remove 2 -> 0 (weight 2); every node reaches 0 using only weight-1 edges.
    assert minMaxWeight(5, edges, 2) == 1


def test_canonical_example_2():
    edges = [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]]

    # Node 2 has no directed path to node 0, so satisfying reachability is impossible.
    assert minMaxWeight(5, edges, 1) == -1


def test_canonical_example_3():
    edges = [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]]

    # Keep 1 -> 2 -> 3 -> 4 -> 0; its largest edge weight is 2.
    assert minMaxWeight(5, edges, 1) == 2


def test_canonical_example_4():
    edges = [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]]

    # Node 3 has no path to node 0, so satisfying reachability is impossible.
    assert minMaxWeight(5, edges, 1) == -1


def test_uses_lower_weight_indirect_path_instead_of_heavy_direct_edge():
    edges = [[1, 0, 10], [1, 2, 4], [2, 0, 4]]

    # Remove 1 -> 0 and keep 1 -> 2 -> 0, making the maximum retained weight 4.
    assert minMaxWeight(3, edges, 1) == 4


def test_selects_lightest_of_multiple_edges_between_same_nodes():
    edges = [[1, 0, 7], [1, 0, 3]]

    # Keep the parallel edge 1 -> 0 of weight 3 and remove the weight-7 edge.
    assert minMaxWeight(2, edges, 1) == 3
