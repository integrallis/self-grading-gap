def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    # Create adjacency list with edge weights
    graph = [[] for _ in range(n)]
    for u, v, w in edges:
        graph[u].append((v, w))

    # Function to check if we can reach node 0 with max edge weight <= max_weight
    def can_reach_with_max_weight(max_weight: int) -> bool:
        in_degree = [0] * n  # Track in-degrees for reachability
        out_degree = [0] * n   # Track out-degrees for threshold check
        visited = [False] * n   # Visited nodes
        queue = []

        # Build modified graph and in-degrees
        for u in range(n):
            for v, w in graph[u]:
                if w <= max_weight:
                    out_degree[u] += 1
                    if out_degree[u] <= threshold:
                        in_degree[v] += 1
                        queue.append(v) if not visited[v] else None

        # Check reachability using BFS
        reachable = [False] * n
        while queue:
            node = queue.pop(0)
            if visited[node]:
                continue
            visited[node] = True
            reachable[node] = True
            for v, w in graph[node]:
                if w <= max_weight:
                    out_degree[node] -= 1
                    if out_degree[node] < threshold:
                        if not visited[v]:
                            queue.append(v)

        return all(reachable)

    # Binary search on maximum edge weight
    left, right = 1, 10**6
    answer = -1

    while left <= right:
        mid = (left + right) // 2
        if can_reach_with_max_weight(mid):
            answer = mid
            right = mid - 1
        else:
            left = mid + 1

    return answer
