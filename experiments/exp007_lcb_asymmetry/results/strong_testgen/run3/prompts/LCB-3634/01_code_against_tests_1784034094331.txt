## Implementation

```python
def calculateScore(s: str) -> int:
    mirror = {chr(i): chr(219 - i) for i in range(97, 123)}  # Mapping for a-z to z-a
    score = 0
    marked = set()  # To keep track of marked indices

    for i in range(len(s)):
        if i in marked:
            continue
        for j in range(i - 1, -1, -1):  # Check for closest unmarked mirror
            if j in marked:
                continue
            if mirror[s[i]] == s[j]:
                score += i - j
                marked.add(i)
                marked.add(j)
                break

    return score

```
