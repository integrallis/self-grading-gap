from solution import calculateScore


def test_canonical_example_1():
    # spec: pairs (0, 2) and (1, 4) contribute 2 + 3 = 5

    # i=2 pairs with closest 'a' at 1 for 1; i=3 pairs with remaining 'a' at 0 for 3
