```python
from typing import List

def maxSum(nums: List[int]) -> int:
    unique_nums = set()
    current_sum = 0
    max_sum = float('-inf')
    for num in nums:
        if num not in unique_nums:
            unique_nums.add(num)
            if num > 0:
                current_sum += num
        else:
            current_sum = sum(n for n in unique_nums if n > 0)
            unique_nums = {num} if num > 0 else set()
            if num > 0:
                current_sum += num
        max_sum = max(max_sum, current_sum)
    if max_sum == float('-inf'):
        return max(nums)
    return max_sum
```
