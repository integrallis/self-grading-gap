from typing import List

def maxSum(nums: List[int]) -> int:
    unique_nums = set()
    max_sum = float('-inf')
    current_sum = 0
    for num in nums:
        if num not in unique_nums:
            unique_nums.add(num)
            current_sum += num
        else:
            current_sum += max(num, 0)  # Accumulate only positive values.
            unique_nums = {num} if num > 0 else set()
        max_sum = max(max_sum, current_sum)
    return max(max_sum, max(nums)) if max_sum < 0 else max_sum
```
