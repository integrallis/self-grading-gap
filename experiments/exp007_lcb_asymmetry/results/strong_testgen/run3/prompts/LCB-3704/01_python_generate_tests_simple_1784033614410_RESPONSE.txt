def test_canonical_example_1():
    # Prefix sums are 10, 20, 23, and 30; each difference with the remaining sum is even.
    assert countPartitions([10, 10, 3, 7, 6]) == 4


def test_canonical_example_2():
    # Split differences are 1 - 4 = -3 and 3 - 2 = 1; neither is even.
    assert countPartitions([1, 2, 2]) == 0


def test_canonical_example_3():
    # The three possible splits all have even differences because the total sum is even.
    assert countPartitions([2, 4, 6, 8]) == 3
