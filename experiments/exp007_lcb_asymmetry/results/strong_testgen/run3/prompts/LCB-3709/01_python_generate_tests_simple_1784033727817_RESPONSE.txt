from solution import hasSpecialSubstring


def test_canonical_example_1():
    # "aaa" at the end has length 3, one character, and preceding "b" differs.
    assert hasSpecialSubstring("aaabaaa", 3) is True


def test_canonical_example_2():
    # No length-2 substring of "abc" contains only one distinct character.
    assert hasSpecialSubstring("abc", 2) is False


def test_entire_string_is_a_special_substring():
    # "zzzz" has exactly length 4 and has no neighboring characters to violate either boundary.
    assert hasSpecialSubstring("zzzz", 4) is True


def test_run_longer_than_k_is_not_special():
    # Every length-3 "aaa" in "aaaa" has an adjacent "a", so neither qualifies.
    assert hasSpecialSubstring("aaaa", 3) is False


def test_exact_interior_run_with_different_neighbors():
    # The interior "aaaa" has length 4 and is bordered by different characters "b" and "c".
    assert hasSpecialSubstring("baaaac", 4) is True


def test_single_character_substring_requires_different_neighbors():
    # Each length-1 substring in "aaa" has an adjacent "a", so no single character qualifies.
    assert hasSpecialSubstring("aaa", 1) is False


def test_returns_true_when_one_valid_run_exists_among_invalid_runs():
    # "bb" has exactly length 2 and is bordered by different characters "a" and "c".
    assert hasSpecialSubstring("aaaabbcc", 2) is True
