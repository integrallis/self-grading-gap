from solution import maxLength


def test_canonical_example_1():
    # [1, 2, 1, 1, 1] has product 2, LCM 2, and GCD 1, so length 5 is valid.
    assert maxLength([1, 2, 1, 2, 1, 1, 1]) == 5


def test_canonical_example_2():
    # [3, 4, 5] has product 60, LCM 60, and GCD 1, so length 3 is valid.
    assert maxLength([2, 3, 4, 5, 6]) == 3


def test_canonical_example_3():
    # [3, 1, 4, 5, 1] has product 60, LCM 60, and GCD 1, so length 5 is valid.
    assert maxLength([1, 2, 3, 1, 4, 5, 1]) == 5


def test_returns_entire_array_when_all_elements_are_one():
    # Product, LCM, and GCD of [1, 1, 1, 1] are all 1, so the full length is valid.
    assert maxLength([1, 1, 1, 1]) == 4


def test_returns_entire_array_for_distinct_prime_factors():
    # [2, 3, 5] has product 30, LCM 30, and GCD 1, so the full length is valid.
    assert maxLength([2, 3, 5]) == 3


def test_returns_two_when_no_length_three_subarray_is_product_equivalent():
    # [2, 2, 2] has product 8, while LCM 2 times GCD 2 is 4; any two elements are valid.
    assert maxLength([2, 2, 2]) == 2
