from solution import maxFrequency


def test_canonical_example_1():
    # Adding -5 to nums[2..5] produces a second 1, so frequency(1) is 2.
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2


def test_canonical_example_2():
    # Adding 8 to nums[1..9] produces three additional 10s plus the original 10: 4 total.
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4


def test_single_element_can_be_changed_to_k():
    # Add -6 to the only element, changing 7 to 1, so frequency(1) is 1.
    assert maxFrequency([7], 1) == 1


def test_zero_addition_preserves_all_existing_k_values():
    # Adding 0 preserves all three existing 4s, so frequency(4) remains 3.
    assert maxFrequency([4, 4, 4], 4) == 3


def test_one_addition_value_can_convert_only_matching_source_values():
    # Adding -1 converts both 2s to 1; the 3 cannot also become 1 with that same addition.
    assert maxFrequency([2, 3, 2], 1) == 2


def test_subarray_cannot_skip_existing_k_between_convertible_values():
    # Converting both 2s requires including the middle 1, which changes it away from 1;
    # thus the best result is 2 ones, either by converting one 2 or converting both 2s.
    assert maxFrequency([2, 1, 2], 1) == 2


def test_best_subarray_can_avoid_losing_existing_k_values():
    # Adding -1 only to the two adjacent 2s yields [1, 1, 1, 1], so frequency(1) is 4.
    assert maxFrequency([1, 2, 2, 1], 1) == 4
