from typing import List

def maxFrequency(nums: List[int], k: int) -> int:
    nums.sort()
    left = 0
    total = 0
    max_freq = 0

    for right in range(len(nums)):
        total += nums[right]
        # Calculate the total number of increments needed to convert all elements in the window to k.
        while (nums[right] * (right - left + 1)) - total > k:
            total -= nums[left]
            left += 1
        # Update max frequency found, which is the size of the current window.
        max_freq = max(max_freq, right - left + 1)

    return max_freq
