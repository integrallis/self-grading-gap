from solution import subarraySum


def test_canonical_example_1():
    # i=0: [2] = 2; i=1: [2, 3] = 5; i=2: [3, 1] = 4; total = 2 + 5 + 4 = 11.
    assert subarraySum([2, 3, 1]) == 11


def test_canonical_example_2():
    # i=0: [3] = 3; i=1: [3, 1] = 4; i=2: [1, 1] = 2; i=3: [1, 1, 2] = 4; total = 13.
    assert subarraySum([3, 1, 1, 2]) == 13


def test_single_element_array():
    # i=0: [1000] = 1000; total = 1000.
    assert subarraySum([1000]) == 1000


def test_subarrays_with_clamped_and_non_clamped_starts():
    # i=0: [1] = 1; i=1: [1, 2] = 3; i=2: [2, 1] = 3;
    # i=3: [1, 2, 1, 3] = 7; i=4: [1, 3, 2] = 6; total = 1 + 3 + 3 + 7 + 6 = 20.
    assert subarraySum([1, 2, 1, 3, 2]) == 20
