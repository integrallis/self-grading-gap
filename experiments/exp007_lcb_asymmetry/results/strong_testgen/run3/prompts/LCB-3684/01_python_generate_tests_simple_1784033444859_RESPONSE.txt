from solution import hasMatch


def test_canonical_example_1():
    # "ee*" can replace '*' with "tcod", producing "eetcode", a substring of "leetcode".
    assert hasMatch("leetcode", "ee*e") is True


def test_canonical_example_2():
    # No replacement for '*' makes a substring of "car" start with "c" and end with "v".
    assert hasMatch("car", "c*v") is False


def test_canonical_example_3():
    # Replacing '*' with "" produces "u", which is a substring of "luck".
    assert hasMatch("luck", "u*") is True


def test_star_matches_zero_characters_between_fixed_parts():
    # Replacing '*' with "" produces "ab", which is a substring of "zabz".
    assert hasMatch("zabz", "a*b") is True


def test_star_matches_characters_between_fixed_parts():
    # Replacing '*' with "xyz" produces "axyzb", which is a substring of "axyzb".
    assert hasMatch("axyzb", "a*b") is True


def test_pattern_can_match_at_start_of_string():
    # Replacing '*' with "bc" produces "abc", which is a substring at the start of "abcdef".
    assert hasMatch("abcdef", "a*c") is True


def test_pattern_can_match_at_end_of_string():
    # Replacing '*' with "bc" produces "bcd", which is a substring at the end of "xabcd".
    assert hasMatch("xabcd", "b*d") is True


def test_star_at_beginning_can_match_empty_sequence():
    # Replacing '*' with "" produces "bc", which is a substring of "abc".
    assert hasMatch("abc", "*bc") is True


def test_star_at_end_can_match_empty_sequence():
    # Replacing '*' with "" produces "ab", which is a substring of "abc".
    assert hasMatch("abc", "ab*") is True


def test_fixed_parts_must_appear_in_order():
    # "ba" occurs before "ab", so no replacement makes "ba*ab" a substring of "ababa".
    assert hasMatch("ababa", "ba*ab") is False
