from typing import List

def maxSum(nums: List[int]) -> int:
    unique_values = set()
    max_sum = float('-inf')  # Start with negative infinity
    current_sum = 0
    left = 0

    for right in range(len(nums)):
        while nums[right] in unique_values:
            unique_values.remove(nums[left])
            current_sum -= nums[left]
            left += 1
        unique_values.add(nums[right])
        current_sum += nums[right]
        max_sum = max(max_sum, current_sum)

    # Ensure we account for the case where all numbers are negative
    return max(max_sum, max(nums))  # Return the best found sum or the max negative value
```
