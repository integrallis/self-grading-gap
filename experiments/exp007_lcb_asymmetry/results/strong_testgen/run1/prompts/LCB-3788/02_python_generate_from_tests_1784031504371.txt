from typing import List

def maxSum(nums: List[int]) -> int:
    unique_values = set()
    max_sum = float('-inf')
    current_sum = 0

    for num in nums:
        if num not in unique_values:
            unique_values.add(num)
            current_sum += num
        else:
            max_sum = max(max_sum, current_sum)
            while num in unique_values:
                unique_values.remove(nums[len(unique_values)])
                current_sum -= nums[len(unique_values)]
            unique_values.add(num)
            current_sum += num

    return max(max_sum, current_sum, 0)
```
