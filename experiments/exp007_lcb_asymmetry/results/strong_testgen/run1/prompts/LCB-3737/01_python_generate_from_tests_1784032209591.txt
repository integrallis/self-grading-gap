from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    dp = [[0] * 3 for _ in range(n)]
    # Initialize the first house's costs
    for j in range(3):
        dp[0][j] = cost[0][j]
    
    # Fill the dp table for all houses considering adjacent and equidistant constraints
    for i in range(1, n):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(dp[i - 1][(j + 1) % 3], dp[i - 1][(j + 2) % 3])

    # Now we need to check the equidistant constraint for the first half vs second half
    min_cost = float('inf')
    for j in range(3):  # color of the last house of the first half
        for k in range(3):  # color of the last house of the second half
            if j != k:
                current_cost = dp[n // 2 - 1][j] + dp[n - 1][k]
                min_cost = min(min_cost, current_cost)

    return min_cost

```
