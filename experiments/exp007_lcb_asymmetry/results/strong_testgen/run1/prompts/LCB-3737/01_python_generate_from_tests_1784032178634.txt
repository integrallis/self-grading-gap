from typing import List


def minCost(n: int, cost: List[List[int]]) -> int:
    dp = [[0] * 3 for _ in range(n)]
    # Initialize the first house's costs
    for j in range(3):
        dp[0][j] = cost[0][j]
    
    # Fill the dp table for the first half
    for i in range(1, n // 2):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(dp[i - 1][(j + 1) % 3], dp[i - 1][(j + 2) % 3])

    first_half_min_cost = dp[n // 2 - 1]

    # Fill the dp table for the second half
    for i in range(n // 2, n - 1):
        for j in range(3):
            dp[i][j] = cost[i][j] + min(dp[i - 1][(j + 1) % 3], dp[i - 1][(j + 2) % 3])

    second_half_min_cost = dp[n - 1]

    # Calculate the minimum cost considering equidistant constraints
    min_cost = float('inf')
    for j in range(3):
        for k in range(3):
            if j != k:
                min_cost = min(min_cost, first_half_min_cost[j] + second_half_min_cost[k])

    return min_cost

```
