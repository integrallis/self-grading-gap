from solution import maxLength


def test_canonical_example_1():
    # [1, 2, 1, 1, 1] has product 2, gcd 1, and lcm 2, so length 5 is valid.
    assert maxLength([1, 2, 1, 2, 1, 1, 1]) == 5


def test_canonical_example_2():
    # [3, 4, 5] has product 60, gcd 1, and lcm 60, so length 3 is valid.
    assert maxLength([2, 3, 4, 5, 6]) == 3


def test_canonical_example_3():
    # [2, 3, 1, 4, 5] has product 120, gcd 1, and lcm 120, so length 5 is valid.
    assert maxLength([1, 2, 3, 1, 4, 5, 1]) == 5


def test_two_element_subarray_is_product_equivalent():
    # For [6, 10], product is 60, gcd is 2, and lcm is 30; 2 * 30 = 60.
    assert maxLength([6, 10]) == 2


def test_entire_array_of_ones_is_product_equivalent():
    # [1, 1, 1, 1] has product 1, gcd 1, and lcm 1, so all 4 elements qualify.
    assert maxLength([1, 1, 1, 1]) == 4


def test_repeated_non_one_values_do_not_form_a_longer_equivalent_subarray():
    # Every pair [2, 2] has product 4, gcd 2, and lcm 2; 2 * 2 = 4.
    # The full [2, 2, 2] has product 8 but gcd * lcm is 2 * 2 = 4.
    assert maxLength([2, 2, 2]) == 2


def test_pairwise_coprime_values_form_a_product_equivalent_subarray():
    # [2, 3, 5, 7] has product 210, gcd 1, and lcm 210, so all 4 qualify.
    assert maxLength([2, 3, 5, 7]) == 4
