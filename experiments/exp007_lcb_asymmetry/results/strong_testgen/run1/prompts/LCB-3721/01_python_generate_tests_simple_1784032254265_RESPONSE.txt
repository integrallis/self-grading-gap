from solution import countMentions


def test_canonical_example_1():
    events = [
        ["MESSAGE", "10", "id1 id0"],
        ["OFFLINE", "11", "0"],
        ["MESSAGE", "71", "HERE"],
    ]

    # At 10 both users are explicitly mentioned; at 71 user 0 is back online, so HERE adds one to each.
    assert countMentions(2, events) == [2, 2]


def test_canonical_example_2():
    events = [
        ["MESSAGE", "10", "id1 id0"],
        ["OFFLINE", "11", "0"],
        ["MESSAGE", "12", "ALL"],
    ]

    # At 10 both users are explicitly mentioned; ALL at 12 adds one to both, including offline user 0.
    assert countMentions(2, events) == [2, 2]


def test_canonical_example_3():
    events = [
        ["OFFLINE", "10", "0"],
        ["MESSAGE", "12", "HERE"],
    ]

    # User 0 is offline at 12, so HERE mentions only online user 1.
    assert countMentions(2, events) == [0, 1]


def test_explicit_mentions_count_duplicates_and_can_mention_offline_users():
    events = [
        ["MESSAGE", "5", "id1 id1 id2"],
        ["OFFLINE", "5", "1"],
    ]

    # The offline change at 5 is processed first, but explicit id1 mentions still count twice; id2 counts once.
    assert countMentions(3, events) == [0, 2, 1]


def test_events_are_processed_by_timestamp_when_listed_out_of_order():
    events = [
        ["MESSAGE", "50", "HERE"],
        ["OFFLINE", "40", "0"],
    ]

    # User 0 goes offline at 40 and remains offline at 50, so HERE at 50 mentions only user 1.
    assert countMentions(2, events) == [0, 1]
