```python
from typing import List


def countMentions(numberOfUsers: int, events: List[List[str]]) -> List[int]:
    mentions = [0] * numberOfUsers  # Initialize mentions count for each user
    online_status = [True] * numberOfUsers  # Track online status of each user
    events.sort(key=lambda x: int(x[1]))  # Sort events by timestamp

    for event in events:
        if event[0] == "MESSAGE":
            timestamp = int(event[1])
            mentions_string = event[2]
            if mentions_string == "ALL":
                for i in range(numberOfUsers):
                    mentions[i] += 1  # Mention all users
            elif mentions_string == "HERE":
                for i in range(numberOfUsers):
                    if online_status[i]:
                        mentions[i] += 1  # Mention only online users
            else:
                ids = mentions_string.split()
                for id_str in ids:
                    user_id = int(id_str[2:])
                    mentions[user_id] += 1  # Count individual user mentions
        elif event[0] == "OFFLINE":
            user_id = int(event[2])
            online_status[user_id] = False  # Set user to offline
            # Set user back online after 60 time units
            for j in range(len(events)):  # Process subsequent events
                if int(events[j][1]) >= int(event[1]) + 60:
                    online_status[user_id] = True
                    break
    return mentions
```
