from solution import maxFrequency


def test_canonical_example_1():
    # Adding -5 to nums[2..5] changes 6 to 1 while the original 1 remains: frequency = 2.
    assert maxFrequency([1, 2, 3, 4, 5, 6], 1) == 2


def test_canonical_example_2():
    # Adding 8 to nums[1..9] changes three 2s to 10; with the original 10, frequency = 4.
    assert maxFrequency([10, 2, 3, 4, 5, 5, 4, 3, 2, 2], 10) == 4


def test_single_non_target_can_be_changed_to_k():
    # Select the only element and add k - nums[0] = 3, producing one 5.
    assert maxFrequency([2], 5) == 1


def test_all_existing_targets_can_be_preserved():
    # Select any subarray and add 0, so all three existing 3s remain 3.
    assert maxFrequency([3, 3, 3], 3) == 3


def test_best_subarray_preserves_targets_outside_it():
    # Adding -1 to the middle two 2s makes them 1 while both outer 1s remain: frequency = 4.
    assert maxFrequency([1, 2, 2, 1], 1) == 4


def test_one_offset_can_convert_only_one_source_value():
    # Adding -1 converts the two 2s to 1, or adding -2 converts the two 3s to 1; either gives 2.
    assert maxFrequency([2, 3, 2, 3], 1) == 2


def test_targets_inside_a_selected_subarray_are_shifted_away():
    # Converting all three 2s with -1 also shifts both intervening 1s; the best possible frequency is 3.
    assert maxFrequency([2, 1, 2, 1, 2], 1) == 3
