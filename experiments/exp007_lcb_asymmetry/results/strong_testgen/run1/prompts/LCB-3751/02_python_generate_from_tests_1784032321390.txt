from typing import List


def maxFrequency(nums: List[int], k: int) -> int:
    count = [0] * 51  # since nums[i] is between 1 and 50
    for num in nums:
        count[num] += 1
    max_freq = 0
    for x in range(1, 51):  # check each possible number in range
        if x <= k:
            needed = k - x
            if needed >= 0:
                frequency = count[x] + sum(count[x + needed:])
                max_freq = max(max_freq, frequency)
    return max_freq
```
