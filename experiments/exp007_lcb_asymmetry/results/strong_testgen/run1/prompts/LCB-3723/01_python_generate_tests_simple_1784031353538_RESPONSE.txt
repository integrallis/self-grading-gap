from solution import sumOfGoodNumbers


def test_canonical_example_1():
    # nums[1] + nums[4] + nums[5] = 3 + 5 + 4 = 12.
    assert sumOfGoodNumbers([1, 3, 2, 1, 5, 4], 2) == 12


def test_canonical_example_2():
    # Only nums[0] is good, so the sum is 2.
    assert sumOfGoodNumbers([2, 1], 1) == 2


def test_requires_greater_than_both_existing_k_distance_neighbors():
    # Good values are 10 and 5, so the sum is 10 + 5 = 15.
    assert sumOfGoodNumbers([1, 10, 5, 1, 4], 2) == 15


def test_equal_k_distance_neighbor_is_not_good():
    # Each value has an equal value at distance k, so no value is good: sum = 0.
    assert sumOfGoodNumbers([5, 1, 5, 1], 2) == 0
