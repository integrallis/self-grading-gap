from typing import List


def maxFreeTime(eventTime: int, k: int, startTime: List[int], endTime: List[int]) -> int:
    n = len(startTime)
    free_periods = []
    last_end = 0

    # Calculate free periods before the first meeting and between meetings
    for i in range(n):
        if i == 0:
            if startTime[i] > 0:
                free_periods.append(startTime[i])  # Free time before the first meeting
        else:
            free_periods.append(startTime[i] - endTime[i - 1])  # Free time between meetings
        last_end = endTime[i]

    # Free time after the last meeting
    if last_end < eventTime:
        free_periods.append(eventTime - last_end)

    # If k is 0, return the maximum free time found
    if k == 0:
        return max(free_periods) if free_periods else 0

    # Maximum free time achievable after moving at most k meetings
    max_free = 0
    total_free = sum(free_periods)  # Start with total free time
    start_moves = [0] * (n + 1)  # To store the possible free time adjustments

    # Try to generate adjustments by moving meetings left as much as possible
    for i in range(n):
        free_left = startTime[i] - (endTime[i - 1] if i > 0 else 0)
        moves = min(k, free_left // (endTime[i] - startTime[i])) if endTime[i] - startTime[i] > 0 else 0
        if moves > 0:
            start_moves[i + 1] += moves

    for i in range(n + 1):
        if i > 0:
            start_moves[i] += start_moves[i - 1]
        if start_moves[i] > k:
            start_moves[i] = k  # Limit to at most k moves

    # Calculate the new free time possible
    for i in range(n):
        if (i > 0):
            moves = start_moves[i]  # Maximum possible moves up to this point
            total_free += moves * (endTime[i] - startTime[i])

    return total_free - (sum(end - start for start, end in zip(startTime, endTime)) - k * (endTime[n - 1] - startTime[0]))
