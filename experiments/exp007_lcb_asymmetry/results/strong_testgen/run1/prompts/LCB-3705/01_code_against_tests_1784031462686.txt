```python
from typing import List


def largestInteger(nums: List[int], k: int) -> int:
    from collections import defaultdict
    count = defaultdict(int)
    n = len(nums)

    # Count occurrences of each integer in each subarray of size k
    for i in range(n - k + 1):
        seen = set()
        for j in range(k):
            if nums[i + j] not in seen:
                count[nums[i + j]] += 1
                seen.add(nums[i + j])

    # Find the largest integer that appears exactly once
    largest = -1
    for num, cnt in count.items():
        if cnt == 1 and num > largest:
            largest = num

    return largest

```
