from solution import largestInteger


def test_canonical_example_1():
    # 3 occurs only in [3, 9, 2] and 7 only in [2, 1, 7]; largest is 7.
    assert largestInteger([3, 9, 2, 1, 7], 3) == 7


def test_canonical_example_2():
    # Only 3 appears in exactly one size-4 subarray: [3, 9, 7, 2].
    assert largestInteger([3, 9, 7, 2, 1, 7], 4) == 3


def test_canonical_example_3():
    # 0 appears in both size-1 subarrays, so no integer appears in exactly one.
    assert largestInteger([0, 0], 1) == -1


def test_entire_array_is_the_only_subarray():
    # The sole size-4 subarray contains 4, 9, and 2; the largest is 9.
    assert largestInteger([4, 9, 9, 2], 4) == 9


def test_size_one_subarrays_require_value_to_occur_once():
    # Size-1 subarrays contain 1, 3, 3, 2, 5, 5; only 1 and 2 occur once, so return 2.
    assert largestInteger([1, 3, 3, 2, 5, 5], 1) == 2


def test_repeated_value_within_one_subarray_counts_once_for_that_subarray():
    # There is one size-3 subarray, [8, 8, 1], containing both 8 and 1; largest is 8.
    assert largestInteger([8, 8, 1], 3) == 8


def test_returns_minus_one_when_every_value_is_in_multiple_subarrays():
    # Both size-2 subarrays, [5, 4] and [4, 5], contain both 5 and 4.
    assert largestInteger([5, 4, 5], 2) == -1
