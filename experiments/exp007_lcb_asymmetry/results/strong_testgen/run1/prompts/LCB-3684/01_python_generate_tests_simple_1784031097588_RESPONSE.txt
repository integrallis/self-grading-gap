from solution import hasMatch


def test_canonical_example_1():
    # "ee" occurs before a later "e" in "leetcode"; '*' can be "tcod".
    assert hasMatch("leetcode", "ee*e") is True


def test_canonical_example_2():
    # "c" occurs, but no later "v" exists for the suffix after '*'.
    assert hasMatch("car", "c*v") is False


def test_canonical_example_3():
    # "u" is a substring; '*' can be replaced with the empty sequence.
    assert hasMatch("luck", "u*") is True


def test_star_can_match_empty_sequence_between_prefix_and_suffix():
    # Replacing '*' with "" makes "ab" a substring of "zabz".
    assert hasMatch("zabz", "a*b") is True


def test_star_can_match_multiple_characters_between_prefix_and_suffix():
    # Replacing '*' with "123" makes "a123b" a substring of "xa123by".
    assert hasMatch("xa123by", "a*b") is True


def test_match_must_be_contiguous_substring():
    # Although "a" appears before "b", no contiguous substring can match "a*b" in "ac".
    assert hasMatch("ac", "a*b") is False


def test_pattern_with_star_at_start_can_match_suffix():
    # Replacing '*' with "ab" makes "abc" a substring of "zabc".
    assert hasMatch("zabc", "*c") is True


def test_pattern_with_star_at_end_can_match_prefix():
    # Replacing '*' with "" makes "ab" a substring of "abc".
    assert hasMatch("abc", "ab*") is True


def test_required_parts_must_appear_in_order():
    # "b" appears before "a", so no substring can match "a*b".
    assert hasMatch("ba", "a*b") is False
