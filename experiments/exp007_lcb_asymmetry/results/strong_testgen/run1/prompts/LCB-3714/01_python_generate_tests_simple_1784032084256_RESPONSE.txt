from solution import minMaxSums


def test_canonical_example_1():
    # Singles contribute 2 + 4 + 6 = 12; pairs contribute 3 + 4 + 5 = 12.
    assert minMaxSums([1, 2, 3], 2) == 24


def test_canonical_example_2():
    # With k = 1, each element contributes itself as both minimum and maximum:
    # 2 * (5 + 0 + 6) = 22.
    assert minMaxSums([5, 0, 6], 1) == 22


def test_canonical_example_3():
    # Three singleton subsequences and three length-two subsequences each contribute 2:
    # 6 * 2 = 12.
    assert minMaxSums([1, 1, 1], 2) == 12


def test_single_element_array():
    # The only subsequence is [7], whose minimum plus maximum is 7 + 7 = 14.
    assert minMaxSums([7], 1) == 14


def test_k_includes_all_nonempty_subsequences():
    # For [1, 2, 3]:
    # singles: 2 + 4 + 6 = 12; pairs: 3 + 4 + 5 = 12; triple: 1 + 3 = 4.
    assert minMaxSums([1, 2, 3], 3) == 28


def test_subsequences_use_original_positions_with_unsorted_values():
    # Singles contribute 8 + 2 + 6 = 16; pairs [4,1], [4,3], [1,3]
    # contribute 5 + 7 + 4 = 16, for a total of 32.
    assert minMaxSums([4, 1, 3], 2) == 32


def test_result_is_returned_modulo_1_billion_and_7():
    # With k = 1, each of two 1_000_000_000 values contributes 2_000_000_000:
    # 4_000_000_000 mod 1_000_000_007 = 999_999_979.
    assert minMaxSums([1_000_000_000, 1_000_000_000], 1) == 999_999_979
