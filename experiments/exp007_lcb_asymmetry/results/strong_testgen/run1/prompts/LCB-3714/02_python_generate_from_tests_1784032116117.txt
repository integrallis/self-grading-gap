from typing import List


def minMaxSums(nums: List[int], k: int) -> int:
    MOD = 10**9 + 7
    n = len(nums)
    total_sum = 0

    # Calculate the contributions for the minimums and maximums
    for i in range(n):
        # Each number nums[i] contributes to all subsequences of length 1 to k
        min_contribution = nums[i] * (1 << min(k, n - 1)) % MOD
        max_contribution = nums[i] * (1 << min(k, n - 1)) % MOD
        total_sum = (total_sum + min_contribution + max_contribution) % MOD

    return total_sum
```
