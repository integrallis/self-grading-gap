```python
from typing import List


def minMaxSums(nums: List[int], k: int) -> int:
    MOD = 10**9 + 7
    n = len(nums)
    total_sum = 0

    # Calculate the contributions for the minimums and maximums
    for i in range(n):
        count_contributions = 1 << min(k, n - 1)
        total_sum += (nums[i] * count_contributions) % MOD
    total_sum = (total_sum * 2) % MOD

    return total_sum

```
