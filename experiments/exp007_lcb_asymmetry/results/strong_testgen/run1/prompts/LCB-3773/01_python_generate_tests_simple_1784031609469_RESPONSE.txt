from solution import minimumPairRemoval


def test_canonical_example_1():
    # (3, 1) has minimum sum 4 -> [5, 2, 4];
    # then (2, 4) has minimum sum 6 -> [5, 6], so 2 operations.
    assert minimumPairRemoval([5, 2, 3, 1]) == 2


def test_canonical_example_2():
    # [1, 2, 2] is already non-decreasing, so no operations are needed.
    assert minimumPairRemoval([1, 2, 2]) == 0


def test_uses_leftmost_pair_when_minimum_sums_tie():
    # Both adjacent pairs in [1, 2, 1] sum to 3.
    # The leftmost pair is merged first -> [3, 1], then -> [4], so 2 operations.
    assert minimumPairRemoval([1, 2, 1]) == 2


def test_handles_negative_values():
    # (-3, 2) has minimum sum -1 -> [1, -1], then -> [0], so 2 operations.
    assert minimumPairRemoval([1, -3, 2]) == 2


def test_single_element_array_is_non_decreasing():
    # A one-element array has no previous element to violate non-decreasing order.
    assert minimumPairRemoval([7]) == 0
