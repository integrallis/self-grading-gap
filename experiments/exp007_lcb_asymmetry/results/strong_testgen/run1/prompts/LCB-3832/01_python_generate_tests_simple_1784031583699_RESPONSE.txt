from solution import minCosts


def test_canonical_example_1():
    # Pay 5, 3, then 1 at positions 0, 1, and 3 respectively; later positions are free.
    assert minCosts([5, 3, 4, 1, 3, 2]) == [5, 3, 3, 1, 1, 1]


def test_canonical_example_2():
    # Paying 1 to swap with person 0 makes every later position reachable for free.
    assert minCosts([1, 2, 4, 6, 7]) == [1, 1, 1, 1, 1]


def test_single_person_line():
    # The only reachable position requires paying that person's cost.
    assert minCosts([100]) == [100]


def test_strictly_decreasing_costs():
    # Each new person is cheaper than all previous people, so each prefix minimum changes.
    assert minCosts([5, 4, 3, 2, 1]) == [5, 4, 3, 2, 1]


def test_repeated_minimum_cost():
    # The minimum cost becomes 2 at index 1 and remains 2 for every later position.
    assert minCosts([7, 2, 2, 5, 2]) == [7, 2, 2, 2, 2]
