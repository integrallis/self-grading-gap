from typing import List


def maximumCoins(coins: List[List[int]], k: int) -> int:
    # Create a map to hold the total coins at each position
    total_coins = {}

    # Fill the total_coins map with the coins from the segments
    for l, r, c in coins:
        for position in range(l, r + 1):
            total_coins[position] = total_coins.get(position, 0) + c

    # Initialize variables to find the maximum coins in k consecutive bags
    max_coins = 0
    current_coins = 0

    # Sort the keys to process positions in order
    sorted_positions = sorted(total_coins.keys())
    left = 0

    # Use a sliding window approach to calculate the maximum coins in k bags
    for right in range(len(sorted_positions)):
        current_coins += total_coins[sorted_positions[right]]

        # Ensure the window size does not exceed k
        while sorted_positions[right] - sorted_positions[left] + 1 > k:
            current_coins -= total_coins[sorted_positions[left]]
            left += 1

        # Update max_coins if the current window is valid
        max_coins = max(max_coins, current_coins)

    return max_coins
```
