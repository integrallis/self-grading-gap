from solution import maximumCoins


def test_canonical_example_1():
    # [3, 4, 5, 6] contains 2 + 0 + 4 + 4 = 10 coins.
    assert maximumCoins([[8, 10, 1], [1, 3, 2], [5, 6, 4]], 4) == 10


def test_canonical_example_2():
    # Any two bags within [1, 10] contain 3 + 3 = 6 coins.
    assert maximumCoins([[1, 10, 3]], 2) == 6


def test_window_can_include_gaps_between_segments():
    # Bags [2, 3, ..., 10] contain 3 + 0 + 0 + 0 + 0 + 0 + 0 + 0 + 4 = 7 coins.
    assert maximumCoins([[2, 2, 3], [10, 10, 4]], 9) == 7


def test_window_can_take_part_of_a_long_segment_to_include_a_higher_value_segment():
    # Bags [92, 93, ..., 101] contain nine 1-coin bags plus one 100-coin bag: 9 + 100 = 109.
    assert maximumCoins([[1, 100, 1], [101, 101, 100]], 10) == 109


def test_large_coordinate_and_total_coin_count():
    # A window of 1,000,000,000 bags covers the whole segment: 1,000,000,000 * 1,000 = 1,000,000,000,000.
    assert maximumCoins([[1, 1_000_000_000, 1000]], 1_000_000_000) == 1_000_000_000_000
