from solution import hasSpecialSubstring


def test_canonical_example_1():
    # "aaa" at the end has length 3 and is preceded by different character 'b'.
    assert hasSpecialSubstring("aaabaaa", 3) is True


def test_canonical_example_2():
    # No length-2 substring of "abc" consists of one repeated character.
    assert hasSpecialSubstring("abc", 2) is False


def test_exact_length_run_in_middle_is_special():
    # "bbb" has length 3 and is bounded by different characters 'a' and 'c'.
    assert hasSpecialSubstring("abbbc", 3) is True


def test_longer_run_does_not_contain_special_shorter_substring():
    # Every length-2 "aa" substring has an adjacent 'a', so none is specially bounded.
    assert hasSpecialSubstring("aaaa", 2) is False


def test_single_character_substring_can_be_special():
    # The "b" has length 1 and both adjacent characters are different ('a' and 'c').
    assert hasSpecialSubstring("abc", 1) is True


def test_entire_string_can_be_special():
    # The entire string "aaaa" has length 4 and has no adjacent characters to violate bounds.
    assert hasSpecialSubstring("aaaa", 4) is True


def test_exact_run_after_a_longer_nonmatching_run_is_special():
    # "bbb" has exactly length 3 and is preceded by different character 'a'.
    assert hasSpecialSubstring("aaaabbb", 3) is True
