from typing import List


def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    # dp[i] will store the maximum length of subsequence ending with nums[i]
    dp = [1] * n
    max_length = 1

    # Iterate through the array to fill dp array
    for i in range(n):
        for j in range(i):
            # Calculate absolute difference
            difference_ij = abs(nums[i] - nums[j])
            # Check if we can form a non-increasing sequence of differences
            if dp[j] > 1 and (j == 0 or abs(nums[j] - nums[j - 1]) >= difference_ij):
                dp[i] = max(dp[i], dp[j] + 1)
                max_length = max(max_length, dp[i])

    return max_length
```
