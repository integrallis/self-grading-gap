```python
from typing import List

def longestSubsequence(nums: List[int]) -> int:
    n = len(nums)
    # dp[i] will store the length of the longest subsequence ending with nums[i]
    dp = [1] * n
    max_length = 1

    # Iterate through the array to fill dp array
    for i in range(n):
        for j in range(i):
            # Calculate absolute difference
            if abs(nums[i] - nums[j]) >= (abs(nums[j] - nums[j - 1]) if j > 0 else float('inf')):
                dp[i] = max(dp[i], dp[j] + 1)
        max_length = max(max_length, dp[i])

    return max_length

```
