from solution import maxDistance


def test_canonical_example_1():
    # Change "NWSE" to "NWNE"; its positions reach (-1, 2), distance 3.
    assert maxDistance("NWSE", 1) == 3


def test_canonical_example_2():
    # Change "NSWWEW" to "NNWWWW"; after all 6 moves, distance is 6.
    assert maxDistance("NSWWEW", 3) == 6


def test_no_changes_uses_greatest_distance_at_any_prefix():
    # "NNSS" has prefix distances 1, 2, 1, 0, so the maximum is 2.
    assert maxDistance("NNSS", 0) == 2


def test_single_change_can_improve_an_intermediate_prefix():
    # Change the second move in "NSNS" from S to N: "NNNS" reaches distance 3.
    assert maxDistance("NSNS", 1) == 3


def test_changes_can_make_every_move_point_in_one_direction():
    # With four changes, "NSWE" can become "NNNN", reaching distance 4.
    assert maxDistance("NSWE", 4) == 4


def test_already_straight_path_needs_no_changes():
    # "NNN" reaches positions with distances 1, 2, 3; the maximum is 3.
    assert maxDistance("NNN", 0) == 3
