```python
def maxDistance(s: str, k: int) -> int:
    # Initialize the coordinates and max distance
    x, y = 0, 0
    max_distance = 0
    # Store the distances at each prefix
    distances = []
    # Count the moves in each direction
    for char in s:
        if char == 'N':
            y += 1
        elif char == 'S':
            y -= 1
        elif char == 'E':
            x += 1
        elif char == 'W':
            x -= 1
        # Calculate the current distance
        current_distance = abs(x) + abs(y)
        distances.append(current_distance)
        max_distance = max(max_distance, current_distance)
    # Calculate the maximum distance with k changes
    for i in range(len(distances)):
        # Calculate potential distance if we can change k moves
        potential_distance = distances[i] + min(k, len(s) - i)
        max_distance = max(max_distance, potential_distance)
    return max_distance
```
