def maxDistance(s: str, k: int) -> int:
    # Initialize the coordinates and max distance
    x, y = 0, 0
    max_distance = 0
    # Count the moves in each direction
    for char in s:
        if char == 'N':
            y += 1
        elif char == 'S':
            y -= 1
        elif char == 'E':
            x += 1
        elif char == 'W':
            x -= 1
        # Calculate the current distance
        current_distance = abs(x) + abs(y)
        max_distance = max(max_distance, current_distance)
    # Calculate the maximum distance with k changes
    total_moves = len(s)
    net_distance = abs(x) + abs(y)
    max_possible_distance = net_distance + k
    return max(max_distance, max_possible_distance)
