from solution import findValidPair


def test_canonical_example_1():
    # "2" appears twice and "3" appears three times; the first valid adjacent pair is "23".
    assert findValidPair("2523533") == "23"


def test_canonical_example_2():
    # "2" appears twice and "1" appears once; the first valid adjacent pair is "21".
    assert findValidPair("221") == "21"


def test_canonical_example_3():
    # Although "2" appears twice, the only adjacent pair is "22", whose digits are equal.
    assert findValidPair("22") == ""


def test_returns_leftmost_when_multiple_valid_pairs_exist():
    # Counts are 1->1, 2->2, and 3->3; both "12" and later pairs are valid, so return "12".
    assert findValidPair("1233322") == "12"


def test_returns_empty_when_an_adjacent_digit_does_not_have_its_required_count():
    # "1" appears once, but "2" appears three times rather than twice, so "12" is not valid.
    assert findValidPair("1222") == ""
