from solution import maximumAmount


def test_canonical_example_1():
    # Path 0 -> 1 -> neutralize -2 -> 3 -> 4 totals 0 + 1 + 0 + 3 + 4 = 8.
    assert maximumAmount([[0, 1, -1], [1, -2, 3], [2, -3, 4]]) == 8


def test_canonical_example_2():
    # A four-cell route contains four 10-value cells: 10 + 10 + 10 + 10 = 40.
    assert maximumAmount([[10, 10, 10], [10, 10, 10]]) == 40


def test_neutralizes_a_robber_at_the_starting_cell():
    # The only cell is a robber worth -7, which can be neutralized for a total of 0.
    assert maximumAmount([[-7]]) == 0


def test_total_can_remain_negative_when_more_than_two_robbers_are_unavoidable():
    # Neutralize -4 and -3; the remaining losses are -1 and -2, totaling -3.
    assert maximumAmount([[-1, -2, -3, -4]]) == -3


def test_chooses_the_best_route_instead_of_using_neutralizations_unnecessarily():
    # The bottom route is 5 + 4 + 4 + 4 = 17, exceeding routes through -100.
    assert maximumAmount([[5, -100, 5], [4, 4, 4]]) == 17
