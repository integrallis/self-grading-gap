from solution import minMaxWeight


def test_canonical_example_1():
    # Keeping 1->0, 3->0, 4->3, and 2->1 lets every node reach 0 with maximum weight 1.
    assert minMaxWeight(
        5,
        [[1, 0, 1], [2, 0, 2], [3, 0, 1], [4, 3, 1], [2, 1, 1]],
        2,
    ) == 1


def test_canonical_example_2():
    # Nodes 2, 3, and 4 have no directed path to node 0, so satisfying reachability is impossible.
    assert minMaxWeight(
        5,
        [[0, 1, 1], [0, 2, 2], [0, 3, 1], [0, 4, 1], [1, 2, 1], [1, 4, 1]],
        1,
    ) == -1


def test_canonical_example_3():
    # Keeping 1->2 (1), 2->3 (2), 3->4 (2), and 4->0 (1) reaches 0 from all nodes;
    # weight 1 cannot work because node 2 then cannot reach 0.
    assert minMaxWeight(
        5,
        [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [3, 4, 2], [4, 0, 1]],
        1,
    ) == 2


def test_canonical_example_4():
    # Node 3 has no directed path to node 0, so satisfying reachability is impossible.
    assert minMaxWeight(
        5,
        [[1, 2, 1], [1, 3, 3], [1, 4, 5], [2, 3, 2], [4, 0, 1]],
        1,
    ) == -1


def test_selects_lower_weight_parallel_edge():
    # Of the two edges from 1 to 0, retaining the weight-4 edge reaches node 0 with maximum weight 4.
    assert minMaxWeight(2, [[1, 0, 9], [1, 0, 4]], 1) == 4
