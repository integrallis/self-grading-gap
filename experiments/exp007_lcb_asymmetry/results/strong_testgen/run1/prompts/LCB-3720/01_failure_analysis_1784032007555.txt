from typing import List


def minMaxWeight(n: int, edges: List[List[int]], threshold: int) -> int:
    from collections import defaultdict, deque

    def can_reach_all(max_weight):
        graph = defaultdict(list)
        out_degree = [0] * n

        for u, v, w in edges:
            if w <= max_weight:
                if out_degree[u] < threshold:
                    graph[u].append((v, w))  # Store outgoing edges instead
                    out_degree[u] += 1

        reachable = [False] * n
        queue = deque([0])
        reachable[0] = True

        while queue:
            node = queue.popleft()
            for neighbor, weight in graph[node]:
                if not reachable[neighbor]:
                    reachable[neighbor] = True
                    queue.append(neighbor)

        return all(reachable)

    left, right = 1, max(w for _, _, w in edges)
    result = -1

    while left <= right:
        mid = (left + right) // 2
        if can_reach_all(mid):
            result = mid
            right = mid - 1
        else:
            left = mid + 1

    return result
