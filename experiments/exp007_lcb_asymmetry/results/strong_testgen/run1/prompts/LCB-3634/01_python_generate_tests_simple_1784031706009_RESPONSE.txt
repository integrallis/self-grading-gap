from solution import calculateScore


def test_canonical_example_1():
    # "aczzx": pair indices (0, 2) for 2 points and (1, 4) for 3 points: 2 + 3 = 5.
    assert calculateScore("aczzx") == 5


def test_canonical_example_2():
    # "abcdef": no character has an earlier unmarked mirror, so the score is 0.
    assert calculateScore("abcdef") == 0


def test_single_character_has_no_earlier_index_to_match():
    # "a": index 0 has no earlier index, so the score is 0.
    assert calculateScore("a") == 0


def test_mirrored_middle_letters_are_paired():
    # "mn": 'n' is the mirror of 'm', so indices (0, 1) contribute 1 - 0 = 1.
    assert calculateScore("mn") == 1


def test_multiple_pairs_accumulate_score():
    # "abzy": pairs are (0, 2) and (1, 3), contributing 2 + 2 = 4.
    assert calculateScore("abzy") == 4


def test_marked_characters_cannot_be_reused():
    # "azza": (0, 1) contributes 1; index 2 cannot reuse marked index 0;
    # then (2, 3) contributes 1, for 1 + 1 = 2.
    assert calculateScore("azza") == 2
